13use tracing::instrument;
14macro_rules! deny_if_true {
15    ($cond:expr, $msg:expr) => {
16        if ($cond) {
17            return Err(IotaError::UserInput {
18                error: UserInputError::TransactionDenied {
19                    error: $msg.to_string(),
20                },
21            });
22        }
23    };
24}

26/// Check that the provided transaction is allowed to be signed according to the
27/// deny config.
28#[instrument(level = "trace", skip_all, fields(tx_digest = ?tx.digest()))]
29pub fn check_transaction_for_validation(
30    tx: &Transaction,
31    tx_signatures: &[UserSignature],
32    input_object_kinds: &[InputObjectKind],
33    receiving_objects: &[ObjectReference],
34    filter_config: &dyn DenyRuleConfig,
35    package_store: &dyn BackingPackageStore,
36) -> IotaResult {
37    check_disabled_features(filter_config, tx, tx_signatures)?;
38
39    check_signers(filter_config, tx)?;
40
41    check_input_objects(filter_config, input_object_kinds)?;
42
43    check_package_dependencies(filter_config, tx, package_store)?;
44
45    check_receiving_objects(filter_config, receiving_objects)?;
46
47    Ok(())
48}

50#[instrument(level = "trace", skip_all)]
51fn check_receiving_objects(
52    filter_config: &dyn DenyRuleConfig,
53    receiving_objects: &[ObjectReference],
54) -> IotaResult {
55    deny_if_true!(
56        filter_config.receiving_objects_disabled() && !receiving_objects.is_empty(),
57        "Receiving objects is temporarily disabled".to_string()
58    );
59    if !filter_config.has_denied_objects() {
60        return Ok(());
61    }
62    for receiving_object in receiving_objects {
63        deny_if_true!(
64            filter_config.is_object_denied(&receiving_object.object_id),
65            format!(
66                "Access to object {:?} is temporarily disabled",
67                receiving_object.object_id
68            )
69        );
70    }
71    Ok(())
72}

74#[instrument(level = "trace", skip_all)]
75fn check_disabled_features(
76    filter_config: &dyn DenyRuleConfig,
77    tx: &Transaction,
78    tx_signatures: &[UserSignature],
79) -> IotaResult {
80    deny_if_true!(
81        filter_config.user_transaction_disabled(),
82        "Transaction signing is temporarily disabled"
83    );
84
85    tx_signatures.iter().try_for_each(|s| {
86        if let UserSignature::MoveAuthenticator(_) = s {
87            deny_if_true!(
88                filter_config.move_authenticator_disabled(),
89                "MoveAuthenticator is temporarily disabled"
90            );
91        }
92        Ok(())
93    })?;
94
95    if !filter_config.package_publish_disabled() && !filter_config.package_upgrade_disabled() {
96        return Ok(());
97    }
98
99    for command in tx.kind().iter_commands() {
100        deny_if_true!(
101            filter_config.package_publish_disabled() && matches!(command, Command::Publish(..)),
102            "Package publish is temporarily disabled"
103        );
104        deny_if_true!(
105            filter_config.package_upgrade_disabled() && matches!(command, Command::Upgrade(..)),
106            "Package upgrade is temporarily disabled"
107        );
108    }
109    Ok(())
110}

112#[instrument(level = "trace", skip_all)]
113fn check_signers(filter_config: &dyn DenyRuleConfig, tx: &Transaction) -> IotaResult {
114    if !filter_config.has_denied_addresses() {
115        return Ok(());
116    }
117    for signer in tx.signers() {
118        deny_if_true!(
119            filter_config.is_address_denied(&signer),
120            format!(
121                "Access to account address {:?} is temporarily disabled",
122                signer
123            )
124        );
125    }
126    Ok(())
127}

129#[instrument(level = "trace", skip_all)]
130fn check_input_objects(
131    filter_config: &dyn DenyRuleConfig,
132    input_object_kinds: &[InputObjectKind],
133) -> IotaResult {
134    let shared_object_disabled = filter_config.shared_object_disabled();
135    if !filter_config.has_denied_objects() && !shared_object_disabled {
136        // No need to iterate through the input objects if no relevant policy is set.
137        return Ok(());
138    }
139    for input_object_kind in input_object_kinds {
140        let id = input_object_kind.object_id();
141        deny_if_true!(
142            filter_config.is_object_denied(&id),
143            format!("Access to input object {id} is temporarily disabled")
144        );
145        deny_if_true!(
146            shared_object_disabled && input_object_kind.is_shared_object(),
147            "Usage of shared object in transactions is temporarily disabled"
148        );
149    }
150    Ok(())
151}

153#[instrument(level = "trace", skip_all)]
154fn check_package_dependencies(
155    filter_config: &dyn DenyRuleConfig,
156    tx: &Transaction,
157    package_store: &dyn BackingPackageStore,
158) -> IotaResult {
159    if !filter_config.has_denied_packages() {
160        return Ok(());
161    }
162    let mut dependencies = vec![];
163    for command in tx.kind().iter_commands() {
164        match command {
165            Command::Publish(cmd) => {
166                // It is possible that the deps list is inaccurate since it's provided
167                // by the user. But that's OK because this publish transaction will fail
168                // to execute in the end. Similar reasoning for Upgrade.
169                dependencies.extend(cmd.dependencies.iter().copied());
170            }
171            Command::Upgrade(cmd) => {
172                dependencies.extend(cmd.dependencies.iter().copied());
173                // It's crucial that we don't allow upgrading a package in the deny list,
174                // otherwise one can bypass the deny list by upgrading a package.
175                dependencies.push(cmd.package);
176            }
177            Command::MoveCall(cmd) => {
178                dependencies.extend(package_and_dependency_ids(cmd.package, package_store)?);
179            }
180            Command::TransferObjects(..)
181            | Command::SplitCoins(..)
182            | Command::MergeCoins(..)
183            | Command::MakeMoveVector(..) => {}
184            _ => unimplemented!("a new Command enum variant was added and needs to be handled"),
185        }
186    }
187    for dep in dependencies {
188        deny_if_true!(
189            filter_config.is_package_denied(&dep),
190            format!("Access to package {dep} is temporarily disabled")
191        );
192    }
193    Ok(())
194}

196/// Check that no `MoveAuthenticator` authenticates through a denied package.
197///
198/// The package holding the authenticate function is named only by the account's
199/// `AuthenticatorFunctionRef`: the call is assembled during execution, and the
200/// reference is a runtime one, so the package appears in neither the
201/// transaction's commands nor any linkage table. It is therefore checked here
202/// rather than alongside a transaction's commands in
203/// [`check_transaction_for_validation`].
204///
205/// Call it once the references have been loaded, over every authenticator the
206/// transaction carries, and before any authenticator runs.
207#[instrument(level = "trace", skip_all)]
208pub fn check_authenticator_packages<'a>(
209    filter_config: &dyn DenyRuleConfig,
210    authenticator_function_refs: impl IntoIterator<Item = &'a AuthenticatorFunctionRef>,
211    package_store: &dyn BackingPackageStore,
212) -> IotaResult {
213    if !filter_config.has_denied_packages() {
214        return Ok(());
215    }
216    for authenticator_function_ref in authenticator_function_refs {
217        let package_id = match authenticator_function_ref {
218            AuthenticatorFunctionRef::V1(v1) => v1.package,
219        };
220        for dep in package_and_dependency_ids(package_id, package_store)? {
221            deny_if_true!(
222                filter_config.is_package_denied(&dep),
223                format!(
224                    "Access to package {dep} from a Move authenticator is temporarily disabled"
225                )
226            );
227        }
228    }
229    Ok(())
230}

232/// The ids the deny list has to see for a call into `package_id`: the package
233/// itself, and the upgraded id of each of its dependencies.
234///
235/// `linkage_table` maps the original id of a dependency to the id actually
236/// loaded, and only the latter is returned. So this establishes that a denied
237/// package is not in current use, and denying one version of a package still
238/// permits a newer one.
239fn package_and_dependency_ids(
240    package_id: ObjectId,
241    package_store: &dyn BackingPackageStore,
242) -> IotaResult<Vec<ObjectId>> {
243    let package = package_store
244        .get_package_object(&package_id)?
245        .ok_or(IotaError::UserInput {
246            error: UserInputError::ObjectNotFound {
247                object_id: package_id,
248                version: None,
249            },
250        })?;
251    let package = package.move_package();
252    Ok(package
253        .linkage_table()
254        .values()
255        .map(|upgrade_info| upgrade_info.upgraded_id)
256        .chain(std::iter::once(package.id()))
257        .collect())
258}

277    const MAX_PACKAGE_SIZE: u64 = 100 * 1024;
278
279    /// A package store holding only the packages a test puts in it; anything
280    /// else reads back as missing.
281    #[derive(Default)]
282    struct TestPackageStore(HashMap<ObjectId, PackageObject>);
283
284    impl BackingPackageStore for TestPackageStore {
285        fn get_package_object(&self, package_id: &ObjectId) -> IotaResult<Option<PackageObject>> {
286            Ok(self.0.get(package_id).cloned())
287        }
288    }
289
290    impl TestPackageStore {
291        /// Adds a module-less package whose linkage table maps each dependency
292        /// onto itself, which is what a package that has never been upgraded
293        /// past its dependencies looks like.
294        fn with_package(
295            mut self,
296            id: ObjectId,
297            dependencies: impl IntoIterator<Item = ObjectId>,
298        ) -> Self {
299            let linkage_table: BTreeMap<_, _> = dependencies
300                .into_iter()
301                .map(|dep| {
302                    (
303                        dep,
304                        UpgradeInfo {
305                            upgraded_id: dep,
306                            upgraded_version: Version::OBJECT_START,
307                        },
308                    )
309                })
310                .collect();
311            let package = MovePackage::new(
312                id,
313                Version::OBJECT_START,
314                BTreeMap::new(),
315                MAX_PACKAGE_SIZE,
316                vec![],
317                linkage_table,
318            )
319            .unwrap();
320            self.0.insert(
321                id,
322                PackageObject::new(Object::new_from_package(
323                    package,
324                    TransactionDigest::genesis_marker(),
325                )),
326            );
327            self
328        }
329    }

335    fn authenticator_ref(package: ObjectId) -> AuthenticatorFunctionRef {
336        AuthenticatorFunctionRef::V1(AuthenticatorFunctionRefV1 {

352            result.unwrap_err(),
353            IotaError::UserInput {
354                error: UserInputError::TransactionDenied { .. }

384    fn denies_when_any_one_of_several_authenticators_is_denied() {
385        // A sponsored transaction carries one authenticator per signer, and the
386        // sender's must be judged even when the sponsor's is fine.
387        let (sender_package, sponsor_package) = (id(1), id(2));
388        let store = TestPackageStore::default()
389            .with_package(sender_package, [])
390            .with_package(sponsor_package, []);

392        assert_denied(check_authenticator_packages(
393            &deny_packages([sender_package]),

395                authenticator_ref(sponsor_package),
