4use async_graphql::{Object, SimpleObject, Union};
5use iota_sdk_types::{
6    AccountClaimKind as NativeAccountClaimKind,
7    ClaimAccountTransaction as NativeClaimAccountTransaction,
8};
9
10use crate::types::base64::Base64;
11
12#[derive(Clone, Eq, PartialEq)]
13pub(crate) struct ClaimAccountTransaction {
14    pub native: NativeClaimAccountTransaction,
15}
16
17/// A transaction that claims the sender's address as an account object.
18#[Object]
19impl ClaimAccountTransaction {
20    /// The type of account created by the claim.
21    async fn kind(&self) -> AccountClaimKind {
22        match &self.native.kind {
23            NativeAccountClaimKind::SmartAccount(smart) => {
24                AccountClaimKind::SmartAccount(SmartAccountClaim {
25                    public_key_scheme: smart.public_key_scheme,
26                    public_key_raw_bytes: Base64::from(smart.public_key_raw_bytes.as_slice()),
27                })
28            }
29            _ => unimplemented!(
30                "a new AccountClaimKind enum variant was added and needs to be handled"
31            ),
32        }
33    }
34}
35
36/// The type of account created by a claim.
37#[derive(Union, Clone, Eq, PartialEq)]
38pub(crate) enum AccountClaimKind {
39    SmartAccount(SmartAccountClaim),
40}
41
42/// Parameters the claimed SmartAccount was created with.
43#[derive(SimpleObject, Clone, Eq, PartialEq)]
44pub(crate) struct SmartAccountClaim {
45    /// The signature scheme flag byte of the public key (0=Ed25519,
46    /// 1=Secp256k1, 2=Secp256r1, 3=MultiSig, 6=Passkey).
47    pub public_key_scheme: u8,
48    /// Raw public key bytes of the claimed address, without the scheme flag
49    /// prefix. For MultiSig this is a BCS-encoded multisig public key.
50    pub public_key_raw_bytes: Base64,
51}
