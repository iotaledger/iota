24#[derive(Clone, Debug, SimpleObject)]
25pub(crate) struct DryRunResult {
26    /// The error that occurred during dry run execution, if any.
27    pub error: Option<String>,
28    /// The intermediate results for each command of the dry run execution,
29    /// including contents of mutated references and return values.
30    pub results: Option<Vec<DryRunEffect>>,
31    /// The transaction block representing the dry run execution.
32    pub transaction: Option<TransactionBlock>,
33    /// If an input object is congested, suggest a gas price to use.
34    pub suggested_gas_price: Option<BigInt>,
35}
36
37#[derive(Clone, Debug, PartialEq, Eq, SimpleObject)]
38pub(crate) struct DryRunEffect {
39    /// Changes made to arguments that were mutably borrowed by each command in
40    /// this transaction.
41    pub mutated_references: Option<Vec<DryRunMutation>>,
42
43    /// Return results of each command in this transaction.
44    pub return_values: Option<Vec<DryRunReturn>>,
45}
46
47#[derive(Clone, Debug, PartialEq, Eq, SimpleObject)]
48pub(crate) struct DryRunMutation {
49    pub input: TransactionArgument,
50
51    pub type_: MoveType,
52
53    pub bcs: Base64,
54}

56#[derive(Clone, Debug, PartialEq, Eq, SimpleObject)]
57pub(crate) struct DryRunReturn {
58    pub type_: MoveType,
59
60    pub bcs: Base64,
61}
62impl TryFrom<IotaExecutionResult> for DryRunEffect {
63    type Error = crate::error::Error;
64
65    fn try_from(result: IotaExecutionResult) -> Result<Self, Self::Error> {
66        let mutated_references = result
67            .mutable_reference_outputs
68            .iter()
69            .map(|(argument, bcs, type_)| {
70                let tag: TypeTag = type_.clone().try_into()?;
71                Ok(DryRunMutation {
72                    input: (*argument).into(),
73                    type_: tag.into(),
74                    bcs: bcs.into(),
75                })
76            })
77            .collect::<Result<Vec<_>, anyhow::Error>>()
78            .map_err(|e| {
79                Error::Internal(format!(
80                    "Failed to parse results returned from the simulation: {e:?}"
81                ))
82            })?;
83        let return_values = result
84            .return_values
85            .iter()
86            .map(|(bcs, type_)| {
87                let tag: TypeTag = type_.clone().try_into()?;
88                Ok(DryRunReturn {
89                    type_: tag.into(),
90                    bcs: bcs.into(),
91                })
92            })
93            .collect::<Result<Vec<_>, anyhow::Error>>()
94            .map_err(|e| {
95                Error::Internal(format!(
96                    "Failed to parse results returned from the simulation: {e:?}"
97                ))
98            })?;
99        Ok(Self {
100            mutated_references: Some(mutated_references),
101            return_values: Some(return_values),
102        })
103    }
104}

106impl TryFrom<SimulationOutput> for DryRunResult {
107    type Error = crate::error::Error;
108
109    fn try_from(simulation: SimulationOutput) -> Result<Self, Self::Error> {
110        // Take the fields GraphQL needs from the raw simulation, ignoring the ones we
111        // are not requesting for graphql.
112        let SimulationOutput {
113            transaction,
114            effects,
115            events: TransactionEvents(events),
116            balance_changes,
117            input_objects,
118            output_objects,
119            command_results,
120            suggested_gas_price,
121            execution_error,
122            ..
123        } = simulation;
124
125        // The node returns per-command results or an execution error, never both,
126        // so `command_results` is `None` for a failed dry run.
127        let results = command_results
128            .map(|results| {
129                results
130                    .into_iter()
131                    .map(DryRunEffect::try_from)
132                    .collect::<Result<Vec<_>, Error>>()
133            })
134            .transpose()?;
135
136        let create_objects_map = |objects: Vec<iota_types::object::Object>| {
137            Arc::new(
138                objects
139                    .into_iter()
140                    .map(|object| (object.id(), object))
141                    .collect::<BTreeMap<_, _>>(),
142            )
143        };
144
145        let tx_data = transaction.expect("transaction is always requested from the simulation");
146        let transaction = Some(TransactionBlock {
147            inner: TransactionBlockInner::Simulated {
148                tx_data,
149                effects,
150                events,
151                balance_changes: balance_changes
152                    .into_iter()
153                    .map(IndexedBalanceChange::from)
154                    .collect(),
155                input_objects: create_objects_map(input_objects),
156                output_objects: create_objects_map(output_objects),
157            },
158            // A simulated transaction uses the fullnode's state, which is typically ahead of
159            // the indexed state, so it is not tied to a checkpoint.
160            checkpoint_viewed_at: UNAVAILABLE_CHECKPOINT_SEQUENCE_NUMBER,
161        });
162        Ok(Self {
163            error: execution_error,
164            results,
165            transaction,
166            suggested_gas_price: suggested_gas_price.map(BigInt::from),
167        })
168    }
169}
