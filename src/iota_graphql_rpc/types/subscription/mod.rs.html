32/// Represents the channel size of the [`InMemory`] broker.
33const BROKER_CHANNEL_SIZE: NonZeroUsize =
34    NonZeroUsize::new(10_000).expect("value should be greater than 0");

36/// Notifies that the subscription consumer has fallen behind the live
37/// subscription stream and missed one or more payloads.
38#[derive(SimpleObject, Clone)]
39pub(crate) struct Lagged {
40    /// Number of missed payloads since the previous emitted one.
41    count: i32,
42}
43
44/// Possible responses from a subscription.
45///
46/// It could be one of the following:
47/// - A successful payload from the subscription stream.
48/// - A notice that the subscription has been lagged behind the network with the number of lost
49///   payloads.
50#[derive(Union, Clone)]
51#[graphql(concrete(name = "EventSubscriptionPayload", params(Event)))]
52#[graphql(concrete(name = "TransactionBlockSubscriptionPayload", params(TransactionBlock)))]
53pub(crate) enum SubscriptionItem<T: OutputType> {
54    /// Successfully received payload from the subscription stream.
55    Payload(T),
56    /// A notice that the subscription has been lagged behind the network.
57    Lagged(Lagged),
58}

60/// Subscribe to events and transactions from the IOTA network.
61pub struct Subscription;
62
63#[Subscription]
64impl Subscription {
65    /// Subscribe to incoming transactions from the IOTA network.
66    ///
67    /// If no filter is provided, all transactions will be returned.
68    ///
69    /// # Stream recovery
70    ///
71    /// When `start_after` is provided with the digest of the last transaction
72    /// the client received, the stream resumes from the transaction
73    /// immediately after it. The identified transaction itself is not
74    /// emitted.
75    async fn transactions(
76        &self,
77        ctx: &Context<'_>,
78        start_after: Option<Digest>,
79        filter: Option<SubscriptionTransactionFilter>,
80    ) -> async_graphql::Result<impl Stream<Item = Result<SubscriptionItem<TransactionBlock>, Error>>>
81    {
82        let stream = ctx.data_unchecked::<GraphQLStream>();
83        Ok(stream.subscribe_transactions(start_after, filter))
84    }
85
86    /// Subscribe to incoming events from the IOTA network.
87    ///
88    /// If no filter is provided, all events will be returned.
89    ///
90    /// # Stream recovery
91    ///
92    /// Events are streamed in transaction order, all events from a single
93    /// transaction arrive contiguously before any events from the next one.
94    ///
95    /// When `start_after` is provided with a transaction digest, the stream
96    /// resumes from the transaction immediately after it.
97    ///
98    /// Because a single transaction can emit multiple events, a disconnection
99    /// may leave the client with only a partial set of events from the
100    /// transaction it was last receiving. A transaction is considered fully
101    /// received only once the client has seen an event from the following
102    /// transaction (a digest change signals the previous transaction is
103    /// complete).
104    ///
105    /// On reconnect, clients should pass the digest of the last fully
106    /// received transaction as `start_after`. The stream resumes from the
107    /// next transaction, so no partial transaction is ever observed.
108    async fn events(
109        &self,
110        ctx: &Context<'_>,
111        start_after: Option<Digest>,
112        filter: Option<SubscriptionEventFilter>,
113    ) -> async_graphql::Result<impl Stream<Item = Result<SubscriptionItem<Event>, Error>>> {
114        let stream = ctx.data_unchecked::<GraphQLStream>();
115        Ok(stream.subscribe_events(start_after, filter))
116    }
117}

119/// Provides real-time data streams for the GraphQL subscription feature.
120///
121/// It wraps the low-level [`InMemory`] streamer and handles the necessary
122/// data processing, filtering, and subscription-specific error handling before
123/// yielding items to GraphQL.
124///
125/// It ensures that when a critical data error occurs during item conversion,
126/// the resulting stream is gracefully terminated by the server.
127pub(crate) struct GraphQLStream {
128    streamer: InMemory,
129}
130
131impl GraphQLStream {
132    pub(crate) async fn new(
133        indexer_reader: IndexerReader,
134        registry: &Registry,
135    ) -> Result<Self, Error> {
136        let config = Config {
137            channel_buffer_size: BROKER_CHANNEL_SIZE,
138            ..Default::default()
139        };
140        let streamer = InMemory::new(config, indexer_reader, InMemoryStreamMetrics::new(registry))
141            .await
142            .map_err(|e| Error::Internal(format!("failed to connect to postgres: {e}")))?;
143        Ok(Self { streamer })
144    }
145
146    /// Checks if the provided filter matches the item.
147    ///
148    /// If no filter is provided, the item is **always** considered a match, and
149    /// the function returns `true`.
150    fn matches_filter<T, F>(filter: Option<&F>, item: &T) -> bool
151    where
152        F: Filter<T>,
153    {
154        filter.as_ref().map(|f| f.matches(item)).unwrap_or(true)
155    }
156
157    /// Subscribe to transactions from IOTA Network.
158    pub(crate) fn subscribe_transactions(
159        &self,
160        start_after: Option<Digest>,
161        filter: Option<SubscriptionTransactionFilter>,
162    ) -> impl Stream<Item = Result<SubscriptionItem<TransactionBlock>, Error>> {
163        self.streamer
164            .subscribe_transactions(
165                start_after.map(|digest| RecoveryPoint::Exclusive(TransactionDigest::from(digest))),
166            )
167            .try_filter(move |stored| future::ready(Self::matches_filter(filter.as_ref(), stored)))
168            .then(|stored| {
169                let subscription_item = match stored {
170                    Ok(stored) => {
171                        let checkpoint_viewed_at = stored.checkpoint_sequence_number as u64;
172                        TransactionBlockInner::try_from(stored).map(|inner| {
173                            SubscriptionItem::Payload(TransactionBlock {
174                                inner,
175                                checkpoint_viewed_at,
176                            })
177                        })
178                    }
179                    Err(IndexerStreamingError::Lagged(BroadcastStreamRecvError::Lagged(count))) => {
180                        warn!("subscriber lagging by {count} messages");
181                        try_into_int(count).map(|count| SubscriptionItem::Lagged(Lagged { count }))
182                    }
183                    Err(err) => Err(err.into()),
184                };
185                future::ready(subscription_item)
186            })
187            // intercept the error, send it to the client and terminate the stream.
188            .scan(false, |should_terminate_stream, subscription_item| {
189                if *should_terminate_stream {
190                    return future::ready(None);
191                }
192                future::ready(Some(
193                    subscription_item.inspect_err(|_| *should_terminate_stream = true),
194                ))
195            })
196    }
197
198    /// Subscribe to events from IOTA Network.
199    pub(crate) fn subscribe_events(
200        &self,
201        start_after: Option<Digest>,
202        filter: Option<SubscriptionEventFilter>,
203    ) -> impl Stream<Item = Result<SubscriptionItem<Event>, Error>> {
204        self.streamer
205            .subscribe_events(
206                start_after.map(|digest| RecoveryPoint::Exclusive(TransactionDigest::from(digest))),
207            )
208            .try_filter(move |stored| future::ready(Self::matches_filter(filter.as_ref(), stored)))
209            .then(|stored| {
210                let subscription_item = match stored {
211                    Ok(stored) => {
212                        Event::try_from_stored_event(stored, 0).map(SubscriptionItem::Payload)
213                    }
214                    Err(IndexerStreamingError::Lagged(BroadcastStreamRecvError::Lagged(count))) => {
215                        warn!("subscriber lagging by {count} messages");
216                        try_into_int(count).map(|count| SubscriptionItem::Lagged(Lagged { count }))
217                    }
218                    Err(err) => Err(err.into()),
219                };
220                future::ready(subscription_item)
221            })
222            // intercept the error, send it to the client and terminate the stream.
223            .scan(false, |should_terminate_stream, subscription_item| {
224                if *should_terminate_stream {
225                    return future::ready(None);
226                }
227                future::ready(Some(
228                    subscription_item.inspect_err(|_| *should_terminate_stream = true),
229                ))
230            })
231    }
