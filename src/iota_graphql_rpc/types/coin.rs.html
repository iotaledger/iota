209    /// The transaction blocks that sent objects to this object.
210    ///
211    /// `scanLimit` restricts the number of candidate transactions scanned when
212    /// gathering a page of results. It is required for queries that apply two
213    /// or more complex filters (on function, affected address, recipient, input
214    /// object, changed object, or wrapped or deleted object), and can be at
215    /// most `serviceConfig.maxScanLimit`. A `kind` filter cannot be
216    /// combined with any of them.
217    ///
218    /// When the scan limit is reached the page will be returned even if it has
219    /// fewer than `first` results when paginating forward (`last` when
220    /// paginating backwards). If there are more transactions to scan,
221    /// `pageInfo.hasNextPage` (or `pageInfo.hasPreviousPage`) will be set to
222    /// `true`, and `PageInfo.endCursor` (or `PageInfo.startCursor`) will be set
223    /// to the last transaction that was scanned as opposed to the last (or
224    /// first) transaction in the page.
225    ///
226    /// Requesting the next (or previous) page after this cursor will resume the
227    /// search, scanning the next `scanLimit` many transactions in the
228    /// direction of pagination, and so on until all transactions in the
229    /// scanning range have been visited.
230    ///
231    /// By default, the scanning range includes all transactions known to
232    /// GraphQL, but it can be restricted by the `after` and `before`
233    /// cursors, and the `beforeCheckpoint`, `afterCheckpoint` and
234    /// `atCheckpoint` filters.
235    ///
236    /// DEPRECATION NOTICE: Support for the combination of two or more complex
237    /// filters as discussed above will stop with the v1.38 release. `scanLimit`
238    /// will thus become obsolete and will be removed as well.
239    #[graphql(
240        complexity = "first.or(last).unwrap_or(DEFAULT_PAGE_SIZE as u64) as usize * child_complexity"
241    )]

243        &self,
244        ctx: &Context<'_>,
245        first: Option<u64>,
246        after: Option<transaction_block::Cursor>,
247        last: Option<u64>,
248        before: Option<transaction_block::Cursor>,
249        filter: Option<TransactionBlockFilter>,
250        #[graphql(
251            deprecation = "`scanLimit` will be removed with v1.38, along with the support for combining complex filters."
252        )]
253        scan_limit: Option<u64>,
254    ) -> Result<ScanConnection<String, TransactionBlock>> {
255        ObjectImpl(&self.super_.super_)
256            .received_transaction_blocks(ctx, first, after, last, before, filter, scan_limit)

343impl Coin {
344    /// Query the database for a `page` of coins. The page uses the bytes of an
345    /// Object ID as the cursor, and can optionally be filtered by an owner.
346    pub(crate) async fn paginate(
347        db: &Db,
348        page: Page<object::Cursor>,
349        coin_type: TypeTag,
350        owner: Option<IotaAddress>,
351        checkpoint_viewed_at: u64,
352    ) -> Result<Connection<String, Coin>, Error> {
353        // If cursors are provided, defer to the `checkpoint_viewed_at` in the cursor if
354        // they are consistent. Otherwise, use the value from the parameter, or
355        // set to None. This is so that paginated queries are consistent with
356        // the previous query that created the cursor.
357        let cursor_viewed_at = page.validate_cursor_consistency()?;
358        let checkpoint_viewed_at = cursor_viewed_at.unwrap_or(checkpoint_viewed_at);
359
360        let max_available_range = db.max_available_range;
361
362        let Some((prev, next, results)) = db
363            .execute_repeatable(move |conn| {
364                if !AvailableRange::is_checkpoint_in_backward_history_range(
365                    conn,
366                    checkpoint_viewed_at,
367                    max_available_range,
368                )? {
369                    return Ok::<_, diesel::result::Error>(None);
370                };
371
372                Ok(Some(page.paginate_raw_query::<StoredBackwardObject>(
373                    conn,
374                    checkpoint_viewed_at,
375                    coins_query(coin_type, owner, checkpoint_viewed_at, &page),
376                )?))
377            })
378            .await?
379        else {
380            return Err(Error::Client(
381                "Requested data is outside the available range".to_string(),
382            ));
383        };
384
385        let mut conn: Connection<String, Coin> = Connection::new(prev, next);
386
387        for stored in results {
388            // To maintain consistency, the returned cursor should have the same upper-bound
389            // as the checkpoint found on the cursor.
390            let cursor = stored.cursor(checkpoint_viewed_at).encode_cursor();
391            let stored_history = stored.into_stored_history(checkpoint_viewed_at);
392            let active_object = ActiveObject::try_from(stored_history)?;
393            let object = Object::from_active_object(active_object, checkpoint_viewed_at, None);
394
395            let move_ = MoveObject::try_from(&object).map_err(|_| {
396                Error::Internal(format!(
397                    "Failed to deserialize as Move object: {}",
398                    object.address
399                ))
400            })?;
401
402            let coin = Coin::try_from(&move_).map_err(|_| {
403                Error::Internal(format!("Failed to deserialize as Coin: {}", object.address))
404            })?;
405
406            conn.edges.push(Edge::new(cursor, coin));
407        }
408
409        Ok(conn)
410    }
411}

413impl TryFrom<&MoveObject> for Coin {
414    type Error = CoinDowncastError;
415
416    fn try_from(move_object: &MoveObject) -> Result<Self, Self::Error> {
417        if !move_object.native.struct_tag().is_coin() {
418            return Err(CoinDowncastError::NotACoin);
419        }
420
421        Ok(Self {
422            super_: move_object.clone(),
423            native: bcs::from_bytes(move_object.native.contents())
424                .map_err(CoinDowncastError::Bcs)?,
425        })
426    }
427}

429/// Constructs a raw query to fetch objects from the database. Since there are
430/// no point lookups for the coin query, objects are filtered out if they
431/// satisfy the criteria but have a later version in the same checkpoint.
432fn coins_query(
433    coin_type: TypeTag,
434    owner: Option<IotaAddress>,
435    checkpoint_viewed_at: u64,
436    page: &Page<object::Cursor>,
437) -> RawQuery {
438    consistent::query(checkpoint_viewed_at, page, move |query| {
439        apply_filter(query, &coin_type, owner)
440    })
441}
442
443fn apply_filter(mut query: RawQuery, coin_type: &TypeTag, owner: Option<IotaAddress>) -> RawQuery {
444    if let Some(owner) = owner {
445        query = filter!(
446            query,
447            format!(
448                "owner_id = '\\x{}'::bytea AND owner_type = {}",
449                hex::encode(owner.into_vec()),
450                OwnerType::Address as i16
451            )
452        );
453    }
454
455    query = filter!(
456        query,
457        "coin_type IS NOT NULL AND coin_type = {}",
458        coin_type.to_canonical_string(/* with_prefix */ true)
459    );
460
461    query
