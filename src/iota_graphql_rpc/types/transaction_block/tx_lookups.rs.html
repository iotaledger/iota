5//! # Transaction Filter Lookup Tables
6//!
7//! ## Schemas
8//!
9//! Tables backing Transaction filters in GraphQL all follow the same rough
10//! shape:
11//!
12//! 1. They each get their own table, mapping the filter value to the transaction sequence number.
13//!
14//! 2. They also include a `sender` column, and a secondary index over the sender, filter values and
15//!    the transaction sequence number.
16//!
17//! 3. They also include a secondary index over the transaction sequence number.
18//!
19//! This pattern allows us to offer a simple rule for users:
20//! - If you are filtering on a **single value**, you can do so without worrying.
21//! - If you want to additionally filter by the sender, that is also possible, but combining any
22//!   other set of filters is not supported.
23//!
24//! ## Query construction
25//!
26//! Queries that filter transactions work in two phases: Identify the
27//! transaction sequence numbers to fetch, and then fetch their contents.
28//! Filtering all happens in the first phase:
29//!
30//! - Firstly filters are broken down into individual queries targeting the appropriate lookup
31//!   table. Each constituent query is expected to return a sorted run of transaction sequence
32//!   numbers.
33//!
34//! - If a `sender` filter is included, then it is incorporated into each constituent query,
35//!   leveraging their secondary indices (2), otherwise each constituent query filters only based on
36//!   its filter value using the primary index (1). `tx_kinds` is the exception: it has no `sender`
37//!   column, so a `kind` filter with a sender is intersected with `tx_senders` instead.
38//!
39//! - The fact that both the primary and secondary indices contain the transaction sequence number
40//!   help to ensure that the output from an index scan is already sorted, which avoids a
41//!   potentially expensive materialize and sort operation.
42//!
43//! - If there are multiple constituent queries, they are intersected using inner joins. Postgres
44//!   can occasionally pick a poor query plan for this merge, so filters resulting in such merges
45//!   are not supported.

69/// Bounds on transaction sequence number, imposed by filters and cursors. The
70/// outermost bounds are determined by the checkpoint filters.
71/// These get translated into bounds in terms of transaction sequence numbers:
72///
73/// ```ignore
74///     tx_lo                                                             tx_hi
75///     [-----------------------------------------------------------------)
76/// ```
77///
78/// If cursors are provided, they further restrict the range of transactions to
79/// scan. Cursors are exclusive, but when issuing database queries, we treat
80/// them inclusively so that we can detect previous and next pages based on the
81/// existence of cursors in the results:
82///
83/// ```ignore
84///             cursor_lo                                  cursor_hi_inclusive
85///             [------------------------------------------]
86/// ```
87///
88/// This data structure can be used to compute the interval of transactions to
89/// look in for candidates to include in a page of results.
90///
91/// NOTE: for consistency, assume that lowerbounds are inclusive and upperbounds
92/// are exclusive. Bounds that do not follow this convention will be annotated
93/// explicitly (e.g. `lo_exclusive` or `hi_inclusive`).
94#[derive(Clone, Debug, Copy)]
95pub(crate) struct TxBounds {
96    /// The inclusive lower bound tx_sequence_number derived from checkpoint
97    /// bounds. If checkpoint bounds are not provided, this will default to
98    /// `0`.
99    tx_lo: u64,
100
101    /// The exclusive upper bound tx_sequence_number derived from checkpoint
102    /// bounds. If checkpoint bounds are not provided, this will default to
103    /// the total transaction count at the checkpoint viewed.
104    tx_hi: u64,
105
106    /// The starting cursor (aka `after`).
107    cursor_lo_exclusive: Option<u64>,
108
109    // The ending cursor (aka `before`).
110    cursor_hi: Option<u64>,
111}
112
113impl TxBounds {
114    /// Determines the `tx_sequence_number` range from the checkpoint bounds for
115    /// a transaction block query. The corresponding `tx_sequence_number` range
116    /// is fetched from db, and further adjusted by cursors. If there are any
117    /// inconsistencies or invalid combinations, i.e. `after` cursor is greater
118    /// than the upper bound, return None.
119    ///
120    /// `cp_bounds` must lie in the range the database has data for. A
121    /// checkpoint missing from `pruner_cp_watermark` fails with `NotFound`.
122    pub(crate) fn query(
123        conn: &mut Conn,
124        cp_bounds: RangeInclusive<u64>,
125        page: &Page<Cursor>,
126    ) -> Result<Option<Self>, diesel::result::Error> {
127        let (cp_lo, cp_hi) = (*cp_bounds.start(), *cp_bounds.end());
128
129        use pruner_cp_watermark::dsl;
130        let tx_ranges: HashMap<i64, (i64, i64)> = conn
131            .results(move || {
132                dsl::pruner_cp_watermark
133                    .select((
134                        dsl::checkpoint_sequence_number,
135                        dsl::min_tx_sequence_number,
136                        dsl::max_tx_sequence_number,
137                    ))
138                    .filter(dsl::checkpoint_sequence_number.eq_any([cp_lo as i64, cp_hi as i64]))
139            })?
140            .into_iter()
141            .map(|(cp, min_tx, max_tx): (i64, i64, i64)| (cp, (min_tx, max_tx)))
142            .collect();
143        // Both checkpoints should be in the range the database has data for.
144        let tx_range = |cp: u64| {
145            tx_ranges
146                .get(&(cp as i64))
147                .copied()
148                .ok_or(diesel::result::Error::NotFound)
149        };
150
151        let (tx_lo, _) = tx_range(cp_lo)?;
152        let (_, tx_hi_inclusive) = tx_range(cp_hi)?;
153        let (tx_lo, tx_hi) = (tx_lo as u64, tx_hi_inclusive as u64 + 1);
154
155        // If the cursors point outside checkpoint bounds, we can return early.
156        if matches!(page.after(), Some(a) if tx_hi <= u64::from(a.tx_sequence_number).saturating_add(1))
157        {
158            return Ok(None);
159        }
160
161        if matches!(page.before(), Some(b) if u64::from(b.tx_sequence_number) <= tx_lo) {
162            return Ok(None);
163        }
164
165        Ok(Some(Self {
166            tx_lo,
167            tx_hi,
168            cursor_lo_exclusive: page.after().map(|a| a.tx_sequence_number.into()),
169            cursor_hi: page.before().map(|b| b.tx_sequence_number.into()),
170        }))
171    }
172
173    /// Inclusive lowerbound for range of transactions to scan, accounting for
174    /// the bounds from filters and the cursor. For the purposes of scanning
175    /// records in the DB, cursors are treated inclusively, even though they are
176    /// exclusive bounds.
177    pub(crate) fn db_lo(&self) -> u64 {
178        max_option([self.cursor_lo_exclusive, Some(self.tx_lo)]).unwrap()
179    }
180
181    /// Exclusive upperbound for range of transactions to scan, accounting for
182    /// the bounds from filters and the cursor. For the purposes of scanning
183    /// records in the DB, cursors are treated inclusively, even though they are
184    /// exclusive bounds.
185    pub(crate) fn db_hi(&self) -> u64 {
186        min_option([
187            self.cursor_hi.map(|h| h.saturating_add(1)),
188            Some(self.tx_hi),
189        ])
190        .unwrap()
191    }
192}

194/// The inclusive range of checkpoints a transaction block query with `filter`
195/// is confined to when viewed at `checkpoint_viewed_at`, or `None` when it is
196/// empty.
197///
198/// The function considers ranges defined by:
199/// - `checkpoint_viewed_at`
200/// - `after_checkpoint`, `before_checkpoint`, `at_checkpoint` properties of the `filter`
201pub(crate) fn checkpoint_bounds_from_filter(
202    filter: &TransactionBlockFilter,
203    checkpoint_viewed_at: u64,
204) -> Option<RangeInclusive<u64>> {
205    fn intersect(a: Range<u64>, b: Range<u64>) -> Range<u64> {
206        a.start.max(b.start)..a.end.min(b.end)
207    }
208
209    // Half-open ranges, so that `before_checkpoint` needs no adjustment.
210    let viewed = Some(0..checkpoint_viewed_at + 1);
211    let after = filter
212        .after_checkpoint
213        .map(|cp| u64::from(cp) + 1..u64::MAX);
214    let before = filter.before_checkpoint.map(|cp| 0..u64::from(cp));
215    let at = filter
216        .at_checkpoint
217        .map(|cp| u64::from(cp)..u64::from(cp) + 1);
218
219    let range = [viewed, after, before, at]
220        .into_iter()
221        .flatten()
222        .fold(0..u64::MAX, intersect);
223    (!range.is_empty()).then(|| range.start..=range.end - 1)
224}

226/// Determines the maximum value in an arbitrary number of Option<impl Ord>.
227fn max_option<T: Ord>(xs: impl IntoIterator<Item = Option<T>>) -> Option<T> {
228    xs.into_iter().flatten().max()
229}
230
231/// Determines the minimum value in an arbitrary number of Option<impl Ord>.
232fn min_option<T: Ord>(xs: impl IntoIterator<Item = Option<T>>) -> Option<T> {
233    xs.into_iter().flatten().min()
234}

236/// Constructs a `RawQuery` as a join over all relevant side tables, filtered on
237/// their own filter condition, plus optionally a sender, plus optionally tx/cp
238/// bounds. An `affected_address` filter without a `sender` filter is a special
239/// case, handled by `select_affected`.
240pub(crate) fn subqueries(
241    filter: &TransactionBlockFilter,
242    tx_bounds: TxBounds,
243    page: &Page<Cursor>,
244) -> Option<RawQuery> {
245    let sender = filter.sent_address;
246
247    let mut subqueries = vec![];
248
249    if let Some(f) = &filter.function {
250        subqueries.push(match f {
251            FqNameFilter::ByModule(filter) => match filter {
252                ModuleFilter::ByPackage(p) => ("tx_calls_pkg", select_pkg(p, sender, tx_bounds)),
253                ModuleFilter::ByModule(p, m) => {
254                    ("tx_calls_mod", select_mod(p, m.clone(), sender, tx_bounds))
255                }
256            },
257            FqNameFilter::ByFqName(p, m, n) => (
258                "tx_calls_fun",
259                select_fun(p, m.clone(), n.clone(), sender, tx_bounds),
260            ),
261        });
262    }
263    if let Some(kind) = &filter.kind {
264        subqueries.push(("tx_kinds", select_kind(*kind, tx_bounds)));
265    }
266    if let Some(recv) = &filter.recv_address {
267        subqueries.push(("tx_recipients", select_recipient(recv, sender, tx_bounds)));
268    }
269    if let Some(input) = &filter.input_object {
270        subqueries.push(("tx_input_objects", select_input(input, sender, tx_bounds)));
271    }
272    if let Some(changed) = &filter.changed_object {
273        subqueries.push((
274            "tx_changed_objects",
275            select_changed(changed, sender, tx_bounds),
276        ));
277    }
278    if let Some(wrapped_or_deleted_object) = &filter.wrapped_or_deleted_object {
279        subqueries.push((
280            "tx_wrapped_or_deleted_objects",
281            select_wrapped_or_deleted(wrapped_or_deleted_object, sender, tx_bounds),
282        ));
283    }
284    if let Some(sender) = &filter.explicit_sender() {
285        subqueries.push(("tx_senders", select_sender(sender, tx_bounds)));
286    }
287    if let Some(txs) = &filter.transaction_ids {
288        subqueries.push(("tx_global_order", select_ids(txs, tx_bounds)));
289    }
290
291    if let Some(affected) = &filter.affected_address {
292        match sender {
293            // A `sender` filter equal to `affected` can only match transactions
294            // where `affected` is the sender.
295            Some(sender) if sender == *affected => {
296                subqueries.push(("tx_affected", select_sender(affected, tx_bounds)));
297            }
298            // A `sender` filter different from `affected` can only match
299            // transactions where `affected` is the recipient.
300            Some(_) => {
301                subqueries.push(("tx_affected", select_recipient(affected, sender, tx_bounds)));
302            }
303            // Otherwise `affected` can be either the sender or a recipient,
304            // so we query both.
305            None => return Some(select_affected(affected, subqueries, tx_bounds, page)),
306        }
307    }
308
309    join_subqueries(subqueries)
310}

312/// The lookup tables [`subqueries`] reads for `filter`.
313pub(crate) fn subquery_tables(filter: &TransactionBlockFilter) -> Vec<CommitterTables> {
314    use CommitterTables::*;
315
316    let sender = filter.sent_address;
317
318    let mut tables = vec![];
319
320    if let Some(f) = &filter.function {
321        tables.push(match f {
322            FqNameFilter::ByModule(ModuleFilter::ByPackage(_)) => TxCallsPkg,
323            FqNameFilter::ByModule(ModuleFilter::ByModule(..)) => TxCallsMod,
324            FqNameFilter::ByFqName(..) => TxCallsFun,
325        });
326    }
327    if filter.kind.is_some() {
328        tables.push(TxKinds);
329    }
330    if filter.recv_address.is_some() {
331        tables.push(TxRecipients);
332    }
333    if filter.input_object.is_some() {
334        tables.push(TxInputObjects);
335    }
336    if filter.changed_object.is_some() {
337        tables.push(TxChangedObjects);
338    }
339    if filter.wrapped_or_deleted_object.is_some() {
340        tables.push(TxWrappedOrDeletedObjects);
341    }
342    if filter.explicit_sender().is_some() {
343        tables.push(TxSenders);
344    }
345    if filter.transaction_ids.is_some() {
346        tables.push(TxGlobalOrder);
347    }
348
349    if let Some(affected) = &filter.affected_address {
350        match sender {
351            Some(sender) if sender == *affected => tables.push(TxSenders),
352            Some(_) => tables.push(TxRecipients),
353            None => tables.extend([TxSenders, TxRecipients]),
354        }
355    }
356
357    tables
358}

360/// Joins `subqueries` on `tx_sequence_number` into a single query, or `None` if
361/// there are no subqueries.
362fn join_subqueries(mut subqueries: Vec<(&str, RawQuery)>) -> Option<RawQuery> {
363    let (_, mut subquery) = subqueries.pop()?;
364
365    if !subqueries.is_empty() {
366        subquery = query!("SELECT tx_sequence_number FROM ({}) AS initial", subquery);
367        while let Some((alias, subselect)) = subqueries.pop() {
368            subquery = inner_join!(subquery, alias => subselect, using: ["tx_sequence_number"]);
369        }
370    }
371
372    Some(subquery)
373}

375fn select_tx(sender: Option<IotaAddress>, bound: TxBounds, from: &str) -> RawQuery {
376    let mut query = filter!(
377        query!(format!("SELECT tx_sequence_number FROM {from}")),
378        format!(
379            "{} <= tx_sequence_number AND tx_sequence_number < {}",
380            bound.db_lo(),
381            bound.db_hi()
382        )
383    );
384
385    if let Some(sender) = sender {
386        query = filter!(
387            query,
388            format!("sender = {}", bytea_literal(sender.as_slice()))
389        );
390    }
391
392    query
393}

436/// Returns a RawQuery that selects transactions of a specific kind from
437/// `tx_kinds`. The sender is not applied here, because `tx_kinds` has no
438/// `sender` column: a `kind` filter with a sender needs to be intersected with
439/// `tx_senders` by the caller.
440fn select_kind(kind: TransactionBlockKindInput, bound: TxBounds) -> RawQuery {
441    filter!(
442        select_tx(None, bound, "tx_kinds"),
443        if TransactionBlockKindInput::SystemTx == kind {
444            // `tx_kinds` stores `IotaTransactionKind` numbers, and every kind
445            // outside its `NON_SYSTEM_KINDS` is a system kind.
446            format!(
447                "tx_kind NOT IN ({})",
448                IotaTransactionKind::NON_SYSTEM_KINDS
449                    .iter()
450                    .map(|kind| (*kind as i16).to_string())
451                    .collect::<Vec<_>>()
452                    .join(", ")
453            )
454        } else {
455            format!("tx_kind = {}", kind as i16)
456        }
457    )
458}

471/// Selects transactions where `affected` is the sender or a recipient, as a
472/// `UNION` of the `tx_senders` and `tx_recipients` tables.
473///
474/// To avoid full table scans on `UNION` we apply order and limit on the
475/// subqueries before `UNION`. For the result to be correct we need to apply all
476/// other filters on the subquery level, by joining each subquery with all
477/// `other_subqueries`.
478fn select_affected(
479    affected: &IotaAddress,
480    other_subqueries: Vec<(&str, RawQuery)>,
481    bound: TxBounds,
482    page: &Page<Cursor>,
483) -> RawQuery {
484    let bounded_subquery = |from: RawQuery, other_subqueries: Vec<(&str, RawQuery)>| {
485        let mut all = vec![("tx_affected", from)];
486        all.extend(other_subqueries);
487        join_subqueries(all)
488            .expect("subqueries list is not empty")
489            .order_by(format!(
490                "tx_sequence_number {}",
491                if page.is_from_front() { "ASC" } else { "DESC" }
492            ))
493            // The same limit that `Page::apply` puts on the overall query: the
494            // page size plus the two rows pointed at by the page's cursors.
495            .limit(page.limit() as i64 + 2)
496    };
497
498    query!(
499        "SELECT tx_sequence_number FROM (({}) UNION ({})) AS affected",
500        bounded_subquery(select_sender(affected, bound), other_subqueries.clone()),
501        bounded_subquery(select_recipient(affected, None, bound), other_subqueries)
502    )
503}

533fn select_ids(ids: &Vec<Digest>, bound: TxBounds) -> RawQuery {
534    let query = select_tx(None, bound, "tx_global_order");
535    if ids.is_empty() {
536        filter!(query, "1=0")
537    } else {
538        let mut inner = String::new();
539        let mut prefix = "tx_digest IN (";
540        for id in ids {
541            write!(&mut inner, "{prefix}{}", bytea_literal(id.as_slice())).unwrap();
542            prefix = ", ";
543        }
544        inner.push(')');
545        filter!(query, inner)
546    }
