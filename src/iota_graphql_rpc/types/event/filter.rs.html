13/// Represents optional available filters for events.
14#[derive(InputObject, Clone, Default)]
15pub(crate) struct EventFilter {
16    /// Filter down to events from transactions sent by this address.
17    pub sender: Option<IotaAddress>,
18    /// Filter down to the events from this transaction (given by its
19    /// transaction digest).
20    pub transaction_digest: Option<Digest>,
21    // Enhancement (post-MVP)
22    // after_checkpoint
23    // at_checkpoint
24    // before_checkpoint
25    /// Events emitted by a particular module. An event is emitted by a
26    /// particular module if some function in the module is called by a
27    /// PTB and emits an event.
28    ///
29    /// Modules can be filtered by their package, or package::module.
30    /// We currently do not support filtering by emitting module and event type
31    /// at the same time so if both are provided in one filter, the query will
32    /// error.
33    pub emitting_module: Option<ModuleFilter>,
34
35    /// This field is used to specify the type of event emitted.
36    ///
37    /// Events can be filtered by their type's package, package::module,
38    /// or their fully qualified type name.
39    ///
40    /// Generic types can be queried by either the generic type name, e.g.
41    /// `0x2::coin::Coin`, or by the full type name, such as
42    /// `0x2::coin::Coin<0x2::iota::IOTA>`.
43    pub event_type: Option<TypeFilter>,
44    // Enhancement (post-MVP)
45    // pub start_time
46    // pub end_time
47
48    // Enhancement (post-MVP)
49    // pub any
50    // pub all
51    // pub not
52}
53
54impl EventFilter {
55    /// Returns the transaction digest when `transaction_digest` is the only
56    /// filter set.
57    pub(crate) fn only_transaction_digest(&self) -> Option<Digest> {
58        if self.sender.is_some() || self.emitting_module.is_some() || self.event_type.is_some() {
59            return None;
60        }
61        self.transaction_digest
62    }
63}
