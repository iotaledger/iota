4use async_graphql::*;
5use iota_json_rpc_types::{IotaMoveValue, IotaMoveViewCallResults};
6use serde::Serialize;
7
8use crate::{error::Error, types::json::Json};

10/// The result of a move-view function call.
11///
12/// Execution errors are captured in the `error` field, in which
13/// case the `results` field will be `None`.
14///
15/// On success, the `results` field will contain the return values of the
16/// move view function, and the `error` field will be `None`.
17#[derive(Clone, Debug, Serialize, Default, SimpleObject)]
18pub(crate) struct MoveViewResult {
19    /// Execution error from executing the move view call.
20    error: Option<String>,
21    /// The return values of the move view function.
22    results: Option<Vec<Json>>,
23}
24
25impl TryFrom<IotaMoveViewCallResults> for MoveViewResult {
26    type Error = Error;
27
28    fn try_from(value: IotaMoveViewCallResults) -> Result<Self, Error> {
29        let mut result = Self::default();
30        match value {
31            IotaMoveViewCallResults::Error(e) => result.error = Some(e),
32            IotaMoveViewCallResults::Results(results) => {
33                result.results = Some(
34                    results
35                        .into_iter()
36                        .map(TryInto::try_into)
37                        .collect::<Result<_, _>>()?,
38                );
39            }
40        }
41        Ok(result)
42    }
43}
44
45impl TryFrom<IotaMoveValue> for Json {
46    type Error = Error;
47
48    fn try_from(value: IotaMoveValue) -> Result<Self, Error> {
49        let json = serde_json::to_value(value).map_err(|e| Error::Internal(e.to_string()))?;
50
51        Value::try_from(json)
52            .map_err(|e| Error::Internal(e.to_string()))
53            .map(Self)
54    }
55}
