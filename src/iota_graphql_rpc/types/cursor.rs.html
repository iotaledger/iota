33/// Connection field parameters parsed into a single type that encodes the
34/// bounds of a single page in a paginated response.
35#[derive(Debug, Clone)]
36pub(crate) struct Page<C> {
37    /// The exclusive lower bound of the page (no bound means start from the
38    /// beginning of the data-set).
39    after: Option<C>,
40
41    /// The exclusive upper bound of the page (no bound means continue to the
42    /// end of the data-set).
43    before: Option<C>,
44
45    /// Maximum number of entries in the page.
46    limit: u64,
47
48    /// In case there are more than `limit` entries in the range described by
49    /// `(after, before)`, this field states whether the entries up to limit
50    /// are taken from the `Front` or `Back` of that range.
51    end: End,
52}
53
54/// Whether the page is extracted from the beginning or the end of the range
55/// bounded by the cursors.
56#[derive(PartialEq, Eq, Debug, Clone, Copy)]
57pub(crate) enum End {
58    Front,
59    Back,
60}

62/// Results from the database that are pointed to by cursors.
63pub(crate) trait Paginated<C: CursorType>: Target<C> {
64    type Source: QuerySource;
65
66    /// Adds a filter to `query` to bound its result to be greater than or equal
67    /// to `cursor` (returning the new query).
68    fn filter_ge<ST, GB>(
69        cursor: &C,
70        query: Query<ST, Self::Source, GB>,
71    ) -> Query<ST, Self::Source, GB>;
72
73    /// Adds a filter to `query` to bound its results to be less than or equal
74    /// to `cursor` (returning the new query).
75    fn filter_le<ST, GB>(
76        cursor: &C,
77        query: Query<ST, Self::Source, GB>,
78    ) -> Query<ST, Self::Source, GB>;
79
80    /// Adds an `ORDER BY` clause to `query` to order rows according to their
81    /// cursor values (returning the new query). The `asc` parameter
82    /// controls whether the ordering is ASCending (`true`) or descending
83    /// (`false`).
84    fn order<ST, GB>(asc: bool, query: Query<ST, Self::Source, GB>) -> Query<ST, Self::Source, GB>;
85}

87/// Results from the database that are pointed to by cursors. Equivalent to
88/// `Paginated`, but for a `RawQuery`.
89pub(crate) trait RawPaginated<C: CursorType>: Target<C> {
90    /// Adds a filter to `query` to bound its result to be greater than or equal
91    /// to `cursor` (returning the new query).
92    fn filter_ge(cursor: &C, query: RawQuery) -> RawQuery;
93
94    /// Adds a filter to `query` to bound its results to be less than or equal
95    /// to `cursor` (returning the new query).
96    fn filter_le(cursor: &C, query: RawQuery) -> RawQuery;
97
98    /// Adds an `ORDER BY` clause to `query` to order rows according to their
99    /// cursor values (returning the new query). The `asc` parameter
100    /// controls whether the ordering is ASCending (`true`) or descending
101    /// (`false`).
102    fn order(asc: bool, query: RawQuery) -> RawQuery;
103}

111/// Interface for dealing with cursors that may come from a scan-limit-ed query.
112pub(crate) trait ScanLimited: Clone + PartialEq {
113    /// Whether the cursor was derived from a scan limit. Only applicable to the
114    /// `startCursor` and `endCursor` returned from a Connection's
115    /// `PageInfo`, and indicates that the cursor may not
116    /// have a corresponding node in the result set.
117    fn is_scan_limited(&self) -> bool {
118        false
119    }
120
121    /// Returns a version of the cursor that is not scan limited.
122    fn unlimited(&self) -> Self {
123        self.clone()
124    }

139impl<C> Page<C> {
140    /// Convert connection parameters into a page. Entries for the page are
141    /// drawn from the range `(after, before)` (Both bounds are optional).
142    /// The number of entries in the page is controlled by `first` and
143    /// `last`.
144    ///
145    /// - Setting both is in an error.
146    /// - Setting `first` indicates that the entries are taken from the front of the range.
147    /// - Setting `last` indicates that the entries are taken from the end of the range.
148    /// - Setting neither defaults the limit to the default page size in `config`, taken from the
149    ///   front of the range.
150    ///
151    /// It is an error to set a limit on page size that is greater than the
152    /// `config`'s max page size.
153    pub(crate) fn from_params(
154        config: &ServiceConfig,
155        first: Option<u64>,
156        after: Option<C>,
157        last: Option<u64>,
158        before: Option<C>,
159    ) -> Result<Self> {
160        let limits = &config.limits;
161        let page = match (first, after, last, before) {
162            (Some(_), _, Some(_), _) => return Err(Error::CursorNoFirstLast.extend()),
163
164            (limit, after, None, before) => Page {
165                after,
166                before,
167                limit: limit.unwrap_or(limits.default_page_size as u64),
168                end: End::Front,
169            },
170
171            (None, after, Some(limit), before) => Page {
172                after,
173                before,
174                limit,
175                end: End::Back,
176            },
177        };
178
179        if page.limit > limits.max_page_size as u64 {
180            return Err(Error::PageTooLarge(page.limit, limits.max_page_size).extend());
181        }
182
183        Ok(page)
184    }
185
186    /// A page that just limits the number of results, without applying any
187    /// other bounds.
188    pub(crate) fn bounded(limit: u64) -> Self {
189        Page {
190            after: None,
191            before: None,
192            limit,
193            end: End::Front,
194        }
195    }
196
197    pub(crate) fn after(&self) -> Option<&C> {
198        self.after.as_ref()
199    }
200
201    pub(crate) fn before(&self) -> Option<&C> {
202        self.before.as_ref()
203    }
204
205    pub(crate) fn limit(&self) -> usize {
206        self.limit as usize
207    }
208
209    pub(crate) fn is_from_front(&self) -> bool {
210        matches!(self.end, End::Front)
211    }
212
213    pub(crate) fn end(&self) -> End {
214        self.end
215    }
216}

218impl<C> Page<C>
219where
220    C: Checkpointed,
221{
222    /// If cursors are provided, defer to the `checkpoint_viewed_at` in the
223    /// cursor if they are consistent. Otherwise, use the value from the
224    /// parameter, or set to None. This is so that paginated queries are
225    /// consistent with the previous query that created the cursor.
226    pub(crate) fn validate_cursor_consistency(&self) -> Result<Option<u64>, Error> {
227        match (self.after(), self.before()) {
228            (Some(after), Some(before)) => {
229                if after.checkpoint_viewed_at() == before.checkpoint_viewed_at() {
230                    Ok(Some(after.checkpoint_viewed_at()))
231                } else {
232                    Err(Error::Client(
233                        "The provided cursors are taken from different checkpoints and cannot be used together in the same query."
234                            .to_string(),
235                    ))
236                }
237            }
238            // If only one cursor is provided, then we can directly use the checkpoint sequence
239            // number on it.
240            (Some(cursor), None) | (None, Some(cursor)) => Ok(Some(cursor.checkpoint_viewed_at())),
241            (None, None) => Ok(None),
242        }
243    }
244}

248pub(crate) struct ConsistentPage<I: Iterator<Item = ConsistentPageCursor>> {
249    /// Whether there is a previous page available
250    pub has_previous_page: bool,
251    /// Whether there is a next page available
252    pub has_next_page: bool,
253    /// The checkpoint viewed at for consistency
254    pub checkpoint_viewed_at: u64,
255    /// Iterator of cursors within the page
256    pub cursors: I,
257}

259impl Page<JsonCursor<ConsistentIndexCursor>> {
260    /// Treat the cursors of this Page as indices into a range [0, total).
261    /// Validates that the cursors of the page are consistent, and returns
262    /// two booleans indicating whether there is a previous or next page in
263    /// the range, the `checkpoint_viewed_at` to set for consistency, and an
264    /// iterator of cursors within that Page.
265    ///
266    /// The caller indexes a collection of `total` elements with the cursors
267    /// that the iterator gives, so each cursor index must be below `total`.
268    /// A cursor that carries `total` itself is a bound, not an element, and
269    /// gives the same page as an absent cursor.
270    ///
271    /// # Errors
272    ///
273    /// Returns [`Error::Client`] if the cursors are taken from different
274    /// checkpoints, or if a cursor index is above `total`.
275    pub(crate) fn paginate_consistent_indices(
276        &self,
277        total: usize,
278        checkpoint_viewed_at: u64,
279    ) -> Result<Option<ConsistentPage<impl Iterator<Item = ConsistentPageCursor>>>, Error> {
280        let cursor_viewed_at = self.validate_cursor_consistency()?;
281        let checkpoint_viewed_at = cursor_viewed_at.unwrap_or(checkpoint_viewed_at);
282
283        if let Some(after) = self.after() {
284            if after.ix > total {
285                return Err(Error::Client(format!(
286                    "`after` cursor (index {}) is above the number of elements, {total}.",
287                    after.ix
288                )));
289            }
290        }
291
292        if let Some(before) = self.before() {
293            if before.ix > total {
294                return Err(Error::Client(format!(
295                    "`before` cursor (index {}) is above the number of elements, {total}.",
296                    before.ix
297                )));
298            }
299        }
300
301        let mut lo = match self.after() {
302            Some(a) => a.ix.checked_add(1).ok_or_else(|| {
303                Error::Client(format!("`after` cursor (index {}) is out of range.", a.ix))
304            })?,
305            None => 0,
306        };
307
308        let mut hi = self.before().map_or(total, |b| b.ix);
309
310        if hi <= lo {
311            return Ok(None);
312        } else if (hi - lo) > self.limit() {
313            if self.is_from_front() {
314                hi = lo + self.limit();
315            } else {
316                lo = hi - self.limit();
317            }
318        }
319
320        Ok(Some(ConsistentPage {
321            has_previous_page: 0 < lo,
322            has_next_page: hi < total,
323            checkpoint_viewed_at,
324            cursors: (lo..hi).map(move |ix| {
325                JsonCursor::new(ConsistentIndexCursor {
326                    ix,
327                    c: UInt53::new_unchecked(checkpoint_viewed_at),
328                })
329            }),
330        }))
331    }
332}

334impl<C: CursorType + ScanLimited + Eq + Clone + Send + Sync + 'static> Page<C> {
335    /// Treat the cursors of this page as upper- and lowerbound filters for a
336    /// database `query`. Returns two booleans indicating whether there is a
337    /// previous or next page in the range, followed by an iterator of
338    /// values in the page, fetched from the database.
339    ///
340    /// The values returned implement `Target<C>`, so are able to compute their
341    /// own cursors.
342    ///
343    /// `checkpoint_viewed_at` is a required parameter to and passed to each
344    /// element to construct a consistent cursor.
345    pub(crate) fn paginate_query<T, Q, ST, GB>(
346        &self,
347        conn: &mut Conn<'_>,
348        checkpoint_viewed_at: u64,
349        query: Q,
350    ) -> QueryResult<(bool, bool, impl Iterator<Item = T>)>
351    where
352        Q: Fn() -> Query<ST, T::Source, GB>,
353        Query<ST, T::Source, GB>: LoadQuery<'static, DieselConn, T>,
354        Query<ST, T::Source, GB>: QueryFragment<DieselBackend>,
355        <T as Paginated<C>>::Source: Send + 'static,
356        <<T as Paginated<C>>::Source as QuerySource>::FromClause: Send + 'static,
357        Q: Send + 'static,
358        T: Send + Paginated<C> + 'static,
359        ST: Send + 'static,
360        GB: Send + 'static,
361    {
362        let page = self.clone();
363        let query = move || {
364            let mut query = query();
365            if let Some(after) = page.after() {
366                query = T::filter_ge(after, query);
367            }
368
369            if let Some(before) = page.before() {
370                query = T::filter_le(before, query);
371            }
372
373            // Load extra rows to detect the existence of pages on either side.
374            query = query.limit(page.limit() as i64 + 2);
375            T::order(page.is_from_front(), query)
376        };
377
378        let results: Vec<T> = if self.limit() == 0 {
379            // Avoid the database roundtrip in the degenerate case.
380            vec![]
381        } else {
382            let mut results = conn.results(query)?;
383            if !self.is_from_front() {
384                results.reverse();
385            }
386            results
387        };
388
389        Ok(self.paginate_results(
390            results.first().map(|f| f.cursor(checkpoint_viewed_at)),
391            results.last().map(|l| l.cursor(checkpoint_viewed_at)),
392            results,
393        ))
394    }
395
396    /// This function is similar to `paginate_query`, but is specifically
397    /// designed for handling `RawQuery`. Treat the cursors of this page as
398    /// upper- and lowerbound filters for a database `query`. Returns two
399    /// booleans indicating whether there is a previous or next page in the
400    /// range, followed by an iterator of values in the page, fetched from the
401    /// database.
402    ///
403    /// `checkpoint_viewed_at` is a required parameter to and passed to each
404    /// element to construct a consistent cursor.
405    pub(crate) fn paginate_raw_query<T>(
406        &self,
407        conn: &mut Conn<'_>,
408        checkpoint_viewed_at: u64,
409        query: RawQuery,
410    ) -> QueryResult<(bool, bool, impl Iterator<Item = T>)>
411    where
412        T: Send + RawPaginated<C> + FromSqlRow<Untyped, DieselBackend> + 'static,
413    {
414        let new_query = move || {
415            let query = self.apply::<T>(query.clone());
416            query.into_boxed()
417        };
418
419        let results: Vec<T> = if self.limit() == 0 {
420            // Avoid the database roundtrip in the degenerate case.
421            vec![]
422        } else {
423            let mut results: Vec<T> = conn.results(new_query)?;
424            if !self.is_from_front() {
425                results.reverse();
426            }
427            results
428        };
429
430        Ok(self.paginate_results(
431            results.first().map(|f| f.cursor(checkpoint_viewed_at)),
432            results.last().map(|l| l.cursor(checkpoint_viewed_at)),
433            results,
434        ))
435    }
436
437    /// Given the results of a database query, determine whether the result set
438    /// has a previous and next page and is consistent with the provided
439    /// cursors. Slightly different logic applies depending on whether the
440    /// provided cursors stem from either tip of the response, or if they
441    /// were derived from a scan limit.
442    ///
443    /// Returns two booleans indicating whether there is a previous or next page
444    /// in the range, followed by an iterator of values in the page, fetched
445    /// from the database. The values returned implement `Target<C>`, so are
446    /// able to compute their own cursors.
447    ///
448    /// `results` must be sorted in ascending cursor order and fetched with
449    /// inclusive cursor bounds, so that the rows at the `after`/`before`
450    /// cursors are part of the result set (see [`Page::apply`]).
451    pub(crate) fn paginate_results<T>(
452        &self,
453        f_cursor: Option<C>,
454        l_cursor: Option<C>,
455        results: Vec<T>,
456    ) -> (bool, bool, impl Iterator<Item = T>)
457    where
458        T: Target<C> + Send + 'static,
459    {
460        // Detect whether the results imply the existence of a previous or next page.
461        let (prev, next, prefix, suffix) =
462            match (self.after(), f_cursor, l_cursor, self.before(), self.end) {
463                // Results came back empty, despite supposedly including the `after` and `before`
464                // cursors, so the bounds must have been invalid, no matter which end the page was
465                // drawn from.
466                (_, None, _, _, _) | (_, _, None, _, _) => {
467                    return (false, false, vec![].into_iter());
468                }
469
470                // Page drawn from the front, and the cursor for the first element does not match
471                // `after`. If that cursor is not from a scan limit, then it must have appeared in
472                // the previous page, and should also be at the tip of the current page. This
473                // absence implies the bound was invalid, so we return an empty result.
474                (Some(a), Some(f), _, _, End::Front) if f != *a && !a.is_scan_limited() => {
475                    return (false, false, vec![].into_iter());
476                }
477
478                // Similar to above case, but for back of results.
479                (_, _, Some(l), Some(b), End::Back) if l != *b && !b.is_scan_limited() => {
480                    return (false, false, vec![].into_iter());
481                }
482
483                // From here onwards, we know that the results are non-empty. In the forward
484                // pagination scenario, the presence of a previous page is determined by whether a
485                // cursor supplied on the end the page is being drawn from is found in the first
486                // position. The presence of a next page is determined by whether we have more
487                // results than the provided limit, and/ or if the end cursor element appears in the
488                // result set.
489                (after, Some(f), Some(l), before, End::Front) => {
490                    let has_previous_page = after.is_some_and(|a| a.unlimited() == f);
491                    let prefix = has_previous_page as usize;
492
493                    // If results end with the before cursor, we will at least need to trim one
494                    // element from the suffix and we trim more off the end if
495                    // there is more after applying the limit.
496                    let mut suffix = before.is_some_and(|b| b.unlimited() == l) as usize;
497                    suffix += results.len().saturating_sub(self.limit() + prefix + suffix);
498                    let has_next_page = suffix > 0;
499
500                    (has_previous_page, has_next_page, prefix, suffix)
501                }
502
503                // Symmetric to the previous case, but drawing from the back.
504                (after, Some(f), Some(l), before, End::Back) => {
505                    // There is a next page if the last element of the results matches the `before`.
506                    // This last element will get pruned from the result set.
507                    let has_next_page = before.is_some_and(|b| b.unlimited() == l);
508                    let suffix = has_next_page as usize;
509
510                    let mut prefix = after.is_some_and(|a| a.unlimited() == f) as usize;
511                    prefix += results.len().saturating_sub(self.limit() + prefix + suffix);
512                    let has_previous_page = prefix > 0;
513
514                    (has_previous_page, has_next_page, prefix, suffix)
515                }
516            };
517
518        // If after trimming, we're going to return no elements, then forget whether
519        // there's a previous or next page, because there will be no start or
520        // end cursor for this page to anchor on.
521        if results.len() == prefix + suffix {
522            return (false, false, vec![].into_iter());
523        }
524
525        // We finally made it -- trim the prefix and suffix rows from the result and
526        // send it!
527        let mut results = results.into_iter();
528        if prefix > 0 {
529            results.nth(prefix - 1);
530        }
531        if suffix > 0 {
532            results.nth_back(suffix - 1);
533        }
534
535        (prev, next, results)
536    }
537
538    pub(crate) fn apply<T>(&self, mut query: RawQuery) -> RawQuery
539    where
540        T: RawPaginated<C>,
541    {
542        if let Some(after) = self.after() {
543            query = T::filter_ge(after, query);
544        }
545
546        if let Some(before) = self.before() {
547            query = T::filter_le(before, query);
548        }
549
550        query = T::order(self.is_from_front(), query);
551
552        query.limit(self.limit() as i64 + 2)
553    }
554}

704    /// A cursor index that is larger than the number of elements it refers to
705    /// must be rejected.
706    #[test]
707    fn test_consistent_indices_rejects_index_past_end() {
708        let config = ServiceConfig::default();
709        let total = 2;
710
711        let past_end = JsonCursor::new(ConsistentIndexCursor {
712            ix: 5,
713            c: UInt53::new_unchecked(0),
714        });
715        let max = JsonCursor::new(ConsistentIndexCursor {
716            ix: usize::MAX,
717            c: UInt53::new_unchecked(0),
718        });
719
720        for (first, after, last, before) in [
721            (None, None, None, Some(past_end.clone())),
722            (None, None, Some(1), Some(past_end.clone())),
723            (None, Some(past_end), None, None),
724            (None, None, None, Some(max.clone())),
725            (None, Some(max), None, None),
726        ] {
727            let page = Page::from_params(&config, first, after, last, before).unwrap();
728            assert!(matches!(
729                page.paginate_consistent_indices(total, 0),
730                Err(Error::Client(_))
731            ));
732        }
733    }
