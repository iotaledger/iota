4//! Consistent view queries: reconstruct object state at a specific checkpoint
5//! by combining unchanged objects from `checkpointed_objects` with previous
6//! versions from `objects_backward_history`.
7
8use crate::{
9    backward_view::{ACTIVE, OBJECT_COLUMNS, merge_and_deduplicate},
10    filter, query,
11    raw_query::RawQuery,
12    types::{
13        cursor::Page,
14        object::{Cursor, StoredBackwardObject},
15    },
16};
17
18/// Builds a consistent view at the given checkpoint by merging objects from
19/// `checkpointed_objects` that haven't changed since the target checkpoint
20/// with previous versions of objects that were superseded after it.
21pub(crate) fn query(
22    checkpoint_viewed_at: u64,
23    page: &Page<Cursor>,
24    filter_fn: impl Fn(RawQuery) -> RawQuery,
25) -> RawQuery {
26    let checkpoint_viewed_at = checkpoint_viewed_at as i64;
27    merge_and_deduplicate(vec![
28        consistent_checkpointed_objects(checkpoint_viewed_at, page, &filter_fn),
29        consistent_historical_objects(checkpoint_viewed_at, page, &filter_fn),
30    ])
31}
32
33/// Returns active objects from `checkpointed_objects` that were consistent
34/// also at the given checkpoint.
35///
36/// Uses a NOT EXISTS subquery against `objects_backward_history` to exclude
37/// objects that have any entry with
38/// `superseded_at_checkpoint > checkpoint_viewed_at`.
39///
40/// # Implementation notes
41///
42/// NOT EXISTS lets Postgres answer "did this object change?" row by row,
43/// with one index lookup each. A LEFT JOIN on a `SELECT DISTINCT` subquery
44/// takes that option away: the full list of changed objects must always be
45/// built first, and in the worst plans it is then also scanned for every
46/// row.
47fn consistent_checkpointed_objects(
48    checkpoint_viewed_at: i64,
49    page: &Page<Cursor>,
50    filter_fn: &impl Fn(RawQuery) -> RawQuery,
51) -> RawQuery {
52    let checkpointed_filtered = filter!(
53        filter_fn(query!(format!(
54            "SELECT {OBJECT_COLUMNS} FROM checkpointed_objects"
55        ))),
56        format!("object_status = {ACTIVE}")
57    );
58
59    let mut source = query!(
60        "SELECT candidates.* FROM ({}) candidates",
61        checkpointed_filtered
62    );
63    source = filter!(
64        source,
65        format!(
66            "NOT EXISTS (\
67                 SELECT 1 FROM objects_backward_history changed \
68                 WHERE changed.object_id = candidates.object_id \
69                   AND changed.superseded_at_checkpoint > {checkpoint_viewed_at})"
70        )
71    );
72    page.apply::<StoredBackwardObject>(source)
73}
74
75/// Returns active objects from `objects_backward_history` that were consistent
76/// at the given checkpoint.
77///
78/// Picks the earliest superseded version (`MIN(object_version)`) per object,
79/// which represents the state just before the first change after the target
80/// checkpoint. Keeps only `Active` entries: when that earliest version is a
81/// tombstone (or `NotYetCreated`), the object had no live state at the target
82/// checkpoint and drops out of the join.
83fn consistent_historical_objects(
84    checkpoint_viewed_at: i64,
85    page: &Page<Cursor>,
86    filter_fn: &impl Fn(RawQuery) -> RawQuery,
87) -> RawQuery {
88    let history_filtered = filter_fn(query!(format!(
89        "SELECT {OBJECT_COLUMNS} FROM objects_backward_history"
90    )));
91
92    let history_window = filter!(
93        history_filtered,
94        format!("superseded_at_checkpoint > {checkpoint_viewed_at} AND object_status = {ACTIVE}")
95    );
96
97    let oldest_subquery = query!(format!(
98        "SELECT object_id, MIN(object_version) AS min_version \
99         FROM objects_backward_history \
100         WHERE superseded_at_checkpoint > {checkpoint_viewed_at} \
101         GROUP BY object_id"
102    ));
103    let source = query!(
104        r#"SELECT candidates.* FROM ({}) candidates
105           JOIN ({}) oldest ON candidates.object_id = oldest.object_id
106               AND candidates.object_version = oldest.min_version"#,
107        history_window,
108        oldest_subquery
109    );
110    page.apply::<StoredBackwardObject>(source)
111}
