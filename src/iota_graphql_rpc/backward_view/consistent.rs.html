4//! Consistent view queries: reconstruct object state at a specific checkpoint
5//! by combining unchanged objects from `checkpointed_objects` with previous
6//! versions from `objects_backward_history`.
7
8use crate::{
9    backward_view::{ACTIVE, OBJECT_COLUMNS, merge_and_deduplicate},
10    filter, query,
11    raw_query::RawQuery,
12    types::{
13        cursor::Page,
14        object::{Cursor, StoredBackwardObject},
15    },
16};
17
18/// Builds a consistent view at the given checkpoint by merging objects from
19/// `checkpointed_objects` that haven't changed since the target checkpoint
20/// with previous versions of objects that were superseded after it.
21pub(crate) fn query(
22    checkpoint_viewed_at: u64,
23    page: &Page<Cursor>,
24    filter_fn: impl Fn(RawQuery) -> RawQuery,
25) -> RawQuery {
26    let checkpoint_viewed_at = checkpoint_viewed_at as i64;
27    merge_and_deduplicate(vec![
28        consistent_checkpointed_objects(checkpoint_viewed_at, page, &filter_fn),
29        consistent_historical_objects(checkpoint_viewed_at, page, &filter_fn),
30    ])
31}
32
33/// Returns active objects from `checkpointed_objects` that were consistent
34/// also at the given checkpoint.
35///
36/// Uses a LATERAL join against `objects_backward_history` to exclude objects
37/// that have any entry with `superseded_at_checkpoint > checkpoint_viewed_at`.
38///
39/// # Implementation notes
40///
41/// The LATERAL join lets Postgres answer "did this object change?" row by row:
42/// the object's newest version is read by `ORDER BY object_version DESC`, to
43/// achieve the backward index scan.
44///
45/// Two alternatives to avoid:
46///
47/// - A join on a `SELECT DISTINCT` subquery: the full list of changed objects
48///   must be built first, and in the worst plans it is then scanned for every
49///   row.
50/// - Filtering directly on `superseded_at_checkpoint > checkpoint_viewed_at`
51///   (e.g. `NOT EXISTS`) without ordering by version: it scans all objects
52///   version in the table, forwards, meaning that query cost grows with
53///   retention period.
54fn consistent_checkpointed_objects(
55    checkpoint_viewed_at: i64,
56    page: &Page<Cursor>,
57    filter_fn: &impl Fn(RawQuery) -> RawQuery,
58) -> RawQuery {
59    let checkpointed_filtered = filter!(
60        filter_fn(query!(format!(
61            "SELECT {OBJECT_COLUMNS} FROM checkpointed_objects"
62        ))),
63        format!("object_status = {ACTIVE}")
64    );
65
66    let mut source = query!(
67        "SELECT candidates.* FROM ({}) candidates \
68         LEFT JOIN LATERAL (\
69             SELECT changed.superseded_at_checkpoint \
70             FROM objects_backward_history changed \
71             WHERE changed.object_id = candidates.object_id \
72             ORDER BY changed.object_version DESC \
73             LIMIT 1\
74         ) latest ON true",
75        checkpointed_filtered
76    );
77    source = filter!(
78        source,
79        format!(
80            "(latest.superseded_at_checkpoint IS NULL \
81              OR latest.superseded_at_checkpoint <= {checkpoint_viewed_at})"
82        )
83    );
84    page.apply::<StoredBackwardObject>(source)
85}

87/// Returns active objects from `objects_backward_history` that were consistent
88/// at the given checkpoint.
89///
90/// Picks the earliest superseded version per object, which represents the state
91/// just before the first change after the target checkpoint. Keeps only
92/// `Active` entries: when that earliest version is a tombstone (or
93/// `NotYetCreated`), the object had no live state at the target checkpoint and
94/// drops out.
95///
96/// # Implementation notes
97///
98/// The candidate object_ids matching the filter are deduplicated first, so the
99/// per-object live-version lookup runs once per object and only matching
100/// objects are visited. The lookup finds the newest version already superseded
101/// at the checkpoint by scanning `object_version DESC` and returns the next
102/// one, so it reads only the object's recent versions. `MIN(object_version)
103/// WHERE object_id = ... AND superseded_at_checkpoint > cp` would instead scan
104/// up from the oldest version through everything superseded by the checkpoint -
105/// slower the more history an object has.
106fn consistent_historical_objects(
107    checkpoint_viewed_at: i64,
108    page: &Page<Cursor>,
109    filter_fn: &impl Fn(RawQuery) -> RawQuery,
110) -> RawQuery {
111    // Distinct object_ids that changed after the checkpoint and match the filter.
112    // The filtered columns are indexed on objects_backward_history, so this scans
113    // only matching objects - except an unfiltered listing, which scans the window.
114    let candidate_ids = filter!(
115        filter_fn(query!(
116            "SELECT DISTINCT object_id FROM objects_backward_history"
117        )),
118        format!("superseded_at_checkpoint > {checkpoint_viewed_at}")
119    );
120    let (candidate_ids_sql, binds) = candidate_ids.finish();
121
122    // The newest version already superseded by the checkpoint; the live version
123    // is the next one after it.
124    let newest_superseded = format!(
125        "SELECT superseded.object_version \
126         FROM objects_backward_history superseded \
127         WHERE superseded.object_id = candidate_ids.object_id \
128           AND superseded.superseded_at_checkpoint <= {checkpoint_viewed_at} \
129         ORDER BY superseded.object_version DESC \
130         LIMIT 1"
131    );
132
133    // For each candidate, fetch the version that was live at the checkpoint: the
134    // earliest one not yet superseded. A LATERAL join fetches that row per
135    // object_id, using the (object_id, object_version) index.
136    //
137    // `object_id` comes from `candidate_ids`, not `live`, so the page's `ORDER BY
138    // object_id` needs no sort after the LATERAL and the LIMIT can early-terminate
139    // after one page.
140    let live_rows = RawQuery::new(
141        format!(
142            "SELECT {OBJECT_COLUMNS} FROM ( \
143                 SELECT candidate_ids.object_id, live.object_version, live.object_status, \
144                        live.object_digest, live.owner_type, live.owner_id, live.object_type, \
145                        live.object_type_package, live.object_type_module, live.object_type_name, \
146                        live.serialized_object, live.coin_type, live.coin_balance, live.df_kind \
147                 FROM ({candidate_ids_sql}) candidate_ids \
148                 CROSS JOIN LATERAL ( \
149                     SELECT objects_backward_history.* \
150                     FROM objects_backward_history \
151                     WHERE objects_backward_history.object_id = candidate_ids.object_id \
152                       AND objects_backward_history.object_version > COALESCE(({newest_superseded}), -1) \
153                     ORDER BY objects_backward_history.object_version ASC \
154                     LIMIT 1 \
155                 ) live \
156             ) AS objects_backward_history"
157        ),
158        binds,
159    );
160
161    // An object is in candidate_ids if any of its versions superseded after the
162    // checkpoint matched the filter, but the version live at the checkpoint might
163    // not. Re-apply the filter and require Active to drop those.
164    let history_window = filter!(filter_fn(live_rows), format!("object_status = {ACTIVE}"));
165
166    let source = query!("SELECT candidates.* FROM ({}) candidates", history_window);
167    page.apply::<StoredBackwardObject>(source)
168}
