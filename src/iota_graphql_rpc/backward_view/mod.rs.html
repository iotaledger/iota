15/// An `ObjectFilter` validated for use against the historical view.
16///
17/// The historical view is keyed on `(object_id, object_version)` lookups
18/// against `checkpointed_objects`, `objects_backward_history`, and (for
19/// keys-only queries) `objects_version`, so it only makes sense when the
20/// filter pins down specific keys. Constructible only via
21/// `TryFrom<ObjectFilter>`, which requires `object_keys` to be set.
22#[derive(Debug)]
23pub(crate) struct HistoricalFilter(ObjectFilter);
24
25#[derive(thiserror::Error, Debug)]
26pub(crate) enum HistoricalFilterError {
27    #[error(
28        "ObjectFilter is missing object_keys; the historical view requires explicit (object_id, object_version) keys"
29    )]
30    MissingKeys,
31}
32
33impl TryFrom<ObjectFilter> for HistoricalFilter {
34    type Error = HistoricalFilterError;
35
36    fn try_from(filter: ObjectFilter) -> Result<Self, Self::Error> {
37        if filter.object_keys.is_some() {
38            Ok(Self(filter))
39        } else {
40            Err(HistoricalFilterError::MissingKeys)
41        }
42    }
43}
44
45impl HistoricalFilter {
46    pub(crate) fn apply(&self, query: RawQuery) -> RawQuery {
47        self.0.apply(query)
48    }
49}

68/// Merges any non-empty set of sources with `UNION ALL` and picks the most
69/// recent version per `object_id` using `DISTINCT ON`.
70///
71/// The result is wrapped so cursor pagination can reference
72/// `candidates.object_id`.
73pub(super) fn merge_and_deduplicate(sources: Vec<RawQuery>) -> RawQuery {
74    assert!(
75        !sources.is_empty(),
76        "merge_and_deduplicate requires at least one source"
77    );
78
79    let mut binds: Vec<String> = Vec::new();
80    let union_terms: Vec<String> = sources
81        .into_iter()
82        .map(|source| {
83            let (sql, source_binds) = source.finish();
84            binds.extend(source_binds);
85            format!("({sql})")
86        })
87        .collect();
88
89    let select = format!(
90        r#"SELECT DISTINCT ON (object_id) * FROM ({}) candidates"#,
91        union_terms.join(" UNION ALL ")
92    );
93
94    let combined = RawQuery::new(select, binds)
95        .order_by("object_id")
96        .order_by("object_version DESC");
97
98    query!("SELECT * FROM ({}) candidates", combined)
99}
