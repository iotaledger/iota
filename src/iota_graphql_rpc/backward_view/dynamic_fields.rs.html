4//! Consistent view of the dynamic fields of a parent at the moment the
5//! parent reached the requested `parent_version`.
6//!
7//! For each candidate DF, the target version is the largest version in
8//! `objects_version` whose value is `<= parent_version`. The state row at
9//! `(object_id, target_version)` is then read from `checkpointed_objects` (if
10//! it's the current state) or `objects_backward_history` (if it's a prior
11//! state). When the target version is a tombstone or `NotYetCreated` marker,
12//! `owner_id`/`df_kind`/etc. are NULL — the `owner_id`/`df_kind` filter
13//! applied internally drops it.
14//!
15//! Returned semantics: for each DF, the earliest state it had while the
16//! parent was at `parent_version`. Note that there can be multiple distinct DF
17//! states observable at the same `parent_version`, we return the first one.
18

21use crate::{

24    raw_query::RawQuery,

26        cursor::Page,
27        iota_address::IotaAddress,
28        object::{Cursor, StoredBackwardObject},
29    },
30};

32/// Builds a consistent view of the dynamic fields owned by `parent` at
33/// `parent_version`.
34///
35/// The DF-shape filter
36/// (`owner_id = parent AND owner_type = Object AND df_kind IS NOT NULL`) is
37/// applied internally — it both narrows the candidate set and excludes
38/// non-Active rows (whose `owner_id`/`df_kind` are NULL).
39pub(crate) fn query(parent: IotaAddress, parent_version: u64, page: &Page<Cursor>) -> RawQuery {
40    let parent_version = parent_version as i64;
41    let parent_filter = parent_dynamic_field_filter(parent);
42    merge_and_deduplicate(vec![
43        dynamic_fields_from_checkpointed_objects(parent_version, page, &parent_filter),
44        dynamic_fields_from_historical_objects(parent_version, page, &parent_filter),
45    ])
46}
47
48/// Returns a filter that constrains a row to be a dynamic field of `parent`.
49///
50/// Applied to both Source A and Source B; excludes non-Active rows by virtue
51/// of their NULL `owner_id`/`df_kind` columns.
52fn parent_dynamic_field_filter(parent: IotaAddress) -> impl Fn(RawQuery) -> RawQuery {
53    move |q| {
54        filter!(
55            q,
56            format!(
57                "owner_id = '\\x{}'::bytea AND owner_type = {} AND df_kind IS NOT NULL",
58                hex::encode(parent.into_vec()),
59                OwnerType::Object as i16
60            )
61        )
62    }
63}

65/// Source A — DFs whose current state in `checkpointed_objects` is at or
66/// before `parent_version`.
67///
68/// Picks rows where `co.object_version <= parent_version`. Since
69/// `checkpointed_objects` tracks each object's latest version, this version
70/// is also `MAX(objects_version.object_version)` for the id, so the simple
71/// `<=` check is equivalent to "the largest known version of this object
72/// is at or before `parent_version`" — no `objects_version` lookup is
73/// needed here.
74///
75/// # Constraints
76///
77/// Excludes rows that also appear in `objects_backward_history` at the same
78/// `(object_id, object_version)`, keeping Source A and Source B disjoint at
79/// the row level. This is needed because a `(id, version)` pair can
80/// briefly exist in both tables during the race window between
81/// `objects_backward_history` and `checkpointed_objects` writes.
82///
83/// # Implementation notes
84///
85/// The merge step uses `DISTINCT ON` to deduplicate *within a page* only;
86/// source-level disjointness is required to keep the result correct across
87/// page boundaries.
88fn dynamic_fields_from_checkpointed_objects(
89    parent_version: i64,
90    page: &Page<Cursor>,
91    filter_fn: &impl Fn(RawQuery) -> RawQuery,
92) -> RawQuery {
93    let checkpointed_filtered = filter_fn(query!(format!(
94        "SELECT {OBJECT_COLUMNS} FROM checkpointed_objects"
95    )));
96
97    let with_target = filter!(
98        checkpointed_filtered,
99        format!("object_version <= {parent_version}")
100    );
101
102    let no_overlap = filter!(
103        with_target,
104        "NOT EXISTS (\
105             SELECT 1 FROM objects_backward_history bh \
106             WHERE bh.object_id = checkpointed_objects.object_id \
107               AND bh.object_version = checkpointed_objects.object_version)"
108    );
109
110    let source = query!("SELECT candidates.* FROM ({}) candidates", no_overlap);
111    page.apply::<StoredBackwardObject>(source)
112}
113
114/// Source B — DFs whose prior state in `objects_backward_history` is the
115/// one current at `parent_version`.
116///
117/// For each `object_id`, picks the row in `objects_backward_history` whose
118/// `object_version` equals `target_v`, where `target_v` is the largest
119/// `objects_version.object_version` entry `<= parent_version`. The chosen
120/// row carries the prior-state data of the DF as it was while the parent
121/// was at `parent_version`.
122///
123/// # Implementation notes
124///
125/// `objects_version` is consulted as the version-timeline authority (it
126/// records every real version, including tombstone versions). When
127/// `target_v` lands on a tombstone version, no Active state row exists at
128/// that key — the candidate naturally drops out of the result.
129///
130/// The query is shaped so that DF `object_id`s from
131/// `objects_backward_history` that match the DF filter
132/// (`owner_id = parent AND owner_type = Object AND df_kind IS NOT NULL`)
133/// are first deduplicated. Without the dedupe, the `MAX(object_version)`
134/// subquery would be evaluated once per version instead of once per found
135/// object, which for objects with many historical versions can kill the
136/// performance.
137///
138/// The dedupe step relies on the partial index
139/// `objects_backward_history_owner_df` on `(owner_id, object_id) WHERE
140/// owner_type = 2 AND df_kind IS NOT NULL`.
141fn dynamic_fields_from_historical_objects(
142    parent_version: i64,
143    page: &Page<Cursor>,
144    filter_fn: &impl Fn(RawQuery) -> RawQuery,
145) -> RawQuery {
146    // Distinct DF object_ids owned by the parent. `filter_fn` adds the
147    // `owner_id = $parent AND owner_type = 2 AND df_kind IS NOT NULL`
148    // predicate, which together with the partial index lets Postgres do
149    // an Index Only Scan to find these ids.
150    let df_ids = filter_fn(query!(
151        "SELECT DISTINCT object_id FROM objects_backward_history"
152    ));
153
154    // For each distinct DF, fetch the backward-history row at target_v
155    // via the correlated MAX subquery.
156    let (df_ids_sql, df_ids_binds) = df_ids.finish();
157    let sql = format!(
158        "SELECT {OBJECT_COLUMNS} FROM ( \
159             SELECT objects_backward_history.* \
160             FROM ({df_ids_sql}) df_ids \
161             JOIN objects_backward_history \
162                 ON objects_backward_history.object_id = df_ids.object_id \
163             WHERE objects_backward_history.object_version = ( \
164                 SELECT MAX(object_version) FROM objects_version \
165                 WHERE object_id = df_ids.object_id \
166                   AND object_version <= {parent_version}) \
167         ) AS objects_backward_history"
168    );
169    let with_target = RawQuery::new(sql, df_ids_binds);
170
171    let source = query!("SELECT candidates.* FROM ({}) candidates", with_target);
172    page.apply::<StoredBackwardObject>(source)
173}
