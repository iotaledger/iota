18pub(crate) static VERSION_HEADER: HeaderName = HeaderName::from_static("x-iota-rpc-version");
19pub(crate) static MIN_SDK_VERSION_HEADER: HeaderName =
20    HeaderName::from_static("x-iota-min-sdk-version");
21
22/// Lowest `iota-sdk-graphql-client` version able to decode this service's
23/// responses, sent in the [`MIN_SDK_VERSION_HEADER`] of every response.
24///
25/// Bump this alongside SDK releases containing wire-visible changes (e.g., new enum variants or
26/// renamed fields) to force older clients to reject the service and fail safely rather than
27/// panicking on un-decodable values.
28pub(crate) const MIN_SDK_VERSION: &str = "1.0.0-rc.1";

30#[expect(unused)]
31pub(crate) struct IotaRpcVersion(Vec<u8>, Vec<Vec<u8>>);
32const NAMED_VERSIONS: [&str; 3] = ["beta", "legacy", "stable"];
33
34impl headers::Header for IotaRpcVersion {
35    fn name() -> &'static HeaderName {
36        &VERSION_HEADER
37    }
38
39    fn decode<'i, I>(values: &mut I) -> Result<Self, headers::Error>
40    where
41        I: Iterator<Item = &'i HeaderValue>,
42    {
43        let mut values = values.map(|v| v.as_bytes().to_owned());
44        let Some(value) = values.next() else {
45            // No values for this header -- it doesn't exist.
46            return Err(headers::Error::invalid());
47        };
48
49        // Extract the header values as bytes.  Distinguish the first value as we expect
50        // there to be just one under normal operation.  Do not attempt to parse
51        // the value, as a header parsing failure produces a generic error.
52        Ok(IotaRpcVersion(value, values.collect()))
53    }
54
55    fn encode<E: Extend<HeaderValue>>(&self, _values: &mut E) {
56        unimplemented!()
57    }
58}

60/// Middleware to check for the existence of a version constraint in the request
61/// header, and confirm that this instance of the RPC matches that version
62/// constraint.  Each RPC instance only supports one version of the RPC
63/// software, and it is the responsibility of the load balancer to make sure
64/// version constraints are met.
65pub(crate) async fn check_version_middleware(
66    version: Option<Path<String>>,
67    State(service_version): State<Version>,
68    request: Request,
69    next: Next,
70) -> Response {
71    let Some(Path(version)) = version else {
72        return next.run(request).await;
73    };
74
75    if NAMED_VERSIONS.contains(&version.as_str()) || version.is_empty() {
76        return next.run(request).await;
77    }
78    let Some((year, month)) = parse_version(&version) else {
79        return (
80                StatusCode::BAD_REQUEST,
81                graphql_error_response(
82                    code::BAD_REQUEST,
83                    format!(
84                        "Failed to parse version path: {version}. Expected either a `beta | legacy | stable` \
85                    version or <YEAR>.<MONTH> version.",
86                    ),
87                ),
88            )
89                .into_response();
90    };
91
92    if year != service_version.year || month != service_version.month {
93        return (
94            StatusCode::MISDIRECTED_REQUEST,
95            graphql_error_response(
96                code::INTERNAL_SERVER_ERROR,
97                format!("Version '{version}' not supported."),
98            ),
99        )
100            .into_response();
101    }
102    next.run(request).await
103}

105/// Mark every outgoing response with a header indicating the precise version of
106/// the RPC that was used (including the patch version and sha), and one with
107/// the lowest SDK version able to decode it.
108pub(crate) async fn set_version_middleware(
109    State(version): State<Version>,
110    request: Request,
111    next: Next,
112) -> Response {
113    let mut response = next.run(request).await;
114    let headers = response.headers_mut();
115    headers.insert(
116        VERSION_HEADER.clone(),
117        HeaderValue::from_static(version.full),
118    );
119    headers.insert(
120        MIN_SDK_VERSION_HEADER.clone(),
121        HeaderValue::from_static(MIN_SDK_VERSION),
122    );
123    response
124}

126/// Split a `version` string into two parts (year and month) separated by a ".".
127///
128/// Confirms that the version specifier contains exactly two components, and
129/// that both components are entirely comprised of digits.
130fn parse_version(version: &str) -> Option<(&str, &str)> {
131    let mut parts = version.split('.');
132    let year = parts.next()?;
133    let month = parts.next()?;
134
135    if year.is_empty() || month.is_empty() {
136        return None;
137    }
138
139    (parts.next().is_none()
140        && year.chars().all(|c| c.is_ascii_digit())
141        && month.chars().all(|c| c.is_ascii_digit()))
142    .then_some((year, month))
143}

163    fn metrics() -> Metrics {
164        let binding_address: SocketAddr = "0.0.0.0:9185".parse().unwrap();
165        let registry = iota_metrics::start_prometheus_server(binding_address).default_registry();
166        Metrics::new(&registry)

168    fn service() -> Router {
169        let version = Version::for_testing();
170        let metrics = metrics();
171        let cancellation_token = CancellationToken::new();
172        let connection_config = ConnectionConfig::default();
173        let service_config = ServiceConfig::default();
174        let state = AppState::new(
175            connection_config,
176            service_config,
177            metrics,
178            cancellation_token,
179            version,
180        );
181
182        Router::new()
183            .route("/", get(|| async { "Hello, Versioning!" }))
184            .route("/{version}", get(|| async { "Hello, Versioning!" }))
185            .route("/graphql", get(|| async { "Hello, Versioning!" }))
186            .route("/graphql/{version}", get(|| async { "Hello, Versioning!" }))
187            .layer(middleware::from_fn_with_state(
188                state.version,
189                check_version_middleware,
190            ))
191            .layer(middleware::from_fn_with_state(
192                state.version,
193                set_version_middleware,
194            ))
195    }

198        Request::builder()
199            .uri("/graphql")
200            .body(Body::empty())

204    fn plain_request() -> Request<Body> {
205        Request::builder().uri("/").body(Body::empty()).unwrap()
206    }

208    fn version_request(version: &str) -> Request<Body> {
209        if version.is_empty() {
210            return plain_request();
211        }
212        Request::builder()
213            .uri(format!("/graphql/{version}"))
214            .body(Body::empty())
215            .unwrap()
216    }

218    async fn response_body(response: Response) -> String {
219        let bytes = response.into_body().collect().await.unwrap();
220        let value: serde_json::Value = serde_json::from_slice(bytes.to_bytes().as_ref()).unwrap();
221        serde_json::to_string_pretty(&value).unwrap()
222    }
