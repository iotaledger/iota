68    /// Create a new instance of the client.
69    ///
70    /// Internally it instantiates a BigTableDB client.
71    pub async fn new(config: KvStoreConfig) -> Result<Self> {
72        let bigtable_client = if let Some(emulator_host) = config.emulator_host {
73            info!(
74                instance_id = %config.instance_id,
75                %emulator_host,
76                "connecting to BigTable emulator"
77            );
78            BigTableClient::new_local(
79                &emulator_host,
80                "iota-rest-kv",
81                config.instance_id,
82                config.column_family,
83            )?
84        } else {
85            info!(
86                instance_id = %config.instance_id,
87                timeout_secs = config.timeout_secs,
88                "connecting to remote BigTable"
89            );
90            BigTableClient::new_remote(
91                config.instance_id,
92                true,
93                Some(Duration::from_secs(config.timeout_secs as u64)),
94                "rest".to_string(),
95                config.column_family,
96                None,
97            )
98            .await?
99            .with_backoff(BigTableClient::default_backoff())
100        };
101
102        Ok(Self {
103            bigtable_client,
104            start_time: Instant::now(),
105        })
106    }

125    /// Builds a [`RowFilter`] that matches only cells whose column qualifier
126    /// equals `column_qualifier` exactly.
127    ///
128    /// Internally the filter is implemented as a regex anchored with `^` and
129    /// `$` to enforce an exact byte match of the provided
130    /// `column_qualifier`, preventing partial or substring matches against
131    /// other columns whose qualifiers happen to contain `column_qualifier`
132    /// as a prefix, suffix, or substring.
133    fn column_qualifier_filter(column_qualifier: &str) -> RowFilter {
134        RowFilter {
135            filter: Some(Filter::ColumnQualifierRegexFilter(
136                format!("^{column_qualifier}$").into_bytes(),
137            )),
138        }
139    }

147    ///
148    /// Based on the provided [`Key`] fetch the data from BigTableDB.
149    pub async fn get(&self, key: Key) -> Result<Option<Bytes>, ApiError> {

154    /// Gets multiple values as [`Vec`]<[`Option`]<[`Bytes`]>> from the kv
155    /// store.
156    ///
157    /// Based on the provided [`Vec`]<[`Key`]> fetch the data from BigTableDB.
158    /// Returns a vector of the same length and order as the input keys.
159    /// Each entry is `Some(bytes)` if the key was found, or `None` if not
160    /// found.
161    ///
162    /// All keys must be of the same type, otherwise [`ApiError::BadRequest`] is
163    /// returned.
164    pub async fn get_items(&self, keys: Vec<Key>) -> Result<Vec<Option<Bytes>>, ApiError> {
165        if keys.is_empty() {
166            return Ok(Vec::new());
167        }
168
169        // Use the first key to determine the type - all keys should be of the same type
170        match keys.first().expect("emptiness was checked earlier") {
171            Key::TransactionDigestsByAddress(_address) => {
172                return Err(ApiError::BadRequest("unsupported key".into()));
173            }
174            Key::Transaction(_) => {
175                let digests = extract_keys(&keys, |k| match k {
176                    Key::Transaction(digest) => Some(*digest),
177                    _ => None,
178                })?;
179
180                let keys = digests.iter().map(|tx| Some(tx.bytes().to_vec())).collect();
181
182                self.fetch_from_bigtable(TRANSACTIONS_TABLE, keys, TRANSACTION_COLUMN_QUALIFIER)
183                    .await
184            }
185            Key::TransactionEffects(_) => {
186                let digests = extract_keys(&keys, |k| match k {
187                    Key::TransactionEffects(digest) => Some(*digest),
188                    _ => None,
189                })?;
190
191                let keys = digests.iter().map(|tx| Some(tx.bytes().to_vec())).collect();
192
193                self.fetch_from_bigtable(TRANSACTIONS_TABLE, keys, EFFECTS_COLUMN_QUALIFIER)
194                    .await
195            }
196            Key::CheckpointContents(_) => {
197                let seq_nums = extract_keys(&keys, |k| match k {
198                    Key::CheckpointContents(seq_num) => Some(*seq_num),
199                    _ => None,
200                })?;
201
202                let keys = seq_nums
203                    .iter()
204                    .map(|sq| Some(sq.to_be_bytes().to_vec()))
205                    .collect();
206
207                self.fetch_from_bigtable(
208                    CHECKPOINTS_TABLE,
209                    keys,
210                    CHECKPOINT_CONTENTS_COLUMN_QUALIFIER,
211                )
212                .await
213            }
214            Key::CheckpointSummary(_) => {
215                let seq_nums = extract_keys(&keys, |k| match k {
216                    Key::CheckpointSummary(seq_num) => Some(*seq_num),
217                    _ => None,
218                })?;
219
220                let keys = seq_nums
221                    .iter()
222                    .map(|sq| Some(sq.to_be_bytes().to_vec()))
223                    .collect();
224
225                self.fetch_from_bigtable(
226                    CHECKPOINTS_TABLE,
227                    keys,
228                    CHECKPOINT_SUMMARY_COLUMN_QUALIFIER,
229                )
230                .await
231            }
232            Key::CheckpointSummaryByDigest(_) => {
233                let checkpoint_digests = extract_keys(&keys, |k| match k {
234                    Key::CheckpointSummaryByDigest(checkpoint_digest) => Some(*checkpoint_digest),
235                    _ => None,
236                })?;
237
238                let digest_keys = checkpoint_digests
239                    .iter()
240                    .map(|digest| Some(digest.bytes().to_vec()))
241                    .collect::<Vec<Option<Vec<u8>>>>();
242
243                self.checkpoint_summary_by_digests(digest_keys).await
244            }
245            Key::TransactionToCheckpoint(_) => {
246                let digests = extract_keys(&keys, |k| match k {
247                    Key::TransactionToCheckpoint(digest) => Some(*digest),
248                    _ => None,
249                })?;
250
251                let keys = digests.iter().map(|tx| Some(tx.bytes().to_vec())).collect();
252
253                self.fetch_from_bigtable(TRANSACTIONS_TABLE, keys, TRANSACTION_TO_CHECKPOINT)
254                    .await
255            }
256            Key::ObjectKey(_) => {
257                let object_keys = extract_keys(&keys, |k| match k {
258                    Key::ObjectKey(object_key) => Some(*object_key),
259                    _ => None,
260                })?;
261
262                let keys = object_keys
263                    .iter()
264                    .map(|key| Some(raw_object_key(key)))
265                    .collect();
266
267                self.fetch_from_bigtable(OBJECTS_TABLE, keys, DEFAULT_COLUMN_QUALIFIER)
268                    .await
269            }
270            Key::EventsByTransactionDigest(_) => {
271                let digests = extract_keys(&keys, |k| match k {
272                    Key::EventsByTransactionDigest(digest) => Some(*digest),
273                    _ => None,
274                })?;
275
276                let keys = digests.iter().map(|tx| Some(tx.bytes().to_vec())).collect();
277
278                let response = self
279                    .fetch_from_bigtable(TRANSACTIONS_TABLE, keys, EVENTS_COLUMN_QUALIFIER)
280                    .await?;
281
282                Ok(response
283                    .into_iter()
284                    .map(|cell| {
285                        cell.and_then(|bytes| {
286                            match bcs::from_bytes::<Option<TransactionEvents>>(&bytes) {
287                                Ok(None) | Err(_) => None,
288                                Ok(Some(events)) => bcs::to_bytes(&events).map(Bytes::from).ok(),
289                            }
290                        })
291                    })
292                    .collect())
293            }
294        }
295        .map_err(Into::into)
296    }

298    /// Fetch multiple values from a BigTable table with a specific key and
299    /// column qualifier.
300    ///
301    /// Keys wrapped in `Option<Vec<u8>>` allow chaining multiple queries: the
302    /// result from one `fetch_from_bigtable` (which contains `None` for missing
303    /// keys) can be directly passed as input to the next call. `None` keys
304    /// are skipped in the query but preserve their position in the result.
305    ///
306    /// The result's length is guaranteed to match the input `keys` length. Each
307    /// position in the result corresponds to the key at the same position in
308    /// the input. This allows the caller to easily determine which
309    /// requested keys have data:
310    /// - `Some(value)` at index `i` means `key[i]` exists and has data
311    /// - `None` at index `i` means `key[i]` was not found or has no matching
312    ///   data
313    #[instrument(
314        level = "debug",
315        skip_all,
316        fields(table = table_name, column_qualifier = column_qualifier, num_keys = keys.len())
317    )]
318    async fn fetch_from_bigtable(
319        &self,
320        table_name: &str,
321        keys: Vec<Option<Vec<u8>>>,
322        column_qualifier: &str,
323    ) -> Result<Vec<Option<Bytes>>, anyhow::Error> {
324        let mut client = self.bigtable_client.clone();
325        // pre-allocate results with None. Matching cells will replace None with
326        // Some(value), and unmatched keys will remain None.
327        let mut results = vec![None; keys.len()];
328
329        let key_to_index = keys
330            .iter()
331            .enumerate()
332            .filter_map(|(index, key)| key.as_ref().map(|k| (k.clone(), index)))
333            .collect::<HashMap<Vec<u8>, usize>>();
334
335        // With no keys to look up, `multi_get` would read the whole table
336        // instead of nothing, so return the empty result without calling it.
337        if key_to_index.is_empty() {
338            return Ok(results);
339        }
340
341        let start = Instant::now();
342        for row in client
343            .multi_get(
344                table_name,
345                key_to_index.keys().cloned().collect(),
346                Some(Self::column_qualifier_filter(column_qualifier)),
347            )
348            .await?
349        {
350            for Cell { name, value } in row.cells {
351                let cell_name = std::str::from_utf8(&name)?;
352                if cell_name == column_qualifier {
353                    if let Some(&index) = key_to_index.get(&row.key) {
354                        results[index] = Some(Bytes::from(value));
355                    }
356                } else {
357                    error!("unexpected column {cell_name:?} in {table_name} table")
358                }
359            }
360        }
361
362        let found = results.iter().filter(|value| value.is_some()).count();
363        debug!(
364            found,
365            missing = results.len() - found,
366            elapsed_ms = start.elapsed().as_millis() as u64,
367            "fetched rows from BigTable"
368        );
369
370        Ok(results)
371    }

374    async fn checkpoint_summary_by_digests(
375        &self,
376        keys: Vec<Option<Vec<u8>>>,
377    ) -> Result<Vec<Option<Bytes>>, anyhow::Error> {

395    /// Performs a single reverse range scan against the objects table and
396    /// returns the bytes of the highest stored version that is **strictly
397    /// less than** the version encoded in [`ObjectRangeKeyBound::end_key`].
398    ///
399    /// Returns `None` if no version below the requested one exists for that
400    /// object (or if the object is not stored at all).
401    #[instrument(level = "debug", skip_all)]
402    pub async fn object_before_version(
403        &self,
404        range: ObjectRangeKeyBound,
405    ) -> Result<Option<Bytes>, anyhow::Error> {
406        if range.is_empty() {
407            return Ok(None);
408        }
409
410        let mut client = self.bigtable_client.clone();
411
412        let start = Instant::now();
413        let reversed = true;
414        let rows_limit = 1;
415        let rows = client
416            .range_scan(
417                OBJECTS_TABLE,
418                Some(range.start_key),
419                Some(range.end_key),
420                rows_limit,
421                reversed,
422                Some(Self::column_qualifier_filter(DEFAULT_COLUMN_QUALIFIER)),
423            )
424            .await?;
425
426        let Some(row) = rows.into_iter().next() else {
427            return Ok(None);
428        };
429
430        let mut value = None;
431        for Cell {
432            name,
433            value: cell_value,
434        } in row.cells
435        {
436            // little optimization, comparing bytes is cheaper than converting it to utf8
437            // string and do string comparison.
438            if name == DEFAULT_COLUMN_QUALIFIER.as_bytes() {
439                value = Some(Bytes::from(cell_value));
440            } else {
441                let cell_name = std::str::from_utf8(&name)?;
442                error!("unexpected column {cell_name:?} in {OBJECTS_TABLE} table");
443            }
444        }
445
446        debug!(
447            found = value.is_some(),
448            elapsed_ms = start.elapsed().as_millis() as u64,
449            "range scanned object before version"
450        );
451        Ok(value)
452    }

454    /// Gets multiple objects returning the latest version strictly less than
455    /// the requested version values as [`Vec`]<[`Option`]<[`Bytes`]>> from the
456    /// kv store.
457    ///
458    /// Based on the provided [`ObjectsBeforeVersionRequest`] fetch the data
459    /// from BigTableDB. Returns a vector of the same length and order as
460    /// the input keys. Each entry is `Some(bytes)` if the key was found, or
461    /// `None` if not found.
462    #[instrument(level = "debug", skip_all)]
463    pub async fn objects_before_version(
464        &self,
465        req: ObjectsBeforeVersionRequest,
466    ) -> Result<Vec<Option<Bytes>>, anyhow::Error> {
467        let req = req.into_inner();
468        debug!(num_keys = req.len(), "fetching objects before version");
469
470        // `multiget_max_items` bounds the size of an incoming request, but it is
471        // an operator-configurable value. Concurrency is capped independently via
472        // `MAX_CONCURRENT_FUTURES` so that increasing the request limit does not
473        // proportionally increase the number of concurrent BigTable calls per request.
474        let max_buffered_concurrent_futures = req.len().clamp(1, MAX_CONCURRENT_FUTURES);
475
476        stream::iter(req)
477            .map(|range| self.object_before_version(range))
478            .buffered(max_buffered_concurrent_futures)
479            .try_collect::<Vec<Option<Bytes>>>()
480            .await
481    }

483    /// Fetches a list `(sequence number, digest)` pairs for transactions
484    /// affecting the given address, ordered by `oldest_first` and capped
485    /// at `limit`.
486    ///
487    /// # Pagination
488    /// `cursor` is **exclusive**.
489    ///
490    /// - oldest_first = `false`: returns entries with `tx_seq < cursor`.
491    /// - oldest_first = `true`: returns entries with `tx_seq > cursor`.
492    ///
493    /// When `None`, the scan starts from the beginning of the requested
494    /// `oldest_first` order.
495    #[instrument(level = "debug", skip(self))]
496    pub async fn transactions_by_address(
497        &self,
498        address: Address,
499        cursor: Option<TransactionSequenceNumber>,
500        limit: usize,
501        oldest_first: bool,
502    ) -> Result<Vec<(TransactionSequenceNumber, TransactionDigest)>, anyhow::Error> {
503        let mut client = self.bigtable_client.clone();
504
505        let order = match oldest_first {
506            true => TransactionsOrder::OldestFirst,
507            false => TransactionsOrder::NewestFirst,
508        };
509
510        let start = Instant::now();
511        let transactions = client
512            .get_transaction_digests_by_address(address, cursor, limit, order)
513            .await?;
514
515        debug!(
516            found = transactions.len(),
517            elapsed_ms = start.elapsed().as_millis() as u64,
518            "fetched transaction digests by address"
519        );
520        Ok(transactions)
521    }

524/// Extracts specific key type from a general [`Key`] type.
525///
526/// Takes:
527/// - `keys`: The list of keys to extract from
528/// - `extractor`: Function that returns Some(extracted_value) for the target
529///   variant, None otherwise
530///
531/// Returns a vector of extracted values. Returns [`ApiError::BadRequest`]
532/// if any extraction returns None value.
533fn extract_keys<T, F>(keys: &[Key], extractor: F) -> Result<Vec<T>, ApiError>
534where
535    F: Fn(&Key) -> Option<T>,
536{
537    keys.iter()
538        .map(|k| {
539            extractor(k).ok_or_else(|| {
540                ApiError::BadRequest("all keys should be of the same type".to_string())
541            })
542        })
543        .collect()
544}

546/// Represents a range of keys for a single object the BigTableDB is allowed to
547/// scan.
548pub(crate) struct ObjectRangeKeyBound {
549    start_key: StartKey,
550    end_key: EndKey,
551}
552
553impl ObjectRangeKeyBound {
554    /// Returns `true` if the range contains no rows.
555    fn is_empty(&self) -> bool {
556        match (&self.start_key, &self.end_key) {
557            (StartKey::StartKeyClosed(start), EndKey::EndKeyClosed(end)) => start > end,
558            (
559                StartKey::StartKeyClosed(start) | StartKey::StartKeyOpen(start),
560                EndKey::EndKeyClosed(end) | EndKey::EndKeyOpen(end),
561            ) => start >= end,
562        }
563    }
564}
565
566impl TryFrom<Key> for ObjectRangeKeyBound {
567    type Error = RangeKeyBoundError;
568
569    fn try_from(key: Key) -> Result<Self, Self::Error> {
570        let Key::ObjectKey(object_key) = key else {
571            return Err(RangeKeyBoundError::UnexpectedItemType {
572                expected: ItemType::Object,
573                detail: "range key must be an ObjectKey".into(),
574            });
575        };
576        Ok(object_key.into())
577    }
578}
579
580impl From<ObjectKey> for ObjectRangeKeyBound {
581    fn from(value: ObjectKey) -> Self {
582        let obj_id = value.0;
583        ObjectRangeKeyBound {
584            start_key: StartKey::StartKeyClosed(raw_object_key(&ObjectKey::min_for_id(&obj_id))),
585            end_key: EndKey::EndKeyOpen(raw_object_key(&value)),
586        }
587    }
588}

590/// Represents a request to fetch objects before a given version.
591///
592/// This struct expects that all keys in the input vector are of type
593/// [`Key::ObjectKey`] in the [`TryFrom`] implementation.
594pub(crate) struct ObjectsBeforeVersionRequest(Vec<ObjectRangeKeyBound>);
595
596impl ObjectsBeforeVersionRequest {
597    fn into_inner(self) -> Vec<ObjectRangeKeyBound> {
598        self.0
599    }
600}
601
602impl TryFrom<Vec<Key>> for ObjectsBeforeVersionRequest {
603    type Error = RangeKeyBoundError;
604
605    fn try_from(keys: Vec<Key>) -> Result<Self, Self::Error> {
606        let object_range_keys = extract_keys(&keys, |k| match k {
607            Key::ObjectKey(object_key) => Some(*object_key),
608            _ => None,
609        })
610        .map_err(|e| RangeKeyBoundError::UnexpectedItemType {
611            expected: ItemType::Object,
612            detail: e.to_string(),
613        })?
614        .into_iter()
615        .map(ObjectRangeKeyBound::from)
616        .collect();
617
618        Ok(ObjectsBeforeVersionRequest(object_range_keys))
619    }
620}
