26/// Configuration for the [`KvStoreClient`] used to access data from S3 and
27/// DynamoDB.
28///
29/// This configuration combines settings for both object storage (S3) and
30/// DynamoDB, matching the storage locations used by the `KVStoreWorker`
31/// in the `iota-data-ingestion` crate.
32///
33/// The client retrieves data from:
34///
35/// - **S3:** Checkpoint contents.
36/// - **DynamoDB:**
37///   - Transactions
38///   - Effects
39///   - Events
40///   - Objects
41///   - Checkpoint summaries
42#[derive(Serialize, Deserialize, Clone, Debug)]
43#[serde(rename_all = "kebab-case")]
44pub struct KvStoreConfig {
45    pub object_store_config: ObjectStoreConfig,
46    pub dynamo_db_config: DynamoDbConfig,
47}

65/// Status of the AWS components used by the [`KvStoreClient`].
66#[derive(Debug, Serialize, Clone)]
67pub struct AwsStatus {
68    pub dynamodb: ServiceStatus,
69    pub s3: ServiceStatus,
70}

72/// Represents the health status of a service.
73///
74/// It captures the current health status (healthy or unhealthy) and
75/// the latency of the service. It's typically used to monitor the status of
76/// external components.
77#[derive(Debug, Serialize, Clone)]
78pub struct ServiceStatus {
79    /// Indicates whether the service is healthy (`true`) or unhealthy
80    /// (`false`).
81    healthy: bool,
82    /// The latency of the service, measured in milliseconds. This represents
83    /// the time it took to check the service's health.
84    latency_ms: u64,
85}
86
87/// Represents a cached status of AWS components.
88///
89/// This struct stores the status of AWS components along with the time it was
90/// cached. It's used to avoid overwhelming the AWS services with frequent
91/// health check requests. The cached status is considered valid for a limited
92/// time (TTL).
93#[derive(Debug)]
94pub struct CachedAwsStatus {
95    /// The status of the AWS components.
96    status: AwsStatus,
97    /// The time at which the status was cached. This is used to determine
98    /// if the cached status is still valid (within the TTL).
99    cached_at: Instant,
100}

102/// Provides read access to data ingested by the `iota-data-ingestion`
103/// crate's `KVStoreWorker`.
104///
105/// It retrieves data from two storage backends:
106///
107/// - **S3:** Used for checkpoint contents.
108/// - **DynamoDB:** Used for:
109///   - Transactions
110///   - Effects
111///   - Events
112///   - Objects
113///   - Checkpoint summaries
114///
115/// The client implements a read-only interface and supports the HTTP fallback
116/// mechanism used by
117/// [`HttpKVStore`](iota_storage::http_key_value_store::HttpKVStore).
118#[derive(Debug, Clone)]
119pub struct KvStoreClient {
120    /// DynamoDb client.
121    dynamo_db_client: Client,
122    /// S3 compatible bucket client.
123    remote_store: Arc<DynObjectStore>,
124    /// DynamoDb table name.
125    table_name: String,
126    /// The representation of the uptime of the service.
127    start_time: Instant,
128    /// Cached AWS components status.
129    cached_status: Arc<RwLock<Option<CachedAwsStatus>>>,
130    /// The TTL of the [`CachedAwsStatus`].
131    cache_duration: Duration,
132}
133
134impl KvStoreClient {
135    /// Create a new instance of the client.
136    ///
137    /// Internally it instantiates a DynamoDb Client and an S3 compatible bucket
138    /// Client.
139    pub async fn new(config: KvStoreConfig) -> Result<Self> {
140        let dynamodb_config = config.dynamo_db_config;
141
142        let credentials = Credentials::new(
143            &dynamodb_config.aws_access_key_id,
144            &dynamodb_config.aws_secret_access_key,
145            None,
146            None,
147            "dynamodb",
148        );
149        let timeout_config = TimeoutConfig::builder()
150            .operation_timeout(OPERATION_TIMEOUT_SECS)
151            .operation_attempt_timeout(OPERATION_ATTEMPT_TIMEOUT_SECS)
152            .connect_timeout(CONNECT_TIMEOUT_SECS)
153            .build();
154        let http_client = aws_smithy_http_client::Builder::new()
155            .tls_provider(aws_smithy_http_client::tls::Provider::Rustls(
156                aws_smithy_http_client::tls::rustls_provider::CryptoMode::Ring,
157            ))
158            .build_https();
159        let mut aws_config_loader = aws_config::defaults(BehaviorVersion::latest())
160            .http_client(http_client)
161            .credentials_provider(credentials)
162            .region(Region::new(dynamodb_config.aws_region))
163            .timeout_config(timeout_config);
164
165        if let Some(url) = dynamodb_config.aws_endpoint {
166            aws_config_loader = aws_config_loader.endpoint_url(url);
167        }
168        let aws_config = aws_config_loader.load().await;
169        let dynamo_db_client = Client::new(&aws_config);
170        let remote_store = config.object_store_config.make()?;
171
172        Ok(Self {
173            dynamo_db_client,
174            remote_store,
175            table_name: dynamodb_config.table_name,
176            start_time: Instant::now(),
177            cache_duration: AWS_STATUS_CACHE_TTL,
178            cached_status: Arc::new(RwLock::new(None)),
179        })
180    }
181
182    /// Get the elapsed time from which the service was instantiated.
183    pub fn get_uptime(&self) -> Duration {
184        self.start_time.elapsed()
185    }
186
187    async fn check_dynamodb_health(&self) -> ServiceStatus {
188        let start = Instant::now();
189
190        let healthy = self
191            .dynamo_db_client
192            .describe_table()
193            .table_name(&self.table_name)
194            .send()
195            .await
196            .inspect_err(|err| tracing::error!("failed describing dynamodb table: {err}"))
197            .is_ok();
198
199        ServiceStatus {
200            healthy,
201            latency_ms: start.elapsed().as_millis() as u64,
202        }
203    }
204
205    async fn check_s3_health(&self) -> ServiceStatus {
206        let start = Instant::now();
207
208        // Just check if we can access the bucket by trying to get a non-existent key
209        let test_path = Path::from("health-check-test");
210
211        let healthy = match self.remote_store.head(&test_path).await {
212            Ok(_) => true,
213            Err(object_store::Error::NotFound { .. }) => true, // Not found is OK
214            Err(err) => {
215                tracing::error!("failed checking file metadata on S3: {err}");
216                false
217            }
218        };
219
220        ServiceStatus {
221            healthy,
222            latency_ms: start.elapsed().as_millis() as u64,
223        }
224    }
225
226    async fn check_aws_health(&self) -> AwsStatus {
227        AwsStatus {
228            dynamodb: self.check_dynamodb_health().await,
229            s3: self.check_s3_health().await,
230        }
231    }
232
233    /// Get AWS service status.
234    pub async fn get_aws_health(&self) -> AwsStatus {
235        // Read lock for checking cache status
236        let should_refresh = {
237            let cached = self.cached_status.read().await;
238            cached.is_none()
239                || cached
240                    .as_ref()
241                    .map(|a| a.cached_at.elapsed() > self.cache_duration)
242                    .unwrap_or(true)
243        };
244
245        if should_refresh {
246            let new_status = self.check_aws_health().await;
247            // Write lock only when updating
248            let mut cached = self.cached_status.write().await;
249            *cached = Some(CachedAwsStatus {
250                status: new_status.clone(),
251                cached_at: Instant::now(),
252            });
253
254            return new_status;
255        }
256
257        // Read lock for getting cached value
258        if let Some(cached) = self.cached_status.read().await.as_ref() {
259            cached.status.clone()
260        } else {
261            // Cache was cleared between our check and here, get fresh status
262            self.check_aws_health().await
263        }
264    }

266    /// Get value as [`Bytes`] from DynamoDb.
267    async fn get_from_dynamodb<T: AsRef<[u8]>>(
268        &self,
269        digest: T,
270        item_type: String,
271    ) -> Result<Option<Bytes>> {
272        let result = self
273            .dynamo_db_client
274            .get_item()
275            .table_name(&self.table_name)
276            .key("digest", AttributeValue::B(Blob::new(digest.as_ref())))
277            .key("type", AttributeValue::S(item_type))
278            .send()
279            .await?;
280
281        if let Some(item) = result.item {
282            if let Some(AttributeValue::B(blob)) = item.get("bcs") {
283                return Ok(Some(Bytes::copy_from_slice(blob.as_ref())));
284            }
285        }
286
287        Ok(None)
288    }
289
290    /// Get value as [`Bytes`] from the S3 compatible bucket.
291    async fn get_from_remote_store<T: AsRef<[u8]>>(&self, digest: &T) -> Result<Option<Bytes>> {
292        let path = Path::from(base64_url::encode(digest));
293
294        // Get the object
295        match self.remote_store.get(&path).await {
296            Ok(response) => {
297                // Get bytes from the response
298                let data = response.bytes().await.map_err(|err| {
299                    anyhow::anyhow!("failed to read data from remote store: {err}")
300                })?;
301
302                Ok(Some(data))
303            }
304            Err(err) => {
305                match err {
306                    // Handle specific object_store errors
307                    object_store::Error::NotFound { .. } => Ok(None),
308                    _ => Err(anyhow::anyhow!("remote store error: {err}")),
309                }
310            }
311        }
312    }

314    /// Get value as [`Bytes`] from the kv store.
315    ///
316    /// Based on the provided [`Key`] fetch the data from DynamoDb or S3
317    /// compatible buckets.
318    pub async fn get(&self, key: Key) -> Result<Option<Bytes>> {
319        let item_type = key.item_type().to_string();
320
321        match key {
322            Key::TransactionDigestsByAddress(_address) => {
323                bail!("unsupported key");
324            }
325            Key::Transaction(transaction_digest) => {
326                self.get_from_dynamodb(transaction_digest, item_type).await
327            }
328            Key::TransactionEffects(transaction_digest) => {
329                self.get_from_dynamodb(transaction_digest, item_type).await
330            }
331            Key::CheckpointContents(chk_seq_num) => {
332                let serialized_checkpoint_number =
333                    bcs::to_bytes(&TaggedKey::CheckpointSequenceNumber(chk_seq_num))?;
334                let data = self
335                    .get_from_dynamodb(&serialized_checkpoint_number, item_type)
336                    .await?;
337                if data.is_none() {
338                    tracing::info!(
339                        "checkpoint contents with sequence number {chk_seq_num} not found in DynamoDB, attempting fetch from remote store",
340                    );
341                    return self
342                        .get_from_remote_store(&serialized_checkpoint_number)
343                        .await;
344                }
345                Ok(data)
346            }
347            Key::CheckpointSummary(chk_seq_num) => {
348                let serialized_checkpoint_number =
349                    bcs::to_bytes(&TaggedKey::CheckpointSequenceNumber(chk_seq_num))?;
350
351                self.get_from_dynamodb(serialized_checkpoint_number, item_type)
352                    .await
353            }
354            Key::CheckpointSummaryByDigest(checkpoint_digest) => {
355                self.get_from_dynamodb(checkpoint_digest, item_type).await
356            }
357            Key::TransactionToCheckpoint(transaction_digest) => {
358                self.get_from_dynamodb(transaction_digest, item_type).await
359            }
360            Key::ObjectKey(object_key) => {
361                let serialized_object_key = bcs::to_bytes(&object_key)?;
362                self.get_from_dynamodb(serialized_object_key, item_type)
363                    .await
364            }
365            Key::EventsByTransactionDigest(transaction_digest) => {
366                self.get_from_dynamodb(transaction_digest, item_type).await
367            }
368        }
369    }
