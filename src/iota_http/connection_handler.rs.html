19/// How long a connection asked to close is given to shut down.
20pub(crate) const SHUTDOWN_GRACE_PERIOD: Duration = Duration::from_secs(1);
21
22// This is moved to its own function as a way to get around
23// https://github.com/rust-lang/rust/issues/102211
24pub async fn serve_connection<IO, S, B, C>(
25    hyper_io: IO,
26    hyper_svc: S,
27    builder: hyper_util::server::conn::auto::Builder<hyper_util::rt::TokioExecutor>,
28    graceful_shutdown_token: tokio_util::sync::CancellationToken,
29    max_connection_age: Option<Duration>,
30    idle_timer: Option<IdleTimer>,
31    metrics: Option<ListenerMetrics>,
32    on_connection_close: C,
33) where
34    B: http_body::Body + Send + 'static,
35    B::Data: Send,
36    B::Error: Into<BoxError>,
37    IO: hyper::rt::Read + hyper::rt::Write + Send + Unpin + 'static,
38    S: hyper::service::Service<Request<hyper::body::Incoming>, Response = Response<B>> + 'static,
39    S::Future: Send + 'static,
40    S::Error: Into<Box<dyn std::error::Error + Send + Sync>>,
41{
42    // `serve_connection_with_upgrades` serves HTTP/1 even when the builder
43    // allows only HTTP/2.
44    if builder.is_http1_available() {
45        serve_until_closed(
46            builder.serve_connection_with_upgrades(hyper_io, hyper_svc),
47            graceful_shutdown_token,
48            max_connection_age,
49            idle_timer,
50            metrics,
51        )
52        .await;
53    } else {
54        serve_until_closed(
55            builder.serve_connection(hyper_io, hyper_svc),
56            graceful_shutdown_token,
57            max_connection_age,
58            idle_timer,
59            metrics,
60        )
61        .await;
62    }
63
64    trace!("connection closed");
65    drop(on_connection_close);
66}

68/// Serves `conn` until it closes. Starts a graceful shutdown of `conn` on the
69/// shutdown signal, at `max_connection_age`, and when `idle_timer` is reached.
70/// Drops `conn` if it has not closed [`SHUTDOWN_GRACE_PERIOD`] later.
71async fn serve_until_closed(
72    conn: impl GracefulConnection<Error = BoxError>,
73    graceful_shutdown_token: tokio_util::sync::CancellationToken,
74    max_connection_age: Option<Duration>,
75    idle_timer: Option<IdleTimer>,
76    metrics: Option<ListenerMetrics>,
77) {
78    let mut sig = pin!(Fuse::new(graceful_shutdown_token.cancelled_owned()));
79
80    let mut conn = pin!(conn);
81
82    let sleep = sleep_or_pending(max_connection_age);
83    let idle = idle_or_pending(idle_timer);
84    tokio::pin!(sleep);
85    tokio::pin!(idle);
86
87    // The peer may ignore a graceful shutdown, so it is bounded by
88    // `force_close`. Once requested, the branches that request it are disabled.
89    let mut shutdown_requested = false;
90    let force_close = sleep_or_pending(None);
91    tokio::pin!(force_close);
92
93    macro_rules! request_shutdown {
94        ($reason:expr) => {{
95            shutdown_requested = true;
96            if let Some(metrics) = &metrics {
97                metrics.record_close($reason);
98            }
99            conn.as_mut().graceful_shutdown();
100            force_close.set(sleep_or_pending(Some(SHUTDOWN_GRACE_PERIOD)));
101        }};
102    }
103
104    loop {
105        tokio::select! {
106            _ = &mut sig, if !shutdown_requested => {
107                request_shutdown!(CloseReason::Shutdown);
108            }
109            rv = &mut conn => {
110                if let Err(err) = rv {
111                    debug!("failed serving connection: {:#}", err);
112                }
113                break;
114            },
115            _ = &mut idle, if !shutdown_requested => {
116                debug!("closing a connection that has been idle past its deadline");
117                request_shutdown!(CloseReason::Idle);
118            }
119            _ = &mut sleep, if !shutdown_requested => {
120                debug!("closing a connection that has reached its maximum age");
121                request_shutdown!(CloseReason::MaxAge);
122            }
123            _ = &mut force_close => {
124                debug!("dropping a connection that did not close when asked");
125                break;
126            }
127        }
128    }
129}

131pub(crate) async fn sleep_or_pending(wait_for: Option<Duration>) {
132    match wait_for {
133        Some(wait) => tokio::time::sleep(wait).await,
134        None => std::future::pending().await,
135    };
136}
137
138/// Waits for `idle_timer`, or forever when there is none.
139async fn idle_or_pending(idle_timer: Option<IdleTimer>) {
140    match idle_timer {
141        Some(idle_timer) => idle_timer.wait().await,
142        None => std::future::pending().await,
143    };
144}

146pub(crate) struct OnConnectionClose<A> {
147    id: ConnectionId,
148    active_connections: ActiveConnections<A>,
149    _count_guard: ConnectionCountGuard,
150}
151
152impl<A> OnConnectionClose<A> {
153    pub(crate) fn new(
154        id: ConnectionId,
155        active_connections: ActiveConnections<A>,
156        count_guard: ConnectionCountGuard,
157    ) -> Self {
158        Self {
159            id,
160            active_connections,
161            _count_guard: count_guard,
162        }
163    }
164}
165
166impl<A> Drop for OnConnectionClose<A> {
167    fn drop(&mut self) {
168        self.active_connections.write().unwrap().remove(&self.id);
169    }
