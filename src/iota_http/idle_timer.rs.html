4//! A timer that measures how long a task has had nothing to do.
5
6use std::{sync::Arc, time::Duration};
7
8use parking_lot::Mutex;
9use tokio::time::Instant;
10
11/// Creates a timer whose [`IdleTimer::wait`] completes once its task has been
12/// idle for `timeout`.
13///
14/// The task is idle whenever it holds no [`IdleGuard`]. It starts idle, so for
15/// one that never takes a guard `wait` completes after a single timeout. Taking
16/// a guard stops the timer, and dropping the last one starts it again from that
17/// moment — so this measures the time since the task last had something to do,
18/// not the time since it was created.
19///
20/// The work is generally not done by the task holding the timer, so guards are
21/// taken through an [`IdleHandle`], which can be cloned and sent wherever the
22/// work happens.
23///
24/// [`IdleTimer::wait`] completes at once while the task has already been idle
25/// for `timeout`, so a loop must stop waiting once it has acted on it.
26///
27/// ```ignore
28/// let timer = idle_timeout(Duration::from_secs(300));
29/// let handle = timer.handle();
30///
31/// // Wherever the work is: the task counts as busy until the guard is dropped.
32/// let guard = handle.guard();
33///
34/// let mut wait = std::pin::pin!(timer.wait());
35/// loop {
36///     tokio::select! {
37///         _ = &mut wait => {
38///             // … wind the task down, and stop polling `wait`
39///         }
40///         // … whatever else the task is waiting for
41///     }
42/// }
43/// ```
44pub fn idle_timeout(timeout: Duration) -> IdleTimer {
45    IdleTimer {
46        timeout,
47        state: Arc::new(Mutex::new(State {
48            counter: 0,
49            idle_since: Instant::now(),
50        })),
51    }
52}
53
54/// A timer that runs only while its task is idle. See [`idle_timeout`].
55pub struct IdleTimer {
56    // Idle timeout: period after which `wait` completes.
57    timeout: Duration,
58    // Whether the task is busy and when it last became idle, shared with
59    // the handles and guards that mark it busy.
60    state: Arc<Mutex<State>>,
61}
62
63impl IdleTimer {
64    /// A handle for marking the task busy, which can be cloned and sent to
65    /// wherever its work is done.
66    pub fn handle(&self) -> IdleHandle {
67        IdleHandle {
68            state: self.state.clone(),
69        }
70    }
71
72    /// Completes once the task has been idle for `timeout`, at once if it
73    /// already has.
74    pub async fn wait(&self) {
75        // Guards do not wake the timer. Taking or dropping one can only move
76        // the moment the task will have been idle for `timeout` later, so this
77        // sleeps until the earliest such moment and works it out again when it
78        // wakes.
79        loop {
80            let now = Instant::now();
81            let deadline = {
82                let state = self.state.lock();
83                if state.counter == 0 {
84                    state.idle_since + self.timeout
85                } else {
86                    now + self.timeout
87                }
88            };
89            if deadline <= now {
90                return;
91            }
92            tokio::time::sleep_until(deadline).await;
93        }
94    }
95}

97/// Marks a task busy for as long as the guards it hands out are held. See
98/// [`idle_timeout`].
99#[derive(Clone, Debug)]
100pub struct IdleHandle {
101    state: Arc<Mutex<State>>,
102}
103
104impl IdleHandle {
105    /// Counts the task as busy until the returned guard is dropped.
106    pub fn guard(&self) -> IdleGuard {
107        self.state.lock().counter += 1;
108        IdleGuard {
109            state: self.state.clone(),
110        }
111    }
112}
113
114/// Counts its task as busy for as long as it is held. See [`idle_timeout`].
115#[must_use = "the task is only counted busy for as long as the guard is held"]
116#[derive(Debug)]
117pub struct IdleGuard {
118    state: Arc<Mutex<State>>,
119}
120
121impl Drop for IdleGuard {
122    fn drop(&mut self) {
123        let mut state = self.state.lock();
124        state.counter -= 1;
125        if state.counter == 0 {
126            // Read under the lock, so a later idle period never stores an
127            // earlier moment than the one before it.
128            state.idle_since = Instant::now();
129        }
130    }
131}
132
133#[derive(Debug)]
134struct State {
135    /// Live [`IdleGuard`]s; the task is idle when this is zero.
136    counter: usize,
137    /// When `counter` last fell to zero, or when the timer was created if it
138    /// never has. Read only while `counter` is zero.
139    idle_since: Instant,
140}
