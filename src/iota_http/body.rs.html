5use std::{
6    pin::Pin,
7    task::{Context, Poll},
8};
9
10use bytes::Bytes;
11use http_body_util::BodyExt;
12use pin_project_lite::pin_project;
13
14use crate::{BoxError, idle_timer::IdleGuard};
15
16pub type BoxBody = http_body_util::combinators::UnsyncBoxBody<Bytes, BoxError>;
17
18pin_project! {
19    /// A response body that keeps its connection counted as busy until the
20    /// body ends.
21    ///
22    /// The guard rides the body rather than the response future because a
23    /// streaming response resolves its future as soon as the headers are
24    /// ready, while the work continues for as long as frames are produced.
25    pub(crate) struct GuardedBody<B> {
26        #[pin]
27        inner: B,
28        guard: IdleGuard,
29    }
30}
31
32impl<B> GuardedBody<B> {
33    pub(crate) fn new(inner: B, guard: IdleGuard) -> Self {
34        Self { inner, guard }
35    }
36}
37
38impl<B: http_body::Body> http_body::Body for GuardedBody<B> {
39    type Data = B::Data;
40    type Error = B::Error;
41
42    fn poll_frame(
43        self: Pin<&mut Self>,
44        cx: &mut Context<'_>,
45    ) -> Poll<Option<Result<http_body::Frame<Self::Data>, Self::Error>>> {
46        self.project().inner.poll_frame(cx)
47    }
48
49    fn is_end_stream(&self) -> bool {
50        self.inner.is_end_stream()
51    }
52
53    fn size_hint(&self) -> http_body::SizeHint {
54        self.inner.size_hint()
55    }
56}
57
58pub fn boxed<B>(body: B) -> BoxBody
59where
60    B: http_body::Body<Data = Bytes> + Send + 'static,
61    B::Error: Into<BoxError>,
62{
63    try_downcast(body).unwrap_or_else(|body| body.map_err(Into::into).boxed_unsync())
64}
65
66pub(crate) fn try_downcast<T, K>(k: K) -> Result<T, K>
67where
68    T: 'static,
69    K: Send + 'static,
70{
71    let mut k = Some(k);
72    if let Some(k) = <dyn std::any::Any>::downcast_mut::<Option<T>>(&mut k) {
73        Ok(k.take().unwrap())
74    } else {
75        Err(k.unwrap())
76    }
77}
