24pub async fn get_balance_changes_from_effect<P: ObjectProvider<Error = E>, E>(
25    object_provider: &P,
26    effects: &TransactionEffects,
27    input_objs: Vec<InputObjectKind>,
28    mocked_coin: Option<ObjectId>,
29) -> Result<Vec<BalanceChange>, E> {
30    let gas_owner = effects.gas_object().owner;
31
32    // Only charge gas when tx fails, skip all object parsing
33    if effects.status() != &ExecutionStatus::Success {
34        return Ok(vec![BalanceChange {
35            owner: gas_owner,
36            coin_type: TypeTag::from(StructTag::new_gas()),
37            amount: effects.gas_cost_summary().net_gas_usage().neg() as i128,
38        }]);
39    }
40
41    let all_mutated = effects
42        .all_changed_objects()
43        .into_iter()
44        .filter_map(|(changed, _)| {
45            let object_ref = changed.reference;
46            if matches!(mocked_coin, Some(coin) if object_ref.object_id == coin) {
47                return None;
48            }
49            Some((
50                object_ref.object_id,
51                object_ref.version,
52                Some(object_ref.digest),
53            ))
54        })
55        .collect::<Vec<_>>();
56
57    let input_objs_to_digest = input_objs
58        .iter()
59        .filter_map(|k| match k {
60            InputObjectKind::ImmOrOwnedMoveObject(o) => Some((o.object_id, o.digest)),
61            InputObjectKind::MovePackage(_) | InputObjectKind::SharedMoveObject { .. } => None,
62        })
63        .collect::<HashMap<ObjectId, ObjectDigest>>();
64    let unwrapped_then_deleted = effects
65        .unwrapped_then_deleted()
66        .iter()
67        .map(|e| e.object_id)
68        .collect::<HashSet<_>>();
69    get_balance_changes(
70        object_provider,
71        &effects
72            .modified_at_versions()
73            .into_iter()
74            .filter_map(|modified| {
75                let (id, version) = (modified.object_id, modified.version);
76                if matches!(mocked_coin, Some(coin) if id == coin) {
77                    return None;
78                }
79                // We won't be able to get dynamic object from object provider today
80                if unwrapped_then_deleted.contains(&id) {
81                    return None;
82                }
83                Some((id, version, input_objs_to_digest.get(&id).cloned()))
84            })
85            .collect::<Vec<_>>(),
86        &all_mutated,
87    )
88    .await
89}

91pub async fn get_balance_changes<P: ObjectProvider<Error = E>, E>(
92    object_provider: &P,
93    modified_at_version: &[(ObjectId, Version, Option<ObjectDigest>)],
94    all_mutated: &[(ObjectId, Version, Option<ObjectDigest>)],
95) -> Result<Vec<BalanceChange>, E> {
96    // 1. subtract all input coins
97    let balances = fetch_coins(object_provider, modified_at_version)
98        .await?
99        .into_iter()
100        .fold(
101            BTreeMap::<_, i128>::new(),
102            |mut acc, (owner, type_, amount)| {
103                *acc.entry((owner, type_)).or_default() -= amount as i128;
104                acc
105            },
106        );
107    // 2. add all mutated coins
108    let balances = fetch_coins(object_provider, all_mutated)
109        .await?
110        .into_iter()
111        .fold(balances, |mut acc, (owner, type_, amount)| {
112            *acc.entry((owner, type_)).or_default() += amount as i128;
113            acc
114        });
115
116    Ok(balances
117        .into_iter()
118        .filter_map(|((owner, coin_type), amount)| {
119            if amount == 0 {
120                return None;
121            }
122            Some(BalanceChange {
123                owner,
124                coin_type,
125                amount,
126            })
127        })
128        .collect())
129}

131async fn fetch_coins<P: ObjectProvider<Error = E>, E>(
132    object_provider: &P,
133    objects: &[(ObjectId, Version, Option<ObjectDigest>)],
134) -> Result<Vec<(Owner, TypeTag, u64)>, E> {
135    let mut all_mutated_coins = vec![];
136    for (id, version, digest_opt) in objects {
137        // TODO: use multi get object
138        let o = object_provider.get_object(id, version).await?;
139        if let Some(object_type) = o.data.opt_object_type() {
140            if object_type.is_coin() {
141                if let Some(digest) = digest_opt {
142                    // TODO: can we return Err here instead?
143                    assert_eq!(
144                        *digest,
145                        o.digest(),
146                        "Object digest mismatch--got bad data from object_provider?"
147                    )
148                }
149                let coin_type = object_type.type_params()[0].clone();
150                all_mutated_coins.push((
151                    o.owner,
152                    coin_type,
153                    // we know this is a coin, safe to unwrap
154                    Coin::extract_balance_if_coin(&o).unwrap().unwrap(),
155                ))
156            }
157        }
158    }
159    Ok(all_mutated_coins)
160}

162#[async_trait]
163pub trait ObjectProvider {
164    type Error;
165    async fn get_object(&self, id: &ObjectId, version: &Version) -> Result<Object, Self::Error>;
166    async fn find_object_lt_or_eq_version(
167        &self,
168        id: &ObjectId,
169        version: &Version,
170    ) -> Result<Option<Object>, Self::Error>;
171}

173pub struct ObjectProviderCache<P> {
174    object_cache: RwLock<BTreeMap<(ObjectId, Version), Object>>,
175    /// Memoizes [`ObjectProvider::find_object_lt_or_eq_version`]: maps an
176    /// object and the version asked for to the version that answered the
177    /// query.
178    resolved_version_cache: RwLock<BTreeMap<(ObjectId, Version), Version>>,
179    provider: P,
180}
181
182impl<P> ObjectProviderCache<P> {
183    pub fn new(provider: P) -> Self {
184        Self {
185            object_cache: Default::default(),
186            resolved_version_cache: Default::default(),
187            provider,
188        }
189    }
190
191    #[instrument(level = "trace", skip_all)]
192    pub fn insert_objects_into_cache(&mut self, objects: Vec<Object>) {
193        let object_cache = self.object_cache.get_mut();
194        let resolved_version_cache = self.resolved_version_cache.get_mut();
195
196        for object in objects {
197            let key = (object.id(), object.version());
198            resolved_version_cache.insert(key, object.version());
199            object_cache.insert(key, object);
200        }
201    }
202
203    pub fn new_with_cache(provider: P, written_objects: &BTreeMap<ObjectId, Object>) -> Self {
204        let mut object_cache = BTreeMap::new();
205        let mut resolved_version_cache = BTreeMap::new();
206
207        for (object_id, object) in written_objects {
208            let key = (*object_id, object.version());
209            resolved_version_cache.insert(key, object.version());
210            object_cache.insert(key, object.clone());
211        }
212
213        Self {
214            object_cache: RwLock::new(object_cache),
215            resolved_version_cache: RwLock::new(resolved_version_cache),
216            provider,
217        }
218    }
219}
220
221#[async_trait]
222impl<P, E> ObjectProvider for ObjectProviderCache<P>
223where
224    P: ObjectProvider<Error = E> + Sync + Send,
225    E: Sync + Send,
226{
227    type Error = P::Error;
228
229    async fn get_object(&self, id: &ObjectId, version: &Version) -> Result<Object, Self::Error> {
230        if let Some(o) = self.object_cache.read().await.get(&(*id, *version)) {
231            return Ok(o.clone());
232        }
233        let o = self.provider.get_object(id, version).await?;
234        self.object_cache
235            .write()
236            .await
237            .insert((*id, *version), o.clone());
238        Ok(o)
239    }
240
241    async fn find_object_lt_or_eq_version(
242        &self,
243        id: &ObjectId,
244        version: &Version,
245    ) -> Result<Option<Object>, Self::Error> {
246        if let Some(resolved) = self
247            .resolved_version_cache
248            .read()
249            .await
250            .get(&(*id, *version))
251        {
252            return Ok(self.get_object(id, resolved).await.ok());
253        }
254        if let Some(o) = self
255            .provider
256            .find_object_lt_or_eq_version(id, version)
257            .await?
258        {
259            self.object_cache
260                .write()
261                .await
262                .insert((*id, o.version()), o.clone());
263            self.resolved_version_cache
264                .write()
265                .await
266                .insert((*id, *version), o.version());
267            Ok(Some(o))
268        } else {
269            Ok(None)
270        }
271    }
