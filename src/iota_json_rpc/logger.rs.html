19/// Defines a logger with callbacks during the RPC request life-cycle. The primary use case for this
20/// is to collect timings for a larger metrics collection solution.
21pub trait Logger: Send + Sync + Clone + 'static {
22    /// Intended to carry timestamp of a request, for example
23    /// `std::time::Instant`. How the trait measures time, if at all, is
24    /// entirely up to the implementation.
25    type Instant: std::fmt::Debug + Send + Sync + Copy;
26
27    /// Called when a new JSON-RPC request comes to the server.
28    fn on_request(&self) -> Self::Instant;
29
30    /// Called on each JSON-RPC method call, batch requests will trigger
31    /// `on_call` multiple times.
32    fn on_call(&self, method_name: &str, params: Params, kind: MethodKind);
33
34    /// Called on each JSON-RPC method completion, batch requests will trigger
35    /// `on_result` multiple times.
36    fn on_result(
37        &self,
38        method_name: &str,
39        success: bool,
40        error_code: Option<i32>,
41        started_at: Self::Instant,
42    );
43}
44
45impl Logger for () {
46    type Instant = ();
47
48    fn on_request(&self) -> Self::Instant {}
49
50    fn on_call(&self, _: &str, _: Params, _: MethodKind) {}
51
52    fn on_result(&self, _: &str, _: bool, _: Option<i32>, _: Self::Instant) {}
53}
54
55impl<A, B> Logger for (A, B)
56where
57    A: Logger,
58    B: Logger,
59{
60    type Instant = (A::Instant, B::Instant);
61
62    fn on_request(&self) -> Self::Instant {
63        (self.0.on_request(), self.1.on_request())
64    }
65
66    fn on_call(&self, method_name: &str, params: Params, kind: MethodKind) {
67        self.0.on_call(method_name, params.clone(), kind);
68        self.1.on_call(method_name, params, kind);
69    }
70
71    fn on_result(
72        &self,
73        method_name: &str,
74        success: bool,
75        error_code: Option<i32>,
76        started_at: Self::Instant,
77    ) {
78        self.0
79            .on_result(method_name, success, error_code, started_at.0);
80        self.1
81            .on_result(method_name, success, error_code, started_at.1);
82    }
83}

85pub(crate) trait FutureWithTracing<O>: Future<Output = RpcInterimResult<O>> {
86    fn trace(self) -> impl Future<Output = RpcResult<O>>
87    where
88        Self: Sized,
89    {
90        self.trace_timeout(Duration::from_secs(1))
91    }
92
93    fn trace_timeout(self, timeout: Duration) -> impl Future<Output = RpcResult<O>>
94    where
95        Self: Sized,
96    {
97        async move {
98            let start = std::time::Instant::now();
99            let interim_result: RpcInterimResult<_> = self.await;
100            let elapsed = start.elapsed();
101            let result = interim_result.map_err(|e| {
102                let anyhow_error = anyhow!("{e:?}");
103
104                let rpc_error: RpcError = e.into();
105                if !matches!(rpc_error, RpcError::Call(_)) {
106                    error!(error=?anyhow_error);
107                }
108                error_object_from_rpc(rpc_error)
109            });
110
111            if elapsed > timeout {
112                info!(?elapsed, "RPC took longer than threshold to complete.");
113            }
114            result
115        }
116        .instrument(Span::current())
117    }
118}
119impl<F: Future<Output = RpcInterimResult<O>>, O> FutureWithTracing<O> for F {}
