5use std::collections::BTreeMap;
6
7use iota_json_rpc_types::ObjectChange;
8use iota_sdk_types::{
9    Address, ObjectReference, ObjectRemoveKind, ObjectVersion, OwnedObjectReference, StructTag,
10    WriteKind,
11};
12
13use crate::ObjectProvider;

15pub async fn get_object_changes<P: ObjectProvider<Error = E>, E>(
16    object_provider: &P,
17    sender: Address,
18    modified_at_versions: Vec<ObjectVersion>,
19    all_changed_objects: Vec<(OwnedObjectReference, WriteKind)>,
20    all_removed_objects: Vec<(ObjectReference, ObjectRemoveKind)>,
21) -> Result<Vec<ObjectChange>, E> {
22    let mut object_changes = vec![];
23
24    let modify_at_version = modified_at_versions
25        .into_iter()
26        .map(|modified| (*modified.object_id(), modified.version()))
27        .collect::<BTreeMap<_, _>>();
28
29    for (changed, kind) in all_changed_objects {
30        let (reference, owner) = (*changed.reference(), *changed.owner());
31        let ObjectReference {
32            object_id,
33            version,
34            digest,
35        } = reference;
36        let o = object_provider.get_object(&object_id, &version).await?;
37        if let Some(object_type) = o.data.opt_object_type() {
38            let object_type: StructTag = object_type.clone().into();
39
40            match kind {
41                WriteKind::Mutate => object_changes.push(ObjectChange::Mutated {
42                    sender,
43                    owner,
44                    object_type,
45                    object_id,
46                    version,
47                    // modify_at_version should always be available for mutated object
48                    previous_version: modify_at_version
49                        .get(&object_id)
50                        .cloned()
51                        .unwrap_or_default(),
52                    digest,
53                }),
54                WriteKind::Create => object_changes.push(ObjectChange::Created {
55                    sender,
56                    owner,
57                    object_type,
58                    object_id,
59                    version,
60                    digest,
61                }),
62                WriteKind::Unwrap => object_changes.push(ObjectChange::Unwrapped {
63                    sender,
64                    owner,
65                    object_type,
66                    object_id,
67                    version,
68                    digest,
69                }),
70                _ => {
71                    unimplemented!("a new WriteKind enum variant was added and needs to be handled")
72                }
73            }
74        } else if let Some(p) = o.data.as_opt_package() {
75            if kind == WriteKind::Create {
76                object_changes.push(ObjectChange::Published {
77                    package_id: p.id(),
78                    version: p.version(),
79                    digest,
80                    modules: p
81                        .serialized_module_map()
82                        .keys()
83                        .map(|k| k.to_string())
84                        .collect(),
85                })
86            }
87        };
88    }
89
90    for (removed_object, kind) in all_removed_objects {
91        let id = removed_object.object_id;
92        let version = removed_object.version;
93        // The version the transaction consumed is the last one below the
94        // removal. Reading it by its exact version lets a provider answer from
95        // the input objects it was given, instead of from a store that may have
96        // pruned it since.
97        let input_object = match modify_at_version.get(&id) {
98            Some(input_version) => object_provider.get_object(&id, input_version).await.ok(),
99            None => None,
100        };
101        let o = match input_object {
102            Some(o) => Some(o),
103            None => {
104                object_provider
105                    .find_object_lt_or_eq_version(&id, &version)
106                    .await?
107            }
108        };
109        if let Some(o) = o {
110            if let Some(object_type) = o.data.opt_object_type() {
111                let object_type: StructTag = object_type.clone().into();
112                match kind {
113                    ObjectRemoveKind::Delete => object_changes.push(ObjectChange::Deleted {
114                        sender,
115                        object_type,
116                        object_id: id,
117                        version,
118                    }),
119                    ObjectRemoveKind::Wrap => object_changes.push(ObjectChange::Wrapped {
120                        sender,
121                        object_type,
122                        object_id: id,
123                        version,
124                    }),
125                    _ => unimplemented!(
126                        "a new ObjectRemoveKind enum variant was added and needs to be handled"
127                    ),
128                }
129            }
130        };
131    }
132
133    Ok(object_changes)
134}

136#[cfg(test)]
137mod tests {
138    use async_trait::async_trait;
139    use iota_sdk_types::{ObjectDigest, ObjectId, Owner, Version};
140    use iota_types::object::Object;
141
142    use super::*;
143
144    /// Serves only the exact versions it was given: every version-bounded
145    /// lookup misses, as in a store that has pruned the older versions.
146    struct ExactVersions(BTreeMap<(ObjectId, Version), Object>);
147
148    #[async_trait]
149    impl ObjectProvider for ExactVersions {
150        type Error = String;
151
152        async fn get_object(&self, id: &ObjectId, version: &Version) -> Result<Object, String> {
153            self.0
154                .get(&(*id, *version))
155                .cloned()
156                .ok_or_else(|| format!("{id} at {version} not found"))
157        }
158
159        async fn find_object_lt_or_eq_version(
160            &self,
161            _id: &ObjectId,
162            _version: &Version,
163        ) -> Result<Option<Object>, String> {
164            Ok(None)
165        }
166    }
167
168    #[tokio::test]
169    async fn a_removed_object_is_described_by_the_version_it_was_removed_at() {
170        let sender = Address::ZERO;
171        let input_version = Version::from(3);
172        let removed_at = Version::from(7);
173        let (deleted, wrapped) = (ObjectId::random(), ObjectId::random());
174        let provider = ExactVersions(
175            [deleted, wrapped]
176                .into_iter()
177                .map(|id| {
178                    let object = Object::with_id_owner_version_for_testing(
179                        id,
180                        input_version,
181                        Owner::Address(sender),
182                    );
183                    ((id, input_version), object)
184                })
185                .collect(),
186        );
187        let removed = |id| ObjectReference {
188            object_id: id,
189            version: removed_at,
190            digest: ObjectDigest::new([0; 32]),
191        };
192
193        let changes = get_object_changes(
194            &provider,
195            sender,
196            vec![
197                ObjectVersion::new(deleted, input_version),
198                ObjectVersion::new(wrapped, input_version),
199            ],
200            vec![],
201            vec![
202                (removed(deleted), ObjectRemoveKind::Delete),
203                (removed(wrapped), ObjectRemoveKind::Wrap),
204            ],
205        )
206        .await
207        .unwrap();
208
209        assert!(
210            matches!(
211                changes.as_slice(),
212                [
213                    ObjectChange::Deleted { object_id: d, version: dv, .. },
214                    ObjectChange::Wrapped { object_id: w, version: wv, .. },
215                ] if *d == deleted && *dv == removed_at && *w == wrapped && *wv == removed_at
216            ),
217            "{changes:?}"
218        );
219    }
