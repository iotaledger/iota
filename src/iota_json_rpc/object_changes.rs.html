5use std::collections::BTreeMap;
6
7use iota_json_rpc_types::ObjectChange;
8use iota_sdk_types::{
9    Address, ObjectReference, ObjectRemoveKind, ObjectVersion, OwnedObjectReference, StructTag,
10    WriteKind,
11};
12
13use crate::ObjectProvider;
14
15pub async fn get_object_changes<P: ObjectProvider<Error = E>, E>(
16    object_provider: &P,
17    sender: Address,
18    modified_at_versions: Vec<ObjectVersion>,
19    all_changed_objects: Vec<(OwnedObjectReference, WriteKind)>,
20    all_removed_objects: Vec<(ObjectReference, ObjectRemoveKind)>,
21) -> Result<Vec<ObjectChange>, E> {
22    let mut object_changes = vec![];
23
24    let modify_at_version = modified_at_versions
25        .into_iter()
26        .map(|modified| (modified.object_id, modified.version))
27        .collect::<BTreeMap<_, _>>();
28
29    for (changed, kind) in all_changed_objects {
30        let OwnedObjectReference { reference, owner } = changed;
31        let ObjectReference {
32            object_id,
33            version,
34            digest,
35        } = reference;
36        let o = object_provider.get_object(&object_id, &version).await?;
37        if let Some(object_type) = o.data.opt_object_type() {
38            let object_type: StructTag = object_type.clone().into();
39
40            match kind {
41                WriteKind::Mutate => object_changes.push(ObjectChange::Mutated {
42                    sender,
43                    owner,
44                    object_type,
45                    object_id,
46                    version,
47                    // modify_at_version should always be available for mutated object
48                    previous_version: modify_at_version
49                        .get(&object_id)
50                        .cloned()
51                        .unwrap_or_default(),
52                    digest,
53                }),
54                WriteKind::Create => object_changes.push(ObjectChange::Created {
55                    sender,
56                    owner,
57                    object_type,
58                    object_id,
59                    version,
60                    digest,
61                }),
62                WriteKind::Unwrap => object_changes.push(ObjectChange::Unwrapped {
63                    sender,
64                    owner,
65                    object_type,
66                    object_id,
67                    version,
68                    digest,
69                }),
70            }
71        } else if let Some(p) = o.data.as_opt_package() {
72            if kind == WriteKind::Create {
73                object_changes.push(ObjectChange::Published {
74                    package_id: p.id(),
75                    version: p.version(),
76                    digest,
77                    modules: p
78                        .serialized_module_map()
79                        .keys()
80                        .map(|k| k.to_string())
81                        .collect(),
82                })
83            }
84        };
85    }
86
87    for (removed_object, kind) in all_removed_objects {
88        let id = removed_object.object_id;
89        let version = removed_object.version;
90        let o = object_provider
91            .find_object_lt_or_eq_version(&id, &version)
92            .await?;
93        if let Some(o) = o {
94            if let Some(object_type) = o.data.opt_object_type() {
95                let object_type: StructTag = object_type.clone().into();
96                match kind {
97                    ObjectRemoveKind::Delete => object_changes.push(ObjectChange::Deleted {
98                        sender,
99                        object_type,
100                        object_id: id,
101                        version,
102                    }),
103                    ObjectRemoveKind::Wrap => object_changes.push(ObjectChange::Wrapped {
104                        sender,
105                        object_type,
106                        object_id: id,
107                        version,
108                    }),
109                }
110            }
111        };
112    }
113
114    Ok(object_changes)
115}
