84/// Create a response body.
85fn from_template<S: Into<axum::body::Body>>(
86    status: hyper::StatusCode,
87    body: S,
88    content_type: &'static str,
89) -> Response {
90    Response::builder()
91        .status(status)
92        .header(
93            "content-type",
94            hyper::header::HeaderValue::from_static(content_type),
95        )
96        .body(body.into())
97        // Parsing `StatusCode` and `HeaderValue` is infallible but
98        // parsing body content is not.
99        .expect("unable to parse response body for type conversion")
100}

102/// Create a valid JSON response.
103pub(crate) fn ok_response(body: String) -> Response {
104    const JSON: &str = "application/json; charset=utf-8";
105    from_template(hyper::StatusCode::OK, body, JSON)
106}

108pub async fn json_rpc_handler<L: Logger>(
109    ConnectInfo(client_addr): ConnectInfo<SocketAddr>,
110    State(service): State<JsonRpcService<L>>,
111    request: HttpRequest,
112) -> Response {
113    let headers = request.headers().clone();
114    let client = match client_ip(&service, &headers, client_addr) {
115        Ok(client) => client,
116        Err(refused_response) => return ok_response(refused_response.into_result()),
117    };
118    // Check whether the client may be served before reading the body, so a
119    // refused client's body is neither read nor parsed.
120    if let Some(traffic_controller) = &service.traffic_controller {
121        if let Err(blocked_response) = handle_traffic_req(traffic_controller.clone(), &client) {
122            return ok_response(blocked_response.into_result());
123        }
124    }
125    let raw_request = match Json::<Box<RawValue>>::from_request(request, &()).await {
126        Ok(Json(raw_request)) => raw_request,
127        Err(rejection) => {
128            // Any body the `Json` extractor rejects (not JSON, no JSON content
129            // type, too large, or not fully received) is tallied like a request
130            // the node cannot parse.
131            if let Some(traffic_controller) = &service.traffic_controller {
132                let parse_error =
133                    MethodResponse::error(Id::Null, ErrorObject::from(ErrorCode::ParseError));
134                handle_traffic_resp(traffic_controller.clone(), client, &parse_error);
135            }
136            return rejection.into_response();
137        }
138    };
139    // Get version from header.
140    let api_version = headers
141        .get(CLIENT_TARGET_API_VERSION_HEADER)
142        .and_then(|h| h.to_str().ok());
143    let response = process_raw_request(&service, api_version, raw_request.get(), client).await;
144
145    ok_response(response.into_result())
146}

148/// Resolves the client IP used for traffic control. Returns the response to
149/// send instead of serving the request when the request must be refused.
150fn client_ip<L>(
151    service: &JsonRpcService<L>,
152    headers: &HeaderMap,
153    client_addr: SocketAddr,
154) -> Result<Option<IpAddr>, MethodResponse> {
155    let Some(source) = service.client_id_source.as_ref() else {
156        return Ok(None);
157    };
158    match get_client_ip(headers, Some(client_addr), source) {
159        ClientIpStatus::Ok(client) => Ok(Some(client)),
160        status => {
161            error!("{status}");
162            if status.should_refuse_request() {
163                let err_obj = ErrorObject::owned(
164                    ErrorCode::InvalidRequest.code(),
165                    status.to_string(),
166                    None::<()>,
167                );
168                return Err(MethodResponse::error(Id::Null, err_obj));
169            }
170            Ok(None)
171        }
172    }
173}

175async fn process_raw_request<L: Logger>(
176    service: &JsonRpcService<L>,
177    api_version: Option<&str>,
178    raw_request: &str,
179    client: Option<IpAddr>,
180) -> MethodResponse {
181    let response = if let Ok(request) = serde_json::from_str::<Request>(raw_request) {
182        process_request(request, api_version, service.call_data()).await
183    } else if let Ok(_batch) = serde_json::from_str::<Vec<&RawValue>>(raw_request) {
184        MethodResponse::error(
185            Id::Null,
186            ErrorObject::borrowed(BATCHES_NOT_SUPPORTED_CODE, BATCHES_NOT_SUPPORTED_MSG, None),
187        )
188    } else {
189        let (id, code) = prepare_error(raw_request);
190        MethodResponse::error(id, ErrorObject::from(code))
191    };
192    // A request the node cannot parse is tallied like any other request, so
193    // that a client cannot send malformed requests without a limit.
194    if let Some(traffic_controller) = &service.traffic_controller {
195        handle_traffic_resp(traffic_controller.clone(), client, &response);
196    }
197    response
198}
199
200fn handle_traffic_req(
201    traffic_controller: Arc<TrafficController>,
202    client: &Option<IpAddr>,
203) -> Result<(), MethodResponse> {
204    if !traffic_controller.check(client, &None) {
205        // Entity in blocklist
206        let err_obj =
207            ErrorObject::borrowed(ErrorCode::ServerIsBusy.code(), TOO_MANY_REQUESTS_MSG, None);
208        Err(MethodResponse::error(Id::Null, err_obj))
209    } else {
210        Ok(())
211    }
212}
213
214fn handle_traffic_resp(
215    traffic_controller: Arc<TrafficController>,
216    client: Option<IpAddr>,
217    response: &MethodResponse,
218) {
219    let error = response.as_error_code().map(ErrorCode::from);
220    traffic_controller.tally(TrafficTally {
221        direct: client,
222        through_fullnode: None,
223        error_info: error.map(|e| {
224            let error_type = e.to_string();
225            let error_weight = normalize(e);
226            (error_weight, error_type)
227        }),
228        // For now, count everything as spam with equal weight
229        // on the rpc node side, including gas-charging endpoints
230        // such as `iota_executeTransactionBlock`, as this can enable
231        // node operators who wish to rate limit their transaction
232        // traffic and incentivize high volume clients to choose a
233        // suitable rpc provider (or run their own). Later we may want
234        // to provide a weight distribution based on the method being called.
235        spam_weight: Weight::one(),
236    });
237}
238
239// TODO: refine error matching here
240fn normalize(err: ErrorCode) -> Weight {
241    match err {
242        ErrorCode::ParseError | ErrorCode::InvalidRequest | ErrorCode::InvalidParams => {
243            Weight::one()
244        }
245        ErrorCode::ServerError(i) if i == TRANSACTION_NOT_FOUND_ERROR_CODE => Weight::zero(),
246        // e.g. invalid client signature
247        ErrorCode::ServerError(i) if i == TRANSACTION_EXECUTION_CLIENT_ERROR_CODE => Weight::one(),
248        _ => Weight::zero(),
249    }
250}

305    async fn send(
306        router: &axum::Router,
307        content_type: Option<&str>,
308        body: impl Into<Body>,
309    ) -> (StatusCode, String) {
310        let mut request = Request::post("/").extension(ConnectInfo(CLIENT));
311        if let Some(content_type) = content_type {
312            request = request.header("content-type", content_type);
313        }
314        let response = router
315            .clone()
316            .oneshot(request.body(body.into()).unwrap())
317            .await
318            .unwrap();
319        let status = response.status();
320        let body = axum::body::to_bytes(response.into_body(), usize::MAX)
321            .await
322            .unwrap();
323        (status, String::from_utf8_lossy(&body).into_owned())
324    }

405async fn process_request<L: Logger>(
406    req: Request<'_>,
407    api_version: Option<&str>,
408    call: CallData<'_, L>,
409) -> MethodResponse {
410    let CallData {
411        methods,
412        rpc_router,
413        logger,
414        extensions,
415        max_response_body_size,
416        request_start,
417    } = call;
418    let conn_id = ConnectionId(0); // unused
419
420    let name = rpc_router.route(&req.method, api_version);
421    let params = Params::new(req.params.as_ref().map(|params| params.get()));
422
423    let id = req.id;
424
425    let response = match methods.method_with_name(name) {
426        None => {
427            logger.on_call(name, params.clone(), MethodKind::NotFound);
428            MethodResponse::error(id, ErrorObject::from(ErrorCode::MethodNotFound))
429        }
430        Some((name, method)) => match method {
431            MethodCallback::Sync(callback) => {
432                logger.on_call(name, params.clone(), MethodKind::MethodCall);
433                (callback)(
434                    id,
435                    params,
436                    max_response_body_size as usize,
437                    extensions.clone(),
438                )
439            }
440            MethodCallback::Async(callback) => {
441                logger.on_call(name, params.clone(), MethodKind::MethodCall);
442
443                let id = id.into_owned();
444                let params = params.into_owned();
445
446                (callback)(
447                    id,
448                    params,
449                    conn_id,
450                    max_response_body_size as usize,
451                    extensions.clone(),
452                )
453                .await
454            }
455            MethodCallback::Subscription(_) | MethodCallback::Unsubscription(_) => {
456                logger.on_call(name, params.clone(), MethodKind::NotFound);
457                // Subscriptions not supported on HTTP
458                MethodResponse::error(id, ErrorObject::from(ErrorCode::InternalError))
459            }
460        },
461    };
462
463    logger.on_result(
464        name,
465        response.is_success(),
466        response.as_error_code(),
467        request_start,
468    );
469    response
470}

472/// Figure out if this is a sufficiently complete request that we can extract an
473/// [`Id`] out of, or just plain unparsable garbage.
474pub fn prepare_error(data: &str) -> (Id<'_>, ErrorCode) {
475    match serde_json::from_str::<InvalidRequest>(data) {
476        Ok(InvalidRequest { id }) => (id, ErrorCode::InvalidRequest),
477        Err(_) => (Id::Null, ErrorCode::ParseError),
478    }
479}
