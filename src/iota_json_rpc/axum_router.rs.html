36pub const MAX_RESPONSE_SIZE: u32 = 2 << 30;
37const TOO_MANY_REQUESTS_MSG: &str = "Too many requests";

39#[derive(Clone, Debug)]
40pub struct JsonRpcService<L> {
41    logger: L,
42
43    /// Registered server methods.
44    methods: Methods,
45    extensions: Extensions,
46    rpc_router: RpcRouter,
47    traffic_controller: Option<Arc<TrafficController>>,
48    client_id_source: Option<ClientIdSource>,
49}

51impl<L> JsonRpcService<L> {
52    pub fn new(
53        methods: Methods,
54        rpc_router: RpcRouter,
55        logger: L,
56        traffic_controller: Option<Arc<TrafficController>>,
57        policy_config: Option<PolicyConfig>,
58        extensions: Extensions,
59    ) -> Self {
60        Self {
61            methods,
62            rpc_router,
63            logger,
64            extensions,
65            traffic_controller,
66            client_id_source: policy_config.map(|policy| policy.client_id_source),
67        }
68    }
69}

84/// Create a response body.
85fn from_template<S: Into<axum::body::Body>>(
86    status: hyper::StatusCode,
87    body: S,
88    content_type: &'static str,
89) -> Response {
90    Response::builder()
91        .status(status)
92        .header(
93            "content-type",
94            hyper::header::HeaderValue::from_static(content_type),
95        )
96        .body(body.into())
97        // Parsing `StatusCode` and `HeaderValue` is infallible but
98        // parsing body content is not.
99        .expect("unable to parse response body for type conversion")
100}
101
102/// Create a valid JSON response.
103pub(crate) fn ok_response(body: String) -> Response {
104    const JSON: &str = "application/json; charset=utf-8";
105    from_template(hyper::StatusCode::OK, body, JSON)
106}

108pub async fn json_rpc_handler<L: Logger>(
109    ConnectInfo(client_addr): ConnectInfo<SocketAddr>,
110    State(service): State<JsonRpcService<L>>,
111    headers: HeaderMap,
112    Json(raw_request): Json<Box<RawValue>>,
113) -> impl axum::response::IntoResponse {
114    let headers_clone = headers.clone();
115    // Get version from header.
116    let api_version = headers
117        .get(CLIENT_TARGET_API_VERSION_HEADER)
118        .and_then(|h| h.to_str().ok());
119    let response = process_raw_request(
120        &service,
121        api_version,
122        raw_request.get(),
123        client_addr,
124        headers_clone,
125    )
126    .await;
127
128    ok_response(response.into_result())
129}
130
131async fn process_raw_request<L: Logger>(
132    service: &JsonRpcService<L>,
133    api_version: Option<&str>,
134    raw_request: &str,
135    client_addr: SocketAddr,
136    headers: HeaderMap,
137) -> MethodResponse {
138    let client = match service.client_id_source.as_ref() {
139        Some(source) => match get_client_ip(&headers, Some(client_addr), source) {
140            ClientIpStatus::Ok(client) => Some(client),
141            status => {
142                error!("{status}");
143                None
144            }
145        },
146        None => None,
147    };
148    // check if either IP is blocked, in which case return early
149    if let Some(traffic_controller) = &service.traffic_controller {
150        if let Err(blocked_response) = handle_traffic_req(traffic_controller.clone(), &client) {
151            return blocked_response;
152        }
153    }
154    let response = if let Ok(request) = serde_json::from_str::<Request>(raw_request) {
155        process_request(request, api_version, service.call_data()).await
156    } else if let Ok(_batch) = serde_json::from_str::<Vec<&RawValue>>(raw_request) {
157        MethodResponse::error(
158            Id::Null,
159            ErrorObject::borrowed(BATCHES_NOT_SUPPORTED_CODE, BATCHES_NOT_SUPPORTED_MSG, None),
160        )
161    } else {
162        let (id, code) = prepare_error(raw_request);
163        MethodResponse::error(id, ErrorObject::from(code))
164    };
165    // A request the node cannot parse is tallied like any other request, so
166    // that a client cannot send malformed requests without a limit.
167    if let Some(traffic_controller) = &service.traffic_controller {
168        handle_traffic_resp(traffic_controller.clone(), client, &response);
169    }
170    response
171}
172
173fn handle_traffic_req(
174    traffic_controller: Arc<TrafficController>,
175    client: &Option<IpAddr>,
176) -> Result<(), MethodResponse> {
177    if !traffic_controller.check(client, &None) {
178        // Entity in blocklist
179        let err_obj =
180            ErrorObject::borrowed(ErrorCode::ServerIsBusy.code(), TOO_MANY_REQUESTS_MSG, None);
181        Err(MethodResponse::error(Id::Null, err_obj))
182    } else {
183        Ok(())
184    }
185}
186
187fn handle_traffic_resp(
188    traffic_controller: Arc<TrafficController>,
189    client: Option<IpAddr>,
190    response: &MethodResponse,
191) {
192    let error = response.as_error_code().map(ErrorCode::from);
193    traffic_controller.tally(TrafficTally {
194        direct: client,
195        through_fullnode: None,
196        error_info: error.map(|e| {
197            let error_type = e.to_string();
198            let error_weight = normalize(e);
199            (error_weight, error_type)
200        }),
201        // For now, count everything as spam with equal weight
202        // on the rpc node side, including gas-charging endpoints
203        // such as `iota_executeTransactionBlock`, as this can enable
204        // node operators who wish to rate limit their transaction
205        // traffic and incentivize high volume clients to choose a
206        // suitable rpc provider (or run their own). Later we may want
207        // to provide a weight distribution based on the method being called.
208        spam_weight: Weight::one(),
209    });
210}
211
212// TODO: refine error matching here
213fn normalize(err: ErrorCode) -> Weight {
214    match err {
215        ErrorCode::ParseError | ErrorCode::InvalidRequest | ErrorCode::InvalidParams => {
216            Weight::one()
217        }
218        ErrorCode::ServerError(i) if i == TRANSACTION_NOT_FOUND_ERROR_CODE => Weight::zero(),
219        // e.g. invalid client signature
220        ErrorCode::ServerError(i) if i == TRANSACTION_EXECUTION_CLIENT_ERROR_CODE => Weight::one(),
221        _ => Weight::zero(),
222    }
223}

256async fn process_request<L: Logger>(
257    req: Request<'_>,
258    api_version: Option<&str>,
259    call: CallData<'_, L>,
260) -> MethodResponse {
261    let CallData {
262        methods,
263        rpc_router,
264        logger,
265        extensions,
266        max_response_body_size,
267        request_start,
268    } = call;
269    let conn_id = ConnectionId(0); // unused
270
271    let name = rpc_router.route(&req.method, api_version);
272    let params = Params::new(req.params.as_ref().map(|params| params.get()));
273
274    let id = req.id;
275
276    let response = match methods.method_with_name(name) {
277        None => {
278            logger.on_call(name, params.clone(), MethodKind::NotFound);
279            MethodResponse::error(id, ErrorObject::from(ErrorCode::MethodNotFound))
280        }
281        Some((name, method)) => match method {
282            MethodCallback::Sync(callback) => {
283                logger.on_call(name, params.clone(), MethodKind::MethodCall);
284                (callback)(
285                    id,
286                    params,
287                    max_response_body_size as usize,
288                    extensions.clone(),
289                )
290            }
291            MethodCallback::Async(callback) => {
292                logger.on_call(name, params.clone(), MethodKind::MethodCall);
293
294                let id = id.into_owned();
295                let params = params.into_owned();
296
297                (callback)(
298                    id,
299                    params,
300                    conn_id,
301                    max_response_body_size as usize,
302                    extensions.clone(),
303                )
304                .await
305            }
306            MethodCallback::Subscription(_) | MethodCallback::Unsubscription(_) => {
307                logger.on_call(name, params.clone(), MethodKind::NotFound);
308                // Subscriptions not supported on HTTP
309                MethodResponse::error(id, ErrorObject::from(ErrorCode::InternalError))
310            }
311        },
312    };
313
314    logger.on_result(
315        name,
316        response.is_success(),
317        response.as_error_code(),
318        request_start,
319    );
320    response
321}

323/// Figure out if this is a sufficiently complete request that we can extract an
324/// [`Id`] out of, or just plain unparsable garbage.
325pub fn prepare_error(data: &str) -> (Id<'_>, ErrorCode) {
326    match serde_json::from_str::<InvalidRequest>(data) {
327        Ok(InvalidRequest { id }) => (id, ErrorCode::InvalidRequest),
328        Err(_) => (Id::Null, ErrorCode::ParseError),
329    }
330}
