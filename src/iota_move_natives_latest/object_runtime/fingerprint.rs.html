5use iota_protocol_config::ProtocolConfig;
6use iota_sdk_types::{ObjectId, StructTag};
7use move_binary_format::errors::{PartialVMError, PartialVMResult};
8use move_core_types::vm_status::StatusCode;
9use move_vm_types::values::Value;
10
11/// This type is used to track if an object has changed since it was read from
12/// storage. Ideally, this would just store the owner ID+type+BCS bytes of the
13/// object; however, due to pending rewrites in the adapter, that would be too
14/// much code churn. Instead, we use the `Value` since the `RuntimeResults`
15/// operate over `Value` and not BCS bytes.
16pub struct ObjectFingerprint(Option<ObjectFingerprint_>);
17
18enum ObjectFingerprint_ {
19    /// The object did not exist (as a child object) in storage at the start of
20    /// the transaction.
21    Empty,
22    // The object was loaded as a child object from storage.
23    Preexisting {
24        owner: ObjectId,
25        ty: StructTag,
26        value: Value,
27    },
28}
29
30impl ObjectFingerprint {
31    #[cfg(debug_assertions)]
32    pub fn is_disabled(&self) -> bool {
33        self.0.is_none()
34    }
35
36    /// Creates a new object fingerprint for a child object not found in
37    /// storage. Will be internally disabled if the feature is not enabled
38    /// in the protocol config.
39    pub fn none(protocol_config: &ProtocolConfig) -> Self {
40        if !protocol_config.minimize_child_object_mutations() {
41            Self(None)
42        } else {
43            Self(Some(ObjectFingerprint_::Empty))
44        }
45    }
46
47    /// Creates a new object fingerprint for a child found in storage.
48    /// Will be internally disabled if the feature is not enabled in the
49    /// protocol config.
50    pub fn preexisting(
51        protocol_config: &ProtocolConfig,
52        preexisting_owner: &ObjectId,
53        preexisting_type: &StructTag,
54        preexisting_value: &Value,
55    ) -> PartialVMResult<Self> {
56        Ok(if !protocol_config.minimize_child_object_mutations() {
57            Self(None)
58        } else {
59            Self(Some(ObjectFingerprint_::Preexisting {
60                owner: *preexisting_owner,
61                ty: preexisting_type.clone(),
62                value: preexisting_value.copy_value()?,
63            }))
64        })
65    }
66
67    /// Checks if the object has changed since it was read from storage.
68    /// Gives an invariant violation if the fingerprint is disabled.
69    /// Gives an invariant violation if the values do not have the same layout,
70    /// but the owner and type are thesame.
71    pub fn object_has_changed(
72        &self,
73        final_owner: &ObjectId,
74        final_type: &StructTag,
75        final_value: &Option<Value>,
76    ) -> PartialVMResult<bool> {
77        use ObjectFingerprint_ as F;
78        let Some(inner) = &self.0 else {
79            return Err(
80                PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR).with_message(
81                    "Object fingerprint not enabled, yet we were asked for the changes".to_string(),
82                ),
83            );
84        };
85        Ok(match (inner, final_value) {
86            (F::Empty, None) => false,
87            (F::Empty, Some(_)) | (F::Preexisting { .. }, None) => true,
88            (
89                F::Preexisting {
90                    owner: preexisting_owner,
91                    ty: preexisting_type,
92                    value: preexisting_value,
93                },
94                Some(final_value),
95            ) => {
96                // owner changed or value changed.
97                // For the value, we must first check if the types are the same before comparing
98                // the values
99                !(preexisting_owner == final_owner
100                    && preexisting_type == final_type
101                    && preexisting_value.equals(final_value)?)
102            }
103        })
104    }
105}
