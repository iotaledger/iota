31/// Raised when the `(flag, raw_bytes)` pair does not form a public key whose
32/// address can be derived. `public_key::create` validates keys before calling
33/// this native, so this abort is not expected to be reachable for
34/// `public_key::PublicKey`-derived calls.
35const E_INVALID_PUBLIC_KEY: u64 = 0;
36
37#[derive(Clone)]
38pub struct PublicKeyToIotaAddressImplCostParams {
39    pub public_key_to_iota_address_impl_cost_base: Option<InternalGas>,
40}
41/// Implementation of the Move native function
42/// `public_key::to_iota_address_impl(flag: u8, raw_bytes: &vector<u8>):
43/// address`
44///
45/// Derives the IOTA address for the `flag`-typed public key with raw key
46/// material `raw_bytes` (no scheme flag prefix), delegating to the node's
47/// canonical address derivation so Move and Rust cannot diverge.
48///
49/// Aborts with `E_INVALID_PUBLIC_KEY` if `flag` is unrecognized or `raw_bytes`
50/// is not a valid public key for it.
51///
52/// gas cost: public_key_to_iota_address_impl_cost_base
53pub fn to_iota_address_impl(
54    context: &mut NativeContext,
55    ty_args: Vec<Type>,
56    mut args: VecDeque<Value>,
57) -> PartialVMResult<NativeResult> {
58    debug_assert!(ty_args.is_empty());
59    debug_assert!(args.len() == 2);
60
61    let cost_base = context
62        .extensions()
63        .get::<NativesCostTable>()?
64        .public_key_to_iota_address_impl_cost_params
65        .public_key_to_iota_address_impl_cost_base
66        .ok_or_else(|| {
67            PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR)
68                .with_message("Gas cost for to_iota_address_impl not available".to_string())
69        })?;
70    native_charge_gas_early_exit!(context, cost_base);
71
72    let raw_bytes = pop_arg!(args, VectorRef);
73    let raw_bytes_ref = raw_bytes.as_bytes_ref();
74    let flag = pop_arg!(args, u8);
75    let cost = context.gas_used();
76
77    let address = match SignatureScheme::from_byte(flag) {
78        // MultiSig is not part of the `PublicKey` enum; derive from its committee bytes.
79        Ok(SignatureScheme::Multisig) => bcs::from_bytes::<MultisigCommittee>(&raw_bytes_ref)
80            .ok()
81            .map(|committee| committee.derive_address()),
82        Ok(scheme) => PublicKey::try_from_bytes(scheme, &raw_bytes_ref)
83            .ok()
84            .map(|public_key| IotaAddress::from(&public_key)),
85        Err(_) => None,
86    };
87
88    Ok(match address {
89        Some(address) => NativeResult::ok(
90            cost,
91            smallvec![Value::address(AccountAddress::new(address.into_bytes()))],
92        ),
93        None => NativeResult::err(cost, E_INVALID_PUBLIC_KEY),
94    })
95}

97#[derive(Clone)]
98pub struct PublicKeyVerifyPersonalMessageSignatureImplCostParams {
99    /// Base cost for invoking the `verify_personal_message_signature_impl` function
100    pub public_key_verify_personal_message_signature_impl_cost_base: Option<InternalGas>,
101    /// Cost per byte of `message`
102    pub public_key_verify_personal_message_signature_impl_msg_cost_per_byte: Option<InternalGas>,
103    /// Cost per byte of `signature`
104    pub public_key_verify_personal_message_signature_impl_sig_cost_per_byte: Option<InternalGas>,
105}
106/// Implementation of the Move native function
107/// `public_key::verify_personal_message_signature_impl(public_key: &PublicKey,
108/// signature: &vector<u8>, message: &vector<u8>): bool`
109///
110/// Returns `true` if `signature` is a valid signature, by the public key, of
111/// `message` signed as an IOTA personal message, using the signature scheme of
112/// the public key. Returns `false` for a MultiSig public key.
113///
114/// gas cost: public_key_verify_personal_message_signature_impl_cost_base
115///     + public_key_verify_personal_message_signature_impl_msg_cost_per_byte * message.len()
116///     + public_key_verify_personal_message_signature_impl_sig_cost_per_byte * signature.len()
117pub fn verify_personal_message_signature_impl(
118    context: &mut NativeContext,
119    ty_args: Vec<Type>,
120    mut args: VecDeque<Value>,
121) -> PartialVMResult<NativeResult> {
122    debug_assert!(ty_args.is_empty());
123    debug_assert!(args.len() == 3);
124
125    let cost_params = get_extension!(context, NativesCostTable)?
126        .public_key_verify_personal_message_signature_impl_cost_params
127        .clone();
128    let (Some(cost_base), Some(msg_cost_per_byte), Some(sig_cost_per_byte)) = (
129        cost_params.public_key_verify_personal_message_signature_impl_cost_base,
130        cost_params.public_key_verify_personal_message_signature_impl_msg_cost_per_byte,
131        cost_params.public_key_verify_personal_message_signature_impl_sig_cost_per_byte,
132    ) else {
133        return Err(
134            PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR).with_message(
135                "Gas cost for verify_personal_message_signature_impl not available".to_string(),
136            ),
137        );
138    };
139    native_charge_gas_early_exit!(context, cost_base);
140
141    let message = pop_arg!(args, VectorRef);
142    let message_ref = message.as_bytes_ref();
143    let signature = pop_arg!(args, VectorRef);
144    let signature_ref = signature.as_bytes_ref();
145    let public_key: MovePublicKey = utils::from_value(
146        pop_arg!(args, StructRef).read_ref()?,
147        &MovePublicKey::layout(),
148    )?;
149
150    native_charge_gas_early_exit!(
151        context,
152        msg_cost_per_byte * (message_ref.len() as u64).into()
153    );
154    // Charged before the check decodes the signature, whose length the caller
155    // chooses.
156    native_charge_gas_early_exit!(
157        context,
158        sig_cost_per_byte * (signature_ref.len() as u64).into()
159    );
160
161    let protocol_config = get_extension!(context, ObjectRuntime)?.protocol_config;
162
163    let is_valid = verify_personal_message_signature(
164        protocol_config,
165        &public_key,
166        &signature_ref,
167        &message_ref,
168    )
169    .is_ok();
170
171    Ok(NativeResult::ok(
172        context.gas_used(),
173        smallvec![Value::bool(is_valid)],
174    ))
175}

177#[derive(Clone)]
178pub struct PublicKeyVerifyBuiltinAuthenticatorSignatureImplCostParams {
179    pub public_key_verify_builtin_authenticator_signature_impl_cost_base: Option<InternalGas>,
180}
181/// Implementation of the Move native function
182/// `public_key::verify_builtin_authenticator_signature_impl(public_key: &PublicKey,
183/// signature: &vector<u8>): bool`
184///
185/// Returns `true` if `signature` is a valid signature, by the public key, of
186/// the transaction being authenticated, using the signature scheme of the
187/// public key. Only the built-in authenticator calls it, while it
188/// authenticates a transaction: the transaction comes from the
189/// `AuthenticationContext`.
190///
191/// gas cost: public_key_verify_builtin_authenticator_signature_impl_cost_base
192pub fn verify_builtin_authenticator_signature_impl(
193    context: &mut NativeContext,
194    ty_args: Vec<Type>,
195    mut args: VecDeque<Value>,
196) -> PartialVMResult<NativeResult> {
197    debug_assert!(ty_args.is_empty());
198    debug_assert!(args.len() == 2);
199
200    let cost_base = get_extension!(context, NativesCostTable)?
201        .public_key_verify_builtin_authenticator_signature_impl_cost_params
202        .public_key_verify_builtin_authenticator_signature_impl_cost_base
203        .ok_or_else(|| {
204            PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR)
205                .with_message("Gas cost for built-in authenticators not available".to_string())
206        })?;
207    native_charge_gas_early_exit!(context, cost_base);
208
209    let signature = pop_arg!(args, VectorRef);
210    let public_key: MovePublicKey = utils::from_value(
211        pop_arg!(args, StructRef).read_ref()?,
212        &MovePublicKey::layout(),
213    )?;
214
215    let protocol_config = get_extension!(context, ObjectRuntime)?.protocol_config;
216
217    let auth_context = get_extension!(context, AuthenticationContext)?
218        .auth_context
219        .borrow();
220
221    let is_valid = verify_builtin_authenticator_signature(
222        protocol_config,
223        &public_key,
224        &signature.as_bytes_ref(),
225        auth_context.tx_data_bytes(),
226    )
227    .is_ok();
228
229    Ok(NativeResult::ok(
230        context.gas_used(),
231        smallvec![Value::bool(is_valid)],
232    ))
