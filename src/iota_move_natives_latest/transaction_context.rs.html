19// TransactionContext is a wrapper around TxContext that is exposed to
20// NativeContextExtensions in order to provide transaction context information
21// to Move native functions. Holds a Rc<RefCell<TxContext>> to allow for
22// mutation of the TxContext.
23#[derive(Tid)]
24pub struct TransactionContext {
25    pub(crate) tx_context: Rc<RefCell<TxContext>>,
26    test_only: bool,
27
28    /// Cached `GlobalValue` containing TxContext data. Caching is used to
29    /// avoid redundant conversions and allocations.
30    cached_digest: Option<GlobalValue>,
31}
32
33impl NativeExtensionMarker<'_> for TransactionContext {}
34
35impl TransactionContext {
36    pub fn new(tx_context: Rc<RefCell<TxContext>>) -> Self {
37        Self {
38            tx_context,
39            test_only: false,
40            cached_digest: None,
41        }
42    }
43
44    pub fn new_for_testing(tx_context: Rc<RefCell<TxContext>>) -> Self {
45        Self {
46            tx_context,
47            test_only: true,
48            cached_digest: None,
49        }
50    }
51
52    pub fn sender(&self) -> Address {
53        self.tx_context.borrow().sender()
54    }
55
56    pub fn epoch(&self) -> EpochId {
57        self.tx_context.borrow().epoch()
58    }
59
60    pub fn epoch_timestamp_ms(&self) -> u64 {
61        self.tx_context.borrow().epoch_timestamp_ms()
62    }
63
64    pub fn digest(&self) -> TransactionDigest {
65        self.tx_context.borrow().digest()
66    }
67
68    /// Returns a `Value` containing a transaction digest ref.
69    /// Caches the result to avoid redundant conversions and allocations on
70    /// subsequent calls.
71    pub fn digest_ref(&mut self) -> PartialVMResult<Value> {
72        if self.cached_digest.is_none() {
73            let tx_context = self.tx_context.borrow();
74
75            // Wrap in a tuple to match the expected Move layout of
76            // `struct TxContext {
77            //     digest: vector<u8>
78            // }`
79            let rust_value = (tx_context.digest(),);
80            let digest_move_layout = MoveTypeLayout::Vector(Box::new(MoveTypeLayout::U8));
81
82            self.cached_digest = Some(utils::to_global_value(&rust_value, digest_move_layout)?.0);
83        }
84
85        self.cached_digest
86            .as_ref()
87            .unwrap()
88            .borrow_global()
89            .inspect_err(|err| assert!(err.major_status() != StatusCode::MISSING_DATA))?
90            .value_as::<StructRef>()?
91            .borrow_field(0)
92    }
93
94    pub fn sponsor(&self) -> Option<Address> {
95        self.tx_context.borrow().sponsor()
96    }
97
98    pub fn chain_identifier(&self) -> ChainIdentifier {
99        self.tx_context.borrow().chain_identifier()
100    }
101
102    pub fn rgp(&self) -> u64 {
103        self.tx_context.borrow().rgp()
104    }
105
106    pub fn gas_price(&self) -> u64 {
107        self.tx_context.borrow().gas_price()
108    }
109
110    pub fn gas_budget(&self) -> u64 {
111        self.tx_context.borrow().gas_budget()
112    }
113
114    pub fn ids_created(&self) -> u64 {
115        self.tx_context.borrow().ids_created()
116    }
117
118    pub fn fresh_id(&self) -> ObjectId {
119        self.tx_context.borrow_mut().fresh_id()
120    }
121
122    pub fn tx_nonce(&self) -> Option<[u8; 32]> {
123        self.tx_context.borrow().nonce()
124    }
125
126    // Test only function: replace all fields of the wrapped TxContext.
127    pub fn replace(
128        &mut self,
129        sender: AccountAddress,
130        tx_hash: Vec<u8>,
131        epoch: u64,
132        epoch_timestamp_ms: u64,
133        ids_created: u64,
134        rgp: u64,
135        gas_price: u64,
136        gas_budget: u64,
137        sponsor: Option<AccountAddress>,
138    ) -> PartialVMResult<()> {
139        if !self.test_only {
140            return Err(
141                PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR)
142                    .with_message("`replace` called on a non testing scenario".to_string()),
143            );
144        }
145        self.tx_context.borrow_mut().replace(
146            sender,
147            tx_hash,
148            epoch,
149            epoch_timestamp_ms,
150            ids_created,
151            rgp,
152            gas_price,
153            gas_budget,
154            sponsor,
155        );
156
157        // Drop cached values to ensure they are recreated with the updated TxContext
158        // data
159        self.cached_digest = None;
160
161        Ok(())
162    }
163
164    // Test only function: replace the chain identifier of the wrapped TxContext.
165    pub fn set_chain_identifier(
166        &mut self,
167        chain_identifier: ChainIdentifier,
168    ) -> PartialVMResult<()> {
169        if !self.test_only {
170            return Err(
171                PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR).with_message(
172                    "`set_chain_identifier` called on a non testing scenario".to_string(),
173                ),
174            );
175        }
176        self.tx_context
177            .borrow_mut()
178            .set_chain_identifier(chain_identifier);
179        Ok(())
180    }
181
182    // Test only function: replace the nonce of the wrapped TxContext.
183    pub fn replace_tx_nonce(&mut self, nonce: Option<[u8; 32]>) -> PartialVMResult<()> {
184        if !self.test_only {
185            return Err(
186                PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR).with_message(
187                    "`replace_tx_nonce` called on a non testing scenario".to_string(),
188                ),
189            );
190        }
191        self.tx_context.borrow_mut().replace_nonce(nonce);
192        Ok(())
193    }
194}
