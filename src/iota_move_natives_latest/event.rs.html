20#[derive(Clone, Debug)]
21pub struct EventEmitCostParams {
22    pub event_emit_cost_base: InternalGas,
23    pub event_emit_value_size_derivation_cost_per_byte: InternalGas,
24    pub event_emit_tag_size_derivation_cost_per_byte: InternalGas,
25    pub event_emit_output_cost_per_byte: InternalGas,
26}
27/// ****************************************************************************
28/// ********************* native fun emit
29/// Implementation of the Move native function `event::emit<T: copy +
30/// drop>(event: T)` Adds an event to the transaction's event log
31///   gas cost: event_emit_cost_base                  |  covers various fixed
32/// costs in the oper
33///              + event_emit_value_size_derivation_cost_per_byte * event_size |
34///                derivation of size
35///              + event_emit_tag_size_derivation_cost_per_byte * tag_size |
36///                converting type
37///              + event_emit_output_cost_per_byte * (tag_size + event_size) |
38///                emitting the actual event
39/// ****************************************************************************
40/// *******************
41pub fn emit(
42    context: &mut NativeContext,
43    mut ty_args: Vec<Type>,
44    mut args: VecDeque<Value>,
45) -> PartialVMResult<NativeResult> {
46    debug_assert!(ty_args.len() == 1);
47    debug_assert!(args.len() == 1);
48
49    let event_emit_cost_params = context
50        .extensions_mut()
51        .get::<NativesCostTable>()?
52        .event_emit_cost_params
53        .clone();
54
55    native_charge_gas_early_exit!(context, event_emit_cost_params.event_emit_cost_base);
56
57    let ty = ty_args.pop().unwrap();
58    let event_value = args.pop_back().unwrap();
59
60    let event_value_size = event_value.legacy_size();
61
62    // Deriving event value size can be expensive due to recursion overhead
63    native_charge_gas_early_exit!(
64        context,
65        event_emit_cost_params.event_emit_value_size_derivation_cost_per_byte
66            * u64::from(event_value_size).into()
67    );
68
69    let tag = match context.type_to_type_tag(&ty)? {
70        TypeTag::Struct(s) => s,
71        _ => {
72            return Err(
73                PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR)
74                    .with_message("IOTA verifier guarantees this is a struct".to_string()),
75            );
76        }
77    };
78    let tag_size = tag.abstract_size_for_gas_metering();
79
80    // Converting type to typetag be expensive due to recursion overhead
81    native_charge_gas_early_exit!(
82        context,
83        event_emit_cost_params.event_emit_tag_size_derivation_cost_per_byte
84            * u64::from(tag_size).into()
85    );
86
87    let obj_runtime: &mut ObjectRuntime = context.extensions_mut().get_mut()?;
88    let max_event_emit_size = obj_runtime.protocol_config.max_event_emit_size();
89    let ev_size = u64::from(tag_size + event_value_size);
90    // Check if the event size is within the limit
91    if ev_size > max_event_emit_size {
92        return Err(PartialVMError::new(StatusCode::MEMORY_LIMIT_EXCEEDED)
93            .with_message(format!(
94                "Emitting event of size {ev_size} bytes. Limit is {max_event_emit_size} bytes."
95            ))
96            .with_sub_status(
97                VMMemoryLimitExceededSubStatusCode::EVENT_SIZE_LIMIT_EXCEEDED as u64,
98            ));
99    }
100
101    // Check that the size contribution of the event is within the total size limit
102    // This feature is guarded as its only present in some versions
103    if let Some(max_event_emit_size_total) = obj_runtime
104        .protocol_config
105        .max_event_emit_size_total_as_option()
106    {
107        let total_events_size = obj_runtime.state.total_events_size() + ev_size;
108        if total_events_size > max_event_emit_size_total {
109            return Err(PartialVMError::new(StatusCode::MEMORY_LIMIT_EXCEEDED)
110                .with_message(format!(
111                    "Reached total event size of size {total_events_size} bytes. Limit is {max_event_emit_size_total} bytes."
112                ))
113                .with_sub_status(
114                    VMMemoryLimitExceededSubStatusCode::TOTAL_EVENT_SIZE_LIMIT_EXCEEDED as u64,
115                ));
116        }
117        obj_runtime.state.incr_total_events_size(ev_size);
118    }
119    // Emitting an event is cheap since its a vector push
120    native_charge_gas_early_exit!(
121        context,
122        event_emit_cost_params.event_emit_output_cost_per_byte * ev_size.into()
123    );
124
125    let obj_runtime: &mut ObjectRuntime = context.extensions_mut().get_mut()?;
126
127    obj_runtime.emit_event(ty, *tag, event_value)?;
128    Ok(NativeResult::ok(context.gas_used(), smallvec![]))
129}

131/// Get the all emitted events of type `T`, starting at the specified index
132pub fn num_events(
133    context: &mut NativeContext,
134    ty_args: Vec<Type>,
135    args: VecDeque<Value>,
136) -> PartialVMResult<NativeResult> {
137    assert!(ty_args.is_empty());
138    assert!(args.is_empty());
139    let object_runtime_ref: &ObjectRuntime = context.extensions().get()?;
140    let num_events = object_runtime_ref.state.events().len();
141    Ok(NativeResult::ok(
142        legacy_test_cost(),
143        smallvec![Value::u32(num_events as u32)],
144    ))
145}

147/// Get the all emitted events of type `T`, starting at the specified index
148pub fn get_events_by_type(
149    context: &mut NativeContext,
150    mut ty_args: Vec<Type>,
151    args: VecDeque<Value>,
152) -> PartialVMResult<NativeResult> {
153    assert_eq!(ty_args.len(), 1);
154    let specified_ty = ty_args.pop().unwrap();
155    let specialization: VectorSpecialization = (&specified_ty).try_into()?;
156    assert!(args.is_empty());
157    let object_runtime_ref: &ObjectRuntime = context.extensions().get()?;
158    let matched_events = object_runtime_ref
159        .state
160        .events()
161        .iter()
162        .filter_map(|(ty, _, event)| {
163            if specified_ty == *ty {
164                Some(event.copy_value().unwrap())
165            } else {
166                None
167            }
168        })
169        .collect::<Vec<_>>();
170    Ok(NativeResult::ok(
171        legacy_test_cost(),
172        smallvec![move_vm_types::values::Vector::pack(
173            specialization,
174            matched_events
175        )?],
176    ))
177}
