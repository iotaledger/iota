19/// Raised when the chain identifier or the nonce is not 32 bytes.
20const E_INVALID_KEY_PROOF_MESSAGE_VALUE: u64 = 5;
21
22#[derive(Clone)]
23pub struct SmartAccountBuiltinAuthenticatorKeyProofMessageCostParams {
24    pub smart_account_builtin_authenticator_key_proof_message_cost_base: Option<InternalGas>,
25}
26/// Implementation of the Move native function
27/// `smart_account_builtin_authenticator::key_proof_message(chain_identifier:
28/// vector<u8>, account: address, new_account: bool, nonce: vector<u8>,
29/// valid_until_epoch: u64): vector<u8>`
30///
31/// Returns the UTF-8 bytes of the text that the SDK's `key_proof_message_v1`
32/// writes for these values, the text a key holder signs to approve attaching
33/// their key to a `SmartAccount`. Clients call the same SDK function, so they
34/// sign the bytes this returns.
35///
36/// Aborts with `E_INVALID_KEY_PROOF_MESSAGE_VALUE` if `chain_identifier` or
37/// `nonce` is not 32 bytes.
38///
39/// gas cost: smart_account_builtin_authenticator_key_proof_message_cost_base
40pub fn key_proof_message(
41    context: &mut NativeContext,
42    ty_args: Vec<Type>,
43    mut args: VecDeque<Value>,
44) -> PartialVMResult<NativeResult> {
45    debug_assert!(ty_args.is_empty());
46    debug_assert!(args.len() == 5);
47
48    let cost_base = context
49        .extensions()
50        .get::<NativesCostTable>()?
51        .smart_account_builtin_authenticator_key_proof_message_cost_params
52        .smart_account_builtin_authenticator_key_proof_message_cost_base
53        .ok_or_else(|| {
54            PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR)
55                .with_message("Gas cost for key_proof_message not available".to_string())
56        })?;
57    native_charge_gas_early_exit!(context, cost_base);
58
59    let valid_until_epoch = pop_arg!(args, u64);
60    let nonce = pop_arg!(args, Vec<u8>);
61    let new_account = pop_arg!(args, bool);
62    let account = pop_arg!(args, AccountAddress);
63    let chain_identifier = pop_arg!(args, Vec<u8>);
64    let cost = context.gas_used();
65
66    let (Ok(chain_identifier), Ok(nonce)) = (
67        <[u8; 32]>::try_from(chain_identifier),
68        <[u8; 32]>::try_from(nonce),
69    ) else {
70        return Ok(NativeResult::err(cost, E_INVALID_KEY_PROOF_MESSAGE_VALUE));
71    };
72
73    let message = key_proof_message_v1(
74        &chain_identifier,
75        &Address::new(account.into_bytes()),
76        new_account,
77        &nonce,
78        valid_until_epoch,
79    );
80
81    Ok(NativeResult::ok(
82        cost,
83        smallvec![Value::vector_u8(message.into_bytes())],
84    ))
85}
