23/// The length of the largest valid BCS-encoded MultiSig committee (353 bytes).
24///
25/// The SDK does not export this bound, so it is derived here from the BCS
26/// layout of `MultisigCommittee`:
27/// - member count: ULEB128, one byte for up to 127 members;
28/// - each of at most `MULTISIG_COMMITTEE_SIZE_MAX` members: a one-byte scheme tag, the key, and a
29///   `u8` weight. The longest member key is 33 bytes (Secp256k1, Secp256r1 and Passkey; Ed25519 is
30///   32), so `Secp256k1PublicKey::LENGTH` stands for it;
31/// - threshold: `u16`, two bytes.
32///
33/// A new member scheme with a key longer than 33 bytes must update this, or
34/// valid committees using it would be refused.
35const MULTISIG_PUBLIC_KEY_LENGTH_MAX: usize =
36    1 + MULTISIG_COMMITTEE_SIZE_MAX * (1 + Secp256k1PublicKey::LENGTH + 1) + 2;
37
38#[derive(Clone)]
39pub struct MultisigValidatePubkeyCostParams {
40    pub multisig_multisig_validate_pubkey_cost_base: Option<InternalGas>,
41}
42
43/// Implementation of the Move native function
44/// `multisig::multisig_validate_pubkey(public_key: &vector<u8>): bool`
45///
46/// Returns `true` if `public_key` deserializes into a MultiSig committee that
47/// passes `MultisigCommittee::validate()` — 1 to 10 distinct members, every
48/// weight greater than zero, threshold greater than zero, total weight at
49/// least the threshold, and no Ed25519 member together with members of other
50/// schemes — and `false` otherwise (including on deserialization failure or
51/// trailing bytes). Member keys are not checked against their curves,
52/// matching the SDK.
53///
54/// An input longer than `MULTISIG_PUBLIC_KEY_LENGTH_MAX` (353 bytes) cannot
55/// hold a valid committee and returns `false` without being decoded, so the
56/// work done is bounded and the flat cost covers it.
57///
58/// gas cost: multisig_multisig_validate_pubkey_cost_base
59pub fn multisig_validate_pubkey(
60    context: &mut NativeContext,
61    ty_args: Vec<Type>,
62    mut args: VecDeque<Value>,
63) -> PartialVMResult<NativeResult> {
64    debug_assert!(ty_args.is_empty());
65    debug_assert!(args.len() == 1);
66
67    let cost_base = context
68        .extensions()
69        .get::<NativesCostTable>()?
70        .multisig_validate_pubkey_cost_params
71        .multisig_multisig_validate_pubkey_cost_base
72        .ok_or_else(|| {
73            PartialVMError::new(StatusCode::UNKNOWN_INVARIANT_VIOLATION_ERROR)
74                .with_message("Gas cost for multisig_validate_pubkey not available".to_string())
75        })?;
76    native_charge_gas_early_exit!(context, cost_base);
77
78    let public_key = pop_arg!(args, VectorRef);
79    let public_key_ref = public_key.as_bytes_ref();
80
81    let is_valid = validate_pubkey(&public_key_ref).is_ok();
82
83    let cost = context.gas_used();
84    Ok(NativeResult::ok(cost, smallvec![Value::bool(is_valid)]))
85}

87#[derive(Debug, PartialEq, Eq)]
88enum MultisigPublicKeyError {
89    TooLong,
90    Deserialize,
91    InvalidCommittee,
92}
93
94fn validate_pubkey(public_key: &[u8]) -> Result<(), MultisigPublicKeyError> {
95    if public_key.len() > MULTISIG_PUBLIC_KEY_LENGTH_MAX {
96        return Err(MultisigPublicKeyError::TooLong);
97    }
98    // BCS rejects trailing bytes and unsupported member schemes.
99    let committee = bcs::from_bytes::<MultisigCommittee>(public_key)
100        .map_err(|_| MultisigPublicKeyError::Deserialize)?;
101    committee
102        .validate()
103        .map_err(|_| MultisigPublicKeyError::InvalidCommittee)
104}

106#[cfg(test)]
107mod tests {
108    use iota_sdk_types::crypto::MultisigMember;
109
110    use super::*;
111
112    fn committee_bytes(size: usize) -> Vec<u8> {
113        let members = (0..size)
114            .map(|i| {
115                let mut key = [u8::try_from(i).unwrap(); Secp256k1PublicKey::LENGTH];
116                key[0] = 0x02;
117                MultisigMember::new(Secp256k1PublicKey::new(key), 1)
118            })
119            .collect();
120        bcs::to_bytes(&MultisigCommittee::new_unchecked(
121            members,
122            size.try_into().unwrap(),
123        ))
124        .unwrap()
125    }
126
127    #[test]
128    fn largest_valid_committee_is_accepted() {
129        let bytes = committee_bytes(MULTISIG_COMMITTEE_SIZE_MAX);
130        assert_eq!(bytes.len(), 353);
131        assert_eq!(bytes.len(), MULTISIG_PUBLIC_KEY_LENGTH_MAX);
132        assert_eq!(validate_pubkey(&bytes), Ok(()));
133    }
134
135    #[test]
136    fn input_one_byte_over_the_bound_is_refused_before_decoding() {
137        let mut bytes = committee_bytes(MULTISIG_COMMITTEE_SIZE_MAX);
138        bytes.push(0);
139        assert_eq!(
140            validate_pubkey(&bytes),
141            Err(MultisigPublicKeyError::TooLong)
142        );
143    }
144
145    #[test]
146    fn committee_with_too_many_members_is_refused_before_decoding() {
147        let too_many_members = committee_bytes(100);
148        assert_eq!(
149            validate_pubkey(&too_many_members),
150            Err(MultisigPublicKeyError::TooLong)
151        );
152    }
153
154    #[test]
155    fn truncated_input_within_the_bound_fails_to_deserialize() {
156        let mut bytes = committee_bytes(MULTISIG_COMMITTEE_SIZE_MAX);
157        bytes.pop();
158        assert_eq!(
159            validate_pubkey(&bytes),
160            Err(MultisigPublicKeyError::Deserialize)
161        );
162    }
163
164    #[test]
165    fn zero_threshold_within_the_bound_fails_committee_validation() {
166        let mut bytes = committee_bytes(2);
167        let len = bytes.len();
168        bytes[len - 2..].fill(0);
169        assert_eq!(
170            validate_pubkey(&bytes),
171            Err(MultisigPublicKeyError::InvalidCommittee)
172        );
173    }
174}
