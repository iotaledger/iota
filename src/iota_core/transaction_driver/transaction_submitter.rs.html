48    #[instrument(level = "debug", skip_all, err(level = "debug"))]
49    pub(crate) async fn submit_transaction<A>(
50        &self,
51        authority_aggregator: &AuthorityAggregator<A>,
52        client_monitor: &ValidatorClientMonitor,
53        amplification_factor: u64,
54        transaction: Option<TransactionEnvelope>,
55        options: &SubmitTransactionOptions,
56    ) -> Result<(AuthorityName, TxStatusUpdate), TransactionDriverError>
57    where
58        A: AuthorityAPI + Send + Sync + 'static,
59    {
60        let start_time = Instant::now();
61
62        self.metrics
63            .submit_amplification_factor
64            .observe(amplification_factor as f64);
65
66        let mut retrier = RequestRetrier::new(
67            authority_aggregator,
68            client_monitor,
69            &options.allowed_validators,
70            &options.blocked_validators,
71        );
72        retrier.exclude(&options.cut_off_validators);
73
74        let mut retries = 0;
75        let mut request_rpcs = FuturesUnordered::new();
76
77        // This loop terminates when there are enough (f+1) non-retriable errors when
78        // submitting the transaction, or all feasible targets returned errors
79        // or timed out.
80        loop {
81            // Try to fill up to amplification_factor concurrent requests
82            while request_rpcs.len() < amplification_factor as usize {
83                match retrier.next_target() {
84                    Ok((name, client)) => {
85                        let display_name = authority_aggregator.get_display_name(&name);
86                        self.metrics
87                            .validator_selections
88                            .with_label_values(&[display_name.as_str()])
89                            .inc();
90
91                        // Create a future that returns the name and display_name along with the
92                        // result
93                        let submit_fut = self.submit_transaction_once(
94                            client,
95                            &transaction,
96                            options,
97                            client_monitor,
98                            name,
99                            display_name.clone(),
100                        );
101
102                        let wrapped_fut = async move {
103                            let result = submit_fut.await;
104                            (name, display_name, result)
105                        };
106
107                        request_rpcs.push(wrapped_fut);
108                    }
109                    Err(_) if request_rpcs.is_empty() => {
110                        // No more targets and no requests in flight
111                        return Err(TransactionDriverError::Aborted {
112                            submission_non_retriable_errors: aggregate_request_errors(
113                                retrier
114                                    .non_retriable_errors_aggregator
115                                    .status_by_authority(),
116                            ),
117                            submission_retriable_errors: aggregate_request_errors(
118                                retrier.retriable_errors_aggregator.status_by_authority(),
119                            ),
120                            observed_effects_digests: AggregatedEffectsDigests {
121                                digests: Vec::new(),
122                            },
123                        });
124                    }
125                    Err(_) => {
126                        // No more targets but still have requests in flight
127                        break;
128                    }
129                }
130            }
131
132            match request_rpcs.next().await {
133                Some((name, display_name, Ok(result))) => {
134                    self.metrics
135                        .validator_submit_transaction_successes
136                        .with_label_values(&[display_name.as_str()])
137                        .inc();
138                    self.metrics
139                        .submit_transaction_retries
140                        .observe(retries as f64);
141                    let elapsed = start_time.elapsed().as_secs_f64();
142                    self.metrics.submit_transaction_latency.observe(elapsed);
143
144                    return Ok((name, result));
145                }
146                Some((name, display_name, Err(e))) => {
147                    let error_type = e.categorize().into();
148                    self.metrics
149                        .validator_submit_transaction_errors
150                        .with_label_values(&[display_name.as_str(), error_type])
151                        .inc();
152
153                    retries += 1;
154                    retrier.add_error(name, e)?;
155                }
156                None => {
157                    // All requests have been processed.
158                    return Err(TransactionDriverError::Aborted {
159                        submission_non_retriable_errors: aggregate_request_errors(
160                            retrier
161                                .non_retriable_errors_aggregator
162                                .status_by_authority(),
163                        ),
164                        submission_retriable_errors: aggregate_request_errors(
165                            retrier.retriable_errors_aggregator.status_by_authority(),
166                        ),
167                        observed_effects_digests: AggregatedEffectsDigests {
168                            digests: Vec::new(),
169                        },
170                    });
171                }
172            };
173
174            // Yield to prevent this retry loop from starving other tasks.
175            tokio::task::yield_now().await;
176        }
177    }

179    #[instrument(level = "debug", skip_all, err(level = "debug"), ret, fields(validator_display_name = ?display_name))]
180    pub(crate) async fn submit_transaction_once<A>(
181        &self,
182        client: Arc<SafeClient<A>>,
183        transaction: &Option<TransactionEnvelope>,
184        options: &SubmitTransactionOptions,
185        client_monitor: &ValidatorClientMonitor,
186        validator: AuthorityName,
187        display_name: String,
188    ) -> Result<TxStatusUpdate, TransactionRequestError>
189    where
190        A: AuthorityAPI + Send + Sync + 'static,
191    {
192        let feedback_builder =
193            &OperationFeedback::builder(validator, display_name, OperationType::Submit);
194        let submit_start = Instant::now();
195
196        let statuses = timeout(
197            SUBMIT_TRANSACTION_TIMEOUT,
198            client.submit_tx(
199                transaction.clone().into_iter().collect(),
200                options.forwarded_client_addr,
201            ),
202        )
203        .await
204        .map_err(|_| {
205            client_monitor.record_interaction_result(feedback_builder.clone().err_now());
206            TransactionRequestError::TimedOutSubmittingTransaction
207        })?
208        .map_err(|error| {
209            if is_validator_error(error.categorize()) {
210                client_monitor.record_interaction_result(feedback_builder.clone().err_now());
211            }
212            TransactionRequestError::RejectedAtValidator(error)
213        })?;
214
215        let result = statuses
216            .into_iter()
217            .next()
218            .map(|(_digest, update)| update)
219            .unwrap_or(TxStatusUpdate::Rejected {
220                error: IotaError::Unknown("No result returned".to_string()),
221            });
222
223        // Since only one transaction is submitted, it is ok to return error when the
224        // submission is rejected, expired or timed out.
225        match &result {
226            TxStatusUpdate::Rejected { error } => {
227                let err = error.clone();
228                if is_validator_error(err.categorize()) {
229                    client_monitor.record_interaction_result(feedback_builder.clone().err_now());
230                }
231                return Err(TransactionRequestError::RejectedAtValidator(err));
232            }
233            TxStatusUpdate::Expired { epoch } => {
234                return Err(TransactionRequestError::StatusExpired(*epoch));
235            }
236            TxStatusUpdate::TimedOut => {
237                return Err(TransactionRequestError::TimedOutSubmittingTransaction);
238            }
239            _ => {}
240        }
241
242        let latency = submit_start.elapsed();
243        client_monitor.record_interaction_result(feedback_builder.clone().ok_now(latency));
244        Ok(result)
245    }

247    /// Pings `validator` with a submit request that carries no transaction
248    /// and records the round trip with `client_monitor` as a submit
249    /// interaction. The validator answers such a request with an empty
250    /// stream, so an answer within `timeout_duration` is a successful ping
251    /// and its latency sample. The sample measures the submit RPC's round
252    /// trip only: nothing is admitted or submitted to consensus. A validator
253    /// error or no answer in time is recorded as a failed submit, as for a
254    /// transaction.
255    #[instrument(level = "debug", skip_all, err(level = "debug"), fields(validator_display_name = ?display_name))]
256    pub(crate) async fn ping<A>(
257        &self,
258        client: Arc<SafeClient<A>>,
259        client_monitor: &ValidatorClientMonitor,
260        validator: AuthorityName,
261        display_name: String,
262        timeout_duration: Duration,
263    ) -> Result<Duration, IotaError>
264    where
265        A: AuthorityAPI + Send + Sync + 'static,
266    {
267        let feedback_builder =
268            OperationFeedback::builder(validator, display_name, OperationType::Submit);
269        let start = Instant::now();
270        match timeout(timeout_duration, client.submit_tx(Vec::new(), None)).await {
271            Ok(Ok(_)) => {
272                let latency = start.elapsed();
273                client_monitor.record_interaction_result(feedback_builder.ok_now(latency));
274                Ok(latency)
275            }
276            Ok(Err(error)) => {
277                if is_validator_error(error.categorize()) {
278                    client_monitor.record_interaction_result(feedback_builder.err_now());
279                }
280                Err(error)
281            }
282            Err(_) => {
283                client_monitor.record_interaction_result(feedback_builder.err_now());
284                Err(IotaError::Timeout)
285            }
286        }
287    }

290// Whether the failure is caused by the peer validator, as opposed to the user
291// or this node.
292fn is_validator_error(category: ErrorCategory) -> bool {
293    matches!(
294        category,
295        ErrorCategory::Aborted
296            | ErrorCategory::Internal
297            | ErrorCategory::ValidatorOverloaded
298            | ErrorCategory::Unavailable
299    )

314    /// Every validator rejects the transaction for an input object it does
315    /// not have. The submission is rejected once a quorum answered, without
316    /// asking the rest, and the error names the object.
317    #[tokio::test(start_paused = true)]
318    async fn submission_rejected_by_a_quorum_for_a_missing_input_is_final() {
319        let aggregator =
320            AuthorityAggregatorBuilder::from_committee_size(4).build_mock_authority_aggregator();
321        let transaction = iota_types::utils::create_fake_transaction();
322        let rejected = TxStatusUpdate::Rejected {
323            error: IotaError::from(UserInputError::ObjectNotFound {
324                object_id: ObjectId::ZERO,
325                version: Some(Version::from_u64(1)),
326            }),
327        };
328        for client in aggregator.authority_clients.values() {
329            client
330                .authority_client()
331                .stub_submit_tx(Ok(vec![(*transaction.digest(), rejected.clone())]));
332        }
333        let submitter =
334            TransactionSubmitter::new(Arc::new(TransactionDriverMetrics::new_for_tests()));
335        let monitor = ValidatorClientMonitor::new_for_test();
336
337        let error = submitter
338            .submit_transaction(
339                &aggregator,
340                &monitor,
341                1,
342                Some(transaction),
343                &SubmitTransactionOptions::default(),
344            )
345            .await
346            .expect_err("a quorum rejecting for a missing input is final");
347
348        assert!(
349            matches!(error, TransactionDriverError::RejectedByValidators { .. }),
350            "{error}"
351        );
352        assert!(!error.is_submission_retriable(), "{error}");
353        assert!(
354            error
355                .to_string()
356                .contains("Could not find the referenced object"),
357            "{error}"
358        );
359        let asked = aggregator
360            .authority_clients
361            .values()
362            .filter(|client| !client.authority_client().submit_requests().is_empty())
363            .count();
364        assert_eq!(asked, 3);
365    }

367    /// Longer than the mock client's delay.
368    const PING_TIMEOUT: Duration = Duration::from_secs(1);
369
370    struct PingSetup {
371        submitter: TransactionSubmitter,
372        monitor: ValidatorClientMonitor,
373        metrics: Arc<ValidatorClientMetrics>,
374        client: Arc<SafeClient<MockAuthorityApi>>,
375        name: AuthorityName,
376        display_name: String,
377    }
378
379    impl PingSetup {
380        fn new() -> Self {
381            let aggregator = AuthorityAggregatorBuilder::from_committee_size(4)
382                .build_mock_authority_aggregator();
383            let (name, client) = aggregator
384                .authority_clients
385                .iter()
386                .next()
387                .expect("the committee has members");
388            let metrics = Arc::new(ValidatorClientMetrics::new_for_tests());
389            Self {
390                submitter: TransactionSubmitter::new(Arc::new(
391                    TransactionDriverMetrics::new_for_tests(),
392                )),
393                monitor: ValidatorClientMonitor::new(
394                    ValidatorClientMonitorConfig::default(),
395                    metrics.clone(),
396                ),
397                metrics,
398                client: client.clone(),
399                name: *name,
400                display_name: aggregator.get_display_name(name),
401            }
402        }
403
404        async fn ping(&self, timeout_duration: Duration) -> Result<Duration, IotaError> {
405            self.submitter
406                .ping(
407                    self.client.clone(),
408                    &self.monitor,
409                    self.name,
410                    self.display_name.clone(),
411                    timeout_duration,
412                )
413                .await
414        }
415
416        /// The successful and failed submit interactions recorded for the
417        /// pinged validator.
418        fn submit_counts(&self) -> (u64, u64) {
419            self.metrics
420                .operation_counts(&self.display_name, OperationType::Submit)
421        }
422    }

424    /// The ping sends no transaction, and the validator answers it with an
425    /// empty stream, which is a successful ping with a submit latency sample.
426    #[tokio::test(start_paused = true)]
427    async fn ping_answered_with_an_empty_stream_is_a_submit_success() {
428        let setup = PingSetup::new();
429        setup
430            .client
431            .authority_client()
432            .stub_submit_tx(Ok(Vec::new()));
433
434        let latency = setup
435            .ping(PING_TIMEOUT)
436            .await
437            .expect("an answered ping succeeds");
438
439        let requests = setup.client.authority_client().submit_requests();
440        assert_eq!(requests.len(), 1);
441        assert!(requests[0].is_empty());
442        assert!(latency > Duration::ZERO);
443        assert_eq!(setup.submit_counts(), (1, 0));
444    }
