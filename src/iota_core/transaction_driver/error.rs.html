16/// Errors emitted from individual validators during transaction driver
17/// operations.
18///
19/// These errors are associated with the transaction and authority externally,
20/// so it is unnecessary to include those information in these messages.
21///
22/// NOTE: these errors will be aggregated across authorities by status and
23/// reported to the caller. So the error messages should not contain authority
24/// specific information, such as authority name.
25#[derive(Eq, PartialEq, Clone, Debug, Error)]
26pub(crate) enum TransactionRequestError {
27    #[error("Request timed out submitting transaction")]
28    TimedOutSubmittingTransaction,
29    #[error("Request timed out getting full effects")]
30    TimedOutGettingFullEffectsAtValidator,
31    #[error("{0}")]
32    ValidatorInternal(String),
33
34    // Rejected by the validator when voting on the transaction.
35    #[error("{0}")]
36    RejectedAtValidator(IotaError),
37    // Transaction status has been dropped from cache at the validator.
38    #[error("Transaction status expired")]
39    StatusExpired(EpochId),
40    // Request to submit transaction or get full effects failed.
41    #[error("{0}")]
42    Aborted(IotaError),
43}
44
45impl TransactionRequestError {
46    /// Retry delay requested by an overloaded validator.
47    fn retry_after_secs(&self) -> Option<u64> {
48        match self {
49            TransactionRequestError::RejectedAtValidator(error)
50            | TransactionRequestError::Aborted(error) => error
51                .is_retryable_overload()
52                .then(|| error.retry_after_secs()),
53            _ => None,
54        }
55    }
56
57    pub(crate) fn categorize(&self) -> ErrorCategory {
58        match self {
59            TransactionRequestError::TimedOutSubmittingTransaction => ErrorCategory::Unavailable,
60            TransactionRequestError::TimedOutGettingFullEffectsAtValidator => {
61                ErrorCategory::Unavailable
62            }
63            TransactionRequestError::ValidatorInternal(_) => ErrorCategory::Internal,
64
65            TransactionRequestError::RejectedAtValidator(error) => error.categorize(),
66            TransactionRequestError::StatusExpired(_) => ErrorCategory::Aborted,
67            TransactionRequestError::Aborted(error) => error.categorize(),
68        }
69    }
70
71    pub(crate) fn is_submission_retriable(&self) -> bool {
72        self.categorize().is_submission_retriable()
73    }
74}

76/// Client facing errors on transaction processing via Transaction Driver.
77///
78/// NOTE: every error should indicate if it is retriable.
79#[derive(Eq, PartialEq, Clone)]
80pub enum TransactionDriverError {
81    /// TransactionDriver encountered an internal error.
82    /// Non-retriable.
83    ClientInternal { error: String },
84    /// The transaction failed validation from local state.
85    /// Non-retriable.
86    ValidationFailed { error: String },
87    /// Transient failure during transaction processing that prevents the
88    /// transaction from finalization. Retriable with new transaction
89    /// submission.
90    Aborted {
91        submission_non_retriable_errors: AggregatedRequestErrors,
92        submission_retriable_errors: AggregatedRequestErrors,
93        observed_effects_digests: AggregatedEffectsDigests,
94    },
95    /// Over validity threshold of validators rejected the transaction as
96    /// invalid. Non-retriable.
97    RejectedByValidators {
98        submission_non_retriable_errors: AggregatedRequestErrors,
99        submission_retriable_errors: AggregatedRequestErrors,
100    },
101    /// Transaction execution observed multiple effects digests, and it is no
102    /// longer possible to certify any of them.
103    /// Non-retriable.
104    ForkedExecution {
105        observed_effects_digests: AggregatedEffectsDigests,
106        submission_non_retriable_errors: AggregatedRequestErrors,
107        submission_retriable_errors: AggregatedRequestErrors,
108    },
109    /// Transaction timed out but we return last retriable error if it exists.
110    /// Non-retriable.
111    TimeoutWithLastRetriableError {
112        last_error: Option<Box<TransactionDriverError>>,
113        attempts: u32,
114        timeout: Duration,
115    },
116    /// Transaction was successfully submitted to consensus, but the
117    /// subsequent effects fetch from the submitting validator failed. The
118    /// tx is expected to be finalized via the local checkpoint executor; the
119    /// caller may recover by waiting for checkpoint inclusion and reading
120    /// effects from the local cache. Retriable at the client level.
121    SubmittedButFetchFailed {
122        validator: AuthorityName,
123        error: String,
124    },
125}

127impl TransactionDriverError {
128    pub(crate) fn is_submission_retriable(&self) -> bool {
129        self.categorize().is_submission_retriable()
130    }
131
132    /// True when the overloaded stake of an aborted attempt exceeds all
133    /// other error stake combined. Single rule shared by the driver's
134    /// overload backoff and the client-facing overload mapping. `categorize`
135    /// is unsuitable here: it reads only the largest message bucket and must
136    /// keep preferring retriable errors for retriability.
137    pub(crate) fn is_overload_dominated(&self) -> bool {
138        let TransactionDriverError::Aborted {
139            submission_non_retriable_errors,
140            submission_retriable_errors,
141            ..
142        } = self
143        else {
144            return false;
145        };
146        let mut overloaded = 0;
147        let mut other = 0;
148        for (_, _, stake, category) in submission_retriable_errors
149            .errors
150            .iter()
151            .chain(submission_non_retriable_errors.errors.iter())
152        {
153            if *category == ErrorCategory::ValidatorOverloaded {
154                overloaded += *stake;
155            } else {
156                other += *stake;
157            }
158        }
159        overloaded > 0 && overloaded > other
160    }
161
162    pub fn categorize(&self) -> ErrorCategory {
163        match self {
164            TransactionDriverError::ClientInternal { .. } => ErrorCategory::Internal,
165            TransactionDriverError::ValidationFailed { .. } => ErrorCategory::InvalidTransaction,
166            TransactionDriverError::Aborted {
167                submission_retriable_errors,
168                submission_non_retriable_errors,
169                ..
170            } => {
171                if let Some((_, _, _, category)) = submission_retriable_errors.errors.first() {
172                    *category
173                } else if let Some((_, _, _, category)) =
174                    submission_non_retriable_errors.errors.first()
175                {
176                    *category
177                } else {
178                    ErrorCategory::Aborted
179                }
180            }
181            TransactionDriverError::RejectedByValidators {
182                submission_non_retriable_errors,
183                submission_retriable_errors,
184                ..
185            } => {
186                if let Some((_, _, _, category)) = submission_non_retriable_errors.errors.first() {
187                    *category
188                } else if let Some((_, _, _, category)) = submission_retriable_errors.errors.first()
189                {
190                    *category
191                } else {
192                    // There should be at least one error.
193                    ErrorCategory::Internal
194                }
195            }
196            TransactionDriverError::ForkedExecution { .. } => ErrorCategory::Internal,
197            TransactionDriverError::TimeoutWithLastRetriableError { .. } => {
198                ErrorCategory::Unavailable
199            }
200            TransactionDriverError::SubmittedButFetchFailed { .. } => ErrorCategory::Unavailable,
201        }
202    }
203
204    fn display_aborted(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
205        let TransactionDriverError::Aborted {
206            submission_non_retriable_errors,
207            submission_retriable_errors,
208            observed_effects_digests,
209        } = self
210        else {
211            return Ok(());
212        };
213        let mut msgs =
214            vec!["Transaction processing aborted (retriable with another submission).".to_string()];
215        if submission_retriable_errors.total_stake > 0 {
216            msgs.push(format!(
217                "Retriable errors: [{submission_retriable_errors}]."
218            ));
219        }
220        if submission_non_retriable_errors.total_stake > 0 {
221            msgs.push(format!(
222                "Non-retriable errors: [{submission_non_retriable_errors}]."
223            ));
224        }
225        if !observed_effects_digests.digests.is_empty() {
226            msgs.push(format!(
227                "Observed effects digests: [{observed_effects_digests}]."
228            ));
229        }
230        write!(f, "{}", msgs.join(" "))
231    }
232
233    fn display_validation_failed(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
234        let TransactionDriverError::ValidationFailed { error } = self else {
235            return Ok(());
236        };
237        write!(f, "Transaction failed validation: {error}")
238    }
239
240    fn display_invalid_transaction(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
241        let TransactionDriverError::RejectedByValidators {
242            submission_non_retriable_errors,
243            submission_retriable_errors,
244        } = self
245        else {
246            return Ok(());
247        };
248        let mut msgs = vec!["Transaction is rejected as invalid by more than 1/3 of validators by stake (non-retriable).".to_string()];
249        if submission_non_retriable_errors.total_stake > 0 {
250            msgs.push(format!(
251                "Non-retriable errors: [{submission_non_retriable_errors}]."
252            ));
253        }
254        if submission_retriable_errors.total_stake > 0 {
255            msgs.push(format!(
256                "Retriable errors: [{submission_retriable_errors}]."
257            ));
258        }
259        write!(f, "{}", msgs.join(" "))
260    }
261
262    fn display_forked_execution(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
263        let TransactionDriverError::ForkedExecution {
264            observed_effects_digests,
265            submission_non_retriable_errors,
266            submission_retriable_errors,
267        } = self
268        else {
269            return Ok(());
270        };
271        let mut msgs =
272            vec!["Transaction execution observed forked outputs (non-retriable).".to_string()];
273        msgs.push(format!(
274            "Observed effects digests: [{observed_effects_digests}]."
275        ));
276        if submission_non_retriable_errors.total_stake > 0 {
277            msgs.push(format!(
278                "Non-retriable errors: [{submission_non_retriable_errors}]."
279            ));
280        }
281        if submission_retriable_errors.total_stake > 0 {
282            msgs.push(format!(
283                "Retriable errors: [{submission_retriable_errors}]."
284            ));
285        }
286        write!(f, "{}", msgs.join(" "))
287    }
288}

290impl std::fmt::Display for TransactionDriverError {
291    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
292        match self {
293            TransactionDriverError::ClientInternal { error } => {
294                write!(f, "TransactionDriver internal error: {error}")
295            }
296            TransactionDriverError::Aborted { .. } => self.display_aborted(f),
297            TransactionDriverError::ValidationFailed { .. } => self.display_validation_failed(f),
298            TransactionDriverError::RejectedByValidators { .. } => {
299                self.display_invalid_transaction(f)
300            }
301            TransactionDriverError::ForkedExecution { .. } => self.display_forked_execution(f),
302            TransactionDriverError::TimeoutWithLastRetriableError {
303                last_error,
304                attempts,
305                timeout,
306            } => {
307                write!(
308                    f,
309                    "Transaction timed out after {} attempts. Timeout: {:?}. Last error: {}",
310                    attempts,
311                    timeout,
312                    last_error
313                        .as_ref()
314                        .map(|e| e.to_string())
315                        .unwrap_or_default()
316                )
317            }
318            TransactionDriverError::SubmittedButFetchFailed { validator, error } => {
319                write!(
320                    f,
321                    "Transaction submitted to consensus but failed to fetch effects from submitter {:?}: {error}",
322                    validator.concise()
323                )
324            }
325        }
326    }
327}

337#[derive(Eq, PartialEq, Clone, Debug, Default)]
338pub struct AggregatedRequestErrors {
339    pub errors: Vec<(String, Vec<AuthorityName>, StakeUnit, ErrorCategory)>,
340    // The total stake of all errors.
341    pub total_stake: StakeUnit,
342    /// Stake per requested retry delay in seconds. Mirrors the aggregator's
343    /// `RetryableOverloadInfo`.
344    pub stake_requested_retry_after: BTreeMap<u64, StakeUnit>,
345}
346
347impl AggregatedRequestErrors {
348    /// Upper stake-weighted median of the requested retry delays over all
349    /// overloaded stake. Returns `None` when no delay is backed by a
350    /// majority of that stake.
351    pub fn median_retry_after_secs(&self) -> Option<u64> {
352        let overloaded_stake: StakeUnit = self
353            .errors
354            .iter()
355            .filter(|(_, _, _, category)| *category == ErrorCategory::ValidatorOverloaded)
356            .map(|(_, _, stake, _)| *stake)
357            .sum();
358        if overloaded_stake == 0 {
359            return None;
360        }
361        let mut cumulative = 0;
362        for (secs, stake) in &self.stake_requested_retry_after {
363            cumulative += stake;
364            if cumulative * 2 > overloaded_stake {
365                return Some(*secs);
366            }
367        }
368        None
369    }
370}

372impl std::fmt::Display for AggregatedRequestErrors {
373    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
374        let msg = self
375            .errors
376            .iter()
377            .map(|(error, names, stake, _category)| {
378                format!(
379                    "{} {{ {} }} with {} stake",
380                    error,
381                    names.iter().map(|n| n.concise_owned()).join(", "),
382                    stake
383                )
384            })
385            .join("; ");
386        write!(f, "{msg}")?;
387        Ok(())
388    }

391fn format_transaction_request_error(error: &TransactionRequestError) -> String {
392    match error {
393        TransactionRequestError::RejectedAtValidator(iota_error) => match iota_error {
394            IotaError::UserInput { error: user_error } => user_error.to_string(),
395            _ => iota_error.to_string(),
396        },
397        _ => error.to_string(),
398    }
399}

401pub(crate) fn aggregate_request_errors(
402    errors: Vec<(AuthorityName, StakeUnit, TransactionRequestError)>,
403) -> AggregatedRequestErrors {
404    let mut total_stake = 0;
405    let mut aggregated_errors =
406        BTreeMap::<String, (Vec<AuthorityName>, StakeUnit, ErrorCategory)>::new();
407    let mut stake_requested_retry_after = BTreeMap::new();
408
409    for (name, stake, error) in errors {
410        total_stake += stake;
411        if let Some(secs) = error.retry_after_secs() {
412            *stake_requested_retry_after.entry(secs).or_default() += stake;
413        }
414        let key = format_transaction_request_error(&error);
415        let entry = aggregated_errors
416            .entry(key)
417            .or_insert_with(|| (vec![], 0, error.categorize()));
418        entry.0.push(name);
419        entry.1 += stake;
420    }
421
422    let mut errors: Vec<_> = aggregated_errors
423        .into_iter()
424        .map(|(error, (names, stake, category))| (error, names, stake, category))
425        .collect();
426    errors.sort_by_key(|(_, _, stake, _)| std::cmp::Reverse(*stake));
427
428    AggregatedRequestErrors {
429        errors,
430        total_stake,
431        stake_requested_retry_after,
432    }
433}

472    /// Aggregation records stake per requested delay. The median ignores
473    /// low-stake outliers.
474    #[test]
475    fn aggregation_records_stake_per_retry_hint() {
476        use fastcrypto::traits::VerifyingKey;
477
478        let name = |i: u8| AuthorityPublicKeyBytes([i; AuthorityPublicKey::LENGTH]);
479        let overloaded = |secs| {
480            TransactionRequestError::RejectedAtValidator(IotaError::ValidatorOverloadedRetryAfter {
481                retry_after_secs: secs,
482            })
483        };
484        let aggregated = aggregate_request_errors(vec![
485            (name(0), 4000, overloaded(10)),
486            (name(1), 3000, overloaded(10)),
487            (name(2), 1000, overloaded(3600)),
488            (
489                name(3),
490                1000,
491                TransactionRequestError::TimedOutSubmittingTransaction,
492            ),
493        ]);
494        assert_eq!(
495            aggregated.stake_requested_retry_after,
496            BTreeMap::from([(10, 7000), (3600, 1000)])
497        );
498        assert_eq!(aggregated.total_stake, 9000);
499        assert_eq!(aggregated.median_retry_after_secs(), Some(10));
500
501        let aggregated = aggregate_request_errors(vec![(
502            name(0),
503            1000,
504            TransactionRequestError::TimedOutSubmittingTransaction,
505        )]);
506        assert!(aggregated.stake_requested_retry_after.is_empty());
507        assert_eq!(aggregated.median_retry_after_secs(), None);
508    }

510    /// Dominance requires overloaded stake to exceed all other error stake
511    /// combined.
512    #[test]
513    fn overload_dominance_sums_stake_per_category() {
514        use fastcrypto::traits::VerifyingKey;
515
516        let name = |i: u8| AuthorityPublicKeyBytes([i; AuthorityPublicKey::LENGTH]);
517        let overloaded = |secs| {
518            TransactionRequestError::RejectedAtValidator(IotaError::ValidatorOverloadedRetryAfter {
519                retry_after_secs: secs,
520            })
521        };
522        let aborted = |retriable: Vec<_>| TransactionDriverError::Aborted {
523            submission_non_retriable_errors: AggregatedRequestErrors::default(),
524            submission_retriable_errors: aggregate_request_errors(retriable),
525            observed_effects_digests: AggregatedEffectsDigests { digests: vec![] },
526        };
527
528        // Overload split across delays still outweighs a larger single
529        // bucket.
530        let dominated = aborted(vec![
531            (
532                name(0),
533                4000,
534                TransactionRequestError::TimedOutSubmittingTransaction,
535            ),
536            (name(1), 2500, overloaded(10)),
537            (name(2), 2500, overloaded(30)),
538        ]);
539        assert!(dominated.is_overload_dominated());
540
541        // An overload plurality that is an aggregate minority does not
542        // dominate.
543        let not_dominated = aborted(vec![
544            (name(0), 3500, overloaded(10)),
545            (
546                name(1),
547                3000,
548                TransactionRequestError::TimedOutSubmittingTransaction,
549            ),
550            (
551                name(2),
552                2500,
553                TransactionRequestError::TimedOutGettingFullEffectsAtValidator,
554            ),
555        ]);
556        assert!(!not_dominated.is_overload_dominated());
557    }
