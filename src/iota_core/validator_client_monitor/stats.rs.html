15/// Ok(latency in sec) or Err(high failure latency in sec)
16type Observation = Result<f64, f64>;
17
18/// Exponential weighted moving average (EWMA) estimator.
19///
20/// Each new observation is weighted by α and the prior estimate by (1 − α):
21/// μ_t ​= α x_t​ + (1−α) μ_{t−1}​
22/// σ_t^2​ = (1−α) σ_{t−1}^2 ​+ α (x_t​−μ_{t−1}​)^2
23#[derive(Clone, Copy, Debug)]
24struct Ewma {
25    /// Current mean estimate: μ_t.
26    mean: f64,
27    /// Current variance estimate: σ_t^2.
28    variance: f64,
29    /// Failure mean estimate.
30    failure: f64,
31    /// Decayed/effective sample size.
32    sample_size: f64,
33}
34
35impl Ewma {
36    fn first_value(observation: Observation) -> Self {
37        Self {
38            mean: observation.unwrap_or_else(|failure| failure),
39            variance: 0.0,
40            failure: observation.map_or(1.0, |_| 0.0),
41            sample_size: 1.0,
42        }
43    }
44
45    fn stddev(&self) -> f64 {
46        self.variance.sqrt()
47    }
48
49    fn update(&mut self, observation: Observation, alpha: f64) {
50        let a1 = 1.0 - alpha;
51        // treat failures as high latency
52        let x = observation.unwrap_or_else(|failure| failure);
53        // μ_t ​= α x_t​ + (1−α) μ_{t−1}​ = μ_{t−1}​ + α (x_t​ - μ_{t−1})
54        let delta = x - self.mean;
55        let ad = alpha * delta;
56        self.mean += ad;
57        // σ_t^2​ = (1−α) σ_{t−1}^2 ​+ α (x_t​−μ_{t−1}​)^2
58        self.variance = a1 * self.variance + ad * delta;
59        // failures EWMA
60        self.failure = a1 * self.failure + observation.map_or(alpha, |_| 0.0);
61        // effective sample size is just EWMA of the observation indicators
62        // w_t = (1 - α) w_{t-1} + α
63        self.sample_size = a1 * self.sample_size + alpha;
64    }
65
66    fn score(&self, k: f64) -> f64 {
67        // Score = mean + k * stddev. Higher k means more penalty for
68        // variability and less confidence in the estimate.
69        self.mean + k * self.stddev()
70    }
71
72    fn get_stats(&self, k: f64) -> (f64, f64, f64) {
73        (self.score(k), self.sample_size, self.failure)
74    }
75}

77/// Time-decayed EWMA-based estimator.
78///
79/// This is a EWMA estimator with a variable weight α_t (interval between
80/// updates): α_t = 1 - exp(-Δt / τ)
81#[derive(Clone, Copy, Debug)]
82struct TimeDecayEwma {
83    /// Base EWMA estimator.
84    ewma: Ewma,
85    /// Timestamp of the last update: t.
86    last_update: Instant,
87    /// EWMA of normalized observations interval (Δt/τ).
88    /// None until the second observation.
89    interval_ewma: Option<f64>,
90}
91
92impl TimeDecayEwma {
93    fn first_value(observation: Observation, now: Instant) -> Self {
94        Self {
95            ewma: Ewma::first_value(observation),
96            last_update: now,
97            interval_ewma: None,
98        }
99    }
100
101    /// `now` is not guaranteed to be at or after [`Self::last_update`]:
102    /// feedback is stamped with `Instant::now()` before its recorder contends
103    /// for the stats lock, so two recorders reporting on the same validator can
104    /// reach the stats in the opposite order to their timestamps. A late sample
105    /// saturates to Δt ≈ 0, which folds it in with a negligible weight rather
106    /// than letting it rewrite the estimate.
107    fn interval_alpha(&self, tau: f64, now: Instant) -> (f64, f64) {
108        // α_t = 1 - exp(-Δt / τ)
109        // avoid zero Δt, otherwise (α_t = 0) observation won't be updated
110        let dt = now
111            .saturating_duration_since(self.last_update)
112            .as_secs_f64()
113            .max(1e-9);
114        let interval = dt / tau;
115        (interval, 1.0 - (-interval).exp())
116    }
117
118    fn update_with_time_decay(&mut self, tau: f64, now: Instant, observation: Observation) {
119        let (interval, alpha) = self.interval_alpha(tau, now);
120        // Update the value EWMA with time-derived alpha.
121        self.ewma.update(observation, alpha);
122        // Never move the clock backwards: a sample that arrived late must not
123        // make the samples that already landed look newer than they are.
124        self.last_update = self.last_update.max(now);
125        // Update the interval EWMA with a fixed weight of 0.1.
126        const ALPHA: f64 = 0.1;
127        self.interval_ewma = Some(
128            self.interval_ewma
129                .map_or(interval, |prev| (1.0 - ALPHA) * prev + ALPHA * interval),
130        );
131    }
132
133    fn get_stats(&self, k: f64, tau: f64, now: Instant) -> (f64, f64, f64, f64, Option<f64>) {
134        let (score, sample_size, failure) = self.ewma.get_stats(k);
135        (
136            score,
137            sample_size,
138            failure,
139            self.interval_alpha(tau, now).1,
140            self.interval_ewma,
141        )
142    }
143}

145/// Latency estimator is based on time-decayed EWMA.
146#[derive(Clone, Copy, Debug, Default)]
147struct LatencyEwma {
148    inner: Option<TimeDecayEwma>,
149}
150
151impl LatencyEwma {
152    fn new() -> Self {
153        Self { inner: None }
154    }
155
156    fn update(&mut self, tau: f64, timestamp: Instant, observation: Observation) {
157        if let Some(inner) = &mut self.inner {
158            inner.update_with_time_decay(tau, timestamp, observation);
159        } else {
160            self.inner = Some(TimeDecayEwma::first_value(observation, timestamp));
161        }
162    }
163
164    fn get_stats(
165        &self,
166        k: f64,
167        tau: f64,
168        now: Instant,
169    ) -> Option<(f64, f64, f64, f64, Option<f64>)> {
170        self.inner
171            .map(|time_decay_ewma| time_decay_ewma.get_stats(k, tau, now))
172    }
173}

175/// Latency estimator with logarithmic observations.
176///
177/// Logarithmic observations help smooth out high variability in latency
178/// measurements. For regular observations LatencyEwma can be used.
179#[derive(Clone, Copy, Debug, Default)]
180struct LogLatencyEwma {
181    inner: LatencyEwma,
182}
183
184impl LogLatencyEwma {
185    fn new() -> Self {
186        Self {
187            inner: LatencyEwma::new(),
188        }
189    }
190
191    /// Update the estimator with a new observation.
192    ///
193    /// Tau argument should match the expected interval between observations.
194    /// A failed observation is still used to update the log latency average.
195    /// Thus the failure value is treated as a large latency value (eg.
196    /// operation timeout).
197    fn update(&mut self, tau: f64, timestamp: Instant, observation: Observation) {
198        // Avoid ln(0) by capping at a small positive value.
199        let log_observation = observation
200            .map(|x| x.max(1e-9).ln())
201            .map_err(|x| x.max(1e-9).ln());
202        self.inner.update(tau, timestamp, log_observation);
203    }
204
205    fn get_stats(
206        &self,
207        k: f64,
208        tau: f64,
209        now: Instant,
210    ) -> Option<(f64, f64, f64, f64, Option<f64>)> {
211        self.inner.get_stats(k, tau, now).map(
212            |(score, sample_size, failure, alpha, interval_ewma)| {
213                (score.exp(), sample_size, failure, alpha, interval_ewma)
214            },
215        )
216    }
217}

229/// Client-observed stats for a single validator.
230#[derive(Debug, Clone)]
231struct ValidatorClientStats {
232    /// Health-check operation stats.
233    stats_health_check: LogLatencyEwma,
234    /// Submit operation stats.
235    stats_submit: LogLatencyEwma,
236    /// Effects operation stats.
237    stats_effects: LogLatencyEwma,
238    /// Consensus operation stats.
239    stats_consensus: LogLatencyEwma,
240}
241
242/// The reference interval for Submit, Effects, and Consensus operations.
243const TAU: f64 = 60.0;
244
245/// Failure penalty (max latency or timeout in seconds) for Submit operation.
246const FAILURE_SUBMIT: f64 = 5.0;
247
248/// Failure penalty (max latency or timeout in seconds) for Effects operation.
249const FAILURE_EFFECTS: f64 = 10.0;
250
251/// Failure penalty (max latency or timeout in seconds) for Consensus operation.
252const FAILURE_CONSENSUS: f64 = FAILURE_SUBMIT + FAILURE_EFFECTS;
253
254/// Default value for maximum weighted interval (in seconds) in case there is
255/// not enough observations. No observations in practice is equivalent to the
256/// last observation being very old.
257const MAX_INTERVAL: f64 = 3600.0;
258
259/// Penalty coefficient k used in latency score calculation: latency_score =
260/// mean + k * stddev. Higher k means more penalty for variability and less
261/// confidence in the estimate. Health-check operation is the most stable.
262const VARIANCE_PENALTY_HEALTH_CHECK: f64 = 0.5;
263
264/// Penalty coefficient k used in latency score calculation: latency_score =
265/// mean + k * stddev. Higher k means more penalty for variability and less
266/// confidence in the estimate. Submit operation is less stable.
267const VARIANCE_PENALTY_SUBMIT: f64 = 2.0;
268
269/// Penalty coefficient k used in latency score calculation: latency_score =
270/// mean + k * stddev. Higher k means more penalty for variability and less
271/// confidence in the estimate. Effects operation is less stable.
272const VARIANCE_PENALTY_EFFECTS: f64 = 3.0;
273
274/// Penalty coefficient k used in latency score calculation: latency_score =
275/// mean + k * stddev. Higher k means more penalty for variability and less
276/// confidence in the estimate. Consensus operation is the least stable.
277const VARIANCE_PENALTY_CONSENSUS: f64 = 4.0;
278
279/// Default latency score (in seconds) for missing observations.
280/// Health-check operation is expected to have the lowest latency.
281const EMPTY_SCORE_HEALTH_CHECK: f64 = 0.5;
282
283/// Default latency score (in seconds) for missing observations.
284/// Submit operation is expected to have lower latency.
285const EMPTY_SCORE_SUBMIT: f64 = 2.0;
286
287/// Default latency score (in seconds) for missing observations.
288/// Effects operation is expected to have higher latency.
289const EMPTY_SCORE_EFFECTS: f64 = 3.0;
290
291/// Default latency score (in seconds) for missing observations.
292/// Consensus operation is expected to have the highest latency.
293const EMPTY_SCORE_CONSENSUS: f64 = 4.0;
294
295/// Penalty score (in seconds) for failed operations.
296const FAILURE_SCORE: f64 = 10.0;
297
298impl ValidatorClientStats {
299    fn new() -> Self {
300        Self {
301            stats_health_check: LogLatencyEwma::new(),
302            stats_submit: LogLatencyEwma::new(),
303            stats_effects: LogLatencyEwma::new(),
304            stats_consensus: LogLatencyEwma::new(),
305        }
306    }
307
308    fn record_interaction_result(
309        &mut self,
310        config: &ValidatorClientMonitorConfig,
311        feedback: &OperationFeedback,
312    ) {
313        let observation = feedback.result.map(|latency| latency.as_secs_f64());
314        match feedback.operation {
315            OperationType::HealthCheck => {
316                let tau_hc = config.health_check_interval.as_secs_f64();
317                let failure_hc = config.health_check_timeout.as_secs_f64() * 4.0;
318                self.stats_health_check.update(
319                    tau_hc,
320                    feedback.timestamp,
321                    observation.map_err(|()| failure_hc),
322                );
323            }
324            OperationType::Submit => {
325                self.stats_submit.update(
326                    TAU,
327                    feedback.timestamp,
328                    observation.map_err(|()| FAILURE_SUBMIT),
329                );
330            }
331            OperationType::Effects => {
332                self.stats_effects.update(
333                    TAU,
334                    feedback.timestamp,
335                    observation.map_err(|()| FAILURE_EFFECTS),
336                );
337            }
338            OperationType::Consensus => {
339                self.stats_consensus.update(
340                    TAU,
341                    feedback.timestamp,
342                    observation.map_err(|()| FAILURE_CONSENSUS),
343                );
344            }
345        }
346    }
347
348    fn performance_score(&self, config: &ValidatorClientMonitorConfig, now: Instant) -> (f64, f64) {
349        let tau_hc = config.health_check_interval.as_secs_f64();
350
351        let (score_hc, _sample_size_hc, failures_hc, _alpha_hc, _interval_hc) = self
352            .stats_health_check
353            .get_stats(VARIANCE_PENALTY_HEALTH_CHECK, tau_hc, now)
354            .unwrap_or((EMPTY_SCORE_HEALTH_CHECK, 0.0, 0.0, 1.0, None));
355        let (score_sub, sample_size_sub, failures_sub, alpha_sub, interval_sub) = self
356            .stats_submit
357            .get_stats(VARIANCE_PENALTY_SUBMIT, TAU, now)
358            .unwrap_or((EMPTY_SCORE_SUBMIT, 0.0, 0.0, 1.0, None));
359        let (score_eff, sample_size_eff, failures_eff, alpha_eff, interval_eff) = self
360            .stats_effects
361            .get_stats(VARIANCE_PENALTY_EFFECTS, TAU, now)
362            .unwrap_or((EMPTY_SCORE_EFFECTS, 0.0, 0.0, 1.0, None));
363        let (score_con, sample_size_con, failures_con, alpha_con, interval_con) = self
364            .stats_consensus
365            .get_stats(VARIANCE_PENALTY_CONSENSUS, TAU, now)
366            .unwrap_or((EMPTY_SCORE_CONSENSUS, 0.0, 0.0, 1.0, None));
367
368        let score = score_hc + score_sub + score_eff + score_con;
369        let sample_size = sample_size_sub.max(sample_size_eff).max(sample_size_con);
370        let failures = failures_hc + failures_sub + failures_eff + failures_con;
371        let recency = alpha_sub.max(alpha_eff).max(alpha_con);
372        // max weighted interval over Submit, Effects, and Consensus operations
373        let interval = interval_sub
374            .unwrap_or(MAX_INTERVAL)
375            .max(interval_eff.unwrap_or(MAX_INTERVAL))
376            .max(interval_con.unwrap_or(MAX_INTERVAL));
377
378        // good performance means:
379        // - small score (latency)
380        // - large sample size
381        // - small failures
382        // - small recency (of the latest observation) (or close to 1-1/e if
383        //   observations interval is close to tau)
384        // - small interval (frequent observations)
385        // exploitation score is constructed such that lower value means better
386        // performance
387        let exploitation = (score + failures * FAILURE_SCORE)
388            * (1.0 + recency)
389            * ((interval + 1e-2) / (sample_size + 1e-2)).sqrt();
390
391        // new validator or stale stats means:
392        // - any score (latency)
393        // - small sample size
394        // - small failures
395        // - recency close to 1
396        // - large interval
397        // exploration score is constructed such that lower value means less info about
398        // the validator
399        let exploration =
400            failures * (1.0 - recency) * ((sample_size + 1e-2) / (interval + 1e-2)).sqrt();
401
402        (exploitation, exploration)
403    }
404}

406impl ClientObservedStats {
407    pub(super) fn new(config: ValidatorClientMonitorConfig) -> Self {
408        Self {
409            validator_stats: HashMap::new(),
410            config,
411        }
412    }
413
414    /// Record client-observed interaction result with a validator.
415    pub(super) fn record_interaction_result(&mut self, feedback: &OperationFeedback) -> (f64, f64) {
416        let validator_stats = self
417            .validator_stats
418            .entry(feedback.authority_name)
419            .or_insert_with(ValidatorClientStats::new);
420        validator_stats.record_interaction_result(&self.config, feedback);
421        validator_stats.performance_score(&self.config, feedback.timestamp)
422    }
423
424    pub(super) fn select_shuffled_preferred_validators<'a>(
425        &self,
426        committee: impl Iterator<Item = &'a AuthorityName>,
427        now: Instant,
428        mut rng: impl rand::Rng,
429    ) -> Vec<&'a AuthorityName> {
430        const UNKNOWN_VALIDATOR_SCORE: (f64, f64) = (1e6, 0.0);
431        // rate committee validators
432        let mut scored_committee: Vec<_> = committee
433            .map(|v| {
434                (
435                    v,
436                    self.validator_stats
437                        .get(v)
438                        .map(|stats| stats.performance_score(&self.config, now))
439                        .unwrap_or(UNKNOWN_VALIDATOR_SCORE),
440                )
441            })
442            .collect();
443        // order by exploitation score, best performing first
444        scored_committee.sort_by(|(_, (e1, _)), (_, (e2, _))| {
445            e1.partial_cmp(e2).unwrap_or(std::cmp::Ordering::Equal)
446        });
447        // select exploitation group
448        let exploitation_group_size =
449            (self.config.exploitation_group_share.min(100) * scored_committee.len()).div_ceil(100);
450        // order the rest by exploration score, new validators first followed by most
451        // outdated ones
452        scored_committee[exploitation_group_size..].sort_by(|(_, (_, e1)), (_, (_, e2))| {
453            e1.partial_cmp(e2).unwrap_or(std::cmp::Ordering::Equal)
454        });
455        // select exploration group
456        let exploration_group_size = (self.config.exploration_group_share.min(100)
457            * scored_committee.len())
458        .div_ceil(100)
459        .min(scored_committee.len() - exploitation_group_size);
460        // shuffle the two groups together to avoid overloading the best performing
461        // validators and guarantee uniform exploration
462        scored_committee[..exploitation_group_size + exploration_group_size].shuffle(&mut rng);
463        // order the rest by combined score
464        scored_committee[exploitation_group_size + exploration_group_size..].sort_by(
465            |(_, e1), (_, e2)| {
466                (e1.0 + e1.1)
467                    .partial_cmp(&(e2.0 + e2.1))
468                    .unwrap_or(std::cmp::Ordering::Equal)
469            },
470        );
471
472        scored_committee.into_iter().map(|(v, _)| v).collect()
473    }

475    /// Retain only the specified validators, removing any others.
476    pub(super) fn retain_validators<'a>(
477        &mut self,
478        validators: impl Iterator<Item = &'a AuthorityName>,
479    ) {
480        let cur_len = self.validator_stats.len();
481        let validator_set: HashSet<_> = validators.collect();
482        self.validator_stats
483            .retain(|validator, _| validator_set.contains(validator));
484        let removed_count = cur_len - self.validator_stats.len();
485        if removed_count > 0 {
486            debug!("Removed {} stale validator data", removed_count);
487        }
488    }
489
490    #[cfg(test)]
491    pub(super) fn has_validator(&self, validator: &AuthorityName) -> bool {
492        self.validator_stats.contains_key(validator)
493    }
494
495    #[cfg(test)]
496    pub(super) fn num_validators(&self) -> usize {
497        self.validator_stats.len()
498    }
