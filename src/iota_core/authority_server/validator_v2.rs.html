126/// Admits a transaction with a transaction dependency input only with the
127/// local evidence post-consensus validation requires of its dependency: a
128/// root or deferred at a commit this validator processed, or submitted to
129/// consensus through this validator and still in flight. A dependency known
130/// to have failed or to have run in an earlier epoch, or recorded as dropped,
131/// is rejected for good. Any other is rejected with a retriable error, also
132/// one that executed with success without being a root, and one recorded as
133/// dropped by the deny rules or the coin deny list: either can change in the
134/// epoch, so the dependency can be kept when it is submitted again. Local by
135/// design: post-consensus validation decides the same way on every
136/// validator.
137fn check_transaction_dependency_for_submission(
138    state: &AuthorityState,
139    epoch_store: &AuthorityPerEpochStore,
140    soft_locks: &PreConsensusSoftLocks,
141    transaction: &VerifiedTransaction,
142) -> IotaResult<()> {
143    let transaction = transaction.data().transaction();
144    let Some(digest) = transaction.transaction_dependency() else {
145        return Ok(());
146    };
147    state.check_transaction_dependency_known_outcome(transaction, epoch_store.epoch())?;
148    if epoch_store.root_commit_of(&digest)?.is_some()
149        || epoch_store.is_deferred_transaction(&digest)
150        || epoch_store.is_pending_consensus_user_transaction(&digest)?
151        || soft_locks.check_in_flight(&digest)
152    {
153        return Ok(());
154    }
155    if epoch_store
156        .recorded_dropped_transaction(&digest)
157        .is_some_and(|error| !is_denial(&error))
158    {
159        return Err(UserInputError::TransactionDependencyFailed { digest }.into());
160    }
161    Err(UserInputError::TransactionDependencyNotFound { digest }.into())
162}

164/// Whether `error` is a denial by the deny rules or the coin deny list.
165fn is_denial(error: &IotaError) -> bool {
166    matches!(
167        error,
168        IotaError::UserInput {
169            error: UserInputError::TransactionDenied { .. }
170                | UserInputError::AddressDeniedForCoin { .. }
171                | UserInputError::CoinTypeGlobalPause { .. },
172        }
173    )
174}

176/// Whether admission checks the transaction's inputs against local state: that
177/// they load at the declared versions, pass the deny checks and get their soft
178/// locks.
179///
180/// Tests can register the `admit-without-input-checks` fail point to have a
181/// validator submit to consensus a transaction it would otherwise reject, as a
182/// faulty validator can.
183fn checks_admitted_inputs() -> bool {
184    let checks = std::cell::Cell::new(true);
185    iota_macros::fail_point_if!("admit-without-input-checks", || checks.set(false));
186    checks.get()
187}

200/// Returns the attestation of `transaction` from a validator that skips its
201/// input checks, see [`checks_admitted_inputs()`], while validator attestation
202/// is on. Without a dry run, it records no object versions, and it claims the
203/// most units the transaction can pay for, the upper bound that post-consensus
204/// validation allows. Post-consensus validation drops the transaction when a
205/// Move authenticator of `transaction` names a shared object, such as a
206/// shared account, since this record has no entry for it. Only tests reach
207/// it, by registering the `admit-without-input-checks` fail point.
208fn attest_without_dry_run(transaction: &VerifiedTransaction) -> AttestationData {
209    let transaction = transaction.data().transaction();
210
211    AttestationData::V1 {
212        computation_units: transaction
213            .gas_budget()
214            .checked_div(transaction.gas_price())
215            .unwrap_or(u64::MAX),
216        shared_object_versions: Vec::new(),
217    }
218}

220/// Whether the validator admits new user transactions in `reconfig_state`. It
221/// stops once its epoch starts closing.
222///
223/// Tests can register the `admit-user-transactions-after-epoch-close` fail
224/// point to keep a validator admitting them, and submitting them to consensus,
225/// after that point.
226fn admits_user_transactions(reconfig_state: &ReconfigState) -> bool {
227    // A `Cell` keeps the binding immutable in builds where the fail point
228    // compiles to nothing.
229    let admits = std::cell::Cell::new(reconfig_state.should_accept_user_certs());
230    iota_macros::fail_point_if!("admit-user-transactions-after-epoch-close", || {
231        admits.set(true)
232    });
233    admits.get()
234}

236/// Checks that `consensus_tx` is no larger than consensus accepts: its BCS size
237/// against [`ProtocolConfig::max_transaction_size_bytes()`], above which
238/// Starfish refuses a submitted transaction.
239fn check_consensus_transaction_size(
240    consensus_tx: &ConsensusTransaction,
241    protocol_config: &ProtocolConfig,
242) -> IotaResult {
243    let size = bcs::serialized_size(consensus_tx).map_err(|error| {
244        IotaError::TransactionSerialization {
245            error: error.to_string(),
246        }
247    })?;
248
249    BlockTransactionLimits::new(protocol_config)
250        .check_transaction_size(size as u64)
251        .map_err(|error| to_iota_error(error.into()))
252}

254/// Rejects a submission because the validator is overloaded, and counts it in
255/// [`ValidatorServiceMetrics::num_rejected_tx_during_overload`]. The
256/// rejection weighs nothing in the traffic controller's tally: the client
257/// turned away is not necessarily the one keeping the validator busy.
258fn reject_for_overload(
259    metrics: &ValidatorServiceMetrics,
260    error: IotaError,
261) -> (TxStatusUpdate, Weight) {
262    metrics
263        .num_rejected_tx_during_overload
264        .with_label_values(&[error.as_ref()])
265        .inc();
266
267    (TxStatusUpdate::Rejected { error }, Weight::zero())
268}

270/// Counts a soft lock refusal in
271/// [`ValidatorServiceMetrics::num_rejected_tx_recently_resubmitted`] when the
272/// transaction is in flight, and in
273/// [`ValidatorServiceMetrics::num_rejected_tx_soft_lock_conflict`] when
274/// another transaction holds a lock on one of its objects. Other errors are
275/// not counted.
276fn count_soft_lock_refusal(metrics: &ValidatorServiceMetrics, error: &IotaError) {
277    match error {
278        IotaError::RecentlyResubmitted { .. } => {
279            metrics.num_rejected_tx_recently_resubmitted.inc();
280        }
281        IotaError::ObjectLockConflict { .. } => {
282            metrics.num_rejected_tx_soft_lock_conflict.inc();
283        }
284        _ => {}
285    }
286}

288/// Lists the owned and immutable objects `transaction` names as inputs, gas
289/// coins included. It reads the transaction alone, before validation, so it
290/// cannot distinguish owned inputs from immutable ones. A check for soft lock
291/// conflicts can use the list anyway: soft locks are taken only on the owned
292/// objects that validation finds once it loads the inputs, so an immutable
293/// object has none. A transaction whose inputs cannot be listed gets an empty
294/// list, and validation refuses it later.
295fn list_owned_or_immutable_inputs(transaction: &TransactionEnvelope) -> Vec<ObjectReference> {
296    transaction
297        .data()
298        .input_objects(false)
299        .unwrap_or_default()
300        .into_iter()
301        .filter_map(|input| match input {
302            InputObjectKind::ImmOrOwnedMoveObject(object_ref) => Some(object_ref),
303            InputObjectKind::SharedMoveObject { .. } | InputObjectKind::MovePackage(_) => None,
304        })
305        .collect()
306}

308/// Returns the traffic weight of a rejection the sender causes:
309/// [`Weight::one()`] after the attestation's dry run, which the validator
310/// then ran for nothing, and otherwise what [`normalize()`] gives `error`.
311fn weigh_sender_rejection(error: &IotaError, after_dry_run: bool) -> Weight {
312    if after_dry_run {
313        Weight::one()
314    } else {
315        normalize(error)
316    }
317}

319/// Returns the traffic weight of a rejection by
320/// [`AuthorityState::attest_transaction()`] or
321/// [`AuthorityState::handle_transaction_validation_checks()`]:
322/// [`Weight::one()`] for `TransactionDependencyNotFound`, and otherwise what
323/// [`normalize()`] gives `error`. The two return that error only for a Move
324/// authentication that failed while the transaction's dependency has not
325/// executed here, so the validator ran the authenticators for nothing.
326fn weigh_validation_rejection(error: &IotaError) -> Weight {
327    if matches!(
328        error,
329        IotaError::UserInput {
330            error: UserInputError::TransactionDependencyNotFound { .. },
331        }
332    ) {
333        Weight::one()
334    } else {
335        normalize(error)
336    }
337}
338
339/// Returns the traffic weight of a rejection by
340/// [`ExecutionCacheWrite::validate_owned_object_versions()`]: what
341/// [`weigh_sender_rejection()`] gives for an [`IotaError::UserInput`] error,
342/// and otherwise what [`normalize()`] gives `error`.
343///
344/// [`ExecutionCacheWrite::validate_owned_object_versions()`]: crate::execution_cache::ExecutionCacheWrite::validate_owned_object_versions
345fn weigh_owned_input_rejection(error: &IotaError, after_dry_run: bool) -> Weight {
346    if matches!(error, IotaError::UserInput { .. }) {
347        // Another transaction of the sender used an owned input meanwhile.
348        weigh_sender_rejection(error, after_dry_run)
349    } else {
350        // The validator failed to read its own store.
351        normalize(error)
352    }
353}

355impl ValidatorService {
356    async fn submit_tx_impl(
357        &self,
358        transactions: Vec<TransactionEnvelope>,
359    ) -> Result<ReceiverStream<TxUpdateItem>, tonic::Status> {
360        let state = self.state.clone();
361        let epoch_store = state.load_epoch_store_one_call_per_task();
362
363        fp_ensure!(
364            !state.is_fullnode(&epoch_store),
365            IotaError::FullNodeCantHandleValidatorV2.into()
366        );
367
368        fp_ensure!(
369            epoch_store.protocol_config().enable_pcool_flow(),
370            IotaError::UnsupportedFeature {
371                error: "P-COOL flow is not enabled in this protocol version".to_string()
372            }
373            .into()
374        );
375
376        fp_ensure!(
377            transactions.len() <= MAX_TRANSACTIONS_PER_SUBMIT,
378            tonic::Status::invalid_argument(format!(
379                "too many transactions: {} exceeds limit of {MAX_TRANSACTIONS_PER_SUBMIT}",
380                transactions.len()
381            ))
382        );
383
384        let (tx_sender, rx) = tokio::sync::mpsc::channel(transactions.len().max(1));
385        let consensus_adapter = self.consensus_adapter.clone();
386        let metrics = self.metrics.clone();
387        let soft_locks = self.soft_locks.clone();
388
389        // Run per-tx tasks concurrently, capped by `MAX_CONCURRENT_SUBMIT_TASKS`.
390        // Spawning lets CPU-heavy work run across worker threads; `buffer_unordered`
391        // forwards results as tasks complete.
392        spawn_monitored_task!(async move {
393            let mut in_flight = stream::iter(transactions)
394                .map(move |transaction| {
395                    let state = state.clone();
396                    let epoch_store = epoch_store.clone();
397                    let consensus_adapter = consensus_adapter.clone();
398                    let metrics = metrics.clone();
399                    let soft_locks = soft_locks.clone();
400                    spawn_monitored_task!(async move {
401                        let tx_digest = *transaction.digest();
402                        let (update, weight) = Self::submit_single_tx(
403                            &state,
404                            &consensus_adapter,
405                            &metrics,
406                            &epoch_store,
407                            &soft_locks,
408                            transaction,
409                        )
410                        .await;
411                        ((tx_digest, update), weight)
412                    })
413                })
414                .buffer_unordered(MAX_CONCURRENT_SUBMIT_TASKS)
415                .map(|join_res| {
416                    join_res.map_err(|e| {
417                        tonic::Status::internal(format!("submit_single_tx task failed: {e}"))
418                    })
419                });
420
421            while let Some(item) = in_flight.next().await {
422                // Stop forwarding on client disconnect; in-flight tasks still
423                // run to completion (dropping a JoinHandle doesn't cancel).
424                if tx_sender.send(item).await.is_err() {
425                    break;
426                }
427            }
428        });
429
430        Ok(ReceiverStream::new(rx))
431    }

433    /// Handles submission of a single transaction. Validates, checks for prior
434    /// execution, verifies signature, runs deny checks, and submits to
435    /// consensus. Returns the terminal status together with the per-item
436    /// traffic weight derived from the outcome: `Weight::one()` for an
437    /// already-executed duplicate (spam-like), for a transaction too large for
438    /// consensus once attested, for a Move authentication that failed while
439    /// the transaction's dependency has not executed here, and for a changed
440    /// owned input or a refused soft lock after the attestation's dry run
441    /// (the locks are taken after the dry run only when the transaction's
442    /// Move authenticators did not run on their own first); `Weight::zero()`
443    /// for a rejection because the validator is overloaded and for a
444    /// successful submission; and `normalize(&error)` for any other rejection
445    /// (signature/epoch errors weigh, others don't).
446    async fn submit_single_tx(
447        state: &Arc<AuthorityState>,
448        consensus_adapter: &Arc<ConsensusAdapter>,
449        metrics: &Arc<ValidatorServiceMetrics>,
450        epoch_store: &Arc<AuthorityPerEpochStore>,
451        soft_locks: &Arc<PreConsensusSoftLocks>,
452        transaction: TransactionEnvelope,
453    ) -> (TxStatusUpdate, Weight) {
454        let tx_digest = *transaction.digest();
455
456        let build_executed = |effects: TransactionEffects| -> TxStatusUpdate {
457            let effects_digest = effects.digest();
458            if let Err(error) = state.check_effects_against_previously_signed(
459                epoch_store,
460                &tx_digest,
461                &effects_digest,
462                "submit_tx",
463            ) {
464                return TxStatusUpdate::Rejected { error };
465            }
466            TxStatusUpdate::Executed {
467                effects_digest,
468                details: Some(Self::build_executed_data(state, &effects)),
469            }
470        };
471
472        // Take this transaction's slot in the consensus queue and run the
473        // overload checks against the queue at that moment, which also counts
474        // admissions still in progress. Returning before the submission below
475        // drops the slot and releases it.
476        let slot = match state.reserve_consensus_queue_slot(consensus_adapter, transaction.data()) {
477            Ok(slot) => slot,
478            Err(e) => return reject_for_overload(metrics, e),
479        };
480
481        // Validate transaction.
482        if checks_admitted_validity() {
483            if let Err(e) = transaction.validity_check(&epoch_store.tx_validity_check_context()) {
484                let weight = normalize(&e);
485                return (TxStatusUpdate::Rejected { error: e }, weight);
486            }
487        }
488
489        // Check if already executed. Transient cache errors are treated as
490        // "not found" — consensus handles dedup, so the normal submission
491        // flow continues safely.
492        if let Some(effects) = state
493            .get_transaction_cache_reader()
494            .try_get_executed_effects(&tx_digest)
495            .ok()
496            .flatten()
497        {
498            return (build_executed(effects), Weight::one());
499        }
500
501        // Suppress duplicate resubmissions of a transaction that is still in
502        // flight on this validator (its soft locks are held). Checked before
503        // signature verification so resubmission storms are short-circuited
504        // cheaply; the soft-lock acquisition below remains the authoritative,
505        // atomic gate for duplicates racing past this probe.
506        if soft_locks.check_in_flight(&tx_digest) {
507            metrics.num_rejected_tx_recently_resubmitted.inc();
508            let error = IotaError::RecentlyResubmitted { digest: tx_digest };
509            let weight = normalize(&error);
510            return (TxStatusUpdate::Rejected { error }, weight);
511        }
512
513        // Refuse a transaction when another transaction in flight on this
514        // validator holds a soft lock on one of its inputs. Before signature
515        // verification, since this takes no lock and every lock it can find
516        // was taken by a transaction that passed validation. With validator
517        // attestation enabled, the check runs again once a dry run is free. The
518        // locks themselves are taken after validation.
519        let inputs = list_owned_or_immutable_inputs(&transaction);
520        if let Err(error) = soft_locks.check_conflicts(&tx_digest, &inputs) {
521            count_soft_lock_refusal(metrics, &error);
522            let weight = normalize(&error);
523
524            return (TxStatusUpdate::Rejected { error }, weight);
525        }
526
527        // Every validator drops a cut-off validator's attested transactions
528        // for the rest of the epoch, so it rejects them here and the client
529        // submits through another validator. Before the signature check,
530        // which a rejected submission need not pay for.
531        //
532        // Tests register the `attest-despite-cutoff` fail point to make a
533        // cut-off validator attest anyway, as a faulty validator can. A `Cell`
534        // keeps the binding immutable in builds where the fail point compiles
535        // to nothing.
536        let rejects_when_cut_off = std::cell::Cell::new(true);
537        iota_macros::fail_point_if!("attest-despite-cutoff", || {
538            rejects_when_cut_off.set(false)
539        });
540        if rejects_when_cut_off.get() {
541            if let Some(attestor) = epoch_store.own_index_if_cut_off() {
542                let error = IotaError::AttestorCutOff { attestor };
543                let weight = normalize(&error);
544                return (TxStatusUpdate::Rejected { error }, weight);
545            }
546        }
547
548        // Verify user signature.
549        let tx_verif_guard = metrics.tx_verification_latency.start_timer();
550        let verified_tx = match epoch_store.verify_transaction(transaction) {
551            Ok(verified) => verified,
552            Err(e) => {
553                metrics.signature_errors.inc();
554                let weight = normalize(&e);
555                return (TxStatusUpdate::Rejected { error: e }, weight);
556            }
557        };
558        drop(tx_verif_guard);
559
560        // Early bail-out during epoch boundary.
561        if !admits_user_transactions(&epoch_store.get_reconfig_state_read_lock_guard()) {
562            metrics.num_rejected_tx_in_epoch_boundary.inc();
563            let error = IotaError::ValidatorHaltedAtEpochEnd;
564            let weight = normalize(&error);
565            return (TxStatusUpdate::Rejected { error }, weight);
566        }
567
568        // A dependency not visible here yet is reported only after the checks
569        // below, so an error that rejects the transaction for good comes
570        // first. They then run without the Move authenticators and without
571        // the attestation's dry run.
572        let mut dependency_not_found = None;
573        if checks_admitted_inputs() {
574            match check_transaction_dependency_for_submission(
575                state,
576                epoch_store,
577                soft_locks,
578                &verified_tx,
579            ) {
580                Ok(()) => {}
581                Err(
582                    error @ IotaError::UserInput {
583                        error: UserInputError::TransactionDependencyNotFound { .. },
584                    },
585                ) => dependency_not_found = Some(error),
586                Err(e) => {
587                    let weight = normalize(&e);
588                    return (TxStatusUpdate::Rejected { error: e }, weight);
589                }
590            }
591        }
592
593        // Deny-rule source for admission. With governance enabled, admission
594        // checks the active governance rules on top of the local config, so
595        // transactions the network will deterministically drop post-consensus
596        // don't occupy consensus slots. The local config always applies.
597        let governance_rules = epoch_store
598            .protocol_config()
599            .enable_pcool_flow()
600            .then(|| epoch_store.get_active_transaction_deny_rules());
601
602        // With validator attestation enabled, `attest_transaction` runs the
603        // deny checks, object loading, input checks, coin deny list and a full
604        // dry run in one call and produces the attestation payload; otherwise
605        // the validation-only function runs. While the dependency is not
606        // found, only its checks before the Move authenticators run: the
607        // transaction is rejected whatever the authenticators answer. The
608        // attestation takes the soft locks itself before its dry run, once
609        // the transaction's signers have authorized it.
610        let mut soft_lock = None;
611        let (attestation_data, owned_objects, verified_tx, permit) = if !checks_admitted_inputs() {
612            // Only tests take this branch, through the fail point of
613            // `checks_admitted_inputs()`: the validator skips its input checks
614            // and the dry run, as a faulty one can. With validator attestation
615            // on, it attaches an attestation built without the dry run, since
616            // block verification refuses a transaction without one.
617            let attestation_data = epoch_store
618                .protocol_config()
619                .enable_validator_attestation()
620                .then(|| attest_without_dry_run(&verified_tx));
621
622            (attestation_data, Vec::new(), verified_tx, None)
623        } else if dependency_not_found.is_none()
624            && epoch_store.protocol_config().enable_validator_attestation()
625        {
626            let permit = match state.acquire_attestation_permit().await {
627                Ok(permit) => permit,
628                Err(e) => return reject_for_overload(metrics, e),
629            };
630            // Another transaction may have locked these inputs while this one
631            // waited for its dry run.
632            if let Err(error) = soft_locks.check_conflicts(&tx_digest, &inputs) {
633                count_soft_lock_refusal(metrics, &error);
634                let weight = normalize(&error);
635
636                return (TxStatusUpdate::Rejected { error }, weight);
637            }
638            let state_for_attest = state.clone();
639            let epoch_store_for_attest = epoch_store.clone();
640            let governance_for_attest = governance_rules.clone();
641            let soft_locks_for_attest = soft_locks.clone();
642            let metrics_for_attest = metrics.clone();
643            let join_result = tokio::task::spawn_blocking(move || {
644                // Built inside the closure: `DenyRuleUnion` borrows, and
645                // `spawn_blocking` requires `'static` captures.
646                let local_deny_config = &state_for_attest.config.transaction_deny_config;
647                let combined_deny_config;
648                let deny_config: &dyn DenyRuleConfig = match governance_for_attest.as_ref() {
649                    Some(rules) => {
650                        combined_deny_config = DenyRuleUnion {
651                            first: local_deny_config,
652                            second: rules.as_ref(),
653                        };
654                        &combined_deny_config
655                    }
656                    None => local_deny_config,
657                };
658                let mut lock_before_dry_run = None;
659                let result = state_for_attest.attest_transaction(
660                    &verified_tx,
661                    &epoch_store_for_attest,
662                    deny_config,
663                    |owned_objects| {
664                        let guard = soft_locks_for_attest
665                            .try_acquire_guard(tx_digest, owned_objects)
666                            .inspect_err(|error| {
667                                count_soft_lock_refusal(&metrics_for_attest, error)
668                            })?;
669                        lock_before_dry_run = Some(guard);
670
671                        Ok(())
672                    },
673                );
674                // The permit comes back with the result, so the admission holds
675                // it until its soft locks are taken. A dry run that panics, or
676                // whose submission was dropped, frees it when it ends.
677                (result, verified_tx, permit, lock_before_dry_run)
678            })
679            .await;
680            let (result, verified_tx, permit, lock_before_dry_run) = match join_result {
681                Ok(dry_run) => dry_run,
682                Err(join_err) => {
683                    // A panic in the dry run reaches this point only where
684                    // panics unwind, as in tests: release builds set
685                    // `panic = 'abort'`, so there it stops the validator. A
686                    // `JoinError` also comes when the runtime shuts down
687                    // before the dry run starts. The transaction is rejected
688                    // rather than panicking again on this submission's task.
689                    tracing::error!(?tx_digest, "attest_transaction task failed: {join_err}");
690                    let err = IotaError::GenericAuthority {
691                        error: format!("attest_transaction task failed: {join_err}"),
692                    };
693                    let weight = normalize(&err);
694                    return (TxStatusUpdate::Rejected { error: err }, weight);
695                }
696            };
697            match result {
698                Ok((data, objs)) => {
699                    soft_lock = lock_before_dry_run;
700
701                    (Some(data), objs, verified_tx, Some(permit))
702                }
703                Err(e) => {
704                    let weight = weigh_validation_rejection(&e);
705                    return (TxStatusUpdate::Rejected { error: e }, weight);
706                }
707            }
708        } else {
709            let local_deny_config = &state.config.transaction_deny_config;
710            let combined_deny_config;
711            let deny_config: &dyn DenyRuleConfig = match governance_rules.as_ref() {
712                Some(rules) => {
713                    combined_deny_config = DenyRuleUnion {
714                        first: local_deny_config,
715                        second: rules.as_ref(),
716                    };
717                    &combined_deny_config
718                }
719                None => local_deny_config,
720            };
721            if let Some(not_found) = dependency_not_found {
722                let error = state
723                    .handle_transaction_validation_checks_without_authentication(
724                        &verified_tx,
725                        epoch_store,
726                        deny_config,
727                    )
728                    .err()
729                    .unwrap_or(not_found);
730                let weight = normalize(&error);
731                return (TxStatusUpdate::Rejected { error }, weight);
732            }
733            match state
734                .handle_transaction_validation_checks(&verified_tx, epoch_store, deny_config)
735                .await
736            {
737                Ok(objs) => (None, objs, verified_tx, None),
738                Err(e) => {
739                    let weight = weigh_validation_rejection(&e);
740                    return (TxStatusUpdate::Rejected { error: e }, weight);
741                }
742            }
743        };
744
745        // From here on, a rejection the sender causes weighs one if the
746        // attestation's dry run ran.
747        let after_dry_run = attestation_data.is_some();
748
749        if let Err(e) = state
750            .get_cache_writer()
751            .validate_owned_object_versions(&owned_objects)
752        {
753            // Edge case: check if executed while being validated.
754            if let Some(effects) = state
755                .get_transaction_cache_reader()
756                .try_get_executed_effects(&tx_digest)
757                .ok()
758                .flatten()
759            {
760                return (build_executed(effects), Weight::one());
761            }
762            let weight = weigh_owned_input_rejection(&e, after_dry_run);
763
764            return (TxStatusUpdate::Rejected { error: e }, weight);
765        }
766
767        // Reconfig check. The guard is held through consensus submission below.
768        let reconfiguration_lock = epoch_store.get_reconfig_state_read_lock_guard();
769        if !admits_user_transactions(&reconfiguration_lock) {
770            metrics.num_rejected_tx_in_epoch_boundary.inc();
771            let error = IotaError::ValidatorHaltedAtEpochEnd;
772            let weight = normalize(&error);
773            return (TxStatusUpdate::Rejected { error }, weight);
774        }
775
776        // Build the consensus transaction.
777        let consensus_tx = if let Some(attestation_data) = attestation_data {
778            // submit_single_tx runs on a validator, so it must be present in its
779            // own committee. Treat absence as a fatal bug, not a user error.
780            let authority_index = epoch_store
781                .committee()
782                .authority_index(&state.name)
783                .expect("validator must be present in its own consensus committee");
784            let authority_index = u8::try_from(authority_index)
785                .expect("consensus committee indices should fit in a u8");
786
787            ConsensusTransaction::new_attested_user_transaction(AttestedTransaction::new(
788                verified_tx.into_inner(),
789                Some(Attestation::new_authority(
790                    attestation_data,
791                    authority_index,
792                )),
793            ))
794        } else {
795            ConsensusTransaction::new_user_transaction(verified_tx.into_inner())
796        };
797
798        // The attestation is added after the transaction passed its own size
799        // check. The protocol config's limits `max_tx_size_bytes` and
800        // `consensus_max_transaction_size_bytes` leave room for the largest
801        // attestation beside the largest transaction, so
802        // `check_consensus_transaction_size()` refuses a transaction only if
803        // the limits no longer leave that room. The consensus adapter refuses
804        // such a transaction as well, but its refusal weighs nothing. A
805        // rejection by `check_consensus_transaction_size()` weighs one: the dry
806        // run has already run, and with no soft lock kept, the in-flight check
807        // does not stop a resend.
808        if let Err(error) =
809            check_consensus_transaction_size(&consensus_tx, epoch_store.protocol_config())
810        {
811            return (TxStatusUpdate::Rejected { error }, Weight::one());
812        }
813
814        // Soft-lock owned objects to prevent conflicting transactions. With
815        // attestation, `attest_transaction()` took the locks before its dry
816        // run, except when the transaction's Move authenticators did not run
817        // on their own first.
818        // Same-digest resubmission while in flight → rejected (duplicate).
819        // Different-digest conflict on same objects → rejected.
820        // A rejection after the locks are taken drops the guard, which
821        // releases them. A lock refused after the dry run means that another
822        // submission on the same objects passed both checks for conflicts at
823        // the same time as this one.
824        let soft_lock = match soft_lock {
825            Some(soft_lock) => soft_lock,
826            None => match soft_locks.try_acquire_guard(tx_digest, &owned_objects) {
827                Ok(soft_lock) => soft_lock,
828                Err(error) => {
829                    count_soft_lock_refusal(metrics, &error);
830                    let weight = weigh_sender_rejection(&error, after_dry_run);
831
832                    return (TxStatusUpdate::Rejected { error }, weight);
833                }
834            },
835        };
836        // Freed only now, so that a submission on the same objects that gets
837        // this permit next finds these locks in its check for conflicts.
838        drop(permit);
839        metrics
840            .soft_lock_table_size
841            .set(soft_locks.lock_count() as i64);
842
843        // Submit to consensus. The slot stays counted until consensus has
844        // processed the transaction, and any other outcome releases the slot.
845        if let Err(e) = consensus_adapter.submit_with_slot(
846            consensus_tx,
847            Some(&reconfiguration_lock),
848            epoch_store,
849            slot,
850        ) {
851            let weight = normalize(&e);
852            // Release soft locks so the transaction can be retried.
853            tracing::debug!(
854                ?tx_digest,
855                error = ?e,
856                "releasing soft lock after consensus submit failure",
857            );
858            drop(soft_lock);
859            metrics
860                .soft_lock_table_size
861                .set(soft_locks.lock_count() as i64);
862            return (TxStatusUpdate::Rejected { error: e }, weight);
863        }
864        soft_lock.keep();
865
866        (TxStatusUpdate::Submitted, Weight::zero())
867    }

869    /// Waits for one or more previously submitted transactions to reach
870    /// finality and streams one status per digest: Executed, Rejected,
871    /// Expired when the epoch ended first, or TimedOut when the wait ended
872    /// first.
873    async fn get_tx_status_impl(
874        &self,
875        request: DomainGetTxStatusRequest,
876    ) -> Result<ReceiverStream<TxUpdateItem>, tonic::Status> {
877        let state = self.state.clone();
878        let epoch_store = state.load_epoch_store_one_call_per_task();
879
880        fp_ensure!(
881            !state.is_fullnode(&epoch_store),
882            IotaError::FullNodeCantHandleValidatorV2.into()
883        );
884
885        fp_ensure!(
886            epoch_store.protocol_config().enable_pcool_flow(),
887            IotaError::UnsupportedFeature {
888                error: "P-COOL flow is not enabled in this protocol version".to_string()
889            }
890            .into()
891        );
892
893        // Empty queries is a valid no-op/ping.
894        fp_ensure!(
895            request.queries.len() <= MAX_QUERIES_PER_GET_TX_STATUS,
896            tonic::Status::invalid_argument(format!(
897                "too many queries: {} exceeds limit of {MAX_QUERIES_PER_GET_TX_STATUS}",
898                request.queries.len()
899            ))
900        );
901
902        let (tx_sender, rx) = tokio::sync::mpsc::channel(request.queries.len().max(1));
903
904        for query in request.queries {
905            let state = state.clone();
906            let epoch_store = epoch_store.clone();
907            let tx_sender = tx_sender.clone();
908            spawn_monitored_task!(async move {
909                tokio::select! {
910                    biased;
911                    _ = tx_sender.closed() => {}
912                    (update, weight) = Self::wait_for_tx_finality(
913                        &state,
914                        &epoch_store,
915                        query.transaction_digest,
916                        query.include_details,
917                    ) => {
918                        let _ = tx_sender
919                            .send(Ok(((query.transaction_digest, update), weight)))
920                            .await;
921                    }
922                }
923            });
924        }
925
926        Ok(ReceiverStream::new(rx))
927    }

929    /// Waits for a single transaction to reach finality. Returns immediately
930    /// if already executed, otherwise blocks up to the timeout. The returned
931    /// `Weight` is the per-item spam contribution decided by the path taken:
932    /// a fast-path cache hit means the client is querying an already-finalized
933    /// tx (duplicate/redundant query, `Weight::one()`), while every wait-path
934    /// outcome corresponds to legitimate polling for an in-flight tx and is
935    /// weighted as `Weight::zero()`.
936    async fn wait_for_tx_finality(
937        state: &Arc<AuthorityState>,
938        epoch_store: &Arc<AuthorityPerEpochStore>,
939        tx_digest: TransactionDigest,
940        include_details: bool,
941    ) -> (TxStatusUpdate, Weight) {
942        // Fast path: already executed → duplicate query, spam-like.
943        let cache = state.get_transaction_cache_reader();
944        match cache.try_get_executed_effects(&tx_digest) {
945            Ok(Some(effects)) => {
946                return (
947                    Self::build_executed_update(state, epoch_store, effects, include_details),
948                    Weight::one(),
949                );
950            }
951            Err(e) => {
952                tracing::warn!(?tx_digest, "failed to read effects cache: {e}");
953                // Fall through to the wait path.
954            }
955            Ok(None) => {}
956        }
957
958        // Wait for execution, rejection, epoch end, or timeout. All outcomes
959        // below are legitimate polling for an in-flight tx → Weight::zero().
960        let result = tokio::time::timeout(
961            std::time::Duration::from_secs(GET_TX_STATUS_TIMEOUT_SECS),
962            epoch_store.within_alive_epoch(async {
963                let digests_to_watch = [tx_digest];
964                tokio::select! {
965                    biased;
966                    effects_digests = cache.notify_read_executed_effects_digests("ValidatorService::notify_read_executed_effects_digests", &digests_to_watch) => {
967                        Either::Left(effects_digests)
968                    }
969                    dropped_error = epoch_store.notify_read_dropped_digests(tx_digest) => {
970                        Either::Right(dropped_error)
971                    }
972                }
973            }),
974        )
975            .await;
976
977        let update = match result {
978            // Epoch ended before execution or rejection.
979            Ok(Err(_)) => TxStatusUpdate::Expired {
980                epoch: epoch_store.epoch(),
981            },
982            Ok(Ok(Either::Left(effects_digests))) => {
983                let Some(effects_digest) = effects_digests.into_iter().next() else {
984                    // The notifier returns one digest per requested digest.
985                    debug_fatal!("empty effects from notify_read for {tx_digest:?}");
986                    return (TxStatusUpdate::TimedOut, Weight::zero());
987                };
988                if let Err(error) = state.check_effects_against_previously_signed(
989                    epoch_store,
990                    &tx_digest,
991                    &effects_digest,
992                    "get_tx_status",
993                ) {
994                    return (TxStatusUpdate::Rejected { error }, Weight::zero());
995                }
996                let details = if include_details {
997                    match cache.try_get_executed_effects(&tx_digest) {
998                        Ok(Some(effects)) => Some(Self::build_executed_data(state, &effects)),
999                        Ok(None) => {
1000                            tracing::warn!(
1001                                ?tx_digest,
1002                                "effects disappeared between notification and fetch, \
1003                                 returning Executed without details"
1004                            );
1005                            None
1006                        }
1007                        Err(e) => {
1008                            tracing::warn!(
1009                                ?tx_digest,
1010                                "failed to read effects: {e}, returning Executed without details"
1011                            );
1012                            None
1013                        }
1014                    }
1015                } else {
1016                    None
1017                };
1018                TxStatusUpdate::Executed {
1019                    effects_digest,
1020                    details,
1021                }
1022            }
1023            Ok(Ok(Either::Right(dropped_error))) => TxStatusUpdate::Rejected {
1024                error: dropped_error,
1025            },
1026            Err(_timeout) => TxStatusUpdate::TimedOut,
1027        };
1028        (update, Weight::zero())
1029    }

1031    /// Builds a `TxStatusUpdate::Executed` from known effects, optionally
1032    /// including full details.
1033    fn build_executed_update(
1034        state: &Arc<AuthorityState>,
1035        epoch_store: &AuthorityPerEpochStore,
1036        effects: TransactionEffects,
1037        include_details: bool,
1038    ) -> TxStatusUpdate {
1039        let effects_digest = effects.digest();
1040        if let Err(error) = state.check_effects_against_previously_signed(
1041            epoch_store,
1042            effects.transaction_digest(),
1043            &effects_digest,
1044            "get_tx_status",
1045        ) {
1046            return TxStatusUpdate::Rejected { error };
1047        }
1048        let details = if include_details {
1049            Some(Self::build_executed_data(state, &effects))
1050        } else {
1051            None
1052        };
1053        TxStatusUpdate::Executed {
1054            effects_digest,
1055            details,
1056        }
1057    }

1059    /// Fetches execution details (events, input/output objects) for a
1060    /// transaction.
1061    fn build_executed_data(
1062        state: &Arc<AuthorityState>,
1063        effects: &TransactionEffects,
1064    ) -> Box<ExecutedData> {
1065        let events = if effects.events_digest().is_some() {
1066            state
1067                .get_transaction_events(effects.transaction_digest())
1068                .ok()
1069        } else {
1070            None
1071        };
1072        let input_objects = state
1073            .get_transaction_input_objects(effects)
1074            .ok()
1075            .unwrap_or_default();
1076        let output_objects = state
1077            .get_transaction_output_objects(effects)
1078            .ok()
1079            .unwrap_or_default();
1080        Box::new(ExecutedData {
1081            effects: effects.clone(),
1082            events,
1083            input_objects,
1084            output_objects,
1085        })
1086    }

1088    async fn notify_capabilities_impl(
1089        &self,
1090        request: HandleCapabilityNotificationRequestV1,
1091    ) -> Result<(HandleCapabilityNotificationResponseV1, Weight), tonic::Status> {
1092        let epoch_store = self.state.load_epoch_store_one_call_per_task();
1093
1094        fp_ensure!(
1095            epoch_store
1096                .protocol_config()
1097                .track_non_committee_eligible_validators(),
1098            IotaError::UnsupportedFeature {
1099                error: "capability notification endpoint is not supported in this Protocol Version"
1100                    .to_string()
1101            }
1102            .into()
1103        );
1104
1105        fp_ensure!(
1106            !self.state.is_fullnode(&epoch_store),
1107            IotaError::FullNodeCantHandleValidatorV2.into()
1108        );
1109
1110        let existing_capabilities = epoch_store.get_capabilities_v1()?;
1111        let incoming_capability = request.message.data();
1112
1113        tracing::info!(
1114            "Received capability notification: {:?}",
1115            incoming_capability
1116        );
1117
1118        if let Some(existing) = existing_capabilities
1119            .iter()
1120            .find(|cap| cap.authority == incoming_capability.authority)
1121        {
1122            if incoming_capability.generation <= existing.generation {
1123                return Ok((
1124                    HandleCapabilityNotificationResponseV1 { _unused: false },
1125                    Weight::one(),
1126                ));
1127            }
1128        }
1129
1130        if let Err(error) = self.consensus_adapter.check_consensus_overload() {
1131            self.metrics
1132                .num_rejected_capability_notifications_during_overload
1133                .with_label_values(&[error.as_ref()])
1134                .inc();
1135            return Err(error.into());
1136        }
1137
1138        let _metrics_guard = self
1139            .metrics
1140            .handle_capability_notification_latency
1141            .start_timer();
1142
1143        let signed_authority_capabilities = request.message;
1144        let verified_authority_capabilities = epoch_store
1145            .verify_authority_capabilities(signed_authority_capabilities)
1146            .inspect_err(|_e| {
1147                self.metrics.signature_errors.inc();
1148            })?;
1149
1150        let authority_name = verified_authority_capabilities.authority;
1151        // Process the verified capabilities
1152        tracing::debug!("Verified capability notification for authority {authority_name:?}");
1153
1154        let transaction = ConsensusTransaction::new_signed_capability_notification_v1(
1155            verified_authority_capabilities.into_inner(),
1156        );
1157
1158        self.consensus_adapter
1159            .submit(transaction, None, &epoch_store)?;
1160
1161        tracing::debug!(
1162            "Submitted capability notification to consensus for authority {authority_name:?}"
1163        );
1164
1165        Ok((
1166            HandleCapabilityNotificationResponseV1 { _unused: false },
1167            Weight::one(),
1168        ))
1169    }

1247        let (req, ip) = self.pre_handle(request).await?;
1248        self.post_handle_unary(ip, self.health_check_impl(req))
1249    }
