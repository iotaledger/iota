28/// Maximum number of transactions allowed in a single `submit_tx` request,
29/// bounding the traffic tallies and execution work one request can create.
30const MAX_TRANSACTIONS_PER_SUBMIT: usize = 32;
31
32/// Maximum number of queries allowed in a single `get_tx_status` request.
33/// Matches `MAX_TRANSACTIONS_PER_SUBMIT` so one status poll can cover a full
34/// submit batch.
35const MAX_QUERIES_PER_GET_TX_STATUS: usize = 32;

37/// Timeout for waiting on transaction execution in `get_tx_status`.
38const GET_TX_STATUS_TIMEOUT_SECS: u64 = 30;

40/// The union of two deny rule sources: denies whatever either source denies.

96/// Maximum number of `submit_single_tx` futures allowed to run concurrently
97/// per `submit_tx` request. Caps pre-consensus work (signature verification,
98/// validation, DB reads) and contention on `consensus_adapter`'s submit
99/// semaphore.
100const MAX_CONCURRENT_SUBMIT_TASKS: usize = 16;
101
102/// A single streamed item in a V2 RPC response. The `Weight` is the per-item
103/// spam-policy contribution decided by the producing code path.
104type TxUpdateItem = Result<((TransactionDigest, TxStatusUpdate), Weight), tonic::Status>;

118impl ValidatorService {
119    async fn submit_tx_impl(
120        &self,
121        transactions: Vec<TransactionEnvelope>,
122    ) -> Result<ReceiverStream<TxUpdateItem>, tonic::Status> {
123        let state = self.state.clone();
124        let epoch_store = state.load_epoch_store_one_call_per_task();
125
126        fp_ensure!(
127            !state.is_fullnode(&epoch_store),
128            IotaError::FullNodeCantHandleValidatorV2.into()
129        );
130
131        fp_ensure!(
132            epoch_store.protocol_config().enable_pcool_flow(),
133            IotaError::UnsupportedFeature {
134                error: "P-COOL flow is not enabled in this protocol version".to_string()
135            }
136            .into()
137        );
138
139        fp_ensure!(
140            transactions.len() <= MAX_TRANSACTIONS_PER_SUBMIT,
141            tonic::Status::invalid_argument(format!(
142                "too many transactions: {} exceeds limit of {MAX_TRANSACTIONS_PER_SUBMIT}",
143                transactions.len()
144            ))
145        );
146
147        let (tx_sender, rx) = tokio::sync::mpsc::channel(transactions.len().max(1));
148        let consensus_adapter = self.consensus_adapter.clone();
149        let metrics = self.metrics.clone();
150        let soft_locks = self.soft_locks.clone();
151
152        // Run per-tx tasks concurrently, capped by `MAX_CONCURRENT_SUBMIT_TASKS`.
153        // Spawning lets CPU-heavy work run across worker threads; `buffer_unordered`
154        // forwards results as tasks complete.
155        spawn_monitored_task!(async move {
156            let mut in_flight = stream::iter(transactions)
157                .map(move |transaction| {
158                    let state = state.clone();
159                    let epoch_store = epoch_store.clone();
160                    let consensus_adapter = consensus_adapter.clone();
161                    let metrics = metrics.clone();
162                    let soft_locks = soft_locks.clone();
163                    spawn_monitored_task!(async move {
164                        let tx_digest = *transaction.digest();
165                        let (update, weight) = Self::submit_single_tx(
166                            &state,
167                            &consensus_adapter,
168                            &metrics,
169                            &epoch_store,
170                            &soft_locks,
171                            transaction,
172                        )
173                        .await;
174                        ((tx_digest, update), weight)
175                    })
176                })
177                .buffer_unordered(MAX_CONCURRENT_SUBMIT_TASKS)
178                .map(|join_res| {
179                    join_res.map_err(|e| {
180                        tonic::Status::internal(format!("submit_single_tx task failed: {e}"))
181                    })
182                });
183
184            while let Some(item) = in_flight.next().await {
185                // Stop forwarding on client disconnect; in-flight tasks still
186                // run to completion (dropping a JoinHandle doesn't cancel).
187                if tx_sender.send(item).await.is_err() {
188                    break;
189                }
190            }
191        });
192
193        Ok(ReceiverStream::new(rx))
194    }

196    /// Handles submission of a single transaction. Validates, checks for prior
197    /// execution, verifies signature, runs deny checks, and submits to
198    /// consensus. Returns the terminal status together with the per-item
199    /// traffic weight derived from the outcome: `Weight::one()` for an
200    /// already-executed duplicate (spam-like), `normalize(&error)` for a
201    /// rejection (signature/epoch errors weigh, others don't), and
202    /// `Weight::zero()` for a successful submission.
203    async fn submit_single_tx(
204        state: &Arc<AuthorityState>,
205        consensus_adapter: &Arc<ConsensusAdapter>,
206        metrics: &Arc<ValidatorServiceMetrics>,
207        epoch_store: &Arc<AuthorityPerEpochStore>,
208        soft_locks: &Arc<PreConsensusSoftLocks>,
209        transaction: TransactionEnvelope,
210    ) -> (TxStatusUpdate, Weight) {
211        let tx_digest = *transaction.digest();
212
213        let build_executed = |effects: TransactionEffects| -> TxStatusUpdate {
214            let effects_digest = effects.digest();
215            if let Err(error) = state.check_effects_against_previously_signed(
216                epoch_store,
217                &tx_digest,
218                &effects_digest,
219                "submit_tx",
220            ) {
221                return TxStatusUpdate::Rejected { error };
222            }
223            TxStatusUpdate::Executed {
224                effects_digest,
225                details: Some(Self::build_executed_data(state, &effects)),
226            }
227        };
228
229        // Check system overload.
230        if let Err(e) = state.check_system_overload(
231            consensus_adapter,
232            transaction.data(),
233            state.check_system_overload_at_signing(),
234            // `true` means P-COOL flow is enabled - ensured by `fp_ensure!` in the caller
235            true,
236        ) {
237            metrics
238                .num_rejected_tx_during_overload
239                .with_label_values(&[e.as_ref()])
240                .inc();
241            let weight = normalize(&e);
242            return (TxStatusUpdate::Rejected { error: e }, weight);
243        }
244
245        // Validate transaction.
246        if let Err(e) = transaction.validity_check(&epoch_store.tx_validity_check_context()) {
247            let weight = normalize(&e);
248            return (TxStatusUpdate::Rejected { error: e }, weight);
249        }
250
251        // Check if already executed. Transient cache errors are treated as
252        // "not found" — consensus handles dedup, so the normal submission
253        // flow continues safely.
254        if let Some(effects) = state
255            .get_transaction_cache_reader()
256            .try_get_executed_effects(&tx_digest)
257            .ok()
258            .flatten()
259        {
260            return (build_executed(effects), Weight::one());
261        }
262
263        // Suppress duplicate resubmissions of a transaction that is still in
264        // flight on this validator (its soft locks are held). Checked before
265        // signature verification so resubmission storms are short-circuited
266        // cheaply; the soft-lock acquisition below remains the authoritative,
267        // atomic gate for duplicates racing past this probe.
268        if soft_locks.check_in_flight(&tx_digest) {
269            metrics.num_rejected_tx_recently_resubmitted.inc();
270            let error = IotaError::RecentlyResubmitted { digest: tx_digest };
271            let weight = normalize(&error);
272            return (TxStatusUpdate::Rejected { error }, weight);
273        }
274
275        // Verify user signature.
276        let tx_verif_guard = metrics.tx_verification_latency.start_timer();
277        let verified_tx = match epoch_store.verify_transaction(transaction) {
278            Ok(verified) => verified,
279            Err(e) => {
280                metrics.signature_errors.inc();
281                let weight = normalize(&e);
282                return (TxStatusUpdate::Rejected { error: e }, weight);
283            }
284        };
285        drop(tx_verif_guard);
286
287        // Early bail-out during epoch boundary.
288        if !epoch_store
289            .get_reconfig_state_read_lock_guard()
290            .should_accept_user_certs()
291        {
292            metrics.num_rejected_tx_in_epoch_boundary.inc();
293            let error = IotaError::ValidatorHaltedAtEpochEnd;
294            let weight = normalize(&error);
295            return (TxStatusUpdate::Rejected { error }, weight);
296        }
297
298        // Content validation: deny checks + owned object version validation.
299        // With governance enabled, admission checks the active governance
300        // rules on top of the local config, so transactions the network will
301        // deterministically drop post-consensus don't occupy consensus slots.
302        // The local config always applies.
303        let local_deny_config = &state.config.transaction_deny_config;
304        let governance_rules;
305        let combined_deny_config;
306        let deny_config: &dyn DenyRuleConfig =
307            if epoch_store.protocol_config().deny_rule_governance() {
308                governance_rules = epoch_store.get_active_transaction_deny_rules();
309                combined_deny_config = DenyRuleUnion {
310                    first: local_deny_config,
311                    second: governance_rules.as_ref(),
312                };
313                &combined_deny_config
314            } else {
315                local_deny_config
316            };
317        let owned_objects = match state
318            .handle_transaction_validation_checks(
319                &verified_tx,
320                epoch_store,
321                deny_config,
322                // Latest-value coin deny-list read: admission is validator-local,
323                // and denials should take effect immediately. This deliberately
324                // differs from the epoch-gated post-consensus read; see the
325                // read-mode notes on `handle_transaction_validation_checks`.
326                false,
327            )
328            .await
329        {
330            Ok(objs) => objs,
331            Err(e) => {
332                let weight = normalize(&e);
333                return (TxStatusUpdate::Rejected { error: e }, weight);
334            }
335        };
336        if let Err(e) = state
337            .get_cache_writer()
338            .validate_owned_object_versions(&owned_objects)
339        {
340            // Edge case: check if executed while being validated.
341            if let Some(effects) = state
342                .get_transaction_cache_reader()
343                .try_get_executed_effects(&tx_digest)
344                .ok()
345                .flatten()
346            {
347                return (build_executed(effects), Weight::one());
348            }
349            let weight = normalize(&e);
350            return (TxStatusUpdate::Rejected { error: e }, weight);
351        }
352
353        // Reconfig check. The guard is held through consensus submission below.
354        let reconfiguration_lock = epoch_store.get_reconfig_state_read_lock_guard();
355        if !reconfiguration_lock.should_accept_user_certs() {
356            metrics.num_rejected_tx_in_epoch_boundary.inc();
357            let error = IotaError::ValidatorHaltedAtEpochEnd;
358            let weight = normalize(&error);
359            return (TxStatusUpdate::Rejected { error }, weight);
360        }
361
362        // Soft-lock owned objects to prevent conflicting transactions.
363        // Same-digest resubmission while in flight → rejected (duplicate).
364        // Different-digest conflict on same objects → rejected.
365        // Placed after the reconfig check so we never acquire locks that would
366        // need releasing on the epoch-halt path.
367        if let Err(error) = soft_locks.try_acquire(tx_digest, &owned_objects) {
368            if matches!(error, IotaError::RecentlyResubmitted { .. }) {
369                metrics.num_rejected_tx_recently_resubmitted.inc();
370            } else {
371                metrics.num_rejected_tx_soft_lock_conflict.inc();
372            }
373            let weight = normalize(&error);
374            return (TxStatusUpdate::Rejected { error }, weight);
375        }
376        metrics
377            .soft_lock_table_size
378            .set(soft_locks.lock_count() as i64);
379
380        // Submit to consensus.
381        if let Err(e) = consensus_adapter.submit(
382            ConsensusTransaction::new_user_transaction(verified_tx.into_inner()),
383            Some(&reconfiguration_lock),
384            epoch_store,
385        ) {
386            let weight = normalize(&e);
387            // Release soft locks so the transaction can be retried.
388            tracing::debug!(
389                ?tx_digest,
390                error = ?e,
391                "releasing soft lock after consensus submit failure",
392            );
393            soft_locks.release(&tx_digest);
394            metrics
395                .soft_lock_table_size
396                .set(soft_locks.lock_count() as i64);
397            return (TxStatusUpdate::Rejected { error: e }, weight);
398        }
399
400        (TxStatusUpdate::Submitted, Weight::zero())
401    }

403    /// Waits for one or more previously submitted transactions to reach
404    /// finality and streams a terminal status per digest (Executed, Rejected,
405    /// or Expired).
406    async fn get_tx_status_impl(
407        &self,
408        request: DomainGetTxStatusRequest,
409    ) -> Result<ReceiverStream<TxUpdateItem>, tonic::Status> {
410        let state = self.state.clone();
411        let epoch_store = state.load_epoch_store_one_call_per_task();
412
413        fp_ensure!(
414            !state.is_fullnode(&epoch_store),
415            IotaError::FullNodeCantHandleValidatorV2.into()
416        );
417
418        fp_ensure!(
419            epoch_store.protocol_config().enable_pcool_flow(),
420            IotaError::UnsupportedFeature {
421                error: "P-COOL flow is not enabled in this protocol version".to_string()
422            }
423            .into()
424        );
425
426        // Empty queries is a valid no-op/ping.
427        fp_ensure!(
428            request.queries.len() <= MAX_QUERIES_PER_GET_TX_STATUS,
429            tonic::Status::invalid_argument(format!(
430                "too many queries: {} exceeds limit of {MAX_QUERIES_PER_GET_TX_STATUS}",
431                request.queries.len()
432            ))
433        );
434
435        let (tx_sender, rx) = tokio::sync::mpsc::channel(request.queries.len().max(1));
436
437        for query in request.queries {
438            let state = state.clone();
439            let epoch_store = epoch_store.clone();
440            let tx_sender = tx_sender.clone();
441            spawn_monitored_task!(async move {
442                tokio::select! {
443                    biased;
444                    _ = tx_sender.closed() => {}
445                    (update, weight) = Self::wait_for_tx_finality(
446                        &state,
447                        &epoch_store,
448                        query.transaction_digest,
449                        query.include_details,
450                    ) => {
451                        let _ = tx_sender
452                            .send(Ok(((query.transaction_digest, update), weight)))
453                            .await;
454                    }
455                }
456            });
457        }
458
459        Ok(ReceiverStream::new(rx))
460    }

462    /// Waits for a single transaction to reach finality. Returns immediately
463    /// if already executed, otherwise blocks up to the timeout. The returned
464    /// `Weight` is the per-item spam contribution decided by the path taken:
465    /// a fast-path cache hit means the client is querying an already-finalized
466    /// tx (duplicate/redundant query, `Weight::one()`), while every wait-path
467    /// outcome corresponds to legitimate polling for an in-flight tx and is
468    /// weighted as `Weight::zero()`.
469    async fn wait_for_tx_finality(
470        state: &Arc<AuthorityState>,
471        epoch_store: &Arc<AuthorityPerEpochStore>,
472        tx_digest: TransactionDigest,
473        include_details: bool,
474    ) -> (TxStatusUpdate, Weight) {
475        // Fast path: already executed → duplicate query, spam-like.
476        let cache = state.get_transaction_cache_reader();
477        match cache.try_get_executed_effects(&tx_digest) {
478            Ok(Some(effects)) => {
479                return (
480                    Self::build_executed_update(state, epoch_store, effects, include_details),
481                    Weight::one(),
482                );
483            }
484            Err(e) => {
485                tracing::warn!(?tx_digest, "failed to read effects cache: {e}");
486                // Fall through to the wait path.
487            }
488            Ok(None) => {}
489        }
490
491        // Wait for execution, rejection, epoch end, or timeout. All outcomes
492        // below are legitimate polling for an in-flight tx → Weight::zero().
493        let result = tokio::time::timeout(
494            std::time::Duration::from_secs(GET_TX_STATUS_TIMEOUT_SECS),
495            epoch_store.within_alive_epoch(async {
496                let digests_to_watch = [tx_digest];
497                tokio::select! {
498                    biased;
499                    effects_digests = cache.notify_read_executed_effects_digests("ValidatorService::notify_read_executed_effects_digests", &digests_to_watch) => {
500                        Either::Left(effects_digests)
501                    }
502                    dropped_error = epoch_store.notify_read_dropped_digests(tx_digest) => {
503                        Either::Right(dropped_error)
504                    }
505                }
506            }),
507        )
508            .await;
509
510        let update = match result {
511            // Epoch ended before execution or rejection.
512            Ok(Err(_)) => TxStatusUpdate::Expired {
513                epoch: epoch_store.epoch(),
514            },
515            Ok(Ok(Either::Left(effects_digests))) => {
516                let Some(effects_digest) = effects_digests.into_iter().next() else {
517                    tracing::warn!(
518                        ?tx_digest,
519                        "empty effects from notify_read, returning Expired"
520                    );
521                    return (
522                        TxStatusUpdate::Expired {
523                            epoch: epoch_store.epoch(),
524                        },
525                        Weight::zero(),
526                    );
527                };
528                if let Err(error) = state.check_effects_against_previously_signed(
529                    epoch_store,
530                    &tx_digest,
531                    &effects_digest,
532                    "get_tx_status",
533                ) {
534                    return (TxStatusUpdate::Rejected { error }, Weight::zero());
535                }
536                let details = if include_details {
537                    match cache.try_get_executed_effects(&tx_digest) {
538                        Ok(Some(effects)) => Some(Self::build_executed_data(state, &effects)),
539                        Ok(None) => {
540                            tracing::warn!(
541                                ?tx_digest,
542                                "effects disappeared between notification and fetch, \
543                                 returning Executed without details"
544                            );
545                            None
546                        }
547                        Err(e) => {
548                            tracing::warn!(
549                                ?tx_digest,
550                                "failed to read effects: {e}, returning Executed without details"
551                            );
552                            None
553                        }
554                    }
555                } else {
556                    None
557                };
558                TxStatusUpdate::Executed {
559                    effects_digest,
560                    details,
561                }
562            }
563            Ok(Ok(Either::Right(dropped_error))) => TxStatusUpdate::Rejected {
564                error: dropped_error,
565            },
566            Err(_timeout) => TxStatusUpdate::Expired {
567                epoch: epoch_store.epoch(),
568            },
569        };
570        (update, Weight::zero())
571    }

573    /// Builds a `TxStatusUpdate::Executed` from known effects, optionally
574    /// including full details.
575    fn build_executed_update(
576        state: &Arc<AuthorityState>,
577        epoch_store: &AuthorityPerEpochStore,
578        effects: TransactionEffects,
579        include_details: bool,
580    ) -> TxStatusUpdate {
581        let effects_digest = effects.digest();
582        if let Err(error) = state.check_effects_against_previously_signed(
583            epoch_store,
584            effects.transaction_digest(),
585            &effects_digest,
586            "get_tx_status",
587        ) {
588            return TxStatusUpdate::Rejected { error };
589        }
590        let details = if include_details {
591            Some(Self::build_executed_data(state, &effects))
592        } else {
593            None
594        };
595        TxStatusUpdate::Executed {
596            effects_digest,
597            details,
598        }
599    }

601    /// Fetches execution details (events, input/output objects) for a
602    /// transaction.
603    fn build_executed_data(
604        state: &Arc<AuthorityState>,
605        effects: &TransactionEffects,
606    ) -> Box<ExecutedData> {
607        let events = if effects.events_digest().is_some() {
608            state
609                .get_transaction_events(effects.transaction_digest())
610                .ok()
611        } else {
612            None
613        };
614        let input_objects = state
615            .get_transaction_input_objects(effects)
616            .ok()
617            .unwrap_or_default();
618        let output_objects = state
619            .get_transaction_output_objects(effects)
620            .ok()
621            .unwrap_or_default();
622        Box::new(ExecutedData {
623            effects: effects.clone(),
624            events,
625            input_objects,
626            output_objects,
627        })
628    }

630    async fn notify_capabilities_impl(
631        &self,
632        request: HandleCapabilityNotificationRequestV1,
633    ) -> Result<(HandleCapabilityNotificationResponseV1, Weight), tonic::Status> {
634        let epoch_store = self.state.load_epoch_store_one_call_per_task();
635
636        fp_ensure!(
637            epoch_store
638                .protocol_config()
639                .track_non_committee_eligible_validators(),
640            IotaError::UnsupportedFeature {
641                error: "capability notification endpoint is not supported in this Protocol Version"
642                    .to_string()
643            }
644            .into()
645        );
646
647        fp_ensure!(
648            !self.state.is_fullnode(&epoch_store),
649            IotaError::FullNodeCantHandleValidatorV2.into()
650        );
651
652        let existing_capabilities = epoch_store.get_capabilities_v1()?;
653        let incoming_capability = request.message.data();
654
655        tracing::info!(
656            "Received capability notification: {:?}",
657            incoming_capability
658        );
659
660        if let Some(existing) = existing_capabilities
661            .iter()
662            .find(|cap| cap.authority == incoming_capability.authority)
663        {
664            if incoming_capability.generation <= existing.generation {
665                return Ok((
666                    HandleCapabilityNotificationResponseV1 { _unused: false },
667                    Weight::one(),
668                ));
669            }
670        }
671
672        if let Err(error) = self.consensus_adapter.check_consensus_overload() {
673            self.metrics
674                .num_rejected_capability_notifications_during_overload
675                .with_label_values(&[error.as_ref()])
676                .inc();
677            return Err(error.into());
678        }
679
680        let _metrics_guard = self
681            .metrics
682            .handle_capability_notification_latency
683            .start_timer();
684
685        let signed_authority_capabilities = request.message;
686        let verified_authority_capabilities = epoch_store
687            .verify_authority_capabilities(signed_authority_capabilities)
688            .inspect_err(|_e| {
689                self.metrics.signature_errors.inc();
690            })?;
691
692        let authority_name = verified_authority_capabilities.authority;
693        // Process the verified capabilities
694        tracing::debug!("Verified capability notification for authority {authority_name:?}");
695
696        let transaction = ConsensusTransaction::new_signed_capability_notification_v1(
697            verified_authority_capabilities.into_inner(),
698        );
699
700        self.consensus_adapter
701            .submit(transaction, None, &epoch_store)?;
702
703        tracing::debug!(
704            "Submitted capability notification to consensus for authority {authority_name:?}"
705        );
706
707        Ok((
708            HandleCapabilityNotificationResponseV1 { _unused: false },
709            Weight::one(),
710        ))
711    }
712
713    fn health_check_impl(
714        &self,
715        _request: ValidatorHealthRequest,
716    ) -> Result<(ValidatorHealthResponse, Weight), tonic::Status> {
717        let epoch_store = self.state.load_epoch_store_one_call_per_task();
718
719        let last_locally_built_checkpoint = epoch_store
720            .last_built_checkpoint_summary()
721            .map_err(|e| {
722                tonic::Status::internal(format!(
723                    "Failed to read last built checkpoint summary: {e}"
724                ))
725            })?
726            .map(|(seq, _)| seq)
727            .unwrap_or(0);
728
729        Ok((
730            ValidatorHealthResponse {
731                num_inflight_execution_transactions: self
732                    .state
733                    .execution_scheduler()
734                    .num_pending_transactions()
735                    as u64,
736                num_inflight_consensus_transactions: self
737                    .consensus_adapter
738                    .num_inflight_transactions(),
739                last_locally_built_checkpoint,
740            },
741            Weight::zero(),
742        ))
743    }
744}
