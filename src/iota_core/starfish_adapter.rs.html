24/// Gets a client to submit transactions to Starfish, or waits for one to be
25/// available. This hides the complexities of async consensus initialization and
26/// submitting to different instances of consensus across epochs.
27#[derive(Default, Clone)]
28pub struct LazyStarfishClient {
29    client: Arc<ArcSwapOption<TransactionClient>>,
30}
31
32impl LazyStarfishClient {
33    pub fn new() -> Self {
34        Self {
35            client: Arc::new(ArcSwapOption::empty()),
36        }
37    }
38
39    async fn get(&self) -> Guard<Option<Arc<TransactionClient>>> {
40        let client = self.client.load();
41        if client.is_some() {
42            return client;
43        }
44
45        // Consensus client is initialized after validators or epoch starts, and cleared
46        // after an epoch ends. But calls to get() can happen during validator
47        // startup or epoch change, before consensus finished initializations.
48
49        // TODO: https://github.com/iotaledger/iota/issues/8359
50        // Consider listening to updates from consensus manager instead of polling.
51        let mut count = 0;
52        let start = Instant::now();
53        const RETRY_INTERVAL: Duration = Duration::from_millis(100);
54        loop {
55            let client = self.client.load();
56            if client.is_some() {
57                return client;
58            } else {
59                sleep(RETRY_INTERVAL).await;
60                count += 1;
61                if count % 100 == 0 {
62                    warn!(
63                        "Waiting for consensus to initialize after {:?}",
64                        Instant::now() - start
65                    );
66                }
67            }
68        }
69    }
70
71    pub fn set(&self, client: Arc<TransactionClient>) {
72        self.client.store(Some(client));
73    }
74
75    pub fn clear(&self) {
76        self.client.store(None);
77    }
78}

80#[async_trait::async_trait]
81impl ConsensusClient for LazyStarfishClient {
82    async fn submit(
83        &self,
84        transactions: &[ConsensusTransaction],
85        _epoch_store: &Arc<AuthorityPerEpochStore>,
86    ) -> IotaResult<BlockStatusReceiver> {
87        // TODO: https://github.com/iotaledger/iota/issues/8354
88        // The retrieved TransactionClient can be from the past epoch. Submit would fail
89        // after Starfish shuts down, so there should be no correctness issue.
90        let client = self.get().await;
91        let transactions_bytes = transactions
92            .iter()
93            .map(|t| bcs::to_bytes(t).expect("Serializing consensus transaction cannot fail"))
94            .collect::<Vec<_>>();
95
96        let (block_ref, status_waiter) = client
97            .as_ref()
98            .expect("Client should always be returned")
99            .submit(transactions_bytes)
100            .await
101            .tap_err(|err| {
102                // Will be logged by caller as well.
103                let msg = format!("Transaction submission failed with: {err:?}");
104                match err {
105                    ClientError::ConsensusShuttingDown(_) => {
106                        info!("{}", msg);
107                    }
108                    ClientError::OversizedTransaction(_, _)
109                    | ClientError::OversizedTransactionBundleBytes(_, _)
110                    | ClientError::OversizedTransactionBundleCount(_, _) => {
111                        if cfg!(debug_assertions) {
112                            panic!("{}", msg);
113                        } else {
114                            error!("{}", msg);
115                        }
116                    }
117                };
118            })
119            .map_err(|err| IotaError::FailedToSubmitToConsensus(err.to_string()))?;
120
121        let is_soft_bundle = transactions.len() > 1;
122
123        if !is_soft_bundle
124            && matches!(
125                transactions[0].kind,
126                ConsensusTransactionKind::EndOfPublish(_)
127                    | ConsensusTransactionKind::CapabilityNotificationV1(_)
128                    | ConsensusTransactionKind::RandomnessDkgMessage(_, _)
129                    | ConsensusTransactionKind::RandomnessDkgConfirmation(_, _)
130                    | ConsensusTransactionKind::OverloadNotificationV1(_, _, _)
131                    | ConsensusTransactionKind::TransactionDenyRuleProposal(_)
132            )
133        {
134            let transaction_key = SequencedConsensusTransactionKey::External(transactions[0].key());
135            tracing::info!("Transaction {transaction_key:?} was included in {block_ref}",)
136        };
137
138        // Transform the status waiter into a receiver that can be used by the consensus
139        // adapter.
140
141        let (tx_internal, rx_internal) =
142            tokio::sync::oneshot::channel::<consensus_adapter::BlockStatusInternal>();
143
144        tokio::spawn(async move {
145            if let Ok(status) = status_waiter.await {
146                let _ = tx_internal.send(status.into());
147            }
148        });
149
150        Ok(rx_internal)
151    }
152}
