18pub struct EpochMetrics {
19    /// The current epoch ID. This is updated only when the AuthorityState
20    /// finishes reconfiguration.
21    pub current_epoch: IntGauge,
22
23    /// Current voting right of the validator in the protocol. Updated at the
24    /// start of epochs.
25    pub current_voting_right: IntGauge,
26
27    /// Total duration of the epoch. This is measured from when the current
28    /// epoch store is opened, until the current epoch store is replaced
29    /// with the next epoch store.
30    pub epoch_total_duration: IntGauge,
31
32    /// Number of checkpoints in the epoch.
33    pub epoch_checkpoint_count: IntGauge,
34
35    /// Number of transactions in the epoch.
36    pub epoch_transaction_count: IntGauge,
37
38    /// Total amount of gas rewards (i.e. computation gas cost) in the epoch.
39    pub epoch_total_gas_reward: IntGauge,
40
41    // An active validator reconfigures through the following steps:
42    // 1. Halt validator (a.k.a. close epoch) and stop accepting user transaction certs.
43    // 2. Finishes processing all pending certificates and then send EndOfPublish message.
44    // 3. Stop accepting messages from consensus after seeing 2f+1 EndOfPublish messages.
45    // 4. Creating the last checkpoint of the epoch by augmenting it with AdvanceEpoch transaction.
46    // 5. CheckpointExecutor finishes executing the last checkpoint, and triggers reconfiguration.
47    // 6. During reconfiguration, we tear down consensus, reconfigure state (at which point we
48    //    opens up user certs), and start consensus again.
49    // 7. After reconfiguration, and eventually consensus starts successfully, at some point the
50    //    first checkpoint of the new epoch will be created.
51    // We introduce various metrics to cover the latency of above steps.
52    /// The duration from when the epoch is closed (i.e. validator halted) to
53    /// when all pending certificates are processed (i.e. ready to send
54    /// EndOfPublish message). This is the duration of (1) through (2)
55    /// above.
56    pub epoch_pending_certs_processed_time_since_epoch_close_ms: IntGauge,
57
58    /// The interval from when the epoch is closed to when we receive 2f+1
59    /// EndOfPublish messages. This is the duration of (1) through (3)
60    /// above.
61    pub epoch_end_of_publish_quorum_time_since_epoch_close_ms: IntGauge,
62
63    /// The interval from when the epoch is closed to when we created the last
64    /// checkpoint of the epoch.
65    /// This is the duration of (1) through (4) above.
66    pub epoch_last_checkpoint_created_time_since_epoch_close_ms: IntGauge,
67
68    /// The interval from when the epoch is closed to when we finished executing
69    /// the last transaction of the checkpoint (and hence triggering
70    /// reconfiguration process). This is the duration of (1) through (5)
71    /// above.
72    pub epoch_reconfig_start_time_since_epoch_close_ms: IntGauge,
73
74    /// The total duration when this validator is halted, and hence does not
75    /// accept certs from users. This is the duration of (1) through (6)
76    /// above, and is the most important latency metric reflecting
77    /// reconfiguration delay for each validator.
78    pub epoch_validator_halt_duration_ms: IntGauge,
79
80    /// The interval from when the epoch begins (i.e. right after state
81    /// reconfigure, when the new epoch_store is created), to when the first
82    /// checkpoint of the epoch is ready for creation locally. This is (7)
83    /// above, and is a good proxy to how long it takes for the validator to
84    /// become useful in the network after reconfiguration.
85    // TODO: This needs to be reported properly.
86    pub epoch_first_checkpoint_created_time_since_epoch_begin_ms: IntGauge,
87
88    /// Whether we are running in safe mode where reward distribution and
89    /// tokenomics are disabled.
90    pub is_safe_mode: IntGauge,
91
92    /// When building the last checkpoint of the epoch, we execute advance epoch
93    /// transaction once without committing results to the store. It's
94    /// useful to know whether this execution leads to safe_mode, since in
95    /// theory the result could be different from checkpoint executor.
96    pub checkpoint_builder_advance_epoch_is_safe_mode: IntGauge,
97
98    /// Buffer stake current in effect for this epoch
99    pub effective_buffer_stake: IntGauge,
100
101    /// Set to 1 while the random beacon DKG protocol is unavailable after its
102    /// timeout.
103    pub epoch_random_beacon_dkg_failed: IntGauge,
104
105    /// Set to 1 if the random beacon DKG protocol completed after its timeout.
106    /// Process-local: not restored after a mid-epoch restart.
107    pub epoch_random_beacon_dkg_completed_after_timeout: IntGauge,
108
109    /// The number of shares held by this node after the random beacon DKG
110    /// protocol completed.
111    pub epoch_random_beacon_dkg_num_shares: IntGauge,
112
113    /// The amount of time taken from epoch start to completion of random beacon
114    /// DKG protocol, for the most recent epoch.
115    pub epoch_random_beacon_dkg_epoch_start_completion_time_ms: IntGauge,
116
117    /// The amount of time taken to complete random beacon DKG protocol from the
118    /// time it was started (which may be a bit after the epcoh began), for
119    /// the most recent epoch.
120    pub epoch_random_beacon_dkg_completion_time_ms: IntGauge,
121
122    /// The amount of time taken to start first phase of the random beacon DKG
123    /// protocol, at which point the node has submitted a DKG Message, for
124    /// the most recent epoch.
125    pub epoch_random_beacon_dkg_message_time_ms: IntGauge,
126
127    /// The amount of time taken to complete first phase of the random beacon
128    /// DKG protocol, at which point the node has submitted a DKG
129    /// Confirmation, for the most recent epoch.
130    pub epoch_random_beacon_dkg_confirmation_time_ms: IntGauge,
131
132    /// The number of consensus output items in the quarantine.
133    pub consensus_quarantine_queue_size: IntGauge,
134
135    /// The highest commit index such that it and every commit below it is
136    /// fully executed. Its lag behind consensus is `last_commit_index` minus
137    /// this.
138    pub handler_object_state_highest_fully_executed_commit: IntGauge,
139
140    /// The number of handler-processed rows held in memory, not yet durable.
141    pub handler_object_state_handler_processed_overlay_entries: IntGauge,
142
143    /// The number of shared inputs, package inputs, dynamic fields and coin
144    /// types of shared inputs post-consensus validation answered from an
145    /// earlier read at the same commit, by kind.
146    pub consensus_handler_validation_reader_reused_verdicts: IntCounterVec,
147
148    /// The number of consensus commits that injected deny-rule update
149    /// transactions.
150    pub deny_rule_updates_injected: IntCounter,
151
152    /// The number of injected deny-rule update transactions (one per chunk
153    /// of the delta).
154    pub deny_rule_update_transactions_injected: IntCounter,
155
156    /// Set to 1, and never cleared, when the `TransactionDenyRules` object
157    /// diverged from the mirrored state at an epoch boundary.
158    pub deny_rule_mirror_divergence: IntGauge,
159
160    /// The number of injected deny-rule update transactions whose execution
161    /// failed — always an invariant violation.
162    pub deny_rule_update_execution_failures: IntCounter,
163
164    /// Milliseconds from the start of the last epoch change to each stage.
165    pub epoch_change_stage_ms: IntGaugeVec,
166
167    /// Unix time in milliseconds at which the last epoch change started.
168    pub epoch_change_start_timestamp_ms: IntGauge,
169
170    /// The epoch entered by the change being recorded, and when it started.
171    epoch_change: Mutex<Option<(EpochId, Instant)>>,
172}

372    /// Starts recording the stages of the change to `target_epoch`, measured
373    /// from now, and clears the stages of the previous change. Call it once
374    /// the last checkpoint of the previous epoch has been executed.
375    pub fn begin_epoch_change(&self, target_epoch: EpochId) {
376        let start_unix_ms = SystemTime::now()
377            .duration_since(UNIX_EPOCH)
378            .unwrap_or_default()
379            .as_millis();
380        let mut epoch_change = self.epoch_change.lock();
381        self.epoch_change_stage_ms.reset();
382        self.epoch_change_start_timestamp_ms
383            .set(start_unix_ms as i64);
384        *epoch_change = Some((target_epoch, Instant::now()));
385    }
386
387    /// Records that the change to `target_epoch` reached `stage` at `at`.
388    /// Ignored unless that change was begun in this process, so a restart or
389    /// a later call from an older epoch never overwrites a newer value.
390    pub(crate) fn record_epoch_change_stage(
391        &self,
392        target_epoch: EpochId,
393        stage: &'static str,
394        at: Instant,
395    ) {
396        let epoch_change = self.epoch_change.lock();
397        let Some((_, start)) = epoch_change
398            .as_ref()
399            .filter(|(epoch, _)| *epoch == target_epoch)
400        else {
401            return;
402        };
403        self.epoch_change_stage_ms
404            .with_label_values(&[stage])
405            .set(at.saturating_duration_since(*start).as_millis() as i64);
406    }
407
408    /// Records that the change to `target_epoch` reached `stage` now.
409    pub fn record_epoch_change_stage_now(&self, target_epoch: EpochId, stage: &'static str) {
410        self.record_epoch_change_stage(target_epoch, stage, Instant::now());
411    }

414#[cfg(test)]
415mod tests {
416    use std::time::Duration;
417
418    use super::*;
419
420    #[test]
421    fn epoch_change_stages_need_a_begun_change_of_the_same_epoch() {
422        let metrics = EpochMetrics::new(&Registry::new());
423        let stage = |name| {
424            metrics
425                .epoch_change_stage_ms
426                .with_label_values(&[name])
427                .get()
428        };
429        metrics.record_epoch_change_stage_now(5, "no_change");
430        metrics.begin_epoch_change(4);
431        metrics.record_epoch_change_stage(
432            4,
433            "previous_change",
434            Instant::now() + Duration::from_millis(1),
435        );
436        assert!(stage("previous_change") >= 1);
437
438        metrics.begin_epoch_change(5);
439        metrics.record_epoch_change_stage_now(4, "other_epoch");
440        metrics.record_epoch_change_stage(5, "stage", Instant::now() + Duration::from_millis(30));
441
442        assert!(stage("stage") >= 30);
443        assert!(metrics.epoch_change_start_timestamp_ms.get() > 0);
444        // Beginning a change clears the stages of the previous one.
445        for ignored in ["no_change", "previous_change", "other_epoch"] {
446            assert_eq!(stage(ignored), 0);
447        }
448    }
