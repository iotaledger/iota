5use iota_config::NodeConfig;
6use tokio::runtime::Runtime;
7
8/// Minimum number of worker threads for the serving runtime.
9const MIN_SERVING_THREADS: usize = 2;
10/// Minimum number of worker threads for the node-core runtime.
11const MIN_NODE_THREADS: usize = 4;
12/// Number of worker threads for the metrics runtime. Small and mostly idle;
13/// sits outside [`size_worker_threads`].
14const METRICS_THREADS: usize = 2;
15
16/// Number of CPU cores available to the process, falling back to 8 when it
17/// cannot be determined.
18pub fn available_cpu_cores() -> usize {
19    std::thread::available_parallelism()
20        .map(|n| n.get())
21        .unwrap_or(8)
22}
23
24/// Sizes the node-core and serving runtimes, returning `(node_threads,
25/// serving_threads)`.
26///
27/// The core always gets one worker per available core, so it can use the
28/// whole machine while serving is idle. The serving workers are added on top,
29/// deliberately oversubscribing the machine: a worker blocked in RocksDB
30/// frees its core for the other runtime, and when both runtimes are saturated
31/// the OS schedules fairly per thread, so the serving pool's size relative to
32/// the core pool bounds the CPU share serving can take. A validator caps
33/// serving at a quarter of the cores (at most ~20% of the machine under
34/// saturation) to protect consensus and execution; a fullnode, whose primary
35/// job is serving reads, allows half (~33%).
36fn size_worker_threads(available: usize, is_validator: bool) -> (usize, usize) {
37    let node_threads = available.max(MIN_NODE_THREADS);
38    let serving_threads = if is_validator {
39        available / 4
40    } else {
41        available / 2
42    }
43    .max(MIN_SERVING_THREADS);
44    (node_threads, serving_threads)
45}
46
47pub struct IotaRuntimes {
48    // Order in this struct is the order in which runtimes are stopped.
49    /// Client-facing servers (validator gRPC, JSON-RPC, gRPC read API) and the
50    /// per-request handlers they spawn. Isolated from `iota_node` so that a
51    /// flood of external requests cannot starve the node core. Stopped first so
52    /// that client requests are cut off before the node core they call into is
53    /// torn down.
54    pub serving: Runtime,
55    /// Node core: consensus, execution, state sync, checkpoints and p2p
56    /// networking.
57    pub iota_node: Runtime,
58    pub metrics: Runtime,
59}
60
61impl IotaRuntimes {
62    pub fn new(config: &NodeConfig) -> Self {
63        let is_validator = config.is_validator();
64        let (node_threads, serving_threads) =
65            size_worker_threads(available_cpu_cores(), is_validator);
66
67        let iota_node = tokio::runtime::Builder::new_multi_thread()
68            .thread_name("iota-node-runtime")
69            .worker_threads(node_threads)
70            .enable_all()
71            .build()
72            .unwrap();
73        let serving = tokio::runtime::Builder::new_multi_thread()
74            .thread_name("serving-runtime")
75            .worker_threads(serving_threads)
76            .enable_all()
77            .build()
78            .unwrap();
79        let metrics = tokio::runtime::Builder::new_multi_thread()
80            .thread_name("metrics-runtime")
81            .worker_threads(METRICS_THREADS)
82            .enable_all()
83            .build()
84            .unwrap();
85
86        Self {
87            serving,
88            iota_node,
89            metrics,
90        }
91    }
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97
98    #[test]
99    fn available_cpu_cores_is_never_zero() {
100        assert!(super::available_cpu_cores() >= 1);
101    }
102
103    #[test]
104    fn validator_serving_gets_a_quarter_of_the_cores() {
105        assert_eq!(size_worker_threads(16, true), (16, 4));
106        assert_eq!(size_worker_threads(8, true), (8, 2));
107    }
108
109    #[test]
110    fn fullnode_serving_gets_half_of_the_cores() {
111        assert_eq!(size_worker_threads(16, false), (16, 8));
112        assert_eq!(size_worker_threads(12, false), (12, 6));
113    }
114
115    #[test]
116    fn the_core_always_gets_one_thread_per_core() {
117        for available in MIN_NODE_THREADS..=256 {
118            for is_validator in [true, false] {
119                let (node, _) = size_worker_threads(available, is_validator);
120                assert_eq!(
121                    node, available,
122                    "core throttled below the machine size with {available} cores \
123                     (is_validator: {is_validator})",
124                );
125            }
126        }
127    }
128
129    #[test]
130    fn both_runtimes_always_get_their_minimum_threads() {
131        for available in 1..=256 {
132            for is_validator in [true, false] {
133                let (node, serving) = size_worker_threads(available, is_validator);
134                assert!(node >= MIN_NODE_THREADS);
135                assert!(serving >= MIN_SERVING_THREADS);
136            }
137        }
138    }
139}
