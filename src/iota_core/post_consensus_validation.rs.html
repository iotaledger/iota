4//! Post-consensus validation and owned-object conflict resolution for
5//! `UserTransactionV1` transactions.
6//!
7//! This module merges two formerly separate pipeline stages into a single pass:
8//!
9//! 1. **Semantic validation** — deduplication, already-executed check,
10//!    structural validity, and deny checks (deny lists, gas, ownership, coin
11//!    deny list, Move authenticator).
12//! 2. **Owned-object conflict resolution** (white-flag) — three-tier lock check
13//!    and lock acquisition.
14//!
15//! Processing both in one loop avoids iterating over every transaction twice
16//! and skips expensive validation for transactions that can't acquire object
17//! locks anyway.
18//!
19//! # Per-transaction order within the loop
20//!
21//! 1. Non-`UserTransactionV1` — pass through unchanged.
22//! 2. Dedup by `ConsensusTransactionKey` — silent drop.
23//! 3. Already executed — **retained** as a committee-agreed winner (registers
24//!    the locks its own effects report, skips re-validation); not dropped. See
25//!    issue #11649.
26//! 4. `validity_check()` — drop with error.
27//! 5. Three-tier lock conflict check (local HashMap → quarantine → DB) — drop
28//!    with error, except a lock held by the same transaction (a deferred tx's
29//!    own prior-round lock), which is exempt. Cheap; performed before expensive
30//!    checks.
31//! 6. `handle_transaction_validation_checks()` — drop with error. Only reached
32//!    when all locks are free.
33//! 7. All passed — acquire locks in the local tracking map, keep transaction.
34//!
35//! Non-`UserTransactionV1` transactions pass through unchanged.
36//!
37//! # Which references get locked
38//!
39//! Step 5 probes every `ImmOrOwnedMoveObject` reference, but with
40//! `pcool_skip_immutable_object_locks` set only genuinely owned references are
41//! locked — an immutable input never takes a new version, so its lock would
42//! never lapse (issue #12602). Step 7 filters via the loaded objects, step 3
43//! via the executed transaction's effects; without the flag both lock the raw
44//! probed set.
45

72/// Validates `UserTransactionV1` transactions and resolves owned-object
73/// conflicts in a single pass.
74///
75/// For each `UserTransactionV1` in consensus order:
76/// - Runs deduplication, structural validity, lock conflict check, and deny
77///   checks (deny list, gas, ownership, coin deny list, Move authenticator).
78/// - If all checks pass, acquires owned-object locks in a local tracking map;
79///   with `pcool_skip_immutable_object_locks` set, immutable inputs are not
80///   locked (issue #12602).
81/// - Drops the transaction (with an error) on any failure.
82/// - An already-executed transaction is **retained** (not dropped): it
83///   registers the locks its effects report (the raw input set without the
84///   flag) and skips re-validation. See issue #11649.
85///
86/// Non-`UserTransactionV1` transactions pass through unchanged.
87///
88/// # Arguments
89///
90/// * `authority_state` — Used for cache reads and deny checks.
91/// * `epoch_store` — Current epoch store (protocol config, lock storage,
92///   governance deny rules).
93/// * `transactions` — All sequenced transactions for this consensus commit;
94///   modified in-place.
95///
96/// # Returns
97///
98/// `Ok((dropped, locks, all_user_tx_digests))` where:
99/// - `dropped` — `(digest, error)` for every semantically-invalid or
100///   lock-conflicting transaction. Silent drops (duplicates) are **not**
101///   included.
102/// - `locks` — Owned-object locks acquired in this commit, to be stored in the
103///   consensus quarantine so subsequent commits can see them.
104/// - `all_user_tx_digests` — Every `UserTransactionV1` digest that passed dedup
105///   (both kept and dropped). Used by the caller to release pre-consensus soft
106///   locks.
107pub async fn validate_and_resolve_conflicts(
108    authority_state: &AuthorityState,
109    epoch_store: &Arc<AuthorityPerEpochStore>,
110    transactions: &mut Vec<VerifiedSequencedConsensusTransaction>,
111) -> IotaResult<(
112    Vec<(TransactionDigest, IotaError)>,
113    HashMap<ObjectReference, LockDetails>,
114    Vec<TransactionDigest>,
115)> {
116    let mut dropped: Vec<(TransactionDigest, IotaError)> = Vec::new();
117    let mut seen_keys: HashSet<SequencedConsensusTransactionKey> = HashSet::new();
118    // Locks acquired within this commit. Populated for every transaction that
119    // passes all checks; used by subsequent transactions' conflict checks.
120    let mut current_commit_locks: HashMap<ObjectReference, LockDetails> = HashMap::new();
121    // Index-parallel keep flags: true = keep, false = remove.
122    let mut keep = vec![true; transactions.len()];
123    // All UserTransactionV1 digests seen in this commit (both kept and dropped),
124    // used by the caller to release pre-consensus soft locks.
125    let mut all_user_tx_digests = Vec::with_capacity(transactions.len());
126
127    // One deny-rule snapshot for the whole commit, so every transaction in it
128    // is judged by the same set. With governance enabled this must be the
129    // consensus-derived set — local config can differ between validators and
130    // would fork the post-consensus decisions.
131    let governance_deny_rules;
132    let deny_config: &dyn DenyRuleConfig = if epoch_store.protocol_config().deny_rule_governance() {
133        governance_deny_rules = epoch_store.get_active_transaction_deny_rules();
134        governance_deny_rules.as_ref()
135    } else {
136        &authority_state.config.transaction_deny_config
137    };
138
139    // Read once for the whole commit: the protocol config is fixed for the epoch.
140    let skip_immutable_locks = epoch_store
141        .protocol_config()
142        .pcool_skip_immutable_object_locks();
143
144    for (i, tx) in transactions.iter().enumerate() {
145        // Check #0: Dedup by ConsensusTransactionKey.
146        // The same UserTransactionV1 may appear in DAG blocks from multiple
147        // validators within the same consensus commit. Only the first occurrence
148        // is kept. Silent drop — not added to `dropped`.
149        if !seen_keys.insert(tx.0.key()) {
150            keep[i] = false;
151            continue;
152        }
153
154        // Only validate UserTransactionV1; pass everything else through unchanged.
155        let Some(transaction) = (match &tx.0.transaction {
156            SequencedConsensusTransactionKind::External(ext) => ext.kind.as_user_transaction(),
157            SequencedConsensusTransactionKind::System(_) => None,
158        }) else {
159            continue;
160        };
161
162        let digest = *transaction.digest();
163        all_user_tx_digests.push(digest);
164
165        // Check #1: Already executed (typically by state-sync before this node's
166        // consensus handler reached the commit). It is a committee-agreed winner, so
167        // keep it in the sequence to flow into checkpoint roots like on every other
168        // validator (dropping it forks — issue #11649). Register its owned-object
169        // locks so double-spend siblings still lose, then skip re-validation (#2/#5);
170        // the active scheduler's enqueue filter suppresses the re-execution.
171        if authority_state
172            .get_transaction_cache_reader()
173            .try_is_tx_already_executed(&digest)?
174        {
175            // Byte-based, so safe even though the inputs are already consumed.
176            let owned_inputs = extract_owned_input_objects(tx)?;
177            let owned_inputs = if skip_immutable_locks {
178                filter_locked_inputs_by_effects(authority_state, &digest, owned_inputs)?
179            } else {
180                owned_inputs
181            };
182            for obj_ref in &owned_inputs {
183                // A winner cannot be out-locked: after the effects filter,
184                // every reference here was consumed by this transaction, so a
185                // lock held by a different tx is a consistency violation, not
186                // a conflict. (Flag off, immutable references stay in the set
187                // and weaken this invariant — issue #12602.)
188                if let Some(other) =
189                    find_existing_lock(obj_ref, &current_commit_locks, epoch_store)?
190                {
191                    if other != digest {
192                        fatal!(
193                            "already-executed transaction {:?} has owned input {:?} \
194                             locked by a different transaction {:?}",
195                            digest,
196                            obj_ref,
197                            other,
198                        );
199                    }
200                }
201                current_commit_locks.insert(*obj_ref, digest);
202            }
203            debug!(
204                ?digest,
205                num_owned_inputs = owned_inputs.len(),
206                "Transaction already executed; retained as checkpoint root, skipping re-validation"
207            );
208            // keep[i] stays true so the transaction remains in the sequence.
209            continue;
210        }
211
212        // Check #2: Structural validity.
213        if let Err(e) = transaction.validity_check(&epoch_store.tx_validity_check_context()) {
214            warn!(
215                ?digest,
216                error = ?e,
217                "UserTransactionV1 failed validity_check post-consensus, dropping"
218            );
219            dropped.push((digest, e));
220            keep[i] = false;
221            continue;
222        }
223
224        // Check #3: Extract owned input objects for lock conflict detection.
225        let owned_inputs = match extract_owned_input_objects(tx) {
226            Ok(inputs) => inputs,
227            Err(e) => {
228                warn!(
229                    ?digest,
230                    error = ?e,
231                    "Failed to extract owned input objects post-consensus, dropping"
232                );
233                dropped.push((digest, e));
234                keep[i] = false;
235                continue;
236            }
237        };
238
239        // Check #4: Three-tier lock conflict check.
240        // Cheap (HashMap + quarantine + DB lookups); performed before the
241        // expensive deny checks so conflicting transactions are filtered first.
242        //
243        // Locks are keyed by full ObjectReference (id + version + digest), not just
244        // ObjectID. Two transactions referencing the same object at different
245        // versions will NOT conflict here — version freshness is validated
246        // later in Check #5 (deny checks load objects from DB and verify
247        // that the transaction's input refs match the current state).
248        //
249        // Probing the raw set is safe though immutable inputs are never
250        // locked: acquisition never inserts one, and the lock tables are
251        // epoch-scoped while the protocol config is fixed per epoch, so no
252        // tier can hold a stale immutable lock.
253        //
254        // Tier 1: Local HashMap (current commit).
255        // Tier 2: Consensus quarantine (previous uncommitted commits).
256        // Tier 3: Persistent DB (committed data).
257        let mut conflict: Option<IotaError> = None;
258        for obj_ref in &owned_inputs {
259            if let Some(locked_by) =
260                find_existing_lock(obj_ref, &current_commit_locks, epoch_store)?
261            {
262                // A lock held by this same transaction is its own lock from a
263                // prior round: it acquired owned-object locks, was then deferred,
264                // and is reloaded this round. A self-held lock is NOT a conflict
265                // - without this guard, the transaction will be dropped with
266                // `ObjectLockConflict` and never executed. Issue #11649 mirrors
267                // this guard for the already-executed branch in Check #1.
268                //
269                // Note that same-commit duplicates cannot slip through here since
270                // Check #0 already dedups by digest, so a same-digest lock here
271                // is always this transaction's own prior-round lock.
272                if locked_by == digest {
273                    continue;
274                }
275
276                debug!(
277                    ?digest,
278                    ?obj_ref,
279                    ?locked_by,
280                    "Transaction conflicts with existing owned-object lock, dropping"
281                );
282                conflict = Some(IotaError::ObjectLockConflict {
283                    obj_ref: *obj_ref,
284                    pending_transaction: locked_by,
285                });
286                break;
287            }
288        }
289        if let Some(e) = conflict {
290            dropped.push((digest, e));
291            keep[i] = false;
292            continue;
293        }
294
295        // Check #5: Deny list, gas, ownership, coin deny list, Move
296        // authenticator. Only reached if all locks are free — skips the
297        // expensive object loading for transactions that would be dropped
298        // by the lock conflict check.
299        //
300        // Safe to skip signature re-verification: the consensus block verifier
301        // (`IotaTxValidator::validate_transactions`) already called
302        // `verify_tx()` on every `UserTransactionV1` before accepting the
303        // block. Re-verifying here would not add safety — if a quorum
304        // committed a bad signature it indicates a protocol-level failure
305        // (2f+1 Byzantine/buggy validators), not something we can recover from
306        // by rejecting the transaction post-consensus. Doing so would also risk
307        // diverging from other honest validators.
308        let verified_tx = VerifiedTransaction::new_from_verified((*transaction).clone());
309        let validated_owned_objects = match authority_state
310            .handle_transaction_validation_checks(
311                &verified_tx,
312                epoch_store,
313                deny_config,
314                // Epoch-gated coin deny-list read: the verdict here decides whether
315                // the transaction stays in the committed set, so it must not depend
316                // on this validator's execution progress.
317                true,
318            )
319            .await
320        {
321            Ok(owned) => owned,
322            Err(e) => {
323                if e.is_storage_or_epoch_error() {
324                    return Err(e);
325                }
326                warn!(
327                    ?digest,
328                    error = ?e,
329                    "UserTransactionV1 failed post-consensus deny checks, dropping"
330                );
331                dropped.push((digest, e));
332                keep[i] = false;
333                continue;
334            }
335        };
336
337        // All checks passed — acquire owned-object locks in local tracking.
338        // The validated set comes from loaded objects, so unlike the
339        // byte-derived `owned_inputs` it excludes immutable inputs, which must
340        // never be locked (issue #12602). It is a subset of `owned_inputs`,
341        // so every reference locked here was probed by Check #4 first.
342        let locked_inputs: Vec<ObjectReference> = if skip_immutable_locks {
343            validated_owned_objects
344        } else {
345            owned_inputs
346        };
347        for obj_ref in &locked_inputs {
348            current_commit_locks.insert(*obj_ref, digest);
349        }
350        // Log the acquired refs, not just their count, so the winner's locks
351        // are attributable per (object_id, version).
352        debug!(
353            ?digest,
354            num_owned_inputs = locked_inputs.len(),
355            owned_inputs = ?locked_inputs,
356            "Transaction passed post-consensus validation, acquired all object locks"
357        );
358    }
359
360    if !dropped.is_empty() {
361        warn!(
362            num_dropped = dropped.len(),
363            num_retained = transactions
364                .iter()
365                .enumerate()
366                .filter(|(i, _)| keep[*i])
367                .count(),
368            "Post-consensus validation dropped transactions"
369        );
370    }
371
372    // Remove invalid/duplicate/conflicting entries using the parallel keep
373    // flags.
374    let mut iter = keep.into_iter();
375    transactions.retain(|_| iter.next().unwrap_or(true));
376
377    Ok((dropped, current_commit_locks, all_user_tx_digests))
378}

380/// Narrows the byte-derived owned inputs of an already-executed transaction to
381/// the references it consumed — the same set the main path locks — by
382/// intersecting them with the pre-execution references in its own effects;
383/// immutable inputs never appear there (issue #12602). A missing effects body
384/// is a storage error, never a "not executed" verdict, which would drop a
385/// committee-agreed winner other validators keep.
386fn filter_locked_inputs_by_effects(
387    authority_state: &AuthorityState,
388    digest: &TransactionDigest,
389    owned_inputs: Vec<ObjectReference>,
390) -> IotaResult<Vec<ObjectReference>> {
391    let effects = authority_state
392        .get_transaction_cache_reader()
393        .try_get_executed_effects(digest)?
394        .ok_or_else(|| IotaError::GenericAuthority {
395            error: format!(
396                "effects of already-executed transaction {digest:?} are unavailable in \
397                 post-consensus validation"
398            ),
399        })?;
400    // Owned inputs always appear in effects and immutable ones never do
401    // (`TemporaryStore::ensure_active_inputs_mutated` force-mutates every
402    // non-immutable owned input); intersecting on the full `ObjectReference`
403    // — the lock table's own key — keeps the sets identical by construction.
404    let consumed: HashSet<ObjectReference> = effects
405        .old_object_metadata()
406        .into_iter()
407        .map(|old| old.reference)
408        .collect();
409    Ok(owned_inputs
410        .into_iter()
411        .filter(|obj_ref| consumed.contains(obj_ref))
412        .collect())
413}

415/// Finds an existing owned-object lock on `obj_ref`, walking three tiers in
416/// order:
417/// 1. `current_commit_locks` — locks acquired earlier in the same commit.
418/// 2. Consensus quarantine — locks from previous uncommitted commits.
419/// 3. Persistent DB — committed locks.
420///
421/// Returns `Ok(Some(locker))` if any tier holds a lock, `Ok(None)` if the
422/// input is free. The caller decides what to do with the result (drop with
423/// `ObjectLockConflict` for a contender, or `fatal!` if a winner is
424/// out-locked).
425fn find_existing_lock(
426    obj_ref: &ObjectReference,
427    current_commit_locks: &HashMap<ObjectReference, LockDetails>,
428    epoch_store: &Arc<AuthorityPerEpochStore>,
429) -> IotaResult<Option<LockDetails>> {
430    if let Some(&locker) = current_commit_locks.get(obj_ref) {
431        return Ok(Some(locker));
432    }
433    if let Some(locker) = epoch_store.get_quarantined_owned_object_lock(obj_ref) {
434        return Ok(Some(locker));
435    }
436    epoch_store.tables()?.get_locked_transaction(obj_ref)
437}

439/// Extracts the `ImmOrOwnedMoveObject` input references of a
440/// `UserTransactionV1` consensus transaction (excludes shared objects and
441/// packages).
442///
443/// This is the set probed for lock conflicts, not the set that gets locked.
444/// It is derived from the transaction bytes alone, where owned and immutable
445/// inputs share one kind, so it cannot tell them apart; the callers that
446/// acquire locks narrow it down first. Probing immutable references is
447/// harmless, since no tier ever holds a lock on one.
448fn extract_owned_input_objects(
449    tx: &VerifiedSequencedConsensusTransaction,
450) -> IotaResult<Vec<ObjectReference>> {
451    let Some(transaction_data) = (match &tx.0.transaction {
452        SequencedConsensusTransactionKind::External(ext) => {
453            ext.kind.as_user_transaction().map(|t| t.data())
454        }
455        SequencedConsensusTransactionKind::System(_) => None,
456    }) else {
457        return Err(IotaError::GenericAuthority {
458            error: "Expected UserTransactionV1 in extract_owned_input_objects".to_string(),
459        });
460    };
461
462    // Use SenderSignedTransaction::input_objects() rather than
463    // Transaction::input_objects() to also include objects coming from
464    // MoveAuthenticator signatures. Those can only be immutable or shared,
465    // so they widen the probe set but never the locked set.
466    let owned_objects = transaction_data
467        .input_objects()?
468        .into_iter()
469        .filter_map(|input| match input {
470            InputObjectKind::ImmOrOwnedMoveObject(obj_ref) => Some(obj_ref),
471            InputObjectKind::SharedMoveObject { .. } => None,
472            InputObjectKind::MovePackage(_) => None,
473        })
474        .collect();
475
476    Ok(owned_objects)
477}
