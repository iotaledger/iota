47/// The horizon of `commit_index`: the highest commit whose rows validation at
48/// `commit_index` relies on, which is the commit minus the protocol config's
49/// horizon distance. The reader treats rows produced above it as missing, and
50/// the handler waits until it is fully executed before validating.
51pub fn horizon(protocol_config: &ProtocolConfig, commit_index: CommitIndex) -> CommitIndex {
52    commit_index.saturating_sub(protocol_config.pcool_deterministic_validation_horizon_distance())
53}

55/// Reads inputs as of one consensus commit. Built once per commit and passed
56/// to every transition that touches a store.
57pub struct CommitIndexedReader {
58    pub(super) cache: Arc<dyn ObjectCacheRead>,
59    pub(super) epoch_store: Arc<AuthorityPerEpochStore>,
60    /// The horizon, `C - K`.
61    horizon: CommitIndex,
62}
63
64impl CommitIndexedReader {
65    pub fn new(
66        cache: Arc<dyn ObjectCacheRead>,
67        epoch_store: Arc<AuthorityPerEpochStore>,
68        commit_index: CommitIndex,
69    ) -> Self {
70        let horizon = horizon(epoch_store.protocol_config(), commit_index);
71        Self {
72            cache,
73            epoch_store,
74            horizon,
75        }
76    }

78    /// Reads one owned or immutable input: the row at the named version, or
79    /// without one the store and then, unless the store holds the named
80    /// reference, the other rows of `id`. A keep from the row first reads the
81    /// rows above it, and every keep loads the object. A storage error is
82    /// returned as `Err`, never as a result.
83    pub fn read_owned(&self, input: ObjectReference) -> IotaResult<OwnedVerdict> {
84        // The row at the named version, when it exists, decides a drop or a
85        // missing. A keep from it goes on to the rows above it.
86        let no_tables_entry =
87            match OwnedReader::start(input, self.horizon).read_handler_row(self)? {
88                TablesLookup::LiveAtVersion(live_at_version) => {
89                    return self.read_rows_above(live_at_version);
90                }
91                TablesLookup::Missing(reason) => return Ok(OwnedVerdict::Missing(reason)),
92                TablesLookup::Drop(reason) => return Ok(OwnedVerdict::Drop(reason)),
93                TablesLookup::NoEntry(no_tables_entry) => no_tables_entry,
94            };
95
96        // No row: read the store, keep its answer, and read the row once more
97        // before using that answer.
98        match no_tables_entry.read_store(self)?.reread_handler_row(self)? {
99            TablesRecheck::LiveAtVersion(live_at_version) => self.read_rows_above(live_at_version),
100            TablesRecheck::NeedBytes(need_bytes) => self.load_bytes(need_bytes),
101            TablesRecheck::NotAtVersion(not_at_version) => self.read_other_rows(not_at_version),
102            TablesRecheck::Missing(reason) => Ok(OwnedVerdict::Missing(reason)),
103            TablesRecheck::Drop(reason) => Ok(OwnedVerdict::Drop(reason)),
104        }
105    }
106
107    /// Drops a kept owned input when a row of `id` above it, produced at or
108    /// below the horizon, shows it consumed. Otherwise loads its object.
109    fn read_rows_above(
110        &self,
111        live_at_version: OwnedReader<LiveAtVersion>,
112    ) -> IotaResult<OwnedVerdict> {
113        self.decide_rows_above(live_at_version.read_rows_above(self)?)
114    }
115
116    /// With no row at the named version and another store answer, the other
117    /// rows of `id` decide, so that a validator ahead of the horizon answers
118    /// as one that is not.
119    fn read_other_rows(
120        &self,
121        not_at_version: OwnedReader<NotAtVersion>,
122    ) -> IotaResult<OwnedVerdict> {
123        self.decide_rows_above(not_at_version.read_other_rows(self)?)
124    }
125
126    fn decide_rows_above(&self, lookup: RowsAboveLookup) -> IotaResult<OwnedVerdict> {
127        match lookup {
128            RowsAboveLookup::NeedBytes(need_bytes) => self.load_bytes(need_bytes),
129            RowsAboveLookup::Missing(reason) => Ok(OwnedVerdict::Missing(reason)),
130            RowsAboveLookup::Drop(reason) => Ok(OwnedVerdict::Drop(reason)),
131        }
132    }
133
134    /// Reads one shared input declared at `initial_shared_version`: whether
135    /// the object exists and is shared at that version. Its contents are not
136    /// read. A drop is checked against the lowest row of `id`, so that a
137    /// validator that executed the creation of `id` above the horizon answers
138    /// as one that did not. A storage error is returned as `Err`, never as a
139    /// result.
140    pub fn read_shared(
141        &self,
142        id: ObjectId,
143        initial_shared_version: Version,
144    ) -> IotaResult<SharedVerdict> {
145        match self.read_shared_rows_and_store(id, initial_shared_version)? {
146            SharedVerdict::Drop(reason) => Ok(
147                match SharedReader::start_drop_check(
148                    id,
149                    initial_shared_version,
150                    self.horizon,
151                    reason,
152                )
153                .read_lowest_row(self)?
154                {
155                    DropCheck::Drop(reason) => SharedVerdict::Drop(reason),
156                    DropCheck::Missing(reason) => SharedVerdict::Missing(reason),
157                },
158            ),
159            verdict => Ok(verdict),
160        }
161    }
162
163    fn read_shared_rows_and_store(
164        &self,
165        id: ObjectId,
166        initial_shared_version: Version,
167    ) -> IotaResult<SharedVerdict> {
168        // The row at the declared initial shared version is the object's
169        // creation row, and decides when it exists. One at or below the
170        // horizon shows the object was created shared at that version, so
171        // only whether the object still exists is left to check.
172        let no_tables_entry = match SharedReader::start(id, initial_shared_version, self.horizon)
173            .read_creation_row(self)?
174        {
175            SharedTablesLookup::Created(created) => {
176                return match created.read_object(self)? {
177                    CreatedObjectLookup::Exists => Ok(SharedVerdict::Exists),
178                    CreatedObjectLookup::Absent(object_absent) => {
179                        self.check_deletion(object_absent)
180                    }
181                };
182            }
183            SharedTablesLookup::Missing(reason) => return Ok(SharedVerdict::Missing(reason)),
184            SharedTablesLookup::Drop(reason) => return Ok(SharedVerdict::Drop(reason)),
185            SharedTablesLookup::NoEntry(no_tables_entry) => no_tables_entry,
186        };
187
188        // No creation row: read the latest object from the store, keep it, and
189        // read the creation row once more before using it.
190        match no_tables_entry
191            .read_object(self)?
192            .reread_creation_row(self)?
193        {
194            SharedTablesRecheck::Created(created) => match created.read_object(self)? {
195                CreatedObjectLookup::Exists => Ok(SharedVerdict::Exists),
196                CreatedObjectLookup::Absent(object_absent) => self.check_deletion(object_absent),
197            },
198            SharedTablesRecheck::Exists => Ok(SharedVerdict::Exists),
199            SharedTablesRecheck::Missing(reason) => Ok(SharedVerdict::Missing(reason)),
200            SharedTablesRecheck::Drop(reason) => Ok(SharedVerdict::Drop(reason)),
201            SharedTablesRecheck::Absent(object_absent) => self.check_deletion(object_absent),
202        }
203    }
204
205    /// Decides a shared input whose object is not in the store, from the
206    /// highest row of `id`. Called after either read of the creation row.
207    fn check_deletion(
208        &self,
209        object_absent: SharedReader<ObjectAbsent>,
210    ) -> IotaResult<SharedVerdict> {
211        Ok(match object_absent.read_deletion_row(self)? {
212            DeletionRowLookup::Deleted(version, digest) => SharedVerdict::Deleted(version, digest),
213            DeletionRowLookup::Drop(reason) => SharedVerdict::Drop(reason),
214            DeletionRowLookup::Missing(reason) => SharedVerdict::Missing(reason),
215        })
216    }

218    /// Reads one package input: whether it is visible at this commit. The
219    /// store is read first, because a package input names no version, then
220    /// the row. The package module doc explains the order. A storage error is
221    /// returned as `Err`, never as a result, and so is an id that names a Move
222    /// object (`UserInputError::MoveObjectAsPackage`).
223    pub fn read_package(&self, id: ObjectId) -> IotaResult<PackageVerdict> {
224        let loaded = match PackageReader::start(id, self.horizon).read_package(self)? {
225            PackageLookup::Loaded(loaded) => loaded,
226            PackageLookup::Missing(reason) => return Ok(PackageVerdict::Missing(reason)),
227        };
228        Ok(match loaded.read_row(self)? {
229            PackageRowLookup::Visible(package) => PackageVerdict::Visible(package),
230            PackageRowLookup::Missing(reason) => PackageVerdict::Missing(reason),
231        })
232    }
233
234    /// Loads the object of a kept input at exactly `(id, V)`. Takes the
235    /// `NeedBytes` step, which only a keep result creates. An object with
236    /// another digest drops the input: a keep through a row that replaced `V`
237    /// has not compared the named digest yet. A shared object named as owned
238    /// is decided from its creation row, see the owned module doc. The pruning
239    /// limit keeps the object of every version a validation can still keep,
240    /// so a missing object is a storage error, which stops commit processing,
241    /// and never a result.
242    fn load_bytes(&self, need_bytes: OwnedReader<NeedBytes>) -> IotaResult<OwnedVerdict> {
243        let input = need_bytes.into_reference();
244        let key = ObjectKey(input.object_id, input.version);
245        if let Some(object) = self.cache.try_get_object_by_key(&key.0, key.1)? {
246            if object.digest() != input.digest {
247                return Ok(OwnedVerdict::Drop(DropReason(
248                    DropKind::StoreDigestMismatch,
249                )));
250            }
251            if let Owner::Shared(initial) = object.owner {
252                let creation_row = self
253                    .epoch_store
254                    .handler_processed_object(&ObjectKey(input.object_id, initial))?;
255                return Ok(
256                    match classify_shared_named_as_owned(creation_row.as_ref(), self.horizon) {
257                        SharedNamedAsOwned::Missing(reason) => OwnedVerdict::Missing(reason),
258                        SharedNamedAsOwned::Drop(reason) => OwnedVerdict::Drop(reason),
259                    },
260                );
261            }
262            return Ok(OwnedVerdict::Keep(KeptObject(object)));
263        }
264        Err(IotaError::Storage(format!(
265            "kept input {key:?} has no bytes in the object store"
266        )))
267    }

269    /// The dynamic field `id` as of the horizon, live or absent. The field
270    /// must change only through its shared parent object, as a field of the
271    /// coin deny list or an account's authenticator function field does.
272    /// Storage errors propagate, and so does a version the read needs but the
273    /// store no longer holds.
274    pub fn read_dynamic_field(&self, id: ObjectId) -> IotaResult<DynamicFieldAtHorizon> {
275        let mut lookup = match DynamicFieldReader::start(id, self.horizon).read_highest_row(self)? {
276            HighestRowLookup::Row(lookup) => lookup,
277            HighestRowLookup::NoRow(no_row) => {
278                no_row.read_object(self)?.reread_highest_row(self)?
279            }
280        };
281        loop {
282            lookup = match lookup {
283                RowLookup::NeedBytes(need_bytes) => return need_bytes.load_bytes(self),
284                RowLookup::Decided(field) => return Ok(field),
285                RowLookup::AboveHorizon(above) => above.read_row_below(self)?,
286            };
287        }
288    }

290    /// Reads the authenticator function field of the account `account_id` as
291    /// of the horizon: [`AccountVerdict::Account`] with the field's object
292    /// when the field was live at the horizon, [`AccountVerdict::Missing`]
293    /// otherwise. Storage errors propagate.
294    pub fn read_account(&self, account_id: ObjectId) -> IotaResult<AccountVerdict> {
295        // Checking the field at the horizon is enough for receives. An object
296        // can receive only while it is not an account, and a receive takes no
297        // lock. So for an account whose field is live at the horizon, every
298        // receive through it is at or below the horizon too, where the owned
299        // input read sees what the receive consumed. The field changes only
300        // through the account object, in the order of the commits that change
301        // it, as `read_dynamic_field` requires.
302        let field_id = derive_authenticator_function_ref_v1_dynamic_field_id(account_id)?;
303        Ok(match self.read_dynamic_field(field_id)? {
304            DynamicFieldAtHorizon::Live(field) => AccountVerdict::Account(field),
305            DynamicFieldAtHorizon::Absent => {
306                AccountVerdict::Missing(MissingReason(MissingKind::AuthenticatorFunctionNotFound))
307            }
308        })
309    }

311    /// The authenticate function that `field`, the authenticator function
312    /// field [`Self::read_account`] read for the account `account_id`, names,
313    /// for the authenticator package deny check at the commit. A package this
314    /// reader sees stays visible to it, so the check reads it again through
315    /// this reader as its package store.
316    ///
317    /// On chain, the field can only be created from the package's metadata
318    /// object, which exists only once the package is published, so a field
319    /// live at the horizon names a package this reader sees. `None` when it
320    /// does not, which only a field written directly to the store can cause;
321    /// the deny check then skips the function.
322    ///
323    /// # Errors
324    ///
325    /// Storage errors, and a field that does not decode.
326    pub fn visible_authenticator_function_ref(
327        &self,
328        account_id: ObjectId,
329        field: &Object,
330    ) -> IotaResult<Option<AuthenticatorFunctionRef>> {
331        let function_ref =
332            authenticator_function_ref_v1_from_dynamic_field_object(account_id, field)?
333                .authenticator_function_ref;
334        let package_id = match &function_ref {
335            AuthenticatorFunctionRef::V1(v1) => v1.package,
336        };
337        Ok(match self.read_package(package_id)? {
338            PackageVerdict::Visible(_) => Some(function_ref),
339            PackageVerdict::Missing(_) => None,
340        })
341    }

343    /// An object of the kept shared input `id` declared at
344    /// `initial_shared_version`, for its type. Storage errors propagate, and
345    /// so does an input with neither a live object nor the version its
346    /// deletion replaced.
347    pub fn read_shared_object_for_type(
348        &self,
349        id: ObjectId,
350        initial_shared_version: Version,
351    ) -> IotaResult<Object> {
352        match SharedReader::start_type_read(id, initial_shared_version, self.horizon)
353            .read_object(self)?
354        {
355            TypeObjectLookup::Live(object) => Ok(object),
356            TypeObjectLookup::Deleted(deleted) => deleted.read_deletion_row(self)?.load_bytes(self),
357        }
358    }
359}

361/// The coin deny list as of a commit's horizon, for the deny check at the
362/// commit. It reads through the commit's one reader, so every transaction in
363/// the commit sees the same deny list. Only latest-object reads are answered:
364/// reading an exact version through it is a storage error.
365pub struct DenyListAtCommit<'a> {
366    reader: &'a CommitIndexedReader,
367    /// The reader's error behind the storage error the last failed read
368    /// returned, which keeps only its text.
369    reader_error: RefCell<Option<IotaError>>,
370}
371
372impl<'a> DenyListAtCommit<'a> {
373    pub fn new(reader: &'a CommitIndexedReader) -> Self {
374        Self {
375            reader,
376            reader_error: RefCell::new(None),
377        }
378    }
379
380    /// [`coin_deny_list_v1_denial`] as of the horizon. A failed read returns
381    /// the reader's own error, so the end of the epoch stays
382    /// `IotaError::EpochEnded`.
383    pub fn denial(
384        &self,
385        address: Address,
386        coin_types: BTreeSet<String>,
387    ) -> IotaResult<Option<UserInputError>> {
388        coin_deny_list_v1_denial(address, coin_types, self)
389            .map_err(|error| self.reader_error.take().unwrap_or(error))
390    }
391}
392
393impl ObjectStore for DenyListAtCommit<'_> {
394    fn try_get_object(&self, object_id: &ObjectId) -> StorageResult<Option<Object>> {
395        self.reader
396            .read_dynamic_field(*object_id)
397            .map(|field| match field {
398                DynamicFieldAtHorizon::Live(object) => Some(object),
399                DynamicFieldAtHorizon::Absent => None,
400            })
401            .map_err(|error| {
402                let storage_error = StorageError::custom(error.to_string());
403                *self.reader_error.borrow_mut() = Some(error);
404                storage_error
405            })
406    }
407
408    fn try_get_object_by_key(
409        &self,
410        object_id: &ObjectId,
411        version: Version,
412    ) -> StorageResult<Option<Object>> {
413        Err(StorageError::custom(format!(
414            "the deny list at a commit reads no exact version, asked for {object_id} at {version}"
415        )))
416    }
417}

419/// Lets the deny checks load packages as of the commit. A package the reader
420/// answers missing reads as absent, which the deny checks report as not
421/// found. So validation runs the deny check of the commands only after every
422/// package input has loaded, and gives the authenticator package check only
423/// functions whose package this reader sees.
424impl BackingPackageStore for CommitIndexedReader {
425    fn get_package_object(&self, package_id: &ObjectId) -> IotaResult<Option<PackageObject>> {
426        Ok(match self.read_package(*package_id)? {
427            PackageVerdict::Visible(package) => Some(package),
428            PackageVerdict::Missing(_) => None,
429        })
430    }
