4//! Reads one owned or immutable input, named by `(id, V)`: the object id and the version the
5//! transaction names. Each step is its own type that offers only the next allowed read, so a read
6//! cannot be skipped, repeated or taken out of order. The one exception is the last read of the
7//! other rows of `id`, whose follow-up reads depend only on what the first one returned. The
8//! decisions on what a read returned are plain functions.
9//!
10//! The row at `(id, V)`, when there is one, decides a drop or a missing, and
11//! a keep from it is checked against the rows above `V` as described below.
12//! Without one, the store's
13//! latest reference for `id` is read and kept, and the row is read once more
14//! before that store answer may decide. The doc of the parent module explains
15//! why one more read is enough.
16//!
17//! A keep from the row at `(id, V)` is final only after the lowest row of `id`
18//! above `V` is read. Every consumption of `V` writes a row of `id` above it,
19//! including a receive and a change through a dynamic-field parent, which
20//! take no lock. If that row was produced at or below the horizon, every
21//! validator has seen the consumption, so `V` drops even when this validator
22//! has already pruned its object. A row above the horizon is ignored, because
23//! a validator that has not executed its commit cannot see it. A keep from
24//! the store skips this read: a row above `V` at or below the horizon belongs
25//! to a commit that is executed before validation starts, so the store would
26//! already have answered a newer version. Mutations of a shared object write
27//! no rows, so a shared object's creation version named as owned drops from
28//! the creation row without this read.
29//!
30//! With no row at `(id, V)`, any other store answer does not decide by
31//! itself: this validator may have executed writes of `id` in commits above
32//! the horizon that a slower validator has not, so its store can be ahead of
33//! the state at the horizon. So the lowest row of `id` above `V` is read, and
34//! decides when there is one. A row at or below the horizon is a consumption
35//! every validator has seen, so `V` drops. A row above the horizon shows what
36//! `id` was before that write, which is the state at the horizon as far as
37//! `V` is concerned, since no row lies between `V` and it: if the version it
38//! replaced is `V`, `V` is kept; if it is above `V`, `V` was replaced before
39//! the epoch and drops; if it is below `V` or there is none, `V` was not
40//! produced yet or `id` was not live, so `V` is missing. The keep then needs
41//! `V`'s object, which stays in the store, as the parent module doc explains.
42//!
43//! With no row above `V`, the store answer decides, read so that it does not
44//! depend on writes above the horizon. A tombstone of a deletion whose row is
45//! above the horizon is missing, as `id` was live below `V` at the horizon.
46//! A tombstone left by wrapping is missing, as the unwrap that ends it writes
47//! a row that names no replaced version, and the reads above then answer
48//! missing.
49//!
50//! Mutations of a shared object write no rows, so its versions can move above
51//! the horizon without a trace. A shared object named as owned therefore
52//! drops whatever its version, once its creation is at or below the horizon
53//! or before the epoch, and is missing while its creation row is above the
54//! horizon, as `id` did not exist at the horizon. This is checked whenever
55//! the store answer is a live object, and on a keep once the loaded object
56//! shows the owner.
57

68/// A step of an owned input's read.
69pub trait OwnedState {}
70
71/// One owned input being read for validation at commit `C`. `S` is the step
72/// reached so far. It offers only the next allowed read, and each read
73/// consumes this value.
74pub struct OwnedReader<S: OwnedState> {
75    /// The `(id, version, digest)` the transaction names.
76    input: ObjectReference,
77    /// The horizon, `C - K`. The row at `(id, V)` answers missing when it was
78    /// produced above it. The module doc says how a row of `id` above `V`
79    /// produced above it counts.
80    horizon: CommitIndex,
81    state: S,
82}
83
84impl<S: OwnedState> OwnedReader<S> {
85    fn into_state<T: OwnedState>(self, state: T) -> OwnedReader<T> {
86        OwnedReader {
87            input: self.input,
88            horizon: self.horizon,
89            state,
90        }
91    }
92
93    fn key(&self) -> ObjectKey {
94        ObjectKey(self.input.object_id, self.input.version)
95    }
96
97    /// Decides from the row at `(id, V)`, or `None` when there is no row.
98    /// Used by both reads of the row.
99    fn row_classification(&self, ctx: &CommitIndexedReader) -> IotaResult<Option<Classification>> {
100        Ok(ctx
101            .epoch_store
102            .handler_processed_object(&self.key())?
103            .map(|row| classify_row(&row, &self.input, self.horizon)))
104    }
105}

111/// Nothing read yet.
112pub struct Start;
113impl OwnedState for Start {}
114
115/// Outcome of the first read of the row at `(id, V)`.
116#[must_use]
117pub enum TablesLookup {
118    /// A live row at or below the horizon with the named digest that is not
119    /// a shared object's creation row. Next: the rows above `V`.
120    LiveAtVersion(OwnedReader<LiveAtVersion>),
121    /// A row produced above the horizon. The store is not asked.
122    Missing(MissingReason),
123    /// A tombstone row, a `Live` row with another digest or a shared
124    /// object's creation row.
125    Drop(DropReason),
126    /// No row at `(id, V)`, so no result yet: the read goes on to the store.
127    NoEntry(OwnedReader<NoTablesEntry>),
128}
129
130impl OwnedReader<Start> {
131    /// Starts reading `input` with the horizon `C - K`. Every later step is
132    /// reached from here.
133    pub fn start(input: ObjectReference, horizon: CommitIndex) -> OwnedReader<Start> {
134        OwnedReader {
135            input,
136            horizon,
137            state: Start,
138        }
139    }
140
141    /// Reads the row at `(id, V)`: the rows held in memory first, then the
142    /// table. Decides when a row exists.
143    pub fn read_handler_row(self, ctx: &CommitIndexedReader) -> IotaResult<TablesLookup> {
144        let Some(classification) = self.row_classification(ctx)? else {
145            return Ok(TablesLookup::NoEntry(self.into_state(NoTablesEntry)));
146        };
147        Ok(match classification {
148            Classification::Keep => TablesLookup::LiveAtVersion(self.into_state(LiveAtVersion)),
149            Classification::Missing(reason) => TablesLookup::Missing(reason),
150            Classification::Drop(reason) => TablesLookup::Drop(reason),
151        })
152    }
153}

174/// The store answered with the latest reference, a tombstone, or nothing.
175/// Kept until the row has been read again.
176pub struct StoreAnswered {
177    latest: Option<ObjectReference>,
178}
179impl OwnedState for StoreAnswered {}
180
181/// Outcome of the second read of the row. A row decides as on the first
182/// read. Otherwise the kept store answer decides: the named reference keeps,
183/// and any other answer goes on to the other rows of `id`.
184#[must_use]
185pub enum TablesRecheck {
186    /// A keep from a row that appeared since the first read. Next: the rows
187    /// above `V`.
188    LiveAtVersion(OwnedReader<LiveAtVersion>),
189    /// A keep from the store. Next: the bytes.
190    NeedBytes(OwnedReader<NeedBytes>),
191    /// The store holds another version, a tombstone or nothing. Next: the
192    /// id's other rows.
193    NotAtVersion(OwnedReader<NotAtVersion>),
194    Missing(MissingReason),
195    Drop(DropReason),
196}
197
198impl OwnedReader<StoreAnswered> {
199    /// Reads the row at `(id, V)` again. A row decides as on the first read.
200    /// Without one, the kept store answer decides: the named reference keeps,
201    /// and any other answer goes on to the other rows of `id`.
202    pub fn reread_handler_row(self, ctx: &CommitIndexedReader) -> IotaResult<TablesRecheck> {
203        // A row decides as on the first read. Without one, the kept store
204        // answer is used.
205        if let Some(classification) = self.row_classification(ctx)? {
206            return Ok(match classification {
207                Classification::Keep => {
208                    TablesRecheck::LiveAtVersion(self.into_state(LiveAtVersion))
209                }
210                Classification::Missing(reason) => TablesRecheck::Missing(reason),
211                Classification::Drop(reason) => TablesRecheck::Drop(reason),
212            });
213        }
214        let latest = self.state.latest;
215        Ok(if is_named_reference(latest.as_ref(), &self.input) {
216            TablesRecheck::NeedBytes(self.into_state(NeedBytes))
217        } else {
218            TablesRecheck::NotAtVersion(self.into_state(NotAtVersion { latest }))
219        })
220    }
221}

227/// No row at `(id, V)`, and the store holds another version, a tombstone or
228/// nothing. The id's other rows have not been read.
229pub struct NotAtVersion {
230    latest: Option<ObjectReference>,
231}
232impl OwnedState for NotAtVersion {}
233
234impl OwnedReader<NotAtVersion> {
235    /// Reads the lowest row of `id` above `V`, then, as needed, the highest
236    /// row of `id`, the latest object's owner and a shared object's creation
237    /// row, each from the rows held in memory first. Decides.
238    pub fn read_other_rows(self, ctx: &CommitIndexedReader) -> IotaResult<RowsAboveLookup> {
239        let id = self.input.object_id;
240        let row_above = ctx
241            .epoch_store
242            .lowest_handler_processed_row_above(&id, self.input.version)?;
243        let class = match (row_above, self.state.latest) {
244            (Some((_, row)), _) => {
245                classify_lowest_row_above(&row, self.input.version, self.horizon)
246            }
247            (None, None) => OtherRows::Decided(missing_classification(MissingKind::StoreNotFound)),
248            (None, Some(latest)) if latest.digest.is_wrapped() => {
249                OtherRows::Decided(missing_classification(MissingKind::StoreWrapped))
250            }
251            (None, Some(latest)) if !latest.digest.is_alive() => {
252                let highest = ctx.epoch_store.highest_handler_processed_row(&id)?;
253                classify_deletion(highest.as_ref().map(|(_, row)| row), self.horizon)
254            }
255            (None, Some(latest)) => match read_initial_shared_version(ctx, &id)? {
256                Some(initial) => OtherRows::Shared(initial),
257                None => OtherRows::Decided(classify_live_latest(&latest, &self.input)),
258            },
259        };
260        let classification = match class {
261            OtherRows::Decided(classification) => classification,
262            OtherRows::Shared(initial) => {
263                let creation_row = ctx
264                    .epoch_store
265                    .handler_processed_object(&ObjectKey(id, initial))?;
266                classify_shared_named_as_owned(creation_row.as_ref(), self.horizon).into()
267            }
268        };
269        Ok(match classification {
270            Classification::Keep => RowsAboveLookup::NeedBytes(self.into_state(NeedBytes)),
271            Classification::Missing(reason) => RowsAboveLookup::Missing(reason),
272            Classification::Drop(reason) => RowsAboveLookup::Drop(reason),
273        })
274    }
275}

277/// The initial shared version of `id`'s latest object, `None` when it is not
278/// shared. An object deleted since the store answer shows its owner in its
279/// deletion row.
280fn read_initial_shared_version(
281    ctx: &CommitIndexedReader,
282    id: &ObjectId,
283) -> IotaResult<Option<Version>> {
284    if let Some(object) = ctx.cache.try_get_object(id)? {
285        return Ok(match object.owner {
286            Owner::Shared(initial) => Some(initial),
287            _ => None,
288        });
289    }
290    Ok(ctx
291        .epoch_store
292        .highest_handler_processed_row(id)?
293        .and_then(|(_, row)| row.initial_shared_version))
294}

300/// The row at `(id, V)` kept the input. The rows of `id` above `V` have not
301/// been read.
302pub struct LiveAtVersion;
303impl OwnedState for LiveAtVersion {}
304
305/// Outcome of the read of the rows above `V`.
306#[must_use]
307pub enum RowsAboveLookup {
308    /// A keep. Next: load the object.
309    NeedBytes(OwnedReader<NeedBytes>),
310    /// Only from [`OwnedReader::read_other_rows`]: `V` is not produced at the
311    /// horizon, or `id` was not live there.
312    Missing(MissingReason),
313    Drop(DropReason),
314}
315
316impl OwnedReader<LiveAtVersion> {
317    /// Reads the lowest row of `id` above `V`: the rows held in memory
318    /// first, then the table. Decides.
319    pub fn read_rows_above(self, ctx: &CommitIndexedReader) -> IotaResult<RowsAboveLookup> {
320        let row_above = ctx
321            .epoch_store
322            .lowest_handler_processed_row_above(&self.input.object_id, self.input.version)?;
323        Ok(
324            match classify_row_above(row_above.map(|(_, row)| row).as_ref(), self.horizon) {
325                Some(reason) => RowsAboveLookup::Drop(reason),
326                None => RowsAboveLookup::NeedBytes(self.into_state(NeedBytes)),
327            },
328        )
329    }
330}

344    }

351/// The result one read gave for `(id, V)`.
352#[must_use]
353#[derive(Clone, Copy, Debug, PartialEq, Eq)]
354enum Classification {
355    Keep,
356    Missing(MissingReason),
357    Drop(DropReason),
358}
359
360/// Decides from the row at `(id, V)`: first the horizon, then the kind, then
361/// the digest, then whether the row is a shared object's creation row.
362fn classify_row(
363    row: &HandlerProcessedObject,
364    input: &ObjectReference,
365    horizon: CommitIndex,
366) -> Classification {
367    if row.produced_at > horizon {
368        return Classification::Missing(MissingReason(MissingKind::HandlerRowAboveHorizon));
369    }
370    match row.kind {
371        HandlerProcessedObjectKind::Deleted | HandlerProcessedObjectKind::Wrapped => {
372            Classification::Drop(DropReason(DropKind::HandlerRowTombstone))
373        }
374        HandlerProcessedObjectKind::Live if row.digest != input.digest => {
375            Classification::Drop(DropReason(DropKind::HandlerRowDigestMismatch))
376        }
377        HandlerProcessedObjectKind::Live if row.initial_shared_version.is_some() => {
378            Classification::Drop(DropReason(DropKind::HandlerRowSharedObject))
379        }
380        HandlerProcessedObjectKind::Live => Classification::Keep,
381    }
382}
383
384/// The lowest row of `id` above a kept `V`: one produced at or below the
385/// horizon consumed `V` and drops it; otherwise `None`, and the keep stands.
386fn classify_row_above(
387    row: Option<&HandlerProcessedObject>,
388    horizon: CommitIndex,
389) -> Option<DropReason> {
390    row.filter(|row| row.produced_at <= horizon)
391        .map(|_| DropReason(DropKind::HandlerRowSuperseded))
392}
393
394/// Whether the kept store answer is the named reference. A tombstone never
395/// is, so a transaction naming a tombstone reference never keeps from the
396/// store.
397fn is_named_reference(latest: Option<&ObjectReference>, input: &ObjectReference) -> bool {
398    latest.is_some_and(|latest| latest.digest.is_alive() && latest == input)
399}
400
401fn missing_classification(kind: MissingKind) -> Classification {
402    Classification::Missing(MissingReason(kind))
403}
404
405fn drop_classification(kind: DropKind) -> Classification {
406    Classification::Drop(DropReason(kind))
407}

409/// What the other rows of `id` decided about `(id, V)`.
410#[must_use]
411#[derive(Clone, Copy, Debug, PartialEq, Eq)]
412enum OtherRows {
413    Decided(Classification),
414    /// A shared object with this initial shared version: its creation row
415    /// decides.
416    Shared(Version),
417}
418
419/// The id's lowest row above `V`, with no row at `(id, V)`. One at or below
420/// the horizon consumed `V`, or superseded a `V` that never existed. One
421/// above the horizon decides from the version it replaced, see the module
422/// doc.
423fn classify_lowest_row_above(
424    row: &HandlerProcessedObject,
425    version: Version,
426    horizon: CommitIndex,
427) -> OtherRows {
428    if row.produced_at <= horizon {
429        let kind = if row.replaced_version == Some(version) {
430            DropKind::HandlerRowSuperseded
431        } else {
432            DropKind::StoreSuperseded
433        };
434        return OtherRows::Decided(drop_classification(kind));
435    }
436    match (row.initial_shared_version, row.replaced_version) {
437        // A shared object's deletion: its versions moved without rows.
438        (Some(initial), Some(_)) => OtherRows::Shared(initial),
439        (_, Some(replaced)) => OtherRows::Decided(match replaced.cmp(&version) {
440            Ordering::Equal => Classification::Keep,
441            Ordering::Greater => drop_classification(DropKind::StoreSuperseded),
442            Ordering::Less => missing_classification(MissingKind::VersionNotLiveAtHorizon),
443        }),
444        // A creation or an unwrap.
445        (_, None) => {
446            OtherRows::Decided(missing_classification(MissingKind::VersionNotLiveAtHorizon))
447        }
448    }
449}
450
451/// The id's highest row, with a deletion tombstone as the store answer and no
452/// row above `V`. A deletion above the horizon leaves `id` live below `V` at
453/// the horizon. Any other deletion every validator has seen.
454fn classify_deletion(row: Option<&HandlerProcessedObject>, horizon: CommitIndex) -> OtherRows {
455    match row {
456        Some(row)
457            if row.kind == HandlerProcessedObjectKind::Deleted && row.produced_at > horizon =>
458        {
459            match row.initial_shared_version {
460                Some(initial) => OtherRows::Shared(initial),
461                None => {
462                    OtherRows::Decided(missing_classification(MissingKind::VersionNotLiveAtHorizon))
463                }
464            }
465        }
466        _ => OtherRows::Decided(drop_classification(DropKind::StoreSuperseded)),
467    }
468}
469
470/// A live store answer that is not the named reference, with no row above
471/// `V` and an object that is not shared. A newer version has no row, so it
472/// is the epoch-start version every validator holds.
473fn classify_live_latest(latest: &ObjectReference, input: &ObjectReference) -> Classification {
474    match latest.version.cmp(&input.version) {
475        Ordering::Greater => drop_classification(DropKind::StoreSuperseded),
476        Ordering::Less => missing_classification(MissingKind::StoreBelowVersion),
477        Ordering::Equal if latest.digest == input.digest => Classification::Keep,
478        Ordering::Equal => drop_classification(DropKind::StoreDigestMismatch),
479    }
480}

482/// The result for a shared object named as owned, from its creation row.
483#[must_use]
484#[derive(Clone, Copy, Debug, PartialEq, Eq)]
485pub(super) enum SharedNamedAsOwned {
486    /// Created above the horizon: `id` did not exist at the horizon.
487    Missing(MissingReason),
488    Drop(DropReason),
489}
490
491impl From<SharedNamedAsOwned> for Classification {
492    fn from(verdict: SharedNamedAsOwned) -> Self {
493        match verdict {
494            SharedNamedAsOwned::Missing(reason) => Classification::Missing(reason),
495            SharedNamedAsOwned::Drop(reason) => Classification::Drop(reason),
496        }
497    }
498}

500/// A shared object named as owned, whatever its version: its creation row
501/// at its initial shared version decides, none meaning it was created before
502/// the epoch.
503pub(super) fn classify_shared_named_as_owned(
504    creation_row: Option<&HandlerProcessedObject>,
505    horizon: CommitIndex,
506) -> SharedNamedAsOwned {
507    match creation_row {
508        Some(row) if row.produced_at > horizon => {
509            SharedNamedAsOwned::Missing(MissingReason(MissingKind::SharedCreationAboveHorizon))
510        }
511        _ => SharedNamedAsOwned::Drop(DropReason(DropKind::SharedObjectNamedAsOwned)),
512    }
513}
