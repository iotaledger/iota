4//! Reads one shared input: whether the object exists and is shared at the
5//! declared initial version. It reads the object's creation row (the row at
6//! the declared initial shared version) and the latest object in the store,
7//! for an object that is gone the highest row of `id`, and after a drop the
8//! lowest row of `id`. These reads
9//! never use the object's contents. As for owned inputs, each step is its own
10//! type, and a store answer is kept until the creation row has been read a
11//! second time. A separate read, once the input is kept, loads an object of
12//! the input for its type, which the coin deny check needs; its section below
13//! explains it.
14
15use iota_sdk_types::{ObjectId, Owner, TransactionDigest, Version};
16use iota_types::{
17    error::{IotaError, IotaResult},
18    object::Object,
19    storage::ObjectKey,
20};
21
22use super::{DropKind, DropReason, MissingKind, MissingReason, reader::CommitIndexedReader};
23use crate::authority::authority_per_epoch_store::handler_object_state::{
24    CommitIndex, HandlerProcessedObject, HandlerProcessedObjectKind,
25};

27/// A step of a shared input's read.
28pub trait SharedState {}
29
30/// One shared input being read for validation at commit `C`. `S` is the step
31/// reached so far. It offers only the next allowed read, and each read
32/// consumes this value.
33pub struct SharedReader<S: SharedState> {
34    id: ObjectId,
35    /// The initial shared version the transaction declares. The object's
36    /// creation row is the row at this version.
37    initial_shared_version: Version,
38    /// The horizon, `C - K`. A creation row produced above it answers
39    /// missing, and a tombstone row produced above it keeps the input as
40    /// deleted.
41    horizon: CommitIndex,
42    state: S,
43}
44
45impl<S: SharedState> SharedReader<S> {
46    fn into_state<T: SharedState>(self, state: T) -> SharedReader<T> {
47        SharedReader {
48            id: self.id,
49            initial_shared_version: self.initial_shared_version,
50            horizon: self.horizon,
51            state,
52        }
53    }
54
55    /// Decides from the creation row, or `None` when there is no row at
56    /// `(id, declared initial version)`. Used by both reads of the creation
57    /// row.
58    fn creation_row_classification(
59        &self,
60        ctx: &CommitIndexedReader,
61    ) -> IotaResult<Option<CreationClass>> {
62        let key = ObjectKey(self.id, self.initial_shared_version);
63        Ok(ctx
64            .epoch_store
65            .handler_processed_object(&key)?
66            .map(|row| classify_creation_row(&row, self.initial_shared_version, self.horizon)))
67    }
68}

74/// Nothing read yet.
75pub struct Start;
76impl SharedState for Start {}

78/// Outcome of the first read of the creation row.
79#[must_use]
80pub enum SharedTablesLookup {
81    /// A live row at or below the horizon that records the declared version
82    /// as the initial shared version. Next: the store object, to check for a
83    /// deletion.
84    Created(SharedReader<CreatedShared>),
85    /// A creation row produced above the horizon. The store is not asked.
86    Missing(MissingReason),
87    /// A row that records another initial shared version or none, or a
88    /// tombstone: the object was not created shared at the declared version.
89    Drop(DropReason),
90    /// No creation row at `(id, declared initial version)`, so no result yet:
91    /// the read goes on to the store.
92    NoEntry(SharedReader<NoTablesEntry>),
93}
94
95impl SharedReader<Start> {
96    /// Starts reading shared input `id` declared at `initial_shared_version`,
97    /// with the horizon `C - K`. Together with
98    /// [`SharedReader::start_type_read`], this is how every later step is
99    /// reached.
100    pub fn start(
101        id: ObjectId,
102        initial_shared_version: Version,
103        horizon: CommitIndex,
104    ) -> SharedReader<Start> {
105        SharedReader {
106            id,
107            initial_shared_version,
108            horizon,
109            state: Start,
110        }
111    }
112
113    /// Reads the creation row: the rows held in memory first, then the table.
114    /// Decides when a row exists.
115    pub fn read_creation_row(self, ctx: &CommitIndexedReader) -> IotaResult<SharedTablesLookup> {
116        Ok(match self.creation_row_classification(ctx)? {
117            Some(CreationClass::Created) => {
118                SharedTablesLookup::Created(self.into_state(CreatedShared))
119            }
120            Some(CreationClass::Missing(reason)) => SharedTablesLookup::Missing(reason),
121            Some(CreationClass::Drop(reason)) => SharedTablesLookup::Drop(reason),
122            None => SharedTablesLookup::NoEntry(self.into_state(NoTablesEntry)),
123        })
124    }
125}

177/// The store answered with the latest object or nothing. Kept until the
178/// creation row has been read again.
179pub struct ObjectAnswered {
180    object: Option<Object>,
181}
182impl SharedState for ObjectAnswered {}
183
184/// Outcome of the second read of the creation row. A creation row decides as
185/// on the first read. Otherwise the kept object decides: owner `Shared` at
186/// the declared version exists, another owner drops, and no object goes on to
187/// the highest row of `id`.
188#[must_use]
189pub enum SharedTablesRecheck {
190    Created(SharedReader<CreatedShared>),
191    Exists,
192    Missing(MissingReason),
193    Drop(DropReason),
194    /// No live object. Next: the deletion row.
195    Absent(SharedReader<ObjectAbsent>),
196}
197
198impl SharedReader<ObjectAnswered> {
199    /// Reads the creation row again and decides.
200    pub fn reread_creation_row(self, ctx: &CommitIndexedReader) -> IotaResult<SharedTablesRecheck> {
201        match self.creation_row_classification(ctx)? {
202            Some(CreationClass::Created) => {
203                return Ok(SharedTablesRecheck::Created(self.into_state(CreatedShared)));
204            }
205            Some(CreationClass::Missing(reason)) => {
206                return Ok(SharedTablesRecheck::Missing(reason));
207            }
208            Some(CreationClass::Drop(reason)) => return Ok(SharedTablesRecheck::Drop(reason)),
209            None => {}
210        }
211        // An `Arc` bump: `Object` wraps its contents in one.
212        Ok(match self.state.object.clone() {
213            // No creation row at the declared version: the kept object
214            // decides, from its owner alone.
215            Some(object) => match classify_object(&object, self.initial_shared_version) {
216                ObjectClass::Exists => SharedTablesRecheck::Exists,
217                ObjectClass::Drop(reason) => SharedTablesRecheck::Drop(reason),
218            },
219            None => SharedTablesRecheck::Absent(self.into_state(ObjectAbsent {
220                origin: AbsentOrigin::NoEntry,
221            })),
222        })
223    }
224}

226// ---------------------------------------------------------------------------
227// Deletion: the highest row of `id`
228// ---------------------------------------------------------------------------
229//
230// Execution writes a deletion record in the store at the tombstone version,
231// and again at the version of every later transaction that names the deleted
232// object mutably. Those later transactions write no row for the object, so
233// the latest deletion record depends on which of them a validator has
234// executed, and the deleting execution writes its tombstone before its
235// deletion record, so a read alongside it can find the one without the
236// other. The deletion is therefore decided from the highest row of `id`
237// alone, which execution writes before the outputs and which names the
238// deleting transaction. The deletion record is never read.
239
240/// No live object for `id` in the store.
241pub struct ObjectAbsent {
242    origin: AbsentOrigin,
243}
244impl SharedState for ObjectAbsent {}
245
246/// Which step the read was at when the store showed no live object.
247#[derive(Clone, Copy, Debug, PartialEq, Eq)]
248enum AbsentOrigin {
249    /// A creation row at or below the horizon.
250    Created,
251    /// No creation row.
252    NoEntry,
253}
254
255/// Outcome of the read of the highest row of `id`.
256#[must_use]
257pub enum DeletionRowLookup {
258    /// Deleted above the horizon, by the transaction. Kept: the transaction
259    /// being validated executes against the deletion.
260    Deleted(Version, TransactionDigest),
261    /// Deleted at or below the horizon, or the deleted object was not shared
262    /// at the declared initial version.
263    Drop(DropReason),
264    /// No creation row and no tombstone row: the id never existed, was
265    /// deleted before this epoch, or was created above the horizon.
266    Missing(MissingReason),
267}
268
269impl SharedReader<ObjectAbsent> {
270    /// Reads the row of `id` at its highest version, which decides. A shared
271    /// object has at most a creation row and a tombstone row this epoch, so a
272    /// tombstone row is a deletion whose commit the handler has assigned, and
273    /// it records the deleting commit, the deleted object's initial shared
274    /// version and the deleting transaction.
275    ///
276    /// A creation row at or below the horizon with no live object and no
277    /// tombstone row is a storage error, which stops commit processing, and
278    /// never a result: executing a deletion writes its row before the
279    /// tombstone reaches the store, so only a broken write leaves that state.
280    pub fn read_deletion_row(self, ctx: &CommitIndexedReader) -> IotaResult<DeletionRowLookup> {
281        let row = ctx.epoch_store.highest_handler_processed_row(&self.id)?;
282        let class = classify_deletion_row(
283            row.as_ref().map(|(_, row)| row),
284            self.initial_shared_version,
285            self.horizon,
286        );
287        let tombstone = row.and_then(|(ObjectKey(_, version), row)| {
288            row.deleted_by.map(|deleted_by| (version, deleted_by))
289        });
290        match (class, tombstone) {
291            (DeletionClass::Drop(reason), _) => Ok(DeletionRowLookup::Drop(reason)),
292            (DeletionClass::AboveHorizon, Some((version, deleted_by))) => {
293                Ok(DeletionRowLookup::Deleted(version, deleted_by))
294            }
295            (DeletionClass::AboveHorizon, None) => Err(IotaError::Storage(format!(
296                "shared input {} declared at {} has a tombstone row above the horizon that \
297                 names no transaction",
298                self.id, self.initial_shared_version
299            ))),
300            (DeletionClass::NotHandlerKnown, _) => match self.state.origin {
301                AbsentOrigin::NoEntry => Ok(DeletionRowLookup::Missing(MissingReason(
302                    MissingKind::SharedNotFound,
303                ))),
304                AbsentOrigin::Created => Err(IotaError::Storage(format!(
305                    "shared input {} declared at {} has no live object and no tombstone row, \
306                     though its creation row is at or below the horizon",
307                    self.id, self.initial_shared_version
308                ))),
309            },
310        }
311    }
312}

314// ---------------------------------------------------------------------------
315// A drop: the lowest row of `id`
316// ---------------------------------------------------------------------------
317//
318// A drop from the store object, or from a tombstone row above the horizon,
319// assumes `id` existed at the horizon. A validator that has not executed the
320// commit that created `id`, or unwrapped it, has no object and no tombstone
321// row for it and answers missing. So a drop is checked against the lowest
322// row of `id`: one above the horizon that replaced no version is that
323// creation or unwrap, and the drop becomes missing. A drop decided from a
324// row at or below the horizon has a lowest row at or below it too, as an
325// id's row versions rise with the commits that write them.
326
327/// A drop decided by the reads above. The lowest row of `id` has not been
328/// read.
329pub struct Dropped {
330    reason: DropReason,
331}
332impl SharedState for Dropped {}
333
334impl SharedReader<Start> {
335    /// Starts the check of a drop the reads above decided for `id`.
336    pub fn start_drop_check(
337        id: ObjectId,
338        initial_shared_version: Version,
339        horizon: CommitIndex,
340        reason: DropReason,
341    ) -> SharedReader<Dropped> {
342        SharedReader {
343            id,
344            initial_shared_version,
345            horizon,
346            state: Dropped { reason },
347        }
348    }
349}
350
351/// Outcome of the read of the id's lowest row after a drop.
352#[must_use]
353pub enum DropCheck {
354    Drop(DropReason),
355    /// `id` did not exist at the horizon.
356    Missing(MissingReason),
357}
358
359impl SharedReader<Dropped> {
360    /// Reads the lowest row of `id`, which decides.
361    pub fn read_lowest_row(self, ctx: &CommitIndexedReader) -> IotaResult<DropCheck> {
362        let row = ctx
363            .epoch_store
364            .lowest_handler_processed_row_above(&self.id, Version::MIN_VALID_INCL)?;
365        Ok(
366            match classify_lowest_row(row.as_ref().map(|(_, row)| row), self.horizon) {
367                Some(reason) => DropCheck::Missing(reason),
368                None => DropCheck::Drop(self.state.reason),
369            },
370        )
371    }
372}

374// ---------------------------------------------------------------------------
375// Type: an object of a kept input, for its type
376// ---------------------------------------------------------------------------
377//
378// A separate read, after the shared input's read kept it. The coin deny
379// check at the commit needs the coin type of every input, and a shared
380// input's object is not read for the keep. An object's type never changes, so
381// any stored version gives it, and the latest store object is used even when
382// rows exist: that answer is the same on every validator. Once the object is
383// gone, the version its deletion replaced gives it. The deletion is above the
384// horizon, since one at or below it drops the input, so pruning still keeps
385// that version.
386
387/// A kept shared input whose type is wanted. Nothing read yet.
388pub struct TypeStart;
389impl SharedState for TypeStart {}
390
391/// Outcome of the store read for the type.
392#[must_use]
393pub enum TypeObjectLookup {
394    /// The live object.
395    Live(Object),
396    /// Gone from the store. Next: the deletion row.
397    Deleted(SharedReader<TypeDeleted>),
398}
399
400impl SharedReader<TypeStart> {
401    /// Begins the type read of the kept shared input `id` declared at
402    /// `initial_shared_version`.
403    pub fn start_type_read(
404        id: ObjectId,
405        initial_shared_version: Version,
406        horizon: CommitIndex,
407    ) -> SharedReader<TypeStart> {
408        SharedReader {
409            id,
410            initial_shared_version,
411            horizon,
412            state: TypeStart,
413        }
414    }
415
416    /// The latest object by id.
417    pub fn read_object(self, ctx: &CommitIndexedReader) -> IotaResult<TypeObjectLookup> {
418        Ok(match ctx.cache.try_get_object(&self.id)? {
419            Some(object) => TypeObjectLookup::Live(object),
420            None => TypeObjectLookup::Deleted(self.into_state(TypeDeleted)),
421        })
422    }
423}

425/// The kept input is gone from the store.
426pub struct TypeDeleted;
427impl SharedState for TypeDeleted {}
428
429impl SharedReader<TypeDeleted> {
430    /// Reads the version the deletion replaced from the highest row of `id`:
431    /// the deleting execution writes its row before the object leaves the
432    /// store, and no later write adds a row. A missing row or version is a
433    /// storage error, never a result.
434    pub fn read_deletion_row(
435        self,
436        ctx: &CommitIndexedReader,
437    ) -> IotaResult<SharedReader<TypeNeedBytes>> {
438        let version = ctx
439            .epoch_store
440            .highest_handler_processed_row(&self.id)?
441            .and_then(|(_, row)| row.replaced_version)
442            .ok_or_else(|| {
443                IotaError::Storage(format!(
444                    "kept shared input {} is not in the object store and has no handler row \
445                     with the version it replaced",
446                    self.id
447                ))
448            })?;
449        Ok(self.into_state(TypeNeedBytes { version }))
450    }
451}

453/// The version the deletion of the kept input replaced.
454pub struct TypeNeedBytes {
455    version: Version,
456}
457impl SharedState for TypeNeedBytes {}
458
459impl SharedReader<TypeNeedBytes> {
460    /// Loads the object at exactly that version. Pruning keeps it, so a
461    /// missing object is a storage error.
462    pub fn load_bytes(self, ctx: &CommitIndexedReader) -> IotaResult<Object> {
463        let key = ObjectKey(self.id, self.state.version);
464        ctx.cache
465            .try_get_object_by_key(&key.0, key.1)?
466            .ok_or_else(|| IotaError::Storage(format!("no bytes for {key:?} in the object store")))
467    }
468}

470// ---------------------------------------------------------------------------

474/// What the creation row decided about `(id, declared initial version)`.
475#[must_use]
476#[derive(Debug, PartialEq, Eq)]
477enum CreationClass {
478    Created,
479    Missing(MissingReason),
480    Drop(DropReason),
481}
482
483/// What a row's `initial_shared_version` decided, once the object that
484/// carried the owner is gone.
485#[must_use]
486#[derive(Debug, PartialEq, Eq)]
487enum RowOwnerClass {
488    /// Shared at the declared initial version.
489    Matches,
490    /// Shared at another initial version, or not shared at all.
491    Drop(DropReason),
492}
493
494/// A row's initial shared version against the declared one. `None` means the
495/// deleted object was not shared.
496fn classify_row_initial_version(row_initial: Option<Version>, declared: Version) -> RowOwnerClass {
497    match row_initial {
498        Some(initial) if initial == declared => RowOwnerClass::Matches,
499        Some(_) => RowOwnerClass::Drop(DropReason(DropKind::SharedInitialVersionMismatch)),
500        None => RowOwnerClass::Drop(DropReason(DropKind::SharedNotCreatedShared)),
501    }
502}
503
504/// What a live store object decided about `id`.
505#[must_use]
506#[derive(Debug, PartialEq, Eq)]
507enum ObjectClass {
508    /// Owner `Shared` at the declared initial version.
509    Exists,
510    /// Another owner, or shared at another initial version.
511    Drop(DropReason),
512}
513
514/// The live object against the declared initial version. Only the owner is
515/// consulted, never the contents.
516fn classify_object(object: &Object, declared: Version) -> ObjectClass {
517    match object.owner {
518        Owner::Shared(initial) if initial == declared => ObjectClass::Exists,
519        Owner::Shared(_) => ObjectClass::Drop(DropReason(DropKind::SharedInitialVersionMismatch)),
520        _ => ObjectClass::Drop(DropReason(DropKind::SharedNotCreatedShared)),
521    }
522}
523
524/// What the highest row of `id` decided about a deletion.
525#[must_use]
526#[derive(Debug, PartialEq, Eq)]
527enum DeletionClass {
528    /// A tombstone row above the horizon: the deletion is not visible at this
529    /// commit.
530    AboveHorizon,
531    /// No tombstone row: deleted by execution the handler has not reached, if
532    /// at all.
533    NotHandlerKnown,
534    /// Deleted at or below the horizon, or not shared at the declared
535    /// initial version.
536    Drop(DropReason),
537}
538
539/// The highest row of `id`. Only a tombstone row is a deletion whose commit
540/// the handler has assigned: a creation row as the highest row, or no row,
541/// means the handler has not reached a deletion, and since a row is always
542/// written before its object, no other reading is possible. The tombstone's
543/// initial shared version is checked first: a wrong declared version drops at
544/// every horizon. Then a tombstone at or below the horizon means every
545/// validator has executed the deletion.
546fn classify_deletion_row(
547    row: Option<&HandlerProcessedObject>,
548    declared: Version,
549    horizon: CommitIndex,
550) -> DeletionClass {
551    let Some(row) = row.filter(|row| row.kind != HandlerProcessedObjectKind::Live) else {
552        return DeletionClass::NotHandlerKnown;
553    };
554    match classify_row_initial_version(row.initial_shared_version, declared) {
555        RowOwnerClass::Drop(reason) => DeletionClass::Drop(reason),
556        RowOwnerClass::Matches if row.produced_at <= horizon => {
557            DeletionClass::Drop(DropReason(DropKind::SharedDeletedAtOrBelowHorizon))
558        }
559        RowOwnerClass::Matches => DeletionClass::AboveHorizon,
560    }
561}
562
563/// The id's lowest row after a drop: missing when it is above the horizon
564/// and replaced no version, a creation or an unwrap there.
565fn classify_lowest_row(
566    row: Option<&HandlerProcessedObject>,
567    horizon: CommitIndex,
568) -> Option<MissingReason> {
569    row.filter(|row| row.produced_at > horizon && row.replaced_version.is_none())
570        .map(|_| MissingReason(MissingKind::SharedNotFound))
571}
572
573/// The creation row at `(id, declared)`: horizon first, then the row must be
574/// live and record the declared version as the initial shared version.
575fn classify_creation_row(
576    row: &HandlerProcessedObject,
577    declared: Version,
578    horizon: CommitIndex,
579) -> CreationClass {
580    if row.produced_at > horizon {
581        return CreationClass::Missing(MissingReason(MissingKind::SharedCreationAboveHorizon));
582    }
583    match (row.kind, row.initial_shared_version) {
584        (HandlerProcessedObjectKind::Live, Some(initial)) if initial == declared => {
585            CreationClass::Created
586        }
587        (HandlerProcessedObjectKind::Live, Some(_)) => {
588            CreationClass::Drop(DropReason(DropKind::SharedInitialVersionMismatch))
589        }
590        (HandlerProcessedObjectKind::Live, None)
591        | (HandlerProcessedObjectKind::Deleted, _)
592        | (HandlerProcessedObjectKind::Wrapped, _) => {
593            CreationClass::Drop(DropReason(DropKind::SharedNotCreatedShared))
594        }
595    }
596}
