21impl PostConsensusTxReorder {
22    pub fn reorder(
23        transactions: &mut [VerifiedSequencedConsensusTransaction],
24        kind: ConsensusTransactionOrdering,
25    ) {
26        // TODO: make the reordering algorithm richer and depend on object hotness as
27        // well. Order transactions based on their gas prices. System
28        // transactions without gas price are put to the beginning of the
29        // sequenced_transactions vector.
30        match kind {
31            ConsensusTransactionOrdering::ByGasPrice => Self::order_by_gas_price(transactions),
32            ConsensusTransactionOrdering::None => (),
33        }
34    }
35
36    /// Orders the list so that every transaction comes after every copy of
37    /// the transaction it depends on, when that one is in the list, by
38    /// repeatedly placing the earliest transaction whose dependency is
39    /// already placed; the list changes only as much as the dependencies
40    /// require. A pure function of the list, so every validator orders a
41    /// commit the same way.
42    pub fn order_dependents_after_dependencies(
43        transactions: &mut Vec<VerifiedSequencedConsensusTransaction>,
44    ) {
45        let _scope = monitored_scope("HandleConsensusOutput::order_dependents_after_dependencies");
46        let dependency_of = |txn: &VerifiedSequencedConsensusTransaction| {
47            txn.0.transaction.transaction_dependency()
48        };
49        if !transactions.iter().any(|txn| dependency_of(txn).is_some()) {
50            return;
51        }
52        // Copies of a transaction share its dependency, so they are placed in
53        // list order, and waiting for the last copy waits for all of them.
54        let last_copy: HashMap<TransactionDigest, usize> = transactions
55            .iter()
56            .enumerate()
57            .filter_map(|(position, txn)| {
58                Some((txn.0.transaction.executable_transaction_digest()?, position))
59            })
60            .collect();
61        let dependency_positions: Vec<Option<usize>> = transactions
62            .iter()
63            .map(|txn| dependency_of(txn).and_then(|digest| last_copy.get(&digest).copied()))
64            .collect();
65        if dependency_positions
66            .iter()
67            .enumerate()
68            .all(|(position, dependency)| dependency.is_none_or(|dependency| dependency < position))
69        {
70            return;
71        }
72
73        let mut placed = vec![false; transactions.len()];
74        let mut waiting: Vec<Vec<usize>> = vec![Vec::new(); transactions.len()];
75        let mut ready = BinaryHeap::new();
76        let mut order = Vec::with_capacity(transactions.len());
77        for (position, dependency) in dependency_positions.into_iter().enumerate() {
78            if let Some(dependency) = dependency.filter(|dependency| !placed[*dependency]) {
79                waiting[dependency].push(position);
80                continue;
81            }
82            ready.push(Reverse(position));
83            while let Some(Reverse(next)) = ready.pop() {
84                placed[next] = true;
85                order.push(next);
86                ready.extend(waiting[next].drain(..).map(Reverse));
87            }
88        }
89        // Only a cycle of digests, which needs a hash collision, leaves a
90        // transaction waiting; such transactions go last, in list order.
91        order.extend((0..placed.len()).filter(|&position| !placed[position]));
92
93        let mut slots: Vec<_> = std::mem::take(transactions).into_iter().map(Some).collect();
94        *transactions = order
95            .into_iter()
96            .filter_map(|position| slots[position].take())
97            .collect();
98    }
99
100    fn order_by_gas_price(transactions: &mut [VerifiedSequencedConsensusTransaction]) {
101        let _scope = monitored_scope("HandleConsensusOutput::order_by_gas_price");
102        transactions.sort_by_key(|txn| {
103            // Reverse order, so transactions with higher gas price sort to the front.
104            // Internal messages and system transactions carry no gas price and sort
105            // to the front via `u64::MAX`.
106            let gas_price = match &txn.0.transaction {
107                SequencedConsensusTransactionKind::External(ext) => {
108                    ext.kind.as_sender_signed_transaction()
109                }
110                SequencedConsensusTransactionKind::System(_) => None,
111            }
112            .map(|data| data.transaction().gas_price())
113            .unwrap_or(u64::MAX);
114
115            std::cmp::Reverse(gas_price)
116        })
117    }
118}

122    use iota_sdk_types::{ProgrammableTransaction, TransactionDigest};

131    /// A user transaction that depends on `dependency`, if any, as sequenced
132    /// by consensus.
133    fn sequenced(dependency: Option<TransactionDigest>) -> VerifiedSequencedConsensusTransaction {
134        let (sender, key) = deterministic_random_account_private_key();
135        let mut builder = TestTransactionBuilder::new(sender, random_object_ref(), 1000)
136            .programmable(ProgrammableTransaction {
137                inputs: vec![],
138                commands: vec![],
139            });
140        if let Some(dependency) = dependency {
141            builder = builder.with_transaction_dependency(dependency, 0);
142        }
143        VerifiedSequencedConsensusTransaction::new_test(ConsensusTransaction::new_user_transaction(
144            builder.build_and_sign(&key),
145        ))
146    }
147
148    fn digests(transactions: &[VerifiedSequencedConsensusTransaction]) -> Vec<TransactionDigest> {
149        transactions
150            .iter()
151            .map(|txn| txn.0.transaction.executable_transaction_digest().unwrap())
152            .collect()
153    }
154
155    fn digest(transaction: &VerifiedSequencedConsensusTransaction) -> TransactionDigest {
156        transaction
157            .0
158            .transaction
159            .executable_transaction_digest()
160            .unwrap()
161    }

163    #[test]
164    fn dependent_moves_behind_its_dependency() {
165        let dependency = sequenced(None);
166        let dependent = sequenced(Some(digest(&dependency)));
167        let other = sequenced(None);
168        let expected = digests(&[dependency.clone(), dependent.clone(), other.clone()]);
169
170        let mut transactions = vec![dependent, dependency, other];
171        PostConsensusTxReorder::order_dependents_after_dependencies(&mut transactions);
172
173        assert_eq!(digests(&transactions), expected);
174    }

178        let first = sequenced(None);
179        let second = sequenced(Some(digest(&first)));
180        let third = sequenced(Some(digest(&second)));
181        let expected = digests(&[first.clone(), second.clone(), third.clone()]);
182
183        let mut transactions = vec![third, second, first];
184        PostConsensusTxReorder::order_dependents_after_dependencies(&mut transactions);

187    }
