275fn compaction_filter_config<T: DeserializeOwned>(
276    name: &str,
277    metrics: Arc<JsonRpcCompactionMetrics>,
278    mut db_options: DBOptions,
279    pruner_watermark: Arc<AtomicU64>,
280    extractor: impl Fn(T) -> TxSequenceNumber + Send + Sync + 'static,
281    by_key: bool,
282) -> DBOptions {
283    let cf = name.to_string();
284    db_options
285        .options
286        .set_compaction_filter(name, move |_, key, value| {
287            let bytes = if by_key { key } else { value };
288            let deserializer = bincode::DefaultOptions::new()
289                .with_big_endian()
290                .with_fixint_encoding();
291            match deserializer.deserialize(bytes) {
292                Ok(key_data) => {
293                    let sequence_number = extractor(key_data);
294                    if sequence_number < pruner_watermark.load(Ordering::Relaxed) {
295                        metrics.key_removed.with_label_values(&[&cf]).inc();
296                        Decision::Remove
297                    } else {
298                        metrics.key_kept.with_label_values(&[&cf]).inc();
299                        Decision::Keep
300                    }
301                }
302                Err(_) => {
303                    metrics.key_error.with_label_values(&[&cf]).inc();
304                    Decision::Keep
305                }
306            }
307        });
308    db_options
309}

482    pub fn index_coin(
483        &self,
484        digest: &TransactionDigest,
485        batch: &mut DBBatch,
486        object_index_changes: &ObjectIndexChanges,
487        tx_coins: Option<TxCoins>,
488    ) -> IotaResult<IndexStoreCacheUpdates> {
489        // In production if this code path is hit, we should expect `tx_coins` to not be
490        // None. However, in many tests today we do not distinguish validator
491        // and/or fullnode, so we gracefully exist here.
492        if tx_coins.is_none() {
493            return Ok(IndexStoreCacheUpdates::default());
494        }
495        // Acquire locks on changed coin owners
496        let mut addresses: HashSet<Address> = HashSet::new();
497        addresses.extend(
498            object_index_changes
499                .deleted_owners
500                .iter()
501                .map(|(owner, _)| *owner),
502        );
503        addresses.extend(
504            object_index_changes
505                .new_owners
506                .iter()
507                .map(|((owner, _), _)| *owner),
508        );
509        let _locks = self.caches.locks.acquire_locks(addresses.into_iter());
510        let mut balance_changes: HashMap<Address, HashMap<TypeTag, TotalBalance>> = HashMap::new();
511        // Index coin info
512        let (input_coins, written_coins) = tx_coins.unwrap();
513        // 1. Delete old owner if the object is deleted or transferred to a new owner,
514        // by looking at `object_index_changes.deleted_owners`.
515        // Objects in `deleted_owners` must be coin type (see
516        // `AuthorityState::commit_transaction`).
517        let coin_delete_keys = object_index_changes
518            .deleted_owners
519            .iter()
520            .filter_map(|(owner, obj_id)| {
521                let object = input_coins.get(obj_id).or(written_coins.get(obj_id))?;
522                let coin_type_tag = object.opt_coin_type().unwrap_or_else(|| {
523                    panic!(
524                        "object_id: {obj_id} is not a coin type, input_coins: {input_coins:?}, written_coins: {written_coins:?}, tx_digest: {digest}"
525                    )
526                });
527                let map = balance_changes.entry(*owner).or_default();
528                let entry = map.entry(coin_type_tag.clone()).or_insert(TotalBalance {
529                    num_coins: 0,
530                    balance: 0
531                });
532                if let Ok(Some(coin_info)) = &self.tables.coin_index.get(&(*owner, coin_type_tag.to_string(), *obj_id)) {
533                    entry.num_coins -= 1;
534                    entry.balance -= coin_info.balance as i128;
535                }
536                Some((*owner, coin_type_tag.to_string(), *obj_id))
537            }).collect::<Vec<_>>();
538        trace!(
539            tx_digest=?digest,
540            "coin_delete_keys: {:?}",
541            coin_delete_keys,
542        );
543        batch.delete_batch(&self.tables.coin_index, coin_delete_keys)?;
544
545        // 2. Upsert new owner, by looking at `object_index_changes.new_owners`.
546        // For a object to appear in `new_owners`, it must be owned by `Owner::Address`
547        // after the tx. It also must not be deleted, hence appear in
548        // written_coins (see `AuthorityState::commit_transaction`) It also must
549        // be a coin type (see `AuthorityState::commit_transaction`).
550        // Here the coin could be transferred to a new address, to simply have the
551        // metadata changed (digest, balance etc) due to a successful or failed
552        // transaction.
553        let coin_add_keys = object_index_changes
554        .new_owners
555        .iter()
556        .filter_map(|((owner, obj_id), obj_info)| {
557            // If it's in written_coins, then it's not a coin. Skip it.
558            let obj = written_coins.get(obj_id)?;
559            let coin_type_tag = obj.opt_coin_type().cloned().unwrap_or_else(|| {
560                panic!(
561                    "object_id: {obj_id} in written_coins is not a coin type, written_coins: {written_coins:?}, tx_digest: {digest}"
562                )
563            });
564            let coin = obj.as_coin_maybe().unwrap_or_else(|| {
565                panic!(
566                    "object_id: {obj_id} in written_coins cannot be deserialized as a Coin, written_coins: {written_coins:?}, tx_digest: {digest}"
567                )
568            });
569            let map = balance_changes.entry(*owner).or_default();
570            let entry = map.entry(coin_type_tag.clone()).or_insert(TotalBalance {
571                num_coins: 0,
572                balance: 0
573            });
574            let result = self.tables.coin_index.get(&(*owner, coin_type_tag.to_string(), *obj_id));
575            if let Ok(Some(coin_info)) = &result {
576                entry.balance -= coin_info.balance as i128;
577                entry.balance += coin.balance.value() as i128;
578            } else if let Ok(None) = &result {
579                entry.num_coins += 1;
580                entry.balance += coin.balance.value() as i128;
581            }
582            Some(((*owner, coin_type_tag.to_string(), *obj_id), (CoinInfo {version: obj_info.version, digest: obj_info.digest, balance: coin.balance.value(), previous_transaction: *digest})))
583        }).collect::<Vec<_>>();
584        trace!(
585            tx_digest=?digest,
586            "coin_add_keys: {:?}",
587            coin_add_keys,
588        );
589
590        batch.insert_batch(&self.tables.coin_index, coin_add_keys)?;
591
592        let per_coin_type_balance_changes: Vec<_> = balance_changes
593            .iter()
594            .flat_map(|(address, balance_map)| {
595                balance_map.iter().map(|(type_tag, balance)| {
596                    (
597                        (*address, type_tag.clone()),
598                        Ok::<TotalBalance, IotaError>(*balance),
599                    )
600                })
601            })
602            .collect();
603        let all_balance_changes: Vec<_> = balance_changes
604            .into_iter()
605            .map(|(address, balance_map)| {
606                (
607                    address,
608                    Ok::<Arc<HashMap<TypeTag, TotalBalance>>, IotaError>(Arc::new(balance_map)),
609                )
610            })
611            .collect();
612        let cache_updates = IndexStoreCacheUpdates {
613            _locks,
614            per_coin_type_balance_changes,
615            all_balance_changes,
616        };
617        Ok(cache_updates)
618    }

620    /// Indexes a transaction by updating various indices in the `IndexStore`
621    /// with the provided transaction details.
622    pub fn index_tx(
623        &self,
624        sender: Address,
625        active_inputs: impl Iterator<Item = ObjectId>,
626        mutated_objects: impl Iterator<Item = (ObjectReference, Owner)> + Clone,
627        move_functions: impl Iterator<Item = (ObjectId, String, String)> + Clone,
628        events: &TransactionEvents,
629        object_index_changes: ObjectIndexChanges,
630        digest: &TransactionDigest,
631        timestamp_ms: u64,
632        tx_coins: Option<TxCoins>,
633    ) -> IotaResult<u64> {
634        let sequence = self.next_sequence_number.fetch_add(1, Ordering::SeqCst);
635        let mut batch = self.tables.transactions_from_addr.batch();
636
637        batch.insert_batch(
638            &self.tables.transaction_order,
639            std::iter::once((sequence, *digest)),
640        )?;
641
642        batch.insert_batch(
643            &self.tables.transactions_seq,
644            std::iter::once((*digest, sequence)),
645        )?;
646
647        batch.insert_batch(
648            &self.tables.transactions_from_addr,
649            std::iter::once(((sender, sequence), *digest)),
650        )?;
651
652        batch.insert_batch(
653            &self.tables.transactions_by_input_object_id,
654            active_inputs.map(|id| ((id, sequence), *digest)),
655        )?;
656
657        batch.insert_batch(
658            &self.tables.transactions_by_mutated_object_id,
659            mutated_objects
660                .clone()
661                .map(|(obj_ref, _)| ((obj_ref.object_id, sequence), *digest)),
662        )?;
663
664        batch.insert_batch(
665            &self.tables.transactions_by_move_function,
666            move_functions
667                .map(|(obj_id, module, function)| ((obj_id, module, function, sequence), *digest)),
668        )?;
669
670        batch.insert_batch(
671            &self.tables.transactions_to_addr,
672            mutated_objects.filter_map(|(_, owner)| {
673                owner
674                    .into_opt_address()
675                    .map(|addr| ((addr, sequence), digest))
676            }),
677        )?;
678
679        // Coin Index
680        let cache_updates = self.index_coin(digest, &mut batch, &object_index_changes, tx_coins)?;
681
682        // Owner index
683        batch.delete_batch(
684            &self.tables.owner_index,
685            object_index_changes.deleted_owners,
686        )?;
687        batch.delete_batch(
688            &self.tables.dynamic_field_index,
689            object_index_changes.deleted_dynamic_fields,
690        )?;
691
692        batch.insert_batch(&self.tables.owner_index, object_index_changes.new_owners)?;
693
694        batch.insert_batch(
695            &self.tables.dynamic_field_index,
696            object_index_changes.new_dynamic_fields,
697        )?;
698
699        // events
700        let event_digest = events.digest();
701        batch.insert_batch(
702            &self.tables.event_order,
703            events
704                .iter()
705                .enumerate()
706                .map(|(i, _)| ((sequence, i), (event_digest, *digest, timestamp_ms))),
707        )?;
708        batch.insert_batch(
709            &self.tables.event_by_move_module,
710            events
711                .iter()
712                .enumerate()
713                .map(|(i, e)| {
714                    (
715                        i,
716                        ModuleId::new(
717                            AccountAddress::new(e.package_id.into_bytes()),
718                            Identifier::new(e.module.as_str()).unwrap(),
719                        ),
720                    )
721                })
722                .map(|(i, m)| ((m, (sequence, i)), (event_digest, *digest, timestamp_ms))),
723        )?;
724        batch.insert_batch(
725            &self.tables.event_by_sender,
726            events.iter().enumerate().map(|(i, e)| {
727                (
728                    (e.sender, (sequence, i)),
729                    (event_digest, *digest, timestamp_ms),
730                )
731            }),
732        )?;
733        batch.insert_batch(
734            &self.tables.event_by_move_event,
735            events.iter().enumerate().map(|(i, e)| {
736                (
737                    (e.struct_tag.clone(), (sequence, i)),
738                    (event_digest, *digest, timestamp_ms),
739                )
740            }),
741        )?;
742
743        batch.insert_batch(
744            &self.tables.event_by_time,
745            events.iter().enumerate().map(|(i, _)| {
746                (
747                    (timestamp_ms, (sequence, i)),
748                    (event_digest, *digest, timestamp_ms),
749                )
750            }),
751        )?;
752
753        batch.insert_batch(
754            &self.tables.event_by_event_module,
755            events.iter().enumerate().map(|(i, e)| {
756                (
757                    (
758                        ModuleId::new(
759                            AccountAddress::new(e.struct_tag.address().into_bytes()),
760                            Identifier::new(e.struct_tag.module().as_str()).unwrap(),
761                        ),
762                        (sequence, i),
763                    ),
764                    (event_digest, *digest, timestamp_ms),
765                )
766            }),
767        )?;
768
769        let invalidate_caches =
770            read_size_from_env(ENV_VAR_INVALIDATE_INSTEAD_OF_UPDATE).unwrap_or(0) > 0;
771
772        if invalidate_caches {
773            // Invalidate cache before writing to db so we always serve latest values
774            self.invalidate_per_coin_type_cache(
775                cache_updates
776                    .per_coin_type_balance_changes
777                    .iter()
778                    .map(|x| x.0.clone()),
779            )?;
780            self.invalidate_all_balance_cache(
781                cache_updates.all_balance_changes.iter().map(|x| x.0),
782            )?;
783        }
784
785        batch.write()?;
786
787        if !invalidate_caches {
788            // We cannot update the cache before updating the db or else on failing to write
789            // to db we will update the cache twice). However, this only means
790            // cache is eventually consistent with the db (within a very short
791            // delay)
792            self.update_per_coin_type_cache(cache_updates.per_coin_type_balance_changes)?;
793            self.update_all_balance_cache(cache_updates.all_balance_changes)?;
794        }
795        Ok(sequence)
796    }

802    pub fn get_transactions(
803        &self,
804        filter: Option<TransactionFilter>,
805        cursor: Option<TransactionDigest>,
806        limit: Option<usize>,
807        reverse: bool,
808    ) -> IotaResult<Vec<TransactionDigest>> {
809        // Lookup TransactionDigest sequence number,
810        let cursor = if let Some(cursor) = cursor {
811            Some(
812                self.get_transaction_seq(&cursor)?
813                    .ok_or(IotaError::TransactionNotFound { digest: cursor })?,
814            )
815        } else {
816            None
817        };
818        match filter {
819            Some(TransactionFilter::MoveFunction {
820                package,
821                module,
822                function,
823            }) => Ok(self.get_transactions_by_move_function(
824                package, module, function, cursor, limit, reverse,
825            )?),
826            Some(TransactionFilter::InputObject(object_id)) => {
827                Ok(self.get_transactions_by_input_object(object_id, cursor, limit, reverse)?)
828            }
829            Some(TransactionFilter::ChangedObject(object_id)) => {
830                Ok(self.get_transactions_by_mutated_object(object_id, cursor, limit, reverse)?)
831            }
832            Some(TransactionFilter::FromAddress(address)) => {
833                Ok(self.get_transactions_from_addr(address, cursor, limit, reverse)?)
834            }
835            Some(TransactionFilter::ToAddress(address)) => {
836                Ok(self.get_transactions_to_addr(address, cursor, limit, reverse)?)
837            }
838            // NOTE: filter via checkpoint sequence number is implemented in
839            // `get_transactions` of authority.rs.
840            Some(_) => Err(IotaError::UserInput {
841                error: UserInputError::Unsupported(format!("{filter:?}")),
842            }),
843            None => {
844                if reverse {
845                    let iter = self
846                        .tables
847                        .transaction_order
848                        .safe_range_iter_reversed(..=cursor.unwrap_or(TxSequenceNumber::MAX))
849                        .skip(usize::from(cursor.is_some()))
850                        .map(|result| result.map(|(_, digest)| digest));
851                    if let Some(limit) = limit {
852                        Ok(iter.take(limit).collect::<Result<Vec<_>, _>>()?)
853                    } else {
854                        Ok(iter.collect::<Result<Vec<_>, _>>()?)
855                    }
856                } else {
857                    let iter = self
858                        .tables
859                        .transaction_order
860                        .safe_iter_with_bounds(Some(cursor.unwrap_or(TxSequenceNumber::MIN)), None)
861                        .skip(usize::from(cursor.is_some()))
862                        .map(|result| result.map(|(_, digest)| digest));
863                    if let Some(limit) = limit {
864                        Ok(iter.take(limit).collect::<Result<Vec<_>, _>>()?)
865                    } else {
866                        Ok(iter.collect::<Result<Vec<_>, _>>()?)
867                    }
868                }
869            }
870        }
871    }

945    pub fn get_transactions_by_move_function(
946        &self,
947        package: ObjectId,
948        module: Option<String>,
949        function: Option<String>,
950        cursor: Option<TxSequenceNumber>,
951        limit: Option<usize>,
952        reverse: bool,
953    ) -> IotaResult<Vec<TransactionDigest>> {
954        // If we are passed a function with no module return a UserInputError
955        if function.is_some() && module.is_none() {
956            return Err(IotaError::UserInput {
957                error: UserInputError::MoveFunctionInput(
958                    "Cannot supply function without supplying module".to_string(),
959                ),
960            });
961        }
962
963        // We cannot have a cursor without filling out the other keys.
964        if cursor.is_some() && (module.is_none() || function.is_none()) {
965            return Err(IotaError::UserInput {
966                error: UserInputError::MoveFunctionInput(
967                    "Cannot supply cursor without supplying module and function".to_string(),
968                ),
969            });
970        }
971
972        let cursor_val = cursor.unwrap_or(if reverse {
973            TxSequenceNumber::MAX
974        } else {
975            TxSequenceNumber::MIN
976        });
977
978        let max_string = "z".repeat(self.max_type_length.try_into().unwrap());
979        let module_val = module.clone().unwrap_or(if reverse {
980            max_string.clone()
981        } else {
982            "".to_string()
983        });
984
985        let function_val =
986            function
987                .clone()
988                .unwrap_or(if reverse { max_string } else { "".to_string() });
989
990        let key = (package, module_val, function_val, cursor_val);
991        let iter = if reverse {
992            Either::Left(
993                self.tables
994                    .transactions_by_move_function
995                    .safe_range_iter_reversed(..=key),
996            )
997        } else {
998            Either::Right(
999                self.tables
1000                    .transactions_by_move_function
1001                    .safe_iter_with_bounds(Some(key), None),
1002            )
1003        };
1004        iter
1005            // skip one more if exclusive cursor is Some
1006            .skip(usize::from(cursor.is_some()))
1007            .try_take_map_while_and_collect(
1008                limit,
1009                |((id, m, f, _), _)| {
1010                    *id == package
1011                        && module.as_ref().map(|x| x == m).unwrap_or(true)
1012                        && function.as_ref().map(|x| x == f).unwrap_or(true)
1013                },
1014                |(_, digest)| digest,
1015            )
1016            .map_err(Into::into)
1017    }

1279    pub fn get_dynamic_field_object_id(
1280        &self,
1281        object: ObjectId,
1282        name_type: TypeTag,
1283        name_bcs_bytes: &[u8],
1284    ) -> IotaResult<Option<ObjectId>> {
1285        debug!(?object, "get_dynamic_field_object_id");
1286        let dynamic_field_id =
1287            dynamic_field::derive_dynamic_field_id(object, &name_type, name_bcs_bytes).map_err(
1288                |e| {
1289                    IotaError::Unknown(format!(
1290                        "Unable to generate dynamic field id. Got error: {e:?}"
1291                    ))
1292                },
1293            )?;
1294
1295        if let Some(info) = self
1296            .tables
1297            .dynamic_field_index
1298            .get(&(object, dynamic_field_id))?
1299        {
1300            // info.object_id != dynamic_field_id ==> is_wrapper
1301            debug_assert!(
1302                info.object_id == dynamic_field_id
1303                    || matches!(name_type, TypeTag::Struct(tag) if DynamicFieldInfo::is_dynamic_object_field_wrapper(&tag))
1304            );
1305            return Ok(Some(info.object_id));
1306        }
1307

1397                    return false;

1400                    return false;

1402                true
1403            })

1411        &self,
1412        owner: Address,
1413        starting_object_id: ObjectId,

1417            .tables
1418            .owner_index
1419            // The object id 0 is the smallest possible
1420            .safe_iter_with_bounds(Some((owner, starting_object_id)), None)

1427                } else {

1429                }

1449    /// This method first gets the balance from `per_coin_type_balance` cache.
1450    /// On a cache miss, it gets the balance for passed in `coin_type` from
1451    /// the `all_balance` cache. Only on the second cache miss, we go to the
1452    /// database (expensive) and update the cache. Notice that db read is
1453    /// done with `spawn_blocking` as that is expected to block
1454    pub fn get_balance(&self, owner: Address, coin_type: TypeTag) -> IotaResult<TotalBalance> {
1455        self.metrics.balance_lookup_from_total.inc();
1456        let force_disable_cache = read_size_from_env(ENV_VAR_DISABLE_INDEX_CACHE).unwrap_or(0) > 0;
1457        let cloned_coin_type = coin_type.clone();
1458        let metrics_cloned = self.metrics.clone();
1459        let coin_index_cloned = self.tables.coin_index.clone();
1460        if force_disable_cache {
1461            return Self::get_balance_from_db(
1462                metrics_cloned,
1463                coin_index_cloned,
1464                owner,
1465                cloned_coin_type,
1466            )
1467            .map_err(|e| IotaError::Execution(format!("Failed to read balance frm DB: {e:?}")));
1468        }
1469
1470        let balance = self
1471            .caches
1472            .per_coin_type_balance
1473            .get(&(owner, coin_type.clone()));
1474        if let Some(balance) = balance {
1475            return balance;
1476        }
1477        // cache miss, lookup in all balance cache
1478        let all_balance = self.caches.all_balances.get(&owner.clone());
1479        if let Some(Ok(all_balance)) = all_balance {
1480            if let Some(balance) = all_balance.get(&coin_type) {
1481                return Ok(*balance);
1482            }
1483        }
1484        let cloned_coin_type = coin_type.clone();
1485        let metrics_cloned = self.metrics.clone();
1486        let coin_index_cloned = self.tables.coin_index.clone();
1487        self.caches
1488            .per_coin_type_balance
1489            .get_with((owner, coin_type), move || {
1490                Self::get_balance_from_db(
1491                    metrics_cloned,
1492                    coin_index_cloned,
1493                    owner,
1494                    cloned_coin_type,
1495                )
1496                .map_err(|e| IotaError::Execution(format!("Failed to read balance frm DB: {e:?}")))
1497            })
1498    }

1500    /// This method gets the balance for all coin types from the `all_balance`
1501    /// cache. On a cache miss, we go to the database (expensive) and update
1502    /// the cache. This cache is dual purpose in the sense that it not only
1503    /// serves `get_AllBalance()` calls but is also used for serving
1504    /// `get_Balance()` queries. Notice that db read is performed with
1505    /// `spawn_blocking` as that is expected to block
1506    pub fn get_all_balance(
1507        &self,
1508        owner: Address,
1509    ) -> IotaResult<Arc<HashMap<TypeTag, TotalBalance>>> {
1510        self.metrics.all_balance_lookup_from_total.inc();
1511        let force_disable_cache = read_size_from_env(ENV_VAR_DISABLE_INDEX_CACHE).unwrap_or(0) > 0;
1512        let metrics_cloned = self.metrics.clone();
1513        let coin_index_cloned = self.tables.coin_index.clone();
1514
1515        if force_disable_cache {
1516            return Self::get_all_balances_from_db(metrics_cloned, coin_index_cloned, owner)
1517                .map_err(|e| {
1518                    IotaError::Execution(format!("Failed to read all balance from DB: {e:?}"))
1519                });
1520        }
1521
1522        self.caches.all_balances.get_with(owner, move || {
1523            Self::get_all_balances_from_db(metrics_cloned, coin_index_cloned, owner).map_err(|e| {
1524                IotaError::Execution(format!("Failed to read all balance from DB: {e:?}"))
1525            })
1526        })

1529    /// Read balance for a `Address` and `CoinType` from the backend
1530    /// database
1531    pub fn get_balance_from_db(
1532        metrics: Arc<IndexStoreMetrics>,
1533        coin_index: DBMap<CoinIndexKey, CoinInfo>,
1534        owner: Address,
1535        coin_type: TypeTag,
1536    ) -> IotaResult<TotalBalance> {
1537        metrics.balance_lookup_from_db.inc();
1538        let coin_type_str = coin_type.to_string();
1539        let coins =
1540            Self::get_owned_coins_iterator(&coin_index, owner, Some(coin_type_str.clone()))?
1541                .map(|(_coin_type, obj_id, coin)| (coin_type_str.clone(), obj_id, coin));
1542
1543        let mut balance = 0i128;
1544        let mut num_coins = 0;
1545        for (_coin_type, _obj_id, coin_info) in coins {
1546            balance += coin_info.balance as i128;
1547            num_coins += 1;
1548        }
1549        Ok(TotalBalance { balance, num_coins })
1550    }

1552    /// Read all balances for a `Address` from the backend database
1553    pub fn get_all_balances_from_db(
1554        metrics: Arc<IndexStoreMetrics>,
1555        coin_index: DBMap<CoinIndexKey, CoinInfo>,
1556        owner: Address,
1557    ) -> IotaResult<Arc<HashMap<TypeTag, TotalBalance>>> {
1558        metrics.all_balance_lookup_from_db.inc();
1559        let mut balances: HashMap<TypeTag, TotalBalance> = HashMap::new();
1560        let coins = Self::get_owned_coins_iterator(&coin_index, owner, None)?
1561            .chunk_by(|(coin_type, _obj_id, _coin)| coin_type.clone());
1562        for (coin_type, coins) in &coins {
1563            let mut total_balance = 0i128;
1564            let mut coin_object_count = 0;
1565            for (_coin_type, _obj_id, coin_info) in coins {
1566                total_balance += coin_info.balance as i128;
1567                coin_object_count += 1;
1568            }
1569
1570            if coin_object_count == 0 {
1571                // we do not want to return coins with 0 balance
1572                continue;
1573            }
1574
1575            let coin_type = TypeTag::Struct(Box::new(parse_iota_struct_tag(&coin_type).map_err(
1576                |e| IotaError::Execution(format!("Failed to parse event sender address: {e:?}")),
1577            )?));
1578            balances.insert(
1579                coin_type,
1580                TotalBalance {
1581                    num_coins: coin_object_count,
1582                    balance: total_balance,
1583                },
1584            );
1585        }
1586
1587        Ok(Arc::new(balances))
1588    }

1644    fn merge_all_balance(
1645        old_balance: &IotaResult<Arc<HashMap<TypeTag, TotalBalance>>>,
1646        balance_delta: &IotaResult<Arc<HashMap<TypeTag, TotalBalance>>>,
1647    ) -> IotaResult<Arc<HashMap<TypeTag, TotalBalance>>> {
1648        if let Ok(old_balance) = old_balance {
1649            if let Ok(balance_delta) = balance_delta {
1650                // create a deep copy of the old balance hashmap
1651                let mut new_balance = old_balance.as_ref().clone();
1652                for (key, delta) in balance_delta.iter() {
1653                    let old = new_balance.get(key).unwrap_or(&TotalBalance {
1654                        balance: 0,
1655                        num_coins: 0,
1656                    });
1657                    let new_total = TotalBalance {
1658                        balance: old.balance + delta.balance,
1659                        num_coins: old.num_coins + delta.num_coins,
1660                    };
1661
1662                    // Remove entries where num_coins becomes zero to prevent cache bloat
1663                    if new_total.num_coins == 0 {
1664                        new_balance.remove(key);
1665                    } else {
1666                        new_balance.insert(key.clone(), new_total);
1667                    }
1668                }
1669                Ok(Arc::new(new_balance))
1670            } else {
1671                balance_delta.clone()
1672            }
1673        } else {
1674            old_balance.clone()
1675        }
1676    }
