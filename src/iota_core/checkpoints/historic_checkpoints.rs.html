5//! epoch that closed the checkpoint (see `epoch_buckets`).

25/// Column-family prefix of the historic checkpoint buckets; a bucket's
26/// family is `{prefix}{epoch}`.
27const HISTORIC_CHECKPOINTS_CF_PREFIX: &str = "hist_ckpt_e";
28
29/// The directory of each epoch, under the historic root, that this
30/// store's buckets keep their files in.
31const BUCKET_DIR: &str = "checkpoints";
32
33/// Tags of the tables inside a bucket's column family. Do not reuse a tag
34/// for a different table: mark it retired in a comment instead.
35const DB_PREFIX_HISTORIC_CHECKPOINT_CONTENT: u8 = 0;
36const DB_PREFIX_HISTORIC_CHECKPOINT_BY_DIGEST: u8 = 1;
37
38/// Column family holding the earliest-retained-epoch marker
39/// [`EpochBuckets`] persists on a prune; empty until the first prune.
40///
41/// The name must not begin with [`HISTORIC_CHECKPOINTS_CF_PREFIX`], which
42/// marks a bucket's column family.
43const EARLIEST_RETAINED_CF: &str = "hist_ckpt_retention";
44
45/// One epoch's checkpoint history.
46pub struct HistoricCheckpointsBucket {
47    pub(crate) checkpoint_content: TaggedDBMap<CheckpointContentsDigest, CheckpointContents>,
48    pub(crate) checkpoint_by_digest: TaggedDBMap<CheckpointDigest, TrustedCheckpoint>,
49}
50
51impl BucketReopen for HistoricCheckpointsBucket {
52    fn reopen(db: &Arc<Database>, cf_name: &str) -> Result<Self, TypedStoreError> {
53        Ok(Self {
54            checkpoint_content: TaggedDBMap::reopen(
55                db,
56                cf_name,
57                DB_PREFIX_HISTORIC_CHECKPOINT_CONTENT,
58                &ReadWriteOptions::default(),
59                true,
60            )?
61            .with_metrics_label(bucket_metrics_label(cf_name)),
62            checkpoint_by_digest: TaggedDBMap::reopen(
63                db,
64                cf_name,
65                DB_PREFIX_HISTORIC_CHECKPOINT_BY_DIGEST,
66                &ReadWriteOptions::default(),
67                true,
68            )?
69            .with_metrics_label(bucket_metrics_label(cf_name)),
70        })
71    }
72
73    /// Writes the contents every empty checkpoint names, if missing. State
74    /// sync stores a checkpoint's contents only when no bucket holds them yet,
75    /// so without its own copy a bucket's empty checkpoints would lose their
76    /// contents once the bucket holding the shared copy expired.
77    fn initialize(&self) -> Result<(), TypedStoreError> {
78        if self
79            .checkpoint_content
80            .contains_key(&EMPTY_CHECKPOINT_CONTENTS_DIGEST)?
81        {
82            return Ok(());
83        }
84        // Synced, like the creation of the bucket it belongs with.
85        let mut batch = self.checkpoint_content.batch();
86        batch.insert_batch_tagged(
87            &self.checkpoint_content,
88            [(
89                *EMPTY_CHECKPOINT_CONTENTS_DIGEST,
90                empty_checkpoint_contents(),
91            )],
92        )?;
93        batch.write_opt(&synced_write_options())
94    }
95}

97/// Checkpoint history, bucketed by the epoch that closed the checkpoint.
98///
99/// A bucket's existence does **not** mean its epoch has been executed: state
100/// sync inserts checkpoints ahead of execution and across epoch boundaries,
101/// so the newest bucket can belong to an epoch this node has yet to execute.
102/// Retention must therefore count from the epoch being executed, as
103/// [`Self::prune`] does.
104pub struct HistoricCheckpoints {
105    buckets: EpochBuckets<HistoricCheckpointsBucket>,
106}
107
108impl HistoricCheckpoints {
109    /// The `(name, options)` pairs of the column families this store needs,
110    /// to be opened alongside the checkpoint store's own tables.
111    pub fn extra_column_family_options(
112        checkpoint_db_path: &Path,
113        db_options: &DBOptions,
114        historic_root: &Path,
115    ) -> Vec<(String, DBOptions)> {
116        extra_column_family_options(
117            checkpoint_db_path,
118            db_options,
119            HISTORIC_CHECKPOINTS_CF_PREFIX,
120            EARLIEST_RETAINED_CF,
121            Some(&BucketPaths::new(historic_root, BUCKET_DIR)),
122        )
123    }
124
125    /// Opens the historic-checkpoint buckets already present among `db`'s
126    /// column families. `db` is the checkpoint store's own handle, and
127    /// `db_options` and `historic_root` must be the ones it was opened with.
128    pub fn open(
129        db: Arc<Database>,
130        db_options: &DBOptions,
131        historic_root: &Path,
132    ) -> Result<Self, TypedStoreError> {
133        let existing_cfs = list_tables(db.path_for_pruning().to_path_buf())
134            .map_err(|e| TypedStoreError::RocksDB(format!("failed to list buckets: {e}")))?;
135
136        let mut buckets = BTreeMap::new();
137        for cf_name in &existing_cfs {
138            if let Some(epoch) = bucket_cf_epoch(HISTORIC_CHECKPOINTS_CF_PREFIX, cf_name) {
139                let bucket = HistoricCheckpointsBucket::reopen(&db, cf_name)?;
140                // A crash right after creating the bucket left it without.
141                bucket.initialize()?;
142                buckets.insert(epoch, Arc::new(bucket));
143            }
144        }
145
146        let cf_options = bucket_cf_options(db_options).options;
147        if db.cf_handle(EARLIEST_RETAINED_CF).is_none() {
148            db.create_cf(EARLIEST_RETAINED_CF, &cf_options)?;
149        }
150        let earliest_retained_table: DBMap<(), EpochId> = DBMap::reopen(
151            &db,
152            Some(EARLIEST_RETAINED_CF),
153            &ReadWriteOptions::default(),
154            true,
155        )?;
156
157        let buckets = EpochBuckets::open(
158            db,
159            "historic checkpoints",
160            HISTORIC_CHECKPOINTS_CF_PREFIX,
161            cf_options,
162            Some(BucketPaths::new(historic_root, BUCKET_DIR)),
163            earliest_retained_table,
164            buckets,
165        )?;
166        Ok(Self { buckets })
167    }

169    /// The oldest epoch this store still holds a bucket for, `None` when it
170    /// holds none. This is what the store holds, not what its retention would
171    /// keep: a node restored from a formal snapshot starts with no bucket.
172    pub fn earliest_bucket_epoch(&self) -> Option<EpochId> {
173        self.buckets.earliest_epoch()
174    }
175
176    /// The earliest epoch [`Self::prune`] retains; the checkpoint history of
177    /// every epoch below it is gone.
178    pub fn earliest_retained_epoch(&self) -> EpochId {
179        self.buckets.earliest_retained()
180    }
181
182    /// The bucket holding `epoch`'s checkpoint history, created if absent.
183    pub fn ensure(
184        &self,
185        epoch: EpochId,
186    ) -> Result<Arc<HistoricCheckpointsBucket>, TypedStoreError> {
187        self.buckets.ensure(epoch)
188    }
189
190    /// The bucket holding `epoch`'s checkpoint history, and `None` once that
191    /// epoch has been expired. See
192    /// `EpochBuckets::ensure_retained`.
193    pub fn ensure_retained(
194        &self,
195        epoch: EpochId,
196    ) -> Result<Option<Arc<HistoricCheckpointsBucket>>, TypedStoreError> {
197        self.buckets.ensure_retained(epoch)
198    }
199
200    /// Drops the buckets of the epochs that have fallen outside
201    /// `epochs_to_retain`, counted back from `current_epoch` and including
202    /// it, and returns the earliest epoch retained and how many buckets were
203    /// dropped.
204    ///
205    /// `current_epoch` must be the epoch this node is executing (see
206    /// [`HistoricCheckpoints`]); buckets above it are left alone. Blocks while
207    /// the drops run, so async callers must use `spawn_blocking`.
208    ///
209    /// `before_drop` runs before each bucket is dropped, once the earliest
210    /// epoch retained has moved past it, so that what advertises the retained
211    /// range can be moved first.
212    pub fn prune(
213        &self,
214        current_epoch: EpochId,
215        epochs_to_retain: u64,
216        before_drop: &dyn Fn(),
217    ) -> Result<Option<PrunedBuckets>, TypedStoreError> {
218        self.buckets.prune(current_epoch, epochs_to_retain, |_, _| {
219            before_drop();
220            Ok(())
221        })
222    }
223
224    /// Compacts the buckets of the epochs below `current_epoch` that are not
225    /// compacted yet (see
226    /// `EpochBuckets::compact_closed`, which asks
227    /// `stop` before each bucket). Blocks while
228    /// the compactions run, so async callers must use `spawn_blocking`.
229    pub fn compact_closed_buckets(
230        &self,
231        current_epoch: EpochId,
232        stop: &dyn Fn() -> bool,
233    ) -> Result<usize, TypedStoreError> {
234        self.buckets.compact_closed(current_epoch, stop)
235    }

237    /// The contents stored under `digest`, newest bucket first, `None` if no
238    /// bucket holds them (or their bucket has since been dropped).
239    pub fn find_contents(
240        &self,
241        digest: &CheckpointContentsDigest,
242    ) -> Result<Option<CheckpointContents>, TypedStoreError> {
243        Ok(self
244            .find_contents_with_epoch(digest)?
245            .map(|(_, contents)| contents))
246    }
247
248    /// [`Self::find_contents`], with the epoch of the bucket holding them,
249    /// which is the epoch of the checkpoint they belong to. Every bucket
250    /// holds the empty contents; for them, the newest bucket's epoch.
251    pub fn find_contents_with_epoch(
252        &self,
253        digest: &CheckpointContentsDigest,
254    ) -> Result<Option<(EpochId, CheckpointContents)>, TypedStoreError> {
255        for (epoch, bucket) in self.buckets.iter_with_epoch(true) {
256            if let Some(contents) = absent_if_dropped(bucket.checkpoint_content.get(digest))? {
257                return Ok(Some((epoch, contents)));
258            }
259        }
260        Ok(None)
261    }
262
263    /// The certified summary stored under `digest`, newest bucket first,
264    /// `None` if no bucket holds it (or its bucket has since been dropped).
265    ///
266    /// A caller that has the sequence number should read
267    /// `certified_checkpoints` instead, which is never pruned.
268    pub fn find_by_digest(
269        &self,
270        digest: &CheckpointDigest,
271    ) -> Result<Option<TrustedCheckpoint>, TypedStoreError> {
272        for bucket in self.buckets.iter(true) {
273            if let Some(checkpoint) = absent_if_dropped(bucket.checkpoint_by_digest.get(digest))? {
274                return Ok(Some(checkpoint));
275            }
276        }
277        Ok(None)
278    }
279
280    /// One page of the rows `cf_name` holds, if it is one of this store's
281    /// column families, for the table dump of `iota-tool`. `None` for any
282    /// other name. `db` may be a read-only or secondary handle.
283    pub fn dump_column_family(
284        db: &Arc<Database>,
285        cf_name: &str,
286        page_size: u16,
287        page_number: usize,
288    ) -> Result<Option<BTreeMap<String, String>>, TypedStoreError> {
289        if bucket_cf_epoch(HISTORIC_CHECKPOINTS_CF_PREFIX, cf_name).is_some() {
290            let bucket = HistoricCheckpointsBucket::reopen(db, cf_name)?;
291            bucket.checkpoint_content.try_catch_up_with_primary()?;
292            bucket.checkpoint_by_digest.try_catch_up_with_primary()?;
293            let rows = dump_rows("content:", bucket.checkpoint_content.safe_iter()).chain(
294                dump_rows("by_digest:", bucket.checkpoint_by_digest.safe_iter()),
295            );
296            return dump_page(rows, page_size, page_number).map(Some);
297        }
298        if cf_name == EARLIEST_RETAINED_CF {
299            return dump_retention_floor(db, cf_name, page_size, page_number).map(Some);
300        }
301        Ok(None)
302    }
303}
