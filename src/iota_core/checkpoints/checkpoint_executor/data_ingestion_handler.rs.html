21pub(crate) fn load_checkpoint_data(
22    checkpoint_exec_data: &CheckpointExecutionData,
23    checkpoint_tx_data: &CheckpointTransactionData,
24    object_store: &dyn ObjectStore,
25    historic_objects: &HistoricObjects,
26    transaction_cache_reader: &dyn TransactionCacheRead,
27) -> IotaResult<CheckpointData> {
28    let event_tx_digests = checkpoint_tx_data
29        .effects
30        .iter()
31        .flat_map(|fx| fx.events_digest().map(|_| fx.transaction_digest()).copied())
32        .collect::<Vec<_>>();
33
34    let mut events = transaction_cache_reader
35        .try_multi_get_events(&event_tx_digests)?
36        .into_iter()
37        .zip(event_tx_digests)
38        .map(|(maybe_event, tx_digest)| {
39            maybe_event
40                .ok_or(IotaError::TransactionEventsNotFound { digest: tx_digest })
41                .map(|event| (tx_digest, event))
42        })
43        .collect::<IotaResult<HashMap<_, _>>>()?;
44
45    let mut full_transactions = Vec::with_capacity(checkpoint_tx_data.transactions.len());
46    for (tx, fx) in checkpoint_tx_data
47        .transactions
48        .iter()
49        .zip(checkpoint_tx_data.effects.iter())
50    {
51        let events = fx.events_digest().map(|_event_digest| {
52            events
53                .remove(fx.transaction_digest())
54                .expect("event was already checked to be present")
55        });
56
57        let input_objects =
58            historic_objects::get_transaction_input_objects(object_store, historic_objects, fx)?;
59        let output_objects =
60            historic_objects::get_transaction_output_objects(object_store, historic_objects, fx)?;
61
62        let full_transaction = CheckpointTransaction {
63            transaction: (*tx).clone().into_unsigned().into(),
64            effects: fx.clone(),
65            events,
66            input_objects,
67            output_objects,
68        };
69        full_transactions.push(full_transaction);
70    }
71    let checkpoint_data = CheckpointData {
72        checkpoint_summary: checkpoint_exec_data.checkpoint.clone().into(),
73        checkpoint_contents: checkpoint_exec_data.checkpoint_contents.clone(),
74        transactions: full_transactions,
75    };
76    Ok(checkpoint_data)
77}

79/// Build a [`CheckpointData`] for the last checkpoint of an epoch that holds
80/// only its last transaction, the epoch-change transaction, with that
81/// transaction's events and objects.
82///
83/// The summary and contents still list every transaction of the checkpoint,
84/// so the result is only fit for `index_epoch_boundary`, which reads nothing
85/// else. Use [`load_checkpoint_data`] for anything that needs the other
86/// transactions.
87pub(crate) fn load_epoch_boundary_data(
88    checkpoint_exec_data: &CheckpointExecutionData,
89    checkpoint_tx_data: &CheckpointTransactionData,
90    object_store: &dyn ObjectStore,
91    historic_objects: &HistoricObjects,
92    transaction_cache_reader: &dyn TransactionCacheRead,
93) -> IotaResult<CheckpointData> {
94    // The epoch-change transaction is always ordered last in the checkpoint.
95    let transactions = match checkpoint_tx_data
96        .transactions
97        .last()
98        .zip(checkpoint_tx_data.effects.last())
99    {
100        Some((tx, fx)) => {
101            let events = match fx.events_digest() {
102                Some(_) => {
103                    let tx_digest = *fx.transaction_digest();
104                    Some(
105                        transaction_cache_reader
106                            .try_multi_get_events(&[tx_digest])?
107                            .pop()
108                            .flatten()
109                            .ok_or(IotaError::TransactionEventsNotFound { digest: tx_digest })?,
110                    )
111                }
112                None => None,
113            };
114
115            let input_objects = historic_objects::get_transaction_input_objects(
116                object_store,
117                historic_objects,
118                fx,
119            )?;
120            let output_objects = historic_objects::get_transaction_output_objects(
121                object_store,
122                historic_objects,
123                fx,
124            )?;
125
126            vec![CheckpointTransaction {
127                transaction: (*tx).clone().into_unsigned().into(),
128                effects: fx.clone(),
129                events,
130                input_objects,
131                output_objects,
132            }]
133        }
134        None => vec![],
135    };
136
137    Ok(CheckpointData {
138        checkpoint_summary: checkpoint_exec_data.checkpoint.clone().into(),
139        checkpoint_contents: checkpoint_exec_data.checkpoint_contents.clone(),
140        transactions,
141    })
142}

144pub(crate) fn store_checkpoint_locally(
145    path: impl AsRef<Path>,
146    checkpoint_data: &CheckpointData,
147) -> IotaResult {
148    let path = path.as_ref();
149    let file_name = format!("{}.chk", checkpoint_data.checkpoint_summary.sequence_number);
150
151    std::fs::create_dir_all(path).map_err(|err| {
152        IotaError::FileIO(format!(
153            "failed to save full checkpoint content locally {err:?}"
154        ))
155    })?;
156
157    Blob::encode(&checkpoint_data, BlobEncoding::Bcs)
158        .map_err(|_| IotaError::TransactionSerialization {
159            error: "failed to serialize full checkpoint content".to_string(),
160        }) // Map the first error
161        .and_then(|blob| {
162            std::fs::write(path.join(file_name), blob.to_bytes()).map_err(|_| {
163                IotaError::FileIO("failed to save full checkpoint content locally".to_string())
164            })
165        })?;
166
167    Ok(())
168}
