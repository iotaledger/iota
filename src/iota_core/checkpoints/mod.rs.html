421    #[instrument(level = "info", skip_all)]
422    pub fn insert_genesis_checkpoint(
423        &self,
424        checkpoint: VerifiedCheckpoint,
425        contents: CheckpointContents,
426        epoch_store: &AuthorityPerEpochStore,
427    ) {
428        assert_eq!(
429            checkpoint.epoch(),
430            0,
431            "can't call insert_genesis_checkpoint with a checkpoint not in epoch 0"
432        );
433        assert_eq!(
434            checkpoint.sequence_number(),
435            0,
436            "can't call insert_genesis_checkpoint with a checkpoint that doesn't have a sequence number of 0"
437        );
438
439        // Only insert the genesis checkpoint if the DB is empty and doesn't have it
440        // already. Checked by sequence number, since the digest-keyed copy is
441        // gone once epoch 0 has expired.
442        if self
443            .get_checkpoint_by_sequence_number(0)
444            .unwrap()
445            .is_none_or(|stored| stored.digest() != checkpoint.digest())
446        {
447            if epoch_store.epoch() == checkpoint.epoch {
448                epoch_store
449                    .put_genesis_checkpoint_in_builder(checkpoint.data(), &contents)
450                    .unwrap();
451            } else {
452                debug!(
453                    validator_epoch =% epoch_store.epoch(),
454                    genesis_epoch =% checkpoint.epoch(),
455                    "Not inserting checkpoint builder data for genesis checkpoint",
456                );
457            }
458            self.insert_checkpoint_contents(&checkpoint, contents)
459                .unwrap();
460            self.insert_verified_checkpoint(&checkpoint).unwrap();
461            self.update_highest_synced_checkpoint(&checkpoint).unwrap();
462        }
463    }

575    /// The row `watermark` holds — a sequence number and a digest — and `None`
576    /// when it has never been written.
577    ///
578    /// Only the verified, synced and executed rows name a checkpoint. The
579    /// pruner writes a random digest for `HighestPruned`, since nothing reads
580    /// it, so that row's sequence number is the only part worth having. While
581    /// the synced watermark readers see is held in memory, `HighestSynced`
582    /// answers with that one instead of the saved row.
583    fn get_watermark(
584        &self,
585        watermark: CheckpointWatermark,
586    ) -> Result<Option<(CheckpointSequenceNumber, CheckpointDigest)>, TypedStoreError> {
587        if matches!(watermark, CheckpointWatermark::HighestSynced) {
588            if let Some(in_memory) = self.synced_in_memory.get() {
589                return Ok(*in_memory.highest_synced.lock());
590            }
591        }
592        self.tables.watermarks.get(&watermark)
593    }

595    /// The checkpoint `watermark` names.
596    ///
597    /// `None` for a checkpoint the store no longer holds, or one whose
598    /// sequence number now names a different checkpoint. A caller that only
599    /// compares positions wants [`Self::get_watermark_seq_number`].
600    fn get_watermark_checkpoint(
601        &self,
602        watermark: CheckpointWatermark,
603    ) -> Result<Option<VerifiedCheckpoint>, TypedStoreError> {
604        let Some((sequence_number, digest)) = self.get_watermark(watermark)? else {
605            return Ok(None);
606        };
607        Ok(self
608            .get_checkpoint_by_sequence_number(sequence_number)?
609            .filter(|checkpoint| *checkpoint.digest() == digest))
610    }

712    #[instrument(level = "trace", skip_all)]
713    fn check_for_checkpoint_fork(
714        &self,
715        local_checkpoint: &CheckpointSummary,
716        verified_checkpoint: &VerifiedCheckpoint,
717    ) {
718        if local_checkpoint != verified_checkpoint.data() {
719            let verified_contents = self
720                .get_checkpoint_contents(&verified_checkpoint.contents_digest)
721                .map(|opt_contents| {
722                    opt_contents
723                        .map(|contents| format!("{contents:?}"))
724                        .unwrap_or_else(|| {
725                            format!(
726                                "Verified checkpoint contents not found, digest: {:?}",
727                                verified_checkpoint.contents_digest,
728                            )
729                        })
730                })
731                .map_err(|e| {
732                    format!(
733                        "Failed to get verified checkpoint contents, digest: {:?} error: {:?}",
734                        verified_checkpoint.contents_digest, e
735                    )
736                })
737                .unwrap_or_else(|err_msg| err_msg);
738
739            let local_contents = self
740                .get_checkpoint_contents(&local_checkpoint.contents_digest)
741                .map(|opt_contents| {
742                    opt_contents
743                        .map(|contents| format!("{contents:?}"))
744                        .unwrap_or_else(|| {
745                            format!(
746                                "Local checkpoint contents not found, digest: {:?}",
747                                local_checkpoint.contents_digest
748                            )
749                        })
750                })
751                .map_err(|e| {
752                    format!(
753                        "Failed to get local checkpoint contents, digest: {:?} error: {:?}",
754                        local_checkpoint.contents_digest, e
755                    )
756                })
757                .unwrap_or_else(|err_msg| err_msg);
758
759            // checkpoint contents may be too large for panic message.
760            error!(
761                verified_checkpoint = ?verified_checkpoint.data(),
762                ?verified_contents,
763                ?local_checkpoint,
764                ?local_contents,
765                "Local checkpoint fork detected!",
766            );
767            fatal!(
768                "Local checkpoint fork detected for sequence number: {}",
769                local_checkpoint.sequence_number()
770            );
771        }
772    }

788    /// Inserts a batch of certified checkpoints in a single write, like
789    /// [`Self::insert_certified_checkpoint`] for each of them.
790    pub fn multi_insert_certified_checkpoints(
791        &self,
792        checkpoints: &[VerifiedCheckpoint],
793    ) -> Result<(), TypedStoreError> {
794        let Some(last) = checkpoints.last() else {
795            return Ok(());
796        };
797        debug!(
798            checkpoint_seq = last.sequence_number(),
799            count = checkpoints.len(),
800            "Inserting certified checkpoints",
801        );
802        let mut batch = self.tables.certified_checkpoints.batch();
803        batch
804            .insert_batch(
805                &self.tables.certified_checkpoints,
806                checkpoints
807                    .iter()
808                    .map(|c| (c.sequence_number(), c.serializable_ref())),
809            )?
810            .insert_batch(
811                &self.tables.epoch_last_checkpoint_map,
812                checkpoints
813                    .iter()
814                    .filter(|c| c.next_epoch_committee().is_some())
815                    .map(|c| (c.epoch(), c.sequence_number())),
816            )?;
817        // State sync runs ahead of execution and across epoch boundaries, so
818        // this can create the bucket of an epoch not yet executed (see
819        // `HistoricCheckpoints`), or arrive after the epoch has expired.
820        for checkpoint in checkpoints {
821            let Some(bucket) = self
822                .historic_checkpoints
823                .ensure_retained(checkpoint.epoch())?
824            else {
825                debug!(
826                    checkpoint_seq = checkpoint.sequence_number(),
827                    epoch = checkpoint.epoch(),
828                    "not filing a checkpoint summary whose epoch has been expired",
829                );
830                continue;
831            };
832            batch.insert_batch_tagged(
833                &bucket.checkpoint_by_digest,
834                [(checkpoint.digest(), checkpoint.serializable_ref())],
835            )?;
836        }
837        // An epoch can expire between the staging above and the write.
838        batch.write_opt(&drop_tolerant_write_options())?;
839
840        for checkpoint in checkpoints
841            .iter()
842            .filter(|c| c.next_epoch_committee().is_some())
843        {
844            self.epoch_last_checkpoint_notify_read
845                .notify(&checkpoint.epoch(), &checkpoint.sequence_number());
846        }
847
848        for checkpoint in checkpoints {
849            if let Some(local_checkpoint) = self
850                .tables
851                .locally_computed_checkpoints
852                .get(&checkpoint.sequence_number())?
853            {
854                self.check_for_checkpoint_fork(&local_checkpoint, checkpoint);
855            }
856        }
857
858        Ok(())
859    }

899    /// Marks a consecutive run of checkpoints as synced: writes the watermark
900    /// once, for the last checkpoint, and notifies the waiters that write
901    /// releases. Does nothing when another writer has already carried the
902    /// watermark past the run, or when the run starts above the checkpoint
903    /// after the watermark. Returns whether the watermark moved.
904    ///
905    /// Tooling that has to set the watermark regardless uses
906    /// [`Self::set_highest_synced_checkpoint_subtle`].
907    pub fn multi_update_highest_synced_checkpoint(
908        &self,
909        checkpoints: &[VerifiedCheckpoint],
910    ) -> Result<bool, TypedStoreError> {
911        let Some(last) = checkpoints.last() else {
912            return Ok(false);
913        };
914
915        // Only ever forwards, like the other two: another writer advances this
916        // as well, and moving it back offers contents that may not be there.
917        let previous = self.get_highest_synced_checkpoint_seq_number()?;
918        // The watermark promises that the contents of every checkpoint up to
919        // it are stored, and a writer stores them only for the run it marks.
920        // A run that starts above the checkpoint after the watermark would
921        // pass checkpoints whose contents may still be missing; they are
922        // marked synced once the checkpoints before them are.
923        let first = checkpoints[0].sequence_number();
924        if previous.is_some_and(|previous_seq_number| first > previous_seq_number + 1) {
925            debug!(
926                checkpoint_seq = first,
927                highest_synced = previous,
928                "Not marking checkpoints synced before the ones below them",
929            );
930            return Ok(false);
931        }
932        if previous.is_none_or(|previous_seq_number| previous_seq_number < last.sequence_number()) {
933            debug!(
934                checkpoint_seq = last.sequence_number(),
935                "Updating highest synced checkpoint",
936            );
937            self.write_highest_synced(last)?;
938
939            // A reader parks only below the watermark, so only what this
940            // write has just passed can have anyone waiting on it.
941            for checkpoint in checkpoints.iter().filter(|c| {
942                previous.is_none_or(|previous_seq_number| previous_seq_number < c.sequence_number())
943            }) {
944                self.synced_checkpoint_notify_read
945                    .notify(&checkpoint.sequence_number(), checkpoint);
946            }
947            return Ok(true);
948        }
949        Ok(false)
950    }

1083    /// Moves the synced watermark readers see to `checkpoint`, and the saved
1084    /// one towards it over the checkpoints whose contents are stored.
1085    fn write_highest_synced(&self, checkpoint: &VerifiedCheckpoint) -> Result<(), TypedStoreError> {
1086        let _writes = self.synced_writes.lock();
1087        let row = (checkpoint.sequence_number(), *checkpoint.digest());
1088        let Some(in_memory) = self.synced_in_memory.get() else {
1089            return self
1090                .tables
1091                .watermarks
1092                .insert(&CheckpointWatermark::HighestSynced, &row);
1093        };
1094        {
1095            let mut highest_synced = in_memory.highest_synced.lock();
1096            if highest_synced
1097                .is_none_or(|(sequence_number, _)| sequence_number < checkpoint.sequence_number())
1098            {
1099                *highest_synced = Some(row);
1100            }
1101        }
1102        self.advance_saved_synced(in_memory)
1103    }

1105    /// Moves the saved synced watermark forwards over the checkpoints, up to
1106    /// the one readers see, that have executed or whose transactions and
1107    /// effects are stored, stopping at the first that has neither. Call with
1108    /// `synced_writes` held.
1109    fn advance_saved_synced(
1110        &self,
1111        in_memory: &SyncedWatermarkInMemory,
1112    ) -> Result<(), TypedStoreError> {
1113        let Some((highest_synced, _)) = *in_memory.highest_synced.lock() else {
1114            return Ok(());
1115        };
1116        let saved = self
1117            .tables
1118            .watermarks
1119            .get(&CheckpointWatermark::HighestSynced)?
1120            .map(|(sequence_number, _)| sequence_number);
1121        let highest_executed = self.get_highest_executed_checkpoint_seq_number()?;
1122        let mut advanced_to = None;
1123        for sequence_number in saved.map_or(0, |saved| saved + 1)..=highest_synced {
1124            let stored = highest_executed.is_some_and(|executed| sequence_number <= executed)
1125                || self.contents_stored(sequence_number, &in_memory.historic_ledger)?;
1126            if !stored {
1127                break;
1128            }
1129            advanced_to = Some(sequence_number);
1130        }
1131        let Some(sequence_number) = advanced_to else {
1132            return Ok(());
1133        };
1134        let checkpoint = self
1135            .get_checkpoint_by_sequence_number(sequence_number)?
1136            .expect("a checkpoint at or below the synced watermark is stored");
1137        self.tables.watermarks.insert(
1138            &CheckpointWatermark::HighestSynced,
1139            &(sequence_number, *checkpoint.digest()),
1140        )
1141    }

1174    /// From now on, holds the synced watermark that callers read in memory.
1175    /// The value saved in the database passes a checkpoint only once that
1176    /// checkpoint has executed or its transactions and effects are in the
1177    /// ledger bucket of its epoch.
1178    ///
1179    /// The synced watermark promises that the contents of every checkpoint up
1180    /// to it can be loaded. A checkpoint can be marked synced while its
1181    /// transactions and effects are only in memory: this node certified it
1182    /// after building it, or state sync found its contents in memory. With
1183    /// the watermark held in memory, a restarted node starts from a synced
1184    /// checkpoint whose contents can be loaded, and state sync marks the later
1185    /// checkpoints synced again.
1186    ///
1187    /// Turning this on does not change the value callers read. At startup,
1188    /// call it once the one-time startup passes are done and before state
1189    /// sync starts: until then only the genesis checkpoint, whose contents are
1190    /// in the ledger already, and the ledger migration's rewind write the
1191    /// synced watermark. A running node may call it at an epoch change, after
1192    /// every checkpoint of the ending epoch has executed. Once on, it stays on
1193    /// until the process stops, also in later epochs where the node is not in
1194    /// the committee. There, state sync marks a checkpoint synced only after
1195    /// its transactions and effects are in the ledger, so the saved value
1196    /// moves with the one callers read. Does nothing if already on.
1197    pub fn hold_synced_watermark_in_memory(
1198        &self,
1199        historic_ledger: Arc<HistoricLedger>,
1200    ) -> Result<(), TypedStoreError> {
1201        let _writes = self.synced_writes.lock();
1202        if self.synced_in_memory.get().is_some() {
1203            return Ok(());
1204        }
1205        let saved = self
1206            .tables
1207            .watermarks
1208            .get(&CheckpointWatermark::HighestSynced)?;
1209        self.synced_in_memory
1210            .get_or_init(|| SyncedWatermarkInMemory {
1211                historic_ledger,
1212                highest_synced: Mutex::new(saved),
1213            });
1214        Ok(())
1215    }

1217    /// Moves `HighestPruned` up to the last checkpoint of the epoch below
1218    /// `earliest_retained_epoch`, after that epoch's checkpoint history has
1219    /// been dropped. Never moves the watermark backwards.
1220    pub fn advance_highest_pruned_checkpoint(
1221        &self,
1222        earliest_retained_epoch: EpochId,
1223    ) -> IotaResult<()> {
1224        let Some(previous_epoch) = earliest_retained_epoch.checked_sub(1) else {
1225            return Ok(());
1226        };
1227        let Some(seq) = self.get_epoch_last_checkpoint_seq_number(previous_epoch)? else {
1228            return Ok(());
1229        };
1230        if self
1231            .get_highest_pruned_checkpoint_seq_number()?
1232            .is_some_and(|pruned| pruned >= seq)
1233        {
1234            return Ok(());
1235        }
1236        // The watermark stores the digest too, so the summary is needed.
1237        let Some(checkpoint) = self.get_epoch_last_checkpoint(previous_epoch)? else {
1238            return Ok(());
1239        };
1240        self.update_highest_pruned_checkpoint(&checkpoint)?;
1241        Ok(())
1242    }

1245    /// it forwards.
1246    ///
1247    /// Only for tooling that deliberately rewinds it. The node writes this row
1248    /// through [`Self::update_highest_verified_checkpoint`], which declines to
1249    /// move it backwards, under a lock its callers share.

1257        )

1260    /// Sets the synced watermark to `checkpoint` whether or not that moves it
1261    /// forwards.
1262    ///
1263    /// Only for tooling that deliberately rewinds it. The node writes this row
1264    /// through [`Self::multi_update_highest_synced_checkpoint`], which declines
1265    /// to move it backwards, under a lock its callers share.
1266    pub fn set_highest_synced_checkpoint_subtle(
1267        &self,
1268        checkpoint: &VerifiedCheckpoint,
1269    ) -> Result<(), TypedStoreError> {
1270        let _writes = self.synced_writes.lock();
1271        let row = (checkpoint.sequence_number(), *checkpoint.digest());
1272        if let Some(in_memory) = self.synced_in_memory.get() {
1273            *in_memory.highest_synced.lock() = Some(row);
1274        }
1275        self.tables
1276            .watermarks
1277            .insert(&CheckpointWatermark::HighestSynced, &row)
1278    }

1280    /// Brings the synced watermark back to the executed one, so that state
1281    /// sync fetches the checkpoints between the two again, and returns the
1282    /// sequence number it now names. Does nothing when execution has already
1283    /// caught up.
1284    ///
1285    /// Writes only the saved watermark, so it must run before
1286    /// [`Self::hold_synced_watermark_in_memory`].
1287    pub fn rewind_highest_synced_to_executed(
1288        &self,
1289    ) -> Result<Option<CheckpointSequenceNumber>, TypedStoreError> {
1290        debug_assert!(
1291            self.synced_in_memory.get().is_none(),
1292            "the synced watermark is rewound before it is held in memory"
1293        );
1294        let Some(executed) = self.get_watermark(CheckpointWatermark::HighestExecuted)? else {
1295            return Ok(None);
1296        };
1297        let synced = self.get_watermark_seq_number(CheckpointWatermark::HighestSynced)?;
1298        if synced.is_none_or(|synced| synced <= executed.0) {
1299            return Ok(synced);
1300        }
1301        self.tables
1302            .watermarks
1303            .insert(&CheckpointWatermark::HighestSynced, &executed)?;
1304        Ok(Some(executed.0))
1305    }

1367    /// Batch variant of [`Self::insert_verified_checkpoint_contents`]:
1368    /// persists all contents rows in a single write, then caches the full
1369    /// contents.
1370    ///
1371    /// INVARIANT: See [`Self::cache_full_checkpoint_contents`].
1372    pub fn multi_insert_verified_checkpoint_contents(
1373        &self,
1374        checkpoints: Vec<(VerifiedCheckpoint, VerifiedCheckpointContents)>,
1375    ) -> Result<(), TypedStoreError> {
1376        let checkpoints: Vec<_> = checkpoints
1377            .into_iter()
1378            .map(|(checkpoint, full_contents)| (checkpoint, full_contents.into_inner()))
1379            .collect();
1380
1381        let mut batch = self.tables.checkpoint_content.batch();
1382        for (checkpoint, full_contents) in &checkpoints {
1383            let contents = full_contents.checkpoint_contents();
1384            assert_eq!(checkpoint.contents_digest, contents.digest());
1385            let Some(bucket) = self
1386                .historic_checkpoints
1387                .ensure_retained(checkpoint.epoch())?
1388            else {
1389                debug!(
1390                    checkpoint_seq = checkpoint.sequence_number(),
1391                    epoch = checkpoint.epoch(),
1392                    "not filing checkpoint contents whose epoch has been expired",
1393                );
1394                continue;
1395            };
1396            batch
1397                .insert_batch_tagged(&bucket.checkpoint_content, [(contents.digest(), contents)])?;
1398        }
1399        // An epoch can expire between the staging above and the write.
1400        batch.write_opt(&drop_tolerant_write_options())?;
1401
1402        for (checkpoint, full_contents) in checkpoints {
1403            self.cache_full_checkpoint_contents(
1404                checkpoint.sequence_number(),
1405                checkpoint.contents_digest,
1406                full_contents,
1407            );
1408        }
1409        Ok(())
1410    }

1420    /// Caches full checkpoint contents in memory without writing anything to
1421    /// disk, so state-sync peers can be served without reconstructing the
1422    /// contents. `contents_digest` must be the digest of `full_contents`.
1423    ///
1424    /// INVARIANT: The caller must have durably written the matching
1425    /// `checkpoint_content` row (and the contained transactions and
1426    /// effects) first for two reasons:
1427    /// 1. once the cache evicts the entry (or after a restart), readers reconstruct the full
1428    ///    contents from those stores.
1429    /// 2. state-sync treats available contents as proof of that row and skips its own durable
1430    ///    write, and the checkpoint executor panics on a missing row.
1431    ///
1432    /// Best-effort: a serialization failure is logged and the insert skipped;
1433    /// readers fall back to reconstructing the contents from the durable
1434    /// stores.
1435    pub fn cache_full_checkpoint_contents(
1436        &self,
1437        sequence_number: CheckpointSequenceNumber,
1438        contents_digest: CheckpointContentsDigest,
1439        full_contents: FullCheckpointContents,
1440    ) {
1441        let size = match bcs::serialized_size(&full_contents) {
1442            Ok(size) => size,
1443            Err(e) => {
1444                warn!(
1445                    sequence_number,
1446                    "failed to serialize full checkpoint contents for caching: {e}"
1447                );
1448                return;
1449            }
1450        };
1451        self.full_checkpoint_contents_cache.insert(
1452            sequence_number,
1453            contents_digest,
1454            Arc::new(full_contents),
1455            size,
1456        );
1457    }

1481    /// The lowest checkpoint whose superseded object versions this node still
1482    /// holds, given the epoch of its oldest historic-object bucket: that
1483    /// epoch's first checkpoint. When there is no bucket, or that checkpoint
1484    /// is unknown, returns one past the highest executed checkpoint. Never
1485    /// below one past `objects_pruned_through`, the checkpoint through which
1486    /// the objects pruner of earlier releases deleted superseded versions.
1487    ///
1488    /// While the oldest bucket is that of `backlog_sweep_epoch`, the epoch
1489    /// whose bucket the one-time sweep moved the versions superseded before
1490    /// the upgrade into, it holds every version since `objects_pruned_through`,
1491    /// or all of them where that pruner never ran.
1492    pub fn lowest_checkpoint_with_retained_objects(
1493        &self,
1494        earliest_bucket_epoch: Option<EpochId>,
1495        objects_pruned_through: Option<CheckpointSequenceNumber>,
1496        backlog_sweep_epoch: Option<EpochId>,
1497    ) -> Result<CheckpointSequenceNumber, TypedStoreError> {
1498        let lowest = match earliest_bucket_epoch {
1499            Some(epoch) if Some(epoch) == backlog_sweep_epoch => {
1500                Some(objects_pruned_through.map_or(0, |pruned| pruned + 1))
1501            }
1502            Some(0) => Some(0),
1503            Some(epoch) => self
1504                .get_epoch_last_checkpoint_seq_number(epoch - 1)?
1505                .map(|seq| seq + 1),
1506            None => None,
1507        };
1508        let lowest = match lowest {
1509            Some(lowest) => lowest,
1510            None => self
1511                .get_highest_executed_checkpoint_seq_number()?
1512                .map(|seq| seq + 1)
1513                .unwrap_or(0),
1514        };
1515        Ok(objects_pruned_through.map_or(lowest, |pruned| lowest.max(pruned + 1)))
1516    }

1723    /// This function first waits for ConsensusCommitHandler to finish
1724    /// reprocessing commits that have been processed before the last
1725    /// restart, if consensus_replay_waiter is supplied. Then it starts
1726    /// building checkpoints in a loop.
1727    ///
1728    /// It is optional to pass in consensus_replay_waiter, to make it easier to
1729    /// attribute if slow recovery of previously built checkpoints is due to
1730    /// consensus replay or checkpoint building.
1731    ///
1732    /// Stops once this node executes downloaded checkpoints for the rest of
1733    /// the epoch, also in the middle of building, so it signs nothing more.
1734    async fn run(self, consensus_replay_waiter: Option<ReplayWaiter>) {
1735        let epoch_store = self.epoch_store.clone();
1736        tokio::select! {
1737            _ = epoch_store.wait_for_synced_checkpoint_execution() => {
1738                info!("CheckpointBuilder stopping: executing synced checkpoints for the rest of the epoch");
1739            }
1740            _ = self.build_until_stopped(consensus_replay_waiter) => {}
1741        }
1742    }

1744    async fn build_until_stopped(mut self, consensus_replay_waiter: Option<ReplayWaiter>) {
1745        if let Some(replay_waiter) = consensus_replay_waiter {
1746            info!("Waiting for consensus commits to replay ...");
1747            replay_waiter.wait_for_replay().await;
1748            info!("Consensus commits finished replaying");
1749        }
1750        info!("Starting CheckpointBuilder");
1751        loop {
1752            match self.maybe_build_checkpoints().await {
1753                Ok(()) => {}
1754                err @ Err(
1755                    CheckpointBuilderError::ChangeEpochTxAlreadyExecuted
1756                    | CheckpointBuilderError::SystemPackagesMissing,
1757                ) => {
1758                    info!("CheckpointBuilder stopping: {:?}", err);
1759                    return;
1760                }
1761                Err(CheckpointBuilderError::Retry(inner)) => {
1762                    let msg = format!("{inner:?}");
1763                    debug_fatal!("Error while making checkpoint, will retry in 1s: {}", msg);
1764                    tokio::time::sleep(Duration::from_secs(1)).await;
1765                    self.metrics.checkpoint_errors.inc();
1766                    continue;
1767                }
1768            }
1769
1770            self.notify.notified().await;
1771        }
1772    }

1774    async fn maybe_build_checkpoints(&mut self) -> CheckpointBuilderResult {
1775        let _scope = monitored_scope("BuildCheckpoints");
1776
1777        // Collect info about the most recently built checkpoint.
1778        let summary = self
1779            .epoch_store
1780            .last_built_checkpoint_builder_summary()
1781            .expect("epoch should not have ended");
1782        let mut last_height = summary.as_ref().and_then(|s| s.checkpoint_height);
1783        let mut last_timestamp = summary.as_ref().map(|s| s.summary.timestamp_ms);
1784        let mut last_seq = summary.map(|s| s.summary.sequence_number);
1785
1786        let min_checkpoint_interval_ms = self
1787            .epoch_store
1788            .protocol_config()
1789            .min_checkpoint_interval_ms_as_option()
1790            .unwrap_or_default();
1791        // When set, the interval may additionally be amortized over this many
1792        // recent checkpoints of the current epoch.
1793        let checkpoint_rate_window_size = self
1794            .epoch_store
1795            .protocol_config()
1796            .checkpoint_rate_window_size_as_option();
1797        let mut grouped_pending_checkpoints = Vec::new();
1798        let mut checkpoints_iter = self
1799            .epoch_store
1800            .get_pending_checkpoints(last_height)
1801            .expect("unexpected epoch store error")
1802            .into_iter()
1803            .peekable();
1804        while let Some((height, pending)) = checkpoints_iter.next() {
1805            let current_timestamp = pending.details().timestamp_ms;
1806            // Strict interval against the immediately preceding checkpoint.
1807            let adjacent_interval_elapsed = match last_timestamp {
1808                Some(last_timestamp) => {
1809                    current_timestamp >= last_timestamp + min_checkpoint_interval_ms
1810                }
1811                None => true,
1812            };
1813            // Windowed arm: also allow building once the checkpoint `window`
1814            // back is at least `window * interval` older. This recycles the
1815            // slack the strict arm loses to discrete commit timestamps,
1816            // holding the sustained rate at the ceiling, while the strict arm
1817            // keeps every quiet gap — and thus checkpoint sizes — within one
1818            // interval. Only checkpoints built in the current epoch are
1819            // consulted: every validator builds all of them, so the look-back
1820            // resolves identically everywhere and the gate stays
1821            // deterministic. Until the window fills (epoch start, chain
1822            // genesis) the arm is inert.
1823            let interval_elapsed = adjacent_interval_elapsed
1824                || checkpoint_rate_window_size.is_some_and(|window| {
1825                    last_seq
1826                        .and_then(|seq| (seq + 1).checked_sub(window))
1827                        .and_then(|window_start_seq| {
1828                            self.epoch_store
1829                                .get_built_checkpoint_summary(window_start_seq)
1830                                .expect("epoch store should not error reading a built checkpoint summary")
1831                        })
1832                        .is_some_and(|window_start| {
1833                            current_timestamp
1834                                >= window_start.timestamp_ms + window * min_checkpoint_interval_ms
1835                        })
1836                });
1837            // Group PendingCheckpoints until:
1838            // - the minimum interval has elapsed ...
1839            let can_build = interval_elapsed
1840                // - or, next PendingCheckpoint is last-of-epoch (since the last-of-epoch checkpoint
1841                //   should be written separately) ...
1842                || checkpoints_iter
1843                .peek()
1844                .is_some_and(|(_, next_pending)| next_pending.details().last_of_epoch)
1845                // - or, we have reached end of epoch.
1846                || pending.details().last_of_epoch;
1847            grouped_pending_checkpoints.push(pending);
1848            if !can_build {
1849                debug!(
1850                    checkpoint_commit_height = height,
1851                    ?last_timestamp,
1852                    ?current_timestamp,
1853                    "waiting for more PendingCheckpoints: minimum interval not yet elapsed"
1854                );
1855                continue;
1856            }
1857
1858            // Min interval has elapsed, we can now coalesce and build a checkpoint.
1859            last_height = Some(height);
1860            last_timestamp = Some(current_timestamp);
1861            let commits_in_checkpoint = grouped_pending_checkpoints.len();
1862            debug!(
1863                checkpoint_commit_height_from = grouped_pending_checkpoints
1864                    .first()
1865                    .unwrap()
1866                    .details()
1867                    .checkpoint_height,
1868                checkpoint_commit_height_to = last_height,
1869                "Making checkpoint with commit height range"
1870            );
1871
1872            let seq = self
1873                .make_checkpoint(std::mem::take(&mut grouped_pending_checkpoints))
1874                .await?;
1875
1876            // Count only on success; a failed build retries the same
1877            // group and would otherwise double-count it.
1878            self.metrics
1879                .commits_per_checkpoint
1880                .observe(commits_in_checkpoint as f64);
1881            // Advance the window anchor to the highest checkpoint just
1882            // built (a single call may emit several when chunked).
1883            last_seq = Some(seq);
1884            self.last_built.send_if_modified(|cur| {
1885                // when rebuilding checkpoints at startup, seq can be for an old checkpoint
1886                if seq > *cur {
1887                    *cur = seq;
1888                    true
1889                } else {
1890                    false
1891                }
1892            });
1893
1894            // Ensure that the task can be cancelled at end of epoch, even if no other await
1895            // yields execution.
1896            tokio::task::yield_now().await;
1897        }
1898        debug!(
1899            "Waiting for more checkpoints from consensus after processing {last_height:?}; {} pending checkpoints left unprocessed until next interval",
1900            grouped_pending_checkpoints.len(),
1901        );
1902
1903        Ok(())
1904    }

1906    #[instrument(level = "debug", skip_all, fields(last_height = pendings.last().unwrap().details().checkpoint_height
1907    ))]
1908    async fn make_checkpoint(
1909        &self,
1910        pendings: Vec<PendingCheckpoint>,
1911    ) -> CheckpointBuilderResult<CheckpointSequenceNumber> {
1912        let _scope = monitored_scope("CheckpointBuilder::make_checkpoint");
1913        let last_details = pendings.last().unwrap().details().clone();
1914
1915        // Stores the transactions that should be included in the checkpoint.
1916        // Transactions will be recorded in the checkpoint in this order.
1917        let highest_executed_sequence = self
1918            .store
1919            .get_highest_executed_checkpoint_seq_number()
1920            .expect("db error")
1921            .unwrap_or(0);
1922
1923        let (poll_count, result) = poll_count(self.resolve_checkpoint_transactions(pendings)).await;
1924        let (sorted_tx_effects_included_in_checkpoint, all_roots) = result?;
1925

1927            .create_checkpoints(
1928                sorted_tx_effects_included_in_checkpoint,
1929                &last_details,
1930                &all_roots,
1931            )

1934        if highest_sequence <= highest_executed_sequence && poll_count > 1 {
1935            debug_fatal!(
1936                "resolve_checkpoint_transactions should be instantaneous when executed checkpoint is ahead of checkpoint builder"
1937            );
1938        }

1941            .await?;

1945    // Given the root transactions of the pending checkpoints, resolve the
1946    // transactions that should be included in the checkpoint, and return them in
1947    // the order they should be included in the checkpoint, together with all the
1948    // resolved roots.
1949    #[instrument(level = "debug", skip_all)]
1950    async fn resolve_checkpoint_transactions(
1951        &self,
1952        pending_checkpoints: Vec<PendingCheckpoint>,
1953    ) -> IotaResult<(Vec<TransactionEffects>, HashSet<TransactionDigest>)> {
1954        let _scope = monitored_scope("CheckpointBuilder::resolve_checkpoint_transactions");
1955
1956        // Keeps track of the effects that are already included in the current
1957        // checkpoint. This is used when there are multiple pending checkpoints
1958        // to create a single checkpoint because in such scenarios, dependencies
1959        // of a transaction may in earlier created checkpoints, or in earlier
1960        // pending checkpoints.
1961        let mut effects_in_current_checkpoint = BTreeSet::new();
1962
1963        let mut tx_effects = Vec::new();
1964        let mut tx_roots = HashSet::new();
1965
1966        for pending_checkpoint in pending_checkpoints.into_iter() {
1967            let pending = pending_checkpoint.into_v1();
1968            debug!(
1969                checkpoint_commit_height = pending.details.checkpoint_height,
1970                roots = ?pending.roots,
1971                "Resolving checkpoint transactions for pending checkpoint.",
1972            );
1973
1974            let roots = &pending.roots;
1975
1976            self.metrics
1977                .checkpoint_roots_count
1978                .inc_by(roots.len() as u64);
1979
1980            let root_digests = self
1981                .epoch_store
1982                .notify_read_tx_key_to_digest(roots)
1983                .in_monitored_scope("CheckpointNotifyDigests")
1984                .await?;
1985            let root_effects = self
1986                .effects_store
1987                .try_notify_read_executed_effects(
1988                    CHECKPOINT_BUILDER_NOTIFY_READ_TASK_NAME,
1989                    &root_digests,
1990                )
1991                .in_monitored_scope("CheckpointNotifyRead")
1992                .await?;
1993
1994            let consensus_commit_prologue = {
1995                // If the roots contains consensus commit prologue transaction, we want to
1996                // extract it, and put it to the front of the checkpoint.
1997
1998                let consensus_commit_prologue =
1999                    self.extract_consensus_commit_prologue(&root_digests, &root_effects)?;
2000
2001                // Get the un-included dependencies of the consensus commit prologue. We should
2002                // expect no other dependencies that haven't been included in
2003                // any previous checkpoints.
2004                if let Some((ccp_digest, ccp_effects)) = &consensus_commit_prologue {
2005                    let unsorted_ccp = self.complete_checkpoint_effects(
2006                        vec![ccp_effects.clone()],
2007                        &mut effects_in_current_checkpoint,
2008                    )?;
2009
2010                    // No other dependencies of this consensus commit prologue that haven't been
2011                    // included in any previous checkpoint.
2012                    if unsorted_ccp.len() != 1 {
2013                        fatal!(
2014                            "Expected 1 consensus commit prologue, got {:?}",
2015                            unsorted_ccp
2016                                .iter()
2017                                .map(|e| e.transaction_digest())
2018                                .collect::<Vec<_>>()
2019                        );
2020                    }
2021                    assert_eq!(unsorted_ccp.len(), 1);
2022                    assert_eq!(unsorted_ccp[0].transaction_digest(), ccp_digest);
2023                }
2024                consensus_commit_prologue
2025            };
2026
2027            let unsorted =
2028                self.complete_checkpoint_effects(root_effects, &mut effects_in_current_checkpoint)?;
2029
2030            let _scope = monitored_scope("CheckpointBuilder::causal_sort");
2031            let mut sorted: Vec<TransactionEffects> = Vec::with_capacity(unsorted.len() + 1);
2032            if let Some((_ccp_digest, ccp_effects)) = consensus_commit_prologue {
2033                #[cfg(debug_assertions)]
2034                {
2035                    // When consensus_commit_prologue is extracted, it should not be included in the
2036                    // `unsorted`.
2037                    for tx in unsorted.iter() {
2038                        assert!(tx.transaction_digest() != &_ccp_digest);
2039                    }
2040                }
2041                sorted.push(ccp_effects);
2042            }
2043            sorted.extend(CausalOrder::causal_sort(unsorted));
2044
2045            #[cfg(msim)]
2046            {
2047                // Check consensus commit prologue invariants in sim test.
2048                self.expensive_consensus_commit_prologue_invariants_check(&root_digests, &sorted);
2049            }
2050
2051            tx_effects.extend(sorted);
2052            tx_roots.extend(root_digests);
2053        }
2054
2055        Ok((tx_effects, tx_roots))
2056    }

2058    // This function is used to extract the consensus commit prologue digest and
2059    // effects from the root transactions.
2060    // The consensus commit prologue is expected to be the first transaction in the
2061    // roots.
2062    fn extract_consensus_commit_prologue(
2063        &self,
2064        root_digests: &[TransactionDigest],
2065        root_effects: &[TransactionEffects],
2066    ) -> IotaResult<Option<(TransactionDigest, TransactionEffects)>> {
2067        let _scope = monitored_scope("CheckpointBuilder::extract_consensus_commit_prologue");
2068        if root_digests.is_empty() {
2069            return Ok(None);
2070        }
2071
2072        // Reads the first transaction in the roots, and checks whether it is a
2073        // consensus commit prologue transaction.
2074        // The consensus commit prologue transaction should be the first transaction
2075        // in the roots written by the consensus handler.
2076        let first_tx = self
2077            .state
2078            .get_transaction_cache_reader()
2079            .try_get_transaction_block(&root_digests[0])?
2080            .expect("Transaction block must exist");
2081
2082        Ok(match first_tx.transaction().kind() {
2083            TransactionKind::ConsensusCommitPrologueV1(_) => {
2084                assert_eq!(first_tx.digest(), root_effects[0].transaction_digest());
2085                Some((*first_tx.digest(), root_effects[0].clone()))
2086            }
2087            _ => None,
2088        })
2089    }

2218    #[expect(clippy::type_complexity)]
2219    fn split_checkpoint_chunks(
2220        &self,
2221        transactions_effects_and_sizes: Vec<(TransactionEnvelope, TransactionEffects, usize)>,
2222        signatures: Vec<Vec<UserSignature>>,
2223    ) -> CheckpointBuilderResult<
2224        Vec<Vec<(TransactionEnvelope, TransactionEffects, Vec<UserSignature>)>>,
2225    > {
2226        let _guard = monitored_scope("CheckpointBuilder::split_checkpoint_chunks");
2227        let mut chunks = Vec::new();
2228        let mut chunk = Vec::new();
2229        let mut chunk_size: usize = 0;
2230        for ((transaction, effects, transaction_size), signatures) in
2231            transactions_effects_and_sizes.into_iter().zip(signatures)
2232        {
2233            // Roll over to a new chunk after either max count or max size is reached.
2234            // The size calculation here is intended to estimate the size of the
2235            // FullCheckpointContents struct. If this code is modified, that struct
2236            // should also be updated accordingly.
2237            let size = transaction_size
2238                + bcs::serialized_size(&effects)?
2239                + bcs::serialized_size(&signatures)?;
2240            if chunk.len() == self.max_transactions_per_checkpoint
2241                || (chunk_size + size) > self.max_checkpoint_size_bytes
2242            {
2243                if chunk.is_empty() {
2244                    // Always allow at least one tx in a checkpoint.
2245                    warn!(
2246                        "Size of single transaction ({size}) exceeds max checkpoint size ({}); allowing excessively large checkpoint to go through.",
2247                        self.max_checkpoint_size_bytes
2248                    );
2249                } else {
2250                    chunks.push(chunk);
2251                    chunk = Vec::new();
2252                    chunk_size = 0;
2253                }
2254            }
2255
2256            chunk.push((transaction, effects, signatures));
2257            chunk_size += size;
2258        }
2259
2260        if !chunk.is_empty() || chunks.is_empty() {
2261            // We intentionally create an empty checkpoint if there is no content provided
2262            // to make a 'heartbeat' checkpoint.
2263            // Important: if some conditions are added here later, we need to make sure we
2264            // always have at least one chunk if last_pending_of_epoch is set
2265            chunks.push(chunk);
2266            // Note: empty checkpoints are ok - they shouldn't happen at all on
2267            // a network with even modest load. Even if they do
2268            // happen, it is still useful as it allows fullnodes to
2269            // distinguish between "no transactions have happened" and "i am not
2270            // receiving new checkpoints".
2271        }
2272        Ok(chunks)
2273    }

2666    /// For the given roots return complete list of effects to include in
2667    /// checkpoint This list includes the roots and all their dependencies,
2668    /// which are not part of checkpoint already. Note that this function
2669    /// may be called multiple times to construct the checkpoint.
2670    /// `existing_tx_digests_in_checkpoint` is used to track the transactions
2671    /// that are already included in the checkpoint. Txs in `roots` that
2672    /// need to be included in the checkpoint will be added to
2673    /// `existing_tx_digests_in_checkpoint` after the call of this function.
2674    #[instrument(level = "debug", skip_all)]
2675    fn complete_checkpoint_effects(
2676        &self,
2677        mut roots: Vec<TransactionEffects>,
2678        existing_tx_digests_in_checkpoint: &mut BTreeSet<TransactionDigest>,
2679    ) -> IotaResult<Vec<TransactionEffects>> {
2680        let _scope = monitored_scope("CheckpointBuilder::complete_checkpoint_effects");
2681        let mut results = vec![];
2682        let mut seen = HashSet::new();
2683        loop {
2684            let mut pending = HashSet::new();
2685
2686            let transactions_included = self
2687                .epoch_store
2688                .builder_included_transactions_in_checkpoint(
2689                    roots.iter().map(|e| e.transaction_digest()),
2690                )?;
2691
2692            for (effect, tx_included) in roots.into_iter().zip(transactions_included.into_iter()) {
2693                let digest = effect.transaction_digest();
2694                // Unnecessary to read effects of a dependency if the effect is already
2695                // processed.
2696                seen.insert(*digest);
2697
2698                // Skip roots that are already included in the checkpoint.
2699                if existing_tx_digests_in_checkpoint.contains(effect.transaction_digest()) {
2700                    continue;
2701                }
2702
2703                // Skip roots already included in checkpoints or roots from previous epochs
2704                if tx_included || effect.epoch() < self.epoch_store.epoch() {
2705                    continue;
2706                }
2707
2708                let existing_effects = self
2709                    .epoch_store
2710                    .transactions_executed_in_cur_epoch(effect.dependencies())?;
2711
2712                for (dependency, effects_signature_exists) in
2713                    effect.dependencies().iter().zip(existing_effects.iter())
2714                {
2715                    // Skip here if dependency not executed in the current epoch.
2716                    // Note that the existence of an effects signature in the
2717                    // epoch store for the given digest indicates that the transaction
2718                    // was locally executed in the current epoch
2719                    if !effects_signature_exists {
2720                        continue;
2721                    }
2722                    if seen.insert(*dependency) {
2723                        pending.insert(*dependency);
2724                    }
2725                }
2726                results.push(effect);
2727            }
2728            if pending.is_empty() {
2729                break;
2730            }
2731            let pending = pending.into_iter().collect::<Vec<_>>();
2732            let effects = self
2733                .effects_store
2734                .try_multi_get_executed_effects(&pending)?;
2735            let effects = effects
2736                .into_iter()
2737                .zip(pending)
2738                .map(|(opt, digest)| match opt {
2739                    Some(x) => x,
2740                    None => panic!(
2741                        "Can not find effect for transaction {digest}, however transaction that depend on it was already executed"
2742                    ),
2743                })
2744                .collect::<Vec<_>>();
2745            roots = effects;
2746        }
2747
2748        existing_tx_digests_in_checkpoint.extend(results.iter().map(|e| e.transaction_digest()));
2749        Ok(results)
2750    }

2846    /// errors and retries gracefully, and allowing for proper shutdown on

2873    fn run_inner(&mut self) -> IotaResult<Vec<CertifiedCheckpointSummary>> {
2874        let _scope = monitored_scope("CheckpointAggregator");
2875        let mut result = vec![];
2876        'outer: loop {
2877            let next_to_certify = self.next_checkpoint_to_certify()?;
2878            let current = if let Some(current) = &mut self.current {
2879                // It's possible that the checkpoint was already certified by
2880                // the rest of the network and we've already received the
2881                // certified checkpoint via StateSync. In this case, we reset
2882                // the current signature aggregator to the next checkpoint to
2883                // be certified
2884                if current.summary.sequence_number < next_to_certify {
2885                    self.current = None;
2886                    continue;
2887                }
2888                current
2889            } else {
2890                let Some(summary) = self
2891                    .epoch_store
2892                    .get_built_checkpoint_summary(next_to_certify)?
2893                else {
2894                    return Ok(result);
2895                };
2896                self.current = Some(CheckpointSignatureAggregator {
2897                    next_index: 0,
2898                    digest: summary.digest(),
2899                    summary,
2900                    signatures_by_digest: MultiStakeAggregator::new(
2901                        self.epoch_store.committee().clone(),
2902                    ),
2903                    store: self.store.clone(),
2904                    state: self.state.clone(),
2905                    metrics: self.metrics.clone(),
2906                });
2907                self.current.as_mut().unwrap()
2908            };
2909
2910            let epoch_tables = self
2911                .epoch_store
2912                .tables()
2913                .expect("should not run past end of epoch");
2914            let iter = epoch_tables
2915                .pending_checkpoint_signatures
2916                .safe_iter_with_bounds(
2917                    Some((current.summary.sequence_number, current.next_index)),
2918                    None,
2919                );
2920            for item in iter {
2921                let ((seq, index), data) = item?;
2922                if seq != current.summary.sequence_number {

2924                        checkpoint_seq =? current.summary.sequence_number,
2925                        "Not enough checkpoint signatures",
2926                    );
2927                    // No more signatures (yet) for this checkpoint
2928                    return Ok(result);
2929                }
2930                trace!(
2931                    checkpoint_seq = current.summary.sequence_number,
2932                    "Processing signature for checkpoint (digest: {:?}) from {:?}",

2937                    .checkpoint_participation

2939                        "{:?}",

2945                        checkpoint_seq = current.summary.sequence_number,
2946                        "Successfully aggregated signatures for checkpoint (digest: {:?})",
2947                        current.summary.digest(),
2948                    );

2951                            current.summary.clone(),
2952                            auth_signature,
2953                        ),
2954                    );
2955
2956                    self.store.insert_certified_checkpoint(&summary)?;
2957                    if summary.next_epoch_committee().is_some() {
2958                        fail_point!("certified-end-of-epoch-checkpoint-before-state-sync");
2959                    }
2960                    self.metrics
2961                        .last_certified_checkpoint
2962                        .set(current.summary.sequence_number as i64);
2963                    current
2964                        .summary
2965                        .report_checkpoint_age(&self.metrics.last_certified_checkpoint_age);
2966                    result.push(summary.into_inner());
2967                    self.current = None;
2968                    continue 'outer;
2969                } else {
2970                    current.next_index = index + 1;
2971                }
2972            }
2973            break;
2974        }
2975        Ok(result)
2976    }

3420    /// can think of this as a WAL replay operation. Upon startup, we may
3421    /// have a number of consensus commits and resulting checkpoints that
3422    /// were built but not committed to disk. We want to reprocess the
3423    /// commits and rebuild the checkpoints before starting normal operation.
3424    pub async fn spawn(&self, consensus_replay_waiter: Option<ReplayWaiter>) -> JoinSet<()> {
3425        let mut tasks = JoinSet::new();
3426
3427        let (builder, aggregator, state_hasher) = self.state.lock().take_unstarted();
3428        let (builder_finished_tx, builder_finished_rx) = tokio::sync::oneshot::channel();
3429        tasks.spawn(monitored_future!(async move {
3430            builder.run(consensus_replay_waiter).await;
3431            builder_finished_tx.send(()).ok();
3432        }));
3433        tasks.spawn(monitored_future!(aggregator.run()));
3434        tasks.spawn(monitored_future!(state_hasher.run()));

3436        // If this times out, the validator may still start up. The worst that can
3437        // happen is that we will crash later on instead of immediately. The eventual
3438        // crash would occur because we may be missing transactions that are below the
3439        // highest_synced_checkpoint watermark, which can cause a crash in
3440        // `CheckpointExecutor::extract_randomness_rounds`.
3441        if tokio::time::timeout(Duration::from_secs(120), async move {
3442            tokio::select! {
3443                _ = builder_finished_rx => { debug!("CheckpointBuilder finished"); }
3444                _ = self.wait_for_rebuilt_checkpoints() => (),
3445            }
3446        })
3447        .await
3448        .is_err()
3449        {
3450            debug_fatal!("Timed out waiting for checkpoints to be rebuilt");
3451        }
3452
3453        tasks
3454    }

3458    /// Waits until all checkpoints had been built before the node restarted
3459    /// are rebuilt. This is required to preserve the invariant that all
3460    /// checkpoints (and their transactions) below the
3461    /// highest_synced_checkpoint watermark are available. Once the
3462    /// checkpoints are constructed, we can be sure that the transactions
3463    /// have also been executed.
3464    pub async fn wait_for_rebuilt_checkpoints(&self) {
3465        let highest_previously_built_seq = self.highest_previously_built_seq;
3466        let mut rx = self.highest_currently_built_seq_tx.subscribe();
3467        let mut highest_currently_built_seq = *rx.borrow_and_update();
3468        info!(
3469            "Waiting for checkpoints to be rebuilt, previously built seq: \
3470            {highest_previously_built_seq}, currently built seq: {highest_currently_built_seq}"
3471        );
3472        loop {
3473            if highest_currently_built_seq >= highest_previously_built_seq {
3474                info!("Checkpoint rebuild complete");
3475                break;
3476            }
3477            rx.changed().await.unwrap();
3478            highest_currently_built_seq = *rx.borrow_and_update();
3479        }
3480    }
