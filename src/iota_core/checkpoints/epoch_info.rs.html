5//! Per-epoch verified metadata (`epoch_info`) held by every node, independent
6//! of any API.
7//!
8//! Each row is an [`EpochInfoV2`]: the start-of-epoch identity plus the
9//! close-of-epoch proof bundle ([`EpochInfoV1Entry`]). A row is seeded when its
10//! epoch opens and finalized when the epoch closes. The table is append-only
11//! and never pruned; the `epoch_info_watermark` tracks the highest
12//! epoch whose contiguous prefix is finalized.
13

30    /// Read the `epoch_info` row for `epoch`, if present.
31    pub fn get_epoch_info(&self, epoch: EpochId) -> Result<Option<EpochInfoV2>, TypedStoreError> {
32        self.tables.epoch_info.get(&epoch)
33    }

35    /// Read `epoch_info_watermark`: the highest epoch whose
36    /// `epoch_info` row is finalized (the full close-of-epoch proof bundle
37    /// present, see [`EpochInfoV2::is_finalized`]). `None` if no epoch has
38    /// been fully indexed yet.
39    pub fn highest_indexed_epoch(&self) -> Result<Option<EpochId>, TypedStoreError> {
40        self.tables.epoch_info_watermark.get(&())
41    }

43    /// Persist fully-populated epoch rows (e.g. restored from a snapshot) and
44    /// advance the `epoch_info_watermark` over the now-contiguous
45    /// prefix.
46    ///
47    /// Must not run concurrently with live boundary indexing: it recomputes the
48    /// watermark outside the per-boundary batch.
49    pub fn insert_epoch_info(&self, rows: Vec<EpochInfoV2>) -> Result<(), StorageError> {
50        let mut batch = self.tables.epoch_info.batch();
51        batch.insert_batch(
52            &self.tables.epoch_info,
53            rows.into_iter().map(|row| (row.epoch, row)),
54        )?;
55        batch.write()?;
56        self.reconcile_epoch_info_watermark()?;
57        Ok(())
58    }

60    /// Finalize the closing epoch with its proof bundle and seed the next
61    /// epoch's row. A no-op on non-boundary checkpoints.
62    ///
63    /// Must be called in epoch order; relies on boundaries arriving
64    /// contiguously so the watermark advances by exactly one each time.
65    pub(crate) fn index_epoch_boundary(
66        &self,
67        checkpoint: &CheckpointData,
68    ) -> Result<(), StorageError> {
69        let mut batch = self.tables.epoch_info.batch();
70        self.index_epoch(checkpoint, &mut batch)?;
71        batch.write()?;
72        Ok(())
73    }

75    /// `None` when `epoch_info` is contiguously populated from genesis through
76    /// at least the last closed epoch this node executed; otherwise
77    /// `Some((highest_indexed, last_executed_epoch))` describing the gap a
78    /// startup guard must reject. Measured against the last executed closed
79    /// epoch, not a target epoch, so a node still catching up isn't flagged.
80    pub fn epoch_info_gap(&self) -> Result<Option<(Option<EpochId>, EpochId)>, StorageError> {
81        let Some(open_epoch) = self.first_open_epoch()? else {
82            return Ok(None); // nothing executed yet
83        };
84        let Some(last_executed) = open_epoch.checked_sub(1) else {
85            return Ok(None); // still in the genesis epoch; no closed epoch
86        };
87        let highest_indexed = self.highest_indexed_epoch()?;
88        // `<`, not `!=`: rows seeded past local execution are a superset.
89        Ok((highest_indexed < Some(last_executed)).then_some((highest_indexed, last_executed)))
90    }

92    /// Seeds the open epoch's `epoch_info` row and guards the chain's
93    /// completeness; called once at node startup, before live execution
94    /// resumes. Every node maintains the chain live at its executed epoch
95    /// boundaries, and new nodes obtain the history through genesis sync or a
96    /// formal-snapshot restore, so a gap can only mean this node's store
97    /// predates the `epoch_info` table. A gap is a fatal error on
98    /// mainnet/testnet — every node must hold the verified chain since
99    /// genesis — and only a warning on other networks, where the missing
100    /// epochs are left unfilled: such a node cannot produce snapshots or
101    /// serve the epoch gRPC API for those epochs.
102    pub fn seed_epoch_info(
103        &self,
104        authority_store: &AuthorityStore,
105        chain_identifier: ChainIdentifier,
106    ) -> Result<(), StorageError> {
107        // Seed the open epoch's row (no-op if already present). Without it the
108        // next executed boundary can't finalize it and the watermark wedges.
109        self.ensure_current_epoch_info(authority_store)?;
110
111        if let Some((highest_indexed, last_executed)) = self.epoch_info_gap()? {
112            let detail = format!(
113                "the epoch_info chain is incomplete (finalized through {highest_indexed:?}, \
114                 executed through epoch {last_executed})"
115            );
116            match chain_identifier.chain() {
117                Chain::Mainnet | Chain::Testnet => {
118                    return Err(StorageError::custom(format!(
119                        "{detail}; restore this node from a formal snapshot or resync it from \
120                         genesis to obtain the verified chain"
121                    )));
122                }
123                Chain::Unknown => warn!(
124                    "{detail}; the missing epochs are left unfilled (live indexing only \
125                     extends the chain forward), so this node cannot produce snapshots or \
126                     serve the epoch gRPC API for those epochs"
127                ),
128            }
129        }
130        Ok(())
131    }

133    /// Seed the current (open) epoch's `epoch_info` row if still missing. No-op
134    /// if already seeded or its start checkpoint can't yet be derived locally;
135    /// call again once the closed-epoch rows are seeded.
136    pub fn ensure_current_epoch_info(
137        &self,
138        authority_store: &AuthorityStore,
139    ) -> Result<(), StorageError> {
140        // Seed the *open* epoch, not the highest executed checkpoint's epoch.
141        // The two differ when that checkpoint closed its epoch — the state a
142        // snapshot restore always lands in. The closed epoch's row comes from
143        // EPOCH_INFO; the next epoch's row has no other writer until its own
144        // close, where `index_epoch` would find it missing, skip the finalize,
145        // and wedge the watermark below it for good. With nothing executed yet
146        // the open epoch is genesis epoch 0.
147        let current_epoch = self.first_open_epoch()?.unwrap_or(0);
148
149        if self.tables.epoch_info.get(&current_epoch)?.is_some() {
150            return Ok(());
151        }
152
153        let Some(start_checkpoint) = self.current_epoch_start_checkpoint(current_epoch)? else {
154            // Skip rather than fail startup; re-seeded once the previous
155            // epoch's row lands (e.g. from a formal-snapshot restore).
156            warn!(
157                epoch = current_epoch,
158                "skipping current-epoch seed: previous epoch's last checkpoint is \
159                 unknown locally"
160            );
161            return Ok(());
162        };
163
164        let system_state = iota_types::iota_system_state::get_iota_system_state(authority_store)
165            .map_err(|e| StorageError::custom(format!("Failed to find system state: {e}")))?;
166
167        let epoch_info = EpochInfoV2 {
168            epoch: current_epoch,
169            start_checkpoint,
170            start_timestamp_ms: system_state.epoch_start_timestamp_ms(),
171            system_state,
172            epoch_close_proof: None,
173        };
174
175        self.tables
176            .epoch_info
177            .insert(&epoch_info.epoch, &epoch_info)?;
178
179        Ok(())
180    }

182    /// The current epoch's start checkpoint, or `None` when it can't be derived
183    /// from local data (no previous-epoch `epoch_info` row and no
184    /// `epoch_last_checkpoint_map` entry). `None` means "skip for now", not an
185    /// error.
186    fn current_epoch_start_checkpoint(
187        &self,
188        current_epoch: EpochId,
189    ) -> Result<Option<CheckpointSequenceNumber>, StorageError> {
190        // The first epoch starts at checkpoint 0.
191        if current_epoch == 0 {
192            return Ok(Some(0));
193        }
194        let previous_epoch = current_epoch - 1;
195
196        // Prefer the previous epoch's recorded end checkpoint.
197        if let Some(end_checkpoint) = self
198            .tables
199            .epoch_info
200            .get(&previous_epoch)?
201            .and_then(|info| info.end_checkpoint())
202        {
203            return Ok(Some(end_checkpoint + 1));
204        }
205
206        // Unlike the checkpoint summaries, the map is never pruned.
207        Ok(self
208            .get_epoch_last_checkpoint_seq_number(previous_epoch)?
209            .map(|seq| seq + 1))
210    }

212    /// Finalize the closing epoch's row with its close-of-epoch proof bundle
213    /// and seed the next epoch's row. A no-op on non-boundary checkpoints.
214    /// Stages all writes into `batch` so the caller commits them atomically.
215    fn index_epoch(
216        &self,
217        checkpoint: &CheckpointData,
218        batch: &mut DBBatch,
219    ) -> Result<(), StorageError> {
220        let Some((epoch_info, end_of_epoch_events)) = checkpoint.epoch_info()? else {
221            return Ok(());
222        };
223        let new_epoch_id = epoch_info.epoch;
224
225        // Finalize `prev_epoch`'s row with the close-of-epoch proof bundle and
226        // advance the watermark. Genesis has no previous epoch to finalize; the
227        // close-of-epoch write for epoch 0 fires when epoch 1 is seeded, so the
228        // watermark stays absent until then.
229        if new_epoch_id > 0 {
230            let prev_epoch = new_epoch_id - 1;
231            // If no row exists for `prev_epoch`, this node didn't see its
232            // start (e.g. bootstrapped mid-epoch). Skip the upsert; the row
233            // stays absent and the watermark stays behind, so the snapshot
234            // writer correctly refuses to publish the incomplete chain.
235            if let Some(mut previous_epoch) = self.tables.epoch_info.get(&prev_epoch)? {
236                // The closing checkpoint's last tx is `prev_epoch`'s
237                // epoch-change tx; its effects and the system-state objects it
238                // wrote (object `0x5` and its inner state object) are this
239                // boundary's proof material.
240                let change_epoch_tx = checkpoint.end_of_epoch_transaction().ok_or_else(|| {
241                    StorageError::custom(format!(
242                        "checkpoint {} closes epoch {prev_epoch} but carries no \
243                         epoch-change transaction",
244                        checkpoint.checkpoint_summary.sequence_number,
245                    ))
246                })?;
247                // Serialized to raw bytes, not the decoded view, so they verify
248                // against the effects' written-object digests at restore time
249                // (see `get_iota_system_state_objects` for why only these two).
250                let next_epoch_start_system_state_objects =
251                    iota_types::iota_system_state::get_iota_system_state_objects(
252                        &change_epoch_tx.output_objects.as_slice(),
253                    )
254                    .map_err(|e| {
255                        StorageError::custom(format!(
256                            "extracting next-epoch start-state objects: {e}"
257                        ))
258                    })?
259                    .iter()
260                    .map(bcs::to_bytes)
261                    .collect::<Result<Vec<_>, _>>()
262                    .map_err(|e| {
263                        StorageError::custom(format!("serializing start-state objects: {e}"))
264                    })?;
265
266                previous_epoch.epoch_close_proof = Some(EpochInfoV1Entry {
267                    last_checkpoint_summary: checkpoint.checkpoint_summary.clone(),
268                    last_checkpoint_contents: checkpoint.checkpoint_contents.clone(),
269                    end_of_epoch_tx_effects: change_epoch_tx.effects.clone(),
270                    // Safe-mode boundaries run `advance_epoch_safe_mode`, which
271                    // mutates `0x5` but emits no events, so the list is empty.
272                    end_of_epoch_tx_events: end_of_epoch_events.unwrap_or_default(),
273                    next_epoch_start_system_state_objects,
274                });
275                batch.insert_batch(&self.tables.epoch_info, [(prev_epoch, previous_epoch)])?;
276                self.try_advance_epoch_info_watermark(prev_epoch, batch)?;
277            }
278        }
279
280        // seed the new epoch's row; its close-of-epoch proof is filled in when
281        // the next epoch's boundary is indexed.
282        let new_info = EpochInfoV2 {
283            epoch: epoch_info.epoch,
284            start_checkpoint: epoch_info.start_checkpoint,
285            start_timestamp_ms: epoch_info.start_timestamp_ms,
286            system_state: epoch_info.system_state,
287            epoch_close_proof: None,
288        };
289        batch.insert_batch(&self.tables.epoch_info, [(new_epoch_id, new_info)])?;
290
291        Ok(())
292    }

294    /// Advance the completeness watermark by exactly one, only if `prev_epoch`
295    /// is the next contiguous epoch. Guards against advancing across a gap
296    /// left by a node that bootstrapped mid-history.
297    fn try_advance_epoch_info_watermark(
298        &self,
299        prev_epoch: EpochId,
300        batch: &mut DBBatch,
301    ) -> Result<(), StorageError> {
302        let next_expected = self
303            .tables
304            .epoch_info_watermark
305            .get(&())?
306            .map_or(0, |e| e.saturating_add(1));
307        if prev_epoch == next_expected {
308            batch.insert_batch(&self.tables.epoch_info_watermark, [((), prev_epoch)])?;
309        }
310        Ok(())
311    }

313    /// Recompute `epoch_info_watermark` = the highest epoch whose
314    /// contiguous prefix `[0, epoch]` is finalized (see
315    /// [`EpochInfoV2::is_finalized`]). Only ever raises it.
316    ///
317    /// Unlike [`Self::try_advance_epoch_info_watermark`] (the live +1 step),
318    /// this can jump the watermark across a whole seeded prefix, which is what
319    /// a snapshot restore needs.
320    fn reconcile_epoch_info_watermark(&self) -> Result<(), TypedStoreError> {
321        // `[0, watermark]` is already known complete, so resume the scan from
322        // `watermark + 1` rather than re-scanning the whole table.
323        let current = self.tables.epoch_info_watermark.get(&())?;
324        let mut next = current.map_or(0, |w| w + 1);
325        for entry in self
326            .tables
327            .epoch_info
328            .safe_iter_with_bounds(Some(next), None)
329        {
330            let (epoch_id, info) = entry?;
331            if epoch_id != next || !info.is_finalized() {
332                break;
333            }
334            next += 1;
335        }
336        if let Some(highest) = next.checked_sub(1) {
337            if Some(highest) > current {
338                self.tables.epoch_info_watermark.insert(&(), &highest)?;
339            }
340        }
341        Ok(())
342    }

344    /// The first not-yet-closed epoch: the highest executed checkpoint's epoch,
345    /// plus one if that checkpoint already closed its epoch.
346    fn first_open_epoch(&self) -> Result<Option<EpochId>, StorageError> {
347        Ok(self
348            .get_highest_executed_checkpoint()?
349            .map(|highest| open_epoch_of(highest.data())))
350    }
351}

353/// The open epoch as of `checkpoint`: its own epoch, plus one if it closed it.
354fn open_epoch_of(checkpoint: &CheckpointSummary) -> EpochId {
355    if checkpoint.is_last_checkpoint_of_epoch() {
356        checkpoint.epoch + 1
357    } else {
358        checkpoint.epoch
359    }

577    /// seeded past local execution both count as complete.
578    #[tokio::test]
579    async fn epoch_info_gap_flags_short_prefix_not_overshoot() {
580        let store = CheckpointStore::new_for_tests();
581
582        // Nothing executed yet -> nothing to guard.
583        assert_eq!(store.epoch_info_gap().unwrap(), None);
584
585        // Genesis epoch still open -> no closed epoch -> no gap.
586        let genesis = executed_checkpoint(0, 0);
587        store.insert_verified_checkpoint(&genesis).unwrap();
588        store.update_highest_executed_checkpoint(&genesis).unwrap();
589        assert_eq!(store.epoch_info_gap().unwrap(), None);
590
591        // Executed into epoch 2 (synthetic jump) -> closed epochs [0, 1]; an
592        // empty index and a prefix short of epoch 1 are both gaps.
593        let in_epoch_2 = executed_checkpoint(2, 1);
594        store.insert_verified_checkpoint(&in_epoch_2).unwrap();
595        store
596            .update_highest_executed_checkpoint(&in_epoch_2)
597            .unwrap();
598        assert_eq!(store.epoch_info_gap().unwrap(), Some((None, 1)));
599        store
600            .insert_epoch_info(vec![complete_epoch_info(0)])
601            .unwrap();
602        assert_eq!(store.epoch_info_gap().unwrap(), Some((Some(0), 1)));
603
604        // Complete through the last closed epoch -> no gap.
605        store
606            .insert_epoch_info(vec![complete_epoch_info(1)])
607            .unwrap();
608        assert_eq!(store.epoch_info_gap().unwrap(), None);
609
610        // Rows seeded beyond local execution are a superset, not a gap.
611        store
612            .insert_epoch_info(vec![complete_epoch_info(2), complete_epoch_info(3)])
613            .unwrap();
614        assert_eq!(store.epoch_info_gap().unwrap(), None);
615    }

617    /// `try_advance_epoch_info_watermark` advances the watermark only when
618    /// `prev_epoch` extends the contiguous prefix by exactly one.
619    #[tokio::test]
620    async fn try_advance_epoch_info_watermark_is_gap_aware() {
621        let store = CheckpointStore::new_for_tests();
622
623        let advance = |epoch| {
624            let mut batch = store.tables.epoch_info_watermark.batch();
625            store
626                .try_advance_epoch_info_watermark(epoch, &mut batch)
627                .unwrap();
628            batch.write().unwrap();
629            store.highest_indexed_epoch().unwrap()
630        };
631
632        // From absent: prev_epoch=5 must NOT advance (bootstrap mid-history).
633        assert_eq!(advance(5), None);
634        // From absent: prev_epoch=0 advances to 0 (genesis close).
635        assert_eq!(advance(0), Some(0));
636        // From 0: prev_epoch=2 must NOT advance (gap at 1).
637        assert_eq!(advance(2), Some(0));
638        // From 0: prev_epoch=1 advances to 1.
639        assert_eq!(advance(1), Some(1));
640        // From 1: prev_epoch=1 again is a no-op (already covered).
641        assert_eq!(advance(1), Some(1));
642        // From 1: prev_epoch=2 advances to 2.
643        assert_eq!(advance(2), Some(2));
644    }
