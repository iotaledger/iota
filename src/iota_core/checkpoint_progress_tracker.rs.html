24/// The `tracing` target of the progress lines to shorten the log lines.
25const LOG_TARGET: &str = "iota_core::sync";

27/// A node whose newest executed checkpoint is within this range is
28/// reported without estimates.
29const CAUGHT_UP_WITHIN: Duration = Duration::from_mins(5);
30
31/// Formats a duration for progress lines, naming its two largest units.
32fn format_duration(duration: Duration) -> String {
33    let secs = duration.as_secs();
34    let (days, hours, minutes, seconds) = (
35        secs / 86400,
36        (secs % 86400) / 3600,
37        (secs % 3600) / 60,
38        secs % 60,
39    );
40    if days > 0 {
41        format!("{days}d {hours}h")
42    } else if hours > 0 {
43        format!("{hours}h {minutes}m")
44    } else if minutes > 0 {
45        format!("{minutes}m {seconds}s")
46    } else {
47        format!("{seconds}s")
48    }
49}

51/// Formats a count for progress lines, e.g. "123.4M" or "1.2k".
52fn format_count(count: u64) -> String {
53    if count >= 1_000_000_000 {
54        format!("{:.1}B", count as f64 / 1e9)
55    } else if count >= 1_000_000 {
56        format!("{:.1}M", count as f64 / 1e6)
57    } else if count >= 1_000 {
58        format!("{:.1}k", count as f64 / 1e3)
59    } else {
60        count.to_string()
61    }
62}

64/// Tracks how fast the node executes chain history, in seconds of checkpoint
65/// timestamps per wall-clock second.
66///
67/// For example, rate = 376.0 means execution of 376 seconds of history every
68/// real second. A node keeping pace with the chain sits at 1.0.
69#[derive(Default)]
70struct HistoryExecutionRate {
71    last_wall: Option<Instant>,
72    last_executed_timestamp_ms: CheckpointTimestamp,
73    rate: f64,
74}
75
76/// How many ticks the rate takes to follow a change in speed, so that the
77/// estimates hold steady while the per-second throughput swings.
78const SMOOTHING_TICKS: f64 = 10.0;
79
80impl HistoryExecutionRate {
81    /// Records the timestamp of the newest executed checkpoint at `now` and
82    /// returns the updated rate. Returns 0.0 until two ticks with advancing
83    /// chain time have been seen; a tick whose chain time did not advance
84    /// leaves the estimate unchanged. The estimate follows changes in speed
85    /// gradually over several ticks.
86    fn update(&mut self, now: Instant, executed_timestamp_ms: CheckpointTimestamp) -> f64 {
87        let Some(last_wall) = self.last_wall else {
88            self.last_wall = Some(now);
89            self.last_executed_timestamp_ms = executed_timestamp_ms;
90            return self.rate;
91        };
92        let wall_elapsed_secs = now.duration_since(last_wall).as_secs_f64();
93        let chain_elapsed_secs =
94            executed_timestamp_ms.saturating_sub(self.last_executed_timestamp_ms) as f64 / 1000.0;
95        // Leaving the mark where it is, so that the next tick measures across
96        // the whole stall rather than reading the rate as though it had not
97        // happened.
98        if wall_elapsed_secs <= 0.0 || chain_elapsed_secs <= 0.0 {
99            return self.rate;
100        }
101
102        let sample = chain_elapsed_secs / wall_elapsed_secs;
103        // Seeded rather than averaged up from zero, so the first line after a
104        // restart reports a usable figure.
105        self.rate = if self.rate == 0.0 {
106            sample
107        } else {
108            self.rate * (1.0 - 1.0 / SMOOTHING_TICKS) + sample / SMOOTHING_TICKS
109        };
110        self.last_wall = Some(now);
111        self.last_executed_timestamp_ms = executed_timestamp_ms;
112        self.rate
113    }
114}

116/// Wall-clock time to close `behind` while moving through chain time at
117/// `rate`, or `None` when the node is not gaining, or is gaining so slowly
118/// that the wait is longer than a [`Duration`] can hold.
119///
120/// Hint: The chain keeps producing a second of history per second, so the gap
121/// closes at whatever `rate` exceeds chain pace by — not at `rate` itself.
122fn time_to_catch_up(behind: Duration, rate: f64) -> Option<Duration> {
123    if rate <= 1.0 {
124        return None;
125    }
126    Duration::try_from_secs_f64(behind.as_secs_f64() / (rate - 1.0)).ok()
127}
128
129/// Wall-clock time to reach the end of the current epoch, whose remaining
130/// history is `remaining`, at `rate`. `None` when the rate is not yet known,
131/// or the wait is longer than a [`Duration`] can hold.
132fn time_to_epoch_end(remaining: Duration, rate: f64) -> Option<Duration> {
133    if rate <= 0.0 {
134        return None;
135    }
136    Duration::try_from_secs_f64(remaining.as_secs_f64() / rate).ok()
137}

139/// Returns how far behind the node is, how long until the current epoch ends,
140/// and how long until the node catches up, or `None` when the node is caught up
141/// or not gaining.
142fn sync_eta(
143    checkpoint_store: &CheckpointStore,
144    epoch: EpochId,
145    chain_ms: CheckpointTimestamp,
146    rate: f64,
147) -> Option<String> {
148    if rate <= 0.0 {
149        return None;
150    }
151    let now_ms = SystemTime::now()
152        .duration_since(UNIX_EPOCH)
153        .ok()?
154        .as_millis() as u64;
155    let behind = Duration::from_millis(now_ms.saturating_sub(chain_ms));
156    if behind < CAUGHT_UP_WITHIN {
157        return None;
158    }
159
160    let epoch_eta = checkpoint_store
161        .get_epoch_info(epoch)
162        .ok()
163        .flatten()
164        .and_then(|info| {
165            let ends_ms = info
166                .start_timestamp_ms
167                .saturating_add(info.system_state.epoch_duration_ms());
168            let remaining = Duration::from_millis(ends_ms.saturating_sub(chain_ms));
169            time_to_epoch_end(remaining, rate)
170        })
171        .map(format_duration)
172        .unwrap_or_else(|| "unknown".to_string());
173
174    let sync_eta = time_to_catch_up(behind, rate)
175        .map(format_duration)
176        .unwrap_or_else(|| "not gaining".to_string());
177
178    Some(format!(
179        ", {} behind, epoch ETA {epoch_eta}, sync ETA {sync_eta} ({rate:.0}x chain time)",
180        format_duration(behind),
181    ))
182}

184/// Shared progress tracker for checkpoint operations. Updated by the
185/// checkpoint executor and pruner, periodically logs a one-line summary.
186///
187/// Passed as `Option<Arc<CheckpointProgressTracker>>` — callers that don't need
188/// progress reporting (CLI tools, tests) simply pass `None`.
189///
190/// All values are accumulated and reset on each logging tick.
191pub struct CheckpointProgressTracker {
192    /// Accumulated checkpoint execution time in nanoseconds.
193    execution_time_ns: AtomicU64,
194    /// Accumulated time spent working in each checkpoint pipeline stage, in
195    /// nanoseconds, indexed by [`PipelineStage`]. Since every stage admits
196    /// one checkpoint at a time, a stage accumulating close to one second
197    /// per second is the throughput bottleneck.
198    stage_time_ns: [AtomicU64; PipelineStage::COUNT],
199    /// Accumulated object pruning time in nanoseconds.
200    object_pruning_time_ns: AtomicU64,
201    /// Accumulated checkpoint/effects pruning time in nanoseconds.
202    checkpoint_pruning_time_ns: AtomicU64,
203}
204
205impl CheckpointProgressTracker {
206    pub fn new() -> Self {
207        Self {
208            execution_time_ns: AtomicU64::new(0),
209            stage_time_ns: std::array::from_fn(|_| AtomicU64::new(0)),
210            object_pruning_time_ns: AtomicU64::new(0),
211            checkpoint_pruning_time_ns: AtomicU64::new(0),
212        }
213    }
214
215    pub fn add_execution_time(&self, duration: Duration) {
216        self.execution_time_ns
217            .fetch_add(duration.as_nanos() as u64, Ordering::Relaxed);
218    }
219
220    pub(crate) fn add_stage_time(&self, stage: PipelineStage, duration: Duration) {
221        self.stage_time_ns[stage as usize].fetch_add(duration.as_nanos() as u64, Ordering::Relaxed);
222    }
223
224    pub fn add_object_pruning_time(&self, duration: Duration) {
225        self.object_pruning_time_ns
226            .fetch_add(duration.as_nanos() as u64, Ordering::Relaxed);
227    }
228
229    pub fn add_checkpoint_pruning_time(&self, duration: Duration) {
230        self.checkpoint_pruning_time_ns
231            .fetch_add(duration.as_nanos() as u64, Ordering::Relaxed);
232    }
233
234    /// Spawns a periodic logging task that prints a one-line checkpoint
235    /// progress summary every second (only when there is actual progress).
236    pub fn spawn_logging_task(
237        self: &Arc<Self>,
238        checkpoint_store: Arc<CheckpointStore>,
239        perpetual_db: Arc<AuthorityPerpetualTables>,
240    ) {
241        let tracker = self.clone();
242        tokio::task::spawn(async move {
243            let mut interval = tokio::time::interval(Duration::from_secs(1));
244            let mut prev_executed: u64 = 0;
245            let mut prev_synced: u64 = 0;
246            let mut prev_total_tx: u64 = 0;
247            let mut prev_obj_pruned: u64 = 0;
248            let mut prev_ckpt_pruned: u64 = 0;
249            let mut history_rate = HistoryExecutionRate::default();
250
251            loop {
252                interval.tick().await;
253
254                let highest_executed_checkpoint = checkpoint_store
255                    .get_highest_executed_checkpoint()
256                    .ok()
257                    .flatten();
258                let epoch = highest_executed_checkpoint
259                    .as_ref()
260                    .map(|c| c.epoch())
261                    .unwrap_or(0);
262                let highest_executed_seq_number = highest_executed_checkpoint
263                    .as_ref()
264                    .map(|c| c.sequence_number())
265                    .unwrap_or(0);
266                let total_tx = highest_executed_checkpoint
267                    .as_ref()
268                    .map(|c| c.network_total_transactions)
269                    .unwrap_or(0);
270                let chain_ms = highest_executed_checkpoint
271                    .as_ref()
272                    .map(|c| c.timestamp_ms)
273                    .unwrap_or(0);
274                let synced_seq_number = checkpoint_store
275                    .get_highest_synced_checkpoint_seq_number()
276                    .ok()
277                    .flatten()
278                    .unwrap_or(0);
279                let object_pruned_seq_number = perpetual_db
280                    .get_highest_pruned_checkpoint()
281                    .ok()
282                    .flatten()
283                    .unwrap_or(0);
284                let checkpoint_pruned_seq_number = checkpoint_store
285                    .get_highest_pruned_checkpoint_seq_number()
286                    .ok()
287                    .flatten()
288                    .unwrap_or(0);
289
290                let exec_delta = highest_executed_seq_number.saturating_sub(prev_executed);
291                let synced_delta = synced_seq_number.saturating_sub(prev_synced);
292                let tx_delta = total_tx.saturating_sub(prev_total_tx);
293                let object_prune_delta = object_pruned_seq_number.saturating_sub(prev_obj_pruned);
294                let checkpoint_prune_delta =
295                    checkpoint_pruned_seq_number.saturating_sub(prev_ckpt_pruned);
296
297                if exec_delta > 0
298                    || synced_delta > 0
299                    || tx_delta > 0
300                    || object_prune_delta > 0
301                    || checkpoint_prune_delta > 0
302                {
303                    let exec_time_delta_ns = tracker.execution_time_ns.swap(0, Ordering::Relaxed);
304                    let exec_time_delta = Duration::from_nanos(exec_time_delta_ns);
305
306                    let object_prune_time_delta_ns =
307                        tracker.object_pruning_time_ns.swap(0, Ordering::Relaxed);
308                    let object_prune_time_delta = Duration::from_nanos(object_prune_time_delta_ns);
309
310                    let checkpoint_prune_time_delta_ns = tracker
311                        .checkpoint_pruning_time_ns
312                        .swap(0, Ordering::Relaxed);
313                    let checkpoint_prune_time_delta =
314                        Duration::from_nanos(checkpoint_prune_time_delta_ns);
315
316                    // Only report pruning when there is some, to avoid cluttering the log
317                    let pruning = if object_prune_delta > 0 || checkpoint_prune_delta > 0 {
318                        format!(
319                            ", objs pruned {object_pruned_seq_number} \
320                             (+{object_prune_delta}, {object_prune_time_delta:.2?}), \
321                             ckpts pruned {checkpoint_pruned_seq_number} \
322                             (+{checkpoint_prune_delta}, {checkpoint_prune_time_delta:.2?})"
323                        )
324                    } else {
325                        String::new()
326                    };
327
328                    let rate = history_rate.update(Instant::now(), chain_ms);
329                    let eta =
330                        sync_eta(&checkpoint_store, epoch, chain_ms, rate).unwrap_or_default();
331
332                    info!(
333                        target: LOG_TARGET,
334                        "[epoch {epoch}]: executed {highest_executed_seq_number}/{synced_seq_number} \
335                         (+{exec_delta}/+{synced_delta}, {} tx/s, {exec_time_delta:.2?}){pruning}{eta}",
336                        format_count(tx_delta),
337                    );
338
339                    // Every pipeline stage admits one checkpoint at a time,
340                    // so the stage whose time approaches one second per tick
341                    // is the execution throughput bottleneck.
342                    let stage_times: Vec<String> = PipelineStage::iter()
343                        .filter_map(|stage| {
344                            let stage_time_ns =
345                                tracker.stage_time_ns[stage as usize].swap(0, Ordering::Relaxed);
346                            (stage_time_ns > 0).then(|| {
347                                format!(
348                                    "{} {:.2?}",
349                                    stage.as_str(),
350                                    Duration::from_nanos(stage_time_ns)
351                                )
352                            })
353                        })
354                        .collect();
355                    if !stage_times.is_empty() {
356                        debug!(
357                            target: LOG_TARGET,
358                            "checkpoint pipeline stage times [epoch {epoch}]: {}",
359                            stage_times.join(", ")
360                        );
361                    }
362
363                    prev_executed = highest_executed_seq_number;
364                    prev_synced = synced_seq_number;
365                    prev_total_tx = total_tx;
366                    prev_obj_pruned = object_pruned_seq_number;
367                    prev_ckpt_pruned = checkpoint_pruned_seq_number;
368                }
369            }
370        });
371    }
372}

384    /// The rate is chain time over wall time: a tick that covers an hour of
385    /// checkpoints in a second is moving at 3600x.
386    #[test]
387    fn the_rate_measures_chain_time_against_wall_time() {
388        let mut rate = HistoryExecutionRate::default();
389        let start = Instant::now();
390        assert_eq!(
391            rate.update(start, 0),
392            0.0,
393            "one sample cannot establish a rate"
394        );
395        let sampled = rate.update(start + Duration::from_secs(1), 3_600_000);
396        assert!(
397            (sampled - 3600.0).abs() < 1.0,
398            "an hour of chain time in a wall second is 3600x, got {sampled}"
399        );
400    }
401
402    /// A node whose checkpoints are not advancing has no rate, so no estimate
403    /// can be made from it.
404    #[test]
405    fn a_stalled_node_reports_no_rate() {
406        let mut rate = HistoryExecutionRate::default();
407        let start = Instant::now();
408        rate.update(start, 5_000);
409        assert_eq!(
410            rate.update(start + Duration::from_secs(1), 5_000),
411            0.0,
412            "chain time that did not move must not produce a rate"
413        );
414    }
415
416    /// The estimate is smoothed, so one slow tick moves it rather than
417    /// replacing it — otherwise the line would swing with every archive file.
418    #[test]
419    fn the_rate_is_smoothed_across_ticks() {
420        let mut rate = HistoryExecutionRate::default();
421        let start = Instant::now();
422        rate.update(start, 0);
423        let first = rate.update(start + Duration::from_secs(1), 100_000);
424        let second = rate.update(start + Duration::from_secs(2), 100_000 + 1_000);
425        assert!(
426            second < first && second > 10.0,
427            "a single slow tick must pull the estimate down without resetting it, \
428             got {first} then {second}"
429        );
430    }
431
432    /// The gap closes at the margin above chain pace, not at the rate: a node
433    /// at 2x chain time needs a day of wall clock to make up a day, because
434    /// the chain adds another day's worth while it works.
435    #[test]
436    fn catching_up_closes_the_gap_at_the_margin_above_chain_pace() {
437        let a_day = Duration::from_secs(86_400);
438        assert_eq!(time_to_catch_up(a_day, 2.0), Some(a_day));
439        assert_eq!(time_to_catch_up(a_day, 3.0), Some(a_day / 2));
440    }
441
442    /// A rate a hair above chain pace divides by nearly nothing, overflowing
443    /// the `Duration` the estimate is held in. That has to report no estimate
444    /// rather than panic: this runs in the node's logging task, and the
445    /// smoothed rate passes through chain pace whenever a node arrives at or
446    /// falls back from it.
447    #[test]
448    fn a_rate_barely_above_chain_pace_reports_no_estimate() {
449        let behind = Duration::from_secs(658 * 86_400);
450        assert_eq!(time_to_catch_up(behind, 1.0 + f64::EPSILON), None);
451    }
452
453    /// The same overflow reaches the epoch estimate through a rate that has
454    /// barely got going.
455    #[test]
456    fn a_rate_barely_above_zero_reports_no_epoch_estimate() {
457        let remaining = Duration::from_secs(86_400);
458        assert_eq!(time_to_epoch_end(remaining, f64::MIN_POSITIVE), None);
459        assert_eq!(time_to_epoch_end(remaining, 0.0), None);
460    }
461
462    /// A node moving at chain pace or slower never arrives, so no estimate is
463    /// offered rather than one that divides by nearly nothing.
464    #[test]
465    fn a_node_at_chain_pace_gets_no_estimate() {
466        let a_day = Duration::from_secs(86_400);
467        assert_eq!(time_to_catch_up(a_day, 1.0), None);
468        assert_eq!(time_to_catch_up(a_day, 0.5), None);
469    }
470
471    #[test]
472    fn format_duration_picks_the_two_largest_units() {
473        assert_eq!(format_duration(Duration::from_secs(45)), "45s");
474        assert_eq!(format_duration(Duration::from_secs(200)), "3m 20s");
475        assert_eq!(format_duration(Duration::from_secs(6120)), "1h 42m");
476        assert_eq!(format_duration(Duration::from_secs(280_800)), "3d 6h");
477    }

479    #[test]
480    fn format_count_abbreviates_large_numbers() {
481        assert_eq!(format_count(999), "999");
482        assert_eq!(format_count(1_200), "1.2k");
483        assert_eq!(format_count(123_400_000), "123.4M");
484        assert_eq!(format_count(2_500_000_000), "2.5B");
