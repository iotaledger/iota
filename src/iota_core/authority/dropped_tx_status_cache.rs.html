4//! In-memory bounded cache for transactions dropped by white-flag conflict
5//! resolution. Allows `wait_for_effects` callers to get an immediate
6//! `Rejected` response instead of hanging until timeout.
7
8use std::collections::{HashMap, VecDeque};
9
10use iota_common::sync::notify_read::NotifyRead;
11use iota_sdk_types::TransactionDigest;
12use iota_types::error::IotaError;
13use parking_lot::RwLock;
14
15/// Maximum number of dropped transaction entries to retain.
16/// At ~200 bytes per entry (digest + error), 100k entries ≈ 20 MB.
17const MAX_DROPPED_ENTRIES: usize = 100_000;
18
19pub(crate) struct DroppedTxStatusCache {
20    inner: RwLock<Inner>,
21    notify_read: NotifyRead<TransactionDigest, IotaError>,
22}
23
24struct Inner {
25    /// Digest → error lookup for the register-then-check pattern.
26    entries: HashMap<TransactionDigest, IotaError>,
27    /// Insertion order for FIFO eviction when at capacity.
28    insertion_order: VecDeque<TransactionDigest>,
29}
30
31impl Inner {
32    fn new() -> Self {
33        Self {
34            entries: HashMap::new(),
35            insertion_order: VecDeque::new(),
36        }
37    }
38
39    fn insert(&mut self, digest: TransactionDigest, error: IotaError) {
40        // If the digest is already present, skip (don't update ordering).
41        if self.entries.contains_key(&digest) {
42            return;
43        }
44        // Evict oldest entries to make room.
45        while self.entries.len() >= MAX_DROPPED_ENTRIES {
46            if let Some(oldest) = self.insertion_order.pop_front() {
47                self.entries.remove(&oldest);
48            }
49        }
50        self.entries.insert(digest, error);
51        self.insertion_order.push_back(digest);
52    }
53}
54
55impl DroppedTxStatusCache {
56    pub(crate) fn new() -> Self {
57        Self {
58            inner: RwLock::new(Inner::new()),
59            notify_read: NotifyRead::new(),
60        }
61    }
62
63    /// Record a batch of dropped transactions and notify any waiters.
64    /// When the cache is at capacity, the oldest entries are evicted first.
65    pub(crate) fn insert_and_notify(&self, dropped: &[(TransactionDigest, IotaError)]) {
66        {
67            let mut inner = self.inner.write();
68            for (digest, error) in dropped {
69                inner.insert(*digest, error.clone());
70            }
71        }
72        // Notify outside the lock — entries are already visible to readers,
73        // so the register-then-check pattern in notify_read_dropped is safe.
74        for (digest, error) in dropped {
75            self.notify_read.notify(digest, error);
76        }
77    }
78
79    /// Wait for a transaction to be dropped, or return immediately if it was
80    /// already dropped. Uses the register-then-check pattern to avoid the
81    /// race where `notify()` fires before the caller registers.
82    pub(crate) async fn notify_read_dropped(&self, digest: TransactionDigest) -> IotaError {
83        let registration = self.notify_read.register_one(&digest);
84        if let Some(error) = self.inner.read().entries.get(&digest) {
85            return error.clone();
86        }
87        registration.await
88    }
89
90    #[cfg(test)]
91    pub(crate) fn num_pending_for_testing(&self) -> usize {
92        self.notify_read.num_pending()
93    }
94}

96#[cfg(test)]
97mod tests {
98    use std::time::Duration;
99
100    use tokio::time::timeout;
101
102    use super::*;
103
104    /// When a transaction is already in the cache, `notify_read_dropped`
105    /// should return immediately instead of hanging.
106    #[tokio::test]
107    async fn test_returns_persisted_error() {
108        let cache = DroppedTxStatusCache::new();
109        let digest = TransactionDigest::random();
110        let expected_error = IotaError::TransactionExpired;
111
112        cache.insert_and_notify(&[(digest, expected_error.clone())]);
113
114        let result = timeout(Duration::from_secs(5), cache.notify_read_dropped(digest))
115            .await
116            .expect("should not timeout");
117
118        assert_eq!(result.to_string(), expected_error.to_string());
119    }
120
121    /// The normal path: client registers before the drop is recorded, and
122    /// the notification resolves the future.
123    #[tokio::test]
124    async fn test_waits_for_notification() {
125        let cache = std::sync::Arc::new(DroppedTxStatusCache::new());
126        let digest = TransactionDigest::random();
127        let expected_error = IotaError::TransactionExpired;
128
129        let cache_clone = cache.clone();
130        let handle = tokio::spawn(async move { cache_clone.notify_read_dropped(digest).await });
131
132        // Small delay so the spawned task registers before we notify.
133        tokio::time::sleep(Duration::from_millis(10)).await;
134
135        cache.insert_and_notify(&[(digest, expected_error.clone())]);
136
137        let result = timeout(Duration::from_secs(5), handle)
138            .await
139            .expect("should not timeout")
140            .expect("task should not panic");
141
142        assert_eq!(result.to_string(), expected_error.to_string());
143    }
144
145    /// When the cache is at capacity, the oldest entries are evicted to make
146    /// room for new ones.
147    #[tokio::test]
148    async fn test_capacity_evicts_oldest() {
149        let cache = DroppedTxStatusCache::new();
150
151        // Fill the cache to capacity.
152        let entries: Vec<_> = (0..MAX_DROPPED_ENTRIES)
153            .map(|_| (TransactionDigest::random(), IotaError::TransactionExpired))
154            .collect();
155        let first_digest = entries[0].0;
156        cache.insert_and_notify(&entries);
157
158        assert_eq!(cache.inner.read().entries.len(), MAX_DROPPED_ENTRIES);
159        assert!(cache.inner.read().entries.contains_key(&first_digest));
160
161        // Insert one more — the oldest (first) entry should be evicted.
162        let new_digest = TransactionDigest::random();
163        cache.insert_and_notify(&[(new_digest, IotaError::TransactionExpired)]);
164
165        assert_eq!(cache.inner.read().entries.len(), MAX_DROPPED_ENTRIES);
166        assert!(
167            !cache.inner.read().entries.contains_key(&first_digest),
168            "oldest entry should have been evicted"
169        );
170        assert!(
171            cache.inner.read().entries.contains_key(&new_digest),
172            "new entry should be present"
173        );
174    }
