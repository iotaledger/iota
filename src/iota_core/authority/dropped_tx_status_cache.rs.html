4//! In-memory bounded cache for transactions dropped by white-flag conflict
5//! resolution. Allows `wait_for_effects` callers to get an immediate
6//! `Rejected` response instead of hanging until timeout.
7
8use std::collections::{HashMap, VecDeque};
9
10use iota_common::sync::notify_read::NotifyRead;
11use iota_sdk_types::TransactionDigest;
12use iota_types::error::IotaError;
13use parking_lot::RwLock;
14
15/// Maximum number of dropped transaction entries to retain.
16/// At ~200 bytes per entry (digest + error), 100k entries ≈ 20 MB.
17const MAX_DROPPED_ENTRIES: usize = 100_000;
18
19pub(crate) struct DroppedTxStatusCache {
20    inner: RwLock<Inner>,
21    notify_read: NotifyRead<TransactionDigest, IotaError>,
22    /// Told about every drop that `notify_read` is told about, and also about
23    /// the drop of one copy of a digest, which is not the digest's status.
24    any_drop_notify_read: NotifyRead<TransactionDigest, IotaError>,
25}
26
27struct Inner {
28    /// Digest → error lookup for the register-then-check pattern.
29    entries: HashMap<TransactionDigest, IotaError>,
30    /// Insertion order for FIFO eviction when at capacity.
31    insertion_order: VecDeque<TransactionDigest>,
32}
33
34impl Inner {
35    fn new() -> Self {
36        Self {
37            entries: HashMap::new(),
38            insertion_order: VecDeque::new(),
39        }
40    }
41
42    fn insert(&mut self, digest: TransactionDigest, error: IotaError) {
43        // If the digest is already present, skip (don't update ordering).
44        if self.entries.contains_key(&digest) {
45            return;
46        }
47        // Evict oldest entries to make room. A digest removed when kept stays
48        // in the order, so the order, not the entries, is what is bounded.
49        while self.insertion_order.len() >= MAX_DROPPED_ENTRIES {
50            if let Some(oldest) = self.insertion_order.pop_front() {
51                self.entries.remove(&oldest);
52            }
53        }
54        self.entries.insert(digest, error);
55        self.insertion_order.push_back(digest);
56    }
57}
58
59impl DroppedTxStatusCache {
60    pub(crate) fn new() -> Self {
61        Self {
62            inner: RwLock::new(Inner::new()),
63            notify_read: NotifyRead::new(),
64            any_drop_notify_read: NotifyRead::new(),
65        }
66    }
67
68    /// Record a batch of dropped transactions and notify any waiters.
69    /// When the cache is at capacity, the oldest entries are evicted first.
70    pub(crate) fn insert_and_notify(&self, dropped: &[(TransactionDigest, IotaError)]) {
71        {
72            let mut inner = self.inner.write();
73            for (digest, error) in dropped {
74                inner.insert(*digest, error.clone());
75            }
76        }
77        // Notify outside the lock — entries are already visible to readers,
78        // so the register-then-check pattern in notify_read_dropped is safe.
79        for (digest, error) in dropped {
80            self.notify_read.notify(digest, error);
81            self.any_drop_notify_read.notify(digest, error);
82        }
83    }
84
85    /// Notify the current waiters of transactions dropped from one commit
86    /// only, without recording the drops: the same digests can still be kept
87    /// when sequenced again, so a later request must wait for that outcome.
88    pub(crate) fn notify_without_recording(&self, dropped: &[(TransactionDigest, IotaError)]) {
89        for (digest, error) in dropped {
90            self.notify_read.notify(digest, error);
91            self.any_drop_notify_read.notify(digest, error);
92        }
93    }
94
95    /// Notify only the waiters of [`Self::notify_read_any_drop`] of the drop
96    /// of one copy of each digest at a commit. Another copy of the same
97    /// digest can still be kept, so the drop is neither recorded nor sent to
98    /// the waiters of [`Self::notify_read_dropped`].
99    pub(crate) fn notify_dropped_copies(&self, dropped: &[(TransactionDigest, IotaError)]) {
100        for (digest, error) in dropped {
101            self.any_drop_notify_read.notify(digest, error);
102        }
103    }
104
105    /// Forget the recorded drops of transactions kept at a commit. The digest
106    /// does not cover the signatures, so a drop recorded earlier may be that
107    /// of another copy, or of an earlier submission, of a transaction that is
108    /// now kept: its status is then the outcome of its execution.
109    pub(crate) fn remove_kept(&self, kept: &[TransactionDigest]) {
110        let mut inner = self.inner.write();
111        for digest in kept {
112            inner.entries.remove(digest);
113        }
114    }
115
116    /// Wait for a transaction to be dropped, or return immediately if it was
117    /// already dropped. Uses the register-then-check pattern to avoid the
118    /// race where `notify()` fires before the caller registers.
119    pub(crate) async fn notify_read_dropped(&self, digest: TransactionDigest) -> IotaError {
120        let registration = self.notify_read.register_one(&digest);
121        if let Some(error) = self.inner.read().entries.get(&digest) {
122            return error.clone();
123        }
124        registration.await
125    }
126
127    /// Like [`Self::notify_read_dropped`], but also returns on the drop of one
128    /// copy of the digest at a commit, which is not the digest's status.
129    pub(crate) async fn notify_read_any_drop(&self, digest: TransactionDigest) -> IotaError {
130        let registration = self.any_drop_notify_read.register_one(&digest);
131        if let Some(error) = self.inner.read().entries.get(&digest) {
132            return error.clone();
133        }
134        registration.await
135    }
136
137    #[cfg(test)]
138    pub(crate) fn num_pending_for_testing(&self) -> usize {
139        self.notify_read.num_pending()
140    }
141}

143#[cfg(test)]
144mod tests {
145    use std::time::Duration;
146
147    use tokio::time::timeout;
148
149    use super::*;
150
151    /// When a transaction is already in the cache, `notify_read_dropped`
152    /// should return immediately instead of hanging.
153    #[tokio::test]
154    async fn test_returns_persisted_error() {
155        let cache = DroppedTxStatusCache::new();
156        let digest = TransactionDigest::random();
157        let expected_error = IotaError::TransactionExpired;
158
159        cache.insert_and_notify(&[(digest, expected_error.clone())]);
160
161        let result = timeout(Duration::from_secs(5), cache.notify_read_dropped(digest))
162            .await
163            .expect("should not timeout");
164
165        assert_eq!(result.to_string(), expected_error.to_string());
166    }
167
168    /// The normal path: client registers before the drop is recorded, and
169    /// the notification resolves the future.
170    #[tokio::test]
171    async fn test_waits_for_notification() {
172        let cache = std::sync::Arc::new(DroppedTxStatusCache::new());
173        let digest = TransactionDigest::random();
174        let expected_error = IotaError::TransactionExpired;
175
176        let cache_clone = cache.clone();
177        let handle = tokio::spawn(async move { cache_clone.notify_read_dropped(digest).await });
178
179        // Small delay so the spawned task registers before we notify.
180        tokio::time::sleep(Duration::from_millis(10)).await;
181
182        cache.insert_and_notify(&[(digest, expected_error.clone())]);
183
184        let result = timeout(Duration::from_secs(5), handle)
185            .await
186            .expect("should not timeout")
187            .expect("task should not panic");
188
189        assert_eq!(result.to_string(), expected_error.to_string());
190    }
191
192    /// A drop recorded before the same digest is kept is no longer answered
193    /// once it is kept.
194    #[tokio::test]
195    async fn test_keep_removes_an_earlier_drop() {
196        let cache = DroppedTxStatusCache::new();
197        let digest = TransactionDigest::random();
198        cache.insert_and_notify(&[(digest, IotaError::TransactionExpired)]);
199
200        cache.remove_kept(&[digest]);
201
202        assert!(
203            timeout(
204                Duration::from_millis(100),
205                cache.notify_read_dropped(digest)
206            )
207            .await
208            .is_err(),
209            "a request after the keep must wait for the kept transaction's outcome"
210        );
211    }
212
213    /// The drop of one copy of a digest ends a wait for any drop, but is not
214    /// sent to a status request nor recorded for a later one.
215    #[tokio::test]
216    async fn test_dropped_copy_ends_only_a_wait_for_any_drop() {
217        let cache = DroppedTxStatusCache::new();
218        let digest = TransactionDigest::random();
219        let any_drop = cache.notify_read_any_drop(digest);
220        let status = cache.notify_read_dropped(digest);
221        futures::pin_mut!(any_drop, status);
222        assert!(futures::poll!(&mut any_drop).is_pending());
223        assert!(futures::poll!(&mut status).is_pending());
224
225        cache.notify_dropped_copies(&[(digest, IotaError::TransactionExpired)]);
226
227        let error = timeout(Duration::from_secs(5), any_drop)
228            .await
229            .expect("a wait for any drop must end at the drop of a copy");
230        assert_eq!(error, IotaError::TransactionExpired);
231        assert!(
232            timeout(Duration::from_millis(100), status).await.is_err(),
233            "a status request must not get the drop of a copy"
234        );
235        assert!(
236            timeout(
237                Duration::from_millis(100),
238                cache.notify_read_dropped(digest)
239            )
240            .await
241            .is_err(),
242            "the drop of a copy must not be recorded"
243        );
244    }
245
246    /// When the cache is at capacity, the oldest entries are evicted to make
247    /// room for new ones.
248    #[tokio::test]
249    async fn test_capacity_evicts_oldest() {
250        let cache = DroppedTxStatusCache::new();
251
252        // Fill the cache to capacity.
253        let entries: Vec<_> = (0..MAX_DROPPED_ENTRIES)
254            .map(|_| (TransactionDigest::random(), IotaError::TransactionExpired))
255            .collect();
256        let first_digest = entries[0].0;
257        cache.insert_and_notify(&entries);
258
259        assert_eq!(cache.inner.read().entries.len(), MAX_DROPPED_ENTRIES);
260        assert!(cache.inner.read().entries.contains_key(&first_digest));
261
262        // Insert one more — the oldest (first) entry should be evicted.
263        let new_digest = TransactionDigest::random();
264        cache.insert_and_notify(&[(new_digest, IotaError::TransactionExpired)]);
265
266        assert_eq!(cache.inner.read().entries.len(), MAX_DROPPED_ENTRIES);
267        assert!(
268            !cache.inner.read().entries.contains_key(&first_digest),
269            "oldest entry should have been evicted"
270        );
271        assert!(
272            cache.inner.read().entries.contains_key(&new_digest),
273            "new entry should be present"
274        );
275    }
