4//! Judges the attestation of a transaction whose Move authentication failed at
5//! execution, by re-running authentication at the object versions the attestor
6//! recorded. The verdict is recorded as `AttestationRecord::verdict` and
7//! certified in the checkpoint summary unless `Valid`. When the gas owner's own
8//! authentication fails, the judge also decides who pays: it finds who is at
9//! fault, as an [`AuthenticationFailureBlame`], and the gas owner pays only when
10//! the account is, because authentication passes at the attested state. When
11//! the judge cannot decide, the record is
12//! [`attestation::AttestationVerdict::Undecided`], certified so that a node
13//! executing from a checkpoint charges nobody as well.
14

60/// When a recorded version stopped being current: the commit of the
61/// transaction that replaced it, the leader round when that transaction is a
62/// commit prologue or a deny-rules update, which carry no commit index, or
63/// only the epoch when it is another system transaction, which carries
64/// neither, or when no window is set.
65#[derive(Clone, Copy, Debug, PartialEq, Eq)]
66pub(crate) enum SupersededAt {
67    Commit(CommitIndex),
68    Round(u64),
69    Epoch,
70}
71
72/// Finds when a recorded version stopped being current this epoch.
73pub(crate) struct AttestedObjectVersions<'a> {
74    pub(crate) object_cache: &'a dyn ObjectCacheRead,
75    pub(crate) transaction_cache: &'a dyn TransactionCacheRead,
76    pub(crate) epoch_store: &'a AuthorityPerEpochStore,
77    /// Where consensus first sequenced the attested transaction.
78    pub(crate) sequenced_at: ExecutionIndices,
79    /// How many commits, or rounds, before `sequenced_at` a supersession may
80    /// be; `None` admits any of the epoch.
81    pub(crate) window: Option<u64>,
82}
83
84impl AttestedObjectVersions<'_> {
85    /// When `attested` stopped being current, if a transaction of this epoch
86    /// replaced it inside the window.
87    ///
88    /// # Panics
89    ///
90    /// If a read fails, or the transaction that replaced `attested` has no
91    /// stored body, no row for its gas coin, or does not record its effects:
92    /// a verdict every validator has to reach alike must not depend on this
93    /// node's storage.
94    pub(crate) fn superseded_within_window(
95        &self,
96        object_id: &ObjectId,
97        attested: Version,
98    ) -> Option<SupersededAt> {
99        let epoch = self.epoch_store.epoch();
100        let (replacing, uncommitted) = self
101            .object_cache
102            .try_find_lowest_version_above_in_epoch(object_id, attested, epoch)
103            .unwrap_or_else(|error| {
104                fatal!("cannot read the version after {object_id}@{attested}: {error}")
105            })?;
106        let object = self
107            .object_cache
108            .try_get_object_by_key_with_historic_fallback(object_id, replacing, epoch)
109            .unwrap_or_else(|error| fatal!("cannot read {object_id}@{replacing}: {error}"));
110        let digest = match object {
111            Some(object) => object.previous_transaction,
112            // Deleted: its row names the transaction. A wrap's row names
113            // none, but neither a shared object nor a dynamic field is ever
114            // wrapped.
115            None => self.row(&ObjectKey(*object_id, replacing))?.deleted_by?,
116        };
117        let effects = self.effects_in_epoch(&digest, uncommitted)?;
118        // Consuming another version, or none, means `attested` was never a
119        // version of the object, or was superseded before this epoch.
120        let consumed = effects
121            .modified_at_versions()
122            .into_iter()
123            .find(|modified| modified.object_id() == object_id)?
124            .version();
125        if consumed != attested {
126            return None;
127        }
128        // Without a window, being superseded this epoch is all that counts.
129        let Some(window) = self.window else {
130            return Some(SupersededAt::Epoch);
131        };
132        let at = self.transaction_position(&digest, &effects);
133        within_freshness_window(at, self.sequenced_at, window).then_some(at)
134    }
135
136    /// The effects of `digest` when it executed in this epoch, `None` when it
137    /// ran in an earlier one, whose records a node may have pruned. Waits for
138    /// them when the version naming `digest` is `uncommitted`: a version is
139    /// readable before its writer records its effects.
140    ///
141    /// # Panics
142    ///
143    /// If a read fails, or the effects waited for are still missing after
144    /// ten seconds.
145    fn effects_in_epoch(
146        &self,
147        digest: &TransactionDigest,
148        uncommitted: bool,
149    ) -> Option<TransactionEffects> {
150        // The writer records its effects in the synchronous call that made
151        // its outputs readable: a wait this long means that order broke.
152        const PAUSE: Duration = Duration::from_millis(1);
153        const MAX_PAUSES: u32 = 10_000;
154
155        let epoch = self.epoch_store.epoch();
156        let read = || {
157            let effects_digest = self
158                .transaction_cache
159                .try_multi_get_executed_effects_digests_in_epoch(epoch, &[*digest])
160                .unwrap_or_else(|error| fatal!("cannot read whether {digest} executed: {error}"))
161                .pop()
162                .flatten()?;
163            self.transaction_cache
164                .try_multi_get_effects_in_epoch(epoch, &[effects_digest])
165                .unwrap_or_else(|error| fatal!("cannot read the effects of {digest}: {error}"))
166                .pop()
167                .flatten()
168        };
169        let mut pauses = 0;
170        loop {
171            match read() {
172                // The committed caches outlive the epoch, so either read can
173                // answer for an earlier one.
174                Some(effects) => return (effects.epoch() == epoch).then_some(effects),
175                None if !uncommitted => return None,
176                None if pauses == MAX_PAUSES => fatal!(
177                    "no effects for {digest} {:?} after its outputs became readable",
178                    PAUSE * MAX_PAUSES
179                ),
180                // The writer is inside the cache write that records them.
181                None => {
182                    pauses += 1;
183                    std::thread::sleep(PAUSE);
184                }
185            }
186        }
187    }
188
189    /// Where `digest` ran: its commit; its round for a commit prologue or a
190    /// deny-rules update; only the epoch for any other system transaction.
191    ///
192    /// # Panics
193    ///
194    /// If a read fails, or `digest` has no stored body or no row for its gas
195    /// coin.
196    fn transaction_position(
197        &self,
198        digest: &TransactionDigest,
199        effects: &TransactionEffects,
200    ) -> SupersededAt {
201        let transaction = self
202            .transaction_cache
203            .try_multi_get_transaction_blocks_in_epoch(self.epoch_store.epoch(), &[*digest])
204            .unwrap_or_else(|error| fatal!("cannot read transaction {digest}: {error}"))
205            .pop()
206            .flatten()
207            .unwrap_or_else(|| fatal!("no transaction {digest} behind its executed effects"));
208        match transaction.inner().transaction().kind() {
209            TransactionKind::ConsensusCommitPrologueV1(prologue) => {
210                return SupersededAt::Round(prologue.round);
211            }
212            TransactionKind::TransactionDenyRulesUpdate(update) => {
213                return SupersededAt::Round(update.round);
214            }
215            _ if transaction.is_system_tx() => return SupersededAt::Epoch,
216            _ => {}
217        }
218        // Recorded before the outputs of `digest` became readable.
219        let gas = effects.gas_object();
220        let gas = gas.reference();
221        let row = self
222            .row(&ObjectKey(gas.object_id, gas.version))
223            .unwrap_or_else(|| fatal!("no row for the gas coin of {digest}"));
224        SupersededAt::Commit(row.produced_at)
225    }
226
227    /// The handler row at `key`.
228    ///
229    /// # Panics
230    ///
231    /// If the read fails.
232    fn row(&self, key: &ObjectKey) -> Option<HandlerProcessedObject> {
233        self.epoch_store
234            .handler_processed_object(key)
235            .unwrap_or_else(|error| fatal!("cannot read the row of {key:?}: {error}"))
236    }
237}

239/// Whether a version superseded at `at` was current recently enough before
240/// `sequenced_at` for an honest dry run to have read it: at or after the
241/// window's start. A supersession known only by its epoch cannot be
242/// measured, so it is admitted.
243fn within_freshness_window(at: SupersededAt, sequenced_at: ExecutionIndices, window: u64) -> bool {
244    match at {
245        SupersededAt::Commit(commit) => commit >= sequenced_at.sub_dag_index.saturating_sub(window),
246        SupersededAt::Round(round) => {
247            round >= sequenced_at.last_committed_round.saturating_sub(window)
248        }
249        SupersededAt::Epoch => true,
250    }
251}

253/// What the authority needs to judge an attestation when the attested
254/// transaction fails Move authentication at execution.
255pub(crate) struct AttestationVerdictContext<'a> {
256    pub attestation: &'a Attestation,
257    pub attested_versions: AttestedObjectVersions<'a>,
258    pub store: &'a dyn BackingStore,
259    pub executor: &'a dyn Executor,
260    pub protocol_config: &'a ProtocolConfig,
261    pub metrics: Arc<LimitsMetrics>,
262    pub chain_identifier: ChainIdentifier,
263    pub epoch_id: EpochId,
264    pub epoch_timestamp_ms: u64,
265    pub reference_gas_price: u64,
266    pub gas_data: GasPayment,
267    /// Each Move authenticator with the input objects execution loaded for it.
268    pub authenticators: Vec<(MoveAuthenticator, InputObjects)>,
269    /// Versions of the authenticator inputs and function-ref fields execution
270    /// ran against.
271    pub executed_versions: BTreeMap<ObjectId, Version>,
272    /// The signed transaction. The entries of the attestation name objects by
273    /// their index in it. Its data is serialized, and its kind cloned, only
274    /// when a re-run needs them.
275    pub transaction: &'a SenderSignedTransaction,
276    pub transaction_signer: Address,
277    pub transaction_digest: TransactionDigest,
278    pub sender_auth_digest: Digest,
279    pub sponsor_auth_digest: Option<Digest>,
280}

282/// The authenticators execution runs, paired with the inputs it loaded.
283pub(crate) fn authenticator_inputs(
284    authenticators: &[&MoveAuthenticator],
285    inputs: &[CheckedInputObjects],
286) -> Vec<(MoveAuthenticator, InputObjects)> {
287    authenticators
288        .iter()
289        .zip(inputs)

291        .collect()

294/// The versions execution authenticates against: every authenticator input
295/// plus each resolved function-ref field object.
296pub(crate) fn executed_versions(
297    inputs: &[CheckedInputObjects],
298    resolutions: &[Result<AuthenticatorFunctionRefForExecution, ExecutionError>],
299) -> BTreeMap<ObjectId, Version> {
300    inputs
301        .iter()
302        .zip(resolutions)
303        .flat_map(|(objects, resolution)| {
304            objects
305                .inner()
306                .iter()
307                .map(|object| (object.id(), object.version()))
308                .chain(resolution.as_ref().ok().map(|function_ref| {
309                    (
310                        function_ref.loaded_object_id,
311                        function_ref.loaded_object_metadata.version,
312                    )
313                }))
314        })
315        .collect()
316}

318/// Who is at fault when the Move authentication of an attested transaction
319/// fails at execution, as the judge finds it.
320#[derive(Clone, Copy, Debug, PartialEq, Eq)]
321pub(crate) enum AuthenticationFailureBlame {
322    /// The attestor is at fault. Authentication fails at the versions it
323    /// recorded too, or the judge blames it without a re-run, as
324    /// [`AttestationVerdictContext::decide_authentication_failure_blame()`]
325    /// describes. `rerun_computation_units` is what the judge's re-run consumed,
326    /// zero when nothing had to be re-run.
327    Attestor { rerun_computation_units: u64 },
328
329    /// The account is at fault. Authentication passes at the recorded versions,
330    /// so the account authorized the transaction at the attested state.
331    Account,
332
333    /// Nobody is charged or penalized. The failure is an invariant violation,
334    /// an internal error, or a cancellation, which the judge does not re-run;
335    /// see [`AttestationVerdictContext::decide_authentication_failure_blame()`].
336    Nobody,
337
338    /// The judge could not decide who is at fault: its own re-run failed with
339    /// an internal error. Recorded as
340    /// [`attestation::AttestationVerdict::Undecided`], which is certified, so a
341    /// node executing from a checkpoint charges nobody as well.
342    Undecided,
343}
344
345impl AttestationVerdictContext<'_> {
346    /// Decides who is at fault for `kind`, the error of a failed Move
347    /// authentication, wrapped in an [`ExecutionErrorKind::MoveAuthentication`]
348    /// status or not. Returns [`AuthenticationFailureBlame::Nobody`] for a
349    /// cancellation or an invariant violation, without a re-run; otherwise, the
350    /// recorded versions decide, through a re-run of authentication at them
351    /// when the executed state drifted from them. A deleted authenticator input
352    /// is a drifted one: the deletion superseded the recorded version.
353    ///
354    /// # Panics
355    ///
356    /// If the recorded state cannot be read. The drift check only admits
357    /// versions superseded this epoch within the freshness window, which the
358    /// epoch's historic bucket keeps, so a missing or unreadable one is a fault
359    /// of this node and must not become a verdict the other validators do not
360    /// share.
361    pub(crate) fn decide_authentication_failure_blame(
362        &self,
363        kind: &ExecutionErrorKind,
364    ) -> AuthenticationFailureBlame {
365        let kind = unwrap_move_authentication_error(kind);
366
367        // An invariant violation is an internal error, so nobody is at fault.
368        // A cancellation was decided at the consensus commit before any
369        // authenticator ran. No validator ran one at execution, and judging
370        // the transaction would mean running one only to charge for that run.
371        if is_cancellation(kind) || is_any_invariant_violation(kind) {
372            return AuthenticationFailureBlame::Nobody;
373        }
374
375        // Nothing to re-run: the failure already reproduces at the recorded
376        // state, or a recorded version refutes on its own.
377        let attested_versions = collect_attested_versions(self.attestation, self.transaction);
378        if !should_reauthenticate(
379            &attested_versions,
380            &self.executed_versions,
381            |object_id, version| {
382                self.attested_versions
383                    .superseded_within_window(object_id, version)
384                    .is_some()
385            },
386        ) {
387            return AuthenticationFailureBlame::Attestor {
388                rerun_computation_units: 0,
389            };
390        }
391
392        // The objects authentication read changed after the attestation, so only
393        // a re-run at the recorded versions shows who is at fault.
394        self.reauthenticate_at_attested_versions(attested_versions.into_iter().collect())
395    }
396}

398/// Returns the object id and the version of each entry that `attestation`
399/// records, in the order of the record. `transaction` is the signed
400/// transaction in which the entries name objects by index, as
401/// [`list_move_authenticator_shared_objects()`] lists them for the check of the
402/// record at the commit. An entry whose index names no shared object there is
403/// left out: the check drops a copy whose record has one.
404fn collect_attested_versions(
405    attestation: &Attestation,
406    transaction: &SenderSignedTransaction,
407) -> Vec<(ObjectId, Version)> {
408    let object_ids: HashMap<SharedInputIndex, ObjectId> =
409        list_move_authenticator_shared_objects(transaction)
410            .into_iter()
411            .map(|(index, object)| (index, object.object_id))
412            .collect();
413
414    attestation
415        .shared_object_versions()
416        .iter()
417        .filter_map(|(index, version)| Some((*object_ids.get(index)?, *version)))
418        .collect()
419}

421/// Whether authentication should be re-run at the recorded versions. `false`
422/// either when nothing the authentication read drifted, so the failure
423/// reproduces at the recorded state, or when a recorded version is refuted
424/// without a re-run.
425fn should_reauthenticate(
426    attested_versions: &[(ObjectId, Version)],
427    executed_versions: &BTreeMap<ObjectId, Version>,
428    superseded_within_window: impl Fn(&ObjectId, Version) -> bool,
429) -> bool {
430    let mut drifted = false;
431    for &(object_id, attested) in attested_versions {
432        let Some(&executed) = executed_versions.get(&object_id) else {
433            continue;
434        };
435        if attested == executed {
436            continue;
437        }
438        // A recorded version ahead of the executed one cannot come from an
439        // honest dry run. A recorded version superseded before the freshness
440        // window was not current when an honest dry run could have read it.
441        if attested > executed || !superseded_within_window(&object_id, attested) {
442            return false;
443        }
444        drifted = true;
445    }
446    drifted
447}

449impl AttestationVerdictContext<'_> {
450    /// Re-runs Move authentication at `attested_versions`. The engine meters
451    /// the re-run as execution does: the gas owner's authenticator first, and
452    /// all of them within `max_auth_gas`. Returns
453    /// [`AuthenticationFailureBlame::Attestor`] when the re-run fails too, with
454    /// the units it consumed, or with zero when the claimed units cannot fund a
455    /// re-run, an authenticator cannot be resolved at the recorded state, or
456    /// the account is immutable where execution rejects one;
457    /// [`AuthenticationFailureBlame::Account`] when the re-run passes;
458    /// [`AuthenticationFailureBlame::Undecided`] when the authenticators' inputs
459    /// cannot be combined, or the re-run ends in a cancellation or an invariant
460    /// violation: here, all three can only be internal errors.
461    fn reauthenticate_at_attested_versions(
462        &self,
463        attested_versions: BTreeMap<ObjectId, Version>,
464    ) -> AuthenticationFailureBlame {
465        // The re-run is metered on its own, capped by the attestor's claimed
466        // computation units: exceeding them refutes the attestation by its own
467        // claim, and the cost never reaches the transaction's gas or effects.
468        let gas_price = self.gas_data.price;
469        let attested_budget = self
470            .attestation
471            .computation_units()
472            .saturating_mul(gas_price);
473        // The engine also holds the authenticators to `max_auth_gas`, as in an
474        // honest dry run, so an authentication that passes only above it
475        // refutes the attestation too.
476        let Ok(gas_status) = IotaGasStatus::new(
477            attested_budget,
478            gas_price,
479            self.reference_gas_price,
480            self.protocol_config,
481        ) else {
482            return AuthenticationFailureBlame::Attestor {
483                rerun_computation_units: 0,
484            };
485        };
486
487        // Resolve every authenticator at the recorded versions first, so the
488        // rebuilt auth context carries the same function refs the re-run
489        // executes.
490        let mut resolved = Vec::with_capacity(self.authenticators.len());
491        for (authenticator, input_objects) in &self.authenticators {
492            let reloaded_input_objects =
493                self.reload_input_objects_at_attested_versions(input_objects, &attested_versions);
494
495            let Ok((account_id, pinned_version, pinned_digest)) =
496                authenticator.object_to_authenticate_components()
497            else {
498                return AuthenticationFailureBlame::Attestor {
499                    rerun_computation_units: 0,
500                };
501            };
502
503            // The function ref is resolved from the recorded state, anchored
504            // at the account exactly like execution resolves it: deriving the
505            // field from the account version means a recorded account/field
506            // pair that never coexisted cannot be judged at.
507            let Some(account_object) = reloaded_input_objects
508                .iter()
509                .find(|object| object.id() == account_id)
510                .and_then(|object| object.as_object())
511            else {
512                return AuthenticationFailureBlame::Attestor {
513                    rerun_computation_units: 0,
514                };
515            };
516            let Ok(account_version) = validate_account_object(
517                account_id,
518                pinned_version,
519                pinned_digest,
520                &authenticator.address(),
521                account_object,
522            ) else {
523                return AuthenticationFailureBlame::Attestor {
524                    rerun_computation_units: 0,
525                };
526            };
527            // Execution rejects an immutable account at every state, so no
528            // honest dry run authenticated against one.
529            if self.protocol_config.reject_immutable_account_objects()
530                && account_object.is_immutable()
531            {
532                return AuthenticationFailureBlame::Attestor {
533                    rerun_computation_units: 0,
534                };
535            }
536            let Ok(field_object_id) =
537                derive_authenticator_function_ref_v1_dynamic_field_id(account_id)
538            else {
539                return AuthenticationFailureBlame::Attestor {
540                    rerun_computation_units: 0,
541                };
542            };
543            let field_object = match self.store.read_child_object(
544                &account_id,
545                &field_object_id,
546                account_version,
547            ) {
548                Ok(Some(field_object)) => field_object,
549                // The structural failure reproduces at the recorded state:
550                // no field, or one that is not the account's child.
551                Ok(None) | Err(IotaError::InvalidChildObjectAccess { .. }) => {
552                    return AuthenticationFailureBlame::Attestor {
553                        rerun_computation_units: 0,
554                    };
555                }
556                Err(error) => fatal!(
557                    "cannot read the authenticator field of {account_id}@{account_version}: {error}"
558                ),
559            };
560            let Ok(function_ref) =
561                authenticator_function_ref_v1_from_dynamic_field_object(account_id, &field_object)
562            else {
563                return AuthenticationFailureBlame::Attestor {
564                    rerun_computation_units: 0,
565                };
566            };
567
568            resolved.push((
569                authenticator.clone(),
570                function_ref.authenticator_function_ref,
571                CheckedInputObjects::new_with_checked_transaction_inputs(reloaded_input_objects),
572            ));
573        }
574
575        let per_authenticator_input_objects: Vec<_> = resolved
576            .iter()
577            .map(|(_, _, input_objects)| input_objects)
578            .collect();
579        // Combining the authenticators' inputs fails only when two of them
580        // declare the same object with conflicting kinds. The reload keeps the
581        // declared kinds, and post-consensus validation already accepted them,
582        // so a failure here is an internal error.
583        let Ok(aggregated_input_objects) =
584            iota_transaction_checks::aggregate_authenticator_input_objects(
585                &per_authenticator_input_objects,
586            )
587        else {
588            return AuthenticationFailureBlame::Undecided;
589        };
590
591        // Rebuild the auth context with the function refs resolved at the
592        // recorded versions.
593        let (sender_authenticator_function_ref, sponsor_authenticator_function_ref) =
594            extract_auth_fun_refs(self.transaction_signer, self.gas_data.owner, |address| {
595                resolved
596                    .iter()
597                    .find(|(authenticator, _, _)| authenticator.address() == address)
598                    .map(|(_, function_ref, _)| function_ref.clone())
599            });
600        let auth_context_data = AuthContextData {
601            transaction_data_bytes: bcs::to_bytes(self.transaction.transaction())
602                .expect("Transaction serialization cannot fail"),
603            sender_auth_digest: self.sender_auth_digest,
604            sponsor_auth_digest: self.sponsor_auth_digest,
605            sender_authenticator_function_ref,
606            sponsor_authenticator_function_ref,
607        };
608        let (transaction_kind, _, _) = self.transaction.transaction().execution_parts();
609
610        // The gas data execution has, gas coins included, so the re-run differs
611        // from execution only in the object versions. It charges nobody all the
612        // same: `authenticate_transaction` charges no gas coin, and the re-run
613        // is metered on `gas_status` alone, with its writes dropped.
614        let (gas_status, result) = self.executor.authenticate_transaction(
615            self.store,
616            self.protocol_config,
617            self.metrics.clone(),
618            self.chain_identifier,
619            &self.epoch_id,
620            self.epoch_timestamp_ms,
621            self.gas_data.clone(),
622            gas_status,
623            resolved,
624            aggregated_input_objects,
625            transaction_kind,
626            self.transaction_signer,
627            self.transaction_digest,
628            auth_context_data,
629            &mut None,
630        );
631
632        // Tests register the `judge-rerun-internal-error` fail point with a
633        // transaction digest to make the re-run of that transaction end in an
634        // internal error, which no transaction can cause. A `Cell` keeps the
635        // binding immutable in builds where the fail point compiles to
636        // nothing.
637        let result = Cell::new(result);
638        iota_macros::fail_point_arg!("judge-rerun-internal-error", |digest: TransactionDigest| {
639            if digest == self.transaction_digest {
640                result.set(Err(ExecutionError::invariant_violation(
641                    "the judge-rerun-internal-error fail point",
642                )));
643            }
644        });
645
646        match result.into_inner() {
647            Ok(()) => AuthenticationFailureBlame::Account,
648            Err(error) => {
649                // The re-run carries no cancelled input, so a cancellation
650                // here is an internal error, like an invariant violation.
651                let kind = unwrap_move_authentication_error(error.kind());
652                if is_cancellation(kind) || is_any_invariant_violation(kind) {
653                    AuthenticationFailureBlame::Undecided
654                } else {
655                    AuthenticationFailureBlame::Attestor {
656                        rerun_computation_units: consumed_units(gas_status),
657                    }
658                }
659            }
660        }
661    }

663    /// Rebuilds the input objects for authentication at the versions the
664    /// attestor recorded, reusing the executed object when the recorded
665    /// version is the one execution loaded.
666    fn reload_input_objects_at_attested_versions(
667        &self,
668        input_objects: &InputObjects,
669        attested_versions: &BTreeMap<ObjectId, Version>,
670    ) -> InputObjects {
671        let mut reloaded = Vec::with_capacity(input_objects.len());
672        for object_read_result in input_objects.iter() {
673            let object_id = object_read_result.id();
674            match attested_versions.get(&object_id) {
675                Some(&version) if version != object_read_result.version() => {
676                    // The drift check only admits versions superseded this
677                    // epoch, which the epoch's historic bucket keeps.
678                    let object = self
679                        .attested_versions
680                        .object_cache
681                        .try_get_object_by_key_with_historic_fallback(
682                            &object_id,
683                            version,
684                            self.epoch_id,
685                        )
686                        .unwrap_or_else(|error| {
687                            fatal!("cannot read {object_id}@{version}: {error}")
688                        })
689                        .unwrap_or_else(|| {
690                            fatal!("{object_id}@{version} was superseded this epoch but is gone")
691                        });
692                    reloaded.push(ObjectReadResult::new(
693                        object_read_result.input_object_kind,
694                        ObjectReadResultKind::Object(object),
695                    ));
696                }
697                _ => reloaded.push(object_read_result.clone()),
698            }
699        }
700        InputObjects::new(reloaded)
701    }
702}

704/// The computation units a metered run consumed, bucketized like a charged
705/// run; the whole budget when it ran out of gas.
706pub(crate) fn consumed_units(mut gas_status: IotaGasStatus) -> u64 {
707    // Out of gas pins the computation cost at the budget.
708    let _ = gas_status.bucketize_computation();
709    attestation::computation_units(&gas_status.summary(), gas_status.gas_price())
710}

712/// What execution established for the verdict on an attested transaction.
713pub(crate) struct ExecutionOutcome {
714    /// Who the judge found at fault when the Move authentication of an
715    /// attested transaction failed; `None` when it passed or nobody judged.
716    pub blame: Option<AuthenticationFailureBlame>,
717    /// The computation units execution's own run consumed, as
718    /// [`consumed_units()`] counts them, whether or not the effects charge them;
719    /// zero without Move authentication.
720    pub consumed_units: u64,
721    /// The body ran, so its computation cost is measurable against the claim.
722    pub body_ran: bool,
723}
724
725/// Whether the transaction body ran, so its computation cost measures the same
726/// work the attestor's dry run did.
727pub(crate) fn body_ran(authentication_failed: bool, error: Option<&ExecutionError>) -> bool {
728    !authentication_failed && error.is_none_or(|error| !is_pre_execution_failure(error.kind()))
729}
730
731/// The checks that run before the transaction body: the input checks on the
732/// authenticator's inputs and the transaction's alike, and the coin deny-list
733/// check that follows authentication.
734fn is_pre_execution_failure(kind: &ExecutionErrorKind) -> bool {
735    matches!(
736        kind,
737        ExecutionErrorKind::InputObjectDeleted
738            | ExecutionErrorKind::TransactionDependencyFailed { .. }
739            | ExecutionErrorKind::AddressDeniedForCoin { .. }
740            | ExecutionErrorKind::CoinTypeGlobalPause { .. }
741    ) || is_cancellation(kind)
742}

859    /// The re-run's units are the bucketized computation in gas units, not
860    /// NANOS, and the whole budget when the run exhausted it.
861    #[test]
862    fn consumed_units_are_bucketized_gas_units_capped_at_the_budget() {
863        let config = ProtocolConfig::get_for_max_version_UNSAFE();
864        let step = config.gas_rounding_step();
865        let (gas_price, reference_gas_price) = (2 * step, step);
