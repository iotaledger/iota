4//! Received-and-deleted object markers, bucketed by the epoch that wrote them.
5//!
6//! A marker guards a race inside the epoch that wrote it, so only the current
7//! epoch's markers are kept, and an earlier epoch's are removed with a
8//! `drop_cf` rather than a range delete the execution path would read across.
9
10use std::{collections::BTreeMap, ops::Bound, path::Path, sync::Arc};
11
12use iota_sdk_types::ObjectId;
13use iota_types::{
14    base_types::VersionNumber,
15    committee::EpochId,
16    error::{IotaError, IotaResult},
17    storage::{MarkerValue, ObjectKey},
18};
19use typed_store::{
20    TypedStoreError,
21    database::Database,
22    rocks::{DBMap, DBOptions, ReadWriteOptions, list_tables},
23    traits::Map,
24};

26use crate::{
27    epoch_buckets::{
28        BucketReopen, EpochBuckets, PrunedBuckets, absent_if_dropped, bucket_cf_epoch,
29        bucket_metrics_label, dump_page, dump_retention_floor, dump_rows,
30    },
31    progress_logger::ProgressLogger,
32};
33
34/// Column-family prefix of the marker buckets; a bucket's family is
35/// `{prefix}{epoch}`.
36const MARKERS_CF_PREFIX: &str = "marker_e";
37
38/// Column family holding the earliest-retained-epoch marker
39/// [`EpochBuckets::prune`] persists.
40const EARLIEST_RETAINED_CF: &str = "marker_earliest_retained";
41
42/// Rows one slice of the migration moves before it writes its batch.
43const KEYS_PER_SLICE: usize = 5_000;
44
45/// Historic epochs kept at a reconfiguration, on top of the epoch being
46/// entered: none, as nothing reads a marker outside the running epoch.
47const HISTORIC_EPOCHS_TO_RETAIN: u64 = 0;
48
49/// One epoch's markers.
50pub struct EpochMarkersBucket {
51    /// The objects received, deleted or wrapped during this epoch, at the
52    /// version it happened at.
53    pub(crate) markers: DBMap<ObjectKey, MarkerValue>,
54}
55
56impl BucketReopen for EpochMarkersBucket {
57    fn reopen(db: &Arc<Database>, cf_name: &str) -> Result<Self, TypedStoreError> {
58        Ok(Self {
59            markers: DBMap::reopen(db, Some(cf_name), &ReadWriteOptions::default(), true)?
60                .with_metrics_label(bucket_metrics_label(cf_name)),
61        })
62    }
63}
64
65/// The markers of the epochs still retained, one column family each.
66pub struct EpochMarkers {
67    buckets: EpochBuckets<EpochMarkersBucket>,
68}
69
70impl EpochMarkers {
71    /// Options for a marker bucket's column family: the base options, without
72    /// the write-heavy tuning of the historic buckets, as markers are read on
73    /// the execution path. Clone the result per column family so they share
74    /// one block cache.
75    fn cf_options(db_options: &DBOptions) -> DBOptions {
76        db_options.clone()
77    }
78
79    /// The `(name, options)` pairs of this store's column families, for the
80    /// perpetual store to open alongside its own tables; one left out would be
81    /// reopened with default options.
82    pub fn extra_column_family_options(
83        perpetual_path: &Path,
84        db_options: &DBOptions,
85    ) -> Vec<(String, DBOptions)> {
86        let cf_options = Self::cf_options(db_options);
87        let mut options = vec![(EARLIEST_RETAINED_CF.to_string(), cf_options.clone())];
88        if !perpetual_path.join("CURRENT").exists() {
89            return options;
90        }
91        let Ok(existing_cfs) = list_tables(perpetual_path.to_path_buf()) else {
92            return options;
93        };
94        options.extend(
95            existing_cfs
96                .into_iter()
97                .filter(|name| bucket_cf_epoch(MARKERS_CF_PREFIX, name).is_some())
98                .map(|name| (name, cf_options.clone())),
99        );
100        options
101    }
102
103    /// Opens the marker buckets already present among `db`'s column families.
104    /// `db` is the perpetual database's handle and `db_options` the options its
105    /// tables were opened with.
106    pub fn open(db: Arc<Database>, db_options: &DBOptions) -> Result<Self, TypedStoreError> {
107        let existing_cfs = list_tables(db.path_for_pruning().to_path_buf())
108            .map_err(|e| TypedStoreError::RocksDB(format!("failed to list marker buckets: {e}")))?;
109
110        let mut buckets = BTreeMap::new();
111        for cf_name in &existing_cfs {
112            if let Some(epoch) = bucket_cf_epoch(MARKERS_CF_PREFIX, cf_name) {
113                buckets.insert(epoch, Arc::new(EpochMarkersBucket::reopen(&db, cf_name)?));
114            }
115        }
116
117        let cf_options = Self::cf_options(db_options).options;
118        if db.cf_handle(EARLIEST_RETAINED_CF).is_none() {
119            db.create_cf(EARLIEST_RETAINED_CF, &cf_options)?;
120        }
121        let earliest_retained_table: DBMap<(), EpochId> = DBMap::reopen(
122            &db,
123            Some(EARLIEST_RETAINED_CF),
124            &ReadWriteOptions::default(),
125            true,
126        )?;
127
128        Ok(Self {
129            buckets: EpochBuckets::open(
130                db,
131                "epoch markers",
132                MARKERS_CF_PREFIX,
133                cf_options,
134                // Markers are working data of the epochs being executed rather
135                // than history, so they stay in the database directory.
136                None,
137                earliest_retained_table,
138                buckets,
139            )?,
140        })
141    }
142
143    /// The bucket `epoch`'s markers are written to, created if absent.
144    pub(crate) fn ensure(&self, epoch: EpochId) -> IotaResult<Arc<EpochMarkersBucket>> {
145        self.buckets
146            .ensure(epoch)
147            .map_err(|e| IotaError::Storage(e.to_string()))
148    }
149
150    /// The marker written for `object_id` at exactly `version` during `epoch`,
151    /// `None` also once that epoch's bucket has been dropped.
152    pub fn get_marker_value(
153        &self,
154        object_id: &ObjectId,
155        version: &VersionNumber,
156        epoch: EpochId,
157    ) -> IotaResult<Option<MarkerValue>> {
158        let Some(bucket) = self.buckets.get(epoch) else {
159            return Ok(None);
160        };
161        Ok(absent_if_dropped(
162            bucket.markers.get(&ObjectKey(*object_id, *version)),
163        )?)
164    }
165
166    /// The newest version of `object_id` marked during `epoch`, with its
167    /// marker, `None` also once that epoch's bucket has been dropped.
168    pub fn get_latest_marker(
169        &self,
170        object_id: &ObjectId,
171        epoch: EpochId,
172    ) -> IotaResult<Option<(VersionNumber, MarkerValue)>> {
173        let Some(bucket) = self.buckets.get(epoch) else {
174            return Ok(None);
175        };
176        let Some(row) = absent_if_dropped(
177            bucket
178                .markers
179                .safe_iter_with_prefix_reversed(object_id)
180                .next()
181                .transpose(),
182        )?
183        else {
184            return Ok(None);
185        };
186        let (key, marker) = row;
187        // The iterator bounds cannot yield another object id.
188        debug_assert_eq!(key.0, *object_id);
189        Ok(Some((key.1, marker)))
190    }
191
192    /// The rows of one of this store's column families, for `iota-tool dump`,
193    /// which cannot reach them through the perpetual table struct. `None` when
194    /// `cf_name` is not one of them.
195    pub fn dump_column_family(
196        db: &Arc<Database>,
197        cf_name: &str,
198        page_size: u16,
199        page_number: usize,
200    ) -> Result<Option<BTreeMap<String, String>>, TypedStoreError> {
201        if bucket_cf_epoch(MARKERS_CF_PREFIX, cf_name).is_some() {
202            let bucket = EpochMarkersBucket::reopen(db, cf_name)?;
203            bucket.markers.try_catch_up_with_primary()?;
204            return dump_page(
205                dump_rows("", bucket.markers.safe_iter()),
206                page_size,
207                page_number,
208            )
209            .map(Some);
210        }
211        if cf_name == EARLIEST_RETAINED_CF {
212            return dump_retention_floor(db, cf_name, page_size, page_number).map(Some);
213        }
214        Ok(None)
215    }

217    /// Moves the markers left in the flat `object_per_epoch_marker_table` into
218    /// the bucket of `epoch`, and deletes the rows of earlier epochs, which
219    /// nothing reads. Safe to rerun after an interrupted run.
220    ///
221    /// Call this before starting any service: until it returns, a marker
222    /// written before the upgrade is unreachable, and a missed marker lets a
223    /// receive or a delete happen twice.
224    // TODO(https://github.com/iotaledger/iota/issues/12712): remove once migrated.
225    pub fn migrate_flat_markers(
226        &self,
227        flat: &DBMap<(EpochId, ObjectKey), MarkerValue>,
228        epoch: EpochId,
229    ) -> IotaResult<()> {
230        let mut progress =
231            ProgressLogger::new("epoch marker migration", "markers", flat.estimated_len()?);
232        // Resume above the last key read: restarting at the front would walk
233        // the tombstones of every row deleted so far again.
234        let mut resume_above = None;
235        loop {
236            let mut moved = Vec::new();
237            let mut keys = Vec::new();
238            let slice = match resume_above {
239                Some(last) => flat.safe_range_iter((Bound::Excluded(last), Bound::Unbounded)),
240                None => flat.safe_iter(),
241            };
242            for row in slice.take(KEYS_PER_SLICE) {
243                let ((row_epoch, key), marker) = row?;
244                if row_epoch == epoch {
245                    moved.push((key, marker));
246                }
247                keys.push((row_epoch, key));
248            }
249            if keys.is_empty() {
250                progress.finish();
251                return Ok(());
252            }
253            let mut batch = flat.batch();
254            if !moved.is_empty() {
255                let bucket = self.ensure(epoch)?;
256                batch.insert_batch(&bucket.markers, moved)?;
257            }
258            let read = keys.len();
259            resume_above = keys.last().copied();
260            batch.delete_batch(flat, keys)?;
261            batch.write()?;
262            progress.advance(read as u64);
263        }
264    }
265
266    /// Drops every bucket below the epoch being entered, after making sure
267    /// that epoch has one. Returns the earliest epoch still retained and how
268    /// many buckets were dropped.
269    ///
270    /// Call only while execution is halted, as at reconfiguration: the drops
271    /// take a write lock the execution path's reads contend on.
272    pub fn expire(&self, new_epoch: EpochId) -> IotaResult<Option<PrunedBuckets>> {
273        self.ensure(new_epoch)?;
274        self.buckets
275            .prune(new_epoch, HISTORIC_EPOCHS_TO_RETAIN, |_, _| Ok(()))
276            .map_err(|e| IotaError::Storage(e.to_string()))
277    }
278}
