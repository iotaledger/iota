17// Versioning process:
18//
19// Object storage versioning is done lazily (at read time) - therefore we must
20// always preserve the code for reading the very first storage version. For all
21// versions, a migration function
22//
23//   f(V_n) -> V_(n+1)
24//
25// must be defined. This way we can iteratively migrate the very oldest version
26// to the very newest version at any point in the future.
27//
28// To change the format of the object table value types (StoreObject and
29// StoreMoveObject), use the following process:
30// - Add a new variant to the enum to store the new version type.
31// - Define `From<StoreObjectV{N}> for StoreObjectV{N+1}` to update older
32//   versions, and extend `migrate()` to chain `V{N}` -> `V{N+1}`.
33// - Advance `pub type StoreObject = StoreObjectV{N+1}` and update
34//   `From<StoreObject> for StoreObjectWrapper` to wrap the new variant.
35// - Update `get_store_object` (and any other writers) to construct the new
36//   value type directly.
37
38/// Enum wrapper for versioning
39#[derive(Eq, PartialEq, Debug, Clone, Deserialize, Serialize, Hash)]
40pub enum StoreObjectWrapper {
41    V1(StoreObjectV1),
42    V2(StoreObjectV2),
43}
44
45// always points to latest version.
46pub type StoreObject = StoreObjectV2;
47
48impl StoreObjectWrapper {
49    pub fn migrate(self) -> Self {
50        match self {
51            Self::V1(v1) => Self::V2(v1.into()),
52            v2 @ Self::V2(_) => v2,
53        }
54    }
55
56    // Always returns the most recent version. Older versions are migrated to the
57    // latest version at read time, so there is never a need to access older
58    // versions.
59    pub fn inner(&self) -> &StoreObject {
60        match self {
61            Self::V1(_) => {
62                panic!("object should have been migrated to latest version at read time")
63            }
64            Self::V2(v2) => v2,
65        }
66    }
67    pub fn into_inner(self) -> StoreObject {
68        match self {
69            Self::V1(_) => {
70                panic!("object should have been migrated to latest version at read time")
71            }
72            Self::V2(v2) => v2,
73        }
74    }
75}
76
77impl From<StoreObject> for StoreObjectWrapper {
78    fn from(o: StoreObject) -> Self {
79        StoreObjectWrapper::V2(o)
80    }
81}

90/// Forked version of [`iota_types::object::Object`]
91/// Used for efficient storing of move objects in the database
92#[derive(Eq, PartialEq, Debug, Clone, Deserialize, Serialize, Hash)]
93pub struct StoreObjectValue {
94    pub data: StoreData,
95    pub owner: Owner,
96    pub previous_transaction: TransactionDigest,
97    pub storage_rebate: u64,
98}

100#[derive(Eq, PartialEq, Debug, Clone, Deserialize, Serialize, Hash)]
101pub enum StoreObjectV2 {
102    Value(Box<StoreObjectValueV2>),
103    Deleted,
104    Wrapped,
105}
106
107/// V2 of [`StoreObjectValue`]. Adds `previous_transaction_checkpoint`,
108/// the checkpoint sequence number that contained `previous_transaction`.
109#[derive(Eq, PartialEq, Debug, Clone, Deserialize, Serialize, Hash)]
110pub struct StoreObjectValueV2 {
111    pub data: StoreData,
112    pub owner: Owner,
113    pub previous_transaction: TransactionDigest,
114    /// Checkpoint sequence number of the checkpoint that contained
115    /// `previous_transaction`. Only needed for the formal snapshot writer. The
116    /// snapshot writer refuses to publish if any live-set row carries
117    /// `None`; a node that wants to publish V2 snapshots must therefore
118    /// have synced from genesis under V2 or been started from a V2 state so
119    /// that no lifted-V1 rows are present.
120    pub previous_transaction_checkpoint: Option<CheckpointSequenceNumber>,
121    pub storage_rebate: u64,
122}
123
124impl From<StoreObjectV1> for StoreObjectV2 {
125    fn from(v1: StoreObjectV1) -> Self {
126        match v1 {
127            StoreObjectV1::Value(v1_value) => Self::Value(Box::new(StoreObjectValueV2 {
128                data: v1_value.data,
129                owner: v1_value.owner,
130                previous_transaction: v1_value.previous_transaction,
131                // Pre-V2 rows never recorded the containing checkpoint;
132                // there is no way to recover it. The snapshot writer
133                // rejects any row carrying `None` rather than emit an
134                // unknown value into the wire format.
135                previous_transaction_checkpoint: None,
136                storage_rebate: v1_value.storage_rebate,
137            })),
138            StoreObjectV1::Deleted => Self::Deleted,
139            StoreObjectV1::Wrapped => Self::Wrapped,
140        }
141    }
142}

144/// Forked version of [`iota_sdk_types::ObjectData`]
145/// Adds extra enum value `IndirectObject`, which represents a reference to an
146/// object stored separately
147#[derive(Eq, PartialEq, Debug, Clone, Deserialize, Serialize, Hash)]
148pub enum StoreData {
149    Move(MoveStruct),
150    Package(MovePackage),
151    IndirectObjectDeprecated,
152    Coin(u64),
153}

155/// Build a `StoreObjectWrapper` for a newly written object version. The caller
156/// supplies the checkpoint sequence number that contains the transaction whose
157/// effects produced this object version.
158pub fn get_store_object(
159    object: Object,
160    previous_transaction_checkpoint: Option<CheckpointSequenceNumber>,
161) -> StoreObjectWrapper {
162    let object = object.into_inner();
163
164    let data = match object.data {
165        ObjectData::Package(package) => StoreData::Package(package),
166        ObjectData::Struct(move_obj) => {
167            if move_obj.struct_tag().is_gas_coin() {
168                StoreData::Coin(
169                    Coin::from_bcs_bytes(move_obj.contents())
170                        .expect("failed to deserialize coin")
171                        .balance
172                        .value(),
173                )
174            } else {
175                StoreData::Move(move_obj)
176            }
177        }
178    };
179    let store_object = StoreObjectValueV2 {
180        data,
181        owner: object.owner,
182        previous_transaction: object.previous_transaction,
183        previous_transaction_checkpoint,
184        storage_rebate: object.storage_rebate,
185    };
186    StoreObject::Value(Box::new(store_object)).into()
187}

189pub(crate) fn try_construct_object(
190    object_key: &ObjectKey,
191    store_object: StoreObjectValueV2,
192) -> Result<Object, IotaError> {
193    let data = match store_object.data {
194        StoreData::Move(object) => ObjectData::Struct(object),
195        StoreData::Package(package) => ObjectData::Package(package),
196        StoreData::Coin(balance) => ObjectData::Struct(MoveStruct::new_from_execution_with_limit(
197            StructTag::new_gas_coin(),
198            object_key.1,
199            bcs::to_bytes(&(object_key.0, balance)).expect("serialization failed"),
200            u64::MAX,
201        )?),
202        _ => {
203            return Err(IotaError::Storage(
204                "corrupted field: inconsistent object representation".to_string(),
205            ));
206        }
207    };
208
209    Ok(ObjectInner {
210        data,
211        owner: store_object.owner,
212        previous_transaction: store_object.previous_transaction,
213        storage_rebate: store_object.storage_rebate,
214    }
215    .into())
216}

233    #[test]
234    fn migrate_v1_value_lifts_with_none_checkpoint() {
235        let v1 = v1_value();
236        let wrapped = StoreObjectWrapper::V1(StoreObjectV1::Value(Box::new(v1.clone()))).migrate();
237        let StoreObjectWrapper::V2(StoreObjectV2::Value(v2_value)) = wrapped else {
238            panic!("expected V2(Value), got {wrapped:?}");
239        };
240        assert_eq!(v2_value.data, v1.data);
241        assert_eq!(v2_value.owner, v1.owner);
242        assert_eq!(v2_value.previous_transaction, v1.previous_transaction);
243        assert_eq!(v2_value.storage_rebate, v1.storage_rebate);
244        // Pre-V2 rows never recorded the checkpoint; lift surfaces that as
245        // `None`. The snapshot writer rejects such rows at the boundary.
246        assert_eq!(v2_value.previous_transaction_checkpoint, None);
247    }

264    #[test]
265    fn v1_wrapper_bcs_stays_at_discriminant_zero() {
266        // Critical invariant: V1 must remain at BCS discriminant 0 so existing
267        // on-disk V1 rows decode correctly after V2 was added to the enum.
268        // Reordering StoreObjectWrapper variants would break this.
269        let v1 = StoreObjectWrapper::V1(StoreObjectV1::Value(Box::new(v1_value())));
270        let bytes = bcs::to_bytes(&v1).unwrap();
271        assert_eq!(bytes[0], 0, "V1 must remain at BCS discriminant 0");
272
273        let decoded: StoreObjectWrapper = bcs::from_bytes(&bytes).unwrap();
274        assert!(matches!(decoded, StoreObjectWrapper::V1(_)));
275    }

313    /// Locks the BCS field-order layout of `StoreObjectValueV2` against a
314    /// golden byte vector. Reordering or renaming any field would silently
315    /// corrupt every on-disk V2 row; this test fails loudly on any such
316    /// change.
317    #[test]
318    fn store_object_value_v2_bcs_layout_is_locked() {
319        // `Some(seq)` case - the production write path.
320        let v2 = StoreObjectValueV2 {
321            data: StoreData::Coin(0x0102_0304_0506_0708),
322            owner: Owner::Immutable,
323            previous_transaction: TransactionDigest::ZERO,
324            previous_transaction_checkpoint: Some(0x1011_1213_1415_1617),
325            storage_rebate: 0x2021_2223_2425_2627,
326        };
327        let bytes = bcs::to_bytes(&v2).unwrap();
328
329        let mut golden: Vec<u8> = Vec::new();
330        // `StoreData::Coin(u64)` - the four-variant `StoreData` enum places
331        // `Coin` at variant tag 3, followed by the u64 in little-endian.
332        golden.push(0x03);
333        golden.extend_from_slice(&0x0102_0304_0506_0708u64.to_le_bytes());
334        // `Owner::Immutable` - Owner uses a custom serializer that maps to
335        // `ReadableOwner` (AddressOwner=0, ObjectOwner=1, Shared=2,
336        // Immutable=3), so Immutable encodes as a single tag byte.
337        golden.push(0x03);
338        // `TransactionDigest::ZERO` - Digest's binary BCS form is
339        // length-prefixed: ULEB128 length 32 (=`0x20`) + 32 zero bytes.
340        golden.push(0x20);
341        golden.extend_from_slice(&[0u8; 32]);
342        // `previous_transaction_checkpoint: Option<u64>` - `Some` discriminant
343        // (0x01) + u64 in little-endian.
344        golden.push(0x01);
345        golden.extend_from_slice(&0x1011_1213_1415_1617u64.to_le_bytes());
346        // `storage_rebate: u64` - little-endian.
347        golden.extend_from_slice(&0x2021_2223_2425_2627u64.to_le_bytes());
348
349        assert_eq!(
350            bytes, golden,
351            "StoreObjectValueV2 BCS layout changed; introduce a new StoreObject \
352             version rather than mutating V2"
353        );
354
355        // `None` case - rows lifted from pre-V2 on-disk format.
356        let v2_none = StoreObjectValueV2 {
357            previous_transaction_checkpoint: None,
358            ..v2
359        };
360        let bytes_none = bcs::to_bytes(&v2_none).unwrap();
361        let mut golden_none: Vec<u8> = Vec::new();
362        golden_none.push(0x03);
363        golden_none.extend_from_slice(&0x0102_0304_0506_0708u64.to_le_bytes());
364        golden_none.push(0x03);
365        golden_none.push(0x20);
366        golden_none.extend_from_slice(&[0u8; 32]);
367        // `previous_transaction_checkpoint: Option<u64>` - `None` discriminant.
368        golden_none.push(0x00);
369        golden_none.extend_from_slice(&0x2021_2223_2425_2627u64.to_le_bytes());
370        assert_eq!(bytes_none, golden_none);
371    }
