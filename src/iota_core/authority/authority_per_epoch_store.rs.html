203/// Container for congestion control parameters commonly used in
204/// `SharedObjectCongestionTracker` and `SuggestedGasPriceCalculator`.
205/// It can be initialized from a `ProtocolConfig` instance.
206#[derive(Debug, Clone, PartialEq, Eq)]
207pub(crate) struct CongestionControlParameters {
208    /// Controls the behavior of per-object congestion control. This
209    /// field determines how the estimated execution duration of a
210    /// transaction is calculated.
211    per_object_congestion_control_mode: PerObjectCongestionControlMode,
212
213    /// This field determines how the start time of a transaction should be
214    /// assigned. If `true`, the tracker will assign the start time according
215    /// to the minimum free execution slot for a transaction over all its
216    /// shared objects. If `false`, the tracker will assign the start time
217    /// according to the maximum end time of the occupied execution slots
218    /// for a transaction over all its shared objects.
219    congestion_control_min_free_execution_slot: bool,
220
221    /// Maximum execution duration per shared object per commit. If `None`,
222    /// it means that shared-object congestion control is disabled.
223    max_execution_duration_per_commit: Option<ExecutionTime>,
224
225    /// Maximum amount that is allowed to overshoot
226    /// `max_execution_duration_per_commit`. If `None`, it means that
227    /// congestion limit overshoot is disabled.
228    max_congestion_limit_overshoot_per_commit: Option<ExecutionTime>,
229
230    /// Maximum number of transactions from a commit that may execute
231    /// concurrently (the execution-worker pool size). `Some(n)` activates
232    /// execution-worker congestion control; `None` disables it.
233    max_concurrent_execution_workers: Option<u16>,
234
235    /// Maximum gas price that can be set in transactions. This field
236    /// is only used in `SuggestedGasPriceCalculator` to prevent
237    /// suggesting feedback gas price larger this value.
238    max_gas_price: u64,
239
240    /// Whether to use congestion limit overshoot in the gas price feedback
241    /// mechanism, i.e., this is only used in `SuggestedGasPriceCalculator`.
242    use_congestion_limit_overshoot_in_gas_price_feedback_mechanism: bool,
243
244    /// Whether to use a separate gas price feedback mechanism for transactions
245    /// using randomness.
246    use_separate_gas_price_feedback_mechanism_for_randomness: bool,
247}

310    /// Depending on the `PerObjectCongestionControlMode`, different metrics are
311    /// used to approximate the expected execution duration of a transaction.
312    /// The expected execution duration is what is used to schedule transactions
313    /// and allocate resources based on how many transactions can be executed
314    /// from a given consensus commit.
315    pub(super) fn get_estimated_execution_duration(
316        &self,
317        transaction: &VerifiedExecutableTransaction,
318    ) -> ExecutionTime {
319        match self.per_object_congestion_control_mode {
320            PerObjectCongestionControlMode::None => 0,
321            PerObjectCongestionControlMode::TotalGasBudget => transaction.gas_budget(),
322            PerObjectCongestionControlMode::TotalTxCount => 1,
323        }
324    }

366    /// using randomness.

368        self.use_separate_gas_price_feedback_mechanism_for_randomness

371    /// Get effective congestion limit per commit, i.e.,
372    /// `max_execution_duration_per_commit` plus
373    /// `max_congestion_limit_overshoot_per_commit`.
374    /// Returns `None` if `max_execution_duration_per_commit` is not set,
375    /// i.e., shared-object congestion control is disabled.
376    pub(super) fn get_effective_congestion_limit_per_commit(&self) -> Option<ExecutionTime> {
377        self.max_execution_duration_per_commit
378            .map(|max_execution_duration_per_commit| {
379                max_execution_duration_per_commit.saturating_add(
380                    // If `max_congestion_limit_overshoot_per_commit` is not set,
381                    // add 0 to `max_execution_duration_per_commit`, that is,
382                    // ignore congestion limit overshoot.
383                    self.max_congestion_limit_overshoot_per_commit
384                        .unwrap_or_default(),
385                )
386            })
387    }

443/// The reason a consensus transaction was cancelled rather than successfully
444/// executed.
445///
446/// Cancelled transactions still pass through the execution engine to unlock
447/// owned objects, but they are not executed as normally. The reason is used by
448/// `SharedObjVerManager` when assigning cancellation versions to determine
449/// the appropriate cancellation behavior (e.g., whether to include a suggested
450/// gas price in the cancellation error).
451///
452/// The legacy name (in the certificate era) was
453/// `CancelConsensusCertificateReason`. It is renamed to
454/// `CancelConsensusTransactionReason` because this type now (in the
455/// P-COOL flow) covers more consensus transaction kinds, not just
456/// certificates. The renaming is safe and backward-compatible since this is a
457/// fully internal type.
458pub enum CancelConsensusTransactionReason {
459    /// Transaction was cancelled due to congestion: either on objects it
460    /// touches, or on the execution-worker pool.
461    Congested {
462        /// IDs of the congested objects the transaction touches. Empty when
463        /// the execution-worker pool, rather than any object, was congested
464        /// and the transaction has no shared inputs.
465        congested_objects: Vec<ObjectId>,
466
467        /// Optional suggested gas price from the gas price feedback
468        /// mechanism.
469        suggested_gas_price: Option<u64>,
470    },
471
472    /// Transaction was cancelled due to randomness unavailable.
473    DkgFailed,
474}
475
476/// The outcome of processing a single consensus transaction.
477///
478/// This is returned by `process_consensus_transaction()` and
479/// `process_consensus_system_transaction()` to tell the consensus commit
480/// handler how to handle each transaction: schedule it for execution,
481/// defer it to a future commit, cancel it, or ignore it.
482///
483/// The legacy name (in the certificate era) was `ConsensusCertificateResult`.
484/// It is renamed to `ConsensusTransactionResult` because this type now (in
485/// the P-COOL flow) covers all consensus transaction kinds, not just
486/// certificates. The renaming is safe and backward-compatible since this is a
487/// fully internal type.
488pub enum ConsensusTransactionResult {
489    /// The consensus message was ignored (e.g. because it has already been
490    /// processed).
491    Ignored,
492
493    /// The transaction is scheduled for execution (can be a user tx or a
494    /// system tx) with start_time. The start_time is an ExecutionTime assigned
495    /// by the SharedObjectCongestionTracker and it implies its
496    /// execution order. Before a batch of scheduled transactions are sent
497    /// for execution, they will be ordered by their start_time
498    /// (ascendingly). start_times of shared object transactions imply
499    /// causal ordering. Owned object transactions will always have
500    /// start_time 0, meaning they are not dependent on another transaction
501    /// and they will not wait for another transaction.
502    Scheduled {
503        transaction: VerifiedExecutableTransaction,
504        start_time: ExecutionTime,
505    },
506
507    /// The transaction should be re-processed at a future commit, specified by
508    /// `deferral_key`. If the gas price feedback is enabled,
509    /// `suggested_gas_price` is `Some(...)` and indicates a gas price that
510    /// the transaction would need to pay to be scheduled in a consensus
511    /// commit. If the feedback mechanism is not enabled and for
512    /// transactions deferred due to "randomness not available",
513    /// the `suggested_gas_price` price field will be set to `None`.
514    Deferred {
515        deferral_key: DeferralKey,
516        suggested_gas_price: Option<u64>,
517    },
518
519    /// A message was processed which updates randomness state.
520    RandomnessConsensusMessage,
521
522    /// Everything else, e.g. AuthorityCapabilities, CheckpointSignatures, etc.
523    ConsensusMessage,
524
525    /// A system message in consensus was ignored (e.g. because of end of
526    /// epoch).
527    IgnoredSystem,
528
529    /// A will-be-cancelled transaction. It'll still go through execution engine
530    /// (but not be executed), unlock any owned objects, and return
531    /// corresponding cancellation error according to
532    /// `CancelConsensusTransactionReason`.
533    Cancelled(
534        (
535            VerifiedExecutableTransaction,
536            CancelConsensusTransactionReason,
537        ),
538    ),
539}

1170        parent_path: &Path,
1171        db_options: Option<Options>,
1172        metrics: Arc<EpochMetrics>,
1173        epoch_start_configuration: EpochStartConfiguration,
1174        backing_package_store: Arc<dyn BackingPackageStore + Send + Sync>,
1175        cache_metrics: Arc<ResolverMetrics>,
1176        signature_verifier_metrics: Arc<SignatureVerifierMetrics>,
1177        expensive_safety_check_config: &ExpensiveSafetyCheckConfig,
1178        chain: (ChainIdentifier, Chain),
1179        highest_executed_checkpoint: CheckpointSequenceNumber,
1180    ) -> IotaResult<Arc<Self>> {
1181        let current_time = Instant::now();
1182        let epoch_id = committee.epoch;
1183        metrics.current_epoch.set(epoch_id as i64);
1184        metrics
1185            .current_voting_right
1186            .set(committee.weight(&name) as i64);
1187
1188        let tables = AuthorityEpochTables::open(epoch_id, parent_path, db_options.clone());
1189        let end_of_publish =
1190            StakeAggregator::from_iter(committee.clone(), tables.end_of_publish.safe_iter())?;
1191        let reconfig_state = tables
1192            .load_reconfig_state()
1193            .expect("Load reconfig state at initialization cannot fail");
1194
1195        // Populate the in-memory cache of signed effects digests from disk. The set is
1196        // bounded by uncommitted (not-yet-checkpoint-finalized) transactions in the
1197        // current epoch.
1198        let signed_effects_digests_cache = DashMap::new();
1199        for item in tables.signed_effects_digests.safe_iter() {
1200            let (tx_digest, effects_digest) = item?;
1201            signed_effects_digests_cache.insert(tx_digest, effects_digest);
1202        }
1203
1204        let epoch_alive_token = CancellationToken::new();
1205        // NOTE: `pending_consensus_certificates` will be built after `protocol_config`
1206        // is available below.
1207        let pending_consensus_transactions = tables.get_all_pending_consensus_transactions()?;
1208        assert_eq!(
1209            epoch_start_configuration.epoch_start_state().epoch(),
1210            epoch_id
1211        );
1212
1213        let epoch_start_configuration = Arc::new(epoch_start_configuration);
1214        info!("epoch flags: {:?}", epoch_start_configuration.flags());
1215        let protocol_version = epoch_start_configuration
1216            .epoch_start_state()
1217            .protocol_version();
1218
1219        let chain_from_id = chain.0.chain();
1220        if chain_from_id == Chain::Mainnet || chain_from_id == Chain::Testnet {
1221            assert_eq!(
1222                chain_from_id, chain.1,
1223                "cannot override chain on production networks!"
1224            );
1225        }
1226        info!(
1227            "initializing epoch store from chain id {chain_from_id:?} to chain id {:?}",
1228            chain.1
1229        );
1230
1231        let protocol_config = ProtocolConfig::get_for_version(protocol_version, chain.1);
1232
1233        let pending_consensus_certificates: HashSet<_> = if protocol_config.enable_pcool_flow() {
1234            HashSet::new() // no certificates in P-COOL mode
1235        } else {
1236            pending_consensus_transactions
1237                .iter()
1238                .filter_map(|transaction| {
1239                    if let ConsensusTransactionKind::CertifiedTransaction(certificate) =
1240                        &transaction.kind
1241                    {
1242                        Some(*certificate.digest())
1243                    } else {
1244                        None
1245                    }
1246                })
1247                .collect()
1248        };
1249
1250        let execution_component = ExecutionComponents::new(
1251            &protocol_config,
1252            backing_package_store,
1253            cache_metrics,
1254            expensive_safety_check_config,
1255        );
1256
1257        // Get all active validators and filter out committee members to get
1258        // non-committee validators
1259        let non_committee_validators: BTreeSet<AuthorityName> = epoch_start_configuration
1260            .epoch_start_state()
1261            .get_active_validators()
1262            .into_iter()
1263            .filter_map(|pubkey| AuthorityName::from_bytes(pubkey.as_bytes()).ok())
1264            .filter(|authority_name| !committee.authority_exists(authority_name))
1265            .collect();
1266
1267        let signature_verifier = SignatureVerifier::new(
1268            committee.clone(),
1269            non_committee_validators,
1270            signature_verifier_metrics,
1271            protocol_config.accept_passkey_in_multisig(),
1272            protocol_config.additional_multisig_checks(),
1273        );
1274
1275        let consensus_output_cache = ConsensusOutputCache::new(&tables);
1276
1277        // Seed the quarantine's in-memory overload-notification cache from the
1278        // persisted table. This is the only point we iterate the table; all
1279        // subsequent reads are served from memory.
1280        let cached_overload_notifications: HashMap<AuthorityName, u8> = tables
1281            .authority_overload_notifications
1282            .safe_iter()
1283            .collect::<Result<HashMap<AuthorityName, u8>, _>>()
1284            .expect("AuthorityEpochTables should contain valid overload notifications");
1285
1286        // Seed the quarantine's deny-rule-proposal cache from the persisted
1287        // table and derive the active rule set. A mid-epoch restart resumes
1288        // with the rules from all flushed commits; proposals from unflushed
1289        // commits are re-recorded by consensus replay.
1290        let cached_deny_rule_proposals: BTreeMap<AuthorityName, TransactionDenyRuleProposal> =
1291            tables
1292                .deny_rule_proposals
1293                .safe_iter()
1294                .collect::<Result<BTreeMap<_, _>, _>>()
1295                .expect("AuthorityEpochTables should contain valid deny rule proposals");
1296        let deny_rule_mirror = Self::initial_deny_rule_mirror(&tables, &epoch_start_configuration)?;
1297        let active_transaction_deny_rules = ArcSwap::from_pointee(union_deny_rule_sets(
1298            Self::compute_active_transaction_deny_rules(&cached_deny_rule_proposals, &committee),
1299            &deny_rule_mirror,
1300        ));
1301        let mirrored_transaction_deny_rules = ArcSwap::from_pointee(deny_rule_mirror);
1302
1303        let committee_size = committee.num_members();
1304        let report_version = MisbehaviorReportVersion::from_protocol(&protocol_config);
1305        let misbehavior_monitor = MisbehaviorMonitor::new(name, report_version, committee_size);
1306        let report_aggregator = ReportAggregator::new(report_version, committee_size);
1307        report_aggregator
1308            .restore_from_tables(&tables)
1309            .expect("AuthorityEpochTables should contain valid ReportAggregator state");
1310        let voting_power = committee.members().map(|(_, v)| *v).collect::<Vec<u64>>();
1311        let scoreboard = Scoreboard::new(voting_power, &protocol_config);
1312        // Prime the published score vector from any restored aggregator state.
1313        // Without this, a freshly constructed Scoreboard starts at [MAX_SCORE;
1314        // committee_size], so the next no-report commit (which both restored
1315        // and never-restarted validators skip) would publish different vectors
1316        // across the network. `update_scores` is a no-op when the aggregator
1317        // is empty (no `received_metrics` rows restored), so this is also
1318        // correct for the fresh-epoch path.
1319        if protocol_config.calculate_validator_scores() {
1320            scoreboard.update_scores(&report_aggregator);
1321        }
1322
1323        let s = Arc::new(Self {
1324            name,
1325            committee,
1326            protocol_config,
1327            tables: ArcSwapOption::new(Some(Arc::new(tables))),
1328            consensus_output_cache,
1329            consensus_quarantine: RwLock::new(ConsensusOutputQuarantine::new(
1330                highest_executed_checkpoint,
1331                cached_overload_notifications,
1332                cached_deny_rule_proposals,
1333                metrics.clone(),
1334            )),
1335            parent_path: parent_path.to_path_buf(),
1336            db_options,
1337            reconfig_state_mem: RwLock::new(reconfig_state),
1338            epoch_alive_token,
1339            user_certs_closed_notify: NotifyOnce::new(),
1340            epoch_alive: tokio::sync::RwLock::new(true),
1341            consensus_notify_read: NotifyRead::new(),
1342            executed_transactions_to_checkpoint_notify_read: NotifyRead::new(),
1343            signature_verifier,
1344            checkpoint_state_notify_read: NotifyRead::new(),
1345            running_root_notify_read: NotifyRead::new(),
1346            executed_digests_notify_read: NotifyRead::new(),
1347            signed_effects_digests_cache,
1348            dropped_tx_status_cache: super::dropped_tx_status_cache::DroppedTxStatusCache::new(),
1349            active_transaction_deny_rules,
1350            mirrored_transaction_deny_rules,
1351            soft_locks: OnceCell::new(),
1352            end_of_publish: Mutex::new(end_of_publish),
1353            pending_consensus_certificates: RwLock::new(pending_consensus_certificates),
1354            mutex_table: MutexTable::new(MUTEX_TABLE_SIZE),
1355            version_assignment_mutex_table: MutexTable::new(MUTEX_TABLE_SIZE),
1356            epoch_open_time: current_time,
1357            epoch_close_time: Default::default(),
1358            metrics,
1359            epoch_start_configuration,
1360            execution_component,
1361            chain,
1362            randomness_manager: OnceCell::new(),
1363            randomness_reporter: OnceCell::new(),
1364            misbehavior_monitor,
1365            report_aggregator,
1366            scoreboard,
1367        });
1368
1369        s.update_buffer_stake_metric();
1370        Ok(s)
1371    }

1373    pub fn tables(&self) -> IotaResult<Arc<AuthorityEpochTables>> {
1374        match self.tables.load_full() {
1375            Some(tables) => Ok(tables),
1376            None => Err(IotaError::EpochEnded(self.epoch())),
1377        }
1378    }
1379
1380    // Ideally the epoch tables handle should have the same lifetime as the outer
1381    // AuthorityPerEpochStore, and this function should be unnecessary. But
1382    // unfortunately, Arc<AuthorityPerEpochStore> outlives the
1383    // epoch significantly right now, so we need to manually release the tables to
1384    // release its memory usage.
1385    pub fn release_db_handles(&self) {
1386        let epoch = self.epoch();
1387        // Swap out the tables Arc so no new references can be obtained via tables().
1388        if let Some(tables) = self.tables.swap(None) {
1389            // strong_count() == 1 means we hold the only reference and the
1390            // RocksDB handles will be freed when this Arc is dropped below.
1391            // A higher count means some code path still holds a reference,
1392            // which delays freeing the underlying RocksDB memory.
1393            let strong_count = Arc::strong_count(&tables);
1394            if strong_count > 1 {
1395                warn!(
1396                    epoch,
1397                    strong_count,
1398                    "Epoch tables Arc has lingering references after release_db_handles(). \
1399                     RocksDB memory for epoch {} will not be freed until all {} references \
1400                     are dropped.",
1401                    epoch,
1402                    strong_count
1403                );
1404            } else {
1405                debug!(epoch, "Epoch tables released cleanly for epoch {}", epoch);
1406            }
1407            // tables Arc is dropped here, freeing the RocksDB handles if
1408            // strong_count == 1
1409        }
1410    }

1735    pub fn insert_effects_digest_and_signature(
1736        &self,
1737        tx_digest: &TransactionDigest,
1738        effects_digest: &TransactionEffectsDigest,
1739        effects_signature: &AuthoritySignInfo,
1740    ) -> IotaResult {
1741        // The entry guard serializes concurrent signers of the same transaction, so at
1742        // most one effects digest can ever be recorded per transaction within an
1743        // epoch.
1744        match self.signed_effects_digests_cache.entry(*tx_digest) {
1745            dashmap::mapref::entry::Entry::Occupied(entry) => {
1746                if entry.get() != effects_digest {
1747                    return Err(IotaError::GenericAuthority {
1748                        error: format!(
1749                            "Refusing to report effects for transaction {tx_digest}: effects \
1750                             digest {effects_digest} differs from previously signed effects \
1751                             digest {}",
1752                            entry.get()
1753                        ),
1754                    });
1755                }
1756            }
1757            dashmap::mapref::entry::Entry::Vacant(entry) => {
1758                entry.insert(*effects_digest);
1759            }
1760        }
1761        let tables = self.tables()?;
1762        let mut batch = tables.effects_signatures.batch();
1763        batch.insert_batch(&tables.effects_signatures, [(tx_digest, effects_signature)])?;
1764        batch.insert_batch(
1765            &tables.signed_effects_digests,
1766            [(tx_digest, effects_digest)],
1767        )?;
1768        batch.write()?;
1769        Ok(())
1770    }

1976        &self,
1977        checkpoint: &CheckpointSummary,
1978        digests: &[TransactionDigest],
1979    ) -> IotaResult<()> {
1980        let tables = match self.tables() {
1981            Ok(tables) => tables,
1982            // After Epoch ends, it is no longer necessary to remove pending transactions
1983            // because the table will not be used anymore and be deleted eventually.
1984            Err(IotaError::EpochEnded(_)) => return Ok(()),
1985            Err(e) => return Err(e),

1988

1990        // we don't need to worry about equivocating.

1997        batch.write()?;

2000            self.signed_effects_digests_cache.remove(digest);
2001        }
2002
2003        self.consensus_output_cache
2004            .remove_executed_in_epoch(digests);

2092    // For each id in objects_to_init, return the next version for that id as
2093    // recorded in the next_shared_object_versions table.
2094    //
2095    // If any ids are missing, then we need to initialize the table. We first check
2096    // if a previous version of that object has been written. If so, then the
2097    // object was written in a previous epoch, and we initialize
2098    // next_shared_object_versions to that value. If no version of the
2099    // object has yet been written, we initialize the object to the initial version
2100    // recorded in the transaction (which is a function of the lamport version
2101    // computation of the transaction that created the shared object originally
2102    // - which transaction may not yet have been executed on this node).
2103    //
2104    // Because all paths that assign shared versions for a shared object transaction
2105    // call this function, it is impossible for parent_sync to be updated before
2106    // this function completes successfully for each affected object id.
2107    pub(crate) fn get_or_init_next_object_versions(
2108        &self,
2109        objects_to_init: &[(ObjectId, Version)],
2110        cache_reader: &dyn ObjectCacheRead,
2111    ) -> IotaResult<HashMap<ObjectId, Version>> {
2112        // get_or_init_next_object_versions can be called
2113        // from consensus or checkpoint executor,
2114        // so we need to protect version assignment with a critical section
2115        let _locks = self
2116            .version_assignment_mutex_table
2117            .acquire_locks(objects_to_init.iter().map(|(id, _)| *id));
2118        let tables = self.tables()?;
2119
2120        let next_versions = self
2121            .consensus_quarantine
2122            .read()
2123            .get_next_shared_object_versions(&tables, objects_to_init)?;
2124
2125        let uninitialized_objects: Vec<(ObjectId, Version)> = next_versions
2126            .iter()
2127            .zip(objects_to_init)
2128            .filter_map(|(next_version, id_and_version)| match next_version {
2129                None => Some(*id_and_version),
2130                Some(_) => None,
2131            })
2132            .collect();
2133
2134        // The common case is that there are no uninitialized versions - this early
2135        // return will happen every time except the first time an object is
2136        // used in an epoch.
2137        if uninitialized_objects.is_empty() {
2138            // unwrap ok - we already verified that next_versions is not missing any keys.
2139            return Ok(izip!(
2140                objects_to_init.iter().map(|(id, _)| *id),
2141                next_versions.into_iter().map(|v| v.unwrap())
2142            )
2143            .collect());
2144        }
2145
2146        let versions_to_write: Vec<_> = uninitialized_objects
2147            .iter()
2148            .map(|(id, initial_version)| {
2149                // Note: we don't actually need to read from the transaction here, as no writer
2150                // can update object_store until after get_or_init_next_object_versions
2151                // completes.
2152                match cache_reader.get_object(id) {
2153                    Some(obj) => (*id, obj.version()),
2154                    None => (*id, *initial_version),
2155                }
2156            })
2157            .collect();
2158
2159        let ret = izip!(objects_to_init.iter().cloned(), next_versions.into_iter(),)
2160            // take all the previously initialized versions
2161            .filter_map(|(id, next_version)| next_version.map(|v| (id.0, v)))
2162            // add all the versions we're going to write
2163            .chain(versions_to_write.iter().cloned())
2164            .collect();
2165
2166        debug!(
2167            ?versions_to_write,
2168            "initializing next_shared_object_versions"
2169        );
2170        let mut batch = tables.next_shared_object_versions.batch();
2171        batch.insert_batch(&tables.next_shared_object_versions, versions_to_write)?;
2172        batch.write()?;
2173
2174        Ok(ret)
2175    }

2384    #[instrument("transactions_sequencing", level = "trace", skip_all, fields(tx_digest = ?transaction.digest(), scheduling_result = tracing::field::Empty))]
2385    fn try_schedule(
2386        &self,
2387        transaction: &VerifiedExecutableTransaction,
2388        commit_round: CommitRound,
2389        dkg_failed: bool,
2390        generating_randomness: bool,
2391        previously_deferred_tx_digests: &PreviouslyDeferredTransactions,
2392        shared_object_congestion_tracker: &mut SharedObjectCongestionTracker,
2393    ) -> SchedulingResult {
2394        // Defer transaction if it uses randomness but we aren't generating any this
2395        // round. Don't defer if DKG has permanently failed; in that case we
2396        // need to ignore.
2397        if !dkg_failed && !generating_randomness && transaction.uses_randomness() {
2398            let deferred_from_round = previously_deferred_tx_digests
2399                .get(transaction.digest())
2400                .map(|previous_key_suggested_gas_price_pair| {
2401                    previous_key_suggested_gas_price_pair
2402                        .0
2403                        .deferred_from_round()
2404                })
2405                .unwrap_or(commit_round);
2406            let result = SchedulingResult::Defer(
2407                DeferralKey::new_for_randomness(deferred_from_round),

2409            );

2412            tracing::Span::current().record("scheduling_result", tracing::field::debug(&result));
2413            return result;
2414        }

2417            .congestion_control_parameters()
2418            .is_congestion_control_enabled()
2419        {
2420            // Initialise the free execution slots for the objects that are not in the
2421            // tracker.
2422            let shared_input_objects = transaction.shared_input_objects();

2424                .initialize_object_execution_slots(&shared_input_objects);
2425            // Defer transaction if it uses shared objects that are congested.
2426            match shared_object_congestion_tracker.try_schedule(
2427                transaction,
2428                previously_deferred_tx_digests,
2429                commit_round,
2430            ) {

2434                        DeferralReason::SharedObjectCongestion(congested_objects),

2438            }
2439        } else {

2441            SchedulingResult::Schedule(0)
2442        };

2444        // Record the result of scheduling for tracing.
2445        tracing::Span::current().record("scheduling_result", tracing::field::debug(&result));

2692    /// Wait for the given transactions to be included in a checkpoint, with a
2693    /// timeout.
2694    ///
2695    /// Returns a vec parallel to `digests` with `Some((seq, timestamp_ms))` for
2696    /// each transaction that was checkpointed within the timeout, and `None`
2697    /// for any that were not.
2698    ///
2699    /// For transactions not yet checkpointed, the `(seq, timestamp_ms)` comes
2700    /// directly from the notification. For transactions already in the DB,
2701    /// the seq comes from the DB table and `get_timestamp` is called to
2702    /// resolve the timestamp for each unique checkpoint.
2703    pub async fn wait_for_transactions_in_checkpoint_with_timeout(
2704        &self,
2705        digests: &[TransactionDigest],
2706        timeout: std::time::Duration,
2707        mut get_timestamp: impl FnMut(CheckpointSequenceNumber) -> u64,
2708    ) -> IotaResult<Vec<Option<(CheckpointSequenceNumber, u64)>>> {
2709        // First register for notifications and read the DB in afterwards, to avoid a
2710        // race where a transaction gets checkpointed after we read the DB but
2711        // before we register for notifications.
2712        let registrations = self
2713            .executed_transactions_to_checkpoint_notify_read
2714            .register_all(digests);
2715
2716        // Now read the DB to see if any of the transactions were already checkpointed
2717        // before we registered. For any that were, we can resolve the timestamp
2718        // immediately via the callback. For any that weren't, we will wait for the
2719        // notification to fire, which guarantees that the timestamp is included.
2720        let already_checkpointed: Vec<Option<CheckpointSequenceNumber>> = self
2721            .tables()?
2722            .executed_transactions_to_checkpoint
2723            .multi_get(digests)?;
2724
2725        let mut results: Vec<Option<(CheckpointSequenceNumber, u64)>> = vec![None; digests.len()];
2726        let mut pending = Vec::new();
2727
2728        for (i, (registration, existing)) in registrations
2729            .into_iter()
2730            .zip(already_checkpointed)
2731            .enumerate()
2732        {
2733            if let Some(seq) = existing {
2734                // Transaction was already checkpointed before we started waiting.
2735                // The notification has already fired so the registration won't
2736                // resolve. Resolve the timestamp via the callback.
2737                results[i] = Some((seq, get_timestamp(seq)));
2738            } else {
2739                pending.push((i, registration));
2740            }
2741        }
2742
2743        // Await pending notifications concurrently. Collect results as they
2744        // arrive until the deadline.
2745        if !pending.is_empty() {
2746            let deadline = tokio::time::sleep(timeout);
2747            tokio::pin!(deadline);
2748
2749            let mut in_flight: FuturesUnordered<_> = pending
2750                .into_iter()
2751                .map(|(i, reg)| async move { (i, reg.await) })
2752                .collect();
2753
2754            // Drain notifications until either every pending registration has
2755            // resolved or the deadline expires.
2756            //
2757            // `biased;` polls branches in source order, so a notification that
2758            // is already ready always wins over a deadline that fires in the
2759            // same poll — we'd rather report inclusion than time out at the
2760            // boundary.
2761            //
2762            // We match `Some`/`None` explicitly instead of using `else => break`
2763            // because `else` only fires when *all* named branches are disabled.
2764            // With `Some(...) = in_flight.next()` and a still-pending `deadline`,
2765            // a `Ready(None)` from an empty `in_flight` disables only that one
2766            // branch — the loop would then park on `deadline` and wait the full
2767            // timeout even though every notification had already resolved.
2768            loop {
2769                tokio::select! {
2770                    biased;
2771                    next_result = in_flight.next() => {
2772                        match next_result {
2773                            Some((i, seq_and_ts)) => results[i] = Some(seq_and_ts),
2774                            // All pending registrations resolved before the deadline.
2775                            None => break,
2776                        }
2777                    }
2778                    () = &mut deadline => break,
2779                }
2780            }
2781        }
2782
2783        Ok(results)
2784    }

2994    /// Computes the stake-weighted 2f+1 percentile from an in-memory
2995    /// notifications map. Used both by `get_quorum_load_shedding_percentage`
2996    /// and by the consensus commit pre-pass that overlays in-batch
2997    /// `OverloadNotificationV1` entries on top of the persisted state.
2998    pub(crate) fn compute_quorum_load_shedding_percentage(
2999        &self,
3000        notifications: &HashMap<AuthorityName, u8>,
3001    ) -> u8 {
3002        let committee = self.committee();
3003
3004        // Build a vec of (percentage, stake) for every committee member.
3005        // Default to 0% for authorities that haven't reported.
3006        let mut weighted_values: Vec<(u8, StakeUnit)> = committee
3007            .members()
3008            .map(|(authority, stake)| {
3009                let percentage = notifications.get(authority).copied().unwrap_or(0);
3010                (percentage, *stake)
3011            })
3012            .collect();
3013
3014        // Sort ascending by percentage.
3015        weighted_values.sort_by_key(|(percentage, _)| *percentage);
3016
3017        // Walk from lowest to highest, accumulating stake. The value where
3018        // cumulative stake first reaches the quorum threshold (2f+1) is the result.
3019        let quorum_threshold = committee.quorum_threshold();
3020        let mut accumulated_stake: StakeUnit = 0;
3021
3022        for (percentage, stake) in weighted_values {
3023            accumulated_stake += stake;
3024            if accumulated_stake >= quorum_threshold {
3025                return percentage;
3026            }
3027        }
3028
3029        // Unreachable with a valid committee (total stake >= quorum threshold),
3030        // but return 0 as a safe fallback.
3031        0

3181    /// Computes the stake-weighted aggregate of the given deny rule proposals:
3182    /// a deny list entry is active with f+1 supporting stake, a kill switch
3183    /// with 2f+1. Pure function of the proposals and committee, so every
3184    /// validator derives the same set from the same consensus state.
3185    pub(crate) fn compute_active_transaction_deny_rules(
3186        proposals: &BTreeMap<AuthorityName, TransactionDenyRuleProposal>,
3187        committee: &Committee,
3188    ) -> DenyRuleSet {
3189        let mut address_stakes: BTreeMap<Address, StakeUnit> = BTreeMap::new();
3190        let mut object_stakes: BTreeMap<ObjectId, StakeUnit> = BTreeMap::new();
3191        let mut package_stakes: BTreeMap<ObjectId, StakeUnit> = BTreeMap::new();
3192
3193        for (authority, proposal) in proposals {
3194            // Non-members weigh 0, so stale proposals cannot contribute.
3195            let stake = committee.weight(authority);
3196            let rules = &proposal.proposed_rules;
3197            for address in &rules.denied_addresses {
3198                *address_stakes.entry(*address).or_default() += stake;
3199            }
3200            for object in &rules.denied_objects {
3201                *object_stakes.entry(*object).or_default() += stake;
3202            }
3203            for package in &rules.denied_packages {
3204                *package_stakes.entry(*package).or_default() += stake;
3205            }
3206        }
3207
3208        fn activated<T: Ord>(stakes: BTreeMap<T, StakeUnit>, threshold: StakeUnit) -> BTreeSet<T> {
3209            stakes
3210                .into_iter()
3211                .filter(|(_, stake)| *stake >= threshold)
3212                .map(|(entry, _)| entry)
3213                .collect()
3214        }
3215
3216        let validity_threshold = committee.validity_threshold();
3217        let quorum_threshold = committee.quorum_threshold();
3218        let switch_active = |enabled: fn(&DenyRuleSet) -> bool| {
3219            proposals
3220                .iter()
3221                .filter(|(_, proposal)| enabled(&proposal.proposed_rules))
3222                .map(|(authority, _)| committee.weight(authority))
3223                .sum::<StakeUnit>()
3224                >= quorum_threshold
3225        };
3226        DenyRuleSet {
3227            denied_addresses: activated(address_stakes, validity_threshold),
3228            denied_objects: activated(object_stakes, validity_threshold),
3229            denied_packages: activated(package_stakes, validity_threshold),
3230            package_publish_disabled: switch_active(|rules| rules.package_publish_disabled),
3231            package_upgrade_disabled: switch_active(|rules| rules.package_upgrade_disabled),
3232            shared_object_disabled: switch_active(|rules| rules.shared_object_disabled),
3233            user_transaction_disabled: switch_active(|rules| rules.user_transaction_disabled),
3234            receiving_objects_disabled: switch_active(|rules| rules.receiving_objects_disabled),
3235            move_authenticator_disabled: switch_active(|rules| rules.move_authenticator_disabled),
3236        }
3237    }

3403    /// This function executes given future until epoch_terminated is called
3404    /// If future finishes before epoch_terminated is called, future result is
3405    /// returned If epoch_terminated is called before future is resolved,
3406    /// `IotaError::EpochEnded` is returned
3407    ///
3408    /// In addition to the early termination guarantee, this function also
3409    /// prevents epoch_terminated() if future is being executed.
3410    pub async fn within_alive_epoch<F: Future + Send>(&self, f: F) -> IotaResult<F::Output> {
3411        // This guard is kept in the future until it resolves, preventing
3412        // `epoch_terminated` to acquire a write lock
3413        let guard = self.epoch_alive.read().await;
3414        if !*guard {
3415            return Err(IotaError::EpochEnded(self.epoch()));
3416        }
3417        // The hot path here -- epoch is alive, future resolves first -- is now a
3418        // single atomic load inside CancellationToken's `cancelled()` future,
3419        // instead of mutex + Arc clone + two `.boxed()` heap allocations per call.
3420        tokio::select! {
3421            _ = self.epoch_alive_token.cancelled() => Err(IotaError::EpochEnded(self.epoch())),
3422            result = f => Ok(result),
3423        }
3424    }

4190                soft_locks.release_for_batch(&soft_lock_release_tx_digests);
4191            } else {
4192                error!(
4193                    count = soft_lock_release_tx_digests.len(),
4194                    "P-COOL flow produced soft-lock release digests but \
4195                         soft_locks OnceCell was never set — wiring bug in \
4196                         start_epoch_specific_validator_components"
4197                );
4198            }
4199        }
4200
4201        // Only after batch is written, notify checkpoint service to start building any
4202        // new pending checkpoints.
4203        if make_checkpoint {
4204            debug!(
4205                ?consensus_commit_info.round,
4206                "Notifying checkpoint service about new pending checkpoint(s)",
4207            );
4208            checkpoint_service.notify_checkpoint()?;
4209        }
4210
4211        // Once commit processing is recorded, kick off randomness generation.
4212        if let Some(randomness_round) = randomness_round {
4213            let epoch = self.epoch();
4214            randomness_manager
4215                .as_ref()
4216                .expect("randomness manager should exist if randomness round is provided")
4217                .generate_randomness(epoch, randomness_round);
4218        }
4219
4220        self.process_notifications(&notifications, &end_of_publish_transactions);
4221
4222        if final_round {
4223            info!(
4224                epoch=?self.epoch(),
4225                // Accessing lock on purpose so that the compiler ensures
4226                // the lock is not yet dropped.
4227                lock=?lock.as_ref(),
4228                final_round=?final_round,
4229                "Notified last checkpoint"
4230            );
4231            self.record_end_of_message_quorum_time_metric();
4232        }
4233
4234        let all_txns = [
4235            verified_non_randomness_transactions,
4236            verified_randomness_transactions,
4237        ]
4238        .concat();
4239
4240        Ok((all_txns, assigned_versions))
4241    }

4247    /// Injects `TransactionDenyRulesUpdate` system transactions bringing the
4248    /// on-chain object to the current proposal aggregate, chunked and gated
4249    /// by `compute_deny_rule_update_chunks`. Advances the mirrored state to
4250    /// the injected target in the same commit output that persists it. Each
4251    /// scheduled update becomes a checkpoint root: nothing else in the commit
4252    /// can depend on the object, so an unrooted update would never be
4253    /// checkpointed.
4254    pub(crate) fn add_deny_rule_update_transactions(
4255        &self,
4256        output: &mut ConsensusCommitOutput,
4257        transactions: &mut VecDeque<Schedulable>,
4258        roots: &mut BTreeSet<TransactionKey>,
4259        consensus_commit_info: &ConsensusCommitInfo,
4260    ) -> IotaResult<()> {
4261        if !self.protocol_config().deny_rule_governance_on_chain() {
4262            return Ok(());
4263        }
4264        // Mirroring starts the epoch after the object is created: without the
4265        // object there is nothing to update.
4266        let Some(initial_shared_version) = self
4267            .epoch_start_config()
4268            .transaction_deny_rules_obj_initial_shared_version()
4269        else {
4270            return Ok(());
4271        };
4272
4273        // The aggregate from every recorded proposal, this commit's included.
4274        let mut proposals = self
4275            .consensus_quarantine
4276            .read()
4277            .current_deny_rule_proposals();
4278        proposals.extend(
4279            output
4280                .deny_rule_proposals()
4281                .iter()
4282                .map(|(authority, proposal)| (*authority, proposal.clone())),
4283        );
4284        // No proposals means an empty aggregate and zero announced stake.
4285        // The diff can produce no chunk from that.
4286        if proposals.is_empty() {
4287            return Ok(());
4288        }
4289        let aggregate = Self::compute_active_transaction_deny_rules(&proposals, self.committee());
4290        let mirror = self.mirrored_transaction_deny_rules.load_full();
4291
4292        // Removals wait for enough of the committee to re-announce, so entries
4293        // do not drop out while announcements are still arriving.
4294        let announced_stake = Self::announced_stake(&proposals, self.committee());
4295        let removals_unlocked = announced_stake >= self.committee().quorum_threshold()
4296            && consensus_commit_info.round
4297                >= self.protocol_config().deny_rule_removal_grace_round_floor();
4298        self.metrics
4299            .deny_rule_removals_unlocked
4300            .set(removals_unlocked as i64);
4301
4302        let (chunks, target) = compute_deny_rule_update_chunks(
4303            &aggregate,
4304            &mirror,
4305            removals_unlocked,
4306            self.protocol_config().deny_rule_update_max_entries_per_tx() as usize,
4307            self.epoch(),
4308            consensus_commit_info.round,
4309            initial_shared_version,
4310        );
4311        if chunks.is_empty() {
4312            return Ok(());
4313        }
4314        let chunk_count = chunks.len();
4315
4316        let mut all_scheduled = true;
4317        for chunk in chunks {
4318            let transaction = VerifiedExecutableTransaction::new_system(
4319                VerifiedTransaction::new_transaction_deny_rules_update(chunk),
4320                self.epoch(),
4321            );
4322            match self.process_consensus_system_transaction(&transaction) {
4323                ConsensusTransactionResult::Scheduled {
4324                    transaction,
4325                    start_time: _,
4326                } => {
4327                    roots.insert(transaction.key());
4328                    transactions.push_front(Schedulable::Transaction(transaction));
4329                }
4330                ConsensusTransactionResult::IgnoredSystem => {
4331                    all_scheduled = false;
4332                }
4333                _ => unreachable!(
4334                    "process_consensus_system_transaction returned unexpected ConsensusTransactionResult."
4335                ),
4336            }
4337            output.record_consensus_message_processed(SequencedConsensusTransactionKey::System(
4338                *transaction.digest(),
4339            ));
4340        }
4341        // Holding the mirror back when a chunk was skipped (epoch closing) makes
4342        // the next commit re-derive the same delta; re-application is a no-op.
4343        if all_scheduled {
4344            self.metrics.deny_rule_updates_injected.inc();
4345            self.metrics
4346                .deny_rule_update_transactions_injected
4347                .inc_by(chunk_count as u64);
4348            output.record_deny_rule_mirror(target.clone());
4349            self.mirrored_transaction_deny_rules.store(Arc::new(target));
4350            // The recompute in `push_consensus_output` only runs for commits
4351            // that record proposals, so a removal unlocking on a proposal-free
4352            // commit would otherwise stay enforced after the object dropped it.
4353            self.store_active_transaction_deny_rules(&proposals);
4354        }
4355
4356        Ok(())
4357    }

4549        // `randomness_transactions` is empty. The returned lists are always

4594                    tracker

4602                .congestion_control_parameters()

4607        // share the same congestion control parameters).

4609            shared_object_congestion_tracker
4610                .congestion_control_parameters()
4611                .clone(),
4612            self.reference_gas_price(),
4613        );
4614
4615        fail_point_arg!(
4616            "initial_suggested_gas_price_calculator",
4617            |calculator: SuggestedGasPriceCalculator| {
4618                info!(
4619                    "Initialize suggested_gas_price_calculator to {:?}",
4620                    calculator
4621                );
4622                suggested_gas_price_calculator = calculator;
4623            }
4624        );

4627        let mut sequenced_non_randomness = Vec::new();
4628        let mut sequenced_randomness = Vec::new();

4631            .iter()
4632            .map(Either::Left)

4645                    tx,
4646                    shared_object_using_randomness_congestion_tracker
4647                        .as_mut()

4649                            "randomness transactions are processed separately only when the \

4653                        .protocol_config
4654                        .separate_gas_price_feedback_mechanism_for_randomness()

4658                        &mut suggested_gas_price_calculator

4664            let mut filter_roots = false;

4666                .process_consensus_transaction(
4667                    output,
4668                    tx,
4669                    checkpoint_service,

4673                    dkg_failed,

4677                    authority_metrics,

4682                    transaction,

4687                    // randomness phase and checkpoint, whichever list they
4688                    // were scheduled from.

4702                        .entry(deferral_key)
4703                        .or_default()

4709                        notifications.push(key.clone());

4716                            .insert(*transaction.digest(), reason)

4718                    );
4719                    let start_time = congestion_tracker.max_occupied_slot_end_time();

4721                        sequenced_randomness.push((transaction, start_time));
4722                    } else {
4723                        sequenced_non_randomness.push((transaction, start_time));

4727                    randomness_state_updated = true;
4728                    notifications.push(key.clone());
4729                }

4731                ConsensusTransactionResult::IgnoredSystem => {

4733                }

4735                // they may not get reverted after restart during epoch change.
4736                ConsensusTransactionResult::Ignored => {
4737                    ignored = true;
4738                    filter_roots = true;
4739                }
4740            }

4742                output.record_consensus_message_processed(key.clone());
4743            }
4744            if filter_roots {

4746                    tx.0.transaction
4747                        .executable_transaction_digest()
4748                        .map(TransactionKey::Digest)
4749                {
4750                    non_randomness_roots.remove(&txn_key);
4751                    randomness_roots.remove(&txn_key);
4752                }
4753            }
4754        }

4757        // sequencing result and add these to the `verified_transactions`.
4758        sequenced_non_randomness.sort_by_key(|(_, start_time)| *start_time);
4759        let mut verified_non_randomness_transactions: VecDeque<_> = sequenced_non_randomness
4760            .into_iter()

4764        let mut verified_randomness_transactions: VecDeque<_> = sequenced_randomness
4765            .into_iter()
4766            .map(|(tx, _)| Schedulable::Transaction(tx))
4767            .collect();
4768
4769        // Front of the queue: the update must be assigned the randomness object's
4770        // version before the transactions below that read it.
4771        if let Some(round) = randomness_round {
4772            verified_randomness_transactions
4773                .push_front(Schedulable::RandomnessStateUpdate(self.epoch(), round));
4774        }
4775        let commit_has_deferred_txns = !deferred_txns.is_empty();
4776        let mut total_deferred_txns = 0;
4777        {
4778            if self
4779                .protocol_config
4780                .congestion_control_gas_price_feedback_mechanism()
4781            {
4782                let mut deferred_transactions =
4783                    self.consensus_output_cache.deferred_transactions_v2.lock();
4784                for (key, txns) in deferred_txns.into_iter() {
4785                    total_deferred_txns += txns.len();
4786                    deferred_transactions.insert(key, txns.clone());
4787                    output.defer_transactions(key, txns);
4788                }
4789            } else {
4790                let mut deferred_transactions =
4791                    self.consensus_output_cache.deferred_transactions.lock();
4792
4793                for (key, txns) in deferred_txns.into_iter() {
4794                    total_deferred_txns += txns.len();
4795                    deferred_transactions.insert(
4796                        key,
4797                        txns.iter()
4798                            .map(|tx| tx.transaction.clone())
4799                            .collect::<Vec<_>>(),
4800                    );
4801                    output.defer_transactions(key, txns);
4802                }
4803            }
4804        }
4805
4806        authority_metrics
4807            .consensus_handler_deferred_transactions
4808            .inc_by(total_deferred_txns as u64);
4809        authority_metrics
4810            .consensus_handler_cancelled_transactions
4811            .inc_by(cancelled_txns.len() as u64);
4812        authority_metrics
4813            .consensus_handler_max_object_costs
4814            .with_label_values(&["regular_commit"])
4815            .set(shared_object_congestion_tracker.max_occupied_slot_end_time() as i64);
4816        if let Some(randomness_tracker) = &shared_object_using_randomness_congestion_tracker {
4817            authority_metrics
4818                .consensus_handler_max_object_costs
4819                .with_label_values(&["randomness_commit"])
4820                .set(randomness_tracker.max_occupied_slot_end_time() as i64);
4821        } else {
4822            // The combined tracker schedules randomness-using transactions
4823            // together with the rest, so there is no separate randomness
4824            // tracker to report. Report zero rather than leaving the last
4825            // value from before the combined tracker was enabled.
4826            authority_metrics
4827                .consensus_handler_max_object_costs
4828                .with_label_values(&["randomness_commit"])
4829                .set(0);
4830        }
4831
4832        // Record accumulated debts from this consensus commit following sequencing.
4833        // This output will be written to consensus quarantine so the debts can be
4834        // loaded in the future consensus commit rounds where the objects are involved.
4835        if let Some(max_execution_duration_per_commit) = shared_object_congestion_tracker
4836            .congestion_control_parameters()
4837            .max_execution_duration_per_commit()
4838        {
4839            // Carry over the execution-worker debt that runs past the
4840            // per-commit limit (computed before the tracker is consumed for
4841            // the per-object debts below). `None` when worker control is off.
4842            if let Some(debt) = shared_object_congestion_tracker
4843                .accumulated_worker_debt(max_execution_duration_per_commit)
4844            {
4845                output.set_congestion_control_worker_debt(debt);
4846            }
4847            output.set_congestion_control_object_debts(
4848                shared_object_congestion_tracker
4849                    .accumulated_object_debts(max_execution_duration_per_commit),
4850            );
4851            if let Some(randomness_tracker) = shared_object_using_randomness_congestion_tracker {
4852                output.set_congestion_control_randomness_object_debts(
4853                    randomness_tracker.accumulated_object_debts(max_execution_duration_per_commit),
4854                );
4855            }
4856        }
4857
4858        // Advance the DKG state machine on every commit while it is still
4859        // pending, even when no new messages or confirmations arrived this
4860        // commit, so it can resolve from already-persisted state -- completing,
4861        // or failing once the timeout round passes. Without this, advance_dkg
4862        // only runs on commits carrying fresh DKG traffic, so a validator that
4863        // sees no new inbound traffic (e.g. one that restarted) can stay Pending
4864        // forever -- deferring all randomness-using transactions and blocking
4865        // epoch close.
4866        let dkg_pending = randomness_manager
4867            .as_ref()
4868            .is_some_and(|rm| rm.dkg_status() == DkgStatus::Pending);
4869        if randomness_state_updated || dkg_pending {
4870            if let Some(randomness_manager) = randomness_manager.as_mut() {
4871                randomness_manager
4872                    .advance_dkg(output, consensus_commit_info.round)
4873                    .await?;
4874            }
4875        }
4876
4877        // Inject deny-rule updates before the prologue is prepended, so the
4878        // prologue stays the first transaction of the commit.
4879        self.add_deny_rule_update_transactions(
4880            output,
4881            &mut verified_non_randomness_transactions,
4882            non_randomness_roots,
4883            consensus_commit_info,
4884        )?;
4885
4886        // Add the consensus commit prologue transaction to the beginning of
4887        // `verified_non_randomness_transactions`.
4888        let consensus_commit_prologue_root = self.add_consensus_commit_prologue_transaction(
4889            output,
4890            &mut verified_non_randomness_transactions,
4891            consensus_commit_info,
4892            &cancelled_txns,
4893        )?;
4894
4895        let verified_non_randomness_transactions: Vec<_> =
4896            verified_non_randomness_transactions.into();
4897        let verified_randomness_transactions: Vec<_> = verified_randomness_transactions.into();
4898
4899        let assigned_tx_and_versions = self.process_consensus_transaction_shared_object_versions(
4900            cache_reader,
4901            &verified_non_randomness_transactions,
4902            &verified_randomness_transactions,
4903            &cancelled_txns,
4904            output,
4905        )?;
4906
4907        let (lock, final_round) = self.process_end_of_publish_transactions_and_reconfig(
4908            output,
4909            end_of_publish_transactions,
4910            commit_has_deferred_txns,
4911        )?;
4912
4913        Ok((
4914            verified_non_randomness_transactions,
4915            verified_randomness_transactions,
4916            notifications,
4917            lock,
4918            final_round,
4919            consensus_commit_prologue_root,
4920            assigned_tx_and_versions,
4921        ))
4922    }

4924    fn process_end_of_publish_transactions_and_reconfig(
4925        &self,
4926        output: &mut ConsensusCommitOutput,
4927        transactions: &[VerifiedSequencedConsensusTransaction],
4928        commit_has_deferred_txns: bool,
4929    ) -> IotaResult<(
4930        Option<RwLockWriteGuard<'_, ReconfigState>>,
4931        bool, // true if final round
4932    )> {
4933        let mut lock = None;
4934
4935        for transaction in transactions {
4936            let VerifiedSequencedConsensusTransaction(SequencedConsensusTransaction {
4937                transaction,
4938                ..
4939            }) = transaction;
4940
4941            if let SequencedConsensusTransactionKind::External(ConsensusTransaction {
4942                kind: ConsensusTransactionKind::EndOfPublish(authority),
4943                ..
4944            }) = transaction
4945            {
4946                debug!(
4947                    "Received EndOfPublish for epoch {} from {:?}",
4948                    self.committee.epoch,
4949                    authority.concise()
4950                );
4951
4952                // It is ok to just release lock here as this function is the only place that
4953                // transition into RejectAllCerts state And this function itself
4954                // is always executed from consensus task
4955                let collected_end_of_publish = if lock.is_none()
4956                    && self
4957                        .get_reconfig_state_read_lock_guard()
4958                        .should_accept_consensus_certs()
4959                {
4960                    output.insert_end_of_publish(*authority);
4961                    self.end_of_publish.try_lock()
4962                        .expect("No contention on Authority::end_of_publish as it is only accessed from consensus handler")
4963                        .insert_generic(*authority, ()).is_quorum_reached()
4964                    // end_of_publish lock is released here.
4965                } else {
4966                    // If we past the stage where we are accepting consensus certificates we also
4967                    // don't record end of publish messages
4968                    debug!(
4969                        "Ignoring end of publish message from validator {:?} as we already collected enough end of publish messages",
4970                        authority.concise()
4971                    );
4972                    false
4973                };
4974
4975                if collected_end_of_publish {
4976                    assert!(lock.is_none());
4977                    debug!(
4978                        "Collected enough end_of_publish messages for epoch {} with last message from validator {:?}",
4979                        self.committee.epoch,
4980                        authority.concise(),
4981                    );
4982                    let mut l = self.get_reconfig_state_write_lock_guard();
4983                    l.close_all_certs();
4984                    output.store_reconfig_state(l.clone());
4985                    // Holding this lock until end of
4986                    // process_consensus_transactions_and_commit_boundary() where we write batch to
4987                    // DB
4988                    lock = Some(l);
4989                };
4990                // Important: we actually rely here on fact that ConsensusHandler panics if its
4991                // operation returns error. If some day we won't panic in ConsensusHandler on
4992                // error we need to figure out here how to revert in-memory
4993                // state of .end_of_publish and .reconfig_state when write
4994                // fails.
4995                output.record_consensus_message_processed(transaction.key());
4996            } else {
4997                panic!(
4998                    "process_end_of_publish_transactions_and_reconfig called with non-end-of-publish transaction"
4999                );
5000            }
5001        }
5002
5003        // Determine if we're ready to advance reconfig state to RejectAllTx.
5004        let is_reject_all_certs = if let Some(lock) = &lock {
5005            lock.is_reject_all_certs()
5006        } else {
5007            // It is ok to just release lock here as this function is the only place that
5008            // transitions into RejectAllTx state, and this function itself is always
5009            // executed from consensus task.
5010            self.get_reconfig_state_read_lock_guard()
5011                .is_reject_all_certs()
5012        };
5013
5014        if !is_reject_all_certs || !self.deferred_transactions_empty() || commit_has_deferred_txns {
5015            // Don't end epoch until all deferred transactions are processed.
5016            if is_reject_all_certs {
5017                debug!(
5018                    "Blocking end of epoch on deferred transactions, from previous commits?={}, from this commit?={commit_has_deferred_txns}",
5019                    !self.deferred_transactions_empty(),
5020                );
5021            }
5022            return Ok((lock, false));
5023        }
5024
5025        // Acquire lock to advance state if we don't already have it.
5026        let mut lock = lock.unwrap_or_else(|| self.get_reconfig_state_write_lock_guard());
5027        lock.close_all_tx();
5028        output.store_reconfig_state(lock.clone());
5029        Ok((Some(lock), true))
5030    }

5405                // The `deny_rule_governance` feature gate is enforced in
5406                // `verify_consensus_transaction`, which filters every
5407                // sequenced transaction before it reaches this point.
5408                if !self
5409                    .get_reconfig_state_read_lock_guard()
5410                    .should_accept_consensus_certs()
5411                {
5412                    debug!(
5413                        "Ignoring TransactionDenyRuleProposal from {:?} because of end of epoch",
5414                        proposal.authority.concise()
5415                    );
5416                } else if self.should_record_deny_rule_proposal(proposal) {
5417                    debug!(
5418                        "Received TransactionDenyRuleProposal from {:?} with generation {}",
5419                        proposal.authority.concise(),
5420                        proposal.generation
5421                    );
5422                    output.record_deny_rule_proposal(proposal.clone());
5423                } else {
5424                    debug!(
5425                        "Ignoring TransactionDenyRuleProposal from {:?} with stale generation {}",
5426                        proposal.authority.concise(),
5427                        proposal.generation
5428                    );
5429                }
5430                Ok(ConsensusTransactionResult::ConsensusMessage)
5431            }
5432            SequencedConsensusTransactionKind::System(system_transaction) => {
5433                Ok(self.process_consensus_system_transaction(system_transaction))

5438    /// Handles `SchedulingResult`, i.e., the output of the
5439    /// `self.try_schedule()` function, for the given
5440    /// `VerifiedExecutableTransaction`.
5441    fn handle_scheduling_result(
5442        &self,
5443        scheduling_result: SchedulingResult,
5444        verified_executable_tx: VerifiedExecutableTransaction,
5445        previously_deferred_tx_digests: &PreviouslyDeferredTransactions,
5446        dkg_failed: bool,
5447        shared_object_congestion_tracker: &mut SharedObjectCongestionTracker,
5448        suggested_gas_price_calculator: &mut SuggestedGasPriceCalculator,
5449        authority_metrics: &Arc<AuthorityMetrics>,
5450    ) -> IotaResult<ConsensusTransactionResult> {
5451        match scheduling_result {
5452            SchedulingResult::Defer(deferral_key, deferral_reason) => {
5453                debug!(
5454                    "Deferring verified executable transaction {:?} until {deferral_key:?}",
5455                    verified_executable_tx.digest(),
5456                );
5457
5458                let deferral_result = match deferral_reason {
5459                    DeferralReason::RandomnessNotReady => {
5460                        // Always defer transaction due to randomness not ready.
5461                        ConsensusTransactionResult::Deferred {
5462                            deferral_key,
5463                            suggested_gas_price: None,
5464                        }
5465                    }
5466                    DeferralReason::SharedObjectCongestion(congested_objects) => {
5467                        authority_metrics
5468                            .consensus_handler_congested_transactions
5469                            .inc();
5470
5471                        let suggested_gas_price = if self
5472                            .protocol_config
5473                            .congestion_control_gas_price_feedback_mechanism()
5474                        {
5475                            let current_commit_suggested_gas_price = suggested_gas_price_calculator
5476                                .calculate_suggested_gas_price(&verified_executable_tx);
5477
5478                            let suggested_gas_price = previously_deferred_tx_digests
5479                                .get(verified_executable_tx.digest())
5480                                .map_or_else(
5481                                    || current_commit_suggested_gas_price,
5482                                    |deferral_key_suggested_gas_price_pair| {
5483                                        deferral_key_suggested_gas_price_pair
5484                                            .1
5485                                            // If None, this could mean the transaction was
5486                                            // deferred due to randomness unavailable in
5487                                            // the previous round, but in the current
5488                                            // round, it gets deferred due to congestion.
5489                                            // Since this is the first round the
5490                                            // transaction is deferred due to congestion,
5491                                            // we return the suggested gas price from the
5492                                            // current round.
5493                                            .unwrap_or(current_commit_suggested_gas_price)
5494                                            .min(current_commit_suggested_gas_price)
5495                                    },
5496                                );
5497
5498                            Some(suggested_gas_price)
5499                        } else {
5500                            None
5501                        };
5502
5503                        if transaction_deferral_within_limit(
5504                            &deferral_key,
5505                            self.protocol_config()
5506                                .max_deferral_rounds_for_congestion_control(),
5507                        ) {
5508                            ConsensusTransactionResult::Deferred {
5509                                deferral_key,
5510                                suggested_gas_price,
5511                            }
5512                        } else {
5513                            // Cancel the transaction that has been deferred for too long.
5514                            //
5515                            // A deferral caused by execution-worker congestion reports no
5516                            // congested objects. Cancellation is signalled through assigned
5517                            // versions on the transaction's shared inputs, so treat all of
5518                            // them as congested; the suggested gas price is what matters to
5519                            // the client either way. A transaction without shared inputs is
5520                            // handled in version assignment via its gas object instead.
5521                            let congested_objects = if congested_objects.is_empty() {
5522                                verified_executable_tx
5523                                    .shared_input_objects()
5524                                    .iter()
5525                                    .map(|obj| obj.object_id)
5526                                    .collect()
5527                            } else {
5528                                congested_objects
5529                            };
5530                            debug!(
5531                                "Cancelling verified executable transaction {:?} with deferral \
5532                                    key {deferral_key:?} due to congestion on objects \
5533                                    {congested_objects:?}: actual gas price: {}, suggested gas \
5534                                    price: {suggested_gas_price:?}",
5535                                verified_executable_tx.digest(),
5536                                verified_executable_tx.transaction().gas_price(),
5537                            );
5538
5539                            ConsensusTransactionResult::Cancelled((
5540                                verified_executable_tx,
5541                                CancelConsensusTransactionReason::Congested {
5542                                    congested_objects,
5543                                    suggested_gas_price,
5544                                },
5545                            ))
5546                        }
5547                    }
5548                };
5549
5550                Ok(deferral_result)
5551            }
5552            SchedulingResult::Schedule(start_time) => {
5553                if dkg_failed && verified_executable_tx.uses_randomness() {
5554                    debug!(
5555                        "Canceling randomness-using verified executable transaction {:?} because \
5556                            DKG failed",
5557                        verified_executable_tx.digest(),
5558                    );
5559
5560                    return Ok(ConsensusTransactionResult::Cancelled((
5561                        verified_executable_tx,
5562                        CancelConsensusTransactionReason::DkgFailed,
5563                    )));
5564                }
5565
5566                // This transaction will be scheduled. We update the congestion
5567                // tracker (execution slots / worker slots) and the suggested
5568                // gas price calculator when it touches a shared object, or — when
5569                // execution-worker congestion control is active — for every
5570                // transaction (including owned-object-only ones, which still
5571                // occupy an execution worker).
5572                let worker_congestion_control_active = shared_object_congestion_tracker
5573                    .congestion_control_parameters()
5574                    .max_concurrent_execution_workers()
5575                    .is_some();
5576                if shared_object_congestion_tracker
5577                    .congestion_control_parameters()
5578                    .is_congestion_control_enabled()
5579                    && (verified_executable_tx.contains_shared_object()
5580                        || worker_congestion_control_active)
5581                {
5582                    // We only need to do this if congestion control is enabled,
5583                    // since otherwise this bumping will panic as object execution
5584                    // slots are only initialized if
5585                    // `max_execution_duration_per_commit` is not `None`.
5586                    let bump_result = shared_object_congestion_tracker
5587                        .bump_object_execution_slots(&verified_executable_tx, start_time);
5588
5589                    suggested_gas_price_calculator.update_congestion_info(bump_result);
5590                }
5591
5592                Ok(ConsensusTransactionResult::Scheduled {
5593                    transaction: verified_executable_tx,
5594                    start_time,
5595                })
5596            }
5597        }
5598    }

6018/// The union of two deny rule sets: an entry or switch is active when it is
6019/// active in either. Enforcement combines the mirrored on-chain state with
6020/// the current epoch's aggregate this way, so on-chain rules keep applying
6021/// before their supporters have re-announced.

6023    rules
6024        .denied_addresses
6025        .extend(other.denied_addresses.iter().copied());
6026    rules
6027        .denied_objects

6029    rules
6030        .denied_packages
6031        .extend(other.denied_packages.iter().copied());
6032    rules.package_publish_disabled |= other.package_publish_disabled;
6033    rules.package_upgrade_disabled |= other.package_upgrade_disabled;
6034    rules.shared_object_disabled |= other.shared_object_disabled;
6035    rules.user_transaction_disabled |= other.user_transaction_disabled;
6036    rules.receiving_objects_disabled |= other.receiving_objects_disabled;
6037    rules.move_authenticator_disabled |= other.move_authenticator_disabled;

6041/// Computes the `TransactionDenyRulesUpdate` transactions that bring the
6042/// on-chain object from `mirror` to `aggregate`: additions and switch
6043/// activations always, removals and switch deactivations only when
6044/// `removals_unlocked`. A delta larger than `max_entries_per_tx` is split into
6045/// disjoint chunks of the sorted delta, each carrying the absolute switch
6046/// states. Returns the chunks (empty when the object is up to date) and the
6047/// state the object holds once they have executed — the next mirror. Pure, so
6048/// every validator derives the same transactions at the same commit.
6049pub(crate) fn compute_deny_rule_update_chunks(
6050    aggregate: &DenyRuleSet,
6051    mirror: &DenyRuleSet,
6052    removals_unlocked: bool,
6053    max_entries_per_tx: usize,
6054    epoch: EpochId,
6055    round: u64,
6056    deny_rules_obj_initial_shared_version: Version,
6057) -> (Vec<TransactionDenyRulesUpdate>, DenyRuleSet) {
6058    let switch = |aggregate_switch: bool, mirror_switch: bool| {
6059        if removals_unlocked {
6060            aggregate_switch
6061        } else {
6062            aggregate_switch || mirror_switch
6063        }
6064    };
6065    let mut delta = TransactionDenyRulesUpdate {
6066        epoch,
6067        round,
6068        added_addresses: aggregate
6069            .denied_addresses
6070            .difference(&mirror.denied_addresses)
6071            .copied()
6072            .collect(),
6073        removed_addresses: BTreeSet::new(),
6074        added_objects: aggregate
6075            .denied_objects
6076            .difference(&mirror.denied_objects)
6077            .copied()
6078            .collect(),
6079        removed_objects: BTreeSet::new(),
6080        added_packages: aggregate
6081            .denied_packages
6082            .difference(&mirror.denied_packages)
6083            .copied()
6084            .collect(),
6085        removed_packages: BTreeSet::new(),
6086        package_publish_disabled: switch(
6087            aggregate.package_publish_disabled,
6088            mirror.package_publish_disabled,
6089        ),
6090        package_upgrade_disabled: switch(
6091            aggregate.package_upgrade_disabled,
6092            mirror.package_upgrade_disabled,
6093        ),
6094        shared_object_disabled: switch(
6095            aggregate.shared_object_disabled,
6096            mirror.shared_object_disabled,
6097        ),
6098        user_transaction_disabled: switch(
6099            aggregate.user_transaction_disabled,
6100            mirror.user_transaction_disabled,
6101        ),
6102        receiving_objects_disabled: switch(
6103            aggregate.receiving_objects_disabled,
6104            mirror.receiving_objects_disabled,
6105        ),
6106        move_authenticator_disabled: switch(
6107            aggregate.move_authenticator_disabled,
6108            mirror.move_authenticator_disabled,
6109        ),
6110        deny_rules_obj_initial_shared_version,
6111    };
6112    if removals_unlocked {
6113        delta.removed_addresses = mirror
6114            .denied_addresses
6115            .difference(&aggregate.denied_addresses)
6116            .copied()
6117            .collect();
6118        delta.removed_objects = mirror
6119            .denied_objects
6120            .difference(&aggregate.denied_objects)
6121            .copied()
6122            .collect();
6123        delta.removed_packages = mirror
6124            .denied_packages
6125            .difference(&aggregate.denied_packages)
6126            .copied()
6127            .collect();
6128    }
6129
6130    // The object state once the delta has executed.
6131    let target = DenyRuleSet {
6132        denied_addresses: mirror
6133            .denied_addresses
6134            .union(&delta.added_addresses)
6135            .filter(|key| !delta.removed_addresses.contains(key))
6136            .copied()
6137            .collect(),
6138        denied_objects: mirror
6139            .denied_objects
6140            .union(&delta.added_objects)
6141            .filter(|key| !delta.removed_objects.contains(key))
6142            .copied()
6143            .collect(),
6144        denied_packages: mirror
6145            .denied_packages
6146            .union(&delta.added_packages)
6147            .filter(|key| !delta.removed_packages.contains(key))
6148            .copied()
6149            .collect(),
6150        package_publish_disabled: delta.package_publish_disabled,
6151        package_upgrade_disabled: delta.package_upgrade_disabled,
6152        shared_object_disabled: delta.shared_object_disabled,
6153        user_transaction_disabled: delta.user_transaction_disabled,
6154        receiving_objects_disabled: delta.receiving_objects_disabled,
6155        move_authenticator_disabled: delta.move_authenticator_disabled,
6156    };
6157
6158    let switches_changed = delta.package_publish_disabled != mirror.package_publish_disabled
6159        || delta.package_upgrade_disabled != mirror.package_upgrade_disabled
6160        || delta.shared_object_disabled != mirror.shared_object_disabled
6161        || delta.user_transaction_disabled != mirror.user_transaction_disabled
6162        || delta.receiving_objects_disabled != mirror.receiving_objects_disabled
6163        || delta.move_authenticator_disabled != mirror.move_authenticator_disabled;
6164    let entry_count = delta.added_addresses.len()
6165        + delta.removed_addresses.len()
6166        + delta.added_objects.len()
6167        + delta.removed_objects.len()
6168        + delta.added_packages.len()
6169        + delta.removed_packages.len();
6170    if entry_count == 0 && !switches_changed {
6171        return (Vec::new(), target);
6172    }
6173
6174    // Deterministic chunking: cut each (already sorted) delta set into runs,
6175    // filling one chunk to the limit before starting the next.
6176    let max_entries_per_tx = max_entries_per_tx.max(1);
6177    let chunk_count = entry_count.div_ceil(max_entries_per_tx).max(1);
6178    let empty = TransactionDenyRulesUpdate {
6179        added_addresses: BTreeSet::new(),
6180        removed_addresses: BTreeSet::new(),
6181        added_objects: BTreeSet::new(),
6182        removed_objects: BTreeSet::new(),
6183        added_packages: BTreeSet::new(),
6184        removed_packages: BTreeSet::new(),
6185        ..delta
6186    };
6187    let mut chunks = vec![empty; chunk_count];
6188    let mut slot = 0;
6189    {
6190        let mut fill =
6191            |set: BTreeSet<Address>,
6192             select: fn(&mut TransactionDenyRulesUpdate) -> &mut BTreeSet<Address>| {
6193                for key in set {
6194                    select(&mut chunks[slot / max_entries_per_tx]).insert(key);
6195                    slot += 1;
6196                }
6197            };
6198        fill(delta.added_addresses, |chunk| &mut chunk.added_addresses);
6199        fill(delta.removed_addresses, |chunk| {
6200            &mut chunk.removed_addresses
6201        });
6202    }
6203    {
6204        let mut fill =
6205            |set: BTreeSet<ObjectId>,
6206             select: fn(&mut TransactionDenyRulesUpdate) -> &mut BTreeSet<ObjectId>| {
6207                for key in set {
6208                    select(&mut chunks[slot / max_entries_per_tx]).insert(key);
6209                    slot += 1;
6210                }
6211            };
6212        fill(delta.added_objects, |chunk| &mut chunk.added_objects);
6213        fill(delta.removed_objects, |chunk| &mut chunk.removed_objects);
6214        fill(delta.added_packages, |chunk| &mut chunk.added_packages);
6215        fill(delta.removed_packages, |chunk| &mut chunk.removed_packages);
6216    }
6217
6218    (chunks, target)
