44/// Column-family prefix of the historic object buckets; a bucket's family
45/// is `{prefix}{epoch}`.
46const HISTORIC_OBJECTS_CF_PREFIX: &str = "hist_obj_e";
47
48/// The directory of each epoch, under the historic root, that this
49/// store's buckets keep their files in.
50const BUCKET_DIR: &str = "objects";
51
52/// Tag of the relocated-versions table inside a bucket's column family.
53/// Do not reuse a tag for a different table: mark it retired in a comment
54/// instead, so an older bucket's rows can never be read as the wrong type.
55const DB_PREFIX_HISTORIC_OBJECTS: u8 = 0;
56
57/// Tag of the tombstones table inside a bucket's column family.
58const DB_PREFIX_HISTORIC_TOMBSTONES: u8 = 1;
59
60/// Tag of the expiring marker inside a bucket's column family.
61const DB_PREFIX_HISTORIC_EXPIRING: u8 = 2;
62
63/// Tombstones deleted from the live `objects` table per write batch when
64/// a bucket expires.
65const TOMBSTONE_DELETE_BATCH_SIZE: usize = 10_000;
66
67/// Column family holding the earliest-retained-epoch marker
68/// [`EpochBuckets`] persists on a prune; empty until the first prune.
69///
70/// The name must not begin with [`HISTORIC_OBJECTS_CF_PREFIX`], since that is
71/// how a bucket's column family is told from every other one in this
72/// database.
73const EARLIEST_RETAINED_CF: &str = "hist_obj_retention";

75/// One epoch's relocated object versions.
76pub struct HistoricObjectsBucket {
77    /// Object versions superseded during this epoch, keyed exactly as the
78    /// live `objects` table keys them.
79    pub(crate) objects: TaggedDBMap<ObjectKey, Object>,
80
81    /// The objects deleted or wrapped during this epoch. Their tombstones
82    /// stay in the live `objects` table until this bucket expires: a
83    /// tombstone has to outlive every version beneath it, and a tombstone
84    /// written in this epoch can only sit above versions relocated in this
85    /// epoch or an earlier one. Hence expiry goes oldest epoch first.
86    ///
87    /// A reader that also consults the live `objects` table must read it
88    /// **before** asking the buckets. An expiry marks the bucket expiring
89    /// before it deletes a single tombstone, and reads skip a bucket marked
90    /// expiring. Reading live first therefore means either the live read
91    /// finds the tombstone, or the deletion it missed came after the mark,
92    /// and the bucket read skips the bucket.
93    pub(crate) tombstones: TaggedDBMap<ObjectKey, ()>,
94
95    /// Present once this bucket is scheduled for expiry, after which reads
96    /// skip it. Write it through [`Self::mark_expiring`].
97    pub(crate) expiring: TaggedDBMap<(), ()>,
98
99    /// Mirrors the `expiring` row, so a query need not look it up.
100    expiring_marked: AtomicBool,
101}

103impl BucketReopen for HistoricObjectsBucket {
104    fn reopen(db: &Arc<Database>, cf_name: &str) -> Result<Self, TypedStoreError> {
105        let expiring: TaggedDBMap<(), ()> = TaggedDBMap::reopen(
106            db,
107            cf_name,
108            DB_PREFIX_HISTORIC_EXPIRING,
109            &ReadWriteOptions::default(),
110            true,
111        )?
112        .with_metrics_label(bucket_metrics_label(cf_name));
113        let expiring_marked = AtomicBool::new(expiring.get(&())?.is_some());
114        Ok(Self {
115            objects: TaggedDBMap::reopen(
116                db,
117                cf_name,
118                DB_PREFIX_HISTORIC_OBJECTS,
119                &ReadWriteOptions::default(),
120                true,
121            )?
122            .with_metrics_label(bucket_metrics_label(cf_name)),
123            tombstones: TaggedDBMap::reopen(
124                db,
125                cf_name,
126                DB_PREFIX_HISTORIC_TOMBSTONES,
127                &ReadWriteOptions::default(),
128                true,
129            )?
130            .with_metrics_label(bucket_metrics_label(cf_name)),
131            expiring,
132            expiring_marked,
133        })
134    }
135}

137impl HistoricObjectsBucket {
138    /// Whether this bucket has been marked expiring, in which case its rows
139    /// must no longer be served: its tombstones may already be deleted
140    /// from the live `objects` table.
141    fn is_expiring(&self) -> bool {
142        // Pairs with the Release in `mark_expiring`: a reader that saw a tombstone
143        // deleted also sees the mark set before the deletion.
144        self.expiring_marked.load(Ordering::Acquire)
145    }
146
147    /// Durably marks this bucket expiring, then stops serving its rows.
148    ///
149    /// Synced, because a column-family drop is durable at once while a
150    /// default write may be lost, which after a crash would make a bucket
151    /// whose tombstones are gone readable again.
152    fn mark_expiring(&self) -> Result<(), TypedStoreError> {
153        let mut batch = self.expiring.batch();
154        batch.insert_batch_tagged(&self.expiring, [((), ())])?;
155        batch.write_opt(&synced_write_options())?;
156        self.expiring_marked.store(true, Ordering::Release);
157        Ok(())
158    }
159}

193    /// Opens the historic-object buckets among the column families of `db`,
194    /// the perpetual database. `db_options` and `historic_root` must be the
195    /// ones `db` was opened with, and `objects` its live objects table.
196    ///
197    /// Before any query can reach them, finishes the expiry of the buckets an
198    /// interrupted prune left behind, oldest first, and fails if it cannot.
199    pub fn open(
200        db: Arc<Database>,
201        db_options: &DBOptions,
202        objects: DBMap<ObjectKey, StoreObjectWrapper>,
203        historic_root: &Path,
204    ) -> Result<Self, TypedStoreError> {
205        let existing_cfs = list_tables(db.path_for_pruning().to_path_buf())
206            .map_err(|e| TypedStoreError::RocksDB(format!("failed to list buckets: {e}")))?;
207
208        let mut buckets = BTreeMap::new();
209        for cf_name in &existing_cfs {
210            if let Some(epoch) = bucket_cf_epoch(HISTORIC_OBJECTS_CF_PREFIX, cf_name) {
211                buckets.insert(
212                    epoch,
213                    Arc::new(HistoricObjectsBucket::reopen(&db, cf_name)?),
214                );
215            }
216        }
217
218        let cf_options = bucket_cf_options(db_options).options;
219        if db.cf_handle(EARLIEST_RETAINED_CF).is_none() {
220            db.create_cf(EARLIEST_RETAINED_CF, &cf_options)?;
221        }
222        let earliest_retained_table: DBMap<(), EpochId> = DBMap::reopen(
223            &db,
224            Some(EARLIEST_RETAINED_CF),
225            &ReadWriteOptions::default(),
226            true,
227        )?;
228        let earliest_retained = earliest_retained_table.get(&())?.unwrap_or(0);
229
230        // A prune persists the floor before it marks anything, so a bucket
231        // below the floor is an unfinished expiry even without its marker.
232        // `EpochBuckets::open` would just drop it, leaving its tombstones
233        // in the live `objects` table for good.
234        //
235        // Oldest first: a bounded read that stops at one of this bucket's
236        // deleted tombstones falls through to the older buckets, so those must be
237        // gone first.
238        let interrupted: Vec<(EpochId, Arc<HistoricObjectsBucket>)> = buckets
239            .iter()
240            .filter(|(&epoch, bucket)| epoch < earliest_retained || bucket.is_expiring())
241            .map(|(&epoch, bucket)| (epoch, bucket.clone()))
242            .collect();
243        for (epoch, bucket) in interrupted {
244            let cf_name = bucket_cf_name(HISTORIC_OBJECTS_CF_PREFIX, epoch);
245            if let Err(e) = Self::expire_bucket(&objects, epoch, &bucket, &AtomicBool::new(false)) {
246                // Named so that an operator whose node will not start can
247                // reach the column family with external tooling.
248                error!(
249                    epoch,
250                    cf_name, "cannot finish the interrupted expiry of a historic bucket: {e}"
251                );
252                return Err(e);
253            }
254            buckets.remove(&epoch);
255            info!(
256                epoch,
257                "dropping the bucket of an interrupted expiry at open"
258            );
259            if let Err(e) = db.drop_cf(&cf_name) {
260                warn!(
261                    epoch,
262                    "failed to drop an expiring bucket column family: {e}"
263                );
264            }
265        }
266
267        let buckets = EpochBuckets::open(
268            db,
269            "historic objects",
270            HISTORIC_OBJECTS_CF_PREFIX,
271            cf_options,
272            Some(BucketPaths::new(historic_root, BUCKET_DIR)),
273            earliest_retained_table,
274            buckets,
275        )?;
276        Ok(Self {
277            buckets,
278            objects,
279            stopping: AtomicBool::new(false),
280            #[cfg(test)]
281            bucket_walks: std::sync::atomic::AtomicU64::new(0),
282        })
283    }

327    /// The object relocated under `key`, `None` if it was never relocated (or

329    pub fn get(&self, key: &ObjectKey) -> IotaResult<Option<Object>> {
330        for bucket in self.readable_buckets(true) {
331            if let Some(object) = absent_if_dropped(bucket.objects.get(key))

336        }

340    /// The object `epoch`'s bucket holds under `key`, `None` when it holds
341    /// none or there is no bucket for `epoch`. Reads only that bucket, so the
342    /// answer does not depend on how many epochs the node retains.
343    pub fn get_in_epoch(&self, epoch: EpochId, key: &ObjectKey) -> IotaResult<Option<Object>> {
344        let Some(bucket) = self.buckets.get(epoch) else {
345            return Ok(None);
346        };
347        match bucket.objects.get(key) {
348            // The bucket was dropped during the read.
349            Err(TypedStoreError::UnregisteredColumn(_)) => Ok(None),
350            read => read.map_err(|e| IotaError::Storage(e.to_string())),
351        }
352    }

354    /// Whether `epoch`'s bucket holds a version under `key`, that is whether a
355    /// commit of that epoch relocated it; `false` also when there is no bucket
356    /// for `epoch`. Reads only that bucket, so the answer does not depend on
357    /// how many epochs the node retains.
358    ///
359    /// One exception: a node upgraded from a release before the per-epoch
360    /// history buckets (1.36) moved every version superseded before the
361    /// upgrade into the bucket of the epoch it upgraded in. For that epoch,
362    /// `true` can also mean the version was superseded before the upgrade.
363    pub fn relocated_in_epoch(&self, epoch: EpochId, key: &ObjectKey) -> IotaResult<bool> {
364        let Some(bucket) = self.buckets.get(epoch) else {
365            return Ok(false);
366        };
367        match bucket.objects.contains_key(key) {
368            // The bucket was dropped during the read.
369            Err(TypedStoreError::UnregisteredColumn(_)) => Ok(false),
370            read => read.map_err(|e| IotaError::Storage(e.to_string())),
371        }
372    }

374    /// Fills each `None` in `objects` — the live table's answers for `keys`, in
375    /// the same order — with the version the buckets hold under that key.
376    ///
377    /// Only for reads that serve a response. Execution reads current versions,
378    /// which never leave the live table, so a miss there is a bug and must stay
379    /// one. Validation at a commit reads earlier versions of its own epoch
380    /// through [`Self::get_in_epoch`] instead.
381    ///
382    /// Reads each bucket once for all the keys still missing, starting with
383    /// `likely_epoch`'s, then newest first. Pass the epoch that superseded
384    /// the versions where it is known, such as a transaction's own epoch for
385    /// its inputs.
386    pub fn fill_missing(
387        &self,
388        keys: &[ObjectKey],
389        objects: &mut [Option<Object>],
390        likely_epoch: Option<EpochId>,
391    ) -> IotaResult<()> {
392        let mut pending: Vec<usize> = (0..keys.len())
393            .filter(|&index| objects[index].is_none())
394            .collect();
395        if pending.is_empty() {
396            return Ok(());
397        }
398        let mut buckets: Vec<(EpochId, Arc<HistoricObjectsBucket>)> = self
399            .buckets
400            .iter_with_epoch(true)
401            .into_iter()
402            .filter(|(_, bucket)| !bucket.is_expiring())
403            .collect();
404        if let Some(position) =
405            likely_epoch.and_then(|likely| buckets.iter().position(|(epoch, _)| *epoch == likely))
406        {
407            let likely = buckets.remove(position);
408            buckets.insert(0, likely);
409        }
410        for (_, bucket) in buckets {
411            if pending.is_empty() {
412                break;
413            }
414            let found = absent_if_dropped(
415                bucket
416                    .objects
417                    .multi_get(pending.iter().map(|&index| &keys[index])),
418            )
419            .map_err(|e| IotaError::Storage(e.to_string()))?;
420            // Empty for a dropped bucket, which leaves every key pending.
421            if found.is_empty() {
422                continue;
423            }
424            let mut still_pending = Vec::with_capacity(pending.len());
425            for (index, object) in pending.into_iter().zip(found) {
426                match object {
427                    Some(object) => objects[index] = Some(object),
428                    None => still_pending.push(index),
429                }
430            }
431            pending = still_pending;
432        }
433        Ok(())
434    }

436    /// The newest relocated version of `id` at or below `version`, `None` if
437    /// there is none (or its bucket has since been dropped).
438    ///
439    /// A caller must read the live `objects` table first (see
440    /// `HistoricObjectsBucket::tombstones`) and take the newer of the two
441    /// answers: this ignores tombstones, so only a live tombstone newer than
442    /// what it returns means the object is gone.
443    pub fn find_lt_or_eq_version(
444        &self,
445        id: ObjectId,
446        version: Version,
447    ) -> IotaResult<Option<Object>> {
448        #[cfg(test)]
449        self.bucket_walks.fetch_add(1, Ordering::Relaxed);
450        // Versions are relocated in increasing order, so the newest bucket
451        // with a match holds the newest version.
452        for bucket in self.readable_buckets(true) {
453            let newest = absent_if_dropped(
454                bucket
455                    .objects
456                    .safe_range_iter_reversed(ObjectKey::min_for_id(&id)..=ObjectKey(id, version))
457                    .next()
458                    .transpose(),
459            )
460            .map_err(|e| IotaError::Storage(e.to_string()))?;
461            if let Some((_, object)) = newest {
462                return Ok(Some(object));
463            }
464        }
465        Ok(None)
466    }

481    /// Keeps the buckets of `current_epoch` and the `epochs_to_retain` epochs
482    /// below it, drops older ones, and deletes the tombstones each
483    /// dropped epoch recorded. Returns the earliest epoch still retained and
484    /// how many buckets were dropped, `None` when there is no bucket at all.
485    ///
486    /// Blocks for as long as the deletions and drops take, so an async caller
487    /// must use `spawn_blocking`. Queries keep being answered meanwhile, from
488    /// the buckets not marked expiring.
489    pub fn prune(
490        &self,
491        current_epoch: EpochId,
492        epochs_to_retain: u64,
493    ) -> IotaResult<Option<PrunedBuckets>> {
494        self.buckets
495            .prune(current_epoch, epochs_to_retain, |epoch, bucket| {
496                Self::expire_bucket(&self.objects, epoch, bucket, &self.stopping).map_err(|e| {
497                    TypedStoreError::RocksDB(format!("expiring the bucket of epoch {epoch}: {e}"))
498                })
499            })
500            .map_err(|e| IotaError::Storage(e.to_string()))
501    }

518    /// Marks `bucket` expiring, then deletes the tombstones it recorded from
519    /// the live `objects` table.
520    ///
521    /// Safe to run again on the same bucket: the marker is rewritten as it
522    /// was and a tombstone already deleted is deleted again.
523    fn expire_bucket(
524        objects: &DBMap<ObjectKey, StoreObjectWrapper>,
525        epoch: EpochId,
526        bucket: &Arc<HistoricObjectsBucket>,
527        stopping: &AtomicBool,
528    ) -> Result<(), TypedStoreError> {
529        bucket.mark_expiring()?;
530
531        // Synced: a column-family drop is durable at once, so a lost deletion
532        // would leave tombstones with nothing left to delete them.
533        let delete = |tombstones: Vec<ObjectKey>| -> Result<(), TypedStoreError> {
534            let mut batch = objects.batch();
535            batch.delete_batch(objects, tombstones)?;
536            batch.write_opt(&synced_write_options())
537        };
538
539        // A version left below a tombstone in the live table would become the
540        // newest again once the tombstone is deleted, bringing the object
541        // back, so such a tombstone is kept. Tombstones below it don't count: an
542        // object wrapped and deleted in one epoch leaves two in this bucket.
543        //
544        // The bucket lists its tombstones by object, so each object's live
545        // rows are scanned once, for its lowest live version: an object
546        // wrapped and unwrapped many times in the epoch has as many tombstones
547        // in the live table, and scanning below each of them would be
548        // quadratic in their number.
549        let lowest_live_version =
550            |object_id: &ObjectId| -> Result<Option<Version>, TypedStoreError> {
551                for row in objects.safe_range_iter(
552                    ObjectKey::min_for_id(object_id)..=ObjectKey::max_for_id(object_id),
553                ) {
554                    let (key, row) = row?;
555                    if matches!(row.migrate().into_inner(), StoreObject::Value(_)) {
556                        return Ok(Some(key.1));
557                    }
558                }
559                Ok(None)
560            };
561
562        let mut deleted = 0;
563        let mut kept = 0;
564        let mut batch = Vec::with_capacity(TOMBSTONE_DELETE_BATCH_SIZE);
565        let mut lowest_live: Option<(ObjectId, Option<Version>)> = None;
566        for row in bucket.tombstones.safe_iter() {
567            let (key, ()) = row?;
568            let lowest = match lowest_live {
569                Some((object_id, lowest)) if object_id == key.0 => lowest,
570                _ => {
571                    let lowest = lowest_live_version(&key.0)?;
572                    lowest_live = Some((key.0, lowest));
573                    lowest
574                }
575            };
576            if lowest.is_some_and(|lowest| lowest < key.1) {
577                kept += 1;
578                continue;
579            }
580            batch.push(key);
581            if batch.len() == TOMBSTONE_DELETE_BATCH_SIZE {
582                deleted += batch.len();
583                delete(std::mem::replace(
584                    &mut batch,
585                    Vec::with_capacity(TOMBSTONE_DELETE_BATCH_SIZE),
586                ))?;
587                // The bucket stays marked expiring, so the next open
588                // finishes the expiry from here.
589                if stopping.load(Ordering::Acquire) {
590                    return Err(TypedStoreError::RocksDB(format!(
591                        "the expiry of epoch {epoch}'s bucket was stopped by a shutdown"
592                    )));
593                }
594            }
595        }
596        deleted += batch.len();
597        delete(batch)?;
598
599        if kept > 0 {
600            error!(
601                epoch,
602                kept,
603                "kept tombstones that still have a live version below them: those \
604                 objects stay deleted, which is why the tombstones stay too"
605            );
606        }
607        info!(
608            epoch,
609            tombstones = deleted,
610            kept,
611            "expired a historic bucket"
612        );
613        Ok(())
614    }

617    /// that expiring that bucket fails.

619    pub(super) fn corrupt_recorded_tombstones_for_testing(

621        epoch: EpochId,

635    /// One page of the rows of `cf_name` if it is one of this store's column
636    /// families, `None` otherwise. For the `iota-tool` table dump, which
637    /// cannot reach these through `AuthorityPerpetualTables`.
638    ///
639    /// Tombstone and expiring-marker rows are prefixed by table name.
640    /// `db` may be a read-only or secondary handle.
641    pub fn dump_column_family(
642        db: &Arc<Database>,
643        cf_name: &str,
644        page_size: u16,
645        page_number: usize,
646    ) -> Result<Option<BTreeMap<String, String>>, TypedStoreError> {
647        if bucket_cf_epoch(HISTORIC_OBJECTS_CF_PREFIX, cf_name).is_some() {
648            let bucket = HistoricObjectsBucket::reopen(db, cf_name)?;
649            bucket.objects.try_catch_up_with_primary()?;
650            bucket.tombstones.try_catch_up_with_primary()?;
651            bucket.expiring.try_catch_up_with_primary()?;
652            let rows = dump_rows("", bucket.objects.safe_iter())
653                .chain(dump_rows("tombstone:", bucket.tombstones.safe_iter()))
654                .chain(dump_rows("expiring:", bucket.expiring.safe_iter()));
655            return dump_page(rows, page_size, page_number).map(Some);
656        }
657        if cf_name == EARLIEST_RETAINED_CF {
658            return dump_retention_floor(db, cf_name, page_size, page_number).map(Some);
659        }
660        Ok(None)
661    }
662}

664/// Like [`iota_types::storage::get_transaction_input_objects`], but also
665/// reads the relocated versions.
666pub fn get_transaction_input_objects(
667    object_store: &dyn ObjectStore,
668    historic_objects: &HistoricObjects,
669    effects: &TransactionEffects,
670) -> IotaResult<Vec<Object>> {
671    let keys = effects
672        .modified_at_versions()
673        .into_iter()
674        .map(|modified| ObjectKey(*modified.object_id(), modified.version()))
675        .collect::<Vec<_>>();
676    // The transaction superseded its inputs, in the epoch it executed in.
677    multi_get_objects_with_historic_fallback(
678        object_store,
679        historic_objects,
680        &keys,
681        effects.transaction_digest(),
682        Some(effects.epoch()),
683    )
684}
685
686/// Like [`iota_types::storage::get_transaction_output_objects`], but also
687/// reads the relocated versions.
688pub fn get_transaction_output_objects(
689    object_store: &dyn ObjectStore,
690    historic_objects: &HistoricObjects,
691    effects: &TransactionEffects,
692) -> IotaResult<Vec<Object>> {
693    let keys = effects
694        .all_changed_objects()
695        .into_iter()
696        .map(|(changed, _kind)| ObjectKey::from(*changed.reference()))
697        .collect::<Vec<_>>();
698    multi_get_objects_with_historic_fallback(
699        object_store,
700        historic_objects,
701        &keys,
702        effects.transaction_digest(),
703        None,
704    )
705}
706
707/// The objects at exactly `keys`, erroring on any the live table and the
708/// buckets both lack. `likely_epoch` is as for
709/// [`HistoricObjects::fill_missing`].
710fn multi_get_objects_with_historic_fallback(
711    object_store: &dyn ObjectStore,
712    historic_objects: &HistoricObjects,
713    keys: &[ObjectKey],
714    transaction_digest: &TransactionDigest,
715    likely_epoch: Option<EpochId>,
716) -> IotaResult<Vec<Object>> {
717    let mut objects = object_store.multi_get_objects_by_key(keys);
718    historic_objects.fill_missing(keys, &mut objects, likely_epoch)?;
719    objects
720        .into_iter()
721        .zip(keys)
722        .map(|(object, key)| {
723            object.ok_or_else(|| {
724                IotaError::Storage(format!(
725                    "missing object key {key:?} from tx {transaction_digest}"
726                ))
727            })
728        })
729        .collect()
730}
