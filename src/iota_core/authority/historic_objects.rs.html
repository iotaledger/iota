45/// Column-family prefix of the historic object buckets; a bucket's family
46/// is `{prefix}{epoch}`.
47const HISTORIC_OBJECTS_CF_PREFIX: &str = "hist_obj_e";
48
49/// The directory of each epoch, under the historic root, that this
50/// store's buckets keep their files in.
51const BUCKET_DIR: &str = "objects";
52
53/// Tag of the relocated-versions table inside a bucket's column family.
54/// Do not reuse a tag for a different table: mark it retired in a comment
55/// instead, so an older bucket's rows can never be read as the wrong type.
56const DB_PREFIX_HISTORIC_OBJECTS: u8 = 0;
57
58/// Tag of the tombstones table inside a bucket's column family.
59const DB_PREFIX_HISTORIC_TOMBSTONES: u8 = 1;
60
61/// Tag of the expiring marker inside a bucket's column family.
62const DB_PREFIX_HISTORIC_EXPIRING: u8 = 2;
63
64/// Tombstones deleted from the live `objects` table per write batch when
65/// a bucket expires.
66const TOMBSTONE_DELETE_BATCH_SIZE: usize = 10_000;
67
68/// Column family holding the earliest-retained-epoch marker
69/// [`EpochBuckets`] persists on a prune; empty until the first prune.
70///
71/// The name must not begin with [`HISTORIC_OBJECTS_CF_PREFIX`], since that is
72/// how a bucket's column family is told from every other one in this
73/// database.
74const EARLIEST_RETAINED_CF: &str = "hist_obj_retention";

76/// One epoch's relocated object versions.
77pub struct HistoricObjectsBucket {
78    /// Object versions superseded during this epoch, keyed exactly as the
79    /// live `objects` table keys them.
80    pub(crate) objects: TaggedDBMap<ObjectKey, Object>,
81
82    /// The objects deleted or wrapped during this epoch. Their tombstones
83    /// stay in the live `objects` table until this bucket expires: a
84    /// tombstone has to outlive every version beneath it, and a tombstone
85    /// written in this epoch can only sit above versions relocated in this
86    /// epoch or an earlier one. Hence expiry goes oldest epoch first.
87    ///
88    /// A reader that also consults the live `objects` table must read it
89    /// **before** asking the buckets. An expiry marks the bucket expiring
90    /// before it deletes a single tombstone, and reads skip a bucket marked
91    /// expiring. Reading live first therefore means either the live read
92    /// finds the tombstone, or the deletion it missed came after the mark,
93    /// and the bucket read skips the bucket.
94    pub(crate) tombstones: TaggedDBMap<ObjectKey, ()>,
95
96    /// Present once this bucket is scheduled for expiry, after which reads
97    /// skip it. Write it through [`Self::mark_expiring`].
98    pub(crate) expiring: TaggedDBMap<(), ()>,
99
100    /// Mirrors the `expiring` row, so a query need not look it up.
101    expiring_marked: AtomicBool,
102}

104impl BucketReopen for HistoricObjectsBucket {
105    fn reopen(db: &Arc<Database>, cf_name: &str) -> Result<Self, TypedStoreError> {
106        let expiring: TaggedDBMap<(), ()> = TaggedDBMap::reopen(
107            db,
108            cf_name,
109            DB_PREFIX_HISTORIC_EXPIRING,
110            &ReadWriteOptions::default(),
111            true,
112        )?
113        .with_metrics_label(bucket_metrics_label(cf_name));
114        let expiring_marked = AtomicBool::new(expiring.get(&())?.is_some());
115        Ok(Self {
116            objects: TaggedDBMap::reopen(
117                db,
118                cf_name,
119                DB_PREFIX_HISTORIC_OBJECTS,
120                &ReadWriteOptions::default(),
121                true,
122            )?
123            .with_metrics_label(bucket_metrics_label(cf_name)),
124            tombstones: TaggedDBMap::reopen(
125                db,
126                cf_name,
127                DB_PREFIX_HISTORIC_TOMBSTONES,
128                &ReadWriteOptions::default(),
129                true,
130            )?
131            .with_metrics_label(bucket_metrics_label(cf_name)),
132            expiring,
133            expiring_marked,
134        })
135    }
136}

138impl HistoricObjectsBucket {
139    /// Whether this bucket has been marked expiring, in which case its rows
140    /// must no longer be served: its tombstones may already be deleted
141    /// from the live `objects` table.
142    fn is_expiring(&self) -> bool {
143        // Pairs with the Release in `mark_expiring`: a reader that saw a tombstone
144        // deleted also sees the mark set before the deletion.
145        self.expiring_marked.load(Ordering::Acquire)
146    }
147
148    /// Durably marks this bucket expiring, then stops serving its rows.
149    ///
150    /// Synced, because a column-family drop is durable at once while a
151    /// default write may be lost, which after a crash would make a bucket
152    /// whose tombstones are gone readable again.
153    fn mark_expiring(&self) -> Result<(), TypedStoreError> {
154        let mut batch = self.expiring.batch();
155        batch.insert_batch_tagged(&self.expiring, [((), ())])?;
156        batch.write_opt(&synced_write_options())?;
157        self.expiring_marked.store(true, Ordering::Release);
158        Ok(())
159    }
160}

202    /// Opens the historic-object buckets among the column families of `db`,
203    /// the perpetual database. `db_options` and `historic_root` must be the
204    /// ones `db` was opened with, and `objects` its live objects table.
205    ///
206    /// Before any query can reach them, finishes the expiry of the buckets an
207    /// interrupted prune left behind, oldest first, and fails if it cannot.
208    pub fn open(
209        db: Arc<Database>,
210        db_options: &DBOptions,
211        objects: DBMap<ObjectKey, StoreObjectWrapper>,
212        historic_root: &Path,
213    ) -> Result<Self, TypedStoreError> {
214        let existing_cfs = list_tables(db.path_for_pruning().to_path_buf())
215            .map_err(|e| TypedStoreError::RocksDB(format!("failed to list buckets: {e}")))?;
216
217        let mut buckets = BTreeMap::new();
218        for cf_name in &existing_cfs {
219            if let Some(epoch) = bucket_cf_epoch(HISTORIC_OBJECTS_CF_PREFIX, cf_name) {
220                buckets.insert(
221                    epoch,
222                    Arc::new(HistoricObjectsBucket::reopen(&db, cf_name)?),
223                );
224            }
225        }
226
227        let cf_options = bucket_cf_options(db_options).options;
228        if db.cf_handle(EARLIEST_RETAINED_CF).is_none() {
229            db.create_cf(EARLIEST_RETAINED_CF, &cf_options)?;
230        }
231        let earliest_retained_table: DBMap<(), EpochId> = DBMap::reopen(
232            &db,
233            Some(EARLIEST_RETAINED_CF),
234            &ReadWriteOptions::default(),
235            true,
236        )?;
237        let earliest_retained = earliest_retained_table.get(&())?.unwrap_or(0);
238
239        // A prune persists the floor before it marks anything, so a bucket
240        // below the floor is an unfinished expiry even without its marker.
241        // `EpochBuckets::open` would just drop it, leaving its tombstones
242        // in the live `objects` table for good.
243        //
244        // Oldest first: a bounded read that stops at one of this bucket's
245        // deleted tombstones falls through to the older buckets, so those must be
246        // gone first.
247        let interrupted: Vec<(EpochId, Arc<HistoricObjectsBucket>)> = buckets
248            .iter()
249            .filter(|(&epoch, bucket)| epoch < earliest_retained || bucket.is_expiring())
250            .map(|(&epoch, bucket)| (epoch, bucket.clone()))
251            .collect();
252        for (epoch, bucket) in interrupted {
253            let cf_name = bucket_cf_name(HISTORIC_OBJECTS_CF_PREFIX, epoch);
254            if let Err(e) = Self::expire_bucket(&objects, epoch, &bucket, &AtomicBool::new(false)) {
255                // Named so that an operator whose node will not start can
256                // reach the column family with external tooling.
257                error!(
258                    epoch,
259                    cf_name, "cannot finish the interrupted expiry of a historic bucket: {e}"
260                );
261                return Err(e);
262            }
263            buckets.remove(&epoch);
264            info!(
265                epoch,
266                "dropping the bucket of an interrupted expiry at open"
267            );
268            if let Err(e) = db.drop_cf(&cf_name) {
269                warn!(
270                    epoch,
271                    "failed to drop an expiring bucket column family: {e}"
272                );
273            }
274        }
275
276        let buckets = EpochBuckets::open(
277            db,
278            "historic objects",
279            HISTORIC_OBJECTS_CF_PREFIX,
280            cf_options,
281            Some(BucketPaths::new(historic_root, BUCKET_DIR)),
282            earliest_retained_table,
283            buckets,
284        )?;
285        Ok(Self {
286            buckets,
287            objects,
288            objects_pruned_through: None,
289            backlog_swept: AtomicBool::new(false),
290            backlog_sweep_epoch: OnceLock::new(),
291            stopping: AtomicBool::new(false),
292            #[cfg(test)]
293            bucket_walks: std::sync::atomic::AtomicU64::new(0),
294        })
295    }

382    /// The object `epoch`'s bucket holds under `key`, `None` when it holds
383    /// none or there is no bucket for `epoch`. Reads only that bucket, so the
384    /// answer does not depend on how many epochs the node retains.
385    pub fn get_in_epoch(&self, epoch: EpochId, key: &ObjectKey) -> IotaResult<Option<Object>> {
386        let Some(bucket) = self.buckets.get(epoch) else {
387            return Ok(None);
388        };
389        match bucket.objects.get(key) {
390            // The bucket was dropped during the read.
391            Err(TypedStoreError::UnregisteredColumn(_)) => Ok(None),
392            read => read.map_err(|e| IotaError::Storage(e.to_string())),
393        }
394    }

396    /// Fills each `None` in `objects` — the live table's answers for `keys`, in
397    /// the same order — with the version the buckets hold under that key.
398    ///
399    /// Only for reads that serve a response. Execution reads current versions,
400    /// which never leave the live table, so a miss there is a bug and must stay
401    /// one. Validation at a commit reads earlier versions of its own epoch
402    /// through [`Self::get_in_epoch`] instead.
403    ///
404    /// Reads each bucket once for all the keys still missing, starting with
405    /// `likely_epoch`'s, then newest first. Pass the epoch that superseded
406    /// the versions where it is known, such as a transaction's own epoch for
407    /// its inputs.
408    pub fn fill_missing(
409        &self,
410        keys: &[ObjectKey],
411        objects: &mut [Option<Object>],
412        likely_epoch: Option<EpochId>,
413    ) -> IotaResult<()> {
414        let mut pending: Vec<usize> = (0..keys.len())
415            .filter(|&index| objects[index].is_none())
416            .collect();
417        if pending.is_empty() {
418            return Ok(());
419        }
420        let mut buckets: Vec<(EpochId, Arc<HistoricObjectsBucket>)> = self
421            .buckets
422            .iter_with_epoch(true)
423            .into_iter()
424            .filter(|(_, bucket)| !bucket.is_expiring())
425            .collect();
426        if let Some(position) =
427            likely_epoch.and_then(|likely| buckets.iter().position(|(epoch, _)| *epoch == likely))
428        {
429            let likely = buckets.remove(position);
430            buckets.insert(0, likely);
431        }
432        for (_, bucket) in buckets {
433            if pending.is_empty() {
434                break;
435            }
436            let found = absent_if_dropped(
437                bucket
438                    .objects
439                    .multi_get(pending.iter().map(|&index| &keys[index])),
440            )
441            .map_err(|e| IotaError::Storage(e.to_string()))?;
442            // Empty for a dropped bucket, which leaves every key pending.
443            if found.is_empty() {
444                continue;
445            }
446            let mut still_pending = Vec::with_capacity(pending.len());
447            for (index, object) in pending.into_iter().zip(found) {
448                match object {
449                    Some(object) => objects[index] = Some(object),
450                    None => still_pending.push(index),
451                }
452            }
453            pending = still_pending;
454        }
455        Ok(())
456    }

458    /// The newest relocated version of `id` at or below `version`, `None` if
459    /// there is none (or its bucket has since been dropped).
460    ///
461    /// A caller must read the live `objects` table first (see
462    /// `HistoricObjectsBucket::tombstones`) and take the newer of the two
463    /// answers: this ignores tombstones, so only a live tombstone newer than
464    /// what it returns means the object is gone.
465    pub fn find_lt_or_eq_version(
466        &self,
467        id: ObjectId,
468        version: Version,
469    ) -> IotaResult<Option<Object>> {
470        #[cfg(test)]
471        self.bucket_walks.fetch_add(1, Ordering::Relaxed);
472        // Versions are relocated in increasing order, so the newest bucket
473        // with a match holds the newest version.
474        for bucket in self.readable_buckets(true) {
475            let newest = absent_if_dropped(
476                bucket
477                    .objects
478                    .safe_range_iter_reversed(ObjectKey::min_for_id(&id)..=ObjectKey(id, version))
479                    .next()
480                    .transpose(),
481            )
482            .map_err(|e| IotaError::Storage(e.to_string()))?;
483            if let Some((_, object)) = newest {
484                return Ok(Some(object));
485            }
486        }
487        Ok(None)
488    }

503    /// Keeps the buckets of `current_epoch` and the `epochs_to_retain` epochs
504    /// below it, drops older ones, and deletes the tombstones each
505    /// dropped epoch recorded. Returns the earliest epoch still retained and
506    /// how many buckets were dropped, `None` when there is no bucket at all.
507    ///
508    /// Blocks for as long as the deletions and drops take, so an async caller
509    /// must use `spawn_blocking`. Queries keep being answered meanwhile, from
510    /// the buckets not marked expiring.
511    pub fn prune(
512        &self,
513        current_epoch: EpochId,
514        epochs_to_retain: u64,
515    ) -> IotaResult<Option<PrunedBuckets>> {
516        self.buckets
517            .prune(current_epoch, epochs_to_retain, |epoch, bucket| {
518                Self::expire_bucket(&self.objects, epoch, bucket, &self.stopping).map_err(|e| {
519                    TypedStoreError::RocksDB(format!("expiring the bucket of epoch {epoch}: {e}"))
520                })
521            })
522            .map_err(|e| IotaError::Storage(e.to_string()))
523    }

540    /// Marks `bucket` expiring, then deletes the tombstones it recorded from
541    /// the live `objects` table.
542    ///
543    /// Safe to run again on the same bucket: the marker is rewritten as it
544    /// was and a tombstone already deleted is deleted again.
545    fn expire_bucket(
546        objects: &DBMap<ObjectKey, StoreObjectWrapper>,
547        epoch: EpochId,
548        bucket: &Arc<HistoricObjectsBucket>,
549        stopping: &AtomicBool,
550    ) -> Result<(), TypedStoreError> {
551        bucket.mark_expiring()?;
552
553        // Synced: a column-family drop is durable at once, so a lost deletion
554        // would leave tombstones with nothing left to delete them.
555        let delete = |tombstones: Vec<ObjectKey>| -> Result<(), TypedStoreError> {
556            let mut batch = objects.batch();
557            batch.delete_batch(objects, tombstones)?;
558            batch.write_opt(&synced_write_options())
559        };
560
561        // A version left below a tombstone in the live table would become the
562        // newest again once the tombstone is deleted, bringing the object
563        // back. That only happens where the backlog sweep did not relocate it,
564        // so such a tombstone is kept. Tombstones below it don't count: an
565        // object wrapped and deleted in one epoch leaves two in this bucket.
566        //
567        // The bucket lists its tombstones by object, so each object's live
568        // rows are scanned once, for its lowest live version: an object
569        // wrapped and unwrapped many times in the epoch has as many tombstones
570        // in the live table, and scanning below each of them would be
571        // quadratic in their number.
572        let lowest_live_version =
573            |object_id: &ObjectId| -> Result<Option<Version>, TypedStoreError> {
574                for row in objects.safe_range_iter(
575                    ObjectKey::min_for_id(object_id)..=ObjectKey::max_for_id(object_id),
576                ) {
577                    let (key, row) = row?;
578                    if matches!(row.migrate().into_inner(), StoreObject::Value(_)) {
579                        return Ok(Some(key.1));
580                    }
581                }
582                Ok(None)
583            };
584
585        let mut deleted = 0;
586        let mut kept = 0;
587        let mut batch = Vec::with_capacity(TOMBSTONE_DELETE_BATCH_SIZE);
588        let mut lowest_live: Option<(ObjectId, Option<Version>)> = None;
589        for row in bucket.tombstones.safe_iter() {
590            let (key, ()) = row?;
591            let lowest = match lowest_live {
592                Some((object_id, lowest)) if object_id == key.0 => lowest,
593                _ => {
594                    let lowest = lowest_live_version(&key.0)?;
595                    lowest_live = Some((key.0, lowest));
596                    lowest
597                }
598            };
599            if lowest.is_some_and(|lowest| lowest < key.1) {
600                kept += 1;
601                continue;
602            }
603            batch.push(key);
604            if batch.len() == TOMBSTONE_DELETE_BATCH_SIZE {
605                deleted += batch.len();
606                delete(std::mem::replace(
607                    &mut batch,
608                    Vec::with_capacity(TOMBSTONE_DELETE_BATCH_SIZE),
609                ))?;
610                // The bucket stays marked expiring, so the next open
611                // finishes the expiry from here.
612                if stopping.load(Ordering::Acquire) {
613                    return Err(TypedStoreError::RocksDB(format!(
614                        "the expiry of epoch {epoch}'s bucket was stopped by a shutdown"
615                    )));
616                }
617            }
618        }
619        deleted += batch.len();
620        delete(batch)?;
621
622        if kept > 0 {
623            error!(
624                epoch,
625                kept,
626                "kept tombstones that still have a live version below them: those \
627                 objects were superseded before this build and the backlog sweep did not \
628                 relocate them. They stay deleted, which is why the tombstones stay too"
629            );
630        }
631        info!(
632            epoch,
633            tombstones = deleted,
634            kept,
635            "expired a historic bucket"
636        );
637        Ok(())
638    }

640    /// Writes a row of the wrong type into `epoch`'s tombstones table, so
641    /// that expiring that bucket fails.
642    #[cfg(test)]
643    pub(super) fn corrupt_recorded_tombstones_for_testing(

645        epoch: EpochId,

659    /// One page of the rows of `cf_name` if it is one of this store's column
660    /// families, `None` otherwise. For the `iota-tool` table dump, which
661    /// cannot reach these through `AuthorityPerpetualTables`.
662    ///
663    /// Tombstone and expiring-marker rows are prefixed by table name.
664    /// `db` may be a read-only or secondary handle.
665    pub fn dump_column_family(
666        db: &Arc<Database>,
667        cf_name: &str,
668        page_size: u16,
669        page_number: usize,
670    ) -> Result<Option<BTreeMap<String, String>>, TypedStoreError> {
671        if bucket_cf_epoch(HISTORIC_OBJECTS_CF_PREFIX, cf_name).is_some() {
672            let bucket = HistoricObjectsBucket::reopen(db, cf_name)?;
673            bucket.objects.try_catch_up_with_primary()?;
674            bucket.tombstones.try_catch_up_with_primary()?;
675            bucket.expiring.try_catch_up_with_primary()?;
676            let rows = dump_rows("", bucket.objects.safe_iter())
677                .chain(dump_rows("tombstone:", bucket.tombstones.safe_iter()))
678                .chain(dump_rows("expiring:", bucket.expiring.safe_iter()));
679            return dump_page(rows, page_size, page_number).map(Some);
680        }
681        if cf_name == EARLIEST_RETAINED_CF {
682            return dump_retention_floor(db, cf_name, page_size, page_number).map(Some);
683        }
684        Ok(None)
685    }

688/// Like [`iota_types::storage::get_transaction_input_objects`], but also
689/// reads the relocated versions.
690pub fn get_transaction_input_objects(
691    object_store: &dyn ObjectStore,
692    historic_objects: &HistoricObjects,
693    effects: &TransactionEffects,
694) -> IotaResult<Vec<Object>> {
695    let keys = effects
696        .modified_at_versions()
697        .into_iter()
698        .map(|modified| ObjectKey(*modified.object_id(), modified.version()))
699        .collect::<Vec<_>>();
700    // The transaction superseded its inputs, in the epoch it executed in.
701    multi_get_objects_with_historic_fallback(
702        object_store,
703        historic_objects,
704        &keys,
705        effects.transaction_digest(),
706        Some(effects.epoch()),
707    )
708}

710/// Like [`iota_types::storage::get_transaction_output_objects`], but also
711/// reads the relocated versions.
712pub fn get_transaction_output_objects(
713    object_store: &dyn ObjectStore,
714    historic_objects: &HistoricObjects,
715    effects: &TransactionEffects,
716) -> IotaResult<Vec<Object>> {
717    let keys = effects
718        .all_changed_objects()
719        .into_iter()
720        .map(|(changed, _kind)| ObjectKey::from(*changed.reference()))
721        .collect::<Vec<_>>();
722    multi_get_objects_with_historic_fallback(
723        object_store,
724        historic_objects,
725        &keys,
726        effects.transaction_digest(),
727        None,
728    )
729}

731/// The objects at exactly `keys`, erroring on any the live table and the
732/// buckets both lack. `likely_epoch` is as for
733/// [`HistoricObjects::fill_missing`].
734fn multi_get_objects_with_historic_fallback(
735    object_store: &dyn ObjectStore,
736    historic_objects: &HistoricObjects,
737    keys: &[ObjectKey],
738    transaction_digest: &TransactionDigest,
739    likely_epoch: Option<EpochId>,
740) -> IotaResult<Vec<Object>> {
741    let mut objects = object_store.multi_get_objects_by_key(keys);
742    historic_objects.fill_missing(keys, &mut objects, likely_epoch)?;
743    objects
744        .into_iter()
745        .zip(keys)
746        .map(|(object, key)| {
747            object.ok_or_else(|| {
748                IotaError::Storage(format!(
749                    "missing object key {key:?} from tx {transaction_digest}"
750                ))
751            })
752        })
753        .collect()
754}
