35/// Column-family prefix of the historic ledger buckets; a bucket's family
36/// is `{prefix}{epoch}`.
37const HISTORIC_LEDGER_CF_PREFIX: &str = "hist_ledger_e";
38
39/// The directory of each epoch, under the historic root, that this
40/// store's buckets keep their files in.
41const BUCKET_DIR: &str = "ledger";
42
43/// Tags of the tables inside a bucket's column family. Do not reuse a tag
44/// for a different table: mark it retired in a comment instead, so an older
45/// bucket's rows can never be read as the wrong type.
46const DB_PREFIX_HISTORIC_TRANSACTIONS: u8 = 0;
47const DB_PREFIX_HISTORIC_EFFECTS: u8 = 1;
48const DB_PREFIX_HISTORIC_EXECUTED_EFFECTS: u8 = 2;
49const DB_PREFIX_HISTORIC_EVENTS: u8 = 3;
50const DB_PREFIX_HISTORIC_TX_TO_CHECKPOINT: u8 = 4;
51const DB_PREFIX_HISTORIC_ATTESTATION_RECORDS: u8 = 5;
52
53/// Column family holding the earliest-retained-epoch marker
54/// [`EpochBuckets`] persists on a prune; empty until the first prune.
55///
56/// The name must not begin with [`HISTORIC_LEDGER_CF_PREFIX`], since that is
57/// how a bucket's column family is told from every other one in this
58/// database.
59const EARLIEST_RETAINED_CF: &str = "hist_ledger_retention";
60
61/// A value read from a bucket, with that bucket; `None` where no bucket held
62/// it.
63type FoundIn<V> = Option<(Arc<HistoricLedgerBucket>, V)>;
64
65/// The keys to read from one bucket, each with its position in the batch.
66type KeysOfBucket<K> = (Arc<HistoricLedgerBucket>, Vec<(usize, K)>);
67
68/// One epoch's transaction history. All rows of a transaction are in the same
69/// bucket.
70pub struct HistoricLedgerBucket {
71    pub(crate) transactions: TaggedDBMap<TransactionDigest, TrustedTransaction>,
72    pub(crate) effects: TaggedDBMap<TransactionEffectsDigest, TransactionEffects>,
73    pub(crate) executed_effects: TaggedDBMap<TransactionDigest, TransactionEffectsDigest>,
74    pub(crate) events: TaggedDBMap<TransactionDigest, TransactionEvents>,
75    pub(crate) tx_to_checkpoint: TaggedDBMap<TransactionDigest, CheckpointSequenceNumber>,
76    pub(crate) attestation_records: TaggedDBMap<TransactionDigest, AttestationRecord>,
77}

134/// Transaction history, bucketed by the epoch that executed it.
135///
136/// A bucket's existence does **not** mean its epoch has been executed: state
137/// sync writes a checkpoint's transactions and effects into the bucket of its
138/// epoch ahead of execution, also across epoch boundaries. Retention must
139/// therefore count from the epoch being executed, not from the newest bucket;
140/// see [`Self::prune`].
141pub struct HistoricLedger {
142    buckets: EpochBuckets<HistoricLedgerBucket>,
143
144    /// Counts the bucket walks this store has done.
145    #[cfg(test)]
146    bucket_walks: std::sync::atomic::AtomicU64,
147}

150    /// The `(name, options)` pairs of the column families this store needs,
151    /// for the perpetual store to open alongside its own tables. See
152    /// `epoch_buckets::extra_column_family_options`.
153    pub fn extra_column_family_options(
154        perpetual_path: &Path,
155        db_options: &DBOptions,
156        historic_root: &Path,
157    ) -> Vec<(String, DBOptions)> {
158        extra_column_family_options(
159            perpetual_path,
160            db_options,
161            HISTORIC_LEDGER_CF_PREFIX,
162            EARLIEST_RETAINED_CF,
163            Some(&BucketPaths::new(historic_root, BUCKET_DIR)),
164        )
165    }

167    /// Opens the historic-ledger buckets among the column families of `db`,
168    /// the perpetual database. `db_options` and `historic_root` must be the
169    /// ones `db` was opened with.
170    pub fn open(
171        db: Arc<Database>,
172        db_options: &DBOptions,
173        historic_root: &Path,
174    ) -> Result<Self, TypedStoreError> {
175        let existing_cfs = list_tables(db.path_for_pruning().to_path_buf())
176            .map_err(|e| TypedStoreError::RocksDB(format!("failed to list buckets: {e}")))?;
177
178        let mut buckets = BTreeMap::new();
179        for cf_name in &existing_cfs {
180            if let Some(epoch) = bucket_cf_epoch(HISTORIC_LEDGER_CF_PREFIX, cf_name) {
181                buckets.insert(epoch, Arc::new(HistoricLedgerBucket::reopen(&db, cf_name)?));
182            }
183        }
184
185        let cf_options = bucket_cf_options(db_options).options;
186        if db.cf_handle(EARLIEST_RETAINED_CF).is_none() {
187            db.create_cf(EARLIEST_RETAINED_CF, &cf_options)?;
188        }
189        let earliest_retained_table: DBMap<(), EpochId> = DBMap::reopen(
190            &db,
191            Some(EARLIEST_RETAINED_CF),
192            &ReadWriteOptions::default(),
193            true,
194        )?;
195
196        let buckets = EpochBuckets::open(
197            db,
198            "historic ledger",
199            HISTORIC_LEDGER_CF_PREFIX,
200            cf_options,
201            Some(BucketPaths::new(historic_root, BUCKET_DIR)),
202            earliest_retained_table,
203            buckets,
204        )?;
205        Ok(Self {
206            buckets,
207            #[cfg(test)]
208            bucket_walks: std::sync::atomic::AtomicU64::new(0),
209        })
210    }

254    /// Keeps the buckets of `current_epoch` and the `epochs_to_retain` epochs
255    /// below it, drops older ones, and returns the earliest epoch retained and
256    /// how many buckets were dropped.
257    ///
258    /// `current_epoch` must be the epoch this node is executing, not the
259    /// newest bucket (see [`HistoricLedger`]); buckets above it are left
260    /// alone. Blocks for as long as the drops take, so a caller on an async
261    /// runtime must use `spawn_blocking`.
262    ///
263    /// `before_drop` runs before each bucket is dropped, once the earliest
264    /// epoch retained has moved past it, so that what advertises the retained
265    /// range can be moved first.
266    pub fn prune(
267        &self,
268        current_epoch: EpochId,
269        epochs_to_retain: u64,
270        before_drop: &dyn Fn(),
271    ) -> IotaResult<Option<PrunedBuckets>> {
272        self.buckets
273            // No live table holds rows that depend on a bucket.
274            .prune(current_epoch, epochs_to_retain, |_, _| {
275                before_drop();
276                Ok(())
277            })
278            .map_err(|e| IotaError::Storage(e.to_string()))
279    }

281    /// Compacts the buckets of the epochs below `current_epoch` that are not
282    /// compacted yet (see
283    /// `EpochBuckets::compact_closed`, which asks
284    /// `stop` before each bucket). Blocks while
285    /// the compactions run, so async callers must use `spawn_blocking`.
286    pub fn compact_closed_buckets(
287        &self,
288        current_epoch: EpochId,
289        stop: &dyn Fn() -> bool,
290    ) -> IotaResult<usize> {
291        self.buckets
292            .compact_closed(current_epoch, stop)
293            .map_err(|e| IotaError::Storage(e.to_string()))
294    }

296    /// For each of `keys`, the newest bucket holding it in the table `table`
297    /// picks, with the value it holds there; `None` where no bucket does.
298    ///
299    /// One batched read per bucket, newest first, for the keys no newer
300    /// bucket held, so a batch costs a read per bucket rather than one per
301    /// key and bucket. A bucket dropped meanwhile holds nothing.
302    fn multi_find<K, V>(
303        &self,
304        keys: &[K],
305        table: impl Fn(&HistoricLedgerBucket) -> &TaggedDBMap<K, V>,
306    ) -> IotaResult<Vec<FoundIn<V>>>
307    where
308        K: Serialize + DeserializeOwned,
309        V: Serialize + DeserializeOwned,
310    {
311        #[cfg(test)]
312        self.count_walk();
313        let mut found: Vec<FoundIn<V>> = keys.iter().map(|_| None).collect();
314        let mut pending: Vec<usize> = (0..keys.len()).collect();
315        for bucket in self.buckets.iter(true) {
316            if pending.is_empty() {
317                break;
318            }
319            let values = absent_if_dropped(
320                table(&bucket).multi_get(pending.iter().map(|&index| &keys[index])),
321            )
322            .map_err(|e| IotaError::Storage(e.to_string()))?;
323            // Empty for a dropped bucket, which leaves every key pending.
324            if values.is_empty() {
325                continue;
326            }
327            let mut still_pending = Vec::new();
328            for (index, value) in pending.into_iter().zip(values) {
329                match value {
330                    Some(value) => found[index] = Some((bucket.clone(), value)),
331                    None => still_pending.push(index),
332                }
333            }
334            pending = still_pending;
335        }
336        Ok(found)
337    }

339    /// For each `(bucket, key)` of `lookups`, the value `bucket` holds under
340    /// `key` in the table `table` picks; `None` for a `None` lookup. One
341    /// batched read per distinct bucket.
342    fn multi_get_in_buckets<K, V>(
343        lookups: Vec<FoundIn<K>>,
344        table: impl Fn(&HistoricLedgerBucket) -> &TaggedDBMap<K, V>,
345    ) -> IotaResult<Vec<Option<V>>>
346    where
347        K: Serialize + DeserializeOwned,
348        V: Serialize + DeserializeOwned,
349    {
350        let mut values: Vec<Option<V>> = lookups.iter().map(|_| None).collect();
351        let mut by_bucket: Vec<KeysOfBucket<K>> = Vec::new();
352        for (index, lookup) in lookups.into_iter().enumerate() {
353            let Some((bucket, key)) = lookup else {
354                continue;
355            };
356            match by_bucket
357                .iter_mut()
358                .find(|(other, _)| Arc::ptr_eq(other, &bucket))
359            {
360                Some((_, keys)) => keys.push((index, key)),
361                None => by_bucket.push((bucket, vec![(index, key)])),
362            }
363        }
364        for (bucket, keys) in by_bucket {
365            let found =
366                absent_if_dropped(table(&bucket).multi_get(keys.iter().map(|(_, key)| key)))
367                    .map_err(|e| IotaError::Storage(e.to_string()))?;
368            for ((index, _), value) in keys.into_iter().zip(found) {
369                values[index] = value;
370            }
371        }
372        Ok(values)
373    }

442    /// The values `keys` hold in `epoch`'s bucket, in the table `table`
443    /// picks; all `None` when there is no such bucket or it was dropped.
444    fn multi_get_in_epoch<K, V>(
445        &self,
446        epoch: EpochId,
447        keys: &[K],
448        table: impl Fn(&HistoricLedgerBucket) -> &TaggedDBMap<K, V>,
449    ) -> IotaResult<Vec<Option<V>>>
450    where
451        K: Serialize + DeserializeOwned,
452        V: Serialize + DeserializeOwned,
453    {
454        let Some(bucket) = self.buckets.get(epoch) else {
455            return Ok(keys.iter().map(|_| None).collect());
456        };
457        let found = absent_if_dropped(table(&bucket).multi_get(keys))
458            .map_err(|e| IotaError::Storage(e.to_string()))?;
459        // Empty for a dropped bucket.
460        if found.is_empty() {
461            return Ok(keys.iter().map(|_| None).collect());
462        }
463        Ok(found)
464    }

515    /// The bucket holding the execution record of `digest`, with its epoch;
516    /// `None` if this node has not executed it or the bucket was dropped.
517    ///
518    /// Read the transaction's effects, events and checkpoint from the returned
519    /// bucket rather than probing again. A transaction stored but not yet
520    /// executed is not found here; see [`Self::get_transaction`].
521    pub fn find_epoch(
522        &self,
523        digest: &TransactionDigest,
524    ) -> IotaResult<Option<(EpochId, Arc<HistoricLedgerBucket>)>> {
525        #[cfg(test)]
526        self.count_walk();
527        for (epoch, bucket) in self.buckets.iter_with_epoch(true) {
528            if absent_if_dropped(bucket.executed_effects.contains_key(digest))
529                .map_err(|e| IotaError::Storage(e.to_string()))?
530            {
531                return Ok(Some((epoch, bucket)));
532            }
533        }
534        Ok(None)
535    }

537    /// The effects of `digest` as this node executed it, `None` if no bucket
538    /// holds an execution record for it.
539    pub fn get_executed_effects(
540        &self,
541        digest: &TransactionDigest,
542    ) -> IotaResult<Option<TransactionEffects>> {
543        let Some((_, bucket)) = self.find_epoch(digest)? else {
544            return Ok(None);
545        };
546        let effects = absent_if_dropped(bucket.executed_effects.get(digest))
547            .map_err(|e| IotaError::Storage(e.to_string()))?
548            .map(|effects_digest| absent_if_dropped(bucket.effects.get(&effects_digest)))
549            .transpose()
550            .map_err(|e| IotaError::Storage(e.to_string()))?
551            .flatten();
552        Ok(effects)
553    }

577    /// The checkpoint that finalized `digest`, with the epoch that checkpoint
578    /// belongs to, `None` while the transaction is executed but not yet
579    /// finalized, and for a transaction no bucket holds.
580    pub fn get_transaction_checkpoint(
581        &self,
582        digest: &TransactionDigest,
583    ) -> IotaResult<Option<(EpochId, CheckpointSequenceNumber)>> {
584        let Some((epoch, bucket)) = self.find_epoch(digest)? else {
585            return Ok(None);
586        };
587        // A bucket's epoch is also the epoch of its checkpoints.
588        Ok(absent_if_dropped(bucket.tx_to_checkpoint.get(digest))
589            .map_err(|e| IotaError::Storage(e.to_string()))?
590            .map(|sequence| (epoch, sequence)))
591    }

593    /// [`Self::get_transaction_checkpoint`] for each of `digests`, in one
594    /// batched read per bucket, newest first, for the digests no newer bucket
595    /// held. A checkpoint is recorded in the bucket of the epoch that executed
596    /// the transaction, which is also the checkpoint's epoch.
597    pub fn multi_get_transaction_checkpoints(
598        &self,
599        digests: &[TransactionDigest],
600    ) -> IotaResult<Vec<Option<(EpochId, CheckpointSequenceNumber)>>> {
601        #[cfg(test)]
602        self.count_walk();
603        let mut found = vec![None; digests.len()];
604        let mut pending: Vec<usize> = (0..digests.len()).collect();
605        for (epoch, bucket) in self.buckets.iter_with_epoch(true) {
606            if pending.is_empty() {
607                break;
608            }
609            let values = absent_if_dropped(
610                bucket
611                    .tx_to_checkpoint
612                    .multi_get(pending.iter().map(|&index| &digests[index])),
613            )
614            .map_err(|e| IotaError::Storage(e.to_string()))?;
615            // Empty for a dropped bucket, which leaves every digest pending.
616            if values.is_empty() {
617                continue;
618            }
619            let mut still_pending = Vec::with_capacity(pending.len());
620            for (index, sequence) in pending.into_iter().zip(values) {
621                match sequence {
622                    Some(sequence) => found[index] = Some((epoch, sequence)),
623                    None => still_pending.push(index),
624                }
625            }
626            pending = still_pending;
627        }
628        Ok(found)
629    }
630
631    /// The transaction stored under `digest`, `None` if no bucket holds it.
632    /// Also finds a transaction stored before it was executed.
633    pub fn get_transaction(
634        &self,
635        digest: &TransactionDigest,
636    ) -> IotaResult<Option<TrustedTransaction>> {
637        #[cfg(test)]
638        self.count_walk();
639        for bucket in self.buckets.iter(true) {
640            if let Some(transaction) = absent_if_dropped(bucket.transactions.get(digest))
641                .map_err(|e| IotaError::Storage(e.to_string()))?
642            {
643                return Ok(Some(transaction));
644            }
645        }
646        Ok(None)
647    }

669    /// Whether `epoch`'s bucket holds every one of `transactions` and
670    /// `effects`, read from the store. `false` when there is no bucket for
671    /// `epoch`.
672    pub fn holds_in_epoch(
673        &self,
674        epoch: EpochId,
675        transactions: &[TransactionDigest],
676        effects: &[TransactionEffectsDigest],
677    ) -> Result<bool, TypedStoreError> {
678        let Some(bucket) = self.buckets.get(epoch) else {
679            return Ok(false);
680        };
681        let holds = || -> Result<bool, TypedStoreError> {
682            // Outputs that are only in memory usually miss all of them, so
683            // one lookup settles most calls before the full check.
684            if let Some(last) = effects.last() {
685                if !bucket.effects.contains_key(last)? {
686                    return Ok(false);
687                }
688            }
689            Ok(bucket
690                .transactions
691                .multi_contains_keys(transactions)?
692                .into_iter()
693                .chain(bucket.effects.multi_contains_keys(effects)?)
694                .all(|held| held))
695        };
696        match holds() {
697            // The bucket was dropped during the read.
698            Err(TypedStoreError::UnregisteredColumn(_)) => Ok(false),
699            other => other,
700        }
701    }

703    /// Whether any bucket holds the effects under `digest`.
704    pub fn contains_effects(&self, digest: &TransactionEffectsDigest) -> IotaResult<bool> {
705        #[cfg(test)]
706        self.count_walk();
707        for bucket in self.buckets.iter(true) {
708            if absent_if_dropped(bucket.effects.contains_key(digest))
709                .map_err(|e| IotaError::Storage(e.to_string()))?
710            {
711                return Ok(true);
712            }
713        }
714        Ok(false)
715    }

717    /// One page of the rows of `cf_name` if it is one of this store's column
718    /// families, `None` otherwise. For the `iota-tool` table dump, which
719    /// cannot reach these through `AuthorityPerpetualTables`.
720    ///
721    /// Rows are prefixed by table name. `db` may be a read-only or secondary
722    /// handle.
723    pub fn dump_column_family(
724        db: &Arc<Database>,
725        cf_name: &str,
726        page_size: u16,
727        page_number: usize,
728    ) -> Result<Option<BTreeMap<String, String>>, TypedStoreError> {
729        if bucket_cf_epoch(HISTORIC_LEDGER_CF_PREFIX, cf_name).is_some() {
730            let bucket = HistoricLedgerBucket::reopen(db, cf_name)?;
731            bucket.transactions.try_catch_up_with_primary()?;
732            bucket.effects.try_catch_up_with_primary()?;
733            bucket.executed_effects.try_catch_up_with_primary()?;
734            bucket.events.try_catch_up_with_primary()?;
735            bucket.tx_to_checkpoint.try_catch_up_with_primary()?;
736            bucket.attestation_records.try_catch_up_with_primary()?;
737            let rows = dump_rows("transaction:", bucket.transactions.safe_iter())
738                .chain(dump_rows("effects:", bucket.effects.safe_iter()))
739                .chain(dump_rows(
740                    "executed_effects:",
741                    bucket.executed_effects.safe_iter(),
742                ))
743                .chain(dump_rows("events:", bucket.events.safe_iter()))
744                .chain(dump_rows(
745                    "tx_to_checkpoint:",
746                    bucket.tx_to_checkpoint.safe_iter(),
747                ))
748                .chain(dump_rows(
749                    "attestation_record:",
750                    bucket.attestation_records.safe_iter(),
751                ));
752            return dump_page(rows, page_size, page_number).map(Some);
753        }
754        if cf_name == EARLIEST_RETAINED_CF {
755            return dump_retention_floor(db, cf_name, page_size, page_number).map(Some);
756        }
757        Ok(None)
758    }
