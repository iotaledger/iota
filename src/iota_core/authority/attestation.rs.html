49impl AuthorityState {
50    /// Dry-runs a verified transaction and returns `AttestationData` together
51    /// with the owned objects validated during execution, ready to be wrapped
52    /// in a validator attestation and attached to the transaction before
53    /// submitting it to consensus.
54    ///
55    /// Shares the validation pipeline with
56    /// `handle_transaction_validation_checks`
57    /// via [`Self::run_validation_checks`] (deny checks, input object loading,
58    /// input checks, coin deny list), and then replaces the
59    /// `authenticate_transaction` call with
60    /// `authenticate_then_execute_transaction_to_effects` so that Move
61    /// authentication and transaction execution run together and produce full
62    /// `TransactionEffects` including the accurate `computation_cost`.
63    pub(crate) fn attest_transaction(
64        &self,
65        transaction: &VerifiedTransaction,
66        epoch_store: &Arc<AuthorityPerEpochStore>,
67        deny_config: &dyn DenyRuleConfig,
68    ) -> IotaResult<(AttestationData, Vec<ObjectReference>)> {
69        let protocol_config = epoch_store.protocol_config();
70        let tx = transaction.data().transaction();
71
72        let ValidationOutputs {
73            per_authenticator_inputs,
74            move_authenticators,
75            gas_status,
76            tx_checked_input_objects,
77            per_authenticator_checked_inputs,
78        } = self.run_validation_checks(transaction, epoch_store, deny_config, true)?;
79
80        let owned_objects = tx_checked_input_objects.inner().filter_owned_objects();
81
82        let epoch_id = epoch_store.epoch_start_config().epoch_data().epoch_id();
83        let epoch_start_timestamp = epoch_store
84            .epoch_start_config()
85            .epoch_data()
86            .epoch_start_timestamp();
87        let backing_store = self.get_backing_store();
88        let (kind, signer, gas_data) = tx.execution_parts();
89        let tx_digest = *transaction.digest();
90
91        // The dry run meters authentication and body against the whole budget,
92        // as execution does, and a failed authentication is refused unpaid. So
93        // the authenticators first run on their own at the signing cap, as at
94        // admission: one that fails or exceeds `max_auth_gas` is refused before
95        // budget-sized work is done on it.
96        if !move_authenticators.is_empty() {
97            let function_refs = extract_auth_fun_refs(signer, gas_data.owner, |address| {
98                move_authenticators
99                    .iter()
100                    .zip(per_authenticator_checked_inputs.iter())
101                    .find(|(move_authenticator, _)| move_authenticator.address() == address)
102                    .map(|(_, (_, auth_fun_ref))| auth_fun_ref.clone())
103            });
104            let authenticators = move_authenticators
105                .iter()
106                .zip(per_authenticator_checked_inputs.iter())
107                .map(|(move_authenticator, (input_objects, function_ref))| {
108                    (
109                        (*move_authenticator).to_owned(),
110                        function_ref.clone(),
111                        CheckedInputObjects::new_with_checked_transaction_inputs(
112                            input_objects.inner().clone(),
113                        ),
114                    )
115                })
116                .collect();
117            let capped_gas_status = IotaGasStatus::new(
118                protocol_config.max_auth_gas(),
119                gas_data.price,
120                epoch_store.reference_gas_price(),
121                protocol_config,
122            )?;
123            self.authenticate_before_consensus(
124                epoch_store,
125                transaction,
126                capped_gas_status,
127                authenticators,
128                function_refs,
129            )?;
130        }
131
132        // Capture the `InnerTemporaryStore` from the dry-run: its object maps are the
133        // source of truth for the input versions the attestor observed. The account
134        // objects are read outside the store, so their observed versions are captured
135        // separately.
136        let (inner_temp_store, effects, authentication_failed, account_object_refs) =
137            if move_authenticators.is_empty() {
138                // No Move authentication, execute directly.
139                let (inner_temp_store, _, effects, _, _) =
140                    epoch_store.executor().execute_transaction_to_effects(
141                        backing_store.as_ref(),
142                        protocol_config,
143                        self.metrics.limits_metrics.clone(),
144                        false, // expensive_checks
145                        self.config.certificate_deny_config.certificate_deny_set(),
146                        &epoch_id,
147                        epoch_start_timestamp,
148                        tx_checked_input_objects,
149                        gas_data,
150                        gas_status,
151                        kind,
152                        signer,
153                        tx_digest,
154                        &mut None,
155                    );
156                // This branch has no Move authentication, so it can never fail auth.
157                (inner_temp_store, effects, false, Vec::new())
158            } else {
159                // Recover the `AuthenticatorFunctionRefForExecution` for each authenticator and
160                // run auth + execution in one pass.
161
162                let (account_object_refs, funcs_for_exec): (Vec<_>, Vec<_>) = move_authenticators
163                    .iter()
164                    .zip(per_authenticator_inputs)
165                    .map(|(ma, (_, account_object))| {
166                        // Immutable accounts cannot drift, so they are not recorded.
167                        let account_object_ref = account_object
168                            .as_object()
169                            .filter(|object| !object.is_immutable())
170                            .map(|object| object.object_ref());
171                        let (id, seq, digest) = ma.object_to_authenticate_components()?;
172                        let signer_addr = ma.address();
173                        self.check_move_account_for_validation(
174                            id,
175                            seq,
176                            digest,
177                            account_object,
178                            &signer_addr,
179                            protocol_config,
180                        )
181                        .map(|function_ref| (account_object_ref, function_ref))
182                    })
183                    .collect::<IotaResult<Vec<_>>>()?
184                    .into_iter()
185                    .unzip();
186                let account_object_refs: Vec<_> =
187                    account_object_refs.into_iter().flatten().collect();
188
189                // Union the checked transaction + authenticator input objects for
190                // execution. Borrow the checked inputs here before they are consumed
191                // by `authenticators_for_exec` below.
192                let checked_for_union: Vec<&CheckedInputObjects> =
193                    std::iter::once(&tx_checked_input_objects)
194                        .chain(per_authenticator_checked_inputs.iter().map(|(c, _)| c))
195                        .collect();
196                let auth_and_tx_checked =
197                    iota_transaction_checks::aggregate_authenticator_input_objects(
198                        &checked_for_union,
199                    )?;
200
201                let authenticators_for_exec = move_authenticators
202                    .into_iter()
203                    .zip(funcs_for_exec)
204                    .zip(per_authenticator_checked_inputs)
205                    .map(
206                        |((ma, function_ref), (input_objects, _))| MoveAuthenticatorForExecution {
207                            authenticator: ma.to_owned(),
208                            function_ref,
209                            input_objects,
210                        },
211                    )
212                    .collect::<Vec<_>>();
213
214                let (sender_authenticator_function_ref, sponsor_authenticator_function_ref) =
215                    extract_auth_fun_refs(signer, gas_data.owner, |address| {
216                        authenticators_for_exec
217                            .iter()
218                            .find(|a| a.authenticator.address() == address)
219                            .map(|a| a.function_ref.authenticator_function_ref.clone())
220                    });
221
222                let (sender_auth_digest, sponsor_auth_digest) =
223                    transaction.data().compute_auth_digests()?;
224
225                // Serialize the TransactionData for the auth context before decomposing.
226                let tx_data_bytes =
227                    bcs::to_bytes(tx).expect("TransactionData serialization cannot fail");
228
229                let auth_context_data = AuthContextData {
230                    transaction_data_bytes: tx_data_bytes,
231                    sender_auth_digest,
232                    sponsor_auth_digest,
233                    sender_authenticator_function_ref,
234                    sponsor_authenticator_function_ref,
235                };
236
237                let (inner_temp_store, _, effects, _, _, authentication_error) = epoch_store
238                    .executor()
239                    .authenticate_then_execute_transaction_to_effects(
240                        backing_store.as_ref(),
241                        protocol_config,
242                        self.metrics.limits_metrics.clone(),
243                        false, // expensive_checks
244                        self.config.certificate_deny_config.certificate_deny_set(),
245                        &epoch_id,
246                        epoch_start_timestamp,
247                        gas_data,
248                        gas_status,
249                        MoveAuthenticatorsForExecution::Resolved(authenticators_for_exec),
250                        auth_and_tx_checked,
251                        None, // gas_owner_pays
252                        kind,
253                        signer,
254                        tx_digest,
255                        auth_context_data,
256                        &mut None,
257                    );
258                (
259                    inner_temp_store,
260                    effects,
261                    authentication_error.is_some(),
262                    account_object_refs,
263                )
264            };
265
266        // Refuse to attest a transaction when its Move authentication did not succeed.
267        // Any other failure (out-of-gas or a Move abort in the transaction body) is the
268        // issuer's responsibility and is still attested, sequenced and charged as a
269        // failed transaction under the normal flow.
270        if authentication_failed {
271            return Err(IotaError::Execution(format!(
272                "refusing to attest transaction with failed Move authentication: {:?}",
273                effects.status()
274            )));
275        }
276
277        // Build `AttestationData`. The claim is in gas units, independent
278        // of the gas price the user chose.
279        let computation_units =
280            attestation::computation_units(effects.gas_cost_summary(), tx.gas_price());
281
282        let object_versions =
283            attestable_object_versions(tx, &inner_temp_store, account_object_refs)?;
284
285        Ok((
286            AttestationData::V1 {
287                computation_units,
288                object_versions,
289            },
290            owned_objects,
291        ))
292    }

294    /// Like [`AuthorityState::check_transaction_inputs_at_commit`], for an
295    /// attested transaction: the input checks execution runs, through the
296    /// same function, so the two cannot drift: Move-authenticator inputs, gas
297    /// (owner, coin type, balance), the transaction's own inputs and their
298    /// union. It also checks the receiving objects the reader loaded, as
299    /// admission does; execution does not, since the Move runtime resolves
300    /// receives. The publish verifier and the account object stay at
301    /// execution, where a failure produces failure effects.
302    pub(crate) fn check_attested_transaction_inputs_at_commit(
303        &self,
304        protocol_config: &ProtocolConfig,
305        reference_gas_price: u64,
306        tx: &Transaction,
307        tx_input_objects: InputObjects,
308        tx_receiving_objects: &ReceivingObjects,
309        move_authenticators: &[&MoveAuthenticator],
310        per_authenticator_inputs: Vec<(InputObjects, ObjectReadResult)>,
311    ) -> IotaResult<(CheckedInputObjects, Vec<CheckedInputObjects>)> {
312        // Execution meters authentication against `max_auth_gas` only when
313        // there is something to authenticate.
314        let authenticator_gas_budget = if move_authenticators.is_empty() {
315            0
316        } else {
317            protocol_config.max_auth_gas()
318        };
319        let per_authenticator_input_objects = per_authenticator_inputs
320            .into_iter()
321            .map(|(objects, _account)| objects)
322            .collect();
323        let (_gas_status, per_authenticator_checked_inputs, tx_checked_input_objects) =
324            iota_transaction_checks::check_certificate_and_move_authenticator_input(
325                tx,
326                tx_input_objects,
327                per_authenticator_input_objects,
328                Some(tx_receiving_objects),
329                authenticator_gas_budget,
330                protocol_config,
331                reference_gas_price,
332            )?;
333        Ok((tx_checked_input_objects, per_authenticator_checked_inputs))
334    }

336    /// The validators cut off for the rest of the epoch, as the highest
337    /// certified checkpoint summary this node holds shows them; empty when the
338    /// cutoff is off or that summary is from an earlier epoch.
339    pub(crate) fn cut_off_validators(&self) -> BTreeSet<AuthorityName> {
340        let epoch_store = self.load_epoch_store_one_call_per_task();
341        let protocol_config = epoch_store.protocol_config();
342        let Some(max_refuted_units_squared) =
343            protocol_config.attestation_max_refuted_units_squared_as_option()
344        else {
345            return BTreeSet::new();
346        };
347        let Ok(Some(checkpoint)) = self
348            .get_checkpoint_store()
349            .get_highest_verified_checkpoint()
350        else {
351            return BTreeSet::new();
352        };
353        if checkpoint.epoch != epoch_store.epoch() {
354            return BTreeSet::new();
355        }
356        match checkpoint.parse_version_specific_data(protocol_config) {
357            Ok(Some(data)) => data
358                .attestation_totals()
359                .map_or_else(BTreeSet::new, |totals| {
360                    cut_off_attestors(totals, max_refuted_units_squared, epoch_store.committee())
361                }),
362            _ => BTreeSet::new(),
363        }
364    }

367/// The record of an attested transaction's execution: the certified record
368/// when executing from a checkpoint, the judged verdict when executing from
369/// consensus, nothing for an unattested transaction.
370pub(super) fn attestation_record(
371    source: &AttestationSource,
372    outcome: &ExecutionOutcome,
373    effects: &TransactionEffects,
374    gas_price: u64,
375    protocol_config: &ProtocolConfig,
376) -> Option<AttestationRecord> {
377    match source {
378        AttestationSource::Unattested => None,
379        AttestationSource::Certified(record) => *record,
380        AttestationSource::Attestation(attestation) => {
381            // The claim is only measurable against a body that ran.
382            let executed_units = outcome
383                .body_ran
384                .then(|| attestation::computation_units(effects.gas_cost_summary(), gas_price));
385            let claimed_units = attestation.computation_units();
386            let tolerance = protocol_config.attestor_accuracy_tolerance_percentage_as_option();
387            let verdict = match outcome.blame {
388                Some(AuthenticationFailureBlame::Attestor {
389                    rerun_computation_units,
390                }) => {
391                    // A refuted authentication wasted both runs: execution's
392                    // own, whether or not the effects charge it, and the
393                    // judge's re-run.
394                    let units = outcome
395                        .consumed_units
396                        .saturating_add(rerun_computation_units);
397
398                    AttestationVerdict::new(Some(units), claimed_units, executed_units, tolerance)
399                }
400                // Certified, so a node executing from the checkpoint charges
401                // nobody, as the validators did.
402                Some(AuthenticationFailureBlame::Undecided) => AttestationVerdict::Undecided,
403                Some(AuthenticationFailureBlame::Account | AuthenticationFailureBlame::Nobody)
404                | None => AttestationVerdict::new(None, claimed_units, executed_units, tolerance),
405            };
406
407            Some(AttestationRecord::new(attestation, verdict))
408        }
409    }
410}

412/// Whether `units_squared` takes the attestor's whole score, which also
413/// stops its attestations for the rest of the epoch.
414fn takes_whole_score(units_squared: u64, max_refuted_units_squared: u64) -> bool {
415    units_squared >= max_refuted_units_squared.max(1)
416}
417
418/// The attestors whose `totals` take their whole score.
419pub(crate) fn attestors_taking_whole_score(
420    totals: &AttestationTotals,
421    max_refuted_units_squared: u64,
422) -> BTreeSet<u8> {
423    totals
424        .refuted_units_squared
425        .iter()
426        .filter(|&(_, &units_squared)| takes_whole_score(units_squared, max_refuted_units_squared))
427        .map(|(&attestor, _)| attestor)
428        .collect()
429}

431/// The committee members whose `totals` take their whole score.
432fn cut_off_attestors(
433    totals: &AttestationTotals,
434    max_refuted_units_squared: u64,
435    committee: &Committee,
436) -> BTreeSet<AuthorityName> {
437    attestors_taking_whole_score(totals, max_refuted_units_squared)
438        .into_iter()
439        .filter_map(|attestor| committee.authority_by_index(u32::from(attestor)).copied())
440        .collect()
441}

443/// Cuts each attestor's score by the squared units its refuted attestations
444/// wasted, relative to `max_refuted_units_squared`: a sum that large takes the
445/// whole score, a smaller one its fraction, and zero takes the whole score on
446/// any fault; the score floors at zero. `scores` and the totals are both
447/// indexed by authority index.
448///
449/// # Panics
450///
451/// If a refuted attestor index is beyond `scores`.
452pub(crate) fn apply_fault_penalties(
453    scores: &mut [u64],
454    totals: &AttestationTotals,
455    max_refuted_units_squared: u64,
456) {
457    let full_cut = u128::from(max_refuted_units_squared.max(1));
458    for (&attestor, &units_squared) in &totals.refuted_units_squared {
459        let cut = u128::from(MAX_SCORE) * u128::from(units_squared).min(full_cut) / full_cut;
460        let score = &mut scores[usize::from(attestor)];
461        *score = score.saturating_sub(u64::try_from(cut).unwrap_or(u64::MAX));
462    }
463}

465/// Versions of the objects an attestation records to be used as the evidence
466/// base for detecting a false attestation. `account_object_refs` carries the
467/// Move-authenticator account objects, which are read outside the temporary
468/// store.
469fn attestable_object_versions(
470    tx: &Transaction,
471    inner_temp_store: &InnerTemporaryStore,
472    account_object_refs: Vec<ObjectReference>,
473) -> IotaResult<Vec<ObjectReference>> {
474    let tx_pinned_ids: HashSet<ObjectId> = tx
475        .input_objects()?
476        .into_iter()
477        .filter_map(|kind| match kind {
478            InputObjectKind::ImmOrOwnedMoveObject(oref) => Some(oref.object_id),
479            InputObjectKind::MovePackage(_) | InputObjectKind::SharedMoveObject { .. } => None,
480        })
481        .chain(tx.gas().iter().map(|oref| oref.object_id))
482        .chain(
483            tx.receiving_objects()
484                .into_iter()
485                .map(|oref| oref.object_id),
486        )
487        .collect();
488
489    Ok(inner_temp_store
490        .input_objects
491        .values()
492        .filter(|obj| !obj.is_immutable())
493        .map(|obj| obj.object_ref())
494        .chain(
495            inner_temp_store
496                .loaded_runtime_objects
497                .iter()
498                .filter(|(_, meta)| !matches!(meta.owner, Owner::Immutable))
499                .map(|(id, meta)| ObjectReference::new(*id, meta.version, meta.digest)),
500        )
501        .chain(account_object_refs)
502        .filter(|oref| !tx_pinned_ids.contains(&oref.object_id))
503        .collect())
504}

587    /// A fault whose squared units reach `max_refuted_units_squared` takes the
588    /// whole score, a smaller one the square of its fraction, and faults beyond
589    /// that saturate.
590    #[test]
591    fn fault_penalties_scale_with_the_square_of_the_wasted_units() {
592        // One fault of 10 units takes the whole score.
593        const MAX_REFUTED_UNITS_SQUARED: u64 = 100;
594        let record = |attestor, verdict| Some(AttestationRecord { attestor, verdict });
595        let refuted = |units| AttestationVerdict::Refuted { units };
596        let totals = AttestationTotals::default().with_records(&[
597            // A full fault: the whole score.
598            record(0, refuted(10)),
599            // Half a fault: a quarter of the score.
600            record(2, refuted(5)),
601            // Far beyond a full fault: still the whole score, nothing more.
602            record(3, refuted(100)),
603            record(1, AttestationVerdict::Valid),
604            None,
605        ]);
606
607        let mut scores = vec![MAX_SCORE; 4];
608        apply_fault_penalties(&mut scores, &totals, MAX_REFUTED_UNITS_SQUARED);
609        assert_eq!(scores, vec![0, MAX_SCORE, MAX_SCORE - MAX_SCORE / 4, 0]);
610    }
