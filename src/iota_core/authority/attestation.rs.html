52impl AuthorityState {
53    /// Dry-runs a verified transaction and returns `AttestationData` together
54    /// with the owned objects validated during execution, ready to be wrapped
55    /// in a validator attestation and attached to the transaction before
56    /// submitting it to consensus.
57    ///
58    /// A transaction that passes the validation checks but whose transaction
59    /// dependency is known here to have failed, or to have run in an earlier
60    /// epoch, is refused without a dry run: it can never execute successfully.
61    ///
62    /// Shares the validation pipeline with
63    /// `handle_transaction_validation_checks`
64    /// via [`Self::run_validation_checks`] (deny checks, input object loading,
65    /// input checks, coin deny list), and then replaces the
66    /// `authenticate_transaction` call with
67    /// `authenticate_then_execute_transaction_to_effects` so that Move
68    /// authentication and transaction execution run together and produce full
69    /// `TransactionEffects` including the accurate `computation_cost`.
70    ///
71    /// `lock_owned_objects` is called with the owned objects among the
72    /// transaction's inputs before the dry run, once they pass validation and
73    /// the signers have authorized the transaction; its error is returned
74    /// without running the dry run. A Move account has authorized the
75    /// transaction once its authenticators pass their own run. If they do not
76    /// run on their own, `lock_owned_objects` is not called: only the dry run
77    /// then shows that the account authorized the transaction.
78    pub(crate) fn attest_transaction(
79        &self,
80        transaction: &VerifiedTransaction,
81        epoch_store: &Arc<AuthorityPerEpochStore>,
82        deny_config: &dyn DenyRuleConfig,
83        lock_owned_objects: impl FnOnce(&[ObjectReference]) -> IotaResult,
84    ) -> IotaResult<(AttestationData, Vec<ObjectReference>)> {
85        let protocol_config = epoch_store.protocol_config();
86        let tx = transaction.data().transaction();
87
88        let ValidationOutputs {
89            per_authenticator_inputs,
90            move_authenticators,
91            gas_status,
92            tx_checked_input_objects,
93            per_authenticator_checked_inputs,
94        } = self.run_validation_checks(transaction, epoch_store, deny_config, true)?;
95
96        // `aggregate_authenticator_input_objects()` asserts that the sets it
97        // combines hold the same read result for every object they share, and
98        // a failed assert stops the validator. It is called in
99        // `authenticate_before_consensus()` and again before the dry run, so
100        // the sets are compared before both: if two of them differ, the
101        // transaction is rejected with an error.
102        if !move_authenticators.is_empty() {
103            let checked_input_sets: Vec<_> = std::iter::once(&tx_checked_input_objects)
104                .chain(
105                    per_authenticator_checked_inputs
106                        .iter()
107                        .map(|(inputs, _)| inputs),
108                )
109                .collect();
110            check_input_read_consistency(&checked_input_sets)?;
111        }
112
113        // A dependent executes only after the transaction it depends on, and
114        // fails without running any command when that transaction failed. A
115        // dependency known here to have failed, or to have run in an earlier
116        // epoch, is final, so the transaction can never succeed and is refused
117        // before the dry run. A dependency not executed here yet is assumed to
118        // have succeeded, as in simulation, and the body is dry-run.
119        self.check_transaction_dependency_known_outcome(tx, epoch_store.epoch())?;
120
121        let owned_objects = tx_checked_input_objects.inner().filter_owned_objects();
122
123        let epoch_id = epoch_store.epoch_start_config().epoch_data().epoch_id();
124        let epoch_start_timestamp = epoch_store
125            .epoch_start_config()
126            .epoch_data()
127            .epoch_start_timestamp();
128        let backing_store = self.get_backing_store();
129        let (kind, signer, gas_data) = tx.execution_parts();
130        let tx_digest = *transaction.digest();
131
132        // Tests register the `attest-ignore-auth-failure` fail point to make a
133        // validator attest a transaction whose Move authentication fails, as a
134        // faulty validator can, claiming the units the test gives. A `Cell`
135        // keeps the binding immutable in builds where the fail point compiles
136        // to nothing.
137        let units_despite_failed_authentication = Cell::new(None);
138        iota_macros::fail_point_arg!("attest-ignore-auth-failure", |units: u64| {
139            units_despite_failed_authentication.set(Some(units))
140        });
141
142        // The dry run meters authentication and body against the whole budget,
143        // as execution does (the authenticators at most `max_auth_gas` of it),
144        // and a failed authentication is refused unpaid. So the authenticators first run on
145        // their own at the signing cap, as at admission: one that fails or
146        // exceeds `max_auth_gas` is refused before budget-sized work is done on
147        // it.
148        let authenticators_run_on_their_own =
149            !move_authenticators.is_empty() && units_despite_failed_authentication.get().is_none();
150        if authenticators_run_on_their_own {
151            let function_refs = extract_auth_fun_refs(signer, gas_data.owner, |address| {
152                move_authenticators
153                    .iter()
154                    .zip(per_authenticator_checked_inputs.iter())
155                    .find(|(move_authenticator, _)| move_authenticator.address() == address)
156                    .map(|(_, (_, auth_fun_ref))| auth_fun_ref.clone())
157            });
158            let authenticators = move_authenticators
159                .iter()
160                .zip(per_authenticator_checked_inputs.iter())
161                .map(|(move_authenticator, (input_objects, function_ref))| {
162                    (
163                        (*move_authenticator).to_owned(),
164                        function_ref.clone(),
165                        CheckedInputObjects::new_with_checked_transaction_inputs(
166                            input_objects.inner().clone(),
167                        ),
168                    )
169                })
170                .collect();
171            let capped_gas_status = IotaGasStatus::new(
172                protocol_config.max_auth_gas(),
173                gas_data.price,
174                epoch_store.reference_gas_price(),
175                protocol_config,
176            )?;
177            self.authenticate_before_consensus(
178                epoch_store,
179                transaction,
180                capped_gas_status,
181                authenticators,
182                function_refs,
183            )?;
184        }
185
186        // Validation checked that the signers own these objects, and by now
187        // the signers have authorized the transaction: signatures made with
188        // keys are verified, and the authenticators of a Move account have
189        // passed their own run. So the locks cannot hold anyone else's object,
190        // and a second transaction on the same objects is refused before it
191        // gets a dry run of its own.
192        if move_authenticators.is_empty() || authenticators_run_on_their_own {
193            lock_owned_objects(&owned_objects)?;
194        }
195
196        self.metrics.attestation_dry_runs_total.inc();
197
198        // The sets agree on each object they share: with more than one set,
199        // `check_input_read_consistency()` has checked it.
200        let read_versions = collect_shared_object_versions(
201            per_authenticator_checked_inputs
202                .iter()
203                .map(|(inputs, _)| inputs),
204        );
205
206        let (effects, authentication_error) = if move_authenticators.is_empty() {
207            // No Move authentication, execute directly.
208            let (_, _, effects, _, _) = epoch_store.executor().execute_transaction_to_effects(
209                backing_store.as_ref(),
210                protocol_config,
211                self.metrics.limits_metrics.clone(),
212                false, // expensive_checks
213                self.config.certificate_deny_config.certificate_deny_set(),
214                &epoch_id,
215                epoch_start_timestamp,
216                tx_checked_input_objects,
217                gas_data,
218                gas_status,
219                kind,
220                signer,
221                tx_digest,
222                &mut None,
223            );
224
225            // This branch has no Move authentication, so it can never fail auth.
226            (effects, None)
227        } else {
228            // Recover the `AuthenticatorFunctionRefForExecution` for each authenticator and
229            // run auth + execution in one pass.
230
231            let funcs_for_exec = move_authenticators
232                .iter()
233                .zip(per_authenticator_inputs)
234                .map(|(ma, (_, account_object))| {
235                    let (id, seq, digest) = ma.object_to_authenticate_components()?;
236                    let signer_addr = ma.address();
237
238                    self.check_move_account_for_validation(
239                        id,
240                        seq,
241                        digest,
242                        account_object,
243                        &signer_addr,
244                        protocol_config,
245                    )
246                })
247                .collect::<IotaResult<Vec<_>>>()?;
248
249            // Union the checked transaction + authenticator input objects for
250            // execution. Borrow the checked inputs here before they are consumed
251            // by `authenticators_for_exec` below.
252            let checked_for_union: Vec<&CheckedInputObjects> =
253                std::iter::once(&tx_checked_input_objects)
254                    .chain(per_authenticator_checked_inputs.iter().map(|(c, _)| c))
255                    .collect();
256            let auth_and_tx_checked =
257                iota_transaction_checks::aggregate_authenticator_input_objects(&checked_for_union)?;
258
259            let authenticators_for_exec = move_authenticators
260                .into_iter()
261                .zip(funcs_for_exec)
262                .zip(per_authenticator_checked_inputs)
263                .map(
264                    |((ma, function_ref), (input_objects, _))| MoveAuthenticatorForExecution {
265                        authenticator: ma.to_owned(),
266                        function_ref,
267                        input_objects,
268                    },
269                )
270                .collect::<Vec<_>>();
271
272            let (sender_authenticator_function_ref, sponsor_authenticator_function_ref) =
273                extract_auth_fun_refs(signer, gas_data.owner, |address| {
274                    authenticators_for_exec
275                        .iter()
276                        .find(|a| a.authenticator.address() == address)
277                        .map(|a| a.function_ref.authenticator_function_ref.clone())
278                });
279
280            let (sender_auth_digest, sponsor_auth_digest) =
281                transaction.data().compute_auth_digests()?;
282
283            // Serialize the TransactionData for the auth context before decomposing.
284            let tx_data_bytes =
285                bcs::to_bytes(tx).expect("TransactionData serialization cannot fail");
286
287            let auth_context_data = AuthContextData {
288                transaction_data_bytes: tx_data_bytes,
289                sender_auth_digest,
290                sponsor_auth_digest,
291                sender_authenticator_function_ref,
292                sponsor_authenticator_function_ref,
293            };
294
295            let (_, _, effects, _, execution_result, authentication_error) = epoch_store
296                .executor()
297                .authenticate_then_execute_transaction_to_effects(
298                    backing_store.as_ref(),
299                    protocol_config,
300                    self.metrics.limits_metrics.clone(),
301                    false, // expensive_checks
302                    self.config.certificate_deny_config.certificate_deny_set(),
303                    &epoch_id,
304                    epoch_start_timestamp,
305                    gas_data,
306                    gas_status,
307                    MoveAuthenticatorsForExecution::Resolved(authenticators_for_exec),
308                    auth_and_tx_checked,
309                    None, // gas_owner_pays
310                    kind,
311                    signer,
312                    tx_digest,
313                    auth_context_data,
314                    &mut None,
315                );
316
317            // The error of the dry run is that of the failed
318            // authentication, with its source, unless another failure of
319            // the dry run took its place; then only the kind is left.
320            let authentication_error = authentication_error.map(|kind| match execution_result {
321                Err(error) if *error.kind() == kind => error,
322                _ => ExecutionError::from_kind(kind),
323            });
324
325            (effects, authentication_error)
326        };
327
328        // Build `AttestationData`. The claim is in gas units, independent
329        // of the gas price the user chose.
330        let computation_units = match authentication_error {
331            // Refuse to attest a transaction when its Move authentication did
332            // not succeed, with the error admission reports for it. Any other
333            // failure (out-of-gas or a Move abort in the transaction body) is
334            // the issuer's responsibility and is still attested, sequenced and
335            // charged as a failed transaction under the normal flow.
336            Some(error) => units_despite_failed_authentication.get().ok_or_else(|| {
337                self.authentication_failure(
338                    tx,
339                    epoch_store.epoch(),
340                    move_authentication_failure(error),
341                )
342            })?,
343            None => attestation::computation_units(effects.gas_cost_summary(), tx.gas_price()),
344        };
345
346        // Tests register the `attest-shared-object-versions` fail point to
347        // make a validator record the entries they give in place of the ones
348        // its dry run read, as a faulty validator can.
349        let shared_object_versions = Cell::new(record_shared_object_versions(
350            transaction.data(),
351            &read_versions,
352        ));
353        iota_macros::fail_point_arg!("attest-shared-object-versions", |versions| {
354            shared_object_versions.set(versions)
355        });
356
357        Ok((
358            AttestationData::V1 {
359                computation_units,
360                shared_object_versions: shared_object_versions.into_inner(),
361            },
362            owned_objects,
363        ))
364    }
365
366    /// Like [`AuthorityState::check_transaction_inputs_at_commit`], for an
367    /// attested transaction: the input checks execution runs, through the
368    /// same function, so the two cannot drift: Move-authenticator inputs, gas
369    /// (owner, coin type, balance), the transaction's own inputs and their
370    /// union. It also checks the receiving objects the reader loaded, as
371    /// admission does; execution does not, since the Move runtime resolves
372    /// receives. The publish verifier and the account object stay at
373    /// execution, where a failure produces failure effects.
374    pub(crate) fn check_attested_transaction_inputs_at_commit(
375        &self,
376        protocol_config: &ProtocolConfig,
377        reference_gas_price: u64,
378        tx: &Transaction,
379        tx_input_objects: InputObjects,
380        tx_receiving_objects: &ReceivingObjects,
381        move_authenticators: &[&MoveAuthenticator],
382        per_authenticator_inputs: Vec<(InputObjects, ObjectReadResult)>,
383    ) -> IotaResult<(CheckedInputObjects, Vec<CheckedInputObjects>)> {
384        // Execution meters authentication against `max_auth_gas` only when
385        // there is something to authenticate.
386        let authenticator_gas_budget = if move_authenticators.is_empty() {
387            0
388        } else {
389            protocol_config.max_auth_gas()
390        };
391        let per_authenticator_input_objects = per_authenticator_inputs
392            .into_iter()
393            .map(|(objects, _account)| objects)
394            .collect();
395        let (_gas_status, per_authenticator_checked_inputs, tx_checked_input_objects) =
396            iota_transaction_checks::check_certificate_and_move_authenticator_input(
397                tx,
398                tx_input_objects,
399                per_authenticator_input_objects,
400                Some(tx_receiving_objects),
401                authenticator_gas_budget,
402                protocol_config,
403                reference_gas_price,
404            )?;
405        Ok((tx_checked_input_objects, per_authenticator_checked_inputs))
406    }

408    /// The validators cut off for the rest of the epoch, as the highest
409    /// certified checkpoint summary this node holds shows them; empty when the
410    /// cutoff is off or that summary is from an earlier epoch.
411    pub(crate) fn cut_off_validators(&self) -> BTreeSet<AuthorityName> {
412        let epoch_store = self.load_epoch_store_one_call_per_task();
413        let protocol_config = epoch_store.protocol_config();
414        let Some(max_refuted_units_squared) =
415            protocol_config.attestation_max_refuted_units_squared_as_option()
416        else {
417            return BTreeSet::new();
418        };
419        let Ok(Some(checkpoint)) = self
420            .get_checkpoint_store()
421            .get_highest_verified_checkpoint()
422        else {
423            return BTreeSet::new();
424        };
425        if checkpoint.epoch != epoch_store.epoch() {
426            return BTreeSet::new();
427        }
428        match checkpoint.parse_version_specific_data(protocol_config) {
429            Ok(Some(data)) => data
430                .attestation_totals()
431                .map_or_else(BTreeSet::new, |totals| {
432                    cut_off_attestors(totals, max_refuted_units_squared, epoch_store.committee())
433                }),
434            _ => BTreeSet::new(),
435        }
436    }
437}

439/// The record of an attested transaction's execution: the certified record
440/// when executing from a checkpoint, the judged verdict when executing from
441/// consensus, nothing for an unattested transaction.
442pub(super) fn attestation_record(
443    source: &AttestationSource,
444    outcome: &ExecutionOutcome,
445    effects: &TransactionEffects,
446    gas_price: u64,
447    protocol_config: &ProtocolConfig,
448) -> Option<AttestationRecord> {
449    match source {
450        AttestationSource::Unattested => None,
451        AttestationSource::Certified(record) => *record,
452        AttestationSource::Attestation { attestation, .. } => {
453            // The claim is only measurable against a body that ran.
454            let executed_units = outcome
455                .body_ran
456                .then(|| attestation::computation_units(effects.gas_cost_summary(), gas_price));
457            let claimed_units = attestation.computation_units();
458            let tolerance = protocol_config.attestor_accuracy_tolerance_percentage_as_option();
459            let verdict = match outcome.blame {
460                Some(AuthenticationFailureBlame::Attestor {
461                    rerun_computation_units,
462                }) => {
463                    // A refuted authentication wasted both runs: execution's
464                    // own, whether or not the effects charge it, and the
465                    // judge's re-run.
466                    let units = outcome
467                        .consumed_units
468                        .saturating_add(rerun_computation_units);
469
470                    AttestationVerdict::new(Some(units), claimed_units, executed_units, tolerance)
471                }
472                // Certified, so a node executing from the checkpoint charges
473                // nobody, as the validators did.
474                Some(AuthenticationFailureBlame::Undecided) => AttestationVerdict::Undecided,
475                Some(AuthenticationFailureBlame::Account | AuthenticationFailureBlame::Nobody)
476                | None => AttestationVerdict::new(None, claimed_units, executed_units, tolerance),
477            };
478
479            Some(AttestationRecord::new(attestation, verdict))
480        }
481    }
482}

484/// Whether `units_squared` takes the attestor's whole score, which also
485/// stops its attestations for the rest of the epoch.
486fn takes_whole_score(units_squared: u64, max_refuted_units_squared: u64) -> bool {
487    units_squared >= max_refuted_units_squared.max(1)
488}
489
490/// The attestors whose `totals` take their whole score.
491pub(crate) fn attestors_taking_whole_score(
492    totals: &AttestationTotals,
493    max_refuted_units_squared: u64,
494) -> BTreeSet<u8> {
495    totals
496        .refuted_units_squared
497        .iter()
498        .filter(|&(_, &units_squared)| takes_whole_score(units_squared, max_refuted_units_squared))
499        .map(|(&attestor, _)| attestor)
500        .collect()
501}
502
503/// The committee members whose `totals` take their whole score.
504fn cut_off_attestors(
505    totals: &AttestationTotals,
506    max_refuted_units_squared: u64,
507    committee: &Committee,
508) -> BTreeSet<AuthorityName> {
509    attestors_taking_whole_score(totals, max_refuted_units_squared)
510        .into_iter()
511        .filter_map(|attestor| committee.authority_by_index(u32::from(attestor)).copied())
512        .collect()
513}
514
515/// Cuts each attestor's score by the squared units its refuted attestations
516/// wasted, relative to `max_refuted_units_squared`: a sum that large takes the
517/// whole score, a smaller one its fraction, and zero takes the whole score on
518/// any fault; the score floors at zero. `scores` and the totals are both
519/// indexed by authority index.
520///
521/// # Panics
522///
523/// If a refuted attestor index is beyond `scores`.
524pub(crate) fn apply_fault_penalties(
525    scores: &mut [u64],
526    totals: &AttestationTotals,
527    max_refuted_units_squared: u64,
528) {
529    let full_cut = u128::from(max_refuted_units_squared.max(1));
530    for (&attestor, &units_squared) in &totals.refuted_units_squared {
531        let cut = u128::from(MAX_SCORE) * u128::from(units_squared).min(full_cut) / full_cut;
532        let score = &mut scores[usize::from(attestor)];
533        *score = score.saturating_sub(u64::try_from(cut).unwrap_or(u64::MAX));
534    }
535}

537/// Returns the version of each shared object that `input_sets` hold as read,
538/// by object id. A shared object whose contents `input_sets` do not hold,
539/// such as one read as deleted, has no version here.
540fn collect_shared_object_versions<'a>(
541    input_sets: impl IntoIterator<Item = &'a CheckedInputObjects>,
542) -> BTreeMap<ObjectId, Version> {
543    input_sets
544        .into_iter()
545        .flat_map(|input_set| input_set.inner().iter())
546        .filter(|input| input.is_shared_object())
547        .filter_map(|input| Some((input.id(), input.as_object()?.version())))
548        .collect()
549}

551/// Returns the entries of an attestation of `transaction`: the index of each
552/// shared object that a Move authenticator of `transaction` names, with the
553/// version that `read_versions` holds for the object. An object with no
554/// version there has no entry, which happens only when the authentication of
555/// the dry run failed, as on a deleted input.
556fn record_shared_object_versions(
557    transaction: &SenderSignedTransaction,
558    read_versions: &BTreeMap<ObjectId, Version>,
559) -> Vec<(SharedInputIndex, Version)> {
560    attestation::list_move_authenticator_shared_objects(transaction)
561        .into_iter()
562        .filter_map(|(index, shared_object)| {
563            let version = read_versions.get(&shared_object.object_id)?;
564
565            Some((index, *version))
566        })
567        .collect()
568}

570/// Checks that `input_sets` hold the same read result for every object they
571/// name more than once, and returns [`UserInputError::InconsistentInput`]
572/// for the first object where they differ.
573///
574/// [`iota_transaction_checks::checked_input_objects_union()`] asserts the
575/// same when it combines two sets.
576fn check_input_read_consistency(input_sets: &[&CheckedInputObjects]) -> IotaResult {
577    let mut read_results = HashMap::new();
578    for input in input_sets.iter().flat_map(|set| set.inner().iter()) {
579        let first_read = *read_results.entry(input.id()).or_insert(&input.object);
580        if *first_read != input.object {
581            return Err(UserInputError::InconsistentInput {
582                object_id: input.id(),
583            }
584            .into());
585        }
586    }
587
588    Ok(())
589}

602    /// A refuted record counts the units execution's own run consumed plus the
603    /// judge's re-run, even when the effects charge nothing.
604    #[test]
605    fn refuted_record_units_do_not_depend_on_the_uncharged_effects() {
606        let attestation = attestation::Attestation::new_authority(
607            AttestationData::V1 {
608                computation_units: 1_000,
609                shared_object_versions: vec![],
610            },
611            0,
612        );

682    /// A fault whose squared units reach `max_refuted_units_squared` takes the
683    /// whole score, a smaller one the square of its fraction, and faults beyond
684    /// that saturate.
685    #[test]
686    fn fault_penalties_scale_with_the_square_of_the_wasted_units() {
687        // One fault of 10 units takes the whole score.
688        const MAX_REFUTED_UNITS_SQUARED: u64 = 100;
689        let record = |attestor, verdict| Some(AttestationRecord { attestor, verdict });
690        let refuted = |units| AttestationVerdict::Refuted { units };
691        let totals = AttestationTotals::default().with_records(&[
692            // A full fault: the whole score.
693            record(0, refuted(10)),
694            // Half a fault: a quarter of the score.
695            record(2, refuted(5)),
696            // Far beyond a full fault: still the whole score, nothing more.
697            record(3, refuted(100)),
698            record(1, AttestationVerdict::Valid),
699            None,
700        ]);
701
702        let mut scores = vec![MAX_SCORE; 4];
703        apply_fault_penalties(&mut scores, &totals, MAX_REFUTED_UNITS_SQUARED);
704        assert_eq!(scores, vec![0, MAX_SCORE, MAX_SCORE - MAX_SCORE / 4, 0]);
705    }
