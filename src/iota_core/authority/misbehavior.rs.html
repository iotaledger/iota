13/// Selects which `VersionedMisbehaviorReport` variant peers may submit for
14/// the current epoch. Loaded once from `ProtocolConfig` and threaded through
15/// `MisbehaviorMonitor` / `ReportAggregator` / `Scoreboard` as a `Copy` token.
16///
17/// The schema itself (which categories exist and their layout) lives in the
18/// `MisbehaviorObservations` variants; this enum only versions the wire
19/// format and gates acceptance.
20#[derive(Clone, Copy, Debug, PartialEq, Eq)]
21pub enum MisbehaviorReportVersion {
22    V1,
23    V2,
24}
25
26impl MisbehaviorReportVersion {
27    pub fn from_protocol(protocol_config: &ProtocolConfig) -> Self {
28        match protocol_config.scorer_version_as_option() {
29            None | Some(1) => Self::V1,
30            Some(2) => Self::V2,
31            Some(version) => panic!("Unsupported scorer version {version}"),
32        }
33    }
34
35    /// Returns `true` if the given report's wire format matches this version.
36    pub fn accepts_report(&self, report: &VersionedMisbehaviorReport) -> bool {
37        match self {
38            Self::V1 => matches!(report.payload, MisbehaviorObservations::V1(_)),
39            Self::V2 => matches!(report.payload, MisbehaviorObservations::V2(_)),
40        }
41    }
42}

44/// Per-version zero observations (all metrics, all authorities = 0).
45pub(crate) fn zero_observations(
46    version: MisbehaviorReportVersion,
47    committee_size: usize,
48) -> MisbehaviorObservations {
49    match version {
50        MisbehaviorReportVersion::V1 => MisbehaviorObservations::V1(MisbehaviorObservationsV1 {
51            faulty_blocks_provable: vec![0u64; committee_size],
52            faulty_blocks_unprovable: vec![0u64; committee_size],
53            missing_proposals: vec![0u64; committee_size],
54            equivocations: vec![0u64; committee_size],
55        }),
56        MisbehaviorReportVersion::V2 => MisbehaviorObservations::V2(MisbehaviorObservationsV2 {
57            faulty_blocks_provable: vec![0u64; committee_size],
58            faulty_blocks_unprovable: vec![0u64; committee_size],
59            missing_proposals: vec![0u64; committee_size],
60            equivocations: vec![0u64; committee_size],
61            invalid_bundle_parts: vec![0u64; committee_size],
62        }),
63    }
64}

66/// Element-wise maximum merge across all metrics. Both inputs are always the
67/// epoch's single report version: the monitor's local observations and the
68/// aggregator's accepted reports are constructed from the same
69/// `MisbehaviorReportVersion`, so cross-version pairs cannot occur.
70pub(crate) fn merge_max(
71    a: &MisbehaviorObservations,
72    b: &MisbehaviorObservations,
73) -> MisbehaviorObservations {
74    fn elem_max(a: &[u64], b: &[u64]) -> Vec<u64> {
75        a.iter().zip(b.iter()).map(|(x, y)| *x.max(y)).collect()
76    }
77    match (a, b) {
78        (MisbehaviorObservations::V1(x), MisbehaviorObservations::V1(y)) => {
79            MisbehaviorObservations::V1(MisbehaviorObservationsV1 {
80                faulty_blocks_provable: elem_max(
81                    &x.faulty_blocks_provable,
82                    &y.faulty_blocks_provable,
83                ),
84                faulty_blocks_unprovable: elem_max(
85                    &x.faulty_blocks_unprovable,
86                    &y.faulty_blocks_unprovable,
87                ),
88                missing_proposals: elem_max(&x.missing_proposals, &y.missing_proposals),
89                equivocations: elem_max(&x.equivocations, &y.equivocations),
90            })
91        }
92        (MisbehaviorObservations::V2(x), MisbehaviorObservations::V2(y)) => {
93            MisbehaviorObservations::V2(MisbehaviorObservationsV2 {
94                faulty_blocks_provable: elem_max(
95                    &x.faulty_blocks_provable,
96                    &y.faulty_blocks_provable,
97                ),
98                faulty_blocks_unprovable: elem_max(
99                    &x.faulty_blocks_unprovable,
100                    &y.faulty_blocks_unprovable,
101                ),
102                missing_proposals: elem_max(&x.missing_proposals, &y.missing_proposals),
103                equivocations: elem_max(&x.equivocations, &y.equivocations),
104                invalid_bundle_parts: elem_max(&x.invalid_bundle_parts, &y.invalid_bundle_parts),
105            })
106        }
107        // Exhaustive on the first component: a new version must add its
108        // diagonal merge arm above rather than fall through to the panic.
109        (a @ MisbehaviorObservations::V1(_), b) | (a @ MisbehaviorObservations::V2(_), b) => {
110            panic!(
111                "cross-version misbehavior observation merge: {a:?} vs {b:?} — \
112                 all observations within an epoch share one report version"
113            )
114        }
115    }
116}

118/// Maps consensus-output counts onto the locally tracked schema. Per-metric
119/// vectors are projected onto `committee_size`: empty (= "not wired by
120/// consensus yet") is silently zero-filled, correct length is used as-is,
121/// any other length is malformed and is logged + zero-filled.
122pub(crate) fn observations_from_consensus_output(
123    counts: ConsensusOutputMisbehaviorCounts,
124    version: MisbehaviorReportVersion,
125    committee_size: usize,
126) -> MisbehaviorObservations {
127    let project = |row: Vec<u64>, name: &'static str| -> Vec<u64> {
128        if row.is_empty() {
129            vec![0u64; committee_size]
130        } else if row.len() == committee_size {
131            row
132        } else {
133            // A non-empty wrong-length row means consensus and iota-core
134            // disagree on committee size — this is a programmer error, not
135            // adversarial input (consensus is local code). Halt in debug /
136            // tests; in production, log loudly and zero-fill so the node
137            // keeps making progress (zero-fill is harmless: merge_max with
138            // zeros leaves prior state untouched).
139            debug_assert_eq!(
140                row.len(),
141                committee_size,
142                "consensus output row for {name} has wrong length — consensus and \
143                 iota-core disagree on committee size"
144            );
145            error!(
146                name,
147                actual = row.len(),
148                expected = committee_size,
149                "consensus output row has wrong length; consensus and iota-core \
150                 disagree on committee size — zero-filling and continuing"
151            );
152            vec![0u64; committee_size]
153        }
154    };
155    match version {
156        MisbehaviorReportVersion::V1 => {
157            // The V1 report format has no bundle-part category; fold those
158            // counts into the unprovable bucket.
159            let unprovable = project(counts.faulty_blocks_unprovable, "faulty_blocks_unprovable");
160            let bundle_parts = project(counts.invalid_bundle_parts, "invalid_bundle_parts");
161            let folded = unprovable
162                .iter()
163                .zip(&bundle_parts)
164                .map(|(unprov, bundle)| unprov.saturating_add(*bundle))
165                .collect();
166            MisbehaviorObservations::V1(MisbehaviorObservationsV1 {
167                faulty_blocks_provable: project(
168                    counts.faulty_blocks_provable,
169                    "faulty_blocks_provable",
170                ),
171                faulty_blocks_unprovable: folded,
172                missing_proposals: project(counts.missing_proposals, "missing_proposals"),
173                equivocations: project(counts.equivocations, "equivocations"),
174            })
175        }
176        MisbehaviorReportVersion::V2 => MisbehaviorObservations::V2(MisbehaviorObservationsV2 {
177            faulty_blocks_provable: project(
178                counts.faulty_blocks_provable,
179                "faulty_blocks_provable",
180            ),
181            faulty_blocks_unprovable: project(
182                counts.faulty_blocks_unprovable,
183                "faulty_blocks_unprovable",
184            ),
185            missing_proposals: project(counts.missing_proposals, "missing_proposals"),
186            equivocations: project(counts.equivocations, "equivocations"),
187            invalid_bundle_parts: project(counts.invalid_bundle_parts, "invalid_bundle_parts"),
188        }),
189    }
190}
