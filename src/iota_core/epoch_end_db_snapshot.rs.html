4//! Lets a consumer take a database snapshot of the perpetual store as each
5//! epoch ends.
6//!
7//! The epoch boundary hands the epoch to the consumer and waits only until the
8//! consumer has taken its [`DbSnapshot`](typed_store::DbSnapshot); whatever the
9//! consumer reads through it afterwards runs while the node executes the next
10//! epoch.
11
12use std::{sync::Arc, time::Duration};
13
14use iota_types::committee::EpochId;
15use tokio::sync::{OwnedSemaphorePermit, Semaphore, mpsc, oneshot};
16use tracing::warn;
17
18/// How long the boundary waits for the consumer to take its database snapshot
19/// before skipping the epoch.
20///
21/// Reconfiguration holds the execution write lock across this wait, so it
22/// must be bounded.
23pub const DB_SNAPSHOT_TIMEOUT: Duration = Duration::from_secs(30);
24
25/// The epoch boundary's end of the hand-over.
26pub struct EpochEndDbSnapshotHandle {
27    requests: mpsc::Sender<EpochEndDbSnapshotRequest>,
28    /// A single permit, held while the consumer is still busy with an epoch.
29    /// An epoch arriving while it is taken is skipped rather than queued.
30    permits: Arc<Semaphore>,
31}
32
33impl EpochEndDbSnapshotHandle {
34    pub fn new(requests: mpsc::Sender<EpochEndDbSnapshotRequest>) -> Self {
35        Self {
36            requests,
37            permits: Arc::new(Semaphore::new(1)),
38        }
39    }
40
41    /// Hands `epoch` to the consumer and returns once the consumer has taken
42    /// its database snapshot of the perpetual store.
43    ///
44    /// Returns [`HandOver::Skipped`] without waiting when the consumer is busy
45    /// or gone, and after [`DB_SNAPSHOT_TIMEOUT`] if the database snapshot is
46    /// never taken.
47    pub async fn hand_over(&self, epoch: EpochId) -> HandOver {
48        let Ok(permit) = self.permits.clone().try_acquire_owned() else {
49            warn!(
50                epoch,
51                "the epoch end database snapshot consumer is still busy with an earlier \
52                 epoch; skipping this one"
53            );
54            return HandOver::Skipped;
55        };
56        let (db_snapshot_taken, db_snapshot_is_taken) = oneshot::channel();
57        let request = EpochEndDbSnapshotRequest {
58            epoch,
59            db_snapshot_taken,
60            permit,
61        };
62        if let Err(err) = self.requests.try_send(request) {
63            warn!(
64                epoch,
65                "not handing this epoch to the epoch end database snapshot consumer: {err}"
66            );
67            return HandOver::Skipped;
68        }
69        match tokio::time::timeout(DB_SNAPSHOT_TIMEOUT, db_snapshot_is_taken).await {
70            Ok(Ok(())) => HandOver::Started,
71            Ok(Err(_)) => {
72                warn!(
73                    epoch,
74                    "the epoch end database snapshot consumer stopped before taking its \
75                     database snapshot"
76                );
77                HandOver::Skipped
78            }
79            // Dropping the receiver makes the consumer's send fail, which tells
80            // it the store is moving again.
81            Err(_) => {
82                warn!(
83                    epoch,
84                    "the epoch end database snapshot consumer did not take its database \
85                     snapshot within {DB_SNAPSHOT_TIMEOUT:?}; skipping this epoch"
86                );
87                HandOver::Skipped
88            }
89        }
90    }
91}
92
93/// What became of an epoch handed to the consumer.
94///
95/// This node hands an epoch over once, when it executes its boundary, so
96/// after `Skipped` the consumer never gets a database snapshot of that epoch
97/// from this node.
98#[derive(Debug, Clone, Copy, PartialEq, Eq)]
99pub enum HandOver {
100    /// The consumer holds a database snapshot of the state the epoch ended
101    /// with.
102    Started,
103    /// The consumer got no database snapshot of the epoch.
104    Skipped,
105}
106
107/// Asks the consumer to take a database snapshot of the perpetual store for
108/// `epoch`.
109///
110/// Sent while execution is paused at the epoch boundary, so that the snapshot
111/// holds the state the epoch ended with.
112pub struct EpochEndDbSnapshotRequest {
113    /// The epoch that has just ended.
114    pub epoch: EpochId,
115    /// Signalled once the database snapshot exists; the boundary waits for
116    /// nothing else. Dropped without a signal if the snapshot could not be
117    /// taken.
118    ///
119    /// A failed send means the boundary stopped waiting and execution has
120    /// resumed, so the snapshot may include writes from the next epoch. A
121    /// successful send means the boundary was still waiting, except for a
122    /// send that lands in the instant [`DB_SNAPSHOT_TIMEOUT`] expires.
123    pub db_snapshot_taken: oneshot::Sender<()>,
124    /// Held until the consumer is done with this epoch. See
125    /// [`EpochEndDbSnapshotHandle`].
126    pub permit: OwnedSemaphorePermit,
127}

129#[cfg(test)]
130mod tests {
131    use futures::poll;
132
133    use super::*;
134
135    /// A consumer that never takes its database snapshot must not hold
136    /// reconfiguration, which runs this under the execution write lock.
137    #[tokio::test(start_paused = true)]
138    async fn a_boundary_gives_up_on_a_consumer_that_never_takes_its_db_snapshot() {
139        let (requests, mut received) = mpsc::channel(1);
140        let handle = EpochEndDbSnapshotHandle::new(requests);
141
142        let handing_over = handle.hand_over(0);
143        tokio::pin!(handing_over);
144        assert!(
145            poll!(&mut handing_over).is_pending(),
146            "the boundary waits while the snapshot could still arrive",
147        );
148        // Held, never signalled: the consumer is stuck before its snapshot.
149        let request = received.try_recv().expect("the epoch was handed over");
150
151        // `start_paused` advances the clock as soon as nothing is runnable, so
152        // this returns without spending the timeout in real time.
153        assert_eq!(handing_over.await, HandOver::Skipped);
154
155        assert!(
156            request.db_snapshot_taken.is_closed(),
157            "giving up must drop the receiver, which is what tells the consumer \
158             that the store is moving again",
159        );
160    }
161
162    /// The boundary waits for the consumer's database snapshot and nothing
163    /// else, and an epoch arriving while the consumer is still busy with an
164    /// earlier one is skipped.
165    #[tokio::test]
166    async fn a_boundary_does_not_wait_for_a_busy_consumer() {
167        let (requests, mut received) = mpsc::channel(1);
168        let handle = EpochEndDbSnapshotHandle::new(requests);
169
170        let first = handle.hand_over(0);
171        tokio::pin!(first);
172        assert!(
173            poll!(&mut first).is_pending(),
174            "the boundary must wait until the consumer has taken its snapshot",
175        );
176        let request = received.try_recv().expect("the epoch was handed over");
177
178        // The consumer is still busy with that epoch, so the next one is
179        // skipped.
180        assert_eq!(
181            handle.hand_over(1).await,
182            HandOver::Skipped,
183            "a busy consumer must not hold the boundary up",
184        );
185        assert!(
186            received.try_recv().is_err(),
187            "the skipped epoch must not be queued behind the busy one",
188        );
189
190        // Taking the snapshot releases the first boundary.
191        request
192            .db_snapshot_taken
193            .send(())
194            .expect("the boundary is waiting");
195        assert_eq!(
196            first.await,
197            HandOver::Started,
198            "taking the snapshot must release the boundary",
199        );
200
201        // Once the consumer is done with that epoch, the next one is handed
202        // over again, and its boundary waits for its own snapshot.
203        drop(request.permit);
204        let third = handle.hand_over(2);
205        tokio::pin!(third);
206        assert!(
207            poll!(&mut third).is_pending(),
208            "the boundary waits for the snapshot of its own epoch",
209        );
210        assert_eq!(
211            received
212                .try_recv()
213                .expect("the epoch was handed over")
214                .epoch,
215            2,
216        );
217    }
