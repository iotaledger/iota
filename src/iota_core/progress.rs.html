4//! Progress reporting for the long passes a node makes over a whole table.
5
6use std::time::Duration;
7
8/// How often a pass writes a progress line.
9pub(crate) const PROGRESS_REPORT_INTERVAL: Duration = Duration::from_secs(1);
10
11/// Cap on a reported time left, so a stalled pass cannot overflow `Duration`.
12const LONGEST_REPORTED_WAIT: f64 = u32::MAX as f64;
13
14/// Formats one progress line, e.g. `"<pass>: ~2.9% done, 26.3M/~80.3M objects
15/// scanned (4.4M objects/s), ETA ~3m 23s"`.
16///
17/// `fraction_done` is taken separately from `done / total` so a pass can report
18/// a more accurate measure of its position, such as its place in the object id
19/// space.
20pub(crate) fn progress_line(
21    pass: &str,
22    unit: &str,
23    fraction_done: f64,
24    done: u64,
25    total: u64,
26    elapsed: Duration,
27) -> String {
28    let rate = format_count(progress_rate(done, elapsed) as u64);
29    // `total` comes from RocksDB's key estimate, which can under-report a
30    // heavily pruned table by any margin. Once the scan has passed it, it
31    // says nothing about what is left.
32    if total <= done {
33        return format!(
34            "{pass}: {} {unit} scanned ({rate} {unit}/s), total unknown",
35            format_count(done),
36        );
37    }
38    format!(
39        "{pass}: ~{:.1}% done, {}/~{} {unit} scanned ({rate} {unit}/s), ETA ~{}",
40        fraction_done * 100.0,
41        format_count(done),
42        format_count(total),
43        eta_display(elapsed, fraction_done),
44    )
45}
46
47/// Items processed per second since the work started.
48pub(crate) fn progress_rate(items: u64, elapsed: Duration) -> f64 {
49    items as f64 / elapsed.as_secs_f64().max(f64::EPSILON)
50}
51
52/// The estimated time remaining, or "unknown" when nothing has been done yet.
53pub(crate) fn eta_display(elapsed: Duration, fraction_done: f64) -> String {
54    if fraction_done <= 0.0 {
55        return "unknown".to_string();
56    }
57    let total = elapsed.as_secs_f64() / fraction_done;
58    let left = (total - elapsed.as_secs_f64()).clamp(0.0, LONGEST_REPORTED_WAIT);
59    format_duration(Duration::from_secs_f64(left))
60}
61
62/// Formats a duration for progress lines, e.g. "1h 42m", "3m 20s", or "45s".
63pub(crate) fn format_duration(duration: Duration) -> String {
64    let secs = duration.as_secs();
65    let (hours, minutes, seconds) = (secs / 3600, (secs % 3600) / 60, secs % 60);
66    if hours > 0 {
67        format!("{hours}h {minutes}m")
68    } else if minutes > 0 {
69        format!("{minutes}m {seconds}s")
70    } else {
71        format!("{seconds}s")
72    }
73}
74
75/// Formats a count for progress lines, e.g. "123.4M" or "1.2k".
76pub(crate) fn format_count(count: u64) -> String {
77    if count >= 1_000_000_000 {
78        format!("{:.1}B", count as f64 / 1e9)
79    } else if count >= 1_000_000 {
80        format!("{:.1}M", count as f64 / 1e6)
81    } else if count >= 1_000 {
82        format!("{:.1}k", count as f64 / 1e3)
83    } else {
84        count.to_string()
85    }
86}

88#[cfg(test)]
89mod tests {
90    use super::*;
91
92    /// Pins the format of a progress line, which every pass shares.
93    #[test]
94    fn a_progress_line_reads_the_same_for_every_pass() {
95        assert_eq!(
96            progress_line(
97                "Indexing live object set",
98                "objects",
99                0.029,
100                26_300_000,
101                803_000_000,
102                Duration::from_secs(210),
103            ),
104            "Indexing live object set: ~2.9% done, 26.3M/~803.0M objects scanned \
105             (125.2k objects/s), ETA ~1h 57m"
106        );
107    }
108
109    /// A total of `0` (no estimate) is reported as unknown, with no percentage
110    /// or ETA.
111    #[test]
112    fn an_unknown_total_is_not_reported_as_complete() {
113        let line = progress_line(
114            "Backfilling history",
115            "transactions",
116            0.0,
117            9_100_000,
118            0,
119            Duration::from_secs(125),
120        );
121        assert_eq!(
122            line,
123            "Backfilling history: 9.1M transactions scanned (72.8k transactions/s), \
124             total unknown"
125        );
126        assert!(!line.contains('%'), "{line}");
127        assert!(!line.contains("ETA"), "{line}");
128
129        // So is an estimate the scan has overtaken.
130        let line = progress_line(
131            "Backfilling history",
132            "effects",
133            1.0,
134            104_000_000,
135            193_200,
136            Duration::from_secs(1432),
137        );
138        assert!(!line.contains('%'), "{line}");
139        assert!(!line.contains("ETA"), "{line}");
140        assert!(line.contains("104.0M effects scanned"), "{line}");
141    }
142
143    /// Nothing done yet says so rather than extrapolating from no data.
144    #[test]
145    fn an_eta_needs_progress_to_estimate_from() {
146        assert_eq!(eta_display(Duration::from_secs(100), 0.0), "unknown");
147        assert_eq!(eta_display(Duration::from_secs(100), 0.25), "5m 0s");
148        assert_eq!(eta_display(Duration::from_secs(100), 1.0), "0s");
149    }
150
151    #[test]
152    fn format_duration_picks_the_two_largest_units() {
153        assert_eq!(format_duration(Duration::from_secs(45)), "45s");
154        assert_eq!(format_duration(Duration::from_secs(200)), "3m 20s");
155        assert_eq!(format_duration(Duration::from_secs(6120)), "1h 42m");
156    }
157
158    #[test]
159    fn format_count_abbreviates_large_numbers() {
160        assert_eq!(format_count(999), "999");
161        assert_eq!(format_count(1_200), "1.2k");
162        assert_eq!(format_count(123_400_000), "123.4M");
163        assert_eq!(format_count(2_500_000_000), "2.5B");
164    }
165}
