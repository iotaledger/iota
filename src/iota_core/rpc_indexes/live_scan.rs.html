5//! Fills the RPC index store's live-state tables from a stream of live
6//! objects: during a rebuild from the local live object set
7//! (`LiveObjectSetIndexer`) and during a formal-snapshot restore
8//! ([`RpcIndexesRestorer`]). Both use one indexer per partition of the stream,
9//! and both write the rows to SST files that are ingested once every
10//! partition is done.

46/// The coin metadata of the objects seen so far. A coin type's metadata,
47/// treasury and regulated metadata objects may land in different partitions,
48/// so the rows are written once every partition has been indexed.
49type CoinMetadata = Mutex<HashMap<CoinIndexKey, CoinIndexInfo>>;

51/// Indexes the live object set into the tables of `groups`, one indexer per
52/// scan partition. [`Self::finish`] ingests the rows once every partition is
53/// done.
54pub(super) struct LiveObjectSetIndexer<'a> {
55    tables: &'a IndexStoreTables,
56    groups: &'a BTreeSet<IndexGroup>,
57    coin_metadata: CoinMetadata,
58    sst_files: SstFiles,
59}
60
61impl<'a> LiveObjectSetIndexer<'a> {
62    pub(super) fn new(
63        tables: &'a IndexStoreTables,
64        groups: &'a BTreeSet<IndexGroup>,
65        sst_files: SstFiles,
66    ) -> Self {
67        Self {
68            tables,
69            groups,
70            coin_metadata: Mutex::new(HashMap::new()),
71            sst_files,
72        }
73    }
74
75    /// Ingests the partitions' SST files and writes the coin metadata
76    /// gathered across them.
77    pub(super) fn finish(self) -> Result<(), StorageError> {
78        self.sst_files.ingest(self.tables)?;
79        write_coin_metadata(self.tables, self.coin_metadata)?;
80        Ok(())
81    }
82}

95/// Where a rebuild or a restore keeps its SST files inside the index
96/// database's directory until it ingests them. One that stops before that
97/// leaves them for the next, which clears the whole directory.
98pub(super) const BULK_INGESTION_SST_DIR: &str = "bulk_ingestion";
99
100/// The rows the partitions of a bulk ingestion index, written to SST files
101/// and moved into the tables by [`Self::ingest`].
102///
103/// A partition takes a set of buffers from a shared pool, fills it, and hands
104/// it back when it is done, for the next partition to keep filling. A buffer
105/// is written to a file only once it is full, so the number of files follows
106/// the amount of data rather than the number of partitions; and there are no
107/// more buffers than partitions indexed at once.
108pub(super) struct SstFiles {
109    dir: PathBuf,
110    /// The options the tables are open with, compressing with LZ4 on every
111    /// level.
112    options: typed_store::rocksdb::Options,
113    /// How much memory one set of buffers may take before it is written out.
114    buffer_limit: usize,
115    next_file: AtomicUsize,
116    /// The buffers no partition is filling right now.
117    idle: Mutex<Vec<SstBuffers>>,
118    owner: Mutex<Vec<PathBuf>>,
119    dynamic_field: Mutex<Vec<PathBuf>>,
120    package_version: Mutex<Vec<PathBuf>>,
121}
122
123impl SstFiles {
124    /// For a bulk ingestion into the database at `db_path`, whose tables are
125    /// open with `table_options`, by `partitions_at_once` partitions indexing
126    /// at the same time, which share the memory budget.
127    pub(super) fn new(
128        db_path: &Path,
129        table_options: &typed_store::rocksdb::Options,
130        partitions_at_once: usize,
131    ) -> Result<Self, StorageError> {
132        Self::with_buffer_limit(
133            db_path,
134            table_options,
135            bulk_ingestion_options().sst_buffer_limit(partitions_at_once),
136        )
137    }
138
139    pub(super) fn with_buffer_limit(
140        db_path: &Path,
141        table_options: &typed_store::rocksdb::Options,
142        buffer_limit: usize,
143    ) -> Result<Self, StorageError> {
144        // Ingested files land in level 0 and are rewritten by compaction, so
145        // they only need to be quick to write. The SST writer would use the
146        // tables' Zstd for the bottommost level.
147        let mut options = table_options.clone();
148        options.set_bottommost_compression_type(typed_store::rocksdb::DBCompressionType::Lz4);
149        let dir = db_path.join(BULK_INGESTION_SST_DIR);
150        std::fs::create_dir_all(&dir).map_err(|e| {
151            StorageError::custom(format!(
152                "unable to create the SST file directory {dir:?}: {e}"
153            ))
154        })?;
155        Ok(Self {
156            dir,
157            options,
158            buffer_limit,
159            next_file: AtomicUsize::new(0),
160            idle: Mutex::new(Vec::new()),
161            owner: Mutex::new(Vec::new()),
162            dynamic_field: Mutex::new(Vec::new()),
163            package_version: Mutex::new(Vec::new()),
164        })
165    }
166
167    fn take_buffers(&self) -> SstBuffers {
168        self.idle.lock().pop().unwrap_or_default()
169    }
170
171    fn return_buffers(&self, buffers: SstBuffers) {
172        self.idle.lock().push(buffers);
173    }
174
175    /// Writes `buffer` to a new file of `table`'s list, if it holds any rows.
176    fn write<K: Serialize, V: Serialize>(
177        &self,
178        buffer: &mut SstFileBuffer<K, V>,
179        table: &Mutex<Vec<PathBuf>>,
180    ) -> Result<(), TypedStoreError> {
181        if buffer.is_empty() {
182            return Ok(());
183        }
184        let path = self.dir.join(format!(
185            "{}.sst",
186            self.next_file.fetch_add(1, Ordering::Relaxed)
187        ));
188        buffer.write_file(&path, &self.options)?;
189        table.lock().push(path);
190        Ok(())
191    }
192
193    /// Writes the rows still buffered, moves every file into its table and
194    /// removes the directory. Every partition indexer must have been dropped.
195    fn ingest(self, tables: &IndexStoreTables) -> Result<(), StorageError> {
196        let idle = std::mem::take(&mut *self.idle.lock());
197        let files = &self;
198        // One thread per set, as after a rebuild every scan task leaves one.
199        std::thread::scope(|scope| {
200            idle.into_iter()
201                .map(|mut buffers| scope.spawn(move || buffers.write(files)))
202                .collect::<Vec<_>>()
203                .into_iter()
204                .try_for_each(|writer| {
205                    writer.join().unwrap_or_else(|_| {
206                        Err(TypedStoreError::RocksDB(
207                            "an SST file writer thread panicked".to_string(),
208                        ))
209                    })
210                })
211        })?;
212        tables.owner.ingest_sst_files(&self.owner.into_inner())?;
213        tables
214            .dynamic_field
215            .ingest_sst_files(&self.dynamic_field.into_inner())?;
216        tables
217            .package_version
218            .ingest_sst_files(&self.package_version.into_inner())?;
219        // Not an error: the index is complete by now, and the ingestion moved
220        // the files into the database, so at most an empty directory is left.
221        if let Err(e) = std::fs::remove_dir_all(&self.dir) {
222            warn!(
223                "unable to remove the SST file directory {:?}: {e}",
224                self.dir
225            );
226        }
227        Ok(())
228    }
229}

231/// One buffer per table.
232#[derive(Default)]
233struct SstBuffers {
234    owner: SstFileBuffer<OwnerIndexKey, OwnerIndexInfo>,
235    dynamic_field: SstFileBuffer<DynamicFieldKey, ()>,
236    package_version: SstFileBuffer<PackageVersionKey, PackageVersionInfo>,
237}
238
239impl SstBuffers {
240    fn size_in_bytes(&self) -> usize {
241        self.owner.size_in_bytes()
242            + self.dynamic_field.size_in_bytes()
243            + self.package_version.size_in_bytes()
244    }
245
246    fn write(&mut self, files: &SstFiles) -> Result<(), TypedStoreError> {
247        files.write(&mut self.owner, &files.owner)?;
248        files.write(&mut self.dynamic_field, &files.dynamic_field)?;
249        files.write(&mut self.package_version, &files.package_version)
250    }
251}

253/// Indexes one partition into a set of buffers taken from `SstFiles`, and
254/// hands them back when dropped.
255pub struct RpcIndexesPartitionIndexer<'a> {
256    groups: &'a BTreeSet<IndexGroup>,
257    coin_metadata: &'a CoinMetadata,
258    files: &'a SstFiles,
259    buffers: SstBuffers,
260}
261
262impl<'a> RpcIndexesPartitionIndexer<'a> {
263    fn new(
264        groups: &'a BTreeSet<IndexGroup>,
265        coin_metadata: &'a CoinMetadata,
266        files: &'a SstFiles,
267    ) -> Self {
268        Self {
269            groups,
270            coin_metadata,
271            files,
272            buffers: files.take_buffers(),
273        }
274    }
275
276    /// Stages `object`'s rows in every table the store's groups maintain.
277    pub fn index_object(&mut self, object: &Object) -> Result<(), StorageError> {
278        match object.owner {
279            Owner::Address(owner) => {
280                if let Some((key, info)) = OwnerIndexKey::for_object(owner, object) {
281                    self.buffers.owner.insert(&key, &info)?;
282                }
283            }
284            Owner::Object(parent) => {
285                if is_dynamic_field(object) {
286                    self.buffers
287                        .dynamic_field
288                        .insert(&DynamicFieldKey::new(parent, object.id()), &())?;
289                }
290            }
291            Owner::Shared(_) | Owner::Immutable => {}
292            _ => unimplemented!("a new Owner enum variant was added and needs to be handled"),
293        }
294
295        if let Some((key, info)) = try_create_coin_index_info(object) {
296            merge_coin_into(&mut self.coin_metadata.lock(), key, info);
297        }
298        if let Some((key, object_id)) = try_create_regulated_coin_info(object) {
299            merge_coin_into(
300                &mut self.coin_metadata.lock(),
301                key,
302                CoinIndexInfo {
303                    regulated_coin_metadata_object_id: Some(object_id),
304                    ..Default::default()
305                },
306            );
307        }
308        if self.groups.contains(&IndexGroup::Grpc) {
309            if let Some((key, info)) = try_create_package_version_info(object) {
310                self.buffers.package_version.insert(&key, &info)?;
311            }
312        }
313
314        if self.buffers.size_in_bytes() >= self.files.buffer_limit {
315            self.buffers.write(self.files)?;
316        }
317        Ok(())
318    }
319
320    /// Ends this partition, handing its buffers back for the next one.
321    pub fn finish(self) -> Result<(), StorageError> {
322        drop(self);
323        Ok(())
324    }
325}
326
327/// Hands the buffers back even when the partition fails, so the rows of the
328/// partitions indexed into them earlier are not lost.
329impl Drop for RpcIndexesPartitionIndexer<'_> {
330    fn drop(&mut self) {
331        self.files.return_buffers(std::mem::take(&mut self.buffers));
332    }
333}

345/// Writes the coin metadata gathered across a stream's partitions. Must run
346/// before the markers are written, so the flush that makes the build durable
347/// covers these rows.
348fn write_coin_metadata(
349    tables: &IndexStoreTables,
350    coin_metadata: CoinMetadata,
351) -> Result<(), TypedStoreError> {
352    tables.coin.multi_insert(coin_metadata.into_inner())
353}

355/// The RPC index tables opened for a formal-snapshot restore.
356///
357/// Feed the live objects, in any order, through [`Self::partition_indexer`],
358/// then call [`Self::finalize`] so a node opens the store in place instead of
359/// rebuilding it.
360pub struct RpcIndexesRestorer {
361    tables: IndexStoreTables,
362    groups: BTreeSet<IndexGroup>,
363    coin_metadata: CoinMetadata,
364    sst_files: SstFiles,
365}
366
367impl RpcIndexesRestorer {
368    /// Opens the store for bulk ingestion and records this schema version and
369    /// `groups`. The watermark is written only by [`Self::finalize`], so a
370    /// node opening the store of an interrupted restore rebuilds it.
371    pub fn open(path: PathBuf, groups: BTreeSet<IndexGroup>) -> Result<Self, StorageError> {
372        // After the open, which clears a database directory it finds in use.
373        let (tables, table_options) = IndexStoreTables::open_for_bulk_ingestion(path.clone());
374        // The restore indexes one partition at a time.
375        let sst_files = SstFiles::new(&path, &table_options, 1)?;
376        tables.meta.insert(
377            &(),
378            &MetadataInfo {
379                version: CURRENT_DB_VERSION,
380                groups: groups.clone(),
381            },
382        )?;
383        Ok(Self {
384            tables,
385            groups,
386            coin_metadata: Mutex::new(HashMap::new()),
387            sst_files,
388        })
389    }
390
391    /// Returns an indexer for one partition of the snapshot's live objects.
392    pub fn partition_indexer(&self) -> RpcIndexesPartitionIndexer<'_> {
393        RpcIndexesPartitionIndexer::new(&self.groups, &self.coin_metadata, &self.sst_files)
394    }
395
396    /// Ingests the restored rows, writes the markers at `restore_checkpoint`,
397    /// the restore's highest executed checkpoint, flushes the bulk writes and
398    /// closes the database, so the caller can move its directory once this
399    /// returns.
400    ///
401    /// Callers must have restored the complete live object set first, through
402    /// [`Self::partition_indexer`], and dropped every partition indexer.
403    pub async fn finalize(
404        self,
405        restore_checkpoint: CheckpointSequenceNumber,
406    ) -> Result<(), StorageError> {
407        let Self {
408            tables,
409            coin_metadata,
410            sst_files,
411            ..
412        } = self;
413        sst_files.ingest(&tables)?;
414        write_coin_metadata(&tables, coin_metadata)?;
415        tables.adopt_bulk_ingestion(Some(restore_checkpoint))?;
416
417        let weak_db = tables.meta.db.close_handle();
418        drop(tables);
419        if !wait_for_database_close(weak_db).await {
420            return Err(StorageError::custom(
421                "unable to close the RPC index database after the restore",
422            ));
423        }
424        Ok(())
425    }
426
427    /// Reopens the finalized store as a node does and checks its markers, and
428    /// that the `owner` table is non-empty if `live_object_count` objects were
429    /// restored, so a store the node would rebuild fails the restore instead.
430    /// `historic_db_path` is as for [`RpcIndexesStore::new`].
431    pub async fn verify_restored(
432        path: &Path,
433        historic_db_path: Option<&Path>,
434        restore_checkpoint: CheckpointSequenceNumber,
435        live_object_count: u64,
436    ) -> Result<(), StorageError> {
437        let reopened = RpcIndexesStore::open_index_db(
438            path,
439            &crate::epoch_buckets::historic_root(path, historic_db_path),
440        )
441        .map_err(|e| {
442            StorageError::custom(format!(
443                "unable to reopen the restored RPC index database: {e}"
444            ))
445        })?;
446        let metadata = reopened.tables.meta.get(&())?.ok_or_else(|| {
447            StorageError::custom("the restored RPC index database has no metadata")
448        })?;
449        if metadata.version != CURRENT_DB_VERSION {
450            return Err(StorageError::custom(format!(
451                "restored RPC index database version mismatch: expected {}, found {}",
452                CURRENT_DB_VERSION, metadata.version
453            )));
454        }
455        if metadata.groups.is_empty() {
456            return Err(StorageError::custom(
457                "the restored RPC index database records no API groups",
458            ));
459        }
460        let watermark = reopened.tables.watermark.get(&())?;
461        if watermark != Some(restore_checkpoint) {
462            return Err(StorageError::custom(format!(
463                "the restored RPC index is watermarked at {watermark:?}, expected \
464                 {restore_checkpoint}"
465            )));
466        }
467        // Not `is_empty`, which reads an unreadable table as non-empty.
468        let owner_is_empty = reopened
469            .tables
470            .owner
471            .safe_iter()
472            .next()
473            .transpose()?
474            .is_none();
475        if live_object_count > 0 && owner_is_empty {
476            return Err(StorageError::custom(format!(
477                "the restored RPC index has an empty owner index after {live_object_count} live \
478                 objects"
479            )));
480        }
481
482        let weak_db = reopened.tables.meta.db.close_handle();
483        drop(reopened);
484        if !wait_for_database_close(weak_db).await {
485            return Err(StorageError::custom(
486                "unable to close the RPC index database after verifying the restore",
487            ));
488        }
489        Ok(())
490    }
