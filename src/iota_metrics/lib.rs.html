210    static SERVER_TIMING: Arc<Mutex<Timer>>;

213/// Create a new task-local ServerTiming context and run the provided future
214/// within it. Should be used at the top-most level of a request handler. Can be
215/// added to an axum router as a layer by using
216/// iota_service::server_timing_middleware.
217pub async fn with_new_server_timing<T>(fut: impl Future<Output = T> + Send + 'static) -> T {
218    let timer = Arc::new(Mutex::new(Timer::new()));
219
220    let mut ret = None;
221    SERVER_TIMING
222        .scope(timer, async {
223            ret = Some(fut.await);
224        })
225        .await;
226
227    ret.unwrap()
228}
229
230/// The `Server-Timing` HTTP header key.
231pub fn server_timing_header_key() -> &'static str {
232    Timer::header_key()
233}
234
235/// Add a final `finish_request` entry and write the collected timings to the
236/// `Server-Timing` response header. No-op outside a server-timing context.
237pub fn finish_and_set_server_timing_header(headers: &mut http::HeaderMap) {
238    let Some(timer) = get_server_timing() else {
239        return;
240    };
241    let header_value = {
242        let mut timer = timer.lock();
243        timer.add("finish_request");
244        timer.header_value()
245    };
246    if let Ok(value) = http::HeaderValue::try_from(header_value) {
247        headers.insert(server_timing_header_key(), value);
248    }
249}
250
251pub async fn server_timing_middleware(request: Request, next: Next) -> Response {
252    with_new_server_timing(async move {
253        let mut response = next.run(request).await;
254        finish_and_set_server_timing_header(response.headers_mut());
255        response
256    })
257    .await
258}

260/// Create a new task-local ServerTiming context and run the provided future
261/// within it. Only intended for use by macros within this module.
262pub async fn with_server_timing<T>(
263    timer: Arc<Mutex<Timer>>,
264    fut: impl Future<Output = T> + Send + 'static,
265) -> T {
266    let mut ret = None;
267    SERVER_TIMING
268        .scope(timer, async {
269            ret = Some(fut.await);
270        })
271        .await;
272
273    ret.unwrap()
274}

276/// Get the currently active ServerTiming context. Only intended for use by
277/// macros within this module.
278pub fn get_server_timing() -> Option<Arc<Mutex<Timer>>> {
279    SERVER_TIMING.try_with(|timer| timer.clone()).ok()
280}

282/// Add a new entry to the ServerTiming header.
283/// If the caller is not currently in a ServerTiming context (created with
284/// `with_new_server_timing`), an error is logged.
285pub fn add_server_timing(name: &str) {
286    let res = SERVER_TIMING.try_with(|timer| {
287        timer.lock().add(name);
288    });
289
290    if res.is_err() {
291        tracing::error!("Server timing context not found");
292    }
293}

295#[macro_export]
296macro_rules! monitored_future {
297    ($fut: expr) => {{ monitored_future!(futures, $fut, "", INFO, false) }};
298
299    ($metric: ident, $fut: expr, $name: expr, $logging_level: ident, $logging_enabled: expr) => {{
300        let location: &str = if $name.is_empty() {
301            concat!(file!(), ':', line!())
302        } else {
303            concat!(file!(), ':', $name)
304        };
305
306        async move {
307            let metrics = $crate::get_metrics();
308
309            let _metrics_guard = if let Some(m) = metrics {
310                m.$metric.with_label_values(&[location]).inc();
311                Some($crate::scopeguard::guard(m, |_| {
312                    m.$metric.with_label_values(&[location]).dec();
313                }))
314            } else {
315                None
316            };
317            let _logging_guard = if $logging_enabled {
318                Some($crate::scopeguard::guard((), |_| {
319                    tracing::event!(
320                        tracing::Level::$logging_level,
321                        "Future {} completed",
322                        location
323                    );
324                }))
325            } else {
326                None
327            };
328
329            if $logging_enabled {
330                tracing::event!(
331                    tracing::Level::$logging_level,
332                    "Spawning future {}",
333                    location
334                );
335            }
336
337            $fut.await
338        }
339    }};
340}

387pub struct MonitoredScopeGuard {
388    metrics: &'static Metrics,
389    name: &'static str,
390    timer: Instant,
391}
392
393impl Drop for MonitoredScopeGuard {
394    fn drop(&mut self) {
395        self.metrics
396            .scope_duration_ns
397            .with_label_values(&[self.name])
398            .add(self.timer.elapsed().as_nanos() as i64);
399        self.metrics
400            .scope_entrance
401            .with_label_values(&[self.name])
402            .dec();
403    }
404}

406/// This function creates a named scoped object, that keeps track of
407/// - the total iterations where the scope is called in the `monitored_scope_iterations` metric.
408/// - and the total duration of the scope in the `monitored_scope_duration_ns` metric.
409///
410/// The monitored scope should be single threaded, e.g. the scoped object
411/// encompass the lifetime of a select loop or guarded by mutex.
412/// Then the rate of `monitored_scope_duration_ns`, converted to the unit of sec
413/// / sec, would be how full the single threaded scope is running.
414pub fn monitored_scope(name: &'static str) -> Option<MonitoredScopeGuard> {
415    let metrics = get_metrics();
416    if let Some(m) = metrics {
417        m.scope_iterations.with_label_values(&[name]).inc();
418        m.scope_entrance.with_label_values(&[name]).inc();
419        Some(MonitoredScopeGuard {
420            metrics: m,
421            name,
422            timer: Instant::now(),
423        })
424    } else {
425        None
426    }
427}

429/// A trait extension for `Future` to allow monitoring the execution of the
430/// future within a specific scope. Provides the `in_monitored_scope` method to
431/// wrap the future in a `MonitoredScopeFuture`, which tracks the future's
432/// execution using a `MonitoredScopeGuard` for monitoring purposes.
433pub trait MonitoredFutureExt: Future + Sized {
434    /// Wraps the current future in a `MonitoredScopeFuture` that is associated
435    /// with a specific monitored scope name. The scope helps track the
436    /// execution of the future for performance analysis and metrics collection.
437    fn in_monitored_scope(self, name: &'static str) -> MonitoredScopeFuture<Self>;
438}
439
440impl<F: Future> MonitoredFutureExt for F {
441    fn in_monitored_scope(self, name: &'static str) -> MonitoredScopeFuture<Self> {
442        MonitoredScopeFuture {
443            f: Box::pin(self),
444            active_duration_metric: get_metrics()
445                .map(|m| m.future_active_duration_ns.with_label_values(&[name])),
446            _scope: monitored_scope(name),
447        }
448    }
449}

451/// A future that runs within a monitored scope. This struct wraps a pinned
452/// future and holds an optional `MonitoredScopeGuard` to measure and monitor
453/// the execution of the future. It forwards polling operations
454/// to the underlying future while maintaining the monitoring scope.
455pub struct MonitoredScopeFuture<F: Sized> {
456    f: Pin<Box<F>>,
457    active_duration_metric: Option<GenericGauge<AtomicI64>>,
458    _scope: Option<MonitoredScopeGuard>,
459}
460
461impl<F: Future> Future for MonitoredScopeFuture<F> {
462    type Output = F::Output;
463
464    fn poll(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
465        let active_timer = Instant::now();
466        let ret = self.f.as_mut().poll(cx);
467        if let Some(m) = &self.active_duration_metric {
468            m.add(active_timer.elapsed().as_nanos() as i64);
469        }
470        ret
471    }
472}

474/// A future that runs within a monitored scope. This struct wraps a pinned
475/// future and holds an optional `MonitoredScopeGuard` to measure and monitor
476/// the execution of the future. It forwards polling operations
477/// to the underlying future while maintaining the monitoring scope.
478pub struct CancelMonitor<F: Sized> {
479    finished: bool,
480    inner: Pin<Box<F>>,
481}
482
483impl<F> CancelMonitor<F>
484where
485    F: Future,
486{
487    /// Creates a new `CancelMonitor` that wraps the given future (`inner`). The
488    /// monitor tracks whether the future has completed.
489    pub fn new(inner: F) -> Self {
490        Self {
491            finished: false,
492            inner: Box::pin(inner),
493        }
494    }
495
496    /// Returns `true` if the future has completed; otherwise, `false`.
497    pub fn is_finished(&self) -> bool {
498        self.finished
499    }
500}
501
502impl<F> Future for CancelMonitor<F>
503where
504    F: Future,
505{
506    type Output = F::Output;
507
508    /// Polls the inner future to determine if it is ready or still pending. For
509    /// `CancelMonitor`, if the future completes (`Poll::Ready`), `finished`
510    /// is set to `true`. If it is still pending, the status remains
511    /// unchanged. This allows monitoring of the future's completion status.
512    fn poll(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
513        match self.inner.as_mut().poll(cx) {
514            Poll::Ready(output) => {
515                self.finished = true;
516                Poll::Ready(output)
517            }
518            Poll::Pending => Poll::Pending,
519        }
520    }
521}
522
523impl<F: Sized> Drop for CancelMonitor<F> {
524    /// When the `CancelMonitor` is dropped, it checks whether the future has
525    /// finished executing. If the future was not completed (`finished` is
526    /// `false`), it records that the future was cancelled by logging the
527    /// cancellation status using the current span.
528    fn drop(&mut self) {
529        if !self.finished {
530            Span::current().record("cancelled", true);
531        }
532    }
533}

543}

556/// A service to manage the prometheus registries. This service allow us to
557/// create a new Registry on demand and keep it accessible for
558/// processing/polling. The service can be freely cloned/shared across threads.
559#[derive(Clone)]
560pub struct RegistryService {
561    // Holds a Registry that is supposed to be used
562    default_registry: Registry,
563    registries_by_id: Arc<DashMap<Uuid, Registry>>,
564    filter: Arc<Filter>,
565}
566
567impl RegistryService {
568    // Creates a new registry service and also adds the main/default registry that
569    // is supposed to be preserved and never get removed
570    pub fn new(default_registry: Registry) -> Self {
571        Self {
572            filter: default_registry.filter(),
573            default_registry,
574            registries_by_id: Arc::new(DashMap::new()),
575        }
576    }
577
578    // Returns the default registry for the service that someone can use
579    // if they don't want to create a new one.
580    pub fn default_registry(&self) -> Registry {
581        self.default_registry.clone()
582    }
583
584    /// Returns the metrics filter shared by the service's registries.
585    pub fn filter(&self) -> Arc<Filter> {
586        self.filter.clone()
587    }
588
589    // Creates a new registry that shares the service's metric filter. Prefer
590    // this over `Registry::new()`/`Registry::new_custom()` for registries added
591    // via `add`, so that the configured filter applies to their metrics too.
592    pub fn new_registry_custom(
593        &self,
594        prefix: Option<String>,
595        labels: Option<std::collections::HashMap<String, String>>,
596    ) -> prometheus_filtered::Result<Registry> {
597        Registry::new_custom(prefix, labels, Some(self.filter.clone()))
598    }
599
600    // Adds a new registry to the service. The corresponding RegistryID is returned
601    // so can later be used for removing the Registry. Method panics if we try
602    // to insert a registry with the same id. As this can be quite serious for
603    // the operation of the node we don't want to accidentally swap an existing
604    // registry - we expected a removal to happen explicitly.
605    pub fn add(&self, registry: Registry) -> RegistryID {
606        let registry_id = Uuid::new_v4();
607        if self
608            .registries_by_id
609            .insert(registry_id, registry)
610            .is_some()
611        {
612            panic!("Other Registry already detected for the same id {registry_id}");
613        }
614
615        registry_id
616    }
617
618    // Removes the registry from the service. If Registry existed then this method
619    // returns true, otherwise false is returned instead.
620    pub fn remove(&self, registry_id: RegistryID) -> bool {
621        self.registries_by_id.remove(&registry_id).is_some()
622    }
623
624    // Returns all the registries of the service
625    pub fn get_all(&self) -> Vec<Registry> {
626        let mut registries: Vec<Registry> = self
627            .registries_by_id
628            .iter()
629            .map(|r| r.value().clone())
630            .collect();
631        registries.push(self.default_registry.clone());
632
633        registries
634    }
635
636    // Returns all the metric families from the registries that a service holds.
637    pub fn gather_all(&self) -> Vec<prometheus_filtered::proto::MetricFamily> {
638        self.get_all().iter().flat_map(|r| r.gather()).collect()
639    }
640
641    /// Sets the runtime override on the shared filter; every registry's next
642    /// gather exposes metrics per the new directives. `filter` may use
643    /// [`MetricGroups`] names, which are expanded for matching while the
644    /// string is echoed back as given. Rejects the whole update if any
645    /// directive is invalid.
646    pub fn set_runtime_filter(&self, filter: &str) -> std::result::Result<(), String> {
647        let expanded = MetricGroups::expand_directives(filter)?;
648        self.filter
649            .set_runtime_filter(prometheus_filtered::FilterSource::with_display(
650                &expanded, filter,
651            ))
652    }
653
654    /// Drops the runtime override on the shared filter, restoring every
655    /// registry to its startup exposure.
656    pub fn reset_runtime_filter(&self) {
657        self.filter.reset_runtime_filter();
658    }
659}

661/// Create a metric that measures the uptime from when this metric was
662/// constructed. The metric is labeled with:
663/// - 'process': the process type, differentiating between validator and fullnode
664/// - 'version': binary version, generally be of the format: 'semver-gitrevision'
665/// - 'chain_identifier': the identifier of the network which this process is part of
666pub fn uptime_metric(
667    process: &str,
668    version: &'static str,
669    chain_identifier: &str,
670) -> Box<dyn prometheus_filtered::core::Collector> {
671    let opts = prometheus_filtered::opts!("uptime", "uptime of the node service in seconds")
672        .variable_label("process")
673        .variable_label("version")
674        .variable_label("chain_identifier")
675        .variable_label("os_version")
676        .variable_label("is_docker");
677
678    let start_time = std::time::Instant::now();
679    let uptime = move || start_time.elapsed().as_secs();
680    let metric = prometheus_closure_metric::ClosureMetric::new(
681        opts,
682        prometheus_closure_metric::ValueType::Counter,
683        uptime,
684        &[
685            process,
686            version,
687            chain_identifier,
688            &sysinfo::System::long_os_version()
689                .unwrap_or_else(|| "os_version_unavailable".to_string()),
690            &is_running_in_docker().to_string(),
691        ],
692    )
693    .unwrap();
694
695    Box::new(metric)
696}

704pub const METRICS_ROUTE: &str = "/metrics";
705
706// Creates a new http server that has as a sole purpose to expose
707// and endpoint that prometheus agent can use to poll for the metrics.
708// A RegistryService is returned that can be used to get access in prometheus
709// Registries.
710pub fn start_prometheus_server(addr: SocketAddr) -> RegistryService {
711    start_prometheus_server_with_filter(addr, Filter::from_env())
712}
713
714pub fn start_prometheus_server_with_filter(addr: SocketAddr, filter: Filter) -> RegistryService {
715    // The default registry has no prefix or labels, so its construction is
716    // infallible.
717    let registry = Registry::new_custom(None, None, Some(Arc::new(filter)))
718        .expect("unprefixed registry is infallible");
719
720    let registry_service = RegistryService::new(registry);
721
722    if cfg!(msim) {
723        // prometheus uses difficult-to-support features such as
724        // TcpSocket::from_raw_fd(), so we can't yet run it in the simulator.
725        warn!("not starting prometheus server in simulator");
726        return registry_service;
727    }
728
729    let app = Router::new()
730        .route(METRICS_ROUTE, get(metrics))
731        .layer(Extension(registry_service.clone()));
732
733    tokio::spawn(async move {
734        let listener = tokio::net::TcpListener::bind(&addr).await.unwrap();
735        axum::serve(listener, app.into_make_service())
736            .await
737            .unwrap();
738    });
739
740    registry_service
741}
742
743/// Handles a request to retrieve metrics, using the provided `RegistryService`
744/// to gather all registered metric families. The metrics are then encoded to a
745/// text format for easy consumption by monitoring systems. If successful, it
746/// returns the metrics string with an `OK` status. If an error occurs during
747/// encoding, it returns an `INTERNAL_SERVER_ERROR` status along with an error
748/// message. Returns a tuple containing the status code and either the metrics
749/// data or an error description.
750pub async fn metrics(
751    Extension(registry_service): Extension<RegistryService>,
752) -> (StatusCode, String) {
753    let metrics_families = registry_service.gather_all();
754    match TextEncoder.encode_to_string(&metrics_families) {
755        Ok(metrics) => (StatusCode::OK, metrics),
756        Err(error) => (
757            StatusCode::INTERNAL_SERVER_ERROR,
758            format!("unable to encode metrics: {error}"),
759        ),
760    }
761}

854    #[test]
855    fn set_runtime_filter_applies_to_all_registries() {
856        use std::sync::Arc;
857
858        use prometheus_filtered::{Filter, MetricLevel};
859
860        fn gathered_names(service: &RegistryService) -> Vec<String> {
861            let mut names: Vec<_> = service
862                .gather_all()
863                .iter()
864                .map(|f| f.name().to_owned())
865                .collect();
866            names.sort();
867            names
868        }
869
870        // Both registries share the service's filter, which starts at `warn`
871        // for this module, hiding the default-level (`debug`) gauges.
872        let filter = Arc::new(Filter::parse("iota_metrics=warn"));
873        let default_registry =
874            Registry::new_custom(Some("default".to_string()), None, Some(filter)).unwrap();
875        let registry_service = RegistryService::new(default_registry.clone());
876        let second_registry = registry_service
877            .new_registry_custom(Some("second".to_string()), None)
878            .unwrap();
879        registry_service.add(second_registry.clone());
880
881        for registry in [&default_registry, &second_registry] {
882            prometheus_filtered::register_int_gauge_with_registry!(
883                "g_warn", "h", registry; MetricLevel::Warn
884            )
885            .unwrap();
886            prometheus_filtered::register_int_gauge_with_registry!("g_debug", "h", registry)
887                .unwrap();
888        }
889        assert_eq!(
890            gathered_names(&registry_service),
891            ["default_g_warn", "second_g_warn"]
892        );
893
894        // One runtime update changes the exposure of every registry in the
895        // service.
896        registry_service
897            .set_runtime_filter("runtime=debug")
898            .unwrap();
899        assert_eq!(
900            gathered_names(&registry_service),
901            [
902                "default_g_debug",
903                "default_g_warn",
904                "second_g_debug",
905                "second_g_warn"
906            ]
907        );
908
909        // Reset restores the startup exposure across all registries.
910        registry_service.reset_runtime_filter();
911        assert_eq!(
912            gathered_names(&registry_service),
913            ["default_g_warn", "second_g_warn"]
914        );
915    }
