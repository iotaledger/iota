30/// Reset period for the exact-count test policies when the blocklist TTL is
31/// zero, long enough that no count recovers over the lifetime of a test.
32const EXACT_COUNT_FALLBACK_RESET_PERIOD: Duration = Duration::from_secs(60 * 60);
33
34/// Largest usable client threshold: a sustained rate above one tally per
35/// nanosecond produces a zero replenish interval, which would silently disable
36/// the limiter. Clamped to this at startup, and rejected by the admin API.
37pub(super) const MAX_CLIENT_THRESHOLD: u64 = 1_000_000_000;
38
39/// Upper bound on client IPs tracked per limiter.
40const MAX_TRACKED_CLIENTS: usize = 100_000;
41
42/// GCRA cell for a single client IP. Uses `std::time::Instant` rather than
43/// `governor`'s default TSC-backed clock so that the deterministic simulator,
44/// which virtualizes `clock_gettime`, also controls rate limiter time.
45type ClientRateLimiter =
46    RateLimiter<NotKeyed, InMemoryState, MonotonicClock, NoOpMiddleware<Instant>>;
47
48/// Sustained `threshold` tallies per second, tolerating `threshold *
49/// burst_secs` back to back.
50fn sustained_quota(threshold: u64, burst_secs: u64) -> Quota {
51    let burst = threshold.saturating_mul(burst_secs);
52    if burst > u32::MAX as u64 {
53        warn!(
54            "freq-threshold burst {burst} exceeds {} cells, clamping",
55            u32::MAX
56        );
57    }
58    Quota::per_second(clamp_to_cells(threshold)).allow_burst(clamp_to_cells(burst))
59}
60
61/// The `threshold`-th tally in a row breaches and earlier ones pass, with the
62/// full count recovering over `reset_period`.
63fn exact_count_quota(threshold: u64, reset_period: Duration) -> Quota {
64    let burst = clamp_to_cells(threshold.saturating_sub(1));
65    Quota::with_period((reset_period / burst.get()).max(Duration::from_nanos(1)))
66        .expect("replenish period is non-zero")
67        .allow_burst(burst)
68}
69
70/// Counts recover over twice the blocklist TTL.
71fn exact_count_reset_period(connection_blocklist_ttl_sec: u64) -> Duration {
72    match connection_blocklist_ttl_sec.saturating_mul(2) {
73        0 => EXACT_COUNT_FALLBACK_RESET_PERIOD,
74        secs => Duration::from_secs(secs),
75    }
76}
77
78fn clamp_threshold(threshold: u64, name: &str) -> u64 {
79    if threshold > MAX_CLIENT_THRESHOLD {
80        warn!("freq-threshold {name} {threshold} exceeds {MAX_CLIENT_THRESHOLD}, clamping");
81        return MAX_CLIENT_THRESHOLD;
82    }
83    threshold
84}
85
86fn clamp_to_cells(value: u64) -> NonZeroU32 {
87    NonZeroU32::new(value.clamp(1, u32::MAX as u64) as u32).expect("clamped value is non-zero")
88}

120/// How a threshold maps to a quota.
121#[derive(Clone, Copy)]
122enum QuotaKind {
123    Sustained { burst_secs: u64 },
124    ExactCount { reset_period: Duration },
125}
126
127impl QuotaKind {
128    fn limiter(&self, threshold: u64, evictions: &IntCounter) -> Limiter {
129        let quota = match self {
130            Self::Sustained { burst_secs } if threshold >= 1 => {
131                sustained_quota(threshold, *burst_secs)
132            }
133            Self::ExactCount { reset_period } if threshold >= 2 => {
134                exact_count_quota(threshold, *reset_period)
135            }
136            // Too low for any burst quota: block on the first tally.
137            _ => return Limiter::BlockAll,
138        };
139        Limiter::Cells(ClientCells::new(quota, evictions.clone()))
140    }
141}
142
143/// Rate limiter for direct or proxied clients.
144enum Limiter {
145    /// Every tally blocks its client; the operator killswitch.
146    BlockAll,
147    Cells(ClientCells),
148}
149
150impl Limiter {
151    fn breaches(&self, client: IpAddr) -> bool {
152        match self {
153            Self::BlockAll => true,
154            Self::Cells(cells) => cells.breaches(client),
155        }
156    }
157}
158
159/// Per-client GCRA cells in an LRU cache bounded at [`MAX_TRACKED_CLIENTS`].
160struct ClientCells {
161    quota: Quota,
162    cells: Mutex<LruCache<IpAddr, ClientRateLimiter>>,
163    evictions: IntCounter,
164}
165
166impl ClientCells {
167    fn new(quota: Quota, evictions: IntCounter) -> Self {
168        Self {
169            quota,
170            cells: Mutex::new(LruCache::new(
171                NonZeroUsize::new(MAX_TRACKED_CLIENTS).expect("capacity is non-zero"),
172            )),
173            evictions,
174        }
175    }
176
177    fn breaches(&self, client: IpAddr) -> bool {
178        let mut cells = self.cells.lock();
179        if let Some(cell) = cells.get_mut(&client) {
180            return cell.check().is_err();
181        }
182        let cell = RateLimiter::direct_with_clock(self.quota, &MonotonicClock);
183        let breached = cell.check().is_err();
184        // `push` returns the entry it displaced; at capacity that is another
185        // client's state, which counts as an eviction.
186        if let Some((evicted, _)) = cells.push(client, cell) {
187            debug_assert_ne!(evicted, client);
188            self.evictions.inc();
189        }
190        breached
191    }
192}

194/// The direct-client threshold and its limiter, swapped as one unit on
195/// reconfiguration.
196struct DirectLimiter {
197    threshold: u64,
198    limiter: Limiter,
199}
200
201impl DirectLimiter {
202    fn new(quota_kind: QuotaKind, threshold: u64, evictions: &IntCounter) -> Self {
203        Self {
204            threshold,
205            limiter: quota_kind.limiter(threshold, evictions),
206        }
207    }
208}

210/// Rate limits direct clients, and proxied clients when the policy configures a
211/// separate threshold for them.
212struct RateLimitPolicy {
213    quota_kind: QuotaKind,
214    direct: ArcSwap<DirectLimiter>,
215    proxied: Option<Limiter>,
216    evictions: IntCounter,
217}
218
219impl RateLimitPolicy {
220    fn new(
221        quota_kind: QuotaKind,
222        direct_threshold: u64,
223        proxied_threshold: Option<u64>,
224        evictions: IntCounter,
225    ) -> Self {
226        Self {
227            direct: ArcSwap::from_pointee(DirectLimiter::new(
228                quota_kind,
229                direct_threshold,
230                &evictions,
231            )),
232            proxied: proxied_threshold.map(|threshold| quota_kind.limiter(threshold, &evictions)),
233            quota_kind,
234            evictions,
235        }
236    }
237
238    fn charge(&self, tally: &TrafficTally) -> PolicyResponse {
239        let direct = self.direct.load();
240        PolicyResponse {
241            block_client: tally
242                .direct
243                .filter(|client| direct.limiter.breaches(*client)),
244            block_proxied_client: tally.through_fullnode.filter(|client| {
245                self.proxied
246                    .as_ref()
247                    .is_some_and(|limiter| limiter.breaches(*client))
248            }),
249        }
250    }
251
252    fn set_direct_threshold(&self, threshold: u64) {
253        // Re-asserting the current threshold keeps every client's state.
254        if self.direct.load().threshold == threshold {
255            return;
256        }
257        self.direct.store(Arc::new(DirectLimiter::new(
258            self.quota_kind,
259            threshold,
260            &self.evictions,
261        )));
262    }
263}
264
265enum Policy {
266    NoOp,
267    Limit(RateLimitPolicy),
268    PanicOnInvocation,
269}
270
271/// The spam or error policy a traffic controller charges its tallies against.
272pub(super) struct TrafficControlPolicy(Policy);
273
274impl TrafficControlPolicy {
275    pub(super) fn from_policy_type(
276        policy_type: &PolicyType,
277        connection_blocklist_ttl_sec: u64,
278        evictions: IntCounter,
279    ) -> Self {
280        Self(match policy_type {
281            PolicyType::NoOp => Policy::NoOp,
282            PolicyType::FreqThreshold(FreqThresholdConfig {
283                client_threshold,
284                proxied_client_threshold,
285                burst_secs,
286            }) => {
287                if *burst_secs == 0 {
288                    fatal!("freq-threshold burst-secs must be more than zero");
289                }
290                Policy::Limit(RateLimitPolicy::new(
291                    QuotaKind::Sustained {
292                        burst_secs: *burst_secs,
293                    },
294                    clamp_threshold(*client_threshold, "client-threshold"),
295                    Some(clamp_threshold(
296                        *proxied_client_threshold,
297                        "proxied-client-threshold",
298                    )),
299                    evictions,
300                ))
301            }
302            // The exact-count test policy only ever blocks the direct client.
303            PolicyType::TestNConnIP(threshold) => Policy::Limit(RateLimitPolicy::new(
304                QuotaKind::ExactCount {
305                    reset_period: exact_count_reset_period(connection_blocklist_ttl_sec),
306                },
307                *threshold,
308                None,
309                evictions,
310            )),
311            PolicyType::TestPanicOnInvocation => Policy::PanicOnInvocation,
312        })
313    }
314
315    /// Charges the tally against the policy, returning which clients breached
316    /// their quota.
317    pub(super) fn charge(&self, tally: &TrafficTally) -> PolicyResponse {
318        match &self.0 {
319            Policy::NoOp => PolicyResponse::default(),
320            Policy::Limit(policy) => policy.charge(tally),
321            Policy::PanicOnInvocation => panic!("Tally for this policy should never be invoked"),
322        }
323    }
324
325    /// Direct-client threshold, or `None` for policies that do not rate limit.
326    pub(super) fn client_threshold(&self) -> Option<u64> {
327        match &self.0 {
328            Policy::Limit(policy) => Some(policy.direct.load().threshold),
329            _ => None,
330        }
331    }
332
333    /// Replaces the direct-client threshold, discarding every tracked client's
334    /// limiter state; re-asserting the current threshold keeps it.
335    pub(super) fn set_client_threshold(&self, threshold: u64) {
336        if let Policy::Limit(policy) = &self.0 {
337            policy.set_direct_threshold(threshold);
338        }
339    }
340}

510    #[test]
511    fn test_reconfigured_threshold_takes_effect() {
512        let policy = freq_threshold(1_000, 1_000, 5);
513        assert_eq!(policy.client_threshold(), Some(1_000));
514
515        // Accumulate state under the old threshold.
516        let client = IpAddr::V4(Ipv4Addr::new(10, 0, 0, 1));
517        for _ in 0..10 {
518            assert_eq!(policy.charge(&direct_tally(client)).block_client, None);
519        }
520
521        policy.set_client_threshold(2);
522        assert_eq!(policy.client_threshold(), Some(2));
523
524        // The new quota tolerates a fresh burst of 2 * 5, so the old state is
525        // gone along with the old limiter.
526        for i in 1..=10 {
527            assert_eq!(
528                policy.charge(&direct_tally(client)).block_client,
529                None,
530                "blocked early after {i} tallies"
531            );
532        }
533        assert_eq!(
534            policy.charge(&direct_tally(client)).block_client,
535            Some(client)
536        );
537
538        // Re-asserting the same threshold keeps the exhausted state.
539        policy.set_client_threshold(2);
540        assert_eq!(
541            policy.charge(&direct_tally(client)).block_client,
542            Some(client)
543        );
544
545        // Zero blocks every client, and a permissive threshold restores
546        // service.
547        policy.set_client_threshold(0);
548        assert_eq!(
549            policy.charge(&direct_tally(client)).block_client,
550            Some(client)
551        );
552        policy.set_client_threshold(1_000);
553        assert_eq!(policy.charge(&direct_tally(client)).block_client, None);
554
555        // An exact-count policy below a threshold of two blocks on the first
556        // tally, whether configured at startup or afterwards.
557        for threshold in [0, 1] {
558            assert_eq!(
559                exact_count(threshold, 60)
560                    .charge(&direct_tally(client))
561                    .block_client,
562                Some(client),
563                "threshold {threshold} did not block on the first tally"
564            );
565            let policy = exact_count(5, 60);
566            policy.set_client_threshold(threshold);
567            assert_eq!(
568                policy.charge(&direct_tally(client)).block_client,
569                Some(client),
570                "threshold {threshold} did not block on the first tally after reconfiguration"
571            );
572        }
573    }

589    #[test]
590    fn test_cell_cache_is_bounded() {
591        let evictions = evictions();
592        let policy = TrafficControlPolicy::from_policy_type(
593            &PolicyType::TestNConnIP(2),
594            60,
595            evictions.clone(),
596        );
597        let evicted = IpAddr::V4(Ipv4Addr::from(0x0A00_0000));
598        assert_eq!(policy.charge(&direct_tally(evicted)).block_client, None);
599
600        // Flood enough distinct clients to fill the cache exactly.
601        for i in 1..MAX_TRACKED_CLIENTS as u32 {
602            policy.charge(&direct_tally(IpAddr::V4(Ipv4Addr::from(0x0A00_0000 + i))));
603        }
604        assert_eq!(evictions.get(), 0);
605
606        // One client past capacity pushes the first one out.
607        let cached = IpAddr::V4(Ipv4Addr::from(0x0A00_0000 + MAX_TRACKED_CLIENTS as u32));
608        policy.charge(&direct_tally(cached));
609        assert_eq!(evictions.get(), 1);
610
611        // The evicted client's count was reset along with its cell, so its
612        // next tally counts as its first again; a still-cached client keeps
613        // its count and blocks on its second.
614        assert_eq!(policy.charge(&direct_tally(evicted)).block_client, None);
615        assert_eq!(
616            policy.charge(&direct_tally(cached)).block_client,
617            Some(cached)
618        );
619        // Re-admitting the evicted client displaced another; the cached one
620        // displaced nothing.
621        assert_eq!(evictions.get(), 2);
622
623        // The counter outlives the limiter that a threshold change rebuilds.
624        policy.set_client_threshold(3);
625        assert_eq!(evictions.get(), 2);
626    }
