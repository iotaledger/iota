65/// The blocklist TTLs in force, which the admin API can change at runtime.

90#[derive(Clone)]
91enum Acl {
92    Tally(Arc<TallyState>),
93    /// If this variant is set, then we do no tallying or running
94    /// of background tasks, and instead simply block all IPs not
95    /// in the allowlist on calls to `check`. The allowlist should
96    /// only be populated once at initialization, and stays sorted so
97    /// that `check` can binary-search it.
98    Allowlist(Vec<IpAddr>),
99}
100
101/// Spam and error policies, along with the state shared by the paths that
102/// charge them. Absent in allowlist mode, which does no tallying.
103struct TallyState {
104    spam_policy: Arc<TrafficControlPolicy>,
105    error_policy: Arc<TrafficControlPolicy>,
106    blocklists: Blocklists,
107    firewall_delegation: Option<FirewallDelegation>,
108    /// Whether the firewall drain file is present, refreshed by the dead man's
109    /// switch. Delegation pauses while it is.
110    drainfile_present: Arc<AtomicBool>,
111    shutdown: CancellationToken,
112}
113
114/// Queue of blocks handed to the remote firewall, so that the request thread
115/// never waits on the delegation request.
116struct FirewallDelegation {
117    sender: mpsc::Sender<Vec<DelegatedBlock>>,
118    /// Clients whose block is queued or in flight, so that a client breaching
119    /// on every request enqueues at most one block per firewall roundtrip.
120    pending: Arc<Mutex<HashSet<IpAddr>>>,
121    destination_port: u16,
122    delegate_spam_blocking: bool,
123    delegate_error_blocking: bool,
124}
125
126/// A block queued for the remote firewall, tagged with whether it targets the
127/// proxied client so that a failed delegation lands in the right local
128/// blocklist.
129struct DelegatedBlock {
130    client: IpAddr,
131    address: BlockAddress,
132    proxied: bool,
133}

135impl Drop for TallyState {
136    fn drop(&mut self) {
137        self.shutdown.cancel();
138    }

174        policy_config: PolicyConfig,
175        metrics: Arc<TrafficControllerMetrics>,
176        fw_config: Option<RemoteFirewallConfig>,
177    ) -> Self {
178        metrics.dry_run_enabled.set(policy_config.dry_run as i64);
179        let dry_run = Arc::new(AtomicBool::new(policy_config.dry_run));

198            policy_config: Arc::new(policy_config),
199            metrics,
200            dry_run,
201            blocklist_ttls,
202        }

237    /// Applies an operator initiated policy change. Changing a threshold
238    /// discards the accumulated rate limiter state of every tracked client.
239    pub fn admin_reconfigure(
240        &self,
241        params: TrafficControlReconfigParams,
242    ) -> Result<TrafficControlReconfigParams, IotaError> {
243        let TrafficControlReconfigParams {
244            error_threshold,
245            spam_threshold,
246            dry_run,
247            connection_blocklist_ttl_sec,
248            proxy_blocklist_ttl_sec,
249        } = params;
250        let updates = [
251            (
252                error_threshold,
253                self.tally_state().map(|state| state.error_policy.as_ref()),
254                &self.metrics.error_client_threshold,
255                "error",
256            ),
257            (
258                spam_threshold,
259                self.tally_state().map(|state| state.spam_policy.as_ref()),
260                &self.metrics.spam_client_threshold,
261                "spam",
262            ),
263        ];
264        // Validate the whole request first, so a rejected one applies nothing.
265        for (threshold, policy, _, kind) in updates {
266            if let Some(threshold) = threshold {
267                validate_threshold(policy, threshold, kind)?;
268            }
269        }
270        for (threshold, policy, gauge, _) in updates {
271            if let (Some(threshold), Some(policy)) = (threshold, policy) {
272                policy.set_client_threshold(threshold);
273                gauge.set(threshold as i64);
274            }
275        }
276        if let Some(dry_run) = dry_run {
277            self.metrics.dry_run_enabled.set(dry_run as i64);
278            self.dry_run.store(dry_run, Ordering::Relaxed);
279        }
280        let ttls = [
281            (
282                connection_blocklist_ttl_sec,
283                &self.blocklist_ttls.connection_blocklist_ttl_sec,
284                &self.metrics.connection_blocklist_ttl_sec,
285            ),
286            (
287                proxy_blocklist_ttl_sec,
288                &self.blocklist_ttls.proxy_blocklist_ttl_sec,
289                &self.metrics.proxy_blocklist_ttl_sec,
290            ),
291        ];
292        for (ttl_sec, live_ttl_sec, gauge) in ttls {
293            if let Some(ttl_sec) = ttl_sec {
294                live_ttl_sec.store(ttl_sec, Ordering::Relaxed);
295                gauge.set(ttl_sec as i64);
296            }
297        }
298
299        Ok(self.get_current_state())
300    }

302    /// Charges the tally against the spam and error policies, applying any
303    /// resulting block before returning. No-op in allowlist mode.
304    pub fn tally(&self, tally: TrafficTally) {
305        let Some(state) = self.tally_state() else {
306            return;
307        };
308        self.metrics.tallies.inc();
309        if tally.spam_weight.is_sampled() && self.policy_config.spam_sample_rate.is_sampled() {
310            let response = state.spam_policy.charge(&tally);
311            self.metrics.tally_handled.inc();
312            self.apply_policy_response(response, state, |delegation| {
313                delegation.delegate_spam_blocking
314            });
315        }
316        if let Some((error_weight, error_type)) = &tally.error_info {
317            if error_weight.is_sampled() {
318                self.metrics
319                    .tally_error_types
320                    .with_label_values(&[error_type.as_str()])
321                    .inc();
322                let response = state.error_policy.charge(&tally);
323                self.metrics.error_tally_handled.inc();
324                self.apply_policy_response(response, state, |delegation| {
325                    delegation.delegate_error_blocking
326                });
327            }
328        }
329    }

331    /// Blocks the breaching clients, either locally or by handing them to the
332    /// remote firewall when the policy delegates that kind of blocking.
333    fn apply_policy_response(
334        &self,
335        response: PolicyResponse,
336        state: &TallyState,
337        delegates: impl FnOnce(&FirewallDelegation) -> bool,
338    ) {
339        if response.block_client.is_none() && response.block_proxied_client.is_none() {
340            return;
341        }
342        // The firewall must receive no blocks during a drain or a dry run.
343        match state.firewall_delegation.as_ref().filter(|delegation| {
344            delegates(delegation)
345                && !state.drainfile_present.load(Ordering::Relaxed)
346                && !self.dry_run.load(Ordering::Relaxed)
347        }) {
348            Some(delegation) => self.delegate_policy_response(&response, state, delegation),
349            None => block_locally(
350                &response,
351                &self.blocklist_ttls,
352                &state.blocklists,
353                &self.metrics,
354            ),
355        }
356    }
357
358    fn delegate_policy_response(
359        &self,
360        response: &PolicyResponse,
361        state: &TallyState,
362        delegation: &FirewallDelegation,
363    ) {
364        let blocks: Vec<_> =
365            block_addresses(response, &self.blocklist_ttls, delegation.destination_port)
366                .into_iter()
367                .filter(|block| delegation.pending.lock().insert(block.client))
368                .collect();
369        if blocks.is_empty() {
370            return;
371        }
372        let dropped = match delegation.sender.try_send(blocks) {
373            Ok(()) => return,
374            Err(TrySendError::Full(dropped)) => {
375                // Not logged: it recurs on every request of a sustained breach.
376                self.metrics.firewall_delegation_overflow.inc();
377                dropped
378            }
379            Err(TrySendError::Closed(dropped)) => {
380                warn!("Firewall delegation queue closed unexpectedly");
381                dropped
382            }
383        };
384        release_pending(
385            &delegation.pending,
386            dropped.into_iter().map(|block| block.client),
387        );
388        block_locally(
389            response,
390            &self.blocklist_ttls,
391            &state.blocklists,
392            &self.metrics,
393        );
394    }

396    /// Handle check with dry-run mode considered. A request whose client IP the
397    /// node could not resolve is refused in allowlist mode, and admitted in the
398    /// rate-limiting modes, where it is charged to no client.
399    pub fn check(&self, client: &Option<IpAddr>, proxied_client: &Option<IpAddr>) -> bool {
400        let dry_run = self.dry_run.load(Ordering::Relaxed);
401        if client.is_none() {
402            self.metrics.unresolved_client_requests.inc();
403        }
404        let allowed = match &self.acl {
405            Acl::Allowlist(allowlist) => {
406                client.is_some_and(|client| allowlist.binary_search(&client).is_ok())
407            }
408            Acl::Tally(state) => check_blocklists(&state.blocklists, client, proxied_client),
409        };
410        match (allowed, dry_run) {
411            (true, _) => true,
412            (false, true) => {
413                debug!("Dry run mode: Blocked request from client {:?}", client);
414                self.metrics.num_dry_run_blocked_requests.inc();
415                true
416            }
417            (false, false) => {
418                debug!("Blocked request from client {:?}", client);
419                self.metrics.requests_blocked_at_protocol.inc();
420                false
421            }
422        }
423    }
424}

426/// Returns the allowlist sorted, so that `check` can binary-search it.
427fn parse_allowlist(allow_list: &[String]) -> Vec<IpAddr> {
428    let mut allowlist: Vec<IpAddr> = allow_list
429        .iter()
430        .map(|ip_str| {
431            parse_ip(ip_str)
432                .unwrap_or_else(|| fatal!("Failed to parse allowlist IP address: {ip_str:?}"))
433        })
434        .collect();
435    allowlist.sort_unstable();
436    allowlist
437}

439/// Builds the tallying state and spawns its background tasks. Must be called
440/// from within a tokio runtime.
441fn spawn_tally_state(
442    policy_config: &PolicyConfig,
443    metrics: &Arc<TrafficControllerMetrics>,
444    fw_config: Option<&RemoteFirewallConfig>,
445) -> TallyState {
446    let blocklists = Blocklists {
447        clients: Arc::new(DashMap::new()),
448        proxied_clients: Arc::new(DashMap::new()),
449    };
450    let spam_policy = Arc::new(TrafficControlPolicy::from_policy_type(
451        &policy_config.spam_policy_type,
452        policy_config.connection_blocklist_ttl_sec,
453        metrics.rate_limiter_evictions.clone(),
454    ));
455    let error_policy = Arc::new(TrafficControlPolicy::from_policy_type(
456        &policy_config.error_policy_type,
457        policy_config.connection_blocklist_ttl_sec,
458        metrics.rate_limiter_evictions.clone(),
459    ));
460    let drainfile_present = Arc::new(AtomicBool::new(false));
461    let shutdown = CancellationToken::new();
462    let mut firewall_delegation = None;
463
464    if let Some(fw_config) = fw_config {
465        // An unreadable path counts as a drain, so the node does not delegate
466        // while it cannot tell.
467        let present = fw_config.drain_path.try_exists().unwrap_or(true);
468        drainfile_present.store(present, Ordering::Relaxed);
469        metrics.deadmans_switch_enabled.set(present as i64);
470
471        let (sender, receiver) = mpsc::channel(FIREWALL_DELEGATION_QUEUE_SIZE);
472        let pending = Arc::new(Mutex::new(HashSet::new()));
473        firewall_delegation = Some(FirewallDelegation {
474            sender,
475            pending: pending.clone(),
476            destination_port: fw_config.destination_port,
477            delegate_spam_blocking: fw_config.delegate_spam_blocking,
478            delegate_error_blocking: fw_config.delegate_error_blocking,
479        });
480        let nodefw_client = NodeFWClient::new(fw_config.remote_fw_url.clone());
481        let delegation_blocklists = blocklists.clone();
482        let delegation_metrics = metrics.clone();
483        spawn_monitored_task!(run_firewall_delegation_loop(
484            receiver,
485            pending,
486            nodefw_client,
487            delegation_blocklists,
488            delegation_metrics
489        ));
490
491        let deadmans_switch_fw_config = fw_config.clone();
492        let deadmans_switch_drainfile = drainfile_present.clone();
493        let deadmans_switch_metrics = metrics.clone();
494        let deadmans_switch_shutdown = shutdown.clone();
495        spawn_monitored_task!(run_deadmans_switch_loop(
496            deadmans_switch_fw_config,
497            deadmans_switch_drainfile,
498            deadmans_switch_metrics,
499            deadmans_switch_shutdown
500        ));
501    }
502
503    let clear_loop_blocklists = blocklists.clone();
504    let clear_loop_metrics = metrics.clone();
505    let clear_loop_shutdown = shutdown.clone();
506    spawn_monitored_task!(run_clear_blocklists_loop(
507        clear_loop_blocklists,
508        clear_loop_metrics,
509        clear_loop_shutdown
510    ));
511
512    TallyState {
513        spam_policy,
514        error_policy,
515        blocklists,
516        firewall_delegation,
517        drainfile_present,
518        shutdown,
519    }
520}

522fn validate_threshold(
523    policy: Option<&TrafficControlPolicy>,
524    threshold: u64,
525    kind: &str,
526) -> Result<(), IotaError> {
527    let Some(policy) = policy else {
528        return Err(IotaError::InvalidAdminRequest(format!(
529            "Cannot reconfigure {kind} policy threshold in allowlist mode"
530        )));
531    };
532    if threshold > MAX_CLIENT_THRESHOLD {
533        return Err(IotaError::InvalidAdminRequest(format!(
534            "Threshold {threshold} exceeds the maximum of {MAX_CLIENT_THRESHOLD}"
535        )));
536    }
537    if policy.client_threshold().is_none() {
538        return Err(IotaError::InvalidAdminRequest(
539            "Unsupported prior policy type during traffic control reconfiguration".to_string(),
540        ));
541    }
542    Ok(())
543}

545/// Reports the thresholds the policies enforce.
546fn set_policy_config_metrics(
547    state: &TallyState,
548    policy_config: &PolicyConfig,
549    metrics: &TrafficControllerMetrics,
550) {
551    if let Some(threshold) = state.spam_policy.client_threshold() {
552        metrics.spam_client_threshold.set(threshold as i64);
553    }
554    if let Some(threshold) = state.error_policy.client_threshold() {
555        metrics.error_client_threshold.set(threshold as i64);
556    }
557    if let PolicyType::FreqThreshold(config) = &policy_config.spam_policy_type {
558        metrics
559            .spam_proxied_client_threshold
560            .set(config.proxied_client_threshold.min(MAX_CLIENT_THRESHOLD) as i64);
561    }
562    if let PolicyType::FreqThreshold(config) = &policy_config.error_policy_type {
563        metrics
564            .error_proxied_client_threshold
565            .set(config.proxied_client_threshold.min(MAX_CLIENT_THRESHOLD) as i64);
566    }
567}

569/// Returns true if neither client is blocked.
570fn check_blocklists(
571    blocklists: &Blocklists,
572    client: &Option<IpAddr>,
573    proxied_client: &Option<IpAddr>,
574) -> bool {
575    !blocked(client, &blocklists.clients) && !blocked(proxied_client, &blocklists.proxied_clients)
576}
577
578fn blocked(client: &Option<IpAddr>, blocklist: &Blocklist) -> bool {
579    client.is_some_and(|client| {
580        blocklist
581            .get(&client)
582            .is_some_and(|expiration| SystemTime::now() < *expiration)
583    })
584}

586/// The client to block and the TTL of that block, for the direct and the
587/// proxied client in that order.
588fn blocks(response: &PolicyResponse, ttls: &BlocklistTtls) -> [(Option<IpAddr>, u64); 2] {
589    [
590        (response.block_client, ttls.connection_blocklist_ttl_sec()),
591        (
592            response.block_proxied_client,
593            ttls.proxy_blocklist_ttl_sec(),
594        ),
595    ]
596}
597
598fn block_locally(
599    response: &PolicyResponse,
600    ttls: &BlocklistTtls,
601    blocklists: &Blocklists,
602    metrics: &TrafficControllerMetrics,
603) {
604    let targets = [
605        (&blocklists.clients, &metrics.connection_ip_blocklist_len),
606        (&blocklists.proxied_clients, &metrics.proxy_ip_blocklist_len),
607    ];
608    for ((client, ttl_secs), (blocklist, len_gauge)) in
609        blocks(response, ttls).into_iter().zip(targets)
610    {
611        let Some(client) = client else { continue };
612        insert_block(blocklist, len_gauge, client, ttl_secs);
613    }
614}

616/// Blocks a client for `ttl_secs`, counting it only when it was not already
617/// blocked so that the gauge matches the blocklist length.
618fn insert_block(blocklist: &Blocklist, len_gauge: &IntGauge, client: IpAddr, ttl_secs: u64) {
619    if blocklist
620        .insert(client, SystemTime::now() + Duration::from_secs(ttl_secs))
621        .is_none()
622    {
623        debug!("Adding client {client:?} to blocklist");
624        len_gauge.inc();
625    }
626}
627
628fn block_addresses(
629    response: &PolicyResponse,
630    ttls: &BlocklistTtls,
631    destination_port: u16,
632) -> Vec<DelegatedBlock> {
633    blocks(response, ttls)
634        .into_iter()
635        .zip([false, true])
636        .filter_map(|((client, ttl), proxied)| {
637            let client = client?;
638            debug!("Delegating blocking of client {client:?} to firewall");
639            Some(DelegatedBlock {
640                client,
641                address: BlockAddress {
642                    source_address: client.to_string(),
643                    destination_port,
644                    ttl,
645                },
646                proxied,
647            })
648        })
649        .collect()
650}

652/// Releases clients whose block never reached the firewall, which would
653/// otherwise never enqueue a block again.
654fn release_pending(pending: &Mutex<HashSet<IpAddr>>, clients: impl IntoIterator<Item = IpAddr>) {
655    let mut pending = pending.lock();
656    for client in clients {
657        pending.remove(&client);
658    }
659}

661/// Waits for the next tick of a background loop, returning false once the last
662/// controller holding the loop's state has been dropped.
663async fn tick(interval: Duration, shutdown: &CancellationToken) -> bool {
664    tokio::select! {
665        _ = tokio::time::sleep(interval) => true,
666        _ = shutdown.cancelled() => false,
667    }
668}

670/// Drops expired blocklist entries and refreshes the length gauges.
671async fn run_clear_blocklists_loop(
672    blocklists: Blocklists,
673    metrics: Arc<TrafficControllerMetrics>,
674    shutdown: CancellationToken,
675) {
676    while tick(CLEAR_BLOCKLIST_INTERVAL, &shutdown).await {
677        let now = SystemTime::now();
678        blocklists.clients.retain(|_, expiration| now < *expiration);
679        blocklists
680            .proxied_clients
681            .retain(|_, expiration| now < *expiration);
682        metrics
683            .connection_ip_blocklist_len
684            .set(blocklists.clients.len() as i64);
685        metrics
686            .proxy_ip_blocklist_len
687            .set(blocklists.proxied_clients.len() as i64);
688    }
689}

691/// Posts delegated blocks to the remote firewall, off the request path.
692async fn run_firewall_delegation_loop(
693    mut receiver: mpsc::Receiver<Vec<DelegatedBlock>>,
694    pending: Arc<Mutex<HashSet<IpAddr>>>,
695    node_fw_client: NodeFWClient,
696    blocklists: Blocklists,
697    metrics: Arc<TrafficControllerMetrics>,
698) {
699    while let Some(batch) = receiver.recv().await {
700        let addresses: Vec<_> = batch.iter().map(|block| block.address.clone()).collect();
701        metrics
702            .blocks_delegated_to_firewall
703            .inc_by(addresses.len() as u64);
704        if let Err(err) = node_fw_client
705            .block_addresses(BlockAddresses { addresses })
706            .await
707        {
708            metrics.firewall_delegation_request_fail.inc();
709            warn!("Failed to delegate blocklist to firewall: {err}");
710            for block in &batch {
711                let (blocklist, len_gauge) = if block.proxied {
712                    (&blocklists.proxied_clients, &metrics.proxy_ip_blocklist_len)
713                } else {
714                    (&blocklists.clients, &metrics.connection_ip_blocklist_len)
715                };
716                insert_block(blocklist, len_gauge, block.client, block.address.ttl);
717            }
718        }
719        release_pending(&pending, batch.into_iter().map(|block| block.client));
720    }
721    info!("TrafficController firewall delegation queue closed by all senders");
722}

724/// Drains the firewall if no tallies arrive for the configured timeout, so it
725/// stops blocking on stale state.
726async fn run_deadmans_switch_loop(
727    fw_config: RemoteFirewallConfig,
728    drainfile_present: Arc<AtomicBool>,
729    metrics: Arc<TrafficControllerMetrics>,
730    shutdown: CancellationToken,
731) {
732    let timeout = Duration::from_secs(fw_config.drain_timeout_secs);
733    let mut last_tallies = metrics.tallies.get();
734    let mut last_tally_at = Instant::now();
735    while tick(DEADMANS_SWITCH_POLL_INTERVAL, &shutdown).await {
736        // The operator can add or remove the drain file at any time, so
737        // delegation restarts after a drain. An I/O error keeps the last known
738        // state, because `exists` cannot tell an error from a removal.
739        match fw_config.drain_path.try_exists() {
740            Ok(present) => {
741                drainfile_present.store(present, Ordering::Relaxed);
742                metrics.deadmans_switch_enabled.set(present as i64);
743            }
744            Err(err) => warn!("Failed to read the nodefw drain file: {err}"),
745        }
746        let tallies = metrics.tallies.get();
747        if tallies != last_tallies {
748            last_tallies = tallies;
749            last_tally_at = Instant::now();
750            continue;
751        }
752        if last_tally_at.elapsed() < timeout || drainfile_present.load(Ordering::Relaxed) {
753            continue;
754        }
755        error!(
756            "No traffic tallies received in {} seconds.",
757            timeout.as_secs()
758        );
759        warn!("Draining Node firewall.");
760        if let Err(err) = File::create(&fw_config.drain_path) {
761            error!("Failed to create node firewall drain file: {err}");
762            continue;
763        }
764        drainfile_present.store(true, Ordering::Relaxed);
765        metrics.deadmans_switch_enabled.set(1);
766    }
767}

769pub fn parse_ip(ip: &str) -> Option<IpAddr> {
770    ip.parse::<IpAddr>().ok().or_else(|| {
771        ip.parse::<SocketAddr>()
772            .ok()
773            .map(|socket_addr| socket_addr.ip())
774            .or_else(|| {
775                error!("Failed to parse value of {:?} to ip address or socket.", ip,);
776                None
777            })
778    })
779}

781/// Outcome of resolving the client IP for an incoming request.
782#[derive(Debug)]
783pub enum ClientIpStatus {
784    Ok(IpAddr),
785    /// `SocketAddr` source but the IO type did not expose a remote address
786    /// (e.g. Unix sockets, custom transports). In tests this is usually a
787    /// programming error; in production it usually means a misconfigured
788    /// transport.
789    SocketAddrMissing,
790    /// `XForwardedFor` source but no `x-forwarded-for` header on the request.
791    XForwardedForHeaderMissing,
792    /// `XForwardedFor` source but the entry this node selects was not valid
793    /// UTF-8. Bytes in the rest of the header do not reach this case.
794    XForwardedForInvalidUtf8,
795    /// `XForwardedFor` configured with `num_hops == 0` (operator misconfig).
796    /// Carries the header entries, which the operator counts to get the hop
797    /// count.
798    XForwardedForZeroHops {
799        entries: Vec<String>,
800    },
801    /// `XForwardedFor` configured with `expected` hops but the header
802    /// only had `actual` entries.
803    XForwardedForConfigMismatch {
804        expected: usize,
805        actual: usize,
806    },
807    /// `XForwardedFor` source but the entry this node selects did not parse as
808    /// an IP address. Entries the node does not select do not reach this case.
809    XForwardedForUnparsable,
810}
811
812/// Reports what the node read, in the words an operator needs to act on it.
813/// The three servers share this text, and the hop-count procedure in the
814/// operator guide reads the entries out of the zero-hop line.
815impl std::fmt::Display for ClientIpStatus {
816    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
817        match self {
818            Self::Ok(client) => write!(f, "The client IP is {client}."),
819            Self::SocketAddrMissing => write!(
820                f,
821                "The request carries no peer address. Check the transport, or use the \
822                `x-forwarded-for` client-id-source if a proxy serves this node."
823            ),
824            Self::XForwardedForHeaderMissing => write!(
825                f,
826                "The request carries no x-forwarded-for header, although this node reads the \
827                client IP from that header. The request reached the node without its proxy."
828            ),
829            Self::XForwardedForInvalidUtf8 => write!(
830                f,
831                "The x-forwarded-for entry this node selects is not valid UTF-8."
832            ),
833            Self::XForwardedForZeroHops { entries } => write!(
834                f,
835                "x-forwarded-for: 0 specified. x-forwarded-for contents: {entries:?}. Please \
836                assign a nonzero number of hops, or use the `socket-addr` client-id-source if \
837                requests do not reach this node through a proxy. Until then the node reads no \
838                client IP."
839            ),
840            Self::XForwardedForConfigMismatch { expected, actual } => write!(
841                f,
842                "The x-forwarded-for header holds {actual} entries, but {expected} hops are \
843                configured. Please set the `x-forwarded-for` value under `client-id-source` to \
844                the number of proxies in front of this node."
845            ),
846            Self::XForwardedForUnparsable => write!(
847                f,
848                "The x-forwarded-for entry this node selects is not an IP address."
849            ),
850        }
851    }
852}
853
854/// The entries of the `x-forwarded-for` header, over every field of it and in
855/// order. Empty when the request carries no such header.
856fn forwarded_entries(headers: &http::HeaderMap) -> Vec<&[u8]> {
857    headers
858        .get_all("x-forwarded-for")
859        .iter()
860        .flat_map(|field| field.as_bytes().split(|byte| *byte == b','))
861        .map(|entry| entry.trim_ascii())
862        .collect()
863}
864
865/// How many proxies the `x-forwarded-for` header of a request reports, or
866/// `None` when it carries no such header. Reported whatever the client-id
867/// source is, so that an operator can tell a misconfigured proxy from a
868/// correct one.
869pub fn forwarded_hop_depth(headers: &http::HeaderMap) -> Option<usize> {
870    let entries = forwarded_entries(headers);
871    (!entries.is_empty()).then(|| entries.len().saturating_sub(1))
872}
873
874/// Resolve the client IP for an incoming request.
875pub fn get_client_ip(
876    headers: &http::HeaderMap,
877    remote_addr: Option<SocketAddr>,
878    source: &ClientIdSource,
879) -> ClientIpStatus {
880    match source {
881        ClientIdSource::SocketAddr => match remote_addr {
882            Some(addr) => ClientIpStatus::Ok(addr.ip()),
883            None => ClientIpStatus::SocketAddrMissing,
884        },
885        ClientIdSource::XForwardedFor(num_hops) => {
886            // A proxy either appends to the value the client sent or adds a
887            // field of its own, so a client could otherwise hide the entry its
888            // proxy wrote.
889            let raw_entries = forwarded_entries(headers);
890            if raw_entries.is_empty() {
891                return ClientIpStatus::XForwardedForHeaderMissing;
892            }
893            if *num_hops == 0 {
894                return ClientIpStatus::XForwardedForZeroHops {
895                    entries: raw_entries
896                        .iter()
897                        .map(|entry| String::from_utf8_lossy(entry).into_owned())
898                        .collect(),
899                };
900            }
901            if raw_entries.len() < *num_hops {
902                return ClientIpStatus::XForwardedForConfigMismatch {
903                    expected: *num_hops,
904                    actual: raw_entries.len(),
905                };
906            }
907            let Ok(entry) = str::from_utf8(raw_entries[raw_entries.len() - num_hops]) else {
908                return ClientIpStatus::XForwardedForInvalidUtf8;
909            };
910            match parse_ip(entry) {
911                Some(ip) => ClientIpStatus::Ok(ip),
912                None => ClientIpStatus::XForwardedForUnparsable,
913            }
914        }
915    }
916}

1345    fn controller_with_delegation_queue(
1346        queue_size: usize,
1347    ) -> (TrafficController, mpsc::Receiver<Vec<DelegatedBlock>>) {
1348        // A threshold of one blocks the direct client on every tally.
1349        let policy_config = PolicyConfig {
1350            spam_policy_type: PolicyType::TestNConnIP(1),
1351            spam_sample_rate: Weight::one(),
1352            dry_run: false,
1353            ..Default::default()
1354        };
1355        let metrics = Arc::new(TrafficControllerMetrics::new_for_tests());
1356        let (sender, receiver) = mpsc::channel(queue_size);
1357        // The delegation loop stays unspawned. Queued blocks stay queued.
1358        let state = TallyState {
1359            spam_policy: Arc::new(TrafficControlPolicy::from_policy_type(
1360                &policy_config.spam_policy_type,
1361                policy_config.connection_blocklist_ttl_sec,
1362                metrics.rate_limiter_evictions.clone(),
1363            )),
1364            error_policy: Arc::new(TrafficControlPolicy::from_policy_type(
1365                &PolicyType::NoOp,
1366                policy_config.connection_blocklist_ttl_sec,
1367                metrics.rate_limiter_evictions.clone(),
1368            )),
1369            blocklists: Blocklists {
1370                clients: Arc::new(DashMap::new()),
1371                proxied_clients: Arc::new(DashMap::new()),
1372            },
1373            firewall_delegation: Some(FirewallDelegation {
1374                sender,
1375                pending: Arc::new(Mutex::new(HashSet::new())),
1376                destination_port: 8080,
1377                delegate_spam_blocking: true,
1378                delegate_error_blocking: true,
1379            }),
1380            drainfile_present: Arc::new(AtomicBool::new(false)),
1381            shutdown: CancellationToken::new(),
1382        };
1383        let controller = TrafficController {
1384            acl: Acl::Tally(Arc::new(state)),
1385            blocklist_ttls: Arc::new(BlocklistTtls::from_config(&policy_config)),
1386            policy_config: Arc::new(policy_config),
1387            metrics,
1388            dry_run: Arc::new(AtomicBool::new(false)),
1389        };
1390        (controller, receiver)
1391    }
