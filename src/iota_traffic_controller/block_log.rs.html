4//! The line the node logs when it starts to refuse a client. Host firewall
5//! tools such as fail2ban and CrowdSec match on this line, so its format is
6//! part of the node's interface to operators: change it only together with
7//! the filters and operator docs that match on it, and mention the change in
8//! the release notes.
9
10use std::{fmt, net::IpAddr};
11
12use tracing::info;
13
14use crate::client_access::ipv6_prefix_len_of;
15
16/// Tracing target of the block log line.
17pub(crate) const BLOCK_LOG_TARGET: &str = "iota_node::client_blocked";
18
19/// The policy whose limit the client exceeded.
20#[derive(Clone, Copy, Debug, PartialEq, Eq)]
21pub enum BlockReason {
22    Spam,
23    Error,
24}
25
26impl fmt::Display for BlockReason {
27    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
28        f.write_str(match self {
29            Self::Spam => "spam",
30            Self::Error => "error",
31        })
32    }
33}
34
35/// Logs that the node refuses `client` for the next `ttl_secs` seconds.
36/// `client` is the blocklist key that `client_prefix` gives with
37/// `ipv6_prefix_len`, and the line names a cut IPv6 key with its prefix
38/// length, as in `2001:db8:1:2::/64`. Call
39/// it once when a block takes effect, not for every refused request, and not
40/// for a block that dry run keeps from taking effect.
41///
42/// In the plain-text log the line carries no span, so that it reads the same
43/// whichever request caused the block. The JSON log may still add the span of
44/// that request after the `target` key.
45pub(crate) fn log_block(client: IpAddr, ipv6_prefix_len: u8, reason: BlockReason, ttl_secs: u64) {
46    info!(
47        target: BLOCK_LOG_TARGET,
48        parent: None,
49        "{}",
50        block_log_line(client, ipv6_prefix_len, reason, ttl_secs)
51    );
52}
53
54/// An IPv4 client that reached a dual-stack socket shows up as an
55/// IPv4-mapped IPv6 address. The line names it by its IPv4 address, which is
56/// the one a host firewall sees.
57fn block_log_line(
58    client: IpAddr,
59    ipv6_prefix_len: u8,
60    reason: BlockReason,
61    ttl_secs: u64,
62) -> String {
63    let client = match client.to_canonical() {
64        IpAddr::V6(prefix) => match ipv6_prefix_len_of(prefix, ipv6_prefix_len) {
65            128 => prefix.to_string(),
66            prefix_len => format!("{prefix}/{prefix_len}"),
67        },
68        client => client.to_string(),
69    };
70    format!("client_blocked client={client} reason={reason} ttl_secs={ttl_secs}")
71}

85    /// Collects what a subscriber writes, for tests to read back.
86    #[derive(Clone, Default)]
87    pub(crate) struct CapturedLog(Arc<Mutex<Vec<u8>>>);
88
89    impl CapturedLog {
90        pub(crate) fn lines(&self) -> Vec<String> {
91            String::from_utf8(self.0.lock().unwrap().clone())
92                .unwrap()
93                .lines()
94                .map(str::to_owned)
95                .collect()
96        }
97
98        /// The block log lines written so far.
99        pub(crate) fn block_lines(&self) -> Vec<String> {
100            self.lines()
101                .into_iter()
102                .filter(|line| line.contains("client_blocked "))
103                .collect()
104        }
105    }
106
107    impl io::Write for CapturedLog {
108        fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
109            self.0.lock().unwrap().extend_from_slice(buf);
110            Ok(buf.len())
111        }
112
113        fn flush(&mut self) -> io::Result<()> {
114            Ok(())
115        }
116    }
117
118    impl<'a> MakeWriter<'a> for CapturedLog {
119        type Writer = Self;
120
121        fn make_writer(&'a self) -> Self::Writer {
122            self.clone()
123        }
124    }
125
126    /// Runs `f` with a subscriber that writes plain text as the node does
127    /// without `RUST_LOG_JSON`, and returns what it wrote.
128    pub(crate) fn capture_plain(f: impl FnOnce()) -> CapturedLog {
129        let log = CapturedLog::default();
130        let subscriber = tracing_subscriber::fmt()
131            .with_ansi(false)
132            .with_writer(log.clone())
133            .finish();
134        tracing::subscriber::with_default(subscriber, f);
135        log
136    }
137
138    /// Runs `f` with a subscriber that writes JSON as the node does with
139    /// `RUST_LOG_JSON`, and returns what it wrote.
140    fn capture_json(f: impl FnOnce()) -> CapturedLog {
141        let log = CapturedLog::default();
142        let subscriber = tracing_subscriber::fmt()
143            .with_file(true)
144            .with_line_number(true)
145            .json()
146            .with_writer(log.clone())
147            .finish();
148        tracing::subscriber::with_default(subscriber, f);
149        log
150    }

152    #[test]
153    fn the_line_names_the_client_the_reason_and_the_ttl() {
154        let v4 = IpAddr::V4(Ipv4Addr::new(203, 0, 113, 7));
155        assert_eq!(
156            block_log_line(v4, 64, BlockReason::Spam, 300),
157            "client_blocked client=203.0.113.7 reason=spam ttl_secs=300"
158        );
159        let v6 = IpAddr::V6(Ipv6Addr::new(0x2001, 0xdb8, 1, 2, 0, 0, 0, 1));
160        assert_eq!(
161            block_log_line(v6, 128, BlockReason::Error, 300),
162            "client_blocked client=2001:db8:1:2::1 reason=error ttl_secs=300"
163        );
164    }

166    #[test]
167    fn an_ipv6_prefix_is_named_with_its_length() {
168        let prefix = IpAddr::V6(Ipv6Addr::new(0x2001, 0xdb8, 1, 2, 0, 0, 0, 0));
169        assert_eq!(
170            block_log_line(prefix, 64, BlockReason::Spam, 300),

193    #[test]
194    fn the_plain_text_output_holds_the_line_under_its_target() {
195        let client = IpAddr::V4(Ipv4Addr::new(203, 0, 113, 7));
196        let log = capture_plain(|| log_block(client, 64, BlockReason::Spam, 300));
197
198        let [line] = log.lines().try_into().unwrap();
199        assert!(
200            line.ends_with(
201                " INFO iota_node::client_blocked: \
202                 client_blocked client=203.0.113.7 reason=spam ttl_secs=300"
203            ),
204            "{line}"
205        );
206    }

208    #[test]
209    fn the_line_carries_no_span_of_the_request() {
210        let client = IpAddr::V4(Ipv4Addr::new(203, 0, 113, 7));
211        let log = capture_plain(|| {
212            tracing::info_span!("request", peer = "203.0.113.7")
213                .in_scope(|| log_block(client, 64, BlockReason::Spam, 300))
214        });
215
216        let [line] = log.lines().try_into().unwrap();
217        assert!(
218            line.ends_with(
219                " INFO iota_node::client_blocked: \
220                 client_blocked client=203.0.113.7 reason=spam ttl_secs=300"
221            ),
222            "{line}"
223        );
224    }

226    #[test]
227    fn the_json_output_holds_the_line_as_its_message() {
228        let client = IpAddr::V4(Ipv4Addr::new(203, 0, 113, 7));
229        let log = capture_json(|| log_block(client, 64, BlockReason::Error, 300));
230
231        let [line] = log.lines().try_into().unwrap();
232        let event: serde_json::Value = serde_json::from_str(&line).unwrap();
233        assert_eq!(event["level"], "INFO");
234        assert_eq!(event["target"], BLOCK_LOG_TARGET);
235        assert_eq!(
236            event["fields"]["message"],
237            "client_blocked client=203.0.113.7 reason=error ttl_secs=300"
238        );
239    }
