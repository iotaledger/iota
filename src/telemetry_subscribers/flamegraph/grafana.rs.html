16/// Frame in nested set form suitable for Grafana Flame Graph panel.
17/// The type is mainly used with serde to serialize into axum::Json.
18#[derive(Clone, Debug, Eq, Hash, PartialEq, Serialize)]
19pub struct NestedSetFrame<N = f64> {
20    #[serde(serialize_with = "serialize_label")]
21    pub label: FrameLabel,
22    pub level: u32,
23    pub value: N,
24    #[serde(rename = "self")]
25    pub self_: N,
26}
27
28fn serialize_label<S: serde::Serializer>(label: &FrameLabel, s: S) -> Result<S::Ok, S::Error> {
29    s.serialize_str(label.name)
30}
31
32pub trait NestedSetCollector<N = f64> {
33    fn total(&self) -> N;
34    fn collect_nested_set(&self) -> Vec<NestedSetFrame<N>>;
35}

37pub trait Dashboard {
38    fn list_nested_sets(&self) -> Vec<(GraphId, f64)>;
39    fn get_nested_set(&self, graph_id: &str, running: bool, completed: bool)
40    -> Vec<NestedSetFrame>;
41    fn get_nested_sets(
42        &self,
43        label: &'static str,
44        running: bool,
45        completed: bool,
46    ) -> Vec<NestedSetFrame>;
47}

49trait FromDuration: Default + Sized {
50    fn from_duration(total: Duration) -> Self;
51}
52
53impl FromDuration for f64 {
54    fn from_duration(total: Duration) -> Self {
55        // Grafana nested set model mandates the use of ms.
56        // We preserve fractional part for f64.
57        total.as_nanos() as f64 / 1_000_000.0
58    }
59}
60
61impl FromDuration for u64 {
62    fn from_duration(total: Duration) -> Self {
63        total.as_millis() as u64
64    }
65}
66
67impl CallGraph<FlameMetric> {
68    fn nested_set_frame<N: FromDuration>(&self, ix: NodeId, level: u32) -> NestedSetFrame<N> {
69        let node = &self.graph[ix];
70        let value = N::from_duration(node.value.metrics.running.total);
71
72        // "self" duration is recomputed: self total duration - sum of children total
73        // durations
74        let children_total = node
75            .children
76            .iter()
77            .copied()
78            .map(|ix| self.graph[ix].value.metrics.running.total)
79            .sum::<Duration>();
80        assert!(node.value.metrics.running.total >= children_total);
81
82        let self_ = N::from_duration(node.value.metrics.running.total - children_total);
83        NestedSetFrame {
84            label: node.value.label,
85            level,
86            value,
87            self_,
88        }
89    }
90}
91
92impl<N: FromDuration> NestedSetCollector<N> for CallGraph<FlameMetric> {
93    fn total(&self) -> N {
94        N::from_duration(
95            self.root()
96                .map(|frame| frame.metrics.running.total)
97                .unwrap_or_default(),
98        )
99    }
100
101    fn collect_nested_set(&self) -> Vec<NestedSetFrame<N>> {
102        self.graph.dfs_fold(Vec::new(), |frames, node_id, level| {
103            frames.push(self.nested_set_frame(node_id, level as u32))
104        })
105    }
106}
107
108impl<S> Dashboard for Flames<S>
109where
110    S: Clone + Default + MergeMetrics + SpanMetrics,
111    CallGraph<S>: NestedSetCollector,
112{
113    fn list_nested_sets(&self) -> Vec<(GraphId, f64)> {
114        let mut graph_ids: Vec<(GraphId, f64)> = self
115            .graphs
116            .read()
117            .values()
118            .map(|Graph { graph_id, mutex }| (*graph_id, mutex.lock().total()))
119            .collect();
120
121        graph_ids.extend(
122            self.completed
123                .read()
124                .iter()
125                .map(|(graph_id, graph)| (*graph_id, graph.total())),
126        );
127
128        graph_ids.sort_by(|(_, m), (_, n)| n.total_cmp(m));
129        graph_ids
130    }
131
132    fn get_nested_set(
133        &self,
134        graph_id: &str,
135        running: bool,
136        completed: bool,
137    ) -> Vec<NestedSetFrame> {
138        self.get_callgraph(&Metadata::from(graph_id), running, completed)
139            .map(|graph| graph.collect_nested_set())
140            .unwrap_or_default()
141    }
142
143    fn get_nested_sets(
144        &self,
145        label: &'static str,
146        running: bool,
147        completed: bool,
148    ) -> Vec<NestedSetFrame> {
149        let callgraphs = self.get_callgraphs(running, completed);
150        let total = callgraphs.values().map(|graph| graph.total()).sum::<f64>();
151
152        std::iter::once(NestedSetFrame {
153            label: label.into(),
154            level: 0,
155            value: total,
156            self_: 0.0,
157        })
158        .chain(
159            callgraphs
160                .into_values()
161                .flat_map(|flame| flame.collect_nested_set().into_iter())
162                .map(|mut frame| {
163                    frame.level += 1;
164                    frame
165                }),
166        )
167        .collect()
168    }
