39const RECORD_EPOCH_BOUNDARY_TIMEOUT_SECS: u64 = 5;
40
41#[derive(Serialize, Deserialize, Clone, Debug)]
42#[serde(rename_all = "kebab-case")]
43pub struct HistoricalWriterConfig {
44    pub object_store_config: ObjectStoreConfig,
45    pub commit_duration_seconds: u64,
46    /// Optional seed for the epoch boundaries. When set, it initializes the
47    /// epoch boundaries if not already initialized.
48    #[serde(default)]
49    pub epoch_boundaries: Option<EpochBoundaries>,
50}
51
52pub struct HistoricalReducer {
53    remote_store: Arc<DynObjectStore>,
54    commit_duration_ms: u64,
55}
56
57impl HistoricalReducer {
58    pub async fn new(config: HistoricalWriterConfig) -> anyhow::Result<Self> {
59        let remote_store = config.object_store_config.make()?;
60
61        let reducer = Self {
62            remote_store,
63            commit_duration_ms: config.commit_duration_seconds * 1000,
64        };
65
66        if let Some(epoch_boundaries) = config.epoch_boundaries {
67            reducer.seed_epoch_boundaries(epoch_boundaries).await?;
68        }
69
70        Ok(reducer)
71    }
72
73    async fn upload(
74        &self,
75        checkpoint_range: Range<CheckpointSequenceNumber>,
76        data: Bytes,
77    ) -> anyhow::Result<()> {
78        let file_metadata =
79            create_file_metadata_from_bytes(data.clone(), checkpoint_range.clone())?;
80        self.remote_store
81            .put(&file_metadata.file_path(), data.into())
82            .await?;
83        let mut manifest = Self::read_manifest(&self.remote_store).await?;
84        manifest.update(checkpoint_range.end, file_metadata);
85
86        let bytes = finalize_manifest(manifest)?;
87        self.remote_store
88            .put(&Manifest::file_path(), bytes.into())
89            .await?;
90        Ok(())
91    }
92
93    fn prepare_data_to_upload(&self, mut checkpoint_data: Vec<u8>) -> anyhow::Result<Bytes> {
94        let mut buffer = vec![0; MAGIC_BYTES];
95        BigEndian::write_u32(&mut buffer, CHECKPOINT_FILE_MAGIC);
96        buffer.push(StorageFormat::Blob.into());
97        buffer.push(FileCompression::Zstd.into());
98        buffer.append(&mut checkpoint_data);
99        let mut compressed_buffer = vec![];
100        let mut cursor = Cursor::new(buffer);
101        compress(&mut cursor, &mut compressed_buffer)?;
102        Ok(Bytes::from(compressed_buffer))
103    }
104
105    pub async fn get_watermark(&self) -> anyhow::Result<CheckpointSequenceNumber> {
106        let manifest = Self::read_manifest(&self.remote_store).await?;
107        Ok(manifest.next_checkpoint_seq_num())
108    }
109
110    async fn read_manifest(remote_store: &dyn ObjectStore) -> anyhow::Result<Manifest> {
111        Ok(match remote_store.get(&Manifest::file_path()).await {
112            Ok(resp) => read_manifest_from_bytes(resp.bytes().await?.to_vec())?,
113            Err(ObjectStoreError::NotFound { .. }) => Manifest::new(0),
114            Err(err) => Err(err)?,
115        })
116    }
117
118    /// Initializes the epoch boundaries from the provided seed if not already
119    /// initialized.
120    async fn seed_epoch_boundaries(&self, epoch_boundaries: EpochBoundaries) -> anyhow::Result<()> {
121        match write_epoch_boundaries(
122            &epoch_boundaries,
123            self.remote_store.clone(),
124            PutMode::Create,
125        )
126        .await
127        {
128            Ok(_) => {
129                tracing::info!("Initialized epoch boundaries");
130                Ok(())
131            }
132            Err(IngestionError::ObjectStore(ObjectStoreError::AlreadyExists { .. })) => Ok(()),
133            Err(e) => Err(e.into()),
134        }
135    }
136
137    /// Adds a new entry in the epoch boundaries maintained in the remote store.
138    ///
139    /// # Errors
140    ///
141    /// Fails if there is a gap between the epochs already recorded and the
142    /// current one.
143    async fn record_epoch_boundary(
144        &self,
145        epoch_id: EpochId,
146        checkpoint_sequence_number: CheckpointSequenceNumber,
147    ) -> anyhow::Result<()> {
148        let mut epoch_boundaries = read_epoch_boundaries(self.remote_store.clone())
149            .await
150            .unwrap_or_default();
151        epoch_boundaries.insert_next(epoch_id, checkpoint_sequence_number)?;
152
153        let backoff = ExponentialBackoffBuilder::<SystemClock>::new()
154            .with_max_interval(Duration::from_secs(RECORD_EPOCH_BOUNDARY_TIMEOUT_SECS))
155            .build();
156        retry(backoff, || async {
157            write_epoch_boundaries(
158                &epoch_boundaries,
159                self.remote_store.clone(),
160                PutMode::Overwrite,
161            )
162            .await
163            .map_err(|e| {
164                error!("failed to write epoch boundaries to the store: {:?}", &e);
165                backoff::Error::transient(e)
166            })
167        })
168        .await?;
169        Ok(())
170    }
171}

173#[async_trait]
174impl Reducer<RelayWorker> for HistoricalReducer {
175    async fn commit(&self, batch: &[Arc<CheckpointData>]) -> Result<(), anyhow::Error> {
176        if batch.is_empty() {
177            anyhow::bail!("commit batch can't be empty");
178        }
179        let mut buffer = vec![];
180        let first_checkpoint = &batch[0];
181        let start_checkpoint = first_checkpoint.checkpoint_summary.sequence_number;
182        let uploaded_range = start_checkpoint..(start_checkpoint + batch.len() as u64);
183        for checkpoint in batch {
184            let data = Blob::encode(&checkpoint, BlobEncoding::Bcs)?;
185            data.write(&mut buffer)?;
186            if checkpoint.checkpoint_summary.is_last_checkpoint_of_epoch() {
187                self.record_epoch_boundary(
188                    checkpoint.checkpoint_summary.epoch,
189                    checkpoint.checkpoint_summary.sequence_number,
190                )
191                .await?;
192            }
193        }
194        self.upload(uploaded_range.clone(), self.prepare_data_to_upload(buffer)?)
195            .await?;
196        tracing::info!("committed checkpoints {uploaded_range:?} to the remote store");
197        Ok(())
198    }
199
200    fn should_close_batch(
201        &self,
202        batch: &[Arc<CheckpointData>],
203        next_item: Option<&Arc<CheckpointData>>,
204    ) -> bool {
205        // never close a batch without a trigger condition
206        if batch.is_empty() || next_item.is_none() {
207            return false;
208        }
209        let first_checkpoint = &batch[0].checkpoint_summary;
210        let next_checkpoint = next_item.expect("invariant's checked");
211        // close batch after genesis
212        if next_checkpoint.checkpoint_summary.sequence_number == 1 {
213            return true;
214        }
215        next_checkpoint.checkpoint_summary.epoch != first_checkpoint.epoch
216            || next_checkpoint.checkpoint_summary.timestamp_ms
217                > (self.commit_duration_ms + first_checkpoint.timestamp_ms)
218    }
