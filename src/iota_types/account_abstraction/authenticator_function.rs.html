17pub const AUTHENTICATOR_FUNCTION_MODULE_NAME: Identifier =
18    Identifier::from_static("authenticator_function");
19pub const AUTHENTICATOR_FUNCTION_REF_V1_STRUCT_NAME: Identifier =
20    Identifier::from_static("AuthenticatorFunctionRefV1");
21
22/// An enum representing different versions of AuthenticatorFunctionRef. This is
23/// used to represent the reference to an authenticator function in Move.
24#[derive(Debug, Serialize, Deserialize, Clone, Eq, PartialEq)]
25pub enum AuthenticatorFunctionRef {
26    V1(AuthenticatorFunctionRefV1),
27}
28
29impl From<AuthenticatorFunctionRef> for Option<AuthenticatorFunctionRefV1> {
30    fn from(authenticator_function_ref: AuthenticatorFunctionRef) -> Self {
31        match authenticator_function_ref {
32            AuthenticatorFunctionRef::V1(v1) => Some(v1),
33        }
34    }
35}
36
37#[derive(Debug, Serialize, Deserialize, Clone, Eq, PartialEq)]
38pub struct AuthenticatorFunctionRefV1 {
39    pub package: ObjectId,
40    pub module: String,
41    pub function: String,
42}
43
44impl AuthenticatorFunctionRefV1 {
45    pub fn type_(type_param: StructTag) -> StructTag {
46        StructTag::new(
47            Address::FRAMEWORK,
48            AUTHENTICATOR_FUNCTION_MODULE_NAME,
49            AUTHENTICATOR_FUNCTION_REF_V1_STRUCT_NAME,
50            vec![TypeTag::Struct(Box::new(type_param))],
51        )
52    }
53
54    pub fn from_bcs_bytes(content: &[u8]) -> Result<Self, IotaError> {
55        bcs::from_bytes(content).map_err(|err| IotaError::ObjectDeserialization {
56            error: format!("Unable to deserialize AuthenticatorFunctionRefV1 object: {err}"),
57        })
58    }
59
60    pub fn is_authenticator_function_ref_v1(tag: &StructTag) -> bool {
61        tag.address() == Address::FRAMEWORK
62            && tag.module() == &AUTHENTICATOR_FUNCTION_MODULE_NAME
63            && tag.name() == &AUTHENTICATOR_FUNCTION_REF_V1_STRUCT_NAME
64    }
65}
66
67impl TryFrom<Object> for AuthenticatorFunctionRefV1 {
68    type Error = IotaError;
69    fn try_from(object: Object) -> Result<Self, Self::Error> {
70        if let Some(o) = object.data.as_opt_struct() {
71            if AuthenticatorFunctionRefV1::is_authenticator_function_ref_v1(o.struct_tag()) {
72                return AuthenticatorFunctionRefV1::from_bcs_bytes(o.contents());
73            }
74        }
75
76        Err(IotaError::Type {
77            error: format!("Object type is not a AuthenticatorFunctionRefV1: {object:?}"),
78        })
79    }
80}

82/// A struct used to hold AuthenticatorFunctionRef and
83/// DynamicallyLoadedObjectMetadata together, in order to pass this information
84/// to the execution side.
85#[derive(Debug, Serialize, Deserialize, Clone, Eq, PartialEq)]
86pub struct AuthenticatorFunctionRefForExecution {
87    pub authenticator_function_ref: AuthenticatorFunctionRef,
88    pub loaded_object_id: ObjectId,
89    pub loaded_object_metadata: DynamicallyLoadedObjectMetadata,
90}
91
92impl AuthenticatorFunctionRefForExecution {
93    pub fn new_v1(
94        authenticator_function_ref: AuthenticatorFunctionRefV1,
95        loaded_object_ref: ObjectReference,
96        owner: Owner,
97        storage_rebate: u64,
98        previous_transaction: TransactionDigest,
99    ) -> Self {
100        Self {
101            authenticator_function_ref: AuthenticatorFunctionRef::V1(authenticator_function_ref),
102            loaded_object_id: loaded_object_ref.object_id,
103            loaded_object_metadata: DynamicallyLoadedObjectMetadata {
104                version: loaded_object_ref.version,
105                digest: loaded_object_ref.digest,
106                owner,
107                storage_rebate,
108                previous_transaction,
109            },
110        }
111    }
112}

114/// Derive the id of the dynamic field on the account object that holds its
115/// [`AuthenticatorFunctionRefV1`].
116pub fn derive_authenticator_function_ref_v1_dynamic_field_id(
117    account_object_id: impl Into<ObjectId>,
118) -> UserInputResult<ObjectId> {
119    let account_object_id = account_object_id.into();
120    dynamic_field::derive_dynamic_field_id(
121        account_object_id,
122        &StructTag::new_authenticator_function_ref_v1_key().into(),
123        &AuthenticatorFunctionRefV1Key::default().to_bcs_bytes(),
124    )
125    .map_err(|_| UserInputError::UnableToGetMoveAuthenticatorId { account_object_id })
126}
127
128/// Decode a loaded authenticator dynamic-field object (see
129/// [`derive_authenticator_function_ref_v1_dynamic_field_id`]) into an
130/// [`AuthenticatorFunctionRefForExecution`].
131pub fn authenticator_function_ref_v1_from_dynamic_field_object(
132    account_object_id: ObjectId,
133    field_obj: &Object,
134) -> UserInputResult<AuthenticatorFunctionRefForExecution> {
135    // A dynamic field is never a package object, so a non-struct here means the
136    // object at the derived id is not the authenticator field.
137    let field_move_object = field_obj
138        .data
139        .as_opt_struct()
140        .ok_or(UserInputError::InvalidAuthenticatorFunctionRefField { account_object_id })?;
141
142    let field: Field<AuthenticatorFunctionRefV1Key, AuthenticatorFunctionRefV1> = field_move_object
143        .to_rust()
144        .map_err(|_| UserInputError::InvalidAuthenticatorFunctionRefField { account_object_id })?;
145
146    Ok(AuthenticatorFunctionRefForExecution::new_v1(
147        field.value,
148        field_obj.object_ref(),
149        field_obj.owner,
150        field_obj.storage_rebate,
151        field_obj.previous_transaction,
152    ))
153}

155/// Extracts the sender's and sponsor's [`AuthenticatorFunctionRef`] by calling
156/// `find_ref` for `sender` and, when the gas owner differs, for `gas_owner`.
157pub fn extract_auth_fun_refs(
158    sender: Address,
159    gas_owner: Address,
160    find_ref: impl Fn(Address) -> Option<AuthenticatorFunctionRef>,
161) -> (
162    Option<AuthenticatorFunctionRef>,
163    Option<AuthenticatorFunctionRef>,
164) {
165    (
166        find_ref(sender),
167        if gas_owner != sender {
168            find_ref(gas_owner)
169        } else {
170            None
171        },
172    )
173}
