4use iota_sdk_types::{Address, Identifier, StructTag};
5use serde::{Deserialize, Serialize};
6
7use crate::{
8    IOTA_FRAMEWORK_PACKAGE_ID,
9    account_abstraction::authenticator_function::AuthenticatorFunctionRefV1,
10};
11
12pub const BUILTIN_AUTHENTICATOR_FUNCTIONS_MODULE_NAME: Identifier =
13    Identifier::from_static("builtin_authenticator_functions");
14
15pub const PUBLIC_KEY_FIELD_NAME_STRUCT_NAME: Identifier =
16    Identifier::from_static("PublicKeyFieldName");
17
18pub const BUILTIN_AUTHENTICATOR_FUNCTION_V1_NAME: &str = "builtin_authenticator_v1";
19
20/// The `#[error]` code of `builtin_authenticator_functions::EInvalidSignature`,
21/// the abort raised when the built-in authenticator rejects a signature.
22pub const INVALID_SIGNATURE_ERROR_CODE: u64 = 20;
23
24#[derive(Debug, Default, Serialize, Deserialize, Clone, Eq, PartialEq)]
25pub struct PublicKeyFieldName {
26    // This field is required to make a Rust struct compatible with an empty Move one.
27    // An empty Move struct contains a 1-byte dummy bool field because empty fields are not
28    // allowed in the bytecode.
29    dummy_field: bool,
30}
31
32impl PublicKeyFieldName {
33    pub fn tag() -> StructTag {
34        StructTag::new(
35            Address::FRAMEWORK,
36            BUILTIN_AUTHENTICATOR_FUNCTIONS_MODULE_NAME,
37            PUBLIC_KEY_FIELD_NAME_STRUCT_NAME,
38            Vec::new(),
39        )
40    }
41
42    pub fn to_bcs_bytes(&self) -> Vec<u8> {
43        bcs::to_bytes(&self).expect("PublicKeyFieldName is always BCS-serializable")
44    }
45}
46
47/// Returns an authentication function that references the built-in
48/// authenticator.
49pub fn builtin_authenticator_function_ref_v1() -> AuthenticatorFunctionRefV1 {
50    AuthenticatorFunctionRefV1 {
51        package: IOTA_FRAMEWORK_PACKAGE_ID,
52        module: BUILTIN_AUTHENTICATOR_FUNCTIONS_MODULE_NAME.to_string(),
53        function: BUILTIN_AUTHENTICATOR_FUNCTION_V1_NAME.to_string(),
54    }
55}
56
57/// Returns true if `authenticator_function_ref` references the built-in
58/// authenticator.
59pub fn is_builtin_authenticator_function_ref(
60    authenticator_function_ref: &AuthenticatorFunctionRefV1,
61) -> bool {
62    *authenticator_function_ref == builtin_authenticator_function_ref_v1()
63}
