30/// Represents a transaction combined with its effects and events for efficient
31/// batch processing
32#[derive(Clone, Debug)]
33pub struct TransactionWithEffectsAndEvents {
34    pub transaction: Arc<VerifiedTransaction>,
35    pub effects: TransactionEffects,
36    pub events: Option<TransactionEvents>,
37}

343    fn try_multi_get_transactions_with_events_and_effects(
344        &self,
345        transaction_digests: Vec<TransactionDigest>,
346    ) -> anyhow::Result<Vec<TransactionWithEffectsAndEvents>> {
347        use crate::effects::TransactionEffectsAPI;
348
349        // Batch read all transactions
350        let transactions = self
351            .try_multi_get_transactions(&transaction_digests)?
352            .into_iter()
353            .map(|maybe_transaction| {
354                maybe_transaction.ok_or_else(|| anyhow::anyhow!("missing transaction"))
355            })
356            .collect::<anyhow::Result<Vec<_>>>()?;
357
358        // Batch read all effects
359        let effects = self
360            .try_multi_get_transaction_effects(&transaction_digests)?
361            .into_iter()
362            .map(|maybe_effects| maybe_effects.ok_or_else(|| anyhow::anyhow!("missing effects")))
363            .collect::<anyhow::Result<Vec<_>>>()?;
364
365        // Extract transaction digests for transactions that have events
366        let event_tx_digests = transaction_digests
367            .iter()
368            .zip(effects.iter())
369            .filter_map(|(tx_digest, fx)| fx.events_digest().map(|_| *tx_digest))
370            .collect::<Vec<_>>();
371
372        // Batch read all events
373        let events = self
374            .try_multi_get_events(&event_tx_digests)?
375            .into_iter()
376            .zip(event_tx_digests)
377            .map(|(maybe_event, tx_digest)| {
378                maybe_event
379                    .ok_or_else(|| anyhow::anyhow!("missing event"))
380                    .map(|event| (tx_digest, event))
381            })
382            .collect::<anyhow::Result<HashMap<_, _>>>()?;
383
384        // Collect the final result
385        let result = transactions
386            .into_iter()
387            .zip(effects)
388            .map(|(transaction, effects)| {
389                let events = effects
390                    .events_digest()
391                    .and_then(|_| events.get(effects.transaction_digest()).cloned());
392
393                TransactionWithEffectsAndEvents {
394                    transaction,
395                    effects,
396                    events,
397                }
398            })
399            .collect();
400
401        Ok(result)
402    }

404    fn get_checkpoint_transaction(
405        &self,
406        tx_with_events_and_effects: TransactionWithEffectsAndEvents,
407    ) -> anyhow::Result<CheckpointTransaction> {
408        let input_objects =
409            get_transaction_input_objects(&self, &tx_with_events_and_effects.effects)?;
410        let output_objects =
411            get_transaction_output_objects(&self, &tx_with_events_and_effects.effects)?;
412
413        let full_transaction = CheckpointTransaction {
414            transaction: (*tx_with_events_and_effects.transaction).clone().into(),
415            effects: tx_with_events_and_effects.effects,
416            events: tx_with_events_and_effects.events,
417            input_objects,
418            output_objects,
419        };
420
421        Ok(full_transaction)
422    }

424    /// Stream checkpoint transactions individually to avoid large memory
425    /// footprint. Returns a stream of individual CheckpointTransaction items
426    /// along with metadata
427    fn stream_checkpoint_transactions(
428        &self,
429        checkpoint_contents: CheckpointContents,
430    ) -> std::pin::Pin<
431        Box<dyn futures::Stream<Item = anyhow::Result<CheckpointTransaction>> + Send + '_>,
432    >
433    where
434        Self: Sync,
435    {
436        Box::pin(async_stream::stream! {
437            let transaction_digests = checkpoint_contents
438                .iter()
439                .map(|execution_digests| execution_digests.transaction)
440                .collect::<Vec<_>>();
441
442            let txs_with_events_and_effects = self
443                .try_multi_get_transactions_with_events_and_effects(transaction_digests)?;
444
445            for tx_with_events_and_effects in txs_with_events_and_effects {
446                yield self.get_checkpoint_transaction(tx_with_events_and_effects);
447            }
448        })
449    }

451    // Fetch all checkpoint data
452    // TODO fix return type to not be anyhow
453    fn try_get_checkpoint_data(
454        &self,
455        checkpoint: VerifiedCheckpoint,
456        checkpoint_contents: CheckpointContents,
457    ) -> anyhow::Result<CheckpointData> {
458        let transaction_digests = checkpoint_contents
459            .iter()
460            .map(|execution_digests| execution_digests.transaction)
461            .collect::<Vec<_>>();
462
463        let txs_with_events_and_effects =
464            self.try_multi_get_transactions_with_events_and_effects(transaction_digests)?;
465
466        let mut transactions = Vec::with_capacity(txs_with_events_and_effects.len());
467        for tx_with_events_and_effects in txs_with_events_and_effects {
468            transactions.push(self.get_checkpoint_transaction(tx_with_events_and_effects)?);
469        }
470
471        let checkpoint_data = CheckpointData {
472            checkpoint_summary: checkpoint.into(),
473            checkpoint_contents,
474            transactions,
475        };
476
477        Ok(checkpoint_data)
478    }

880        input_objects: &[Object],
881        output_objects: &[Object],

893        }
894    }

897/// Epoch information structure for indexing.
898///
899/// Contains metadata about an epoch including timing, checkpoints, protocol
900/// version, and a snapshot of the system state at the start of the epoch.
901#[derive(Clone, Serialize, Deserialize, PartialEq, Eq, Debug)]
902pub struct EpochInfo {
903    pub epoch: u64,
904    pub protocol_version: u64,
905    pub start_timestamp_ms: u64,
906    pub end_timestamp_ms: Option<u64>,
907    pub start_checkpoint: u64,
908    pub end_checkpoint: Option<u64>,
909    pub reference_gas_price: u64,
910    /// `IotaSystemState` of object `0x5` right after the AdvanceEpoch tx of
911    /// the previous epoch (or the genesis tx for epoch 0).
912    pub system_state: crate::iota_system_state::IotaSystemState,
913}
914
915/// Epoch information structure for indexing.
916///
917/// Stores only the start-of-epoch identity (`epoch`, `start_checkpoint`,
918/// `start_timestamp_ms`, `system_state`) plus the close-of-epoch
919/// `epoch_close_proof`. Everything derivable from those — `protocol_version`,
920/// `reference_gas_price`, `end_timestamp_ms`, `end_checkpoint` — is a method,
921/// not a stored field.
922#[derive(Clone, Debug, Serialize, Deserialize)]
923pub struct EpochInfoV2 {
924    pub epoch: u64,
925    pub start_checkpoint: CheckpointSequenceNumber,
926    pub start_timestamp_ms: u64,
927    /// `IotaSystemState` of object `0x5` at this epoch's start.
928    pub system_state: IotaSystemState,
929    /// Close-of-epoch proof bundle; `None` until this epoch's boundary is
930    /// indexed. The row is finalized exactly when this is `Some`.
931    pub epoch_close_proof: Option<EpochInfoV1Entry>,
932}
933
934impl EpochInfoV2 {
935    /// Whether this epoch's boundary has been indexed.
936    pub fn is_finalized(&self) -> bool {
937        self.epoch_close_proof.is_some()
938    }
939
940    /// Protocol version in effect this epoch (from the start system state).
941    pub fn protocol_version(&self) -> u64 {
942        self.system_state.protocol_version()
943    }
944
945    /// Reference gas price for this epoch (from the start system state).
946    pub fn reference_gas_price(&self) -> u64 {
947        self.system_state.reference_gas_price()
948    }
949
950    /// Timestamp of this epoch's last checkpoint; `None` until finalized.
951    pub fn end_timestamp_ms(&self) -> Option<u64> {
952        self.epoch_close_proof
953            .as_ref()
954            .map(|entry| entry.last_checkpoint_summary.data().timestamp_ms)
955    }
956
957    /// This epoch's last checkpoint sequence number; `None` until finalized.
958    pub fn end_checkpoint(&self) -> Option<CheckpointSequenceNumber> {
959        self.epoch_close_proof
960            .as_ref()
961            .map(|entry| entry.last_checkpoint_summary.data().sequence_number())
962    }
963}

965/// Per-epoch entry of the snapshot `EPOCH_INFO` file. Every field is anchored
966/// to the certified `last_checkpoint_summary`.
967#[derive(Clone, Debug, Serialize, Deserialize)]
968pub struct EpochInfoV1Entry {
969    /// Certified summary of this epoch's closing checkpoint — the signed
970    /// anchor every other field is proven against.
971    pub last_checkpoint_summary: CertifiedCheckpointSummary,
972    /// Contents of that closing checkpoint; `hash == last_checkpoint_summary`'s
973    /// `contents_digest`.
974    pub last_checkpoint_contents: CheckpointContents,
975    /// Effects of the epoch-change tx (the last tx of the closing checkpoint);
976    /// its `(transaction, effects)` digest pair is the last entry of
977    /// `last_checkpoint_contents`.
978    pub end_of_epoch_tx_effects: TransactionEffects,
979    /// Events from the epoch-change tx (carries `SystemEpochInfoEvent`); `hash
980    /// == end_of_epoch_tx_effects.events_digest`, or empty on safe-mode
981    /// boundaries where that digest is `None`.
982    pub end_of_epoch_tx_events: TransactionEvents,
983    /// Raw serialized bytes of system-state wrapper object and its inner
984    /// system-state object as written by this boundary — the next epoch's
985    /// start state. Each object's digest matches a written-object entry in
986    /// `end_of_epoch_tx_effects`.
987    pub next_epoch_start_system_state_objects: Vec<Vec<u8>>,
988}
