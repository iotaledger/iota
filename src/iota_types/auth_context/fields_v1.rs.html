35/// Mirrors [`iota_sdk_types::MoveCall`] for use in
36/// [`MoveCommand`], substituting [`TypeTag`] for a string in the type arguments
37/// so that the type matches the BCS layout expected by the Move-side
38/// `ptb_command::ProgrammableMoveCall`.
39#[serde_as]
40#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
41pub struct MoveProgrammableMoveCall {
42    pub package: ObjectId,
43    pub module: String,
44    pub function: String,
45    #[serde_as(as = "Vec<TypeName>")]
46    pub type_arguments: Vec<TypeTag>,
47    pub arguments: Vec<Argument>,
48}

50// ---------------------------------------------------------------------------
51// MoveCommand
52// ---------------------------------------------------------------------------
53
54/// Mirrors [`iota_sdk_types::Command`], substituting [`TypeTag`] for
55/// a string in `MoveCall` and `MakeMoveVec` so that
56/// the type matches the BCS layout expected by the Move-side
57/// `ptb_command::Command`.
58#[serde_as]
59#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
60pub enum MoveCommand {
61    MoveCall(Box<MoveProgrammableMoveCall>),
62    TransferObjects(Vec<Argument>, Argument),
63    SplitCoins(Argument, Vec<Argument>),
64    MergeCoins(Argument, Vec<Argument>),
65    Publish(Vec<Vec<u8>>, Vec<ObjectId>),
66    MakeMoveVec(
67        #[serde_as(as = "Option<TypeName>")] Option<TypeTag>,
68        Vec<Argument>,
69    ),
70    Upgrade(Vec<Vec<u8>>, Vec<ObjectId>, ObjectId, Argument),
71}
72
73impl From<&Command> for MoveCommand {
74    fn from(cmd: &Command) -> Self {
75        match cmd {
76            Command::MoveCall(cmd) => MoveCommand::MoveCall(Box::new(MoveProgrammableMoveCall {
77                package: cmd.package,
78                module: cmd.module.to_string(),
79                function: cmd.function.to_string(),
80                type_arguments: cmd.type_arguments.clone(),
81                arguments: cmd.arguments.clone(),
82            })),
83            Command::TransferObjects(cmd) => {
84                MoveCommand::TransferObjects(cmd.objects.clone(), cmd.address)
85            }
86            Command::SplitCoins(cmd) => MoveCommand::SplitCoins(cmd.coin, cmd.amounts.clone()),
87            Command::MergeCoins(cmd) => {
88                MoveCommand::MergeCoins(cmd.coin, cmd.coins_to_merge.clone())
89            }
90            Command::Publish(cmd) => {
91                MoveCommand::Publish(cmd.modules.clone(), cmd.dependencies.clone())
92            }
93            Command::MakeMoveVector(cmd) => {
94                MoveCommand::MakeMoveVec(cmd.type_tag.clone(), cmd.elements.clone())
95            }
96            Command::Upgrade(cmd) => MoveCommand::Upgrade(
97                cmd.modules.clone(),
98                cmd.dependencies.clone(),
99                cmd.package,
100                cmd.ticket,
101            ),
102            _ => unimplemented!("a new Command enum variant was added and needs to be handled"),
103        }
104    }
105}

107impl MoveCommand {
108    pub fn type_() -> StructTag {
109        StructTag {
110            address: IOTA_FRAMEWORK_ADDRESS,
111            module: COMMAND_MODULE_NAME.to_owned(),
112            name: COMMAND_STRUCT_NAME.to_owned(),
113            type_params: vec![],
114        }
115    }
116}

146/// Mirrors [`crate::transaction::CallArg`], matching the BCS layout expected
147/// by the Move-side `ptb_call_arg::CallArg`.
148///
149/// A transaction dependency input has no variant of its own. It is shown as
150/// `Pure` holding the BCS bytes of the digest, exactly like an unused pure
151/// input with the same bytes. A Move authenticator therefore cannot use the
152/// inputs to check that a transaction depends on a given digest; only the
153/// bytes returned by `auth_context::tx_data_bytes` tell the two apart.
154#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
155pub enum MoveCallArg {
156    Pure(Vec<u8>),
157    Object(MoveObjectArg),
158}
159
160impl From<&CallArg> for MoveCallArg {
161    fn from(arg: &CallArg) -> Self {
162        match arg {
163            CallArg::Pure(bytes) => MoveCallArg::Pure(bytes.clone()),
164            CallArg::ImmutableOrOwned(obj_arg) => {
165                MoveCallArg::Object(MoveObjectArg::ImmOrOwnedObject(*obj_arg))
166            }
167            CallArg::Shared(obj_arg) => MoveCallArg::Object(MoveObjectArg::SharedObject {
168                id: obj_arg.object_id,
169                initial_shared_version: obj_arg.initial_shared_version,
170                mutable: obj_arg.mutable,
171            }),
172            CallArg::Receiving(obj_arg) => MoveCallArg::Object(MoveObjectArg::Receiving(*obj_arg)),
173            // The Move enum cannot gain a variant in a framework upgrade, so a
174            // transaction dependency is shown as pure data: the BCS bytes of
175            // its digest, which Move reads as a `vector<u8>`. The input keeps
176            // its index, and no command can use it.
177            CallArg::TransactionDependency(digest) => MoveCallArg::Pure(
178                bcs::to_bytes(digest).expect("a transaction digest serializes to BCS"),
179            ),
180            _ => unimplemented!("a new CallArg enum variant was added and needs to be handled"),
181        }
182    }
183}

193    }

222    /// BCS round-trip helper.
223    fn round_trip<T>(value: &T) -> T
224    where
225        T: serde::Serialize + for<'de> serde::Deserialize<'de> + PartialEq + std::fmt::Debug,
226    {
227        let bytes = bcs::to_bytes(value).unwrap();
228        bcs::from_bytes(&bytes).unwrap()
229    }

391    /// Primitive TypeTag variants (Bool, U8, …) must be converted to their
392    /// canonical string representation as TypeTag.
393    #[test]
394    fn command_from_move_call_primitive_type_tag() {
395        let cases = [
396            (TypeTag::Bool, "bool"),
397            (TypeTag::U8, "u8"),
398            (TypeTag::U64, "u64"),
399            (TypeTag::U128, "u128"),
400            (TypeTag::U16, "u16"),
401            (TypeTag::U32, "u32"),
402            (TypeTag::U256, "u256"),
403            (TypeTag::Address, "address"),
404        ];
405        for (type_tag, expected_name) in cases {
406            let cmd = Command::new_move_call(
407                obj_id(),
408                Identifier::new_unchecked("m"),
409                Identifier::new_unchecked("f"),
410                vec![type_tag],
411                vec![],
412            );
413            let MoveCommand::MoveCall(call) = MoveCommand::from(&cmd) else {
414                panic!("expected MoveCall");
415            };
416            assert_eq!(
417                call.type_arguments,
418                vec![TypeTag::from_str(expected_name).unwrap()],
419                "failed for {expected_name}"
420            );
421        }
422    }
