38/// Returns `Ok(())` if no input or receiving object's coin type is on a deny
39/// list for the given `address`.
40///
41/// The check reads the latest deny-list value, so denials take effect
42/// immediately; the result depends on how far the validator's execution has
43/// progressed, making it suitable only for validator-local decisions, such as
44/// pre-consensus admission. Post-consensus validation uses
45/// [`coin_deny_list_v1_denial`] instead.
46#[instrument(level = "trace", skip_all)]
47pub fn check_coin_deny_list_v1(
48    address: Address,
49    tx_input_objects: &InputObjects,
50    tx_receiving_objects: &ReceivingObjects,
51    per_authenticator_input_objects: &Vec<&InputObjects>,
52    object_store: &dyn ObjectStore,
53) -> UserInputResult {
54    let coin_types = input_object_coin_types_for_denylist_check(
55        tx_input_objects,
56        tx_receiving_objects,
57        per_authenticator_input_objects,
58    );
59
60    match coin_type_denial(address, coin_types, object_store) {
61        Some(CoinTypeDenial::GlobalPause(coin_type)) => {
62            Err(UserInputError::CoinTypeGlobalPause { coin_type })
63        }
64        Some(CoinTypeDenial::AddressDenied(coin_type)) => {
65            Err(UserInputError::AddressDeniedForCoin { address, coin_type })
66        }
67        None => Ok(()),
68    }
69}

71/// Fails if `sender` is denied for one of `coin_types`, or one of them is
72/// paused for every address. `coin_types` are the coin types of the coins the
73/// Move code received. This is the check that [`check_coin_deny_list_v1`] makes
74/// before submission, here with the deny list read as it stood at the start of
75/// `cur_epoch`, so every validator reaches the same result.
76///
77/// # Panics
78///
79/// Panics if `object_store` fails to read the deny list. Reading the failure
80/// as not denied, or as an execution error, would give effects that differ
81/// from those of the other validators.
82pub fn check_coin_deny_list_v1_for_sender_during_execution(
83    sender: Address,
84    coin_types: BTreeSet<String>,
85    cur_epoch: EpochId,
86    object_store: &dyn ObjectStore,
87) -> Result<(), ExecutionError> {
88    let denial = try_coin_type_denial(sender, coin_types, object_store, Some(cur_epoch))
89        .expect("the deny list must be readable during execution");
90    match denial {
91        Some(CoinTypeDenial::GlobalPause(coin_type)) => Err(ExecutionError::new(
92            ExecutionErrorKind::CoinTypeGlobalPause { coin_type },
93            None,
94        )),
95        Some(CoinTypeDenial::AddressDenied(coin_type)) => Err(ExecutionError::new(
96            ExecutionErrorKind::AddressDeniedForCoin {
97                address: sender,
98                coin_type,
99            },
100            None,
101        )),
102        None => Ok(()),
103    }
104}

106/// The error [`check_coin_deny_list_v1`] gives when `address` may not use one
107/// of `coin_types`, or `None` if it may use them all. Types are checked in
108/// sorted order, and for each type a global pause before a denial of `address`,
109/// so the same deny list always gives the same error.
110///
111/// Meant for post-consensus validation, where `object_store` answers as of a
112/// commit: a missing deny-list entry reads as not denied, and any other error
113/// of `object_store` is returned instead of being read as not denied, which
114/// would differ between validators.
115pub fn coin_deny_list_v1_denial(
116    address: Address,
117    coin_types: BTreeSet<String>,
118    object_store: &dyn ObjectStore,
119) -> IotaResult<Option<UserInputError>> {
120    for coin_type in coin_types {
121        let Some(deny_list) = try_get_per_type_coin_deny_list_v1(&coin_type, object_store)? else {
122            continue;
123        };
124
125        if try_read_config_setting(object_store, &deny_list, GlobalPauseKey::default(), None)?
126            .unwrap_or(false)
127        {
128            return Ok(Some(UserInputError::CoinTypeGlobalPause { coin_type }));
129        }
130
131        if try_read_config_setting(object_store, &deny_list, AddressKey(address), None)?
132            .unwrap_or(false)
133        {
134            return Ok(Some(UserInputError::AddressDeniedForCoin {
135                address,
136                coin_type,
137            }));
138        }
139    }
140
141    Ok(None)
142}

144/// A coin type an address may not use, and the reason: the type is paused
145/// for every address, or the address is denied for it.
146enum CoinTypeDenial {
147    GlobalPause(String),
148    AddressDenied(String),
149}
150
151/// Returns the first of `coin_types` (in sorted order) that `address` may
152/// not use, with the reason, or `None` if it may use them all. For each type,
153/// a global pause is checked before a denial of `address`, in the latest deny
154/// list. Every error of `object_store` reads as not denied, as for
155/// [`get_per_type_coin_deny_list_v1`]; [`try_coin_type_denial`] returns them.
156fn coin_type_denial(
157    address: Address,
158    coin_types: BTreeSet<String>,
159    object_store: &dyn ObjectStore,
160) -> Option<CoinTypeDenial> {
161    for coin_type in coin_types {
162        let Some(deny_list) = get_per_type_coin_deny_list_v1(&coin_type, object_store) else {
163            continue;
164        };
165
166        if check_global_pause(&deny_list, object_store, None) {
167            return Some(CoinTypeDenial::GlobalPause(coin_type));
168        }
169
170        if check_address_denied_by_config(&deny_list, address, object_store, None) {
171            return Some(CoinTypeDenial::AddressDenied(coin_type));
172        }
173    }
174
175    None
176}

178/// [`coin_type_denial`] that returns the errors of `object_store` other than
179/// a missing or undecodable entry.
180fn try_coin_type_denial(
181    address: Address,
182    coin_types: BTreeSet<String>,
183    object_store: &dyn ObjectStore,
184    cur_epoch: Option<EpochId>,
185) -> IotaResult<Option<CoinTypeDenial>> {
186    for coin_type in coin_types {
187        let Some(deny_list) = try_get_per_type_coin_deny_list_v1(&coin_type, object_store)? else {
188            continue;
189        };
190
191        if try_read_config_setting(
192            object_store,
193            &deny_list,
194            GlobalPauseKey::default(),
195            cur_epoch,
196        )?
197        .unwrap_or(false)
198        {
199            return Ok(Some(CoinTypeDenial::GlobalPause(coin_type)));
200        }
201
202        if try_read_config_setting(object_store, &deny_list, AddressKey(address), cur_epoch)?
203            .unwrap_or(false)
204        {
205            return Ok(Some(CoinTypeDenial::AddressDenied(coin_type)));
206        }
207    }
208
209    Ok(None)
210}

212/// Returns 1) whether the coin deny list check passed,
213///         2) the deny lists checked
214///         2) the number of regulated coin owners checked.
215pub fn check_coin_deny_list_v1_during_execution(
216    written_objects: &BTreeMap<ObjectId, Object>,
217    cur_epoch: EpochId,
218    object_store: &dyn ObjectStore,
219) -> DenyListResult {
220    let mut new_coin_owners = BTreeMap::new();
221    for obj in written_objects.values() {
222        if obj.is_gas_coin() {
223            continue;
224        }
225        let Some(coin_type) = obj.opt_coin_type() else {
226            continue;
227        };
228        let Some(owner) = obj.owner.as_opt_address() else {
229            continue;
230        };
231        new_coin_owners
232            .entry(coin_type.to_canonical_string(false))
233            .or_insert_with(BTreeSet::new)
234            .insert(*owner);
235    }
236    let num_non_gas_coin_owners = new_coin_owners.values().map(|v| v.len() as u64).sum();
237    let new_regulated_coin_owners = new_coin_owners
238        .into_iter()
239        .filter_map(|(coin_type, owners)| {
240            let deny_list_config = get_per_type_coin_deny_list_v1(&coin_type, object_store)?;
241            Some((coin_type, (deny_list_config, owners)))
242        })
243        .collect::<BTreeMap<_, _>>();
244    let result =
245        check_new_regulated_coin_owners(new_regulated_coin_owners, cur_epoch, object_store);
246    // `num_non_gas_coin_owners` is used to charge for gas. As such we must be
247    // extremely careful to not use a number that is not consistent across all
248    // validators. For example, relying on the number of coins with a deny list
249    // is _not_ consistent since the deny list is created on the first addition
250    // to the deny list. But the total number of coins/owners denied would
251    // be consistent since we rely on the results from the last epoch (i.e. relying
252    // on the Config's internal invariants)
253    DenyListResult {
254        result,
255        num_non_gas_coin_owners,
256    }
257}
258
259fn check_new_regulated_coin_owners(
260    new_regulated_coin_owners: BTreeMap<String, (Config<ConfigWriteCap>, BTreeSet<Address>)>,
261    cur_epoch: EpochId,
262    object_store: &dyn ObjectStore,
263) -> Result<(), ExecutionError> {
264    for (coin_type, (deny_list, owners)) in new_regulated_coin_owners {
265        if check_global_pause(&deny_list, object_store, Some(cur_epoch)) {
266            return Err(ExecutionError::new(
267                ExecutionErrorKind::CoinTypeGlobalPause { coin_type },
268                None,
269            ));
270        }
271        for owner in owners {
272            if check_address_denied_by_config(&deny_list, owner, object_store, Some(cur_epoch)) {
273                return Err(ExecutionError::new(
274                    ExecutionErrorKind::AddressDeniedForCoin {
275                        address: owner,
276                        coin_type,
277                    },
278                    None,
279                ));
280            }
281        }
282    }
283    Ok(())
284}

286/// The deny-list config of `coin_type`, or `None` if it has none. Every error
287/// of `object_store` also reads as `None`, so as not denied. That is
288/// deliberate for the callers outside post-consensus validation at a commit,
289/// whose behaviour must not change. A read that every validator must agree
290/// on goes through [`coin_deny_list_v1_denial`], which returns those errors.
291#[instrument(level = "trace", skip_all)]
292pub fn get_per_type_coin_deny_list_v1(
293    coin_type: &String,
294    object_store: &dyn ObjectStore,
295) -> Option<Config<ConfigWriteCap>> {
296    try_get_per_type_coin_deny_list_v1(coin_type, object_store)
297        .ok()
298        .flatten()
299}
300
301/// [`get_per_type_coin_deny_list_v1`] that returns the errors of
302/// `object_store` other than a missing or undecodable entry.
303fn try_get_per_type_coin_deny_list_v1(
304    coin_type: &String,
305    object_store: &dyn ObjectStore,
306) -> IotaResult<Option<Config<ConfigWriteCap>>> {
307    let config_key = Wrapper {
308        name: ConfigKey {
309            per_type_index: DENY_LIST_COIN_TYPE_INDEX,
310            per_type_key: coin_type.as_bytes().to_vec(),
311        },
312    };
313    // TODO: Consider caching the config object UID to avoid repeat deserialization.
314    absent_if_missing(get_dynamic_field_from_store(
315        object_store,
316        IOTA_DENY_LIST_OBJECT_ID,
317        &config_key,
318    ))
319}

321/// A dynamic-field read with a missing or undecodable field as `None`, both
322/// the same on every store holding the same objects, and every other error
323/// returned.
324fn absent_if_missing<V>(read: Result<V, IotaError>) -> IotaResult<Option<V>> {
325    match read {
326        Ok(value) => Ok(Some(value)),
327        Err(IotaError::DynamicFieldNotExists { .. } | IotaError::DynamicFieldRead(_)) => Ok(None),
328        Err(error) => Err(error),
329    }
330}

373/// Fetches the setting from a particular config.
374/// Reads the value of the setting, giving `newer_value` if the current epoch is
375/// greater than `newer_value_epoch`, and `older_value_opt` otherwise.
376/// If `cur_epoch` is `None`, the `newer_value` is always returned.
377/// Every error of `object_store` reads as `None`, deliberately, as for
378/// [`get_per_type_coin_deny_list_v1`]; [`try_read_config_setting`] returns
379/// them.
380fn read_config_setting<K, V>(
381    object_store: &dyn ObjectStore,
382    config: &Config<ConfigWriteCap>,
383    setting_name: K,
384    cur_epoch: Option<EpochId>,
385) -> Option<V>
386where
387    K: MoveTypeTagTrait + Serialize + DeserializeOwned + fmt::Debug,
388    V: Clone + Serialize + DeserializeOwned + fmt::Debug,
389{
390    try_read_config_setting(object_store, config, setting_name, cur_epoch)
391        .ok()
392        .flatten()
393}
394
395/// [`read_config_setting`] that returns the errors of `object_store` other
396/// than a missing or undecodable setting.
397fn try_read_config_setting<K, V>(
398    object_store: &dyn ObjectStore,
399    config: &Config<ConfigWriteCap>,
400    setting_name: K,
401    cur_epoch: Option<EpochId>,
402) -> IotaResult<Option<V>>
403where
404    K: MoveTypeTagTrait + Serialize + DeserializeOwned + fmt::Debug,
405    V: Clone + Serialize + DeserializeOwned + fmt::Debug,
406{
407    let setting: Option<Setting<V>> = absent_if_missing(get_dynamic_field_from_store(
408        object_store,
409        *config.id.object_id(),
410        &setting_name,
411    ))?;
412    Ok(setting.and_then(|setting| setting.read_value(cur_epoch).cloned()))
413}

415/// Returns all unique coin types in canonical string form from the input
416/// objects and receiving objects. It filters out IOTA coins since it's known
417/// that it's not a regulated coin.
418#[instrument(level = "trace", skip_all)]
419fn input_object_coin_types_for_denylist_check(
420    tx_input_objects: &InputObjects,
421    tx_receiving_objects: &ReceivingObjects,
422    per_authenticator_input_objects: &Vec<&InputObjects>,
423) -> BTreeSet<String> {
424    let all_objects =
425        tx_input_objects
426            .iter_objects()
427            .chain(tx_receiving_objects.iter_objects())
428            .chain(per_authenticator_input_objects.iter().flat_map(
429                |authenticator_input_objects| authenticator_input_objects.iter_objects(),
430            ));
431    coin_types_for_denylist_check(all_objects)
432}
433
434/// Returns all unique coin types of `objects` in canonical string form. It
435/// filters out IOTA coins since it's known that it's not a regulated coin.
436pub fn coin_types_for_denylist_check<'a>(
437    objects: impl IntoIterator<Item = &'a Object>,
438) -> BTreeSet<String> {
439    objects
440        .into_iter()
441        .filter_map(|obj| {
442            obj.as_opt_struct()
443                .and_then(|move_object| coin_type_for_denylist_check(move_object.struct_tag()))
444        })
445        .collect()
446}
447
448/// Returns the coin type of `struct_tag` in canonical string form. It returns
449/// `None` for a type that is not a coin, and for the IOTA coin since it's known
450/// that it's not a regulated coin.
451pub fn coin_type_for_denylist_check(struct_tag: &StructTag) -> Option<String> {
452    if struct_tag.is_gas_coin() {
453        None
454    } else {
455        struct_tag
456            .opt_coin_type()
457            .map(|coin_type| coin_type.to_canonical_string(false))
458    }
459}
