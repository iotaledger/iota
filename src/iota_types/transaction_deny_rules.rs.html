33/// The initial shared version of the `TransactionDenyRules` object, or `None`
34/// while the object has not been created yet (the
35/// `TransactionDenyRulesCreate` end-of-epoch transaction has not run).
36///
37/// # Panics
38///
39/// Panics if the object exists but is not shared, which would mean the
40/// system invariant on the reserved object is broken.
41pub fn get_transaction_deny_rules_obj_initial_shared_version(
42    object_store: &dyn ObjectStore,
43) -> IotaResult<Option<Version>> {
44    Ok(object_store
45        .try_get_object(&IOTA_TRANSACTION_DENY_RULES_OBJECT_ID)?
46        .map(|obj| {
47            obj.owner
48                .into_opt_shared()
49                .expect("TransactionDenyRules object must be shared")
50        }))
51}

79/// The full deny rule state read back from the `TransactionDenyRules` object,
80/// or `None` while the object has not been created yet.
81///
82/// Each deny list is reconstructed by walking its `LinkedTable` (`head` →
83/// `node.next`) with derived-id child reads, so any node can rebuild the
84/// state from its object store alone. Validators seed enforcement and the
85/// mirrored on-chain state from this at epoch start.
86pub fn get_transaction_deny_rules(
87    object_store: &dyn ObjectStore,
88) -> IotaResult<Option<DenyRuleSet>> {
89    let Some(object) = object_store.try_get_object(&IOTA_TRANSACTION_DENY_RULES_OBJECT_ID)? else {
90        return Ok(None);
91    };
92    let iota_sdk_types::ObjectData::Struct(move_object) = &object.data else {
93        return Err(IotaError::ObjectDeserialization {
94            error: "TransactionDenyRules object must be a Move object".to_string(),
95        });
96    };
97    let rules: TransactionDenyRules = bcs::from_bytes(move_object.contents()).map_err(|err| {
98        IotaError::ObjectDeserialization {
99            error: format!("failed to decode TransactionDenyRules: {err}"),
100        }
101    })?;
102    if rules.inner.version != TRANSACTION_DENY_RULES_INNER_V1 {
103        return Err(IotaError::ObjectDeserialization {
104            error: format!(
105                "unsupported TransactionDenyRules inner version {}",
106                rules.inner.version
107            ),
108        });
109    }
110    let inner: TransactionDenyRulesInnerV1 =
111        get_dynamic_field_from_store(object_store, rules.inner.id.id.bytes, &rules.inner.version)?;
112
113    Ok(Some(DenyRuleSet {
114        denied_addresses: walk_linked_table(object_store, &inner.denied_addresses)?
115            .into_iter()
116            .collect(),
117        denied_objects: walk_linked_table(object_store, &inner.denied_objects)?
118            .into_iter()
119            .map(|id| id.bytes)
120            .collect(),
121        denied_packages: walk_linked_table(object_store, &inner.denied_packages)?
122            .into_iter()
123            .map(|id| id.bytes)
124            .collect(),
125        package_publish_disabled: inner.package_publish_disabled,
126        package_upgrade_disabled: inner.package_upgrade_disabled,
127        shared_object_disabled: inner.shared_object_disabled,
128        user_transaction_disabled: inner.user_transaction_disabled,
129        receiving_objects_disabled: inner.receiving_objects_disabled,
130        move_authenticator_disabled: inner.move_authenticator_disabled,
131    }))
132}

134/// Collects a `LinkedTable`'s keys in list order by following the `next`
135/// links, one derived-id child read per entry.
136fn walk_linked_table<K>(
137    object_store: &dyn ObjectStore,
138    table: &LinkedTable<K>,
139) -> IotaResult<Vec<K>>
140where
141    K: MoveTypeTagTrait + Serialize + DeserializeOwned + Clone + fmt::Debug,
142{
143    let mut keys = Vec::with_capacity(table.size as usize);
144    let mut next = table.head.clone();
145    while let Some(key) = next {
146        // A cycle in the links would otherwise never terminate; the walk can
147        // only visit `size` distinct entries.
148        if keys.len() as u64 == table.size {
149            return Err(IotaError::ObjectDeserialization {
150                error: format!(
151                    "LinkedTable {} has more linked entries than its size {}",
152                    table.id, table.size
153                ),
154            });
155        }
156        let node: LinkedTableNode<K, bool> =
157            get_dynamic_field_from_store(object_store, table.id, &key)?;
158        keys.push(key);
159        next = node.next;
160    }
161    if keys.len() as u64 != table.size {
162        return Err(IotaError::ObjectDeserialization {
163            error: format!(
164                "LinkedTable {} links {} entries but its size is {}",
165                table.id,
166                keys.len(),
167                table.size
168            ),
169        });
170    }
171    Ok(keys)
172}
