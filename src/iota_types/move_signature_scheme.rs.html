4use eyre::ensure;
5use iota_sdk_types::{Address, Identifier, SignatureScheme, StructTag};
6use serde::{Deserialize, Serialize};
7
8pub const SIGNATURE_SCHEME_MODULE_NAME: Identifier = Identifier::from_static("signature_scheme");
9pub const SIGNATURE_SCHEME_STRUCT_NAME: Identifier = Identifier::from_static("SignatureScheme");
10
11/// Rust mirror of the Move `signature_scheme::SignatureScheme` struct.
12#[derive(Debug, Serialize, Deserialize, Clone, Copy, Eq, PartialEq)]
13pub struct MoveSignatureScheme {
14    /// The raw flag byte that identifies the scheme.
15    flag: u8,
16}
17
18impl MoveSignatureScheme {
19    pub fn tag() -> StructTag {
20        StructTag::new(
21            Address::FRAMEWORK,
22            SIGNATURE_SCHEME_MODULE_NAME,
23            SIGNATURE_SCHEME_STRUCT_NAME,
24            Vec::new(),
25        )
26    }
27
28    /// Returns the raw flag byte that identifies this scheme.
29    pub fn flag(&self) -> u8 {
30        self.flag
31    }
32}
33
34impl TryFrom<SignatureScheme> for MoveSignatureScheme {
35    type Error = eyre::Report;
36
37    /// Converts a `SignatureScheme` to a `MoveSignatureScheme`.
38    ///
39    /// Returns an error for schemes that are not valid for account public keys:
40    /// BLS12381 and MoveAuthenticator.
41    fn try_from(scheme: SignatureScheme) -> Result<Self, eyre::Report> {
42        ensure!(
43            matches!(
44                scheme,
45                SignatureScheme::Ed25519
46                    | SignatureScheme::Secp256k1
47                    | SignatureScheme::Secp256r1
48                    | SignatureScheme::Multisig
49                    | SignatureScheme::PasskeyAuthenticator
50            ),
51            "Unsupported signature scheme for account public key: {scheme:?}"
52        );
53
54        Ok(Self {
55            flag: scheme.to_u8(),
56        })
57    }
58}
59
60impl From<MoveSignatureScheme> for SignatureScheme {
61    fn from(move_scheme: MoveSignatureScheme) -> Self {
62        SignatureScheme::from_byte(move_scheme.flag).expect("invariant: scheme flag valid")
63    }
64}
