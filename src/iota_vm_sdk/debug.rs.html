4//! Debug capture: [`DebugConfig`] toggles the Move VM gas profiler and
5//! instruction tracing independently; a run returns the matching
6//! [`DebugArtifacts`].
7
8use std::path::PathBuf;
9
10use move_trace_format::format::{MoveTraceBuilder, MoveTraceReader, TraceEvent, TraceVersion};
11
12use crate::error::TraceError;
13
14/// Configuration for a single debug-enabled run.
15///
16/// The [`Default`] disables all capture.
17#[derive(Debug, Default, Clone)]
18#[non_exhaustive]
19pub struct DebugConfig {
20    /// Enable the Move VM gas profiler and choose where the Speedscope JSON
21    /// ends up.
22    pub profile: Option<ProfileSink>,
23    /// Enable instruction-level execution tracing. Only captured on the
24    /// `MoveAuthenticator` path; see [`with_tracing`](Self::with_tracing).
25    pub trace: bool,
26}
27
28impl DebugConfig {
29    /// Enable the gas profiler, writing to `sink`.
30    ///
31    /// Requires the crate's `tracing` feature; without it the run captures no
32    /// profile and [`DebugArtifacts::profile`] stays `None`.
33    #[must_use]
34    pub fn with_profiling(mut self, sink: ProfileSink) -> Self {
35        self.profile = Some(sink);
36        self
37    }
38
39    /// Enable instruction-level execution tracing.
40    ///
41    /// Requires the crate's `tracing` feature; without it
42    /// [`DebugArtifacts::trace`] stays `None`.
43    ///
44    /// Tracing is only captured for signed transactions that authorize via a
45    /// `MoveAuthenticator`. The unsigned
46    /// [`LocalVm::execute`](crate::LocalVm::execute) path and
47    /// standard-signature transactions run through the engine's dev-inspect
48    /// entry point, which accepts no trace builder; for those,
49    /// [`DebugArtifacts::trace`] stays `None` even when tracing was
50    /// requested.
51    #[must_use]
52    pub fn with_tracing(mut self) -> Self {
53        self.trace = true;
54        self
55    }
56
57    /// `true` if any debug capture is enabled.
58    pub(crate) fn any_enabled(&self) -> bool {
59        self.profile.is_some() || self.trace
60    }
61}
62
63/// Where to write the Speedscope-format gas profile.
64#[derive(Debug, Clone)]
65#[non_exhaustive]
66pub enum ProfileSink {
67    /// Write the merged profile JSON to the given path on disk. A path that
68    /// can't be written fails the run with a
69    /// [`VmSdkError`](crate::VmSdkError).
70    Path(PathBuf),
71    /// Write the profile to a temporary location and read its bytes back into
72    /// [`ProfileOutput::Json`] after execution. (Native only — on wasm32 the
73    /// filesystem is unavailable and this behaves as a no-op.)
74    Capture,
75}
76
77/// The resulting gas profile: either a filesystem path the VM wrote to or the
78/// raw JSON bytes read back after execution.
79#[derive(Debug)]
80#[non_exhaustive]
81pub enum ProfileOutput {
82    /// Speedscope JSON written to this path (from [`ProfileSink::Path`]).
83    Path(PathBuf),
84    /// Merged Speedscope JSON bytes read back (from [`ProfileSink::Capture`]).
85    Json(Vec<u8>),
86}

88/// An instruction-level execution trace, in the format the Move trace debugger
89/// reads. Decode its events with [`events`](Self::events), or hand the encoded
90/// bytes to a file or another tool with [`bytes`](Self::bytes).
91pub struct ExecutionTrace {
92    version: TraceVersion,
93    event_count: usize,
94    bytes: Vec<u8>,
95}
96
97impl ExecutionTrace {
98    /// Build from a finished [`MoveTraceBuilder`]: reads the version and event
99    /// count while the builder is still alive (`into_trace` moves the trace
100    /// out, and encoding consumes it), then encodes.
101    pub(crate) fn from_builder(builder: MoveTraceBuilder) -> Self {
102        let version = builder.trace.version;
103        let event_count = builder.current_trace_offset();
104        Self {
105            version,
106            event_count,
107            bytes: builder.into_trace().into_compressed_json_bytes(),
108        }
109    }
110
111    /// Version of the trace format the run was captured with.
112    pub fn version(&self) -> TraceVersion {
113        self.version
114    }
115
116    /// How many events the VM emitted. Counted during the run, so reading it
117    /// decodes nothing.
118    pub fn event_count(&self) -> usize {
119        self.event_count
120    }
121
122    /// The trace as the VM encoded it: a version header line followed by one
123    /// JSON-encoded event per line, zstd-compressed. Write it to a file with
124    /// the `json.zst` extension to open the run in the Move trace debugger.
125    pub fn bytes(&self) -> &[u8] {
126        &self.bytes
127    }
128
129    /// Take the encoded trace, as described on [`bytes`](Self::bytes).
130    pub fn into_bytes(self) -> Vec<u8> {
131        self.bytes
132    }
133}
134
135impl ExecutionTrace {
136    /// The events the VM emitted, in order, decoded from the encoded trace.
137    ///
138    /// Each call decodes the trace again, so collect the events if you need
139    /// them more than once. Decoding a large trace costs many times what the
140    /// encoded bytes do (a run of ~15,000 events decodes from 66 KB of
141    /// compressed bytes to about 4 MB), which is why they are not decoded up
142    /// front.
143    ///
144    /// # Errors
145    ///
146    /// Returns [`TraceError`] if the encoded trace cannot be opened; the
147    /// returned iterator yields one per event that cannot be decoded.
148    pub fn events(&self) -> Result<TraceEvents<'_>, TraceError> {
149        let reader = MoveTraceReader::new(std::io::Cursor::new(self.bytes.as_slice()))
150            .map_err(|source| TraceError { source })?;
151        Ok(TraceEvents(reader))
152    }
153}
154
155/// The events of an [`ExecutionTrace`], decoded as the iterator advances. See
156/// [`ExecutionTrace::events`].
157pub struct TraceEvents<'a>(MoveTraceReader<'static, std::io::Cursor<&'a [u8]>>);
158
159impl Iterator for TraceEvents<'_> {
160    type Item = Result<TraceEvent, TraceError>;
161
162    fn next(&mut self) -> Option<Self::Item> {
163        match self.0.next_event() {
164            Ok(Some(event)) => Some(Ok(event)),
165            Ok(None) => None,
166            Err(source) => Some(Err(TraceError { source })),
167        }
168    }
169}
170
171// Summarised rather than derived: a derived impl would dump `bytes` as a flat
172// list of numbers, which is useless and can run to hundreds of kilobytes for a
173// real transaction. Reach for `events` to see the decoded events.
174impl std::fmt::Debug for ExecutionTrace {
175    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
176        f.debug_struct("ExecutionTrace")
177            .field("version", &self.version)
178            .field("event_count", &self.event_count)
179            .field("bytes_len", &self.bytes.len())
180            .finish()
181    }
182}

184/// Artifacts captured from a run, present when any [`DebugConfig`] toggle was
185/// enabled.
186#[derive(Debug, Default)]
187#[non_exhaustive]
188pub struct DebugArtifacts {
189    /// Gas profile output, if [`DebugConfig::profile`] was set.
190    pub profile: Option<ProfileOutput>,
191    /// Instruction-level execution trace. `None` unless the run went through
192    /// the `MoveAuthenticator` path (see [`DebugConfig::with_tracing`]).
193    pub trace: Option<ExecutionTrace>,
194}
