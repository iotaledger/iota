17/// A [`Store`] backed by a remote node over gRPC, resolving objects on demand.
18///
19/// Clones share the same cache and client. Objects resolve at their latest
20/// version; see [`Store::get_child_object`] for the dynamic-field version-bound
21/// caveat.
22///
23/// On-demand object resolution (via the synchronous [`Store`] impl) requires a
24/// multi-threaded Tokio runtime (e.g. `#[tokio::main]`); outside one, a cache
25/// miss fails with a [`StoreError`] instead of fetching.
26#[derive(Clone)]
27pub struct GrpcStore {
28    cache: CachingStore<GrpcFetcher>,
29}
30
31impl GrpcStore {
32    /// Wrap an existing client. The store starts empty and fetches every
33    /// object a run needs, framework packages included, from the node.
34    pub fn new(client: GrpcClient) -> Self {
35        Self {
36            cache: CachingStore::new(GrpcFetcher { client }),
37        }
38    }
39
40    /// Connect to a gRPC endpoint (by URL) and create an empty store.
41    ///
42    /// # Errors
43    ///
44    /// Returns [`VmSdkError::Store`] if the client cannot be created for `url`.
45    pub fn connect(url: impl Into<String>) -> Result<Self, VmSdkError> {
46        let url: String = url.into();
47        let client = GrpcClient::new(url).map_err(|e| StoreError::new("connect gRPC", e))?;
48        Ok(Self::new(client))
49    }
50
51    /// A snapshot clone of the objects fetched or inserted so far.
52    pub fn store(&self) -> InMemoryStore {
53        self.cache.store()
54    }
55
56    /// Fetch the chain parameters a [`LocalVm`](crate::LocalVm) needs.
57    ///
58    /// The [`Chain`](crate::Chain) is resolved from the node's service info so
59    /// chain-gated protocol features match the real chain; an unrecognised
60    /// chain identifier maps to [`Chain::Unknown`](crate::Chain).
61    ///
62    /// # Errors
63    ///
64    /// Returns [`VmSdkError::Store`] if the epoch or service-info RPC fails or
65    /// can't be decoded.
66    pub async fn fetch_chain_context(&self) -> Result<ChainContext, VmSdkError> {
67        let client = &self.cache.fetcher().client;
68        let (epoch_response, service_info_response) =
69            tokio::join!(client.epoch(), client.service_info());
70        let epoch = epoch_response
71            .map_err(|e| StoreError::new("fetch epoch", e))?
72            .into_inner();
73        let epoch_id = epoch
74            .epoch_id()
75            .map_err(|e| StoreError::new("epoch id", e))?;
76        let reference_gas_price = epoch
77            .gas_price()
78            .map_err(|e| StoreError::new("gas price", e))?;
79        let epoch_timestamp_ms = epoch
80            .start_ms()
81            .map_err(|e| StoreError::new("epoch start", e))?;
82        let protocol_version = epoch
83            .protocol_config()
84            .and_then(|pc| pc.version())
85            .map_err(|e| StoreError::new("protocol version", e))?;
86        let chain = service_info_response
87            .map_err(|e| StoreError::new("fetch service info", e))?
88            .body()
89            .chain_id
90            .as_ref()
91            .and_then(|d| Digest::try_from(d).ok())
92            .map(|digest| ChainIdentifier::from(CheckpointDigest::from(digest)).chain())
93            .unwrap_or(iota_protocol_config::Chain::Unknown);
94        Ok(ChainContext {
95            protocol_version: iota_protocol_config::ProtocolVersion::new(protocol_version),
96            reference_gas_price,
97            epoch_id,
98            epoch_timestamp_ms,
99            chain,
100        })
101    }
102}

104impl Store for GrpcStore {
105    fn get_object(
106        &self,
107        id: &ObjectId,
108        version: Option<Version>,
109    ) -> Result<Option<Object>, StoreError> {
110        self.cache.get_object(id, version)
111    }
112
113    fn get_child_object(
114        &self,
115        parent: &ObjectId,
116        child: &ObjectId,
117        version_upper_bound: Version,
118    ) -> Result<Option<Object>, StoreError> {
119        self.cache
120            .get_child_object(parent, child, version_upper_bound)
121    }
122
123    fn insert(&mut self, object: Object) {
124        self.cache.insert(object);
125    }
126
127    fn remove(&mut self, id: &ObjectId) {
128        self.cache.remove(id);
129    }
130}

132/// gRPC transport for [`CachingStore`].
133#[derive(Clone)]
134struct GrpcFetcher {
135    client: GrpcClient,
136}
137
138impl ObjectFetcher for GrpcFetcher {
139    async fn fetch_objects(
140        &self,
141        refs: &[(ObjectId, Option<Version>)],
142    ) -> Result<Vec<Object>, StoreError> {
143        let results = self
144            .client
145            .objects_with_versions(refs.iter().copied())
146            .await
147            .map_err(|e| StoreError::new("fetch objects via gRPC", e))?
148            .into_inner();
149
150        // The proto helper yields the SDK `Object`, which the node's
151        // `iota_types::object::Object` is a newtype over.
152        skip_not_found(results)?
153            .into_iter()
154            .map(|proto_obj| {
155                proto_obj
156                    .object()
157                    .map(Into::into)
158                    .map_err(|e| StoreError::new("decode gRPC object", e))
159            })
160            .collect()
161    }
162}

164/// Keep the items the node returned, dropping the ones it reported as
165/// `NOT_FOUND`.
166///
167/// The `Store` contract treats absence as `Ok(None)` — the VM's child-object
168/// resolver relies on this, since a dynamic field that does not exist must read
169/// as absent rather than fault. The batched read reports a missing object per
170/// requested ref, so the refs the node could serve survive a missing one.
171fn skip_not_found<T>(
172    results: Vec<Result<T, iota_grpc_client::GrpcError>>,
173) -> Result<Vec<T>, StoreError> {
174    let mut items = Vec::with_capacity(results.len());
175    for result in results {
176        match result {
177            Ok(item) => items.push(item),
178            Err(e) if e.is_not_found() => {}
179            Err(e) => return Err(StoreError::new("fetch objects via gRPC", e)),
180        }
181    }
182    Ok(items)
183}

185#[cfg(test)]
186mod tests {
187    use iota_grpc_client::{GrpcError, RpcStatus};
188
189    use super::skip_not_found;
190
191    fn server_error(code: tonic::Code) -> GrpcError {
192        let mut status = RpcStatus::default();
193        status.code = code.into();
194        GrpcError::Server(status)
195    }
196
197    #[test]
198    fn a_missing_ref_is_skipped_without_dropping_the_rest() {
199        let results = vec![Ok(1), Err(server_error(tonic::Code::NotFound)), Ok(3)];
200
201        assert_eq!(skip_not_found(results).unwrap(), vec![1, 3]);
202    }
203
204    #[test]
205    fn every_ref_missing_yields_no_objects() {
206        let results: Vec<Result<u32, GrpcError>> = vec![
207            Err(server_error(tonic::Code::NotFound)),
208            Err(server_error(tonic::Code::NotFound)),
209        ];
210
211        assert!(skip_not_found(results).unwrap().is_empty());
212    }
213
214    #[test]
215    fn an_error_other_than_not_found_fails_the_fetch() {
216        let results: Vec<Result<u32, GrpcError>> =
217            vec![Ok(1), Err(server_error(tonic::Code::Internal))];
218
219        assert!(skip_not_found(results).is_err());
220    }
