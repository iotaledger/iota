43/// The local Move VM executor. Owns a [`Store`] and the execution engine for a
44/// single [`ChainContext`].
45pub struct LocalVm {
46    pub(super) protocol_config: ProtocolConfig,
47    pub(super) reference_gas_price: u64,
48    pub(super) epoch_id: u64,
49    pub(super) epoch_timestamp_ms: u64,
50    pub(super) limits_metrics: Arc<LimitsMetrics>,
51    pub(super) bytecode_verifier_metrics: Arc<BytecodeVerifierMetrics>,
52    store: Box<dyn Store>,
53    /// Profiler-free executor shared by `decode_events`'s layout resolver and
54    /// all non-profiled `execute*` runs, built lazily on first use. It depends
55    /// only on the immutable `protocol_config`, so one instance is reused
56    /// across calls; a profiled run builds its own executor via
57    /// [`ExecutionEnv`].
58    cached_executor: OnceLock<Arc<dyn Executor + Send + Sync>>,
59}

77    ///
78    /// # Errors
79    ///
80    /// Returns [`VmSdkError::UnsupportedProtocolVersion`] when this build does

82    /// fetched from a node running a newer protocol).
83    pub fn new(ctx: ChainContext, store: impl Store + 'static) -> Result<Self, VmSdkError> {
84        let protocol_config =
85            ProtocolConfig::get_for_version_if_supported(ctx.protocol_version, ctx.chain).ok_or(

87                    version: ctx.protocol_version,
88                    feature: None,

95            epoch_timestamp_ms: ctx.epoch_timestamp_ms,
96            limits_metrics: Arc::new(new_limits_metrics()),
97            bytecode_verifier_metrics: Arc::new(new_bytecode_verifier_metrics()),
98            store: Box::new(store),
99            cached_executor: OnceLock::new(),
100        })
101    }

121    /// Run an unsigned transaction.
122    ///
123    /// No signatures are checked: the result reports
124    /// [`SignatureStatus::NotChecked`], and with [`ExecutionMode::Execute`] the
125    /// effects are committed to the store regardless of whether the transaction
126    /// would be authorized on-chain. Use [`LocalVm::execute_signed`] when
127    /// signature verification is required.
128    ///
129    /// # Errors
130    ///
131    /// Returns a [`VmSdkError`] on preparation or VM faults; a Move-level abort
132    /// is reported via [`ExecutionResult::status`], not as an error.
133    pub fn execute(
134        &mut self,
135        tx: Transaction,
136        opts: ExecuteOptions,
137    ) -> Result<ExecutionResult, VmSdkError> {
138        let env = ExecutionEnv::new(self, &opts.debug)?;
139        let prepared = {
140            let backend = StoreBackend::new(self.store.as_ref());
141            prepare_transaction(
142                &env,
143                &backend,
144                tx,
145                opts.mode,
146                &opts.deny_config,
147                &[],
148                &[],
149                opts.check_coin_deny_list,
150            )?
151        };
152        let (sim, transaction) = {
153            let backend = StoreBackend::new(self.store.as_ref());
154            execute_prepared(&env, &backend, prepared, opts.mode)?
155        };
156        // The dev-inspect entry point accepts no `MoveTraceBuilder`, so this path
157        // never captures a trace; pass `None`. See `DebugConfig::with_tracing`.
158        let artifacts = env.collect_artifacts(None)?;
159        self.finish(
160            sim,
161            transaction,
162            opts.mode,
163            SignatureStatus::NotChecked,
164            artifacts,
165        )
166    }

168    /// Run a signed transaction, verifying signatures first.
169    ///
170    /// Standard schemes are verified cryptographically first. Every
171    /// [`MoveAuthenticator`](iota_sdk_types::MoveAuthenticator)
172    /// — the sender's and, for a sponsored tx, the sponsor's — is verified by
173    /// running its function in the VM. On failure the authenticators are re-run
174    /// alone to tell a rejection from a body abort.
175    ///
176    /// `opts.mode` governs input-check relaxation
177    /// ([`ExecutionMode::DevInspect`]) and commit
178    /// ([`ExecutionMode::Execute`]) as for [`execute`](Self::execute),
179    /// but the authenticators and transaction body always execute under full
180    /// (non-dev-inspect) VM semantics.
181    ///
182    /// Signatures are verified against the transaction as supplied. Gas and
183    /// owned-input references are then resolved against the store's versions,
184    /// so the executed bytes (and digest) can differ from the signed bytes when
185    /// the store holds other versions; a Move authenticator reading the
186    /// transaction bytes from its auth context observes the resolved form.
187    ///
188    /// # Errors
189    ///
190    /// Returns [`VmSdkError::SignatureVerification`] for an invalid
191    /// standard-scheme signature, or another [`VmSdkError`] on preparation/VM
192    /// faults. A rejected `MoveAuthenticator` is reported via
193    /// [`ExecutionResult::signature_status`], not as an error.
194    pub fn execute_signed(
195        &mut self,
196        signed: SenderSignedTransaction,
197        opts: ExecuteOptions,
198    ) -> Result<ExecutionResult, VmSdkError> {
199        let env = ExecutionEnv::new(self, &opts.debug)?;
200        self.verify_standard_signatures(&signed)?;
201
202        let move_authenticators: Vec<_> =
203            signed.move_authenticators().into_iter().cloned().collect();
204        // The deny checks inspect the signatures (e.g. `move_authenticator_disabled`,
205        // deprecated zkLogin), so they must survive `signed` being consumed.
206        let tx_signatures = signed.signatures().to_vec();
207        // The auth digests must be computed from the signed data before it is
208        // consumed; the `MoveAuthenticator` execution path needs them in its
209        // `AuthContextData`.
210        let auth_digests = signed
211            .compute_auth_digests()
212            .map_err(|e| VmSdkError::SignatureVerification(e.into()))?;
213        let transaction = signed.into_signed_transaction().transaction;
214
215        // A `MoveAuthenticator` on a protocol version that predates Move
216        // authentication cannot be run; reject it with a typed error rather
217        // than reaching the engine.
218        ensure_move_authentication_supported(
219            &self.protocol_config,
220            !move_authenticators.is_empty(),
221        )?;
222
223        let prepared = {
224            let backend = StoreBackend::new(self.store.as_ref());
225            prepare_transaction(
226                &env,
227                &backend,
228                transaction,
229                opts.mode,
230                &opts.deny_config,
231                &tx_signatures,
232                &move_authenticators,
233                opts.check_coin_deny_list,
234            )?
235        };
236        let (sim, executed, signature_status, trace) = {
237            let backend = StoreBackend::new(self.store.as_ref());
238            if move_authenticators.is_empty() {
239                // Standard schemes were verified cryptographically above; the
240                // run's outcome cannot retroactively invalidate them. Runs
241                // through the dev-inspect entry point, so no trace is captured.
242                let (sim, executed) = execute_prepared(&env, &backend, prepared, opts.mode)?;
243                (sim, executed, SignatureStatus::Verified, None)
244            } else {
245                // Only the authenticator path threads a `MoveTraceBuilder`
246                // through the engine, so a trace is built only here.
247                let mut trace_builder = env.trace_enabled().then(MoveTraceBuilder::new);
248
249                let (sim, authenticator_outcome, executed) = execute_with_move_authenticators(
250                    &env,
251                    &backend,
252                    prepared,
253                    move_authenticators,
254                    auth_digests,
255                    opts.check_coin_deny_list,
256                    &mut trace_builder,
257                )?;
258                (
259                    sim,
260                    executed,
261                    SignatureStatus::from_authentication(authenticator_outcome),
262                    trace_builder.map(ExecutionTrace::from_builder),
263                )
264            }
265        };
266        let artifacts = env.collect_artifacts(trace)?;
267
268        self.finish(sim, executed, opts.mode, signature_status, artifacts)
269    }

271    /// Check whether the transaction's `MoveAuthenticator`(s) would be admitted
272    /// at signing: runs the pre-consensus authenticator set alone — all
273    /// authenticators, or only the sponsor's for a sponsored transaction when
274    /// the protocol enables `pre_consensus_sponsor_only_move_authentication` —
275    /// via
276    /// [`authenticate_transaction`](iota_execution::Executor::authenticate_transaction),
277    /// metered at the signing gas cap (`max_auth_gas`); commits nothing.
278    ///
279    /// Complements [`execute_signed`](Self::execute_signed), which models the
280    /// post-consensus path and is never capped at `max_auth_gas`; a transaction
281    /// is accepted on-chain only if it passes both. See
282    /// `docs/execution-model.md` for the full phase/budget mapping.
283    ///
284    /// Standard-scheme signatures are verified cryptographically first, as at
285    /// signing; a transaction with no `MoveAuthenticator` is admitted once they
286    /// verify. Deny-list and input policies are not checked.
287    ///
288    /// # Errors
289    ///
290    /// Returns [`VmSdkError::SignatureVerification`] for an invalid
291    /// standard-scheme signature, [`VmSdkError::UnsupportedProtocolVersion`]
292    /// when the protocol version predates Move authentication, or another
293    /// [`VmSdkError`] on a VM fault. An authenticator that rejects or exceeds
294    /// `max_auth_gas` is reported via the returned
295    /// [`SignatureStatus::Failed`], not as an error.
296    pub fn check_signing_authentication(
297        &self,
298        signed: SenderSignedTransaction,
299    ) -> Result<SignatureStatus, VmSdkError> {
300        self.verify_standard_signatures(&signed)?;
301
302        let move_authenticators: Vec<_> =
303            signed.move_authenticators().into_iter().cloned().collect();
304        // Without a `MoveAuthenticator` nothing is gas-capped at signing; the
305        // standard-scheme signatures verified above are all a validator checks.
306        if move_authenticators.is_empty() {
307            return Ok(SignatureStatus::Verified);
308        }
309        ensure_move_authentication_supported(&self.protocol_config, true)?;
310
311        // The subset the node runs before consensus. Each authenticator
312        // authorizes a distinct address, so the subset is identified by address.
313        let pre_consensus_addresses =
314            pre_consensus_authenticator_addresses(&signed, &self.protocol_config);
315        let auth_digests = signed
316            .compute_auth_digests()
317            .map_err(|e| VmSdkError::SignatureVerification(e.into()))?;
318        let transaction = signed.into_signed_transaction().transaction;
319
320        let env = ExecutionEnv::new(self, &DebugConfig::default())?;
321        let backend = StoreBackend::new(self.store.as_ref());
322
323        // Resolve all authenticators so the auth context carries both the
324        // sender's and the sponsor's function refs, matching the node, then run
325        // only the pre-consensus subset.
326        let prepared_auths = prepare_authenticators(&backend, move_authenticators)?;
327        let auth_context_data =
328            build_auth_context_data(&transaction, &prepared_auths, auth_digests)?;
329        let to_run: Vec<_> = prepared_auths
330            .into_iter()
331            .filter(|(a, _, _)| pre_consensus_addresses.contains(&a.address()))
332            .collect();
333
334        let gas_status = IotaGasStatus::new(
335            self.protocol_config.max_auth_gas(),
336            transaction.gas_price(),
337            self.reference_gas_price,
338            &self.protocol_config,
339        )
340        .map_err(|e| ValidationError::new("signing gas status", e))?;
341
342        let authenticator_outcome = authenticate_only(
343            &env,
344            &backend,
345            &transaction,
346            &to_run,
347            gas_status,
348            auth_context_data,
349        )?;
350        Ok(SignatureStatus::from_authentication(authenticator_outcome))
351    }

353    /// The profiler-free executor shared by `decode_events` and non-profiled
354    /// runs, built on first use and cached for the lifetime of this `LocalVm`.
355    pub(super) fn cached_executor(&self) -> Result<&Arc<dyn Executor + Send + Sync>, VmSdkError> {
356        if let Some(executor) = self.cached_executor.get() {
357            return Ok(executor);
358        }
359        // `OnceLock::get_or_try_init` is unstable, so build first and let the
360        // first writer win; a redundant build from a racing caller is dropped.
361        let executor = build_executor(&self.protocol_config)?;
362        Ok(self.cached_executor.get_or_init(|| executor))
363    }

365    /// Decode a [`TransactionEvents`] payload into fully-annotated
366    /// [`DecodedEvent`]s, in event order, using this VM's type-layout resolver
367    /// and the store.
368    ///
369    /// # Errors
370    ///
371    /// Returns [`VmSdkError`] when the layout resolver cannot be built or an
372    /// event fails to decode (the error names the event's type). To decode
373    /// events individually, use [`LocalVm::decode_value`] on an event's
374    /// contents and type.
375    pub fn decode_events(
376        &self,
377        events: &TransactionEvents,
378    ) -> Result<Vec<DecodedEvent>, VmSdkError> {
379        let executor = self.cached_executor()?;
380        let backend = StoreBackend::new(self.store.as_ref());
381        let mut resolver = executor.type_layout_resolver(Box::new(&backend));
382        events
383            .0
384            .iter()
385            .map(|ev| decode_one_event(ev, resolver.as_mut()))
386            .collect()
387    }

389    /// Decode a single BCS-encoded value of the given
390    /// [`TypeTag`](iota_sdk_types::TypeTag) into an annotated
391    /// [`MoveValue`](move_core_types::annotated_value::MoveValue), resolving
392    /// any struct layouts from the packages in the store. Turns raw
393    /// `(bytes, type)` pairs — dev-inspect return values and mutable
394    /// reference outputs — into readable values.
395    ///
396    /// # Errors
397    ///
398    /// Returns [`VmSdkError::Execution`] if the layout can't be resolved or
399    /// `bytes` don't deserialize against it.
400    pub fn decode_value(
401        &self,
402        bytes: &[u8],
403        type_tag: &iota_sdk_types::TypeTag,
404    ) -> Result<move_core_types::annotated_value::MoveValue, VmSdkError> {
405        let core_tag = iota_types::iota_sdk_types_conversions::type_tag_sdk_to_core(type_tag);
406        let layout = TypeLayoutBuilder::build_with_types(&core_tag, self)
407            .map_err(|e| ExecutionError::new(format!("build layout for {type_tag}: {e}")))?;
408        // `BoundedVisitor` bounds the deserialized value's depth and
409        // allocation, like every node-side decoder of externally-sourced
410        // bytes.
411        iota_types::object::bounded_visitor::BoundedVisitor::deserialize_value(bytes, &layout)
412            .map_err(|e| {
413                ExecutionError::new(format!("decode value of type {type_tag}: {e}")).into()
414            })
415    }

417    /// Verify the standard-scheme signatures on `signed`.
418    fn verify_standard_signatures(
419        &self,
420        signed: &SenderSignedTransaction,
421    ) -> Result<(), VmSdkError> {
422        // Match the node's verifier, which derives these from the protocol
423        // config (see `AuthorityPerEpochStore`); `VerifyParams::default()` would
424        // hardcode both off and diverge for passkey-in-multisig / additional
425        // multisig checks.
426        let verify_params = VerifyParams::new(
427            self.protocol_config.accept_passkey_in_multisig(),
428            self.protocol_config.additional_multisig_checks(),
429        );
430        verify_sender_signed_data_message_signatures(signed, &verify_params)
431            .map_err(VmSdkError::SignatureVerification)
432    }

434    /// Assemble an [`ExecutionResult`] from a raw simulation, committing the
435    /// effects to the store when the mode is [`ExecutionMode::Execute`] and the
436    /// run succeeded.
437    fn finish(
438        &mut self,
439        sim: SimulateTransactionResult,
440        transaction: Transaction,
441        mode: ExecutionMode,
442        signature_status: SignatureStatus,
443        artifacts: Option<DebugArtifacts>,
444    ) -> Result<ExecutionResult, VmSdkError> {
445        let gas_summary = sim.effects.gas_cost_summary().clone();
446        let status = sim.effects.status().clone();
447
448        let committed = matches!(mode, ExecutionMode::Execute) && status.is_success();
449        if committed {
450            self.apply_effects(&sim);
451        }
452
453        // Without a mutable shared input there is no congestion to price in, so
454        // a node suggests exactly the reference gas price; with one, the
455        // suggestion depends on congestion only a node observes.
456        let suggested_gas_price = (!transaction
457            .shared_input_objects()
458            .iter()
459            .any(|object| object.mutable))
460        .then_some(self.reference_gas_price);
461
462        let (command_results, execution_error) = match sim.execution_result {
463            Ok(results) => (results, None),
464            Err(error) => (Vec::new(), Some(error)),
465        };
466
467        Ok(ExecutionResult {
468            transaction,
469            effects: sim.effects,
470            events: sim.events,
471            command_results,
472            execution_error,
473            input_objects: sim.input_objects.into_values().collect(),
474            output_objects: sim.output_objects.into_values().collect(),
475            gas_summary,
476            suggested_gas_price,
477            mock_gas_id: sim.mock_gas_id,
478            status,
479            signature_status,
480            committed,
481            debug: artifacts,
482        })
483    }
484
485    /// Apply created/mutated/deleted/wrapped changes back into the store so a
486    /// subsequent run sees them.
487    fn apply_effects(&mut self, sim: &SimulateTransactionResult) {
488        // `output_objects` is authoritative for what survives; then drop what
489        // was deleted or wrapped. `unwrapped_then_deleted` objects were nested,
490        // never standalone store entries, so they need no removal.
491        for obj in sim.output_objects.values() {
492            self.store.insert(obj.clone());
493        }
494        for objref in sim.effects.deleted() {
495            self.store.remove(&objref.object_id);
496        }
497        for objref in sim.effects.wrapped() {
498            self.store.remove(&objref.object_id);
499        }
500        // The one-shot mock gas coin is never persisted.
501        if let Some(id) = sim.mock_gas_id {
502            self.store.remove(&id);
503        }
504    }
505}

507/// Addresses whose `MoveAuthenticator` the node runs before consensus: all
508/// authenticators, or only the sponsor's for a sponsored transaction when the
509/// protocol enables `pre_consensus_sponsor_only_move_authentication`. Mirrors
510/// the node's `pre_consensus_move_authenticators`.
511fn pre_consensus_authenticator_addresses(
512    signed: &SenderSignedTransaction,
513    protocol_config: &ProtocolConfig,
514) -> Vec<Address> {
515    let selected: Vec<&MoveAuthenticator> = if protocol_config
516        .pre_consensus_sponsor_only_move_authentication()
517        && signed.transaction().is_sponsored_tx()
518    {
519        signed.sponsor_move_authenticator().into_iter().collect()
520    } else {
521        signed.move_authenticators()
522    };
523    selected.iter().map(|a| a.address()).collect()
524}

526/// Check that the protocol version supports Move authentication when the
527/// transaction carries `MoveAuthenticator`s.
528///
529/// Support is signalled by `max_auth_gas` being set; it is unset on versions
530/// predating Move authentication, where the panicking getter would crash, so a
531/// typed [`VmSdkError::UnsupportedProtocolVersion`] is returned instead. When
532/// no authenticators are present nothing is read, so any version is accepted.
533fn ensure_move_authentication_supported(
534    protocol_config: &ProtocolConfig,
535    has_authenticators: bool,
536) -> Result<(), VmSdkError> {
537    if has_authenticators && protocol_config.max_auth_gas_as_option().is_none() {
538        return Err(VmSdkError::UnsupportedProtocolVersion {
539            version: protocol_config.version,
540            feature: Some("MoveAuthenticator signatures"),
541        });
542    }
543    Ok(())
544}

555        // No authenticators: `max_auth_gas` is never read, so even a version

558        assert!(ensure_move_authentication_supported(&old, false).is_ok());

561    #[test]
562    fn move_authentication_support_errors_before_move_authentication() {
563        // Protocol v1 predates Move authentication, so `max_auth_gas` is unset:
564        // an authenticator transaction must surface a typed error, not panic.
565        let old = ProtocolConfig::get_for_version(ProtocolVersion::new(1), Chain::Unknown);
566        assert!(old.max_auth_gas_as_option().is_none());
567        assert!(matches!(
568            ensure_move_authentication_supported(&old, true),
569            Err(VmSdkError::UnsupportedProtocolVersion {
570                feature: Some(_),
571                ..
572            })
573        ));
574    }
