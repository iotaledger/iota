14/// The chain parameters a [`LocalVm`](super::LocalVm) needs.
15///
16/// Usually obtained from a node via `GrpcStore::fetch_chain_context` or
17/// `GraphQLStore::fetch_chain_context`; construct it manually only for
18/// offline runs.
19#[derive(Debug, Clone)]
20#[non_exhaustive]
21pub struct ChainContext {
22    /// Protocol version the run should use; selects the protocol config.
23    pub protocol_version: ProtocolVersion,
24    /// Reference gas price for the epoch, in NANOS.
25    pub reference_gas_price: u64,
26    /// Epoch the transaction runs in.
27    pub epoch_id: u64,
28    /// Epoch start timestamp in milliseconds (the VM clock).
29    pub epoch_timestamp_ms: u64,
30    /// Which chain this is (`Mainnet` / `Testnet` / `Unknown`), used for
31    /// chain-specific protocol behaviour.
32    pub chain: Chain,
33}
34
35impl ChainContext {
36    /// Start from the protocol version and chain. The epoch fields default to
37    /// `0`; set them with the `with_*` methods.
38    pub fn new(protocol_version: ProtocolVersion, chain: Chain) -> Self {
39        Self {
40            protocol_version,
41            reference_gas_price: 0,
42            epoch_id: 0,
43            epoch_timestamp_ms: 0,
44            chain,
45        }
46    }
47
48    /// Set the reference gas price for the epoch, in NANOS.
49    #[must_use]
50    pub fn with_reference_gas_price(mut self, reference_gas_price: u64) -> Self {
51        self.reference_gas_price = reference_gas_price;
52        self
53    }
54
55    /// Set the epoch the transaction runs in.
56    #[must_use]
57    pub fn with_epoch_id(mut self, epoch_id: u64) -> Self {
58        self.epoch_id = epoch_id;
59        self
60    }
61
62    /// Set the epoch start timestamp in milliseconds (the VM clock).
63    #[must_use]
64    pub fn with_epoch_timestamp_ms(mut self, epoch_timestamp_ms: u64) -> Self {
65        self.epoch_timestamp_ms = epoch_timestamp_ms;
66        self
67    }
68}

70/// How a transaction is run.
71#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
72#[non_exhaustive]
73pub enum ExecutionMode {
74    /// Relaxed Move VM checks; the store is never modified.
75    DevInspect,
76    /// Full sign-time checks; the store is never modified. The default.
77    ///
78    /// Object references in the transaction (gas payments and owned inputs)
79    /// are resolved against whatever versions the store holds, so a stale
80    /// version or digest that a node would reject at signing time still runs
81    /// here.
82    #[default]
83    DryRun,
84    /// Full sign-time checks; on success, effects are applied back to the
85    /// store and [`ExecutionResult::committed`] is `true`.
86    ///
87    /// Requires a real gas payment — a gasless transaction is rejected rather
88    /// than funded with the mock simulation coin, since its effects would be
89    /// committed.
90    ///
91    /// A transaction that aborts commits nothing — not even the gas charge — so
92    /// across multiple runs the store does not reflect a node's post-abort
93    /// state.
94    Execute,
95}

97/// The outcome of signature verification for a run.
98#[derive(Debug)]
99#[non_exhaustive]
100pub enum SignatureStatus {
101    /// No signature was supplied (unsigned
102    /// [`LocalVm::execute`](super::LocalVm::execute)).
103    NotChecked,
104    /// Signatures verified successfully. For a
105    /// [`MoveAuthenticator`](iota_sdk_types::MoveAuthenticator)
106    /// this means the authenticator function did not abort during execution.
107    Verified,
108    /// Signature verification failed.
109    Failed(crate::error::SignatureError),
110}
111
112impl SignatureStatus {
113    pub(crate) fn from_authentication(
114        outcome: Result<(), iota_types::error::ExecutionError>,
115    ) -> Self {
116        match outcome {
117            Ok(()) => Self::Verified,
118            Err(e) => Self::Failed(crate::error::SignatureError::new(e)),
119        }
120    }
121}

123/// Options for a single run.
124#[derive(Debug, Clone, Default)]
125#[non_exhaustive]
126pub struct ExecuteOptions {
127    pub mode: ExecutionMode,
128    pub debug: DebugConfig,
129    /// Deny-list configuration applied during preparation. Defaults to an
130    /// empty deny-list; set it to the target network's configuration to match
131    /// a live validator's deny checks.
132    pub deny_config: TransactionDenyConfig,
133    /// Whether to run the regulated-coin deny-list check (a validator's
134    /// `check_coin_deny_list_v1`) during preparation. Off by default: it does a
135    /// store lookup per coin type in the transaction, so it is opt-in and only
136    /// meaningful when the store holds the on-chain `DenyList`. Distinct from
137    /// [`deny_config`](Self::deny_config), which is the operator's local
138    /// policy.
139    pub check_coin_deny_list: bool,
140}
141
142impl ExecuteOptions {
143    /// Dev-inspect mode (relaxed checks).
144    pub fn dev_inspect() -> Self {
145        Self {
146            mode: ExecutionMode::DevInspect,
147            ..Self::default()
148        }
149    }
150
151    /// Dry-run mode (full checks, no commit).
152    pub fn dry_run() -> Self {
153        Self {
154            mode: ExecutionMode::DryRun,
155            ..Self::default()
156        }
157    }
158
159    /// Execute mode (full checks, commit on success).
160    pub fn execute() -> Self {
161        Self {
162            mode: ExecutionMode::Execute,
163            ..Self::default()
164        }
165    }
166
167    /// Attach a [`DebugConfig`] to capture a gas profile and/or an execution
168    /// trace.
169    #[must_use]
170    pub fn with_debug(mut self, cfg: DebugConfig) -> Self {
171        self.debug = cfg;
172        self
173    }
174
175    /// Set the deny-list configuration applied during preparation.
176    #[must_use]
177    pub fn with_deny_config(mut self, deny_config: TransactionDenyConfig) -> Self {
178        self.deny_config = deny_config;
179        self
180    }
181
182    /// Enable the regulated-coin deny-list check during preparation. Requires
183    /// the on-chain `DenyList` object in the store to have any effect; see
184    /// [`check_coin_deny_list`](Self::check_coin_deny_list).
185    #[must_use]
186    pub fn with_coin_deny_list_check(mut self) -> Self {
187        self.check_coin_deny_list = true;
188        self
189    }
190}

196/// The full result of a run.
197#[derive(Debug)]
198#[non_exhaustive]
199pub struct ExecutionResult {
200    /// The transaction effects (object changes, gas, status digest).
201    pub effects: TransactionEffects,
202    /// Emitted events, if the run produced any.
203    pub events: Option<TransactionEvents>,
204    /// Per-PTB-command `(mutable_reference_outputs, return_values)`.
205    ///
206    /// Empty for `MoveAuthenticator`-signed runs (the authenticator engine
207    /// entry point does not return per-command results) and for failed runs
208    /// (the engine reports them only for a successful execution).
209    pub command_results: Vec<CommandResult>,
210    /// Objects read as inputs to the run.
211    pub input_objects: Vec<Object>,
212    /// Objects written by the run (created or mutated).
213    pub output_objects: Vec<Object>,
214    /// Gas ledger for the run (computation / storage / rebate).
215    pub gas_summary: GasCostSummary,
216    /// Id of the mock gas coin minted for a gas-less transaction, if any.
217    pub mock_gas_id: Option<ObjectId>,
218    /// The Move-level execution status (success or abort).
219    pub status: iota_sdk_types::ExecutionStatus,
220    /// The outcome of signature verification for the run.
221    pub signature_status: SignatureStatus,
222    /// `true` if and only if [`ExecutionMode::Execute`] ran successfully and
223    /// the effects were applied back to the store.
224    pub committed: bool,
225    /// Captured debug artifacts (profile / trace), if requested.
226    pub debug: Option<DebugArtifacts>,
227}
