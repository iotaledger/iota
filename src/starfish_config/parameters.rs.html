7use serde::{Deserialize, Serialize};

9/// Ceiling on the number of headers, and on the number of shards, a bundle may
10/// declare. Enforced while decoding, before the entries are materialised, so a
11/// peer cannot make the decoder build a vector far larger than the configured
12/// caps allow it to keep. Both caps are validated against it at startup.
13pub const MAX_HEADERS_OR_SHARDS_PER_BUNDLE: usize = 1024;
14
15/// Ceiling on the headers a header-sync fetch response may carry. The server
16/// truncates its response to its own `max_headers_per_header_sync_fetch`,
17/// validated against this at startup, so a client can reject a response past
18/// it as misbehaviour while still accepting a peer configured above its own
19/// cap.
20pub const MAX_HEADERS_PER_HEADER_SYNC_FETCH: usize = 1024;
21
22/// Operational configurations of a consensus authority.
23///
24/// All fields should tolerate inconsistencies among authorities, without
25/// affecting safety of the protocol. Otherwise, they need to be part of IOTA
26/// protocol config or epoch state on-chain.
27///
28/// NOTE: fields with default values are specified in the serde default
29/// functions. Most operators should not need to specify any field, except
30/// db_path.
31#[derive(Clone, Debug, Deserialize, Serialize)]
32pub struct Parameters {
33    /// Path to consensus DB for this epoch. Required when initializing
34    /// consensus. This is calculated based on user configuration for base
35    /// directory.
36    #[serde(skip)]
37    pub db_path: PathBuf,
38
39    /// Time to wait for parent round leader before sealing a block, from when
40    /// parent round has a quorum.
41    #[serde(default = "Parameters::default_leader_timeout")]
42    pub leader_timeout: Duration,
43
44    /// Sustained spacing between own blocks: long-run production never exceeds
45    /// one block per `min_block_delay`. This avoids generating too many rounds
46    /// when latency is low. This is especially necessary for tests running
47    /// locally. If setting a non-default value, it should be set low enough
48    /// to avoid reducing round rate and increasing latency in realistic and
49    /// distributed configurations.
50    #[serde(default = "Parameters::default_min_block_delay")]
51    pub min_block_delay: Duration,
52
53    /// Soft counterpart of `leader_timeout`: after this duration we are
54    /// willing to propose a block even without a strong-vote quorum, to avoid
55    /// liveness stalls when leader data is slow to propagate. Fires earlier
56    /// than `leader_timeout` and does not force block creation on its own.
57    #[serde(default = "Parameters::default_soft_leader_timeout")]
58    pub soft_leader_timeout: Duration,
59
60    /// Window bounding own block production together with `min_block_delay`:
61    /// idle time accrues budget for bursts of up to `block_rate_window /
62    /// min_block_delay` back-to-back blocks, letting a validator that fell
63    /// behind catch up on rounds instead of skipping them. Set at or below
64    /// `min_block_delay` to disable bursting (fixed spacing between blocks).
65    #[serde(default = "Parameters::default_block_rate_window")]
66    pub block_rate_window: Duration,
67
68    /// Number of block headers to fetch per commit sync request.
69    #[serde(default = "Parameters::default_max_headers_per_commit_sync_fetch")]
70    pub max_headers_per_commit_sync_fetch: usize,
71
72    /// Number of transactions to fetch per commit sync request.
73    #[serde(default = "Parameters::default_max_transactions_per_commit_sync_fetch")]
74    pub max_transactions_per_commit_sync_fetch: usize,
75
76    /// Number of block headers to fetch per header sync (periodic or live)
77    /// request.
78    #[serde(default = "Parameters::default_max_headers_per_header_sync_fetch")]
79    pub max_headers_per_header_sync_fetch: usize,
80
81    /// Number of transactions to fetch per transaction sync request.
82    #[serde(default = "Parameters::default_max_transactions_per_transaction_sync_fetch")]
83    pub max_transactions_per_transaction_sync_fetch: usize,
84
85    /// Time to wait during node start up until the node has synced the last
86    /// proposed block via the network peers. When set to `0` the sync
87    /// mechanism is disabled. This property is meant to be used for amnesia
88    /// recovery.
89    #[serde(default = "Parameters::default_sync_last_known_own_block_timeout")]
90    pub sync_last_known_own_block_timeout: Duration,
91
92    /// The number of rounds of blocks to be kept in the Dag state cache per
93    /// authority. The larger the number the more the blocks that will be
94    /// kept in memory allowing minimising any potential disk access.
95    /// Value should be at minimum 50 rounds to ensure node performance, but
96    /// being too large can be expensive in memory usage.
97    #[serde(default = "Parameters::default_dag_state_cached_rounds")]
98    pub dag_state_cached_rounds: u32,
99
100    /// Rounds a header from a far-future-bounded source may lead the locally
101    /// accepted frontier, in addition to `dag_state_cached_rounds`, before it
102    /// is dropped as too far ahead to connect.
103    #[serde(default = "Parameters::default_peer_round_ahead_margin")]
104    pub peer_round_ahead_margin: u32,
105
106    // Number of authorities commit syncer fetches in parallel.
107    // Both commits in a range and blocks referenced by the commits are fetched per authority.
108    #[serde(default = "Parameters::default_commit_sync_parallel_fetches")]
109    pub commit_sync_parallel_fetches: usize,
110
111    // Number of commits to fetch in a batch, also the maximum number of commits returned per
112    // fetch. If this value is set too small, fetching becomes inefficient.
113    // If this value is set too large, it can result in load imbalance and stragglers.
114    #[serde(default = "Parameters::default_commit_sync_batch_size")]
115    pub commit_sync_batch_size: u32,
116
117    // This affects the maximum number of commit batches being fetched, and those fetched but not
118    // processed as consensus output, before throttling of outgoing commit fetches starts.
119    #[serde(default = "Parameters::default_commit_sync_batches_ahead")]
120    pub commit_sync_batches_ahead: usize,
121
122    /// Maximum number of commits scanned and replayed per batch during
123    /// recovery, bounding peak memory when a large unprocessed range is
124    /// replayed at startup.
125    #[serde(default = "Parameters::default_commit_recovery_batch_size")]
126    pub commit_recovery_batch_size: u32,
127
128    /// Maximum number of headers to be included in a bundle. Headers exceeding
129    /// the max allowed limit will be truncated.
130    #[serde(default = "Parameters::default_max_headers_per_bundle")]
131    pub max_headers_per_bundle: usize,
132
133    /// Maximum number of transaction shards to be included in a bundle. Shards
134    /// exceeding the max allowed limit will be truncated.
135    #[serde(default = "Parameters::default_max_shards_per_bundle")]
136    pub max_shards_per_bundle: usize,
137
138    /// Tonic network settings.
139    #[serde(default = "TonicParameters::default")]
140    pub tonic: TonicParameters,
141
142    /// Number of commits requested per fast commit sync fetch. A response may
143    /// carry up to twice this many, since it extends past the requested end to
144    /// reach a certifiable commit. Set too small, fetching is inefficient and
145    /// the server has little room to reach one; set too large, it causes load
146    /// imbalance and stragglers.
147    #[serde(default = "Parameters::default_fast_commit_sync_batch_size")]
148    pub fast_commit_sync_batch_size: u32,
149
150    // Gap threshold for switching between commit syncers. When the gap between quorum and local
151    // commit index is larger than this threshold, FastCommitSyncer fetches. Otherwise,
152    // CommitSyncer fetches.
153    #[serde(default = "Parameters::default_commit_sync_gap_threshold")]
154    pub commit_sync_gap_threshold: u32,
155
156    /// Enable FastCommitSyncer for faster recovery from large commit gaps.
157    /// Enabled by default; operators can disable it locally if bugs are
158    /// discovered.
159    #[serde(default = "Parameters::default_enable_fast_commit_syncer")]
160    pub enable_fast_commit_syncer: bool,
161
162    /// Reconnect block streams after fast commit sync reinitializes consensus,
163    /// discarding bundles buffered against the previous state. Enabled by
164    /// default; operators can disable it locally. Has no effect unless
165    /// `enable_fast_commit_syncer` is also enabled, since only fast sync
166    /// signals the reset.
167    #[serde(default = "Parameters::default_enable_block_stream_reset_on_fast_sync_exit")]
168    pub enable_block_stream_reset_on_fast_sync_exit: bool,
169
170    /// Ask commit-sync peers that have voted for the end of the requested
171    /// range before those that have not, since a vote means the peer has
172    /// solidified every commit in the range. Peers without an observed vote
173    /// are ordered behind; each fetch round tries a bounded number of peers,
174    /// so on a committee larger than that bound they can stay outside the
175    /// round until their votes are observed. Enabled by default; disabling it
176    /// restores a plain uniform order.
177    #[serde(default = "Parameters::default_enable_commit_sync_peer_selection_by_commit_votes")]
178    pub enable_commit_sync_peer_selection_by_commit_votes: bool,
179
180    /// Enable adaptive acknowledgment filtering for StarfishSpeed.
181    /// Local heuristic that drops acks for authorities persistently blamed
182    /// by recent strong-vote masks. Effective only when the protocol-level
183    /// `consensus_starfish_speed` flag is also on. Enabled by default;
184    /// operators can disable it locally without a protocol change.
185    #[serde(default = "Parameters::default_enable_starfish_speed_adaptive_acknowledgments")]
186    pub enable_starfish_speed_adaptive_acknowledgments: bool,
187
188    /// Prefer more responsive peers when the transactions synchronizer, the
189    /// commit syncer and the header synchronizer select peers to fetch from.
190    /// Responses are verified the same way regardless, so it cannot affect
191    /// safety. Enabled by default; disabling it restores the previous
192    /// selection: for the transactions synchronizer a uniform random order
193    /// that excludes the most recently failed peers (up to less than f+1 by
194    /// stake), and for the commit syncer and the header synchronizer a uniform
195    /// random order.
196    #[serde(default = "Parameters::default_enable_peer_responsiveness_ranking")]
197    pub enable_peer_responsiveness_ranking: bool,
198
199    /// Ask only a bounded number of peers that recently supplied a new header
200    /// or referenced a missing one to push more headers from that author.
201    #[serde(default = "Parameters::default_enable_bounded_header_advertisement")]
202    pub enable_bounded_header_advertisement: bool,
203
204    /// Ask only the fastest peers that together provide enough shards and
205    /// stake for reconstruction. Enabled by default; disabling it asks every
206    /// peer.
207    #[serde(default = "Parameters::default_enable_shard_peer_selection")]
208    pub enable_shard_peer_selection: bool,
209
210    /// Port for the DAG visualizer gRPC server (localhost only).
211    /// When set, starts a debugging server for real-time DAG visualization.
212    /// Only has an effect when the `dag-visualizer` feature is compiled in.
213    /// Disabled by default (None).
214    #[serde(default)]
215    pub dag_visualizer_port: Option<u16>,
216
217    /// Maximum number of rounds the last commit's leader may run ahead of the
218    /// last solid commit's leader before the node stops accepting new block
219    /// bundles and restricts header fetching, until solidification catches up.
220    #[serde(default = "Parameters::default_solid_commit_lag_threshold")]
221    pub solid_commit_lag_threshold: u32,
222
223    /// Maximum number of shards from one relaying authority retained across
224    /// all pending shard accumulators. At the budget, admitting a new shard
225    /// from the authority evicts its oldest retained one, so reconstructor
226    /// memory stays bounded by `committee size × budget × shard size`
227    /// regardless of how far commits run ahead of solidification.
228    #[serde(default = "Parameters::default_shard_budget_per_authority")]
229    pub shard_budget_per_authority: u32,
230
231    /// Maximum transaction payload bytes one fast commit-sync response carries.
232    /// A fetch covering more commits than this is answered with the commits
233    /// whose payloads fit, and the requester asks for the rest in its next
234    /// fetch. When the range's first commit exceeds this on its own it is
235    /// still served whole, one such response at a time.
236    #[serde(default = "Parameters::default_max_fast_commit_sync_transaction_bytes")]
237    pub max_fast_commit_sync_transaction_bytes: usize,
238}

240impl Parameters {
241    /// Threshold for the number of commits sent to the consumer but not yet
242    /// handled, above which commit producers (commit syncers, commit observer
243    /// recovery) pause to let the consumer catch up.
244    pub fn unhandled_commits_threshold(&self) -> u32 {
245        self.commit_sync_batch_size * (self.commit_sync_batches_ahead as u32)
246    }
247
248    pub(crate) fn default_leader_timeout() -> Duration {
249        Duration::from_millis(200)
250    }
251
252    pub(crate) fn default_min_block_delay() -> Duration {
253        if cfg!(msim) || std::env::var("__TEST_ONLY_CONSENSUS_USE_LONG_MIN_BLOCK_DELAY").is_ok() {
254            // Checkpoint building and execution cannot keep up with high commit rate in
255            // simtests, leading to long reconfiguration delays. This is because
256            // simtest is single threaded, and spending too much time in
257            // consensus can lead to starvation elsewhere.
258            Duration::from_millis(400)
259        } else if cfg!(test) {
260            // Avoid excessive CPU, data and logs in tests.
261            Duration::from_millis(250)
262        } else {
263            // For production, use min delay between block being set to 50ms, reducing the
264            // block rate to 20 blocks/sec
265            Duration::from_millis(50)
266        }
267    }
268
269    pub(crate) fn default_soft_leader_timeout() -> Duration {
270        Duration::from_millis(5)
271    }
272
273    pub(crate) fn default_block_rate_window() -> Duration {
274        Duration::from_secs(2)
275    }

277    /// Burst capacity: maximum number of own blocks within `block_rate_window`
278    /// (40 in production, 5 in msim, 8 in tests with the default window).
279    pub fn block_rate_burst(&self) -> u64 {
280        let interval_ms = self.min_block_delay.as_millis().max(1) as u64;
281        (self.block_rate_window.as_millis() as u64 / interval_ms).max(1)
282    }
283
284    /// Highest round a header from a far-future-bounded source may have,
285    /// relative to the accepted `frontier`, to still be close enough to
286    /// connect; headers above this are too far ahead and dropped.
287    pub fn far_future_round_ceiling(&self, frontier: u32) -> u32 {
288        frontier
289            .saturating_add(self.dag_state_cached_rounds)
290            .saturating_add(self.peer_round_ahead_margin)
291    }
292
293    /// Maximum number of block headers served per fetch request, depending on
294    /// whether the request comes from commit sync or the header synchronizer.
295    pub fn max_headers_per_fetch(&self, commit_sync: bool) -> usize {
296        if commit_sync {
297            self.max_headers_per_commit_sync_fetch
298        } else {
299            self.max_headers_per_header_sync_fetch
300        }
301    }

303    /// Validates local consensus parameters, rejecting zero values that can
304    /// lead to synchronization problems. Returns a description of the first
305    /// offending field.
306    pub fn validate(&self) -> Result<(), String> {
307        let positive_fields = [
308            (
309                "max_headers_per_commit_sync_fetch",
310                self.max_headers_per_commit_sync_fetch as u128,
311            ),
312            (
313                "max_transactions_per_commit_sync_fetch",
314                self.max_transactions_per_commit_sync_fetch as u128,
315            ),
316            (
317                "max_headers_per_header_sync_fetch",
318                self.max_headers_per_header_sync_fetch as u128,
319            ),
320            (
321                "max_transactions_per_transaction_sync_fetch",
322                self.max_transactions_per_transaction_sync_fetch as u128,
323            ),
324            (
325                "dag_state_cached_rounds",
326                self.dag_state_cached_rounds as u128,
327            ),
328            (
329                "commit_sync_parallel_fetches",
330                self.commit_sync_parallel_fetches as u128,
331            ),
332            (
333                "commit_sync_batch_size",
334                self.commit_sync_batch_size as u128,
335            ),
336            (
337                "commit_recovery_batch_size",
338                self.commit_recovery_batch_size as u128,
339            ),
340            (
341                "commit_sync_batches_ahead",
342                self.commit_sync_batches_ahead as u128,
343            ),
344            (
345                "max_headers_per_bundle",
346                self.max_headers_per_bundle as u128,
347            ),
348            ("max_shards_per_bundle", self.max_shards_per_bundle as u128),
349            (
350                "fast_commit_sync_batch_size",
351                self.fast_commit_sync_batch_size as u128,
352            ),
353            (
354                "max_fast_commit_sync_transaction_bytes",
355                self.max_fast_commit_sync_transaction_bytes as u128,
356            ),
357            (
358                "tonic.connection_buffer_size",
359                self.tonic.connection_buffer_size as u128,
360            ),
361            (
362                "tonic.excessive_message_size",
363                self.tonic.excessive_message_size as u128,
364            ),
365            (
366                "tonic.message_size_limit",
367                self.tonic.message_size_limit as u128,
368            ),
369            (
370                "tonic.keepalive_interval",
371                self.tonic.keepalive_interval.as_nanos(),
372            ),
373            (
374                "shard_budget_per_authority",
375                self.shard_budget_per_authority as u128,
376            ),
377        ];
378        for (name, value) in positive_fields {
379            if value == 0 {
380                return Err(format!("{name} must be positive"));
381            }
382        }
383        let bundle_fields = [
384            ("max_headers_per_bundle", self.max_headers_per_bundle),
385            ("max_shards_per_bundle", self.max_shards_per_bundle),
386        ];
387        for (name, value) in bundle_fields {
388            if value > MAX_HEADERS_OR_SHARDS_PER_BUNDLE {
389                return Err(format!(
390                    "{name} must not exceed {MAX_HEADERS_OR_SHARDS_PER_BUNDLE}"
391                ));
392            }
393        }
394        if self.max_headers_per_header_sync_fetch > MAX_HEADERS_PER_HEADER_SYNC_FETCH {
395            return Err(format!(
396                "max_headers_per_header_sync_fetch must not exceed {MAX_HEADERS_PER_HEADER_SYNC_FETCH}"
397            ));
398        }
399        Ok(())
400    }

622    ///
623    /// If unspecified, this will default to 5s.
624    #[serde(default = "TonicParameters::default_keepalive_interval")]
625    pub keepalive_interval: Duration,
626
627    /// Size of various per-connection buffers.
628    ///
629    /// If unspecified, this will default to 32MiB.

634    /// increment a counter.

636    /// If unspecified, this will default to 16MiB.

638    pub excessive_message_size: usize,
639

753/// Per-peer, per-RPC concurrency caps for the inbound consensus gRPC server.
754///
755/// Each cap bounds how many concurrent requests of one RPC group a single
756/// committee peer (keyed on its authenticated authority index) may have in
757/// flight; a peer cannot consume another peer's budget. These are local,
758/// non-protocol parameters — heterogeneous values across authorities are safe,
759/// so they can be rolled out and tuned per node.
760///
761/// The defaults are sized for ~100-validator committees and the local
762/// synchronizer fan-out toward one server. `0` turns a cap off; each cap is
763/// checked on its own.
764#[derive(Clone, Debug, Deserialize, Serialize)]
765pub struct AdmissionParameters {
766    /// Max concurrent block-subscription streams per peer.
767    ///
768    /// If unspecified, this will default to 2.
769    #[serde(default = "AdmissionParameters::default_max_subscriptions_per_peer")]
770    pub max_subscriptions_per_peer: u32,
771
772    /// Max concurrent header fetches per peer
773    /// (`fetch_block_headers` + `fetch_latest_block_headers`).
774    ///
775    /// If unspecified, this will default to 32.
776    #[serde(default = "AdmissionParameters::default_max_header_fetches_per_peer")]
777    pub max_header_fetches_per_peer: u32,
778
779    /// Max concurrent transaction fetches per peer (`fetch_transactions`).
780    ///
781    /// If unspecified, this will default to 16.
782    #[serde(default = "AdmissionParameters::default_max_transaction_fetches_per_peer")]
783    pub max_transaction_fetches_per_peer: u32,
784
785    /// Max concurrent commit fetches per peer
786    /// (`fetch_commits` + `fetch_commits_and_transactions`).
787    ///
788    /// If unspecified, this will default to 8.
789    #[serde(default = "AdmissionParameters::default_max_commit_fetches_per_peer")]
790    pub max_commit_fetches_per_peer: u32,
791
792    /// Max concurrent commit fetches across all peers. A fast commit-sync
793    /// response is held in memory until it has been sent, so this caps what
794    /// serving them can cost the node at once.
795    ///
796    /// If unspecified, this will default to 16.
797    #[serde(default = "AdmissionParameters::default_max_commit_fetches_total")]
798    pub max_commit_fetches_total: u32,
799}
800
801impl AdmissionParameters {
802    fn default_max_subscriptions_per_peer() -> u32 {
803        2
804    }
805
806    fn default_max_header_fetches_per_peer() -> u32 {
807        32
808    }
809
810    fn default_max_transaction_fetches_per_peer() -> u32 {
811        16
812    }
813
814    fn default_max_commit_fetches_per_peer() -> u32 {
815        Parameters::default_commit_sync_parallel_fetches() as u32
816    }
817
818    fn default_max_commit_fetches_total() -> u32 {
819        2 * AdmissionParameters::default_max_commit_fetches_per_peer()
820    }
821}

827            max_header_fetches_per_peer: AdmissionParameters::default_max_header_fetches_per_peer(),
828            max_transaction_fetches_per_peer:
829                AdmissionParameters::default_max_transaction_fetches_per_peer(),
830            max_commit_fetches_per_peer: AdmissionParameters::default_max_commit_fetches_per_peer(),
831            max_commit_fetches_total: AdmissionParameters::default_max_commit_fetches_total(),
