30/// Managing and processing checkpoint files in a directory.
31pub(crate) trait LocalRead {
32    /// Path is used as the source for reading checkpoint files.
33    fn path(&self) -> &Path;
34
35    /// Returns the current checkpoint sequence number.
36    fn current_checkpoint_number(&self) -> CheckpointSequenceNumber;
37
38    fn update_last_pruned_watermark(&mut self, watermark: CheckpointSequenceNumber);
39
40    /// Returns `true` if the given checkpoint sequence number exceeds the
41    /// allowed capacity.
42    fn exceeds_capacity(&self, checkpoint_number: CheckpointSequenceNumber) -> bool;
43
44    /// Returns `true` if the checkpoint's sequence number is ahead of the
45    /// expected sequence number, indicating a gap in the processed
46    /// checkpoints.
47    fn is_checkpoint_ahead(
48        &self,
49        checkpoint: &CheckpointData,
50        expected_sequence_number: CheckpointSequenceNumber,
51    ) -> bool {
52        checkpoint.checkpoint_summary.sequence_number > expected_sequence_number
53    }
54
55    /// Lists unprocessed checkpoint files in the specified directory.
56    ///
57    /// Scans the checkpoint directory for files whose sequence number is
58    /// greater than or equal to the current checkpoint number. Returns a
59    /// map of sequence numbers to file paths, sorted in ascending order.
60    fn list_unprocessed_checkpoint_files(
61        &self,
62    ) -> IngestionResult<BTreeMap<CheckpointSequenceNumber, PathBuf>> {
63        let mut files = BTreeMap::new();
64        for entry in fs::read_dir(self.path())? {
65            let entry = entry?;
66            let filename = entry.file_name();
67            if let Some(sequence_number) = self.checkpoint_number_from_file_path(&filename) {
68                if sequence_number >= self.current_checkpoint_number() {
69                    files.insert(sequence_number, entry.path());
70                }
71            }
72        }
73        Ok(files)
74    }
75
76    /// Reads and deserializes unprocessed checkpoint files from the directory,
77    /// up to capacity.
78    ///
79    /// Iterates over unprocessed checkpoint files, deserializing each into a
80    /// [`CheckpointData`]. Stops early if the capacity is exceeded, as
81    /// determined by [`LocalRead::exceeds_capacity`], or when
82    /// [`MAX_CHECKPOINTS_IN_PROGRESS`] files have been processed.
83    fn read_local_files(&self) -> IngestionResult<Vec<Arc<CheckpointData>>> {
84        // files are already sorted by sequence number in ascending order
85        let files = self.list_unprocessed_checkpoint_files()?;
86        debug!("unprocessed local files {files:?}");
87        let mut checkpoints = vec![];
88        for (_, filename) in files.iter().take(MAX_CHECKPOINTS_IN_PROGRESS) {
89            let checkpoint = self.read_checkpoint_file(filename)?;
90            if self.exceeds_capacity(checkpoint.checkpoint_summary.sequence_number) {
91                break;
92            }
93            checkpoints.push(checkpoint);
94        }
95        Ok(checkpoints)
96    }
97
98    /// Reads and deserializes unprocessed checkpoint files with retry and
99    /// capacity check.
100    ///
101    /// This method wraps [`LocalRead::read_local_files`] with an
102    /// exponential backoff retry mechanism to handle transient read errors.
103    /// Retries are performed according to the default
104    /// [`backoff::ExponentialBackoff`] policy.
105    async fn read_local_files_with_retry(&self) -> IngestionResult<Vec<Arc<CheckpointData>>> {
106        let backoff = backoff::ExponentialBackoff::default();
107        backoff::future::retry(backoff, || async {
108            self.read_local_files().map_err(|err| {
109                info!("transient local read error {err:?}");
110                backoff::Error::transient(err)
111            })
112        })
113        .await
114    }
115
116    /// Reads and deserializes a checkpoint file.
117    fn read_checkpoint_file(&self, filename: &Path) -> IngestionResult<Arc<CheckpointData>> {
118        let data = fs::read(filename)?;
119        Blob::from_bytes::<Arc<CheckpointData>>(&data)
120            .map_err(|err| IngestionError::DeserializeCheckpoint(err.to_string()))
121    }
122
123    fn checkpoint_number_from_file_path(
124        &self,
125        file_name: &OsString,
126    ) -> Option<CheckpointSequenceNumber> {
127        file_name
128            .to_str()
129            .and_then(|s| s.rfind('.').map(|pos| &s[..pos]))
130            .and_then(|s| s.parse().ok())
131    }
132
133    /// Cleans the local directory by removing all processed checkpoint files.
134    fn gc_processed_files(&mut self, watermark: CheckpointSequenceNumber) -> IngestionResult<()> {
135        debug!("cleaning processed files, watermark is {watermark}");
136        self.update_last_pruned_watermark(watermark);
137        for entry in fs::read_dir(self.path())? {
138            let entry = entry?;
139            let filename = entry.file_name();
140            if let Some(sequence_number) = self.checkpoint_number_from_file_path(&filename) {
141                if sequence_number < watermark {
142                    fs::remove_file(entry.path())?;
143                }
144            }
145        }
146        Ok(())
147    }
148}

150/// Sets up an inotify watcher on the given path and returns the watcher. The
151/// Receiver yields a unit value `()` whenever a filesystem event occurs
152/// in the watched directory
153#[cfg(not(target_os = "macos"))]
154pub(crate) fn init_watcher(
155    inotify_sender: tokio::sync::mpsc::Sender<()>,
156    path: &Path,
157) -> RecommendedWatcher {
158    use notify::Watcher;
159
160    let mut watcher = notify::recommended_watcher(move |res| {
161        if let Err(err) = res {
162            eprintln!("watch error: {err:?}");
163        }
164        inotify_sender
165            .blocking_send(())
166            .expect("failed to send inotify update");
167    })
168    .expect("failed to init inotify");
169    watcher
170        .watch(path, RecursiveMode::NonRecursive)
171        .expect("inotify watcher failed");
172    watcher
173}

175#[derive(Debug, Clone, Copy)]
176pub enum ReadSource {
177    Local,
178    Remote,
179}
180
181impl Display for ReadSource {
182    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
183        match self {
184            ReadSource::Local => write!(f, "local"),
185            ReadSource::Remote => write!(f, "remote"),
186        }
187    }
188}

190/// Fetches and deserializes a checkpoint from an object store.
191pub async fn fetch_from_object_store(
192    store: &dyn ObjectStore,
193    checkpoint_number: CheckpointSequenceNumber,
194) -> CheckpointResult {
195    let path = ObjectStorePath::from(format!("{checkpoint_number}.{CHECKPOINT_FILE_SUFFIX}"));
196    debug!("fetch {path} from live");
197    let response = store.get(&path).await?;
198    let bytes = response.bytes().await?;
199    Ok((
200        Blob::from_bytes::<Arc<CheckpointData>>(&bytes)
201            .map_err(|err| IngestionError::DeserializeCheckpoint(err.to_string()))?,
202        bytes.len(),
203    ))
204}
