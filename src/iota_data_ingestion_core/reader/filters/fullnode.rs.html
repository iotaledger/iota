4//! Server-side filters for fullnode gRPC connections.
5//!
6//! Filters tell the fullnode which transactions to include in each
7//! checkpoint payload. Filtering happens entirely on the server-side, the
8//! ingestion framework performs no client-side filtering.
9//!
10//! # Filter types
11//!
12//! - [`TransactionFilter`]: applied to a checkpoint's transactions.
13//! - [`EventFilter`]: matches a transaction by its emitted events. Used as
14//!   input to [`TransactionFilter::event`].
15//! - [`CommandFilter`]: matches a command within a programmable transaction.
16//!   Used as input to [`TransactionFilter::command`].
17//!
18//! # Composition
19//!
20//! [`TransactionFilter`] and [`EventFilter`] use a chained builder pattern:
21//! each leaf method (e.g. `TransactionFilter::new().kinds().sender()`)
22//! implicitly `AND`s the new condition with everything accumulated so far.
23//! Explicit `OR` composition is available via the [`or`](TransactionFilter::or)
24//! method on each type. `NOT` is available via
25//! [`negate`](TransactionFilter::negate), the `!` operator, or `.not()` when
26//! [`std::ops::Not`] is in scope.
27//!
28//! [`CommandFilter`] is composed only at the [`TransactionFilter`] level by
29//! passing each command to [`TransactionFilter::command`] within a chain.
30//! Multiple [`command`](TransactionFilter::command) calls implicitly `AND`,
31//! while `OR` and `NOT` follow the rules described above.
32//!
33//! # Examples
34//!
35//! Successful programmable transactions:
36//!
37//! ```rust
38//! use iota_data_ingestion_core::filters::fullnode::{TransactionFilter, TransactionKind};
39//!
40//! let filter = TransactionFilter::new()
41//!     .kinds([TransactionKind::Programmable])
42//!     .execution_status(true);
43//! ```
44//!
45//! Transactions emitting a specific event type:
46//!
47//! ```rust
48//! use iota_data_ingestion_core::filters::fullnode::{EventFilter, TransactionFilter};
49//!
50//! let filter =
51//!     TransactionFilter::new().event(EventFilter::new().event_type("0xabcd::my_module::Foo"));
52//! ```
53//!
54//! Transactions NOT sent by a specific address:
55//!
56//! ```ignore
57//! let filter = !TransactionFilter::new().sender(some_address);
58//! ```
59

68/// Filter applied to transactions in a fullnode checkpoint stream.
69///
70/// Built with a chained builder pattern. Start from [`TransactionFilter::new`]
71/// and add leaves with the named methods (e.g. [`TransactionFilter::kinds`],
72/// [`TransactionFilter::sender`], [`TransactionFilter::execution_status`]).
73///
74/// Each leaf implicitly `AND`s with everything accumulated so far. Use
75/// [`TransactionFilter::or`] for `OR` composition and
76/// [`TransactionFilter::negate`] (or the `!` operator) for `NOT`.
77///
78/// # Example
79///
80/// Implicit AND composition
81///
82/// ```rust
83/// use iota_data_ingestion_core::filters::fullnode::{TransactionFilter, TransactionKind};
84///
85/// let filter = TransactionFilter::new()
86///     .kinds([TransactionKind::Programmable])
87///     .execution_status(true);
88/// ```
89/// Composition with AND, OR and NOT:
90///
91/// ```ignore
92/// // (Programmable AND success AND sender == Alice) OR NOT(receiver == Bob)
93/// let filter = TransactionFilter::new()
94///     .kinds([TransactionKind::Programmable])
95///     .execution_status(true)
96///     .sender(alice)
97///     .or(!TransactionFilter::new().receiver(bob));
98/// ```
99#[derive(Clone, Debug, Default)]
100pub struct TransactionFilter(proto::TransactionFilter);
101
102impl TransactionFilter {
103    /// Creates an empty filter.
104    ///
105    /// An empty filter is rejected by the fullnode. Add at least one leaf via
106    /// the builder methods (e.g. [`TransactionFilter::kinds`],
107    /// [`TransactionFilter::sender`]) before passing the filter to the
108    /// ingestion framework.
109    ///
110    /// # Example
111    ///
112    /// ```rust
113    /// use iota_data_ingestion_core::filters::fullnode::{TransactionFilter, TransactionKind};
114    ///
115    /// let filter = TransactionFilter::new().kinds([TransactionKind::Programmable]);
116    /// ```
117    pub fn new() -> Self {
118        Self::default()
119    }
120
121    /// Pushes a leaf into self via implicit `AND`, flattening when self is
122    /// already an `All`.
123    fn and_with(self, leaf: proto::TransactionFilter) -> Self {
124        match self.0.filter {
125            None => Self(leaf),
126            Some(proto::transaction_filter::Filter::All(mut all)) => {
127                all.filters.push(leaf);
128                Self(proto::TransactionFilter::default().with_all(all))
129            }
130            Some(_) => {
131                Self(proto::TransactionFilter::default().with_all(
132                    proto::AllTransactionFilter::default().with_filters(vec![self.0, leaf]),
133                ))
134            }
135        }
136    }
137
138    /// Combines self with `leaf` via `OR`, flattening when self is already an
139    /// `Any`.
140    fn or_with(self, leaf: proto::TransactionFilter) -> Self {
141        match self.0.filter {
142            None => Self(leaf),
143            Some(proto::transaction_filter::Filter::Any(mut any)) => {
144                any.filters.push(leaf);
145                Self(proto::TransactionFilter::default().with_any(any))
146            }
147            Some(_) => {
148                Self(proto::TransactionFilter::default().with_any(
149                    proto::AnyTransactionFilter::default().with_filters(vec![self.0, leaf]),
150                ))
151            }
152        }
153    }
154
155    /// Matches transactions of any of the given [`TransactionKind`]s.
156    ///
157    /// Passing an empty iterator produces a filter that matches no
158    /// transactions (it is accepted by the fullnode but yields no results).
159    /// Pass at least one [`TransactionKind`].
160    pub fn kinds(self, kinds: impl IntoIterator<Item = TransactionKind>) -> Self {
161        let transaction_kinds_filter =
162            kinds
163                .into_iter()
164                .fold(proto::TransactionKindsFilter::default(), |mut acc, kind| {
165                    acc.push_kinds(kind);
166                    acc
167                });
168
169        self.and_with(
170            proto::TransactionFilter::default().with_transaction_kinds(transaction_kinds_filter),
171        )
172    }
173
174    /// Matches transactions by execution status.
175    ///
176    /// - `true` for successful transactions.
177    /// - `false` for failed transactions.
178    pub fn execution_status(self, success: bool) -> Self {
179        self.and_with(
180            proto::TransactionFilter::default().with_execution_status(
181                proto::ExecutionStatusFilter::default().with_success(success),
182            ),
183        )
184    }
185
186    /// Matches transactions sent by the given address.
187    pub fn sender(self, address: Address) -> Self {
188        self.and_with(
189            proto::TransactionFilter::default()
190                .with_sender(proto::AddressFilter::default().with_address(address)),
191        )
192    }
193
194    /// Matches transactions whose recipient is the given address.
195    pub fn receiver(self, address: Address) -> Self {
196        self.and_with(
197            proto::TransactionFilter::default()
198                .with_receiver(proto::AddressFilter::default().with_address(address)),
199        )
200    }
201
202    /// Matches transactions that touch the given object id.
203    pub fn affected_object(self, object_id: ObjectId) -> Self {
204        let object_ref = ObjectReference::default().with_object_id(object_id);
205        self.and_with(
206            proto::TransactionFilter::default()
207                .with_affected_object(proto::ObjectIdFilter::default().with_object_ref(object_ref)),
208        )
209    }
210
211    /// Matches transactions that touch the given object id and version.
212    pub fn affected_object_version(self, object_id: ObjectId, version: Version) -> Self {
213        let object_ref = ObjectReference::default()
214            .with_object_id(object_id)
215            .with_version(version.as_u64());
216        self.and_with(
217            proto::TransactionFilter::default()
218                .with_affected_object(proto::ObjectIdFilter::default().with_object_ref(object_ref)),
219        )
220    }
221
222    /// Matches transactions that touch the given object digest.
223    pub fn affected_object_digest(self, object_digest: ObjectDigest) -> Self {
224        let object_ref = ObjectReference::default().with_digest(object_digest);
225        self.and_with(
226            proto::TransactionFilter::default()
227                .with_affected_object(proto::ObjectIdFilter::default().with_object_ref(object_ref)),
228        )
229    }
230
231    /// Matches transactions containing a command that satisfies the given
232    /// [`CommandFilter`].
233    pub fn command(self, filter: CommandFilter) -> Self {
234        self.and_with(proto::TransactionFilter::default().with_command(filter))
235    }
236
237    /// Matches transactions that contain at least one event satisfying the
238    /// given [`EventFilter`].
239    pub fn event(self, filter: EventFilter) -> Self {
240        self.and_with(proto::TransactionFilter::default().with_event(filter))
241    }
242
243    /// Logical `OR` with another filter.
244    pub fn or(self, other: Self) -> Self {
245        self.or_with(other.0)
246    }
247
248    /// Logical `NOT` of this filter.
249    ///
250    /// Equivalent to `!self` (via [`std::ops::Not`]).
251    ///
252    /// # Scoping
253    ///
254    /// `negate` wraps the entire filter accumulated so far, not just the most
255    /// recent leaf. This follows from the chained builder pattern: each step
256    /// transforms `self` into a new accumulated filter, and `negate` always
257    /// operates on whatever `self` represents at that point.
258    ///
259    /// The two snippets below produce different filters because of where
260    /// `negate` is placed in the chain:
261    ///
262    /// ```ignore
263    /// // (NOT kinds) AND sender(alice)
264    /// TransactionFilter::new()
265    ///     .kinds([TransactionKind::Programmable])
266    ///     .negate()
267    ///     .sender(alice);
268    ///
269    /// // NOT (kinds AND sender(alice))
270    /// TransactionFilter::new()
271    ///     .kinds([TransactionKind::Programmable])
272    ///     .sender(alice)
273    ///     .negate();
274    /// ```
275    ///
276    /// For complex expressions where the scoping is not visually obvious,
277    /// build sub-filters as named bindings and combine them with
278    /// [`TransactionFilter::or`].
279    pub fn negate(self) -> Self {
280        Self(
281            proto::TransactionFilter::default()
282                .with_negation(proto::NotTransactionFilter::default().with_filter(self.0)),
283        )
284    }
285}
286
287impl Not for TransactionFilter {
288    type Output = Self;
289    fn not(self) -> Self {
290        self.negate()
291    }
292}

296        value.0
297    }

401/// Filter for events emitted by transactions.
402///
403/// Used as input to [`TransactionFilter::event`] to match transactions that
404/// contain at least one event satisfying this filter.
405///
406/// Built with a chained builder pattern. Start from [`EventFilter::new`] and
407/// add leaves with the named methods (e.g. [`EventFilter::event_type`],
408/// [`EventFilter::sender`], [`EventFilter::emitted_in`]).
409///
410/// Each leaf implicitly `AND`s with everything accumulated so far. Use
411/// [`EventFilter::or`] for `OR` composition and [`EventFilter::negate`] (or the
412/// `!` operator) for `NOT`.
413///
414/// # Example
415///
416/// ```rust
417/// use iota_data_ingestion_core::filters::fullnode::EventFilter;
418///
419/// let filter = EventFilter::new().event_type("0xabcd::my_module::Foo");
420/// ```
421///
422/// More complex composition with OR and NOT
423///
424/// ```rust,ignore
425/// use iota_data_ingestion_core::filters::fullnode::EventFilter;
426///
427/// // NOT (sender == Alice OR sender == Bob)
428/// let filter = EventFilter::new()
429///     .sender(alice)
430///     .or(EventFilter::new().sender(bob))
431///     .negate();
432/// ```
433#[derive(Clone, Debug, Default)]
434pub struct EventFilter(proto::EventFilter);
435
436impl EventFilter {
437    /// Creates an empty filter.
438    ///
439    /// An empty filter is rejected by the fullnode. Add at least one leaf via
440    /// the builder methods (e.g. [`EventFilter::event_type`]) before passing
441    /// the filter to the ingestion framework.
442    ///
443    /// # Example
444    ///
445    /// ```rust
446    /// use iota_data_ingestion_core::filters::fullnode::EventFilter;
447    ///
448    /// let filter = EventFilter::new().event_type("0xabcd::my_module::Foo");
449    /// ```
450    pub fn new() -> Self {
451        Self::default()
452    }
453
454    fn and_with(self, leaf: proto::EventFilter) -> Self {
455        match self.0.filter {
456            None => Self(leaf),
457            Some(proto::event_filter::Filter::All(mut all)) => {
458                all.filters.push(leaf);
459                Self(proto::EventFilter::default().with_all(all))
460            }
461            Some(_) => Self(
462                proto::EventFilter::default()
463                    .with_all(proto::AllEventFilter::default().with_filters(vec![self.0, leaf])),
464            ),
465        }
466    }
467
468    fn or_with(self, leaf: proto::EventFilter) -> Self {
469        match self.0.filter {
470            None => Self(leaf),
471            Some(proto::event_filter::Filter::Any(mut any)) => {
472                any.filters.push(leaf);
473                Self(proto::EventFilter::default().with_any(any))
474            }
475            Some(_) => Self(
476                proto::EventFilter::default()
477                    .with_any(proto::AnyEventFilter::default().with_filters(vec![self.0, leaf])),
478            ),
479        }
480    }
481
482    /// Matches events whose enclosing transaction was sent by the given
483    /// address.
484    pub fn sender(self, address: Address) -> Self {
485        self.and_with(
486            proto::EventFilter::default()
487                .with_sender(proto::AddressFilter::default().with_address(address)),
488        )
489    }
490
491    /// Matches events emitted by a transaction whose top-level `MoveCall`
492    /// targets the given package.
493    ///
494    /// This matches the package the event was *emitted from*,
495    /// not where the event struct is defined. For the latter, use
496    /// [`EventFilter::defined_in`] / [`EventFilter::defined_in_module`].
497    pub fn emitted_in(self, package_id: ObjectId) -> Self {
498        self.and_with(proto::EventFilter::default().with_move_package_and_module(
499            proto::MovePackageAndModuleFilter::default().with_package_id(package_id),
500        ))
501    }
502
503    /// Matches events emitted by a transaction whose top-level `MoveCall`
504    /// targets the given package and module.
505    ///
506    /// This matches the package and module the event was *emitted from*,
507    /// not where the event struct is defined. For the latter, use
508    /// [`EventFilter::defined_in`] / [`EventFilter::defined_in_module`].
509    pub fn emitted_in_module(self, package_id: ObjectId, module: impl Into<String>) -> Self {
510        self.and_with(
511            proto::EventFilter::default().with_move_package_and_module(
512                proto::MovePackageAndModuleFilter::default()
513                    .with_package_id(package_id)
514                    .with_module(module),
515            ),
516        )
517    }
518
519    /// Matches events whose struct is defined in the given package.
520    ///
521    /// This matches the package the event struct is *defined
522    /// in*, not where it was emitted from. For the latter, use
523    /// [`EventFilter::emitted_in`] / [`EventFilter::emitted_in_module`].
524    pub fn defined_in(self, package_id: ObjectId) -> Self {
525        self.and_with(
526            proto::EventFilter::default().with_move_event_package_and_module(
527                proto::MovePackageAndModuleFilter::default().with_package_id(package_id),
528            ),
529        )
530    }
531
532    /// Matches events whose struct is defined in the given package and module.
533    ///
534    /// This matches the package and module the event struct is *defined
535    /// in*, not where it was emitted from. For the latter, use
536    /// [`EventFilter::emitted_in`] / [`EventFilter::emitted_in_module`].
537    pub fn defined_in_module(self, package_id: ObjectId, module: impl Into<String>) -> Self {
538        self.and_with(
539            proto::EventFilter::default().with_move_event_package_and_module(
540                proto::MovePackageAndModuleFilter::default()
541                    .with_package_id(package_id)
542                    .with_module(module),
543            ),
544        )
545    }
546
547    /// Matches events with the given Move event struct tag (e.g.
548    /// `"0xabcd::my_module::Foo"`).
549    pub fn event_type(self, struct_tag: impl Into<String>) -> Self {
550        self.and_with(proto::EventFilter::default().with_move_event_type(
551            proto::MoveEventTypeFilter::default().with_struct_tag(struct_tag),
552        ))
553    }
554
555    /// Logical `OR` with another filter.
556    pub fn or(self, other: Self) -> Self {
557        self.or_with(other.0)
558    }
559
560    /// Logical `NOT` of this filter.
561    ///
562    /// Equivalent to `!self` (via [`std::ops::Not`]).
563    ///
564    /// # Scoping
565    ///
566    /// `negate` wraps the entire filter accumulated so far, not just the most
567    /// recent leaf. This follows from the chained builder pattern: each step
568    /// transforms `self` into a new accumulated filter, and `negate` always
569    /// operates on whatever `self` represents at that point.
570    ///
571    /// The two snippets below produce different filters because of where
572    /// `negate` is placed in the chain:
573    ///
574    /// ```rust,ignore
575    /// use iota_data_ingestion_core::filters::fullnode::EventFilter;
576    ///
577    /// // (NOT event_type Foo) AND sender(alice)
578    /// EventFilter::new()
579    ///     .event_type("0xabcd::my_module::Foo")
580    ///     .negate()
581    ///     .sender(alice);
582    ///
583    /// // NOT (event_type Foo AND sender(alice))
584    /// EventFilter::new()
585    ///     .event_type("0xabcd::my_module::Foo")
586    ///     .sender(alice)
587    ///     .negate();
588    /// ```
589    ///
590    /// For complex expressions where the scoping is not visually obvious,
591    /// build sub-filters as named bindings and combine them with
592    /// [`EventFilter::or`].
593    pub fn negate(self) -> Self {
594        Self(
595            proto::EventFilter::default()
596                .with_negation(proto::NotEventFilter::default().with_filter(self.0)),
597        )
598    }
599}

604        self.negate()
605    }
