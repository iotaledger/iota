44    pub fn generate<R: ::rand::RngCore + ::rand::CryptoRng>(
45        mut rng: R,
46        epoch: EpochId,
47        committee_size: usize,
48    ) -> Self {
49        let validators = (0..committee_size)
50            .map(|_| iota_types::crypto::get_key_pair_from_rng::<AuthorityKeyPair, _>(&mut rng).1)
51            .map(|keypair| (keypair.public().into(), (keypair, 1)))
52            .collect::<HashMap<_, _>>();
53
54        let committee = Committee::new_for_testing_with_normalized_voting_power(
55            epoch,
56            validators
57                .iter()
58                .map(|(name, (_, stake))| (*name, *stake))
59                .collect(),
60        );
61
62        Self {
63            epoch,
64            validators,
65            committee,
66            genesis: None,
67        }
68    }

101    fn create_root_checkpoint(&self) -> (VerifiedCheckpoint, VerifiedCheckpointContents) {
102        assert_eq!(self.epoch, 0, "root checkpoint must be epoch 0");
103
104        let mut contents = empty_contents();
105        if let Some(genesis) = &self.genesis {
106            // if genesis is provided, create checkpoint contents with the genesis
107            // transaction
108            let tx = genesis.transaction().clone();
109            let effects = genesis.effects().clone();
110            let execution_data = ExecutionData::new(tx, effects);
111
112            contents = VerifiedCheckpointContents::new_unchecked(
113                FullCheckpointContents::new_with_causally_ordered_transactions(std::iter::once(
114                    execution_data,
115                )),
116            );
117        }
118
119        let contents_digest = contents
120            .clone()
121            .into_inner()
122            .into_checkpoint_contents()
123            .digest();
124
125        let checkpoint = CheckpointSummary {
126            epoch: 0,
127            sequence_number: 0,
128            network_total_transactions: contents.num_of_transactions() as u64,
129            contents_digest,
130            previous_digest: None,
131            epoch_rolling_gas_cost_summary: Default::default(),
132            end_of_epoch_data: None,
133            timestamp_ms: 0,
134            version_specific_data: bcs::to_bytes(&CheckpointVersionSpecificData::empty_for_tests())
135                .unwrap(),
136            checkpoint_commitments: Default::default(),
137        };
138
139        (self.create_certified_checkpoint(checkpoint), contents)
140    }

142    fn create_certified_checkpoint(&self, checkpoint: CheckpointSummary) -> VerifiedCheckpoint {
143        let signatures = self
144            .validators
145            .iter()
146            .map(|(name, (key, _))| {
147                let intent_msg = IntentMessage::new(
148                    Intent::iota_app(IntentScope::CheckpointSummary),
149                    checkpoint.clone(),
150                );
151                let signature = AuthoritySignature::new_secure(&intent_msg, &checkpoint.epoch, key);
152                AuthoritySignInfo {
153                    epoch: checkpoint.epoch,
154                    authority: *name,
155                    signature,
156                }
157            })
158            .collect();
159
160        let checkpoint = CertifiedCheckpointSummary::new(checkpoint, signatures, self.committee())
161            .unwrap()
162            .try_into_verified(self.committee())
163            .unwrap();
164
165        checkpoint
166    }

184    pub fn make_checkpoints<F: Fn() -> VerifiedCheckpointContents>(
185        &self,
186        number_of_checkpoints: usize,
187        previous_checkpoint: Option<VerifiedCheckpoint>,
188        content_generator: F,
189    ) -> MakeCheckpointResults {
190        // Only skip the first one if it was supplied
191        let skip = previous_checkpoint.is_some() as usize;
192        let first = previous_checkpoint
193            .map(|c| (c, empty_contents()))
194            .unwrap_or_else(|| self.create_root_checkpoint());
195
196        let (ordered_checkpoints, contents): (Vec<_>, Vec<_>) =
197            std::iter::successors(Some(first), |prev| {
198                let contents = content_generator();
199                let contents_digest = contents
200                    .clone()
201                    .into_inner()
202                    .into_checkpoint_contents()
203                    .digest();
204                let summary = CheckpointSummary {
205                    epoch: self.epoch,
206                    sequence_number: prev.0.sequence_number + 1,
207                    network_total_transactions: prev.0.network_total_transactions
208                        + contents.num_of_transactions() as u64,
209                    contents_digest,
210                    previous_digest: Some(*prev.0.digest()),
211                    epoch_rolling_gas_cost_summary: Default::default(),
212                    end_of_epoch_data: None,
213                    timestamp_ms: 0,
214                    version_specific_data: bcs::to_bytes(
215                        &CheckpointVersionSpecificData::empty_for_tests(),
216                    )
217                    .unwrap(),
218                    checkpoint_commitments: Default::default(),
219                };
220
221                let checkpoint = self.create_certified_checkpoint(summary);
222
223                Some((checkpoint, contents))
224            })
225            .skip(skip)
226            .take(number_of_checkpoints)
227            .unzip();
228
229        let (sequence_number_to_digest, checkpoints) = ordered_checkpoints
230            .iter()
231            .cloned()
232            .map(|checkpoint| {
233                let digest = *checkpoint.digest();
234                ((checkpoint.sequence_number, digest), (digest, checkpoint))
235            })
236            .unzip();
237
238        (
239            ordered_checkpoints,
240            contents,
241            sequence_number_to_digest,
242            checkpoints,
243        )
244    }

246    /// Builds a chain of empty checkpoints assigning each the given timestamp
247    /// (in order), for tests that need control over checkpoint timestamps.
248    pub fn make_checkpoints_with_timestamps(
249        &self,
250        timestamps_ms: &[CheckpointTimestamp],
251        previous_checkpoint: Option<VerifiedCheckpoint>,
252    ) -> Vec<VerifiedCheckpoint> {
253        let mut prev = previous_checkpoint.unwrap_or_else(|| self.create_root_checkpoint().0);
254        let mut checkpoints = Vec::with_capacity(timestamps_ms.len());
255        for &timestamp_ms in timestamps_ms {
256            let contents_digest = empty_contents()
257                .into_inner()
258                .into_checkpoint_contents()
259                .digest();
260            let summary = CheckpointSummary {
261                epoch: self.epoch,
262                sequence_number: prev.sequence_number + 1,
263                network_total_transactions: prev.network_total_transactions,
264                contents_digest,
265                previous_digest: Some(*prev.digest()),
266                epoch_rolling_gas_cost_summary: Default::default(),
267                end_of_epoch_data: None,
268                timestamp_ms,
269                version_specific_data: bcs::to_bytes(
270                    &CheckpointVersionSpecificData::empty_for_tests(),
271                )
272                .unwrap(),
273                checkpoint_commitments: Default::default(),
274            };
275            let checkpoint = self.create_certified_checkpoint(summary);
276            prev = checkpoint.clone();
277            checkpoints.push(checkpoint);
278        }
279        checkpoints
280    }

282    pub fn make_end_of_epoch_checkpoint(
283        &self,
284        previous_checkpoint: VerifiedCheckpoint,
285        end_of_epoch_data: Option<EndOfEpochData>,
286    ) -> (
287        CheckpointSequenceNumber,
288        CheckpointDigest,
289        VerifiedCheckpoint,
290    ) {
291        let summary = CheckpointSummary {
292            epoch: self.epoch,
293            sequence_number: previous_checkpoint.sequence_number + 1,
294            network_total_transactions: 0,
295            contents_digest: empty_contents()
296                .into_inner()
297                .into_checkpoint_contents()
298                .digest(),
299            previous_digest: Some(*previous_checkpoint.digest()),
300            epoch_rolling_gas_cost_summary: Default::default(),
301            end_of_epoch_data,
302            timestamp_ms: 0,
303            version_specific_data: bcs::to_bytes(&CheckpointVersionSpecificData::empty_for_tests())
304                .unwrap(),
305            checkpoint_commitments: Default::default(),
306        };
307
308        let checkpoint = self.create_certified_checkpoint(summary);
309
310        (checkpoint.sequence_number, *checkpoint.digest(), checkpoint)
311    }
