437/// Starts a local network with the given configuration.
438async fn start(
439    config_dir: Option<PathBuf>,
440    with_faucet: Option<String>,
441    faucet_amount: Option<u64>,
442    faucet_coin_count: Option<usize>,
443    with_grpc: Option<String>,
444    #[cfg(feature = "indexer")] indexer_feature_args: IndexerFeatureArgs,
445    force_regenesis: bool,
446    epoch_duration_ms: Option<u64>,
447    fullnode_rpc_port: u16,
448    #[cfg(feature = "indexer")] mut data_ingestion_dir: Option<PathBuf>,
449    no_full_node: bool,
450    disable_fullnode_pruning: bool,
451    node_config_override: Vec<String>,
452    committee_size: Option<usize>,
453    write_config: Option<PathBuf>,
454) -> Result<(), anyhow::Error> {
455    // Parsed here rather than by clap, whose error would echo the whole
456    // argument. These errors name the path only.
457    let mut node_config_overrides = node_config_override
458        .iter()
459        .map(|input| input.parse::<NodeConfigOverride>())
460        .collect::<Result<Vec<_>, _>>()?;
461
462    if force_regenesis {
463        ensure!(
464            config_dir.is_none(),
465            "Cannot pass `--force-regenesis` and `--network.config` at the same time."
466        );
467    }
468
469    if with_grpc.is_some() {
470        ensure!(!no_full_node, "Cannot enable gRPC without a fullnode.");
471    }
472
473    if write_config.is_some() {
474        ensure!(
475            with_faucet.is_none(),
476            "Cannot pass `--with-faucet` and `--write-config` at the same time: the faucet is a \
477             service, and `--write-config` starts nothing."
478        );
479        ensure!(
480            !force_regenesis,
481            "Cannot pass `--force-regenesis` and `--write-config` at the same time: the written \
482             configs would point at a temporary directory that is removed on exit. Pass \
483             `--network.config <DIR>` or run `iota-localnet genesis` first."
484        );
485        #[cfg(feature = "indexer")]
486        ensure!(
487            data_ingestion_dir.is_some()
488                || (indexer_feature_args.with_indexer.is_none()
489                    && indexer_feature_args.with_graphql.is_none()),
490            "Cannot pass `--with-indexer` or `--with-graphql` and `--write-config` at the same \
491             time: the written fullnode config would point at a temporary data ingestion \
492             directory that is removed on exit. Pass `--data-ingestion-dir <DIR>`."
493        );
494    }
495
496    #[cfg(feature = "indexer")]
497    let IndexerFeatureArgs {
498        mut with_indexer,
499        with_graphql,
500        graphql_metrics_address,
501        pg_port,
502        pg_host,
503        pg_db_name,
504        pg_user,
505        pg_password,
506        pruning_options,
507    } = indexer_feature_args;
508
509    #[cfg(feature = "indexer")]
510    if with_graphql.is_some() {
511        with_indexer = Some(with_indexer.unwrap_or_default());
512    }
513
514    #[cfg(feature = "indexer")]
515    if with_indexer.is_some() {
516        ensure!(
517            !no_full_node,
518            "Cannot start the indexer without a fullnode."
519        );
520    }
521
522    // The service addresses are resolved here rather than where each service
523    // starts. An unparsable one then fails before genesis, and the port check
524    // knows every address this run binds.
525    let faucet_address = with_faucet
526        .map(|input| service_address(input, DEFAULT_FAUCET_PORT, "faucet"))
527        .transpose()?;
528    // The faucet builds its config from an `Ipv4Addr`.
529    ensure!(
530        faucet_address.is_none_or(|address| address.is_ipv4()),
531        "Invalid faucet host and port: the faucet needs an IPv4 address"
532    );
533    #[cfg(feature = "indexer")]
534    let (indexer_address, graphql_address, graphql_metrics_address) = {
535        let indexer_address = with_indexer
536            .map(|input| service_address(input, DEFAULT_INDEXER_PORT, "indexer"))
537            .transpose()?;
538        let graphql_address = with_graphql
539            .map(|input| service_address(input, DEFAULT_GRAPHQL_PORT, "graphql"))
540            .transpose()?;
541        // The GraphQL service binds a Prometheus listener beside its main
542        // port. Its own default is the fullnode metrics port, so the address
543        // is pinned here, where the port check also reads it.
544        let graphql_metrics_address = graphql_address
545            .map(|_| -> Result<SocketAddr, anyhow::Error> {
546                let address = parse_host_port_with_default_host(
547                    graphql_metrics_address.unwrap_or_default(),
548                    &Ipv4Addr::LOCALHOST.to_string(),
549                    DEFAULT_GRAPHQL_METRICS_PORT,
550                )
551                .map_err(|_| anyhow!("Invalid graphql metrics host and port"))?;
552                // The service rebuilds the address as `host:port`, which an
553                // IPv6 host needs brackets to survive.
554                ensure!(
555                    address.is_ipv4(),
556                    "graphql metrics configuration requires an IPv4 address"
557                );
558                Ok(address)
559            })
560            .transpose()?;
561        (indexer_address, graphql_address, graphql_metrics_address)
562    };
563
564    if epoch_duration_ms.is_some() && genesis_blob_exists(config_dir.clone()) && !force_regenesis {
565        bail!(
566            "epoch duration can only be set when passing the `--force-regenesis` flag, or when \
567            there is no genesis configuration in the default IOTA configuration folder or the given \
568            network.config argument.",
569        );
570    }
571
572    // Resolve the configuration directory.
573    let config_path = config_dir.clone().map_or_else(iota_config_dir, Ok)?;
574
575    // Deprecated --with-grpc becomes node config overrides. They are
576    // prepended so that explicit --node-config-override values win.
577    if let Some(input) = with_grpc.clone() {
578        eprintln!(
579            "{}",
580            "[warning] The --with-grpc flag is deprecated. Use `--node-config-override \
581             fullnode:enable-grpc-api=true`, which serves the API at the default address, \
582             and add `--node-config-override fullnode:grpc-api-config={address: <HOST:PORT>}` \
583             to serve it elsewhere."
584                .yellow()
585                .bold()
586        );
587        node_config_overrides.splice(0..0, with_grpc_overrides(input)?);
588    }
589
590    let mut swarm_builder = Swarm::builder();
591    // The entry the fullnode's config is derived from. A persisted network
592    // keeps it, so that the fullnode's key pairs, ports and db path are the
593    // same on every run.
594    let mut fullnode_config_info = None;
595
596    if disable_fullnode_pruning {
597        swarm_builder = swarm_builder.with_disable_fullnode_pruning();
598    }
599
600    // If this is set, then no data will be persisted between runs, and a new
601    // genesis will be generated each run.
602    if force_regenesis {
603        let committee_size = NonZeroUsize::new(committee_size.unwrap_or(DEFAULT_COMMITTEE_SIZE))
604            .ok_or_else(|| anyhow!("Committee size must be at least 1."))?;
605
606        swarm_builder = swarm_builder
607            .committee_size(committee_size)
608            .with_deterministic_validator_ports(VALIDATOR_PORT_BASE);
609        let genesis_config = GenesisConfig::custom_genesis(1, 100);
610        swarm_builder = swarm_builder.with_genesis_config(genesis_config);
611        let epoch_duration_ms = epoch_duration_ms.unwrap_or(DEFAULT_EPOCH_DURATION_MS);
612        swarm_builder = swarm_builder.with_epoch_duration_ms(epoch_duration_ms);
613    } else {
614        let network_config_path = config_path.join(IOTA_NETWORK_CONFIG);
615        // Auto genesis if no configuration exists in the configuration directory.
616        if !network_config_path.exists() {
617            if !config_path.exists() {
618                fs::create_dir(&config_path).map_err(|err| {
619                    anyhow!(err).context(format!(
620                        "Cannot create network config dir {}",
621                        config_path.display()
622                    ))
623                })?;
624            }
625            genesis(
626                None,
627                None,
628                Some(config_path.clone()),
629                false,
630                epoch_duration_ms,
631                None,
632                None,
633                false,
634                committee_size.unwrap_or(DEFAULT_COMMITTEE_SIZE),
635                None,
636                None,
637            )
638            .await
639            .map_err(|e| anyhow!("{e}: {}. \n\n\
640            If you are trying to run a local network without persisting the data (so a new genesis that is \
641            randomly generated and will not be saved once the network is shut down), use --force-regenesis flag. \n\
642            If you are trying to persist the network data and start from a new genesis, use iota-localnet genesis --help \
643            to see how to generate a new genesis.", config_path.display()))?;
644        } else if committee_size.is_some() {
645            eprintln!(
646                "{}",
647                "[warning] The committee-size arg will be ignored as a network configuration \
648                        already exists. To change the committee size, you'll have to adjust the \
649                        network configuration file or regenerate a genesis with the desired \
650                        committee size. See `iota-localnet genesis --help` for more information."
651                    .yellow()
652                    .bold()
653            );
654        }
655
656        let mut persisted_network_config = PersistedNetworkConfig::read(&config_path)?;
657        fullnode_config_info = persisted_network_config
658            .genesis_config
659            .fullnode_config_info
660            .take();
661        // Without the persisted entry the fullnode would get a fresh random
662        // identity, and so a fresh database, on every run.
663        ensure!(
664            no_full_node || fullnode_config_info.is_some(),
665            "the network config in {} has no fullnode entry. Re-create the configuration with `iota-localnet genesis --force`.",
666            config_path.display()
667        );
668        // The genesis blob is read, never rebuilt: it is the network's
669        // identity, and the node configs only point at it.
670        let genesis = Genesis::new_from_file(config_path.join(IOTA_GENESIS_FILENAME));
671        let network_config = persisted_network_config.into_network_config(&config_path, genesis)?;
672
673        swarm_builder = swarm_builder
674            .dir(config_path.clone())
675            .with_network_config(network_config);
676    }
677
678    // A derived node config is a real node's, so it gets the default
679    // denial-of-service protection the swarm builders leave unset. Set in both
680    // modes, and before the overrides, so `policy-config=null` still clears it.
681    let policy_config = PolicyConfig::default_dos_protection_policy();
682    swarm_builder = swarm_builder
683        .with_validator_policy_config(Some(policy_config.clone()))
684        .with_fullnode_policy_config(Some(policy_config));
685
686    // the indexer and GraphQL services communicate with the fullnode via gRPC, we
687    // must enable it by default.
688    #[cfg(feature = "indexer")]
689    if indexer_address.is_some() || graphql_address.is_some() {
690        // The gRPC API config is given rather than left out. The builder
691        // would otherwise put the API on a free port, which differs on every
692        // run. A `--node-config-override` still wins over it.
693        swarm_builder = swarm_builder
694            .with_fullnode_enable_grpc_api(true)
695            .with_fullnode_grpc_api_config(GrpcApiConfig::default());
696    }
697
698    // the indexer requires to set the fullnode's data ingestion directory
699    // note that this overrides the default configuration that is set when running
700    // the genesis command, which sets data_ingestion_dir to None.
701    //
702    // The directory is owned here until the network launches, so that
703    // `--write-config`, which starts nothing, leaves none behind.
704    #[cfg(feature = "indexer")]
705    let data_ingestion_tempdir = if indexer_address.is_some() && data_ingestion_dir.is_none() {
706        let tempdir = tempdir()?;
707        data_ingestion_dir = Some(tempdir.path().to_path_buf());
708        Some(tempdir)
709    } else {
710        None
711    };
712
713    #[cfg(feature = "indexer")]
714    if let Some(ref dir) = data_ingestion_dir {
715        swarm_builder = swarm_builder.with_data_ingestion_dir(dir.clone());
716    }
717
718    check_fullnode_override_scopes(&node_config_overrides, !no_full_node)?;
719    swarm_builder = swarm_builder.with_node_config_overrides(node_config_overrides);
720
721    let mut fullnode_url = iota_config::node::default_json_rpc_address();
722    fullnode_url.set_port(fullnode_rpc_port);
723
724    if no_full_node {
725        swarm_builder = swarm_builder.with_fullnode_count(0);
726    } else {
727        swarm_builder = swarm_builder
728            .with_fullnode_count(1)
729            .with_fullnode_rpc_addr(fullnode_url)
730            .with_fullnode_genesis_config(
731                fullnode_config_info.unwrap_or_else(|| fullnode_genesis_config(&mut OsRng, None)),
732            );
733    }
734
735    let mut swarm = tokio::task::spawn_blocking(move || swarm_builder.try_build()).await??;
736    log_applied_node_config_overrides(&swarm);
737
738    if let Some(directory) = write_config {
739        return write_node_configs(&swarm, &directory);
740    }
741
742    let mut addresses = addresses_bound_at_launch(&swarm);
743    if let Some(faucet_address) = faucet_address {
744        addresses.push(BoundAddress::service(
745            "faucet",
746            "--with-faucet",
747            faucet_address,
748        ));
749    }
750    #[cfg(feature = "indexer")]
751    {
752        if let Some(indexer_address) = indexer_address {
753            addresses.push(BoundAddress::service(
754                "indexer",
755                "--with-indexer",
756                indexer_address,
757            ));
758        }
759        if let Some(graphql_address) = graphql_address {
760            addresses.push(BoundAddress::service(
761                "GraphQL",
762                "--with-graphql",
763                graphql_address,
764            ));
765        }
766        if let Some(graphql_metrics_address) = graphql_metrics_address {
767            addresses.push(BoundAddress::service(
768                "GraphQL metrics",
769                "--graphql-metrics-address",
770                graphql_metrics_address,
771            ));
772        }
773    }
774    check_ports_are_free(&addresses)?;
775
776    // The fullnode writes to the data ingestion directory for as long as it
777    // runs, so it outlives this function from here on.
778    #[cfg(feature = "indexer")]
779    if let Some(tempdir) = data_ingestion_tempdir {
780        let _ = tempdir.keep();
781    }
782
783    swarm.launch().await?;
784    // Let nodes connect to one another
785    tokio::time::sleep(tokio::time::Duration::from_secs(2)).await;
786    info!("Cluster started");
787
788    // Taken off the built config rather than off `--fullnode-rpc-port`, since an
789    // override can move the address.
790    let fullnode_url = swarm
791        .fullnodes()
792        .next()
793        .map(|node| node.config().json_rpc_address)
794        .unwrap_or(fullnode_url);
795    // the indexer requires a fullnode url with protocol specified
796    let fullnode_url = format!("http://{fullnode_url}");
797    info!("Fullnode URL: {}", fullnode_url);
798
799    // Reported off the built config rather than off `--with-grpc`, since an
800    // override can turn the API on too.
801    if let Some(grpc_url) = swarm.fullnodes().next().and_then(|node| {
802        let config = node.config();
803        config.enable_grpc_api.then(|| {
804            config
805                .grpc_api_config
806                .as_ref()
807                .map(|grpc| grpc.address)
808                .unwrap_or_else(|| GrpcApiConfig::default().address)
809        })
810    }) {
811        info!("gRPC URL: http://{grpc_url}");
812    }
813
814    #[cfg(feature = "indexer")]
815    let pg_address = format!("postgres://{pg_user}:{pg_password}@{pg_host}:{pg_port}/{pg_db_name}");
816
817    #[cfg(feature = "indexer")]
818    let fullnode_grpc_url = {
819        let socket_addr = swarm
820            .fullnodes()
821            .next()
822            .and_then(|node| {
823                node.config()
824                    .grpc_api_config
825                    .as_ref()
826                    .map(|grpc| grpc.address)
827            })
828            .unwrap_or_else(|| GrpcApiConfig::default().address);
829        format!("http://{socket_addr}")
830    };
831
832    #[cfg(feature = "indexer")]
833    if let Some(indexer_address) = indexer_address {
834        tracing::info!("Starting the indexer service at {indexer_address}");
835        // Start in writer mode
836        start_test_indexer(
837            pg_address.clone(),
838            // reset the existing db
839            true,
840            None,
841            fullnode_grpc_url.clone(),
842            IndexerTypeConfig::writer_mode(Some(pruning_options)),
843            data_ingestion_dir.clone(),
844        )
845        .await;
846        info!("Indexer in writer mode started");
847
848        // Start in reader mode
849        start_test_indexer(
850            pg_address.clone(),
851            false,
852            None,
853            fullnode_grpc_url.clone(),
854            IndexerTypeConfig::reader_mode(indexer_address.to_string()),
855            data_ingestion_dir.clone(),
856        )
857        .await;
858        info!("Indexer in reader mode started");
859
860        // Start in analytical worker mode
861        start_test_indexer(
862            pg_address.clone(),
863            false,
864            None,
865            fullnode_grpc_url.clone(),
866            IndexerTypeConfig::AnalyticalWorker,
867            data_ingestion_dir,
868        )
869        .await;
870        info!("Indexer in analytical worker mode started");
871    }
872
873    #[cfg(feature = "indexer")]
874    if let Some(graphql_address) = graphql_address {
875        tracing::info!("Starting the GraphQL service at {graphql_address}");
876        let graphql_metrics_address =
877            graphql_metrics_address.expect("resolved with the GraphQL address");
878        tracing::info!("Serving the GraphQL metrics at {graphql_metrics_address}");
879        let graphql_connection_config = ConnectionConfig {
880            port: graphql_address.port(),
881            // The server joins host and port with `format!`, so an IPv6 host
882            // needs its brackets.
883            host: graphql_host(graphql_address),
884            db_url: pg_address,
885            prom_host: graphql_host(graphql_metrics_address),
886            prom_port: graphql_metrics_address.port(),
887            ..Default::default()
888        };
889        start_graphql_server_with_fn_rpc(
890            graphql_connection_config,
891            Some(fullnode_grpc_url),
892            None, // it will be initialized by default
893            None, // resolves to default service config
894        )
895        .await;
896        info!("GraphQL started");
897    }
898
899    if let Some(faucet_address) = faucet_address {
900        tracing::info!("Starting the faucet service at {faucet_address}");
901        let faucet_config_dir = if force_regenesis {
902            // tempdir is used so the faucet file is cleaned up afterwards
903            tempdir()?.keep()
904        } else {
905            config_path
906        };
907
908        let host_ip = match faucet_address {
909            SocketAddr::V4(addr) => *addr.ip(),
910            // Rejected where the address is parsed, before genesis.
911            _ => bail!("faucet configuration requires an IPv4 address"),
912        };
913
914        let config = FaucetConfig {
915            host_ip,
916            port: faucet_address.port(),
917            num_coins: faucet_coin_count.unwrap_or(DEFAULT_FAUCET_NUM_COINS),
918            amount: faucet_amount.unwrap_or(DEFAULT_FAUCET_NANOS_AMOUNT),
919            ..Default::default()
920        };
921
922        let prometheus_registry = prometheus_filtered::Registry::new();
923        if force_regenesis {
924            let kp = swarm.config_mut().account_keys.swap_remove(0);
925            let keystore_path = faucet_config_dir.join(IOTA_KEYSTORE_FILENAME);
926            let mut keystore = Keystore::from(FileBasedKeystore::new(&keystore_path).unwrap());
927            let kp = SimpleKeypair::from(kp);
928            let address: Address = kp.public_key().derive_address();
929            keystore.add_key(None, kp).unwrap();
930            IotaClientConfig::new(keystore)
931                .with_envs([IotaEnv::new("localnet", fullnode_url)])
932                .with_active_address(address)
933                .with_active_env("localnet".to_string())
934                .persisted(faucet_config_dir.join(IOTA_CLIENT_CONFIG).as_path())
935                .save()
936                .unwrap();
937        }
938        let faucet_wal = faucet_config_dir.join("faucet.wal");
939        let simple_faucet = SimpleFaucet::new(
940            create_wallet_context(config.wallet_client_timeout_secs, faucet_config_dir)?,
941            &prometheus_registry,
942            faucet_wal.as_path(),
943            config.clone(),
944        )
945        .await
946        .unwrap();
947
948        let app_state = Arc::new(AppState {
949            faucet: simple_faucet,
950            config,
951        });
952
953        start_faucet(app_state, CONCURRENCY_LIMIT, &prometheus_registry).await?;
954    }
955
956    let mut interval = tokio::time::interval(std::time::Duration::from_secs(3));
957    let mut unhealthy_cnt = 0;
958    loop {
959        for node in swarm.validator_nodes() {
960            if let Err(err) = node.health_check(true).await {
961                unhealthy_cnt += 1;
962                if unhealthy_cnt > 3 {
963                    // The network could temporarily go down during reconfiguration.
964                    // If we detect a failed validator 3 times in a row, give up.
965                    return Err(err.into());
966                }
967                // Break the inner loop so that we could retry latter.
968                break;
969            } else {
970                unhealthy_cnt = 0;
971            }
972        }
973
974        interval.tick().await;
975    }
976}
977
978async fn genesis(
979    from_config: Option<PathBuf>,
980    write_config: Option<PathBuf>,
981    working_dir: Option<PathBuf>,
982    force: bool,
983    epoch_duration_ms: Option<u64>,
984    chain_start_timestamp_ms: Option<u64>,
985    benchmark_ips: Option<Vec<String>>,
986    with_faucet: bool,
987    committee_size: usize,
988    num_additional_gas_accounts: Option<usize>,
989    admin_interface_address: Option<String>,
990) -> Result<(), anyhow::Error> {
991    let iota_config_dir = &match working_dir {
992        // if a directory is specified, it must exist (it
993        // will not be created)
994        Some(v) => v,
995        // create default IOTA config dir if not specified
996        // on the command line and if it does not exist
997        // yet
998        None => iota_config_dir()?,
999    };
1000
1001    // if IOTA config dir is not empty then either clean it
1002    // up (if --force/-f option was specified or report an
1003    // error
1004    let dir = iota_config_dir.read_dir().map_err(|err| {
1005        anyhow!(err).context(format!("Cannot open IOTA config dir {iota_config_dir:?}"))
1006    })?;
1007    let files = dir.collect::<Result<Vec<_>, _>>()?;
1008
1009    let client_path = iota_config_dir.join(IOTA_CLIENT_CONFIG);
1010    let keystore_path = iota_config_dir.join(IOTA_KEYSTORE_FILENAME);
1011
1012    if write_config.is_none() && !files.is_empty() {
1013        if force {
1014            // check old keystore and client.yaml is compatible
1015            let is_compatible = FileBasedKeystore::new(&keystore_path).is_ok()
1016                && PersistedConfig::<IotaClientConfig>::read(&client_path).is_ok();
1017            // Keep keystore and client.yaml if they are compatible
1018            if is_compatible {
1019                for file in files {
1020                    let path = file.path();
1021                    if path != client_path && path != keystore_path {
1022                        if path.is_file() {
1023                            fs::remove_file(path)
1024                        } else {
1025                            fs::remove_dir_all(path)
1026                        }
1027                        .map_err(|err| {
1028                            anyhow!(err)
1029                                .context(format!("Cannot remove file {}", file.path().display()))
1030                        })?;
1031                    }
1032                }
1033            } else {
1034                fs::remove_dir_all(iota_config_dir).map_err(|err| {
1035                    anyhow!(err).context(format!(
1036                        "Cannot remove IOTA config dir {}",
1037                        iota_config_dir.display()
1038                    ))
1039                })?;
1040                fs::create_dir(iota_config_dir).map_err(|err| {
1041                    anyhow!(err).context(format!(
1042                        "Cannot create IOTA config dir {}",
1043                        iota_config_dir.display()
1044                    ))
1045                })?;
1046            }
1047        } else if files.len() != 2 || !client_path.exists() || !keystore_path.exists() {
1048            bail!(
1049                "Cannot run genesis with non-empty IOTA config directory {}. \n
1050                Please use the --force/-f option to remove the existing configuration",
1051                iota_config_dir.display()
1052            );
1053        }
1054    }
1055
1056    let network_path = iota_config_dir.join(IOTA_NETWORK_CONFIG);
1057    let genesis_path = iota_config_dir.join(IOTA_GENESIS_FILENAME);
1058
1059    let mut genesis_conf = match from_config {
1060        Some(path) => PersistedConfig::read(&path)?,
1061        None => {
1062            if let Some(ips) = benchmark_ips {
1063                // Make a keystore containing the key for the genesis gas object.
1064                let path = iota_config_dir.join(IOTA_BENCHMARK_GENESIS_GAS_KEYSTORE_FILENAME);
1065                let mut keystore = FileBasedKeystore::new(&path)?;
1066                let num_validators = ips.len();
1067                let num_accounts = num_validators + num_additional_gas_accounts.unwrap_or(0);
1068                for gas_key in GenesisConfig::benchmark_gas_keys(num_accounts) {
1069                    keystore.add_key(None, gas_key)?;
1070                }
1071                keystore.save()?;
1072
1073                // Calculate extra allocations (validator, faucet)
1074                let validator_low_stake_threshold = GenesisConfig::default()
1075                    .protocol_config()
1076                    .validator_low_stake_threshold();
1077                let validator_extra = num_validators as u64
1078                    * (iota_swarm_config::genesis_config::DEFAULT_GAS_AMOUNT
1079                        + validator_low_stake_threshold);
1080                let mut faucet_extra = 0u64;
1081                if with_faucet {
1082                    faucet_extra = iota_swarm_config::genesis_config::DEFAULT_GAS_AMOUNT
1083                        * iota_swarm_config::genesis_config::DEFAULT_NUMBER_OF_OBJECT_PER_ACCOUNT
1084                            as u64;
1085                }
1086                // `u64::MAX - 1` is the max total supply value acceptable by
1087                // `iota::balance::increase_supply`
1088                let total_available_amount = (u64::MAX - 1)
1089                    .saturating_sub(validator_extra)
1090                    .saturating_sub(faucet_extra);
1091
1092                // Make a new genesis config from the provided ip addresses with given epoch
1093                // duration and timestamp.
1094                GenesisConfig::new_for_benchmarks(
1095                    &ips,
1096                    epoch_duration_ms,
1097                    chain_start_timestamp_ms,
1098                    num_additional_gas_accounts,
1099                    total_available_amount,
1100                )
1101            } else if keystore_path.exists() {
1102                let existing_keys = FileBasedKeystore::new(&keystore_path)?.addresses();
1103                GenesisConfig::for_local_testing_with_addresses(existing_keys)
1104            } else {
1105                GenesisConfig::for_local_testing()
1106            }
1107        }
1108    };
1109
1110    // Adds an extra faucet account to the genesis
1111    if with_faucet {
1112        info!("Adding faucet account in genesis config...");
1113        genesis_conf = genesis_conf.add_faucet_account();
1114    }
1115
1116    if let Some(path) = write_config {
1117        let persisted = genesis_conf.persisted(&path);
1118        persisted.save()?;
1119        return Ok(());
1120    }
1121
1122    let validator_info = genesis_conf.validator_config_info.take();
1123    let ssfn_info = genesis_conf.ssfn_config_info.take();
1124    // A genesis config that names its validators is a deployment's, and the
1125    // node config files of that deployment are written below. A plain `genesis`
1126    // names none and writes none: the configs a local network runs come from
1127    // `start --write-config`.
1128    let write_deployment_configs = validator_info.is_some();
1129
1130    if let Some(epoch_duration_ms) = epoch_duration_ms {
1131        genesis_conf.parameters.epoch_duration_ms = epoch_duration_ms;
1132    }
1133
1134    let admin_interface_address_with_port = admin_interface_address
1135        .map(|input| {
1136            let default_port = iota_config::node::default_admin_interface_address().port();
1137            service_address(input, default_port, "admin interface")
1138        })
1139        .transpose()?;
1140
1141    let mut builder = ConfigBuilder::new(iota_config_dir)
1142        .with_genesis_config(genesis_conf)
1143        .with_empty_validator_genesis();
1144    builder = if let Some(validators) = validator_info {
1145        builder.with_validators(validators)
1146    } else {
1147        builder
1148            .committee_size(NonZeroUsize::new(committee_size).unwrap())
1149            .with_deterministic_ports(VALIDATOR_PORT_BASE)
1150    };
1151
1152    if let Some(address) = admin_interface_address_with_port {
1153        builder = builder.with_admin_interface_address(address);
1154    }
1155
1156    let (network_config, mut genesis_config) =
1157        tokio::task::spawn_blocking(move || builder.build_with_genesis_config()).await?;
1158    let mut keystore = FileBasedKeystore::new(&keystore_path)?;
1159    for key in &network_config.account_keys {
1160        keystore.add_key(None, SimpleKeypair::from(key.clone()))?;
1161    }
1162    let active_address = keystore.addresses().pop();
1163
1164    let NetworkConfig {
1165        validator_configs,
1166        account_keys,
1167        genesis,
1168    } = network_config;
1169    genesis.save(&genesis_path)?;
1170    let genesis = Genesis::new_from_file(&genesis_path);
1171
1172    // The fullnode is not part of the genesis committee, but its entry is
1173    // persisted with the validators'. `start` then derives the same fullnode
1174    // config, and reuses its database, on every run.
1175    genesis_config.fullnode_config_info.get_or_insert_with(|| {
1176        fullnode_genesis_config(&mut OsRng, admin_interface_address_with_port)
1177    });
1178
1179    info!("Network genesis completed.");
1180    PersistedNetworkConfig {
1181        version: PersistedNetworkConfig::VERSION,
1182        genesis_config,
1183        account_keys,
1184    }
1185    .save(&network_path)?;
1186    info!("Network config file is stored in {:?}.", network_path);
1187
1188    info!("Client keystore is stored in {:?}.", keystore_path);
1189
1190    if write_deployment_configs {
1191        let ssfn_seed_peers = match ssfn_info {
1192            Some(ssfn_info) => write_state_sync_fullnode_configs(
1193                iota_config_dir,
1194                ssfn_info,
1195                &validator_configs,
1196                admin_interface_address_with_port,
1197            )?,
1198            None => Vec::new(),
1199        };
1200        write_validator_configs(
1201            iota_config_dir,
1202            validator_configs,
1203            &genesis,
1204            &ssfn_seed_peers,
1205        )?;
1206    }
1207
1208    let mut client_config = if client_path.exists() {
1209        PersistedConfig::read(&client_path)?
1210    } else {
1211        IotaClientConfig::new(keystore).with_default_envs()
1212    };
1213
1214    if client_config.active_address().is_none() {
1215        client_config.set_active_address(active_address);
1216    }
1217
1218    // On windows, using 0.0.0.0 will usually yield in an networking error. This
1219    // localnet ip address must bind to 127.0.0.1 if the default 0.0.0.0 is
1220    // used.
1221    let json_rpc_address = iota_config::node::default_json_rpc_address();
1222    let localnet_ip = if json_rpc_address.ip() == IpAddr::V4(Ipv4Addr::new(0, 0, 0, 0)) {
1223        "127.0.0.1".to_string()
1224    } else {
1225        json_rpc_address.ip().to_string()
1226    };
1227    client_config.set_env(IotaEnv::new(
1228        "localnet",
1229        format!("http://{}:{}", localnet_ip, json_rpc_address.port()),
1230    ));
1231    client_config.add_env(IotaEnv::devnet());
1232
1233    if client_config.active_env().is_none() {
1234        client_config.set_active_env(client_config.envs().first().map(|env| env.alias().clone()));
1235    }
1236
1237    client_config.save(&client_path)?;
1238    info!("Client config file is stored in {:?}.", client_path);
1239
1240    Ok(())
1241}

1266/// Write a node config file per state sync fullnode entry, and return the seed
1267/// peers the validators reach them through.
1268///
1269/// These are templates for a deployment, not configs a local network runs:
1270/// their paths and addresses are the ones a packaged node uses.
1271fn write_state_sync_fullnode_configs(
1272    config_directory: &Path,
1273    ssfn_info: Vec<SsfnGenesisConfig>,
1274    validator_configs: &[NodeConfig],
1275    admin_interface_address: Option<SocketAddr>,
1276) -> Result<Vec<SeedPeer>, anyhow::Error> {
1277    let mut ssfn_configs = vec![];
1278    for (index, ssfn) in ssfn_info.into_iter().enumerate() {
1279        let path = config_directory.join(iota_config::ssfn_config_file(
1280            ssfn.p2p_address.clone(),
1281            index,
1282        ));
1283        // join base fullnode config with each SsfnGenesisConfig entry
1284        let deployed_genesis = Genesis::new_from_file("/opt/iota/config/genesis.blob");
1285        let ssfn_config = FullnodeConfigBuilder::new()
1286            .with_config_directory(config_directory.to_path_buf())
1287            .with_p2p_external_address(ssfn.p2p_address)
1288            .with_network_key_pair(ssfn.network_key_pair)
1289            .with_p2p_listen_address(([0, 0, 0, 0], 8084))
1290            .with_db_path(PathBuf::from("/opt/iota/db/authorities_db/full_node_db"))
1291            .with_network_address("/ip4/0.0.0.0/tcp/8080/http".parse()?)
1292            .with_metrics_address(([0, 0, 0, 0], 9184))
1293            .with_admin_interface_address(admin_interface_address)
1294            .with_json_rpc_address(([0, 0, 0, 0], 9000))
1295            .with_genesis(deployed_genesis.clone())
1296            .with_policy_config(Some(PolicyConfig::default_dos_protection_policy()))
1297            .try_build_from_parts(&mut OsRng, validator_configs, deployed_genesis)?;
1298        ssfn_config.save(path)?;
1299        ssfn_configs.push(ssfn_config);
1300    }
1301
1302    ssfn_configs
1303        .iter()
1304        .enumerate()
1305        .map(|(index, config)| {
1306            let address = config.p2p_config.external_address.clone().ok_or_else(|| {
1307                anyhow!(
1308                    "state sync fullnode {index} has no `p2p-config.external-address`, which the \
1309                     validators need to derive their seed peers"
1310                )
1311            })?;
1312            Ok(SeedPeer {
1313                peer_id: Some(anemo::PeerId(
1314                    config.network_key_pair().public().0.to_bytes(),
1315                )),
1316                address,
1317            })
1318        })
1319        .collect()
1320}

1346/// Prepended to every node config `--write-config` writes. `serde_yaml` drops
1347/// comments, so the file is this text followed by the serialized config.
1348const NODE_CONFIG_HEADER: &str = "\
1349# Generated by `iota-localnet start --write-config`. Editing this file does not
1350# change what `iota-localnet start` runs; pass `--node-config-override` to that
1351# command instead.
1352#
1353# A key that is absent from this file is at its default. For `policy-config`
1354# and `grpc-api-config` an explicit `null`, not an absent key, is what turns
1355# the feature off.
1356#
1357# The key pairs below are not redacted: a node config only runs with them, and
1358# they are the keys a persisted network already holds in plaintext in its
1359# `network.yaml`.
1360";

1362/// Write the config of every node of `swarm` to `directory`, under the file
1363/// names the config directory of a local network uses.
1364fn write_node_configs(swarm: &Swarm, directory: &Path) -> Result<(), anyhow::Error> {
1365    fs::create_dir_all(directory).map_err(|err| {
1366        anyhow!(err).context(format!(
1367            "Cannot create node config dir {}",
1368            directory.display()
1369        ))
1370    })?;
1371
1372    // Named by index rather than by host and port: a caller that reads these
1373    // files back knows the validator index, not the address genesis gave it.
1374    for (index, config) in swarm.config().validator_configs().iter().enumerate() {
1375        let path = directory.join(format!("validator-{index}.yaml"));
1376        write_node_config(config, &path)?;
1377    }
1378
1379    // The swarm keeps its nodes in a map, so sort them to give the same node
1380    // the same file name on every run.
1381    let mut fullnodes: Vec<&Node> = swarm.fullnodes().collect();
1382    fullnodes.sort_by_key(|fullnode| fullnode.name());
1383    for (index, fullnode) in fullnodes.iter().enumerate() {
1384        let name = if index == 0 {
1385            IOTA_FULLNODE_CONFIG.to_owned()
1386        } else {
1387            format!("fullnode-{index}.yaml")
1388        };
1389        write_node_config(&fullnode.config(), &directory.join(name))?;
1390    }
1391
1392    info!("Node configs written to {}", directory.display());
1393    Ok(())
1394}

1396fn write_node_config(config: &NodeConfig, path: &Path) -> Result<(), anyhow::Error> {
1397    let mut contents = NODE_CONFIG_HEADER.to_owned();
1398    contents.push('\n');
1399    contents.push_str(&serde_yaml::to_string(config)?);
1400    fs::write(path, contents)
1401        .with_context(|| format!("Cannot write node config to {}", path.display()))
1402}

1404/// Translate the deprecated `--with-grpc` value into node config overrides.
1405///
1406/// The API is off on a generated fullnode, which leaves `grpc-api-config`
1407/// an explicit `null`. The address is therefore set with a value that
1408/// replaces the whole section. The fields that value does not mention come
1409/// out at their default.
1410fn with_grpc_overrides(input: String) -> Result<[NodeConfigOverride; 2], anyhow::Error> {
1411    let grpc_address = service_address(input, DEFAULT_GRPC_PORT, "gRPC")?;
1412    Ok([
1413        "fullnode:enable-grpc-api=true".parse()?,
1414        // Quoted: an IPv6 address like `[::1]:50051` is YAML structure
1415        // when unquoted.
1416        format!("fullnode:grpc-api-config={{address: '{grpc_address}'}}").parse()?,
1417    ])
1418}

1438/// Log, per node the overrides reached, how many applied and which fields
1439/// they set. Later overrides win per field.
1440fn log_applied_node_config_overrides(swarm: &Swarm) {
1441    let node_config_overrides = swarm.node_config_overrides();
1442    for index in 0..swarm.config().validator_configs().len() {

1445            overrides_for_validator(node_config_overrides, index),
1446        );
1447    }

1449        log_applied_overrides_for_node("fullnode", overrides_for_fullnode(node_config_overrides));
1450    }

1453fn log_applied_overrides_for_node<'a>(
1454    name: &str,
1455    node_config_overrides: impl IntoIterator<Item = &'a NodeConfigOverride>,
1456) {
1457    let node_config_overrides: Vec<&NodeConfigOverride> =
1458        node_config_overrides.into_iter().collect();
1459    if node_config_overrides.is_empty() {
1460        return;
1461    }
1462    // The values are not echoed: they may carry a credential, and the
1463    // fields alone say what the run deviates on.
1464    let fields = winning_field_paths(node_config_overrides.iter().copied());
1465    let noun = if node_config_overrides.len() == 1 {
1466        "override"
1467    } else {
1468        "overrides"
1469    };
1470    info!(
1471        "applied {} {noun} to {name}: {}",
1472        node_config_overrides.len(),
1473        fields.join(", ")
1474    );
1475}

1497/// Parse the input string into a SocketAddr, with a default port if none is
1498/// provided.
1499pub fn parse_host_port(
1500    input: String,
1501    default_port_if_missing: u16,
1502) -> Result<SocketAddr, AddrParseError> {
1503    parse_host_port_with_default_host(input, "0.0.0.0", default_port_if_missing)
1504}
1505
1506/// Same as [`parse_host_port`], for an endpoint whose host defaults to
1507/// something other than `0.0.0.0`.
1508pub fn parse_host_port_with_default_host(
1509    input: String,
1510    default_host: &str,
1511    default_port_if_missing: u16,
1512) -> Result<SocketAddr, AddrParseError> {
1513    let mut input = input;
1514    if input.contains("localhost") {
1515        input = input.replace("localhost", "127.0.0.1");
1516    }
1517    if input.contains(':') {
1518        input.parse::<SocketAddr>()
1519    } else if input.contains('.') {
1520        format!("{input}:{default_port_if_missing}").parse::<SocketAddr>()
1521    } else if !input.is_empty() {
1522        format!("{default_host}:{input}").parse::<SocketAddr>()
1523    } else {
1524        format!("{default_host}:{default_port_if_missing}").parse::<SocketAddr>()
1525    }
1526}
