4use std::str::FromStr;
5
6use iota_sdk_types::{Address, ObjectId, StructTag, TypeTag};
7use iota_types::supported_protocol_versions::Chain;
8use serde::{Deserialize, Serialize};
9
10use crate::Name;
11
12#[derive(Clone, Debug, Serialize, Deserialize, Eq, PartialEq)]
13#[serde(rename_all = "kebab-case")]
14pub struct IotaNamesConfig {
15    /// Address of the `iota_names` package.
16    pub package_address: Address,
17    /// ID of the `IotaNames` object.
18    pub object_id: ObjectId,
19    /// Address of the `payments` package.
20    pub payments_package_address: Address,
21    /// ID of the registry table.
22    pub registry_id: ObjectId,
23    /// ID of the reverse registry table.
24    pub reverse_registry_id: ObjectId,
25}
26
27impl Default for IotaNamesConfig {
28    fn default() -> Self {
29        Self::mainnet()
30    }
31}
32
33impl IotaNamesConfig {
34    pub fn new(
35        package_address: Address,
36        object_id: ObjectId,
37        payments_package_address: Address,
38        registry_id: ObjectId,
39        reverse_registry_id: ObjectId,
40    ) -> Self {
41        Self {
42            package_address,
43            object_id,
44            payments_package_address,
45            registry_id,
46            reverse_registry_id,
47        }
48    }
49
50    pub fn from_env() -> anyhow::Result<Self> {
51        Ok(Self::new(
52            std::env::var("IOTA_NAMES_PACKAGE_ADDRESS")?.parse()?,
53            std::env::var("IOTA_NAMES_OBJECT_ID")?.parse()?,
54            std::env::var("IOTA_NAMES_PAYMENTS_PACKAGE_ADDRESS")?.parse()?,
55            std::env::var("IOTA_NAMES_REGISTRY_ID")?.parse()?,
56            std::env::var("IOTA_NAMES_REVERSE_REGISTRY_ID")?.parse()?,
57        ))
58    }
59
60    pub fn from_chain(chain: &Chain) -> Self {
61        match chain {
62            Chain::Mainnet => IotaNamesConfig::mainnet(),
63            Chain::Testnet => IotaNamesConfig::testnet(),
64            Chain::Unknown => IotaNamesConfig::devnet(),
65        }
66    }
67
68    pub fn record_field_id(&self, name: &Name) -> ObjectId {
69        let name_type_tag = StructTag::new_name(self.package_address);
70        let name_bytes = bcs::to_bytes(name).unwrap();
71
72        iota_types::dynamic_field::derive_dynamic_field_id(
73            self.registry_id,
74            &TypeTag::Struct(Box::new(name_type_tag)),
75            &name_bytes,
76        )
77        .unwrap()
78    }
79
80    pub fn reverse_record_field_id(&self, address: &Address) -> ObjectId {
81        iota_types::dynamic_field::derive_dynamic_field_id(
82            self.reverse_registry_id,
83            &TypeTag::Address,
84            address.as_ref(),
85        )
86        .unwrap()
87    }
