6//! This module provides a TransactionExecutor implementation that allows
7//! transaction execution and simulation via gRPC without requiring quorum
8//! consensus.

22    },

31/// Transaction executor implementation for simulacrum
32/// This allows transaction execution and simulation via gRPC without requiring
33/// quorum consensus
34pub struct TransactionExecutor {
35    simulacrum: Arc<Simulacrum>,
36}
37
38impl TransactionExecutor {
39    pub fn new(simulacrum: Arc<Simulacrum>) -> Self {
40        Self { simulacrum }
41    }
42}
43
44#[async_trait]
45impl TransactionExecutorTrait for TransactionExecutor {
46    async fn execute_transaction(
47        &self,
48        request: ExecuteTransactionRequestV1,
49        _skip_certification: bool,
50        _client_addr: Option<std::net::SocketAddr>,
51    ) -> Result<ExecuteTransactionResponseV1, QuorumDriverError> {
52        let simulacrum = &*self.simulacrum;
53
54        // Execute the transaction directly
55        let (effects, _execution_error) = simulacrum
56            .execute_transaction(request.transaction.clone())
57            .map_err(|e| {
58                QuorumDriverError::QuorumDriverInternal(iota_types::error::IotaError::Unknown(
59                    e.to_string(),
60                ))
61            })?;
62
63        // Create a checkpoint to finalize the transaction
64        let checkpoint = simulacrum.create_checkpoint();
65
66        tracing::debug!(
67            tx_digest = ?effects.transaction_digest(),
68            checkpoint = checkpoint.sequence_number(),
69            "Transaction executed and finalized in simulacrum"
70        );
71
72        // For simulacrum, we create a dummy certified effects since there's no real
73        // validator consensus. We use
74        // CertifiedTransactionEffects::new_from_data_and_sig with empty
75        // signatures.
76        let (test_committee, _) = iota_types::committee::Committee::new_simple_test_committee();
77        let effects_cert = iota_types::effects::CertifiedTransactionEffects::new_from_data_and_sig(
78            effects.clone(),
79            iota_types::crypto::AuthorityQuorumSignInfo::new_from_auth_sign_infos(
80                vec![],
81                &test_committee,
82            )
83            .unwrap(),
84        );
85        let verified_effects =
86            iota_types::effects::VerifiedCertifiedTransactionEffects::new_unchecked(effects_cert);
87
88        // Build response
89        let response = ExecuteTransactionResponseV1 {
90            effects: FinalizedEffects::new_from_effects_cert(verified_effects.into()),
91            events: if request.include_events {
92                self.simulacrum.with_store(|store| {
93                    store
94                        .get_transaction_events(effects.transaction_digest())
95                        .cloned()
96                })
97            } else {
98                None
99            },
100            input_objects: if request.include_input_objects {
101                self.simulacrum.with_store(|store| {
102                    storage::get_transaction_input_objects(store, &effects).ok()
103                })
104            } else {
105                None
106            },
107            output_objects: if request.include_output_objects {
108                self.simulacrum.with_store(|store| {
109                    storage::get_transaction_output_objects(store, &effects).ok()
110                })
111            } else {
112                None
113            },
114            auxiliary_data: if request.include_auxiliary_data {
115                // We don't have any aux data generated presently, also in the real network
116                None
117            } else {
118                None
119            },
120        };
121
122        Ok(response)
123    }
124
125    fn simulate_transaction(
126        &self,
127        transaction: Transaction,
128        checks: VmChecks,
129    ) -> Result<SimulateTransactionResult, iota_types::error::IotaError> {
130        self.simulacrum.simulate_transaction(transaction, checks)
131    }
132
133    /// Wait for the given transactions to be included in a checkpoint.
134    ///
135    /// Returns a mapping from transaction digest to
136    /// `(checkpoint_sequence_number, checkpoint_timestamp_ms)`.
137    /// On timeout, returns partial results for any transactions that were
138    /// already checkpointed.
139    async fn wait_for_checkpoint_inclusion(
140        &self,
141        _digests: &[TransactionDigest],
142        _timeout: Duration,
143    ) -> Result<BTreeMap<TransactionDigest, (CheckpointSequenceNumber, u64)>, IotaError> {
144        Err(IotaError::UnsupportedFeature {
145            error: "wait_for_checkpoint_inclusion not supported by this executor".into(),
146        })
147    }
148}
