179pub async fn execute_db_tool_command(db_path: PathBuf, cmd: DbToolCommand) -> anyhow::Result<()> {
180    match cmd {
181        DbToolCommand::ListTables => print_db_all_tables(db_path),
182        DbToolCommand::Dump(d) => print_all_entries(
183            d.store_name,
184            d.epoch,
185            db_path,
186            &d.table_name,
187            d.page_size,
188            d.page_number,
189        ),
190        DbToolCommand::TableSummary(d) => {
191            print_db_table_summary(d.store_name, d.epoch, db_path, &d.table_name)
192        }
193        DbToolCommand::DuplicatesSummary => print_db_duplicates_summary(db_path),
194        DbToolCommand::PrintCheckpointWatermarks => print_checkpoint_watermarks(&db_path),
195        DbToolCommand::ListDBMetadata(d) => {
196            print_table_metadata(d.store_name, d.epoch, db_path, &d.table_name)
197        }
198        DbToolCommand::PrintLastConsensusIndex => print_last_consensus_index(&db_path),
199        DbToolCommand::PrintTransaction(d) => print_transaction(&db_path, d),
200        DbToolCommand::PrintObject(o) => print_object(&db_path, o),
201        DbToolCommand::PrintCheckpoint(d) => print_checkpoint(&db_path, d),
202        DbToolCommand::PrintCheckpointContent(d) => print_checkpoint_content(&db_path, d),
203        DbToolCommand::RewindCheckpointExecution(d) => {
204            rewind_checkpoint_execution(&db_path, d.epoch, d.checkpoint_sequence_number)
205        }
206        DbToolCommand::Compact => compact(db_path),
207        DbToolCommand::MarkObjectBacklogSwept => mark_object_backlog_swept(&db_path),
208        DbToolCommand::IndexSearchKeyRange(rg) => {
209            let res = search_index(
210                db_path,
211                rg.table_name,
212                rg.start,
213                SearchRange::ExclusiveLastKey(rg.end_key),
214            )?;
215            for (k, v) in res {
216                println!("{k}: {v}");
217            }
218            Ok(())
219        }
220        DbToolCommand::IndexSearchCount(sc) => {
221            let res = search_index(
222                db_path,
223                sc.table_name,
224                sc.start,
225                SearchRange::Count(sc.count),
226            )?;
227            for (k, v) in res {
228                println!("{k}: {v}");
229            }
230            Ok(())
231        }
232        DbToolCommand::SetCheckpointWatermark(d) => set_checkpoint_watermark(&db_path, d),
233    }
234}

236/// The directory holding the per-epoch history of the node whose live
237/// database directory is `live_db_path`. Every open of the node's stores must
238/// be given it.
239pub(crate) fn historic_db_path(live_db_path: &Path) -> PathBuf {
240    live_db_path.join(HISTORIC_DB_DIR)

243/// [`historic_db_path`] for a store whose directory, `store_path`, is inside
244/// a node's live database directory.
245pub(crate) fn historic_db_path_of_store(store_path: &Path) -> PathBuf {
246    historic_db_path(store_path.parent().unwrap_or(store_path))
247}

284/// Fails for the perpetual store under `store` until a node on this release
285/// has finished its first start on it: before that the history is still in the
286/// flat tables the commands here no longer read, and a release before the
287/// per-epoch storage may still be started on it.
288pub fn ensure_migrated(store: &Path) -> anyhow::Result<()> {
289    let perpetual = store.join("perpetual");
290    if !perpetual.join("CURRENT").exists() {
291        return Ok(());
292    }
293    let not_migrated = || {
294        anyhow!(
295            "the database under {} has not been migrated to the per-epoch storage yet; \
296             start the node on this release until it serves before using this command, or \
297             use a read-only command such as `db-tool dump`",
298            store.display()
299        )
300    };
301    if typed_store::rocks::list_tables(perpetual)
302        .map_err(|e| anyhow!("cannot list the column families of the perpetual store: {e}"))?
303        .iter()
304        .any(|table| table == "pruned_checkpoint")
305    {
306        return Err(not_migrated());
307    }
308    let tables =
309        AuthorityPerpetualTables::open_readonly(store, Some(&historic_db_path_of_store(store)));
310    if !tables.ledger_backlog_migrated()? {
311        return Err(not_migrated());
312    }
313    Ok(())
314}

316pub fn print_transaction(path: &Path, opt: PrintTransactionOptions) -> anyhow::Result<()> {
317    ensure_migrated(&path.join("store"))?;
318    let (_perpetual_db, _historic_objects, historic_ledger, _epoch_markers) =
319        AuthorityPerpetualTables::open_with_historic_objects(
320            &path.join("store"),
321            perpetual_options(path),
322        )?;
323    if let Some((epoch, checkpoint_seq_num)) =
324        historic_ledger.get_transaction_checkpoint(&opt.digest)?
325    {
326        println!(
327            "Transaction {:?} executed in epoch {} checkpoint {}",
328            opt.digest, epoch, checkpoint_seq_num
329        );
330    };
331    if let Some(effects) = historic_ledger.get_executed_effects(&opt.digest)? {
332        println!(
333            "Transaction {:?} dependencies: {:#?}",
334            opt.digest,
335            effects.dependencies(),
336        );
337    };
338    Ok(())
339}

341pub fn print_object(path: &Path, opt: PrintObjectOptions) -> anyhow::Result<()> {
342    ensure_migrated(&path.join("store"))?;
343    let perpetual_db = AuthorityPerpetualTables::open(&path.join("store"), perpetual_options(path));
344
345    let obj = if let Some(version) = opt.version {
346        perpetual_db.try_get_object_by_key(&opt.id, version.into())?
347    } else {
348        perpetual_db.try_get_object(&opt.id)?
349    };
350
351    if let Some(obj) = obj {
352        println!("Object {:?}:\n{:#?}", opt.id, obj);
353    } else {
354        println!("Object {:?} not found", opt.id);
355    }
356
357    Ok(())
358}

360pub fn print_checkpoint(path: &Path, opt: PrintCheckpointOptions) -> anyhow::Result<()> {
361    ensure_migrated(&path.join("store"))?;
362    let checkpoint_store = CheckpointStore::new_with_historic_db_path(
363        &path.join("checkpoints"),
364        &historic_db_path(path),
365    );
366    let checkpoint = checkpoint_store
367        .get_checkpoint_by_digest(&opt.digest)?
368        .ok_or(anyhow!(
369            "Checkpoint digest {:?} not found in checkpoint store",
370            opt.digest
371        ))?;
372    println!("Checkpoint: {checkpoint:?}");
373    drop(checkpoint_store);
374    print_checkpoint_content(
375        path,
376        PrintCheckpointContentOptions {
377            digest: checkpoint.contents_digest,
378        },
379    )
380}

401/// Force sets the highest executed checkpoint.
402/// NOTE: Does not force re-execution of transactions.
403/// Run with: cargo run --package iota-tool -- db-tool --db-path
404/// /opt/iota/db/authorities_db/live rewind-checkpoint-execution --epoch 3
405/// --checkpoint-sequence-number 300000
406pub fn rewind_checkpoint_execution(
407    path: &Path,
408    epoch: EpochId,
409    checkpoint_sequence_number: u64,
410) -> anyhow::Result<()> {
411    ensure_migrated(&path.join("store"))?;
412    let checkpoint_db = CheckpointStore::new_with_historic_db_path(
413        &path.join("checkpoints"),
414        &historic_db_path(path),
415    );
416    let Some(checkpoint) =
417        checkpoint_db.get_checkpoint_by_sequence_number(checkpoint_sequence_number)?
418    else {
419        bail!("Checkpoint {checkpoint_sequence_number} not found!");
420    };
421    if epoch != checkpoint.epoch() {
422        bail!(
423            "Checkpoint {checkpoint_sequence_number} is in epoch {} not {epoch}!",
424            checkpoint.epoch()
425        );
426    }
427
428    let highest_executed_sequence_number = checkpoint_db
429        .get_highest_executed_checkpoint_seq_number()?
430        .unwrap_or_default();
431    if checkpoint_sequence_number > highest_executed_sequence_number {
432        bail!(
433            "Must rewind checkpoint execution to be not later than highest executed ({checkpoint_sequence_number} > {highest_executed_sequence_number})!"
434        );
435    }
436    checkpoint_db.set_highest_executed_checkpoint_subtle(&checkpoint)?;
437    Ok(())
438}

440pub fn print_db_table_summary(
441    store: StoreName,
442    epoch: Option<EpochId>,
443    path: PathBuf,
444    table_name: &str,
445) -> anyhow::Result<()> {
446    let summary = table_summary(store, epoch, path, table_name)?;
447    let quantiles = [25, 50, 75, 90, 99];
448    println!(
449        "Total num keys = {}, total key bytes = {}, total value bytes = {}",
450        summary.num_keys, summary.key_bytes_total, summary.value_bytes_total
451    );
452    println!("Key size distribution:\n");
453    quantiles.iter().for_each(|q| {
454        println!(
455            "p{:?} -> {:?} bytes\n",
456            q,
457            summary.key_hist.value_at_quantile(*q as f64 / 100.0)
458        );
459    });
460    println!("Value size distribution:\n");
461    quantiles.iter().for_each(|q| {
462        println!(
463            "p{:?} -> {:?} bytes\n",
464            q,
465            summary.value_hist.value_at_quantile(*q as f64 / 100.0)
466        );
467    });
468    Ok(())
469}

485/// Records the one-time object backlog sweep as skipped without running it,
486/// for a node that cannot start because the sweep fails on a row it cannot
487/// decode.
488///
489/// The versions the sweep had not reached stay in the live `objects` table,
490/// where no retention reclaims them, until the node is rebuilt from a
491/// snapshot. Bounded object reads on the node keep checking every retained
492/// bucket for good.
493///
494/// Run with (for example):
495/// cargo run --package iota-tool -- db-tool --db-path
496/// /opt/iota/db/authorities_db/live mark-object-backlog-swept
497pub fn mark_object_backlog_swept(path: &Path) -> anyhow::Result<()> {
498    let store = path.join("store");
499    // Opening creates the database, so a mistyped path would otherwise leave
500    // an empty one behind and report success.
501    if !store.join("perpetual").exists() {
502        bail!("no perpetual store under {}", store.display());
503    }
504    ensure_migrated(&store)?;
505    let perpetual = AuthorityPerpetualTables::open(&store, perpetual_options(path));
506    perpetual.mark_object_backlog_sweep_skipped()?;
507    println!(
508        "recorded the object backlog sweep as skipped; the versions it had not reached stay in \
509         the live objects table, and bounded object reads keep checking every retained bucket"
510    );
511    Ok(())
512}

514/// Prints the checkpoint watermark rows as stored, without resolving them to
515/// checkpoints, so a watermark naming a checkpoint the store no longer holds
516/// still shows.
517pub fn print_checkpoint_watermarks(path: &Path) -> anyhow::Result<()> {
518    let checkpoint_db = CheckpointStore::new_with_historic_db_path(
519        &path.join("checkpoints"),
520        &historic_db_path(path),
521    );
522    let mut table = Table::new();
523    table
524        .set_content_arrangement(ContentArrangement::Dynamic)
525        .set_header(vec!["watermark", "sequence number", "digest"]);
526    for (mark, sequence_number, digest) in checkpoint_db.get_checkpoint_watermarks()? {
527        table.add_row(vec![
528            format!("{mark:?}"),
529            sequence_number.to_string(),
530            format!("{digest}"),
531        ]);
532    }
533    println!("{table}");
534    Ok(())
535}

537/// Force sets state sync checkpoint watermarks.
538/// Run with (for example):
539/// cargo run --package iota-tool -- db-tool --db-path
540/// /opt/iota/db/authorities_db/live set_checkpoint_watermark --highest-synced
541/// 300000
542pub fn set_checkpoint_watermark(
543    path: &Path,
544    options: SetCheckpointWatermarkOptions,
545) -> anyhow::Result<()> {
546    ensure_migrated(&path.join("store"))?;
547    let checkpoint_db = CheckpointStore::new_with_historic_db_path(
548        &path.join("checkpoints"),
549        &historic_db_path(path),
550    );
551
552    if let Some(highest_verified) = options.highest_verified {
553        let Some(checkpoint) = checkpoint_db.get_checkpoint_by_sequence_number(highest_verified)?
554        else {
555            bail!("Checkpoint {highest_verified} not found");
556        };
557        checkpoint_db.set_highest_verified_checkpoint_subtle(&checkpoint)?;
558    }
559    if let Some(highest_synced) = options.highest_synced {
560        let Some(checkpoint) = checkpoint_db.get_checkpoint_by_sequence_number(highest_synced)?
561        else {
562            bail!("Checkpoint {highest_synced} not found");
563        };
564        // Verify that checkpoint contents exist, since the checkpoint executor
565        // will panic if it tries to execute a checkpoint whose contents are
566        // missing from the store.
567        if checkpoint_db
568            .get_checkpoint_contents(&checkpoint.contents_digest)?
569            .is_none()
570        {
571            bail!(
572                "Checkpoint contents not found for checkpoint {highest_synced}. \
573                 Setting highest_synced without contents would cause the executor to panic."
574            );
575        }
576        checkpoint_db.set_highest_synced_checkpoint_subtle(&checkpoint)?;
577    }
578    Ok(())
579}

589    fn create_perpetual(store: &Path, column_families: &[&str]) {
590        let mut options = rocksdb::Options::default();
591        options.create_if_missing(true);
592        options.create_missing_column_families(true);
593        rocksdb::DB::open_cf(&options, store.join("perpetual"), column_families).unwrap();
594    }

596    /// A store a release before the per-epoch storage wrote last is refused,
597    /// and so is one whose first start has not finished the migration; a
598    /// migrated one and a missing one are not.
599    #[tokio::test]
600    async fn only_a_migrated_store_is_accepted() {
601        let old = tempfile::tempdir().unwrap();
602        create_perpetual(old.path(), &["objects", "pruned_checkpoint"]);
603        assert!(ensure_migrated(old.path()).is_err());
604
605        let unfinished = tempfile::tempdir().unwrap();
606        drop(AuthorityPerpetualTables::open(unfinished.path(), None));
607        assert!(ensure_migrated(unfinished.path()).is_err());
608
609        let migrated = tempfile::tempdir().unwrap();
610        AuthorityPerpetualTables::open(migrated.path(), None)
611            .mark_ledger_backlog_migrated()
612            .unwrap();
613        ensure_migrated(migrated.path()).unwrap();
614
615        ensure_migrated(&tempfile::tempdir().unwrap().path().join("missing")).unwrap();
616    }
