178pub async fn execute_db_tool_command(db_path: PathBuf, cmd: DbToolCommand) -> anyhow::Result<()> {
179    match cmd {
180        DbToolCommand::ListTables => print_db_all_tables(db_path),
181        DbToolCommand::Dump(d) => print_all_entries(
182            d.store_name,
183            d.epoch,
184            db_path,
185            &d.table_name,
186            d.page_size,
187            d.page_number,
188        ),
189        DbToolCommand::TableSummary(d) => {
190            print_db_table_summary(d.store_name, d.epoch, db_path, &d.table_name)
191        }
192        DbToolCommand::DuplicatesSummary => print_db_duplicates_summary(db_path),
193        DbToolCommand::PrintCheckpointWatermarks => print_checkpoint_watermarks(&db_path),
194        DbToolCommand::ListDBMetadata(d) => {
195            print_table_metadata(d.store_name, d.epoch, db_path, &d.table_name)
196        }
197        DbToolCommand::PrintLastConsensusIndex => print_last_consensus_index(&db_path),
198        DbToolCommand::PrintTransaction(d) => print_transaction(&db_path, d),
199        DbToolCommand::PrintObject(o) => print_object(&db_path, o),
200        DbToolCommand::PrintCheckpoint(d) => print_checkpoint(&db_path, d),
201        DbToolCommand::PrintCheckpointContent(d) => print_checkpoint_content(&db_path, d),
202        DbToolCommand::RewindCheckpointExecution(d) => {
203            rewind_checkpoint_execution(&db_path, d.epoch, d.checkpoint_sequence_number)
204        }
205        DbToolCommand::Compact => compact(db_path),
206        DbToolCommand::IndexSearchKeyRange(rg) => {
207            let res = search_index(
208                db_path,
209                rg.table_name,
210                rg.start,
211                SearchRange::ExclusiveLastKey(rg.end_key),
212            )?;
213            for (k, v) in res {
214                println!("{k}: {v}");
215            }
216            Ok(())
217        }
218        DbToolCommand::IndexSearchCount(sc) => {
219            let res = search_index(
220                db_path,
221                sc.table_name,
222                sc.start,
223                SearchRange::Count(sc.count),
224            )?;
225            for (k, v) in res {
226                println!("{k}: {v}");
227            }
228            Ok(())
229        }
230        DbToolCommand::SetCheckpointWatermark(d) => set_checkpoint_watermark(&db_path, d),
231    }
232}

234/// The directory holding the per-epoch history of the node whose live
235/// database directory is `live_db_path`. Every open of the node's stores must
236/// be given it.
237pub(crate) fn historic_db_path(live_db_path: &Path) -> PathBuf {
238    live_db_path.join(HISTORIC_DB_DIR)
239}
240
241/// [`historic_db_path`] for a store whose directory, `store_path`, is inside
242/// a node's live database directory.
243pub(crate) fn historic_db_path_of_store(store_path: &Path) -> PathBuf {
244    historic_db_path(store_path.parent().unwrap_or(store_path))
245}

248/// directory is `live_db_path`.

250    Some(AuthorityPerpetualTablesOptions {

252        ..Default::default()

282pub fn print_transaction(path: &Path, opt: PrintTransactionOptions) -> anyhow::Result<()> {
283    AuthorityPerpetualTables::ensure_per_epoch_layout(&path.join("store"))?;
284    let (_perpetual_db, _historic_objects, historic_ledger, _epoch_markers) =
285        AuthorityPerpetualTables::open_with_historic_objects(
286            &path.join("store"),
287            perpetual_options(path),
288        )?;
289    if let Some((epoch, checkpoint_seq_num)) =
290        historic_ledger.get_transaction_checkpoint(&opt.digest)?
291    {
292        println!(
293            "Transaction {:?} executed in epoch {} checkpoint {}",
294            opt.digest, epoch, checkpoint_seq_num
295        );
296    };
297    if let Some(effects) = historic_ledger.get_executed_effects(&opt.digest)? {
298        println!(
299            "Transaction {:?} dependencies: {:#?}",
300            opt.digest,
301            effects.dependencies(),
302        );
303    };
304    Ok(())
305}

307pub fn print_object(path: &Path, opt: PrintObjectOptions) -> anyhow::Result<()> {
308    AuthorityPerpetualTables::ensure_per_epoch_layout(&path.join("store"))?;
309    let perpetual_db = AuthorityPerpetualTables::open(&path.join("store"), perpetual_options(path));
310
311    let obj = if let Some(version) = opt.version {
312        perpetual_db.try_get_object_by_key(&opt.id, version.into())?
313    } else {
314        perpetual_db.try_get_object(&opt.id)?
315    };
316
317    if let Some(obj) = obj {
318        println!("Object {:?}:\n{:#?}", opt.id, obj);
319    } else {
320        println!("Object {:?} not found", opt.id);
321    }
322
323    Ok(())
324}

326pub fn print_checkpoint(path: &Path, opt: PrintCheckpointOptions) -> anyhow::Result<()> {
327    AuthorityPerpetualTables::ensure_per_epoch_layout(&path.join("store"))?;
328    let checkpoint_store = CheckpointStore::new_with_historic_db_path(
329        &path.join("checkpoints"),
330        &historic_db_path(path),
331    );
332    let checkpoint = checkpoint_store
333        .get_checkpoint_by_digest(&opt.digest)?
334        .ok_or(anyhow!(
335            "Checkpoint digest {:?} not found in checkpoint store",
336            opt.digest
337        ))?;
338    println!("Checkpoint: {checkpoint:?}");
339    drop(checkpoint_store);
340    print_checkpoint_content(
341        path,
342        PrintCheckpointContentOptions {
343            digest: checkpoint.contents_digest,
344        },
345    )
346}

367/// Force sets the highest executed checkpoint.
368/// NOTE: Does not force re-execution of transactions.
369/// Run with: cargo run --package iota-tool -- db-tool --db-path
370/// /opt/iota/db/authorities_db/live rewind-checkpoint-execution --epoch 3
371/// --checkpoint-sequence-number 300000
372pub fn rewind_checkpoint_execution(
373    path: &Path,
374    epoch: EpochId,
375    checkpoint_sequence_number: u64,
376) -> anyhow::Result<()> {
377    AuthorityPerpetualTables::ensure_per_epoch_layout(&path.join("store"))?;
378    let checkpoint_db = CheckpointStore::new_with_historic_db_path(
379        &path.join("checkpoints"),
380        &historic_db_path(path),
381    );
382    let Some(checkpoint) =
383        checkpoint_db.get_checkpoint_by_sequence_number(checkpoint_sequence_number)?
384    else {
385        bail!("Checkpoint {checkpoint_sequence_number} not found!");
386    };
387    if epoch != checkpoint.epoch() {
388        bail!(
389            "Checkpoint {checkpoint_sequence_number} is in epoch {} not {epoch}!",
390            checkpoint.epoch()
391        );
392    }
393
394    let highest_executed_sequence_number = checkpoint_db
395        .get_highest_executed_checkpoint_seq_number()?
396        .unwrap_or_default();
397    if checkpoint_sequence_number > highest_executed_sequence_number {
398        bail!(
399            "Must rewind checkpoint execution to be not later than highest executed ({checkpoint_sequence_number} > {highest_executed_sequence_number})!"
400        );
401    }
402    checkpoint_db.set_highest_executed_checkpoint_subtle(&checkpoint)?;
403    Ok(())
404}

406pub fn print_db_table_summary(
407    store: StoreName,
408    epoch: Option<EpochId>,
409    path: PathBuf,
410    table_name: &str,
411) -> anyhow::Result<()> {
412    let summary = table_summary(store, epoch, path, table_name)?;
413    let quantiles = [25, 50, 75, 90, 99];
414    println!(
415        "Total num keys = {}, total key bytes = {}, total value bytes = {}",
416        summary.num_keys, summary.key_bytes_total, summary.value_bytes_total
417    );
418    println!("Key size distribution:\n");
419    quantiles.iter().for_each(|q| {
420        println!(
421            "p{:?} -> {:?} bytes\n",
422            q,
423            summary.key_hist.value_at_quantile(*q as f64 / 100.0)
424        );
425    });
426    println!("Value size distribution:\n");
427    quantiles.iter().for_each(|q| {
428        println!(
429            "p{:?} -> {:?} bytes\n",
430            q,
431            summary.value_hist.value_at_quantile(*q as f64 / 100.0)
432        );
433    });
434    Ok(())
435}

451/// Prints the checkpoint watermark rows as stored, without resolving them to
452/// checkpoints, so a watermark naming a checkpoint the store no longer holds
453/// still shows.
454pub fn print_checkpoint_watermarks(path: &Path) -> anyhow::Result<()> {
455    AuthorityPerpetualTables::ensure_per_epoch_layout(&path.join("store"))?;
456    let checkpoint_db = CheckpointStore::new_with_historic_db_path(
457        &path.join("checkpoints"),
458        &historic_db_path(path),
459    );
460    let mut table = Table::new();
461    table
462        .set_content_arrangement(ContentArrangement::Dynamic)
463        .set_header(vec!["watermark", "sequence number", "digest"]);
464    for (mark, sequence_number, digest) in checkpoint_db.get_checkpoint_watermarks()? {
465        table.add_row(vec![
466            format!("{mark:?}"),
467            sequence_number.to_string(),
468            format!("{digest}"),
469        ]);
470    }
471    println!("{table}");
472    Ok(())
473}

475/// Force sets state sync checkpoint watermarks.
476/// Run with (for example):
477/// cargo run --package iota-tool -- db-tool --db-path
478/// /opt/iota/db/authorities_db/live set_checkpoint_watermark --highest-synced
479/// 300000
480pub fn set_checkpoint_watermark(
481    path: &Path,
482    options: SetCheckpointWatermarkOptions,
483) -> anyhow::Result<()> {
484    AuthorityPerpetualTables::ensure_per_epoch_layout(&path.join("store"))?;
485    let checkpoint_db = CheckpointStore::new_with_historic_db_path(
486        &path.join("checkpoints"),
487        &historic_db_path(path),
488    );
489
490    if let Some(highest_verified) = options.highest_verified {
491        let Some(checkpoint) = checkpoint_db.get_checkpoint_by_sequence_number(highest_verified)?
492        else {
493            bail!("Checkpoint {highest_verified} not found");
494        };
495        checkpoint_db.set_highest_verified_checkpoint_subtle(&checkpoint)?;
496    }
497    if let Some(highest_synced) = options.highest_synced {
498        let Some(checkpoint) = checkpoint_db.get_checkpoint_by_sequence_number(highest_synced)?
499        else {
500            bail!("Checkpoint {highest_synced} not found");
501        };
502        // Verify that checkpoint contents exist, since the checkpoint executor
503        // will panic if it tries to execute a checkpoint whose contents are
504        // missing from the store.
505        if checkpoint_db
506            .get_checkpoint_contents(&checkpoint.contents_digest)?
507            .is_none()
508        {
509            bail!(
510                "Checkpoint contents not found for checkpoint {highest_synced}. \
511                 Setting highest_synced without contents would cause the executor to panic."
512            );
513        }
514        checkpoint_db.set_highest_synced_checkpoint_subtle(&checkpoint)?;
515    }
516    Ok(())
