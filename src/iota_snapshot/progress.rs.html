4//! Progress reporting for snapshot downloads.
5//!
6//! When the total byte size of a download is known — from a size sweep over
7//! the remote files with [`fetch_total_bytes`] — the bar is byte-denominated
8//! and indicatif renders the size totals, download speed, and ETA natively,
9//! while the file counts move to the bar message. When the sweep fails, the
10//! bar falls back to file counts.
11//!
12//! Bars can only be drawn on a terminal. Whenever they can't be — output is
13//! piped or redirected, or `--disable-progress-bar` was passed — every bar's
14//! [`ProgressTicker`] logs the same information as a status line once per
15//! second instead.
16

40/// The `tracing` target every status line is logged under, so that a caller
41/// which otherwise silences its logs can keep them.
42pub const LOG_TARGET_PROGRESS: &str = module_path!();
43
44/// Report `msg` on the progress display, or log it when the display is hidden
45/// — [`MultiProgress::println`] writes nothing at all there, so the phase
46/// announcements around the bars would otherwise be lost in exactly the runs
47/// that depend on the status lines.
48pub fn println_or_log(m: &MultiProgress, msg: impl AsRef<str>) -> std::io::Result<()> {
49    if m.is_hidden() {
50        info!(target: LOG_TARGET_PROGRESS, "{}", msg.as_ref());
51        Ok(())
52    } else {
53        m.println(msg)
54    }
55}

57/// The progress display for a restore: one that draws its bars on the
58/// terminal, or a hidden one — whose bars report through periodic status logs
59/// instead — when `disable_progress_bar` is set.
60///
61/// A display bound to a non-terminal output stream is hidden either way, so
62/// callers get the status logs there without passing the flag.
63pub fn make_multi_progress(disable_progress_bar: bool) -> MultiProgress {
64    if disable_progress_bar {
65        MultiProgress::with_draw_target(ProgressDrawTarget::hidden())
66    } else {
67        MultiProgress::new()
68    }
69}

71/// What a progress bar's position counts, used to render its status lines.
72#[derive(Clone, Copy)]
73pub enum ProgressUnit {
74    /// The position is a number of bytes.
75    Bytes,
76    /// The position counts items named by the given plural noun, e.g. "files".
77    Count(&'static str),
78}

80/// Reports a progress bar's state once per second for as long as it runs:
81/// mirrors `counter` into the bar's position if one was given, and logs a
82/// status line carrying the same totals, rate and ETA the bar shows whenever
83/// the bar itself can't be drawn.
84///
85/// Finish the bar through [`Self::finish_with_message`] or
86/// [`Self::finish_and_clear`] rather than on the bar itself, so the final
87/// status line is logged even when the bar finishes before the first second is
88/// up. Dropping the ticker stops the reporting.
89pub struct ProgressTicker {
90    bar: ProgressBar,
91    phase: &'static str,
92    unit: ProgressUnit,
93    task: Option<JoinHandle<()>>,
94}
95
96impl ProgressTicker {
97    /// Starts reporting `bar`, whose position counts `unit` and whose progress
98    /// is logged under `phase`, e.g. "Downloading files".
99    ///
100    /// When a `counter` is given, the bar's position is taken from it once per
101    /// second rather than being advanced by the caller.
102    pub fn spawn(
103        bar: ProgressBar,
104        phase: &'static str,
105        unit: ProgressUnit,
106        counter: Option<Arc<AtomicU64>>,
107    ) -> Self {
108        // A drawn bar renders its own totals, so there is only something to do
109        // once per second if the bar is hidden or its position has to be
110        // mirrored from a counter.
111        let log_status = bar.is_hidden();
112        let task = (log_status || counter.is_some()).then(|| {
113            let bar = bar.clone();
114            tokio::spawn(async move {
115                while !bar.is_finished() {
116                    if let Some(counter) = &counter {
117                        bar.set_position(counter.load(Ordering::Relaxed));
118                    }
119                    if log_status {
120                        info!(
121                            target: LOG_TARGET_PROGRESS,
122                            "{}",
123                            status_line(&bar, phase, &unit, None)
124                        );
125                    }
126                    tokio::time::sleep(Duration::from_secs(1)).await;
127                }
128            })
129        });
130        Self {
131            bar,
132            phase,
133            unit,
134            task,
135        }
136    }
137
138    /// Finishes the bar with `msg`, leaving it on screen.
139    pub fn finish_with_message(&self, msg: &'static str) {
140        self.finish(msg, false);
141    }
142
143    /// Finishes the bar with `msg` and removes it from the display, for a
144    /// phase whose completed bar isn't worth keeping.
145    pub fn finish_and_clear(&self, msg: &'static str) {
146        self.finish(msg, true);
147    }
148
149    /// The underlying bar, to advance it or set its message.
150    pub fn bar(&self) -> &ProgressBar {
151        &self.bar
152    }
153
154    fn finish(&self, msg: &'static str, clear: bool) {
155        // Stop the ticker first so it can't log a periodic line after the
156        // final one.
157        self.stop();
158        if clear {
159            self.bar.finish_and_clear();
160        } else {
161            self.bar.finish_with_message(msg);
162        }
163        if self.bar.is_hidden() {
164            info!(
165                target: LOG_TARGET_PROGRESS,
166                "{}",
167                status_line(&self.bar, self.phase, &self.unit, Some(msg))
168            );
169        }
170    }
171
172    fn stop(&self) {
173        if let Some(task) = &self.task {
174            task.abort();
175        }
176    }
177}
178
179impl Drop for ProgressTicker {
180    fn drop(&mut self) {
181        self.stop();
182    }
183}

185/// Renders `bar`'s state as one log line: elapsed time, phase, how far it has
186/// got, its rate and ETA, and finally either `done_msg` or the bar's own
187/// message.
188///
189/// The rate and ETA are left out until the bar has moved far enough to
190/// estimate them, and the ETA also once the bar has finished.
191fn status_line(
192    bar: &ProgressBar,
193    phase: &str,
194    unit: &ProgressUnit,
195    done_msg: Option<&str>,
196) -> String {
197    let pos = bar.position();
198    // Every snapshot bar is created with a length; falling back to the
199    // position keeps a lengthless one from reporting a total of zero.
200    let len = bar.length().unwrap_or(pos);
201    let progress = match unit {
202        ProgressUnit::Bytes => format!("{}/{}", HumanBytes(pos), HumanBytes(len)),
203        ProgressUnit::Count(unit) => format!("{}/{} {unit}", HumanCount(pos), HumanCount(len)),
204    };
205    let mut line = format!("[{}] {phase}: {progress}", FormattedDuration(bar.elapsed()));
206    let rate = bar.per_sec();
207    if rate > 0.0 {
208        let rate = match unit {
209            ProgressUnit::Bytes => format!("{}/s", HumanBytes(rate as u64)),
210            ProgressUnit::Count(unit) => format!("{rate:.1} {unit}/s"),
211        };
212        if bar.is_finished() {
213            line.push_str(&format!(" ({rate})"));
214        } else {
215            line.push_str(&format!(" ({rate}, ETA {})", HumanDuration(bar.eta())));
216        }
217    }
218    let msg: Cow<'_, str> = done_msg.map_or_else(|| bar.message().into(), Into::into);
219    if !msg.is_empty() {
220        line.push_str(&format!(" — {msg}"));
221    }
222    line
223}

225/// Fetch the combined size in bytes of every path in `paths` from the remote
226/// store with bounded concurrency, rendering a progress bar on `m` while the
227/// lookups run. Returns `None` after logging a warning if any lookup fails —
228/// a partial sum would misstate the total: progress reporting is best-effort
229/// and must never fail a download.
230pub async fn fetch_total_bytes(
231    remote_object_store: &Arc<dyn ObjectStoreGetExt>,
232    paths: Vec<Path>,
233    concurrency: usize,
234    m: &MultiProgress,
235) -> Option<u64> {
236    let bar = m.add(
237        ProgressBar::new(paths.len() as u64).with_style(
238            ProgressStyle::with_template(
239                "[{elapsed_precise}] {wide_bar} Fetching file sizes: {pos}/{len} (ETA {eta})",
240            )
241            .unwrap(),
242        ),
243    );
244    steady_tick_when_drawn(&bar);
245    let ticker = ProgressTicker::spawn(
246        bar.clone(),
247        "Fetching file sizes",
248        ProgressUnit::Count("files"),
249        None,
250    );
251    let mut stream = futures::stream::iter(paths)
252        .map(|path| {
253            let store = remote_object_store.clone();
254            let bar = bar.clone();
255            async move {
256                // Sizes are best-effort: retry briefly so a single transient
257                // error out of thousands of lookups doesn't hide the totals,
258                // but give up quickly rather than stall the restore.
259                let size = retry(size_lookup_backoff(), || async {
260                    store.object_size(&path).await.map_err(classify_for_retry)
261                })
262                .await;
263                bar.inc(1);
264                (path, size)
265            }
266        })
267        .buffer_unordered(concurrency);
268    let mut total_bytes = 0u64;
269    while let Some((path, size)) = stream.next().await {
270        match size {
271            Ok(size) => total_bytes += size,
272            Err(err) => {
273                warn!("size lookup for {path} failed, download totals will not be shown: {err:?}");
274                // Dropping the stream cancels the remaining lookups rather
275                // than waiting for every one to finish first.
276                ticker.finish_and_clear("Size lookup failed");
277                return None;
278            }
279        }
280    }
281    ticker.finish_and_clear("File sizes fetched");
282    Some(total_bytes)
283}

285/// Repaints `bar` on a timer so it stays current while its work is in flight,
286/// rather than only when the caller advances it. A hidden bar is never
287/// repainted; its [`ProgressTicker`] reports it instead.
288fn steady_tick_when_drawn(bar: &ProgressBar) {
289    if !bar.is_hidden() {
290        bar.enable_steady_tick(Duration::from_millis(100));
291    }
292}

294/// Wrap `err` for [`retry`]: an error retrying cannot fix — a missing object
295/// or denied access — gives up immediately, everything else is retried.
296fn classify_for_retry(err: anyhow::Error) -> backoff::Error<anyhow::Error> {
297    let is_permanent = err.chain().any(|cause| {
298        cause
299            .downcast_ref::<reqwest::Error>()
300            .and_then(reqwest::Error::status)
301            .is_some_and(|status| {
302                matches!(
303                    status,
304                    reqwest::StatusCode::NOT_FOUND
305                        | reqwest::StatusCode::FORBIDDEN
306                        | reqwest::StatusCode::UNAUTHORIZED
307                )
308            })
309    });
310
311    if is_permanent {
312        backoff::Error::permanent(err)
313    } else {
314        backoff::Error::transient(err)
315    }
316}
317
318fn size_lookup_backoff() -> backoff::ExponentialBackoff {
319    backoff::ExponentialBackoff {
320        max_elapsed_time: Some(Duration::from_secs(3)),
321        ..Default::default()
322    }
323}

325/// A download progress bar added to `m`: byte-denominated when `total_bytes`
326/// is known (indicatif renders the size totals, download speed, and ETA; the
327/// file counts go to the message), file-count based otherwise.
328///
329/// `phase` is rendered verbatim at the start of the bar line, e.g.
330/// "Downloading files".
331pub struct DownloadProgressBar {
332    bar: ProgressBar,
333    num_files: u64,
334    files_done: AtomicU64,
335    byte_denominated: bool,
336    ticker: ProgressTicker,
337}
338
339impl DownloadProgressBar {
340    pub fn new(
341        m: &MultiProgress,
342        phase: &'static str,
343        num_files: u64,
344        total_bytes: Option<u64>,
345    ) -> Self {
346        // A single-file download has no file counts to carry alongside its
347        // bytes, so it leaves the bar's message out entirely.
348        let msg = if num_files > 1 { " — {msg}" } else { "" };
349        let bar = match total_bytes {
350            Some(total_bytes) => ProgressBar::new(total_bytes).with_style(
351                ProgressStyle::with_template(&format!(
352                    "[{{elapsed_precise}}] {{wide_bar}} {phase}: \
353                     {{binary_bytes}}/{{binary_total_bytes}} ({{binary_bytes_per_sec}}, \
354                     ETA {{eta}}){msg}"
355                ))
356                .unwrap(),
357            ),
358            None => ProgressBar::new(num_files).with_style(
359                ProgressStyle::with_template(&format!(
360                    "[{{elapsed_precise}}] {{wide_bar}} {phase}: {{pos}}/{{len}} files \
361                     (ETA {{eta}}){msg}"
362                ))
363                .unwrap(),
364            ),
365        };
366        let bar = m.add(bar);
367        steady_tick_when_drawn(&bar);
368        let unit = match total_bytes {
369            Some(_) => ProgressUnit::Bytes,
370            None => ProgressUnit::Count("files"),
371        };
372        Self {
373            bar: bar.clone(),
374            num_files,
375            files_done: AtomicU64::new(0),
376            byte_denominated: total_bytes.is_some(),
377            ticker: ProgressTicker::spawn(bar, phase, unit, None),
378        }
379    }
380
381    /// Record `n` freshly received bytes of an in-flight download.
382    pub fn add_bytes(&self, n: u64) {
383        if self.byte_denominated {
384            self.bar.inc(n);
385        }
386    }
387
388    /// Roll back bytes recorded by a failed download attempt, so the retry
389    /// doesn't count them twice.
390    pub fn remove_bytes(&self, n: u64) {
391        if self.byte_denominated {
392            // The read-modify-write races with concurrent `add_bytes` calls
393            // from other in-flight downloads (indicatif has no atomic
394            // decrement); a lost increment only skews the displayed position
395            // slightly, which is acceptable for a progress bar.
396            self.bar.set_position(self.bar.position().saturating_sub(n));
397        }
398    }
399
400    /// Record one fully downloaded file, whose bytes were already recorded
401    /// with [`Self::add_bytes`] as they arrived.
402    pub fn file_done(&self) {
403        let files_done = self.files_done.fetch_add(1, Ordering::Relaxed) + 1;
404        if self.byte_denominated {
405            self.bar
406                .set_message(format!("{files_done}/{} files", self.num_files));
407        } else {
408            self.bar.inc(1);
409        }
410    }
411
412    /// Records that both download phases are done, under the message `msg`.
413    pub fn finish_with_message(&self, msg: &'static str) {
414        self.ticker.finish_with_message(msg);
415    }
416
417    /// The underlying bar, e.g. to read its position in tests.
418    pub fn bar(&self) -> &ProgressBar {
419        &self.bar
420    }
421}

423/// Download `src` from `store` with retries, recording received chunks on
424/// `progress` as they arrive. A failed attempt's bytes are rolled back so the
425/// retry doesn't count them twice. An empty response counts as a failed
426/// attempt: no snapshot file is legitimately empty, and a store may serve one
427/// transiently.
428pub async fn get_with_progress<S: ObjectStoreGetExt>(
429    store: &S,
430    src: &Path,
431    progress: &DownloadProgressBar,
432) -> Result<Bytes> {
433    retry(backoff::ExponentialBackoff::default(), || async {
434        let attempt_bytes = AtomicU64::new(0);
435        store
436            .get_bytes_with_progress(src, &|n| {
437                attempt_bytes.fetch_add(n, Ordering::Relaxed);
438                progress.add_bytes(n);
439            })
440            .await
441            .and_then(|bytes| {
442                ensure!(!bytes.is_empty(), "Downloaded empty file {src}");
443                Ok(bytes)
444            })
445            .map_err(|e| {
446                progress.remove_bytes(attempt_bytes.load(Ordering::Relaxed));
447                error!("Failed to read file {src} from object store with error: {e:?}");
448                classify_for_retry(e)
449            })
450    })
451    .await
452}

454/// Download the single file `src` from `store` with retries, reporting the
455/// bytes as they arrive on a bar of its own under `phase` — for a file big
456/// enough that a restore shouldn't sit silent while it downloads.
457pub async fn get_single_file_with_progress<S: ObjectStoreGetExt>(
458    store: &S,
459    src: &Path,
460    phase: &'static str,
461    m: &MultiProgress,
462) -> Result<Bytes> {
463    // The size only feeds the bar's totals, so a failed lookup costs the size
464    // and ETA rather than the download.
465    let total_bytes = store.object_size(src).await.ok();
466    let progress = DownloadProgressBar::new(m, phase, 1, total_bytes);
467    let bytes = get_with_progress(store, src, &progress).await?;
468    progress.finish_with_message("Download complete");
469    Ok(bytes)
470}

472/// Copy every `(path, sha3_digest)` in `files` from `src_store` to
473/// `dest_store` in parallel, recording downloaded chunks and each completed
474/// file on `progress` and failing on the first copy that fails. A download
475/// whose bytes don't hash to its manifest digest — a truncated file, say — is
476/// rejected instead of being written to `dest_store`.
477pub async fn copy_files_with_progress<S: ObjectStoreGetExt, D: ObjectStorePutExt>(
478    files: &[(Path, [u8; 32])],
479    src_store: &S,
480    dest_store: &D,
481    concurrency: usize,
482    progress: &DownloadProgressBar,
483) -> Result<()> {
484    futures::stream::iter(files)
485        .map(|(path, sha3_digest)| async move {
486            let bytes = get_with_progress(src_store, path, progress)
487                .await
488                .with_context(|| format!("Failed to download {path}"))?;
489            let computed = Sha3_256::digest(&bytes).digest;
490            ensure!(
491                computed == *sha3_digest,
492                "Downloaded {path} does not match the manifest checksum: computed {computed:?}, \
493                 expected {sha3_digest:?}"
494            );
495            put(dest_store, path, bytes)
496                .await
497                .with_context(|| format!("Failed to write {path}"))?;
498            Ok::<_, anyhow::Error>(())
499        })
500        .boxed()
501        .buffer_unordered(concurrency)
502        .try_for_each(|()| {
503            progress.file_done();
504            futures::future::ready(Ok(()))
505        })
506        .await
507}

591    /// Everything logged by `f` on the calling thread.
592    fn captured_logs(f: impl FnOnce()) -> String {
593        #[derive(Clone)]
594        struct Sink(Arc<Mutex<Vec<u8>>>);
595
596        impl io::Write for Sink {
597            fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
598                self.0.lock().unwrap().extend_from_slice(buf);
599                Ok(buf.len())
600            }
601
602            fn flush(&mut self) -> io::Result<()> {
603                Ok(())
604            }
605        }
606
607        impl MakeWriter<'_> for Sink {
608            type Writer = Self;
609
610            fn make_writer(&self) -> Self::Writer {
611                self.clone()
612            }
613        }
614
615        let buf = Arc::new(Mutex::new(Vec::new()));
616        let subscriber = tracing_subscriber::fmt()
617            .with_writer(Sink(buf.clone()))
618            .with_ansi(false)
619            .without_time()
620            .finish();
621        tracing::subscriber::with_default(subscriber, f);
622        let logs = buf.lock().unwrap().clone();
623        String::from_utf8(logs).expect("log output is not valid UTF-8")
624    }

814    /// Store whose first download attempt goes wrong — it returns
815    /// `first_response` as is, or emits a partial chunk and then fails when
816    /// that is `None` — and whose every later attempt returns `payload`.
817    struct FlakyStore {
818        first_response: Option<Bytes>,
819        payload: Bytes,
820        failed_once: AtomicBool,
821    }
822
823    impl std::fmt::Display for FlakyStore {
824        fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
825            write!(f, "flaky")
826        }
827    }
828
829    #[async_trait]
830    impl ObjectStoreGetExt for FlakyStore {
831        async fn get_bytes(&self, src: &Path) -> Result<Bytes> {
832            self.get_bytes_with_progress(src, &|_| {}).await
833        }
834
835        async fn get_bytes_with_progress(
836            &self,
837            _src: &Path,
838            on_bytes: &(dyn Fn(u64) + Send + Sync),
839        ) -> Result<Bytes> {
840            if !self.failed_once.swap(true, Ordering::Relaxed) {
841                let Some(first_response) = &self.first_response else {
842                    on_bytes(3);
843                    anyhow::bail!("transient failure after a partial chunk");
844                };
845                on_bytes(first_response.len() as u64);
846                return Ok(first_response.clone());
847            }
848            on_bytes(self.payload.len() as u64);
849            Ok(self.payload.clone())
850        }
851
852        async fn exists(&self, _src: &Path) -> Result<bool> {
853            Ok(true)
854        }
855
856        async fn object_size(&self, _src: &Path) -> Result<u64> {
857            Ok(self.payload.len() as u64)
858        }
859    }
