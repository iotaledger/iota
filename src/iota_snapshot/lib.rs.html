66/// The following describes the format of an object file (*.obj) used for
67/// persisting live iota objects. The maximum size per .obj file is 128MB. State
68/// snapshot will be taken at the end of every epoch. Live object set is split
69/// into and stored across multiple hash buckets. The hashing function used
70/// for bucketing objects is the same as the one used to build the accumulator
71/// tree for computing state root hash. Buckets are further subdivided into
72/// partitions. A partition is a smallest storage unit which holds a subset of
73/// objects in one bucket. Each partition is a single *.obj file where
74/// objects are appended to in an append-only fashion. A new partition is
75/// created when the current one reaches its maximum size. i.e. 128MB.
76/// Partitions allow a single hash bucket to be consumed in parallel. Partition
77/// files are optionally compressed with the zstd compression format. Partition
78/// filenames follows the format <bucket_number>_<partition_number>.obj. Object
79/// references for hash. There is one single ref file per hash bucket. Object
80/// references are written in an append-only manner as well. Finally, the
81/// MANIFEST file contains per file metadata of every file in the snapshot
82/// directory.
83///
84/// Snapshot-format V2 additions over V1:
85/// - OBJECT file magic is `0x00B7EC76` (V1 was `0x00B7EC75`); a V2 reader fails
86///   fast on a V1 magic and vice versa. Encoded records are BCS-serialized
87///   `SnapshotLiveObject` carrying the per-object
88///   `previous_transaction_checkpoint` inline. The writer rejects rows whose
89///   checkpoint is `None` (lifted from pre-V2 store rows) at the publish
90///   boundary, so any record present in a published `.obj` file carries a
91///   concrete checkpoint sequence number.
92/// - REFERENCE file format is unchanged from V1.
93/// - A per-snapshot `EPOCH_INFO` file carries one [`EpochInfoV1Entry`] per
94///   epoch in `[0, snapshot_epoch]` from the CheckpointStore's `epoch_info`
95///   table. The writer refuses to publish unless that table's completeness
96///   watermark covers `snapshot_epoch`.
97/// - `MANIFEST` is now [`ManifestV2`], adding a `chain_id` field so a restore
98///   can reject a foreign-chain snapshot.
99///
100/// State Snapshot Directory Layout
101///  - snapshot/
102///     - epoch_0/
103///        - 1_1.obj
104///        - 1_2.obj
105///        - 1_3.obj
106///        - 2_1.obj
107///        - ...
108///        - 1000_1.obj
109///        - REFERENCE-1
110///        - REFERENCE-2
111///        - ...
112///        - REFERENCE-1000
113///        - EPOCH_INFO
114///        - MANIFEST
115///     - epoch_1/
116///       - 1_1.obj
117///       - ...
118///
119/// Object File Disk Format
120/// ┌──────────────────────────────┐
121/// │  magic(0x00B7EC76) <4 byte>  │
122/// ├──────────────────────────────┤
123/// │ ┌──────────────────────────┐ │
124/// │ │         Object 1         │ │
125/// │ ├──────────────────────────┤ │
126/// │ │          ...             │ │
127/// │ ├──────────────────────────┤ │
128/// │ │         Object N         │ │
129/// │ └──────────────────────────┘ │
130/// └──────────────────────────────┘
131/// Object
132/// ┌───────────────┬───────────────────┬──────────────┐
133/// │ len <uvarint> │ encoding <1 byte> │ data <bytes> │
134/// └───────────────┴───────────────────┴──────────────┘
135///
136/// REFERENCE File Disk Format
137/// ┌────────────────────────────────────┐
138/// │     magic(0xDEADBEEF) <4 byte>     │
139/// ├────────────────────────────────────┤
140/// │ ┌────────────────────────────────┐ │
141/// │ │       ObjectReference 1        │ │
142/// │ ├────────────────────────────────┤ │
143/// │ │              ...               │ │
144/// │ ├────────────────────────────────┤ │
145/// │ │       ObjectReference N        │ │
146/// │ └────────────────────────────────┘ │
147/// └────────────────────────────────────┘
148/// ObjectReference (ObjectId, Version, ObjectDigest)
149/// ┌───────────────┬───────────────────┬──────────────┐
150/// │         data (<(address_len + 8 + 32) bytes>)    │
151/// └───────────────┴───────────────────┴──────────────┘
152///
153/// EPOCH_INFO File Disk Format
154/// ┌──────────────────────────────┐
155/// │  magic(0x9000C001) <4 byte>  │
156/// ├──────────────────────────────┤
157/// │   bcs(EpochInfo)             │
158/// └──────────────────────────────┘
159/// See [`EpochInfo`] for the schema. `FileMetadata::sha3_digest` in the
160/// MANIFEST can be used to verify file integrity.
161///
162/// MANIFEST File Disk Format
163/// ┌──────────────────────────────┐
164/// │  magic(0x00C0FFEE) <4 byte>  │
165/// ├──────────────────────────────┤
166/// │   serialized manifest        │
167/// ├──────────────────────────────┤
168/// │      sha3 <32 bytes>         │
169/// └──────────────────────────────┘
170const OBJECT_FILE_MAGIC: u32 = 0x00B7EC76;
171const REFERENCE_FILE_MAGIC: u32 = 0xDEADBEEF;
172const EPOCH_INFO_FILE_MAGIC: u32 = 0x9000C001;
173const MANIFEST_FILE_MAGIC: u32 = 0x00C0FFEE;
174const MAGIC_BYTES: usize = 4;
175const SNAPSHOT_VERSION_BYTES: usize = 1;
176const ADDRESS_LENGTH_BYTES: usize = 8;
177const PADDING_BYTES: usize = 3;
178const MANIFEST_FILE_HEADER_BYTES: usize =
179    MAGIC_BYTES + SNAPSHOT_VERSION_BYTES + ADDRESS_LENGTH_BYTES + PADDING_BYTES;
180const FILE_MAX_BYTES: usize = 128 * 1024 * 1024;
181const OBJECT_ID_BYTES: usize = ObjectId::LENGTH;
182const SEQUENCE_NUM_BYTES: usize = 8;
183const OBJECT_DIGEST_BYTES: usize = 32;
184const OBJECT_REF_BYTES: usize = OBJECT_ID_BYTES + SEQUENCE_NUM_BYTES + OBJECT_DIGEST_BYTES;
185const FILE_TYPE_BYTES: usize = 1;
186const BUCKET_BYTES: usize = 4;
187const BUCKET_PARTITION_BYTES: usize = 4;
188const COMPRESSION_TYPE_BYTES: usize = 1;
189const FILE_METADATA_BYTES: usize =
190    FILE_TYPE_BYTES + BUCKET_BYTES + BUCKET_PARTITION_BYTES + COMPRESSION_TYPE_BYTES + SHA3_BYTES;

192pub fn default_download_concurrency() -> NonZeroUsize {
193    const MAX_PARALLEL_DOWNLOADS: usize = 8;
194    let cores = std::thread::available_parallelism().map_or(1, NonZeroUsize::get);
195    NonZeroUsize::new(cores.saturating_sub(1).clamp(1, MAX_PARALLEL_DOWNLOADS))
196        .unwrap_or(NonZeroUsize::MIN)
197}

210#[derive(Debug, Clone, Serialize, Deserialize, Eq, PartialEq)]
211/// FileMetadata holds either an object or a reference file metadata.
212pub struct FileMetadata {
213    pub file_type: FileType,
214    pub bucket_num: u32,
215    pub part_num: u32,
216    pub file_compression: FileCompression,
217    pub sha3_digest: [u8; 32],
218}
219
220impl FileMetadata {
221    pub fn file_path(&self, dir_path: &Path) -> Path {
222        match self.file_type {
223            FileType::Object => {
224                dir_path.child(&*format!("{}_{}.obj", self.bucket_num, self.part_num))
225            }
226            FileType::Reference => {
227                dir_path.child(&*format!("{}_{}.ref", self.bucket_num, self.part_num))
228            }
229            // EPOCH_INFO is a singleton per snapshot, so bucket/part numbers
230            // do not contribute to the filename.
231            FileType::EpochInfo => dir_path.child("EPOCH_INFO"),
232        }
233    }
234    pub fn local_file_path(&self, root_path: &std::path::Path, dir_path: &Path) -> Result<PathBuf> {
235        path_to_filesystem(root_path.to_path_buf(), &self.file_path(dir_path))
236    }
237}

325/// Chain-verified `EPOCH_INFO`: the `chain_id` matched and every entry was
326/// anchored to its certified `last_checkpoint_summary` — the committee chain
327/// walked from the genesis committee, every byte hash-checked back to that
328/// signed summary. The only constructor is `verify_epoch_info_chain`, so
329/// holding one is proof the data is anchored to the operator-provided genesis.
330#[derive(Debug)]
331pub struct VerifiedEpochInfo {
332    epoch_info: EpochInfo,
333    committees: Vec<Committee>,
334    start_system_states: Vec<IotaSystemState>,
335}
336
337impl VerifiedEpochInfo {
338    /// Entries for epochs `[0, snapshot_epoch]`, in epoch order.
339    ///
340    /// These values represent data at the boundary of each epoch, including the
341    /// committee and start system state of the next epoch.
342    ///
343    /// The last entry represents the snapshot epoch boundary, which carries
344    /// information about the next epoch. Thus [`Self::committees`], and
345    /// [`Self::start_system_states`] have one more item for the committee and
346    /// the start system state of the epoch following the snapshot epoch.
347    pub fn entries(&self) -> &[EpochInfoV1Entry] {
348        self.epoch_info.entries()
349    }
350
351    /// Committees for epochs `[0, snapshot_epoch + 1]`: the genesis committee
352    /// plus one handed forward by each entry's `end_of_epoch_data`.
353    pub fn committees(&self) -> &[Committee] {
354        &self.committees
355    }
356
357    /// Digest-verified start system state per epochs `[0, snapshot_epoch + 1].
358    ///
359    /// Contains the genesis start system state, plus the ones derived by the
360    /// epoch boundaries represented by [`Self::entries`].
361    pub fn start_system_states(&self) -> &[IotaSystemState] {
362        &self.start_system_states
363    }
364
365    /// Consumes the verified info into its components: the epoch info entries,
366    /// the per-epoch committees, and the per-epoch start system states.
367    ///
368    /// See [`Self::entries`], [`Self::committees`], and
369    /// [`Self::start_system_states`].
370    pub fn into_parts(self) -> (EpochInfo, Vec<Committee>, Vec<IotaSystemState>) {
371        (self.epoch_info, self.committees, self.start_system_states)
372    }
373
374    /// Convert the verified entries `[0, snapshot_epoch]` into `EpochInfoV2`
375    /// index rows. Each row's `system_state` is the digest-verified start state
376    /// of its own epoch (`start_system_states[i]`).
377    pub(crate) fn into_epoch_info_v2_rows(self) -> Vec<EpochInfoV2> {
378        let VerifiedEpochInfo {
379            epoch_info,
380            start_system_states,
381            ..
382        } = self;
383        // `start_system_states[i]` is epoch `i`'s start state; `zip` drops the
384        // trailing one (the state the last boundary proves, which has no row).
385        // Each epoch's start checkpoint is the previous epoch's last + 1 (0 for
386        // epoch 0), derived inline from the signed summaries.
387        let mut previous_end_checkpoint: Option<u64> = None;
388        epoch_info
389            .into_entries()
390            .into_iter()
391            .zip(start_system_states)
392            .map(|(entry, start_system_state)| {
393                let start_checkpoint = previous_end_checkpoint.map_or(0, |seq| seq + 1);
394                previous_end_checkpoint =
395                    Some(entry.last_checkpoint_summary.data().sequence_number());
396                epoch_info_v2_row(entry, start_system_state, start_checkpoint)
397            })
398            .collect()
399    }
400
401    /// Restore the verified rows `[0, snapshot_epoch]` into the given
402    /// consumer's epoch store.
403    pub async fn restore_epoch_info(self, db: &impl RestoreEpochInfo) -> anyhow::Result<()> {
404        let rows = self.into_epoch_info_v2_rows();
405        db.restore_epoch_info(rows).await
406    }
407}

409/// Verify a snapshot's `EPOCH_INFO` against the operator's trust roots: the
410/// expected `chain_id`, the committee chain walked from `genesis_committee`,
411/// and `genesis_system_state` (epoch 0's start state, which no entry proves).
412/// Each entry must be the contiguous certified close of its epoch, signed by
413/// the committee the previous entry handed forward, with its proof bundle
414/// hashing back to the signed summary (see `verify_epoch_boundary_proof`).
415/// Nothing is written; the returned `VerifiedEpochInfo` is the witness
416/// consumers require.
417pub fn verify_epoch_info_chain(
418    epoch_info: EpochInfo,
419    genesis_committee: Committee,
420    genesis_system_state: IotaSystemState,
421    snapshot_chain_id: ChainIdentifier,
422    expected_chain_id: ChainIdentifier,
423) -> anyhow::Result<VerifiedEpochInfo> {
424    anyhow::ensure!(
425        snapshot_chain_id == expected_chain_id,
426        "snapshot chain_id {snapshot_chain_id} does not match this node's chain \
427         {expected_chain_id} (snapshot from the wrong network's bucket?)"
428    );
429    anyhow::ensure!(
430        genesis_committee.epoch == 0,
431        "the trust root must be the genesis committee, got epoch {}",
432        genesis_committee.epoch
433    );
434
435    let mut chain_verifier = CommitteeChainVerifier::new(genesis_committee);
436    let mut committees = vec![chain_verifier.committee().clone()];
437    // `start_system_states[i]` is epoch `i`'s start state; epoch 0's is the
438    // genesis root, every later one is derived from the previous boundary.
439    let mut start_system_states = vec![genesis_system_state];
440    for (index, entry) in epoch_info.entries().iter().enumerate() {
441        // EPOCH_INFO-specific: entries must be the contiguous epochs from 0.
442        // The epoch comes from the signed summary, not a stored field, so it
443        // can't disagree with the data it anchors.
444        let epoch = entry.last_checkpoint_summary.epoch();
445        anyhow::ensure!(
446            epoch == index as u64,
447            "EPOCH_INFO entry at index {index} carries a summary for epoch {epoch}",
448        );
449
450        // Defense in depth: the committee in epoch `index`'s start state must
451        // match the one the chain certified — catches a tampered validator set
452        // that left the (separately signed) committee handover intact.
453        let start_committee = start_system_states[index].get_current_epoch_committee();
454        anyhow::ensure!(
455            start_committee.committee() == chain_verifier.committee(),
456            "EPOCH_INFO entry for epoch {index}: the committee in its start system \
457             state does not match the certified committee",
458        );
459
460        chain_verifier
461            .verify_epoch_close(entry.last_checkpoint_summary.clone())
462            .map_err(|e| {
463                anyhow::anyhow!("EPOCH_INFO entry for epoch {index} failed verification: {e}")
464            })?;
465
466        // Anchor the rest of the entry to the now-verified summary and derive
467        // epoch `index + 1`'s start state from the boundary objects.
468        let next_start_state = verify_epoch_boundary_proof(entry)
469            .map_err(|e| anyhow::anyhow!("EPOCH_INFO entry for epoch {index}: {e}"))?;
470
471        committees.push(chain_verifier.committee().clone());
472        start_system_states.push(next_start_state);
473    }
474
475    Ok(VerifiedEpochInfo {
476        epoch_info,
477        committees,
478        start_system_states,
479    })
480}

482/// Anchor an entry's proof bundle to its (already signature-verified)
483/// `last_checkpoint_summary` and return the next epoch's digest-verified start
484/// state — the system-state objects this boundary wrote. Each link (contents,
485/// effects, events, start-state objects) is checked below.
486fn verify_epoch_boundary_proof(entry: &EpochInfoV1Entry) -> anyhow::Result<IotaSystemState> {
487    let summary = entry.last_checkpoint_summary.data();
488
489    // 1. Contents hash to the signed summary.
490    anyhow::ensure!(
491        entry.last_checkpoint_contents.digest() == summary.contents_digest,
492        "last_checkpoint_contents does not hash to the signed contents_digest",
493    );
494
495    // 2. The epoch-change effects are the last tx of the verified contents.
496    let expected_execution_digest = entry
497        .last_checkpoint_contents
498        .transactions()
499        .last()
500        .map(|info| ExecutionDigests::new(info.transaction, info.effects))
501        .ok_or_else(|| anyhow::anyhow!("the closing checkpoint has no transactions"))?;
502    let effects = &entry.end_of_epoch_tx_effects;
503    anyhow::ensure!(
504        effects.execution_digests() == expected_execution_digest,
505        "end_of_epoch_tx_effects digest pair does not match the closing checkpoint's last transaction",
506    );
507
508    // 3. Events hash to the effects' events_digest (`None` ⇒ events empty, the
509    // safe-mode boundary case).
510    match effects.events_digest() {
511        Some(events_digest) => anyhow::ensure!(
512            entry.end_of_epoch_tx_events.digest() == *events_digest,
513            "end_of_epoch_tx_events does not hash to the effects' events_digest",
514        ),
515        None => anyhow::ensure!(
516            entry.end_of_epoch_tx_events.is_empty(),
517            "the epoch-change effects carry no events_digest but \
518             end_of_epoch_tx_events is non-empty",
519        ),
520    }
521
522    // 4. Each start-state object's digest is one the effects wrote; `0x5` must
523    // be present. Decode `IotaSystemState` only from these verified bytes.
524    let written: HashSet<ObjectReference> = effects
525        .all_changed_objects()
526        .into_iter()
527        .map(|(changed, _)| changed.reference)
528        .collect();
529    let mut objects = Vec::with_capacity(entry.next_epoch_start_system_state_objects.len());
530    for raw in &entry.next_epoch_start_system_state_objects {
531        let object: Object = bcs::from_bytes(raw)
532            .map_err(|e| anyhow::anyhow!("decoding a next-epoch start-state object: {e}"))?;
533        anyhow::ensure!(
534            written.contains(&object.object_ref()),
535            "a next-epoch start-state object is not written by the epoch-change tx",
536        );
537        objects.push(object);
538    }
539    anyhow::ensure!(
540        objects
541            .iter()
542            .any(|o| o.id() == IOTA_SYSTEM_STATE_OBJECT_ID),
543        "the next-epoch start-state objects do not include the system-state object 0x5",
544    );
545    get_iota_system_state(&objects.as_slice())
546        .map_err(|e| anyhow::anyhow!("decoding the next-epoch system state: {e}"))
547}

549/// Build an `EpochInfoV2` index row from a verified entry, its digest-verified
550/// start system state, and its start checkpoint. The `epoch` comes from the
551/// entry's signed summary; the `end_*` facts are derived from the embedded
552/// entry by `EpochInfoV2`'s methods.
553fn epoch_info_v2_row(
554    entry: EpochInfoV1Entry,
555    system_state: IotaSystemState,
556    start_checkpoint: CheckpointSequenceNumber,
557) -> EpochInfoV2 {
558    EpochInfoV2 {
559        epoch: entry.last_checkpoint_summary.epoch(),
560        start_checkpoint,
561        start_timestamp_ms: system_state.epoch_start_timestamp_ms(),
562        system_state,
563        epoch_close_proof: Some(entry),
564    }
565}

567/// Creates a FileMetadata of the provided file path, which is overwritten with
568/// compressed data of the original file.
569pub fn create_file_metadata(
570    file_path: &std::path::Path,
571    file_compression: FileCompression,
572    file_type: FileType,
573    bucket_num: u32,
574    part_num: u32,
575) -> Result<FileMetadata> {
576    // Overwrites the file with compressed data of the original file.
577    file_compression.compress(file_path)?;
578    // Computes the sha3 checksum of the compressed file.
579    let sha3_digest = compute_sha3_checksum(file_path)?;
580    let file_metadata = FileMetadata {
581        file_type,
582        bucket_num,
583        part_num,
584        file_compression,
585        sha3_digest,
586    };
587    Ok(file_metadata)
588}

590pub async fn setup_db_state(
591    epoch: u64,
592    state_hash: GlobalStateHash,
593    perpetual_db: Arc<AuthorityPerpetualTables>,
594    checkpoint_store: Arc<CheckpointStore>,
595    committee_store: Arc<CommitteeStore>,
596    verify: bool,
597    num_live_objects: u64,
598    m: MultiProgress,
599) -> Result<()> {
600    // This function should be called once state accumulator based hash verification
601    // is complete and live object set state is downloaded to local store
602    let system_state_object = get_iota_system_state(&perpetual_db)?;
603    let new_epoch_start_state = system_state_object.into_epoch_start_state();
604    let next_epoch_committee = new_epoch_start_state.get_iota_committee();
605    let root_digest: ECMHLiveObjectSetDigest = state_hash.digest().into();
606    let last_checkpoint = checkpoint_store
607        .get_epoch_last_checkpoint(epoch)
608        .expect("Error loading last checkpoint for current epoch")
609        .expect("Could not load last checkpoint for current epoch");
610    let flags = EpochFlag::default_for_no_config();
611    let epoch_start_configuration = EpochStartConfiguration::new(
612        new_epoch_start_state,
613        *last_checkpoint.digest(),
614        &perpetual_db,
615        flags,
616    )
617    .unwrap();
618    perpetual_db.set_epoch_start_configuration(&epoch_start_configuration)?;
619    perpetual_db.insert_root_state_hash(epoch, last_checkpoint.sequence_number, state_hash)?;
620    perpetual_db.set_highest_pruned_checkpoint_without_wb(last_checkpoint.sequence_number)?;
621    committee_store.insert_new_committee(&next_epoch_committee)?;
622    checkpoint_store.update_highest_executed_checkpoint(&last_checkpoint)?;
623
624    if verify {
625        let iter = perpetual_db.iter_live_object_set();
626        let local_digest = ECMHLiveObjectSetDigest::from(
627            accumulate_live_object_iter(Box::new(iter), m.clone(), num_live_objects)
628                .await
629                .digest(),
630        );
631        assert_eq!(
632            root_digest, local_digest,
633            "End of epoch {} root state digest {} does not match \
634                local root state hash {} after restoring db from formal snapshot",
635            epoch, root_digest.digest, local_digest.digest,
636        );
637        println!("DB live object state verification completed successfully!");
638    }
639
640    Ok(())
641}

643pub async fn accumulate_live_object_iter(
644    iter: Box<dyn Iterator<Item = LiveObject> + '_>,
645    m: MultiProgress,
646    num_live_objects: u64,
647) -> GlobalStateHash {
648    // Monitor progress of live object accumulation, whose position the ticker
649    // takes from the counter the loop below advances
650    let accum_counter = Arc::new(AtomicU64::new(0));
651    let accum_ticker = ProgressTicker::spawn(
652        m.add(
653            ProgressBar::new(num_live_objects).with_style(
654                ProgressStyle::with_template(
655                    "[{elapsed_precise}] {wide_bar} Accumulating DB live objects: {pos}/{len} \
656                     ({per_sec}, ETA {eta})",
657                )
658                .unwrap(),
659            ),
660        ),
661        "Accumulating DB live objects",
662        ProgressUnit::Count("objects"),
663        Some(accum_counter.clone()),
664    );
665
666    // Accumulate live objects
667    let mut acc = GlobalStateHash::default();
668    for live_object in iter {
669        GlobalStateHasher::accumulate_live_object(&mut acc, &live_object);
670        accum_counter.fetch_add(1, Ordering::Relaxed);
671    }
672    let num_accumulated = accum_counter.load(Ordering::Relaxed);
673    assert!(
674        num_accumulated <= num_live_objects,
675        "Accumulated more objects ({num_accumulated}) than expected ({num_live_objects})"
676    );
677    accum_ticker.finish_with_message("DB live object accumulation completed");
678    acc
