156    pub fn add_validator_signature(mut self, keypair: &AuthorityKeyPair) -> Self {
157        let name = keypair.public().into();
158        assert!(
159            self.validators.contains_key(&name),
160            "provided keypair does not correspond to a validator in the validator set"
161        );
162
163        let UnsignedGenesis { checkpoint, .. } = self.get_or_build_unsigned_genesis();
164
165        let checkpoint_signature = {
166            let intent_msg = IntentMessage::new(
167                Intent::iota_app(IntentScope::CheckpointSummary),
168                checkpoint.clone(),
169            );
170            let signature = AuthoritySignature::new_secure(&intent_msg, &checkpoint.epoch, keypair);
171            AuthoritySignInfo {
172                epoch: checkpoint.epoch,
173                authority: name,
174                signature,
175            }
176        };
177
178        self.signatures.insert(name, checkpoint_signature);
179
180        self
181    }

187    /// Evaluate the genesis [`TokenDistributionSchedule`]: use the schedule
188    /// given as input, if any, or instantiate a default schedule for the
189    /// validators otherwise.
190    fn resolve_token_distribution_schedule(&mut self) -> TokenDistributionSchedule {
191        self.token_distribution_schedule.take().unwrap_or_else(|| {
192            TokenDistributionSchedule::new_for_validators_with_default_allocation(
193                self.validators.values().map(|v| v.info.iota_address()),
194                self.parameters.protocol_version,
195            )
196        })
197    }
198
199    fn build_and_cache_unsigned_genesis(&mut self) {
200        // Verify that all input data is valid.
201        // Check that if extra objects are present then it is allowed by the parameters
202        // to add extra objects and it also validates the validator info
203        self.validate_inputs().unwrap();
204
205        let token_distribution_schedule = self.resolve_token_distribution_schedule();
206
207        // Verify that token distribution schedule is valid
208        token_distribution_schedule.validate();
209        token_distribution_schedule
210            .check_minimum_stake_for_validators(
211                self.validators.values().map(|v| v.info.iota_address()),
212                self.parameters.protocol_version,
213            )
214            .expect("all validators should have the required stake");
215
216        let unsigned_genesis = build_unsigned_genesis_data(
217            &self.parameters,
218            &token_distribution_schedule,
219            self.validators.values(),
220            self.objects.clone().into_values().collect::<Vec<_>>(),
221        );
222
223        self.built_genesis = Some(unsigned_genesis);
224        self.token_distribution_schedule = Some(token_distribution_schedule);
225    }

227    pub fn get_or_build_unsigned_genesis(&mut self) -> &UnsignedGenesis {
228        if self.built_genesis.is_none() {
229            self.build_and_cache_unsigned_genesis();
230        }
231        self.built_genesis
232            .as_ref()
233            .expect("genesis should have been built and cached")
234    }

249    pub fn build(mut self) -> Genesis {
250        if self.built_genesis.is_none() {
251            self.build_and_cache_unsigned_genesis();
252        }
253
254        // Verify that all on-chain state was properly created
255        self.validate().unwrap();
256
257        let UnsignedGenesis {
258            checkpoint,
259            checkpoint_contents,
260            transaction,
261            effects,
262            events,
263            objects,
264        } = self
265            .built_genesis
266            .take()
267            .expect("genesis should have been built");
268
269        let committee = Self::committee(&objects);
270
271        let checkpoint = {
272            let signatures = self.signatures.clone().into_values().collect();
273
274            CertifiedCheckpointSummary::new(checkpoint, signatures, &committee).unwrap()
275        };
276
277        Genesis::new(
278            checkpoint,
279            checkpoint_contents,
280            transaction,
281            effects,
282            events,
283            objects,
284        )
285    }

287    /// Validates the entire state of the build, no matter what the internal
288    /// state is (input collection phase or output phase)
289    pub fn validate(&self) -> anyhow::Result<(), anyhow::Error> {
290        self.validate_inputs()?;
291        self.validate_token_distribution_schedule()?;
292        self.validate_output();
293        Ok(())
294    }
295
296    /// Runs through validation checks on the input values present in the
297    /// builder
298    fn validate_inputs(&self) -> anyhow::Result<(), anyhow::Error> {
299        if !self.parameters.allow_insertion_of_extra_objects && !self.objects.is_empty() {
300            bail!("extra objects are disallowed");
301        }
302
303        for validator in self.validators.values() {
304            validator.validate().with_context(|| {
305                format!(
306                    "metadata for validator {} is invalid",
307                    validator.info.name()
308                )
309            })?;
310        }
311
312        Ok(())
313    }
314
315    /// Runs through validation checks on the input token distribution schedule
316    fn validate_token_distribution_schedule(&self) -> anyhow::Result<(), anyhow::Error> {
317        if let Some(token_distribution_schedule) = &self.token_distribution_schedule {
318            token_distribution_schedule.validate();
319            token_distribution_schedule.check_minimum_stake_for_validators(
320                self.validators.values().map(|v| v.info.iota_address()),
321                self.parameters.protocol_version,
322            )?;
323        }
324
325        Ok(())
326    }

328    /// Runs through validation checks on the generated output (the initial
329    /// chain state) based on the input values present in the builder
330    fn validate_output(&self) {
331        // If genesis hasn't been built yet, just early return as there is nothing to
332        // validate yet
333        let Some(unsigned_genesis) = self.unsigned_genesis_checkpoint() else {
334            return;
335        };
336
337        let GenesisChainParameters {
338            protocol_version,
339            chain_start_timestamp_ms,
340            epoch_duration_ms,
341            max_validator_count,
342            min_validator_joining_stake,
343            validator_low_stake_threshold,
344            validator_very_low_stake_threshold,
345            validator_low_stake_grace_period,
346        } = self.parameters.to_genesis_chain_parameters();
347
348        // In non-testing code, genesis type must always be V1.
349        let system_state = match unsigned_genesis.iota_system_object() {
350            IotaSystemState::V1(inner) => inner,
351            IotaSystemState::V2(_) => unreachable!(),
352            #[cfg(msim)]
353            _ => {
354                // Types other than V1 used in simtests do not need to be validated.
355                return;
356            }
357        };
358
359        assert!(unsigned_genesis.has_randomness_state_object());
360
361        assert!(unsigned_genesis.has_coin_deny_list_object());
362
363        assert_eq!(
364            self.validators.len(),
365            system_state.validators.active_validators.len()
366        );
367        let mut address_to_pool_id = BTreeMap::new();
368        for (validator, onchain_validator) in self
369            .validators
370            .values()
371            .zip(system_state.validators.active_validators.iter())
372        {
373            let metadata = onchain_validator.verified_metadata();
374
375            // Validators should not have duplicate addresses so the result of insertion
376            // should be None.
377            assert!(
378                address_to_pool_id
379                    .insert(metadata.iota_address, onchain_validator.staking_pool.id)
380                    .is_none()
381            );
382            assert_eq!(validator.info.iota_address(), metadata.iota_address);
383            assert_eq!(validator.info.authority_key(), metadata.iota_pubkey_bytes());
384            assert_eq!(validator.info.network_key, metadata.network_pubkey);
385            assert_eq!(validator.info.protocol_key, metadata.protocol_pubkey);
386            assert_eq!(
387                validator.proof_of_possession.as_ref().to_vec(),
388                metadata.proof_of_possession_bytes
389            );
390            assert_eq!(validator.info.name(), &metadata.name);
391            assert_eq!(validator.info.description, metadata.description);
392            assert_eq!(validator.info.image_url, metadata.image_url);
393            assert_eq!(validator.info.project_url, metadata.project_url);
394            assert_eq!(validator.info.network_address(), &metadata.net_address);
395            assert_eq!(validator.info.p2p_address, metadata.p2p_address);
396            assert_eq!(validator.info.primary_address, metadata.primary_address);
397
398            assert_eq!(validator.info.gas_price, onchain_validator.gas_price);
399            assert_eq!(
400                validator.info.commission_rate,
401                onchain_validator.commission_rate
402            );
403        }
404
405        assert_eq!(system_state.epoch, 0);
406        assert_eq!(system_state.protocol_version, protocol_version);
407        assert_eq!(system_state.storage_fund.non_refundable_balance.value(), 0);
408        assert_eq!(
409            system_state
410                .storage_fund
411                .total_object_storage_rebates
412                .value(),
413            0
414        );
415
416        assert_eq!(system_state.parameters.epoch_duration_ms, epoch_duration_ms);
417        assert_eq!(
418            system_state.parameters.max_validator_count,
419            max_validator_count,
420        );
421        assert_eq!(
422            system_state.parameters.min_validator_joining_stake,
423            min_validator_joining_stake,
424        );
425        assert_eq!(
426            system_state.parameters.validator_low_stake_threshold,
427            validator_low_stake_threshold,
428        );
429        assert_eq!(
430            system_state.parameters.validator_very_low_stake_threshold,
431            validator_very_low_stake_threshold,
432        );
433        assert_eq!(
434            system_state.parameters.validator_low_stake_grace_period,
435            validator_low_stake_grace_period,
436        );
437
438        assert!(!system_state.safe_mode);
439        assert_eq!(
440            system_state.epoch_start_timestamp_ms,
441            chain_start_timestamp_ms,
442        );
443        assert_eq!(system_state.validators.pending_removals.len(), 0);
444        assert_eq!(
445            system_state
446                .validators
447                .pending_active_validators
448                .contents
449                .size,
450            0
451        );
452        assert_eq!(system_state.validators.inactive_validators.size, 0);
453        assert_eq!(system_state.validators.validator_candidates.size, 0);
454
455        // Check distribution is correct
456        let token_distribution_schedule = self.token_distribution_schedule.clone().unwrap();
457
458        let allocations_amount: u64 = token_distribution_schedule
459            .allocations
460            .iter()
461            .map(|allocation| allocation.amount_nanos)
462            .sum();
463
464        assert_eq!(
465            system_state.iota_treasury_cap.total_supply().value,
466            token_distribution_schedule.pre_minted_supply + allocations_amount
467        );
468
469        let mut gas_objects: BTreeMap<ObjectId, (&Object, GasCoin)> = unsigned_genesis
470            .objects()
471            .iter()
472            .filter_map(|o| GasCoin::try_from(o).ok().map(|g| (o.id(), (o, g))))
473            .collect();
474        let mut staked_iota_objects: BTreeMap<ObjectId, (&Object, StakedIota)> = unsigned_genesis
475            .objects()
476            .iter()
477            .filter_map(|o| StakedIota::try_from(o).ok().map(|s| (o.id(), (o, s))))
478            .collect();
479
480        for allocation in token_distribution_schedule.allocations {
481            if let Some(staked_with_validator) = allocation.staked_with_validator {
482                let staking_pool_id = *address_to_pool_id
483                    .get(&staked_with_validator)
484                    .expect("staking pool should exist");
485                let staked_iota_object_id = staked_iota_objects
486                    .iter()
487                    .find(|(_k, (o, s))| {
488                        let Owner::Address(owner) = &o.owner else {
489                            panic!("gas object owner must be address owner");
490                        };
491                        *owner == allocation.recipient_address
492                            && s.principal() == allocation.amount_nanos
493                            && s.pool_id() == staking_pool_id
494                    })
495                    .map(|(k, _)| *k)
496                    .expect("all allocations should be present");
497                let staked_iota_object =
498                    staked_iota_objects.remove(&staked_iota_object_id).unwrap();
499                assert_eq!(
500                    staked_iota_object.0.owner,
501                    Owner::Address(allocation.recipient_address)
502                );
503                assert_eq!(staked_iota_object.1.principal(), allocation.amount_nanos);
504                assert_eq!(staked_iota_object.1.pool_id(), staking_pool_id);
505                assert_eq!(staked_iota_object.1.activation_epoch(), 0);
506            } else {
507                let gas_object_id = gas_objects
508                    .iter()
509                    .find(|(_k, (o, g))| {
510                        if let Owner::Address(owner) = &o.owner {
511                            *owner == allocation.recipient_address
512                                && g.value() == allocation.amount_nanos
513                        } else {
514                            false
515                        }
516                    })
517                    .map(|(k, _)| *k)
518                    .expect("all allocations should be present");
519                let gas_object = gas_objects.remove(&gas_object_id).unwrap();
520                assert_eq!(
521                    gas_object.0.owner,
522                    Owner::Address(allocation.recipient_address)
523                );
524                assert_eq!(gas_object.1.value(), allocation.amount_nanos,);
525            }
526        }
527
528        // All Gas and staked objects should be accounted for
529        if !self.parameters.allow_insertion_of_extra_objects {
530            assert!(gas_objects.is_empty());
531            assert!(staked_iota_objects.is_empty());
532        }
533
534        let committee = system_state.get_current_epoch_committee();
535        for signature in self.signatures.values() {
536            if !self.validators.contains_key(&signature.authority) {
537                panic!("found signature for unknown validator: {signature:#?}");
538            }
539
540            signature
541                .verify_secure(
542                    unsigned_genesis.checkpoint(),
543                    Intent::iota_app(IntentScope::CheckpointSummary),
544                    committee.committee(),
545                )
546                .expect("signature should be valid");
547        }
548    }

550    pub fn load<P: AsRef<Path>>(path: P) -> anyhow::Result<Self, anyhow::Error> {
551        let path = path.as_ref();
552        let path: &Utf8Path = path.try_into()?;
553        trace!("Reading Genesis Builder from {}", path);
554
555        if !path.is_dir() {
556            bail!("path must be a directory");
557        }
558
559        // Load parameters
560        let parameters_file = path.join(GENESIS_BUILDER_PARAMETERS_FILE);
561        let parameters = serde_yaml::from_slice(&fs::read(&parameters_file).context(format!(
562            "unable to read genesis parameters file {parameters_file}"
563        ))?)
564        .context("unable to deserialize genesis parameters")?;
565
566        let token_distribution_schedule_file =
567            path.join(GENESIS_BUILDER_TOKEN_DISTRIBUTION_SCHEDULE_FILE);
568        let token_distribution_schedule = if token_distribution_schedule_file.exists() {
569            Some(TokenDistributionSchedule::from_csv(fs::File::open(
570                token_distribution_schedule_file,
571            )?)?)
572        } else {
573            None
574        };
575
576        // Load validator infos
577        let mut committee = BTreeMap::new();
578        for entry in path.join(GENESIS_BUILDER_COMMITTEE_DIR).read_dir_utf8()? {
579            let entry = entry?;
580            if entry.file_name().starts_with('.') {
581                continue;
582            }
583
584            let path = entry.path();
585            let validator_info: GenesisValidatorInfo = serde_yaml::from_slice(&fs::read(path)?)
586                .with_context(|| format!("unable to load validator info for {path}"))?;
587            committee.insert(validator_info.info.authority_key(), validator_info);
588        }
589
590        // Load Signatures
591        let mut signatures = BTreeMap::new();
592        for entry in path.join(GENESIS_BUILDER_SIGNATURE_DIR).read_dir_utf8()? {
593            let entry = entry?;
594            if entry.file_name().starts_with('.') {
595                continue;
596            }
597
598            let path = entry.path();
599            let sigs: AuthoritySignInfo = bcs::from_bytes(&fs::read(path)?)
600                .with_context(|| format!("unable to load validator signature for {path}"))?;
601            signatures.insert(sigs.authority, sigs);
602        }
603
604        let mut builder = Self {
605            parameters,
606            token_distribution_schedule,
607            objects: Default::default(),
608            validators: committee,
609            signatures,
610            built_genesis: None, // Leave this as none, will build and compare below
611        };
612
613        let unsigned_genesis_file = path.join(GENESIS_BUILDER_UNSIGNED_GENESIS_FILE);
614        if unsigned_genesis_file.exists() {
615            let reader = BufReader::new(File::open(unsigned_genesis_file)?);
616            let loaded_genesis: UnsignedGenesis = bcs::from_reader(reader)?;
617
618            // If we have a built genesis, then we must have a token_distribution_schedule
619            // present as well.
620            assert!(
621                builder.token_distribution_schedule.is_some(),
622                "If a built genesis is present, then there must also be a token-distribution-schedule present"
623            );
624
625            // Verify loaded genesis matches one build from the constituent parts
626            loaded_genesis.checkpoint_contents.digest(); // cache digest before compare
627            assert!(
628                *builder.get_or_build_unsigned_genesis() == loaded_genesis,
629                "loaded genesis does not match built genesis"
630            );
631
632            // Just to double check that its set after building above
633            assert!(builder.unsigned_genesis_checkpoint().is_some());
634        }
635
636        Ok(builder)
637    }

639    pub fn save<P: AsRef<Path>>(self, path: P) -> anyhow::Result<(), anyhow::Error> {
640        let path = path.as_ref();
641        trace!("Writing Genesis Builder to {}", path.display());
642
643        fs::create_dir_all(path)?;
644
645        // Write parameters
646        let parameters_file = path.join(GENESIS_BUILDER_PARAMETERS_FILE);
647        fs::write(parameters_file, serde_yaml::to_string(&self.parameters)?)?;
648
649        if let Some(token_distribution_schedule) = &self.token_distribution_schedule {
650            token_distribution_schedule.to_csv(fs::File::create(
651                path.join(GENESIS_BUILDER_TOKEN_DISTRIBUTION_SCHEDULE_FILE),
652            )?)?;
653        }
654
655        // Write Signatures
656        let signature_dir = path.join(GENESIS_BUILDER_SIGNATURE_DIR);
657        std::fs::create_dir_all(&signature_dir)?;
658        for (pubkey, sigs) in self.signatures {
659            let name = self.validators.get(&pubkey).unwrap().info.name();
660            fs::write(signature_dir.join(name), &bcs::to_bytes(&sigs)?)?;
661        }
662
663        // Write validator infos
664        let committee_dir = path.join(GENESIS_BUILDER_COMMITTEE_DIR);
665        fs::create_dir_all(&committee_dir)?;
666
667        for (_pubkey, validator) in self.validators {
668            fs::write(
669                committee_dir.join(validator.info.name()),
670                &serde_yaml::to_string(&validator)?,
671            )?;
672        }
673
674        if let Some(genesis) = &self.built_genesis {
675            let mut write = BufWriter::new(File::create(
676                path.join(GENESIS_BUILDER_UNSIGNED_GENESIS_FILE),
677            )?);
678            bcs::serialize_into(&mut write, &genesis)?;
679        }
680
681        Ok(())
682    }

685// Create a Genesis Txn Context to be used when generating genesis objects by
686// hashing all of the inputs into genesis ans using that as our "Txn Digest".
687// This is done to ensure that coin objects created between chains are unique
688fn create_genesis_context(
689    epoch_data: &EpochData,
690    genesis_chain_parameters: &GenesisChainParameters,
691    genesis_validators: &[GenesisValidatorMetadata],
692    token_distribution_schedule: &TokenDistributionSchedule,
693    system_packages: &[SystemPackage],
694    protocol_config: &ProtocolConfig,
695) -> Rc<RefCell<TxContext>> {
696    let mut hasher = DefaultHash::default();
697    hasher.update(b"iota-genesis");
698    hasher.update(bcs::to_bytes(genesis_chain_parameters).unwrap());
699    hasher.update(bcs::to_bytes(genesis_validators).unwrap());
700    hasher.update(bcs::to_bytes(token_distribution_schedule).unwrap());
701    for system_package in system_packages {
702        hasher.update(bcs::to_bytes(&system_package.bytes).unwrap());
703    }
704
705    let hash = hasher.finalize();
706    let genesis_transaction_digest = TransactionDigest::new(hash.into());
707
708    let tx_context = TxContext::new(
709        &Address::ZERO,
710        &genesis_transaction_digest,
711        epoch_data,
712        0,
713        0,
714        0,
715        None,
716        protocol_config,
717    );
718
719    Rc::new(RefCell::new(tx_context))
720}

722fn build_unsigned_genesis_data<'info>(
723    parameters: &GenesisCeremonyParameters,
724    token_distribution_schedule: &TokenDistributionSchedule,
725    validators: impl Iterator<Item = &'info GenesisValidatorInfo>,
726    objects: Vec<Object>,
727) -> UnsignedGenesis {
728    if !parameters.allow_insertion_of_extra_objects && !objects.is_empty() {
729        panic!(
730            "insertion of extra objects at genesis time is prohibited due to 'allow_insertion_of_extra_objects' parameter"
731        );
732    }
733
734    let genesis_chain_parameters = parameters.to_genesis_chain_parameters();
735    let genesis_validators = validators
736        .cloned()
737        .map(GenesisValidatorMetadata::from)
738        .collect::<Vec<_>>();
739
740    let epoch_data = EpochData::new_genesis(genesis_chain_parameters.chain_start_timestamp_ms);
741
742    // Get the correct system packages for our protocol version. If we cannot find
743    // the snapshot that means that we must be at the latest version and we
744    // should use the latest version of the framework.
745    let mut system_packages =
746        iota_framework_snapshot::load_bytecode_snapshot(parameters.protocol_version.as_u64())
747            .unwrap_or_else(|_| BuiltInFramework::iter_system_packages().cloned().collect());
748
749    // if system packages are provided in `objects`, update them with the provided
750    // bytes. This is a no-op under normal conditions and only an issue with
751    // certain tests.
752    update_system_packages_from_objects(&mut system_packages, &objects);
753
754    let protocol_config = get_genesis_protocol_config(parameters.protocol_version);
755
756    let genesis_ctx = create_genesis_context(
757        &epoch_data,
758        &genesis_chain_parameters,
759        &genesis_validators,
760        token_distribution_schedule,
761        &system_packages,
762        &protocol_config,
763    );
764
765    // Use a throwaway metrics registry for genesis transaction execution.
766    let registry = prometheus_filtered::Registry::new();
767    let metrics = Arc::new(LimitsMetrics::new(&registry));
768
769    // In here the main genesis objects are created. This means the main system
770    // objects and the ones that are created at genesis like the network coin.
771    let (genesis_objects, events) = create_genesis_objects(
772        genesis_ctx,
773        objects,
774        &genesis_validators,
775        &genesis_chain_parameters,
776        token_distribution_schedule,
777        system_packages,
778        metrics.clone(),
779    );
780
781    // Create the main genesis transaction of kind `GenesisTransaction`
782    let (genesis_transaction, genesis_effects, genesis_events, genesis_objects) =
783        create_genesis_transaction(
784            genesis_objects,
785            events,
786            &protocol_config,
787            metrics,
788            &epoch_data,
789        );
790
791    let (checkpoint, checkpoint_contents) = create_genesis_checkpoint(
792        &protocol_config,
793        parameters,
794        &genesis_transaction,
795        &genesis_effects,
796    );
797
798    UnsignedGenesis {
799        checkpoint,
800        checkpoint_contents,
801        transaction: genesis_transaction,
802        effects: genesis_effects,
803        events: genesis_events,
804        objects: genesis_objects,
805    }
806}

808// Some tests provide an override of the system packages via objects to the
809// genesis builder. When that happens we need to update the system packages with
810// the new bytes provided. Mock system packages in protocol config tests are an
811// example of that (today the only example).
812// The problem here arises from the fact that if regular system packages are
813// pushed first *AND* if any of them is loaded in the loader cache, there is no
814// way to override them with the provided object (no way to mock properly).
815// System packages are loaded only from internal dependencies (a system package
816// depending on some other), and in that case they would be loaded in the
817// VM/loader cache. The Bridge is an example of that and what led to this code.
818// The bridge depends on `iota_system` which is mocked in some tests, but would
819// be in the loader cache courtesy of the Bridge, thus causing the problem.
820fn update_system_packages_from_objects(
821    system_packages: &mut Vec<SystemPackage>,
822    objects: &[Object],
823) {
824    // Filter `objects` for system packages, and make `SystemPackage`s out of them.
825    let system_package_overrides: BTreeMap<ObjectId, Vec<Vec<u8>>> = objects
826        .iter()
827        .filter_map(|obj| {
828            let pkg = obj.data.as_opt_package()?;
829            pkg.id().is_system_package().then(|| {
830                (
831                    pkg.id(),
832                    pkg.serialized_module_map().values().cloned().collect(),
833                )
834            })
835        })
836        .collect();
837
838    // Replace packages in `system_packages` that are present in `objects` with
839    // their counterparts from the previous step.
840    for package in system_packages {
841        if let Some(overrides) = system_package_overrides.get(&package.id).cloned() {
842            package.bytes = overrides;
843        }
844    }
845}

847fn create_genesis_checkpoint(
848    protocol_config: &ProtocolConfig,
849    parameters: &GenesisCeremonyParameters,
850    system_genesis_transaction: &TransactionEnvelope,
851    system_genesis_tx_effects: &TransactionEffects,
852) -> (CheckpointSummary, CheckpointContents) {
853    let genesis_execution_digests = ExecutionDigests {
854        transaction: *system_genesis_transaction.digest(),
855        effects: system_genesis_tx_effects.digest(),
856    };
857
858    let contents = CheckpointContents::new_with_digests_and_signatures(
859        vec![genesis_execution_digests],
860        vec![vec![]],
861    );
862    let version_specific_data =
863        match protocol_config.checkpoint_summary_version_specific_data_as_option() {
864            None | Some(0) => Vec::new(),
865            Some(1) => bcs::to_bytes(&CheckpointVersionSpecificData::V1(
866                CheckpointVersionSpecificDataV1::default(),
867            ))
868            .unwrap(),
869            _ => unimplemented!("unrecognized version_specific_data version for CheckpointSummary"),
870        };
871    let checkpoint = CheckpointSummary {
872        epoch: 0,
873        sequence_number: 0,
874        network_total_transactions: contents.len().try_into().unwrap(),
875        contents_digest: contents.digest(),
876        previous_digest: None,
877        epoch_rolling_gas_cost_summary: Default::default(),
878        end_of_epoch_data: None,
879        timestamp_ms: parameters.chain_start_timestamp_ms,
880        version_specific_data,
881        checkpoint_commitments: Default::default(),
882    };
883
884    (checkpoint, contents)
885}

887fn create_genesis_transaction(
888    objects: Vec<Object>,
889    events: Vec<Event>,
890    protocol_config: &ProtocolConfig,
891    metrics: Arc<LimitsMetrics>,
892    epoch_data: &EpochData,
893) -> (
894    TransactionEnvelope,
895    TransactionEffects,
896    TransactionEvents,
897    Vec<Object>,
898) {
899    let genesis_transaction = {
900        let genesis_objects = objects
901            .into_iter()
902            .map(|mut object| {
903                if let Some(o) = object.data.as_opt_mut_struct() {
904                    o.decrement_version_to(Version::MIN_VALID_INCL);
905                }
906
907                if let Owner::Shared(initial_shared_version) = &mut object.owner {
908                    *initial_shared_version = Version::MIN_VALID_INCL;
909                }
910
911                let object = object.into_inner();
912                GenesisObject::new(object.data, object.owner)
913            })
914            .collect();
915
916        iota_types::transaction::VerifiedTransaction::new_genesis_transaction(
917            genesis_objects,
918            events,
919        )
920        .into_inner()
921    };
922
923    // execute txn to effects
924    let (effects, events, objects) =
925        execute_genesis_transaction(epoch_data, protocol_config, metrics, &genesis_transaction);
926
927    (genesis_transaction, effects, events, objects)
928}

930fn create_genesis_objects(
931    genesis_ctx: Rc<RefCell<TxContext>>,
932    input_objects: Vec<Object>,
933    validators: &[GenesisValidatorMetadata],
934    parameters: &GenesisChainParameters,
935    token_distribution_schedule: &TokenDistributionSchedule,
936    system_packages: Vec<SystemPackage>,
937    metrics: Arc<LimitsMetrics>,
938) -> (Vec<Object>, Vec<Event>) {
939    let mut store = InMemoryStorage::new(Vec::new());
940    let mut events = Vec::new();
941    // We don't know the chain ID here since we haven't yet created the genesis
942    // checkpoint. However since we know there are no chain specific protocol
943    // config options in genesis, we use Chain::Unknown here.
944    let protocol_config = ProtocolConfig::get_for_version(
945        ProtocolVersion::new(parameters.protocol_version),
946        Chain::Unknown,
947    );
948
949    let silent = true;
950    let executor = iota_execution::executor(&protocol_config, silent, None)
951        .expect("Creating an executor should not fail here");
952
953    for system_package in system_packages.into_iter() {
954        let tx_events = process_package(
955            &mut store,
956            executor.as_ref(),
957            genesis_ctx.clone(),
958            &system_package.modules(),
959            system_package.dependencies,
960            &protocol_config,
961            metrics.clone(),
962        )
963        .expect("Processing a package should not fail here");
964
965        events.extend(tx_events.0);
966    }
967
968    for object in input_objects {
969        store.insert_object(object);
970    }
971
972    generate_genesis_system_object(
973        &mut store,
974        executor.as_ref(),
975        validators,
976        genesis_ctx,
977        parameters,
978        token_distribution_schedule,
979        metrics,
980    )
981    .expect("Genesis creation should not fail here");
982
983    (store.into_inner().into_values().collect(), events)
984}

986pub(crate) fn process_package(
987    store: &mut InMemoryStorage,
988    executor: &dyn Executor,
989    ctx: Rc<RefCell<TxContext>>,
990    modules: &[CompiledModule],
991    dependencies: Vec<ObjectId>,
992    protocol_config: &ProtocolConfig,
993    metrics: Arc<LimitsMetrics>,
994) -> anyhow::Result<TransactionEvents> {
995    let dependency_objects = store.get_objects(&dependencies);
996    // When publishing genesis packages, since the std framework packages all have
997    // non-zero addresses, they will be considered as dependencies even though they
998    // are not. Hence input_objects contain objects that don't exist on-chain
999    // because they are yet to be published.
1000    #[cfg(debug_assertions)]
1001    {
1002        use std::collections::HashSet;
1003
1004        use move_core_types::account_address::AccountAddress;
1005
1006        let to_be_published_addresses: HashSet<_> = modules
1007            .iter()
1008            .map(|module| *module.self_id().address())
1009            .collect();
1010        assert!(
1011            // An object either exists on-chain, or is one of the packages to be published.
1012            dependencies
1013                .iter()
1014                .zip(dependency_objects.iter())
1015                .all(|(dependency, obj_opt)| obj_opt.is_some()
1016                    || to_be_published_addresses
1017                        .contains(&AccountAddress::new(dependency.into_bytes())))
1018        );
1019    }
1020    let loaded_dependencies: Vec<_> = dependencies
1021        .iter()
1022        .zip(dependency_objects)
1023        .filter_map(|(dependency, object)| {
1024            Some(ObjectReadResult::new(
1025                InputObjectKind::MovePackage(*dependency),
1026                object?.clone().into(),
1027            ))
1028        })
1029        .collect();
1030
1031    let module_bytes = modules
1032        .iter()
1033        .map(|m| {
1034            let mut buf = vec![];
1035            m.serialize_with_version(m.version, &mut buf).unwrap();
1036            buf
1037        })
1038        .collect();
1039    let pt = {
1040        let mut builder = ProgrammableTransactionBuilder::new();
1041        // executing in Genesis mode does not create an `UpgradeCap`.
1042        builder.command(Command::new_publish(module_bytes, dependencies));
1043        builder.finish()
1044    };
1045    let InnerTemporaryStore {
1046        written, events, ..
1047    } = executor.update_genesis_state(
1048        &*store,
1049        protocol_config,
1050        metrics,
1051        ctx,
1052        CheckedInputObjects::new_for_genesis(loaded_dependencies),
1053        pt,
1054    )?;
1055
1056    store.finish(written);
1057
1058    Ok(events)
1059}

1061pub fn generate_genesis_system_object(
1062    store: &mut InMemoryStorage,
1063    executor: &dyn Executor,
1064    genesis_validators: &[GenesisValidatorMetadata],
1065    genesis_ctx: Rc<RefCell<TxContext>>,
1066    genesis_chain_parameters: &GenesisChainParameters,
1067    token_distribution_schedule: &TokenDistributionSchedule,
1068    metrics: Arc<LimitsMetrics>,
1069) -> anyhow::Result<()> {
1070    let protocol_config = ProtocolConfig::get_for_version(
1071        ProtocolVersion::new(genesis_chain_parameters.protocol_version),
1072        ChainIdentifier::default().chain(),
1073    );
1074
1075    let pt = {
1076        let mut builder = ProgrammableTransactionBuilder::new();
1077        // Step 1: Create the IotaSystemState UID
1078        let iota_system_state_uid = builder.programmable_move_call(
1079            ObjectId::FRAMEWORK,
1080            Identifier::OBJECT_MODULE,
1081            Identifier::from_static("iota_system_state"),
1082            vec![],
1083            vec![],
1084        );
1085
1086        // Step 2: Create and share the Clock.
1087        builder.move_call(
1088            ObjectId::FRAMEWORK,
1089            Identifier::CLOCK_MODULE,
1090            Identifier::from_static("create"),
1091            vec![],
1092            vec![],
1093        )?;
1094
1095        // Create the randomness state_object
1096        builder.move_call(
1097            ObjectId::FRAMEWORK,
1098            Identifier::RANDOM_MODULE,
1099            RANDOMNESS_STATE_CREATE_FUNCTION_NAME,
1100            vec![],
1101            vec![],
1102        )?;
1103
1104        // Create the deny list
1105        builder.move_call(
1106            ObjectId::FRAMEWORK,
1107            Identifier::DENY_LIST_MODULE,
1108            DENY_LIST_CREATE_FUNC,
1109            vec![],
1110            vec![],
1111        )?;
1112
1113        // Step 4: Create the IOTA Coin Treasury Cap.
1114        let iota_treasury_cap = builder.programmable_move_call(
1115            ObjectId::FRAMEWORK,
1116            Identifier::IOTA_MODULE,
1117            Identifier::from_static("new"),
1118            vec![],
1119            vec![],
1120        );
1121
1122        // Step 5: Create System Admin Cap.
1123        let system_admin_cap = builder.programmable_move_call(
1124            ObjectId::FRAMEWORK,
1125            Identifier::SYSTEM_ADMIN_CAP_MODULE,
1126            Identifier::from_static("new_system_admin_cap"),
1127            vec![],
1128            vec![],
1129        );
1130
1131        // Step 6: Run genesis.
1132        // The first argument is the system state uid we got from step 1 and the second
1133        // one is the IOTA `TreasuryCap` we got from step 4.
1134        let mut arguments = vec![iota_system_state_uid, iota_treasury_cap];
1135        let mut call_arg_arguments = vec![
1136            CallArg::pure(&genesis_chain_parameters),
1137            CallArg::pure(&genesis_validators),
1138            CallArg::pure(&token_distribution_schedule),
1139        ]
1140        .into_iter()
1141        .map(|a| builder.input(a))
1142        .collect::<anyhow::Result<_, _>>()?;
1143        arguments.append(&mut call_arg_arguments);
1144        if genesis_chain_parameters.protocol_version
1145            < GENESIS_BUILDER_MIGRATION_LOGIC_REMOVAL_PROTOCOL_VERSION
1146        {
1147            // For older protocol versions, e.g., for running some specific tests, we need
1148            // to pass the timelock genesis label as an argument, but as a None value.
1149            arguments.push(builder.input(CallArg::pure(&None::<String>))?);
1150        }
1151        arguments.push(system_admin_cap);
1152        builder.programmable_move_call(
1153            ObjectId::SYSTEM,
1154            Identifier::from_static("genesis"),
1155            Identifier::from_static("create"),
1156            vec![],
1157            arguments,
1158        );
1159
1160        builder.finish()
1161    };
1162
1163    let InnerTemporaryStore { mut written, .. } = executor.update_genesis_state(
1164        &*store,
1165        &protocol_config,
1166        metrics,
1167        genesis_ctx,
1168        CheckedInputObjects::new_for_genesis(vec![]),
1169        pt,
1170    )?;
1171
1172    // update the value of the clock to match the chain start time
1173    {
1174        let object = written.get_mut(&ObjectId::CLOCK).unwrap();
1175        object
1176            .data
1177            .as_opt_mut_struct()
1178            .unwrap()
1179            .set_clock_timestamp_ms_unchecked(genesis_chain_parameters.chain_start_timestamp_ms);
1180    }
1181
1182    store.finish(written);
1183
1184    Ok(())
1185}
