24/// A checkpoint downloaded from the archive whose content digests — and,
25/// unless deferred, authority signatures — have been verified by a
26/// [`StateSyncWorker`], pending the chain-linkage check and insertion in the
27/// [`StateSyncReducer`].
28pub(crate) struct VerifiedArchiveCheckpoint {
29    summary: CertifiedCheckpointSummary,
30    contents: VerifiedCheckpointContents,
31    /// False when the committee for the checkpoint's epoch was not yet in the
32    /// store, because the previous epoch's last checkpoint had not been
33    /// committed; the reducer verifies the signatures for those instead.
34    signatures_verified: bool,
35}
36
37/// Verifies checkpoints downloaded from the archive.
38///
39/// Multiple workers run concurrently, so this performs only the CPU-heavy
40/// per-checkpoint verification that doesn't depend on the previous checkpoint:
41/// authority signatures and content digests. The [`StateSyncReducer`] receives
42/// the results ordered by sequence number and performs the chain-linkage check
43/// and the store insertion.
44pub(crate) struct StateSyncWorker<S>(pub(crate) S);
45
46#[async_trait]
47impl<S: WriteStore + Clone + Send + Sync + 'static> Worker for StateSyncWorker<S> {
48    type Error = anyhow::Error;
49    type Message = VerifiedArchiveCheckpoint;
50
51    async fn process_checkpoint(
52        &self,
53        checkpoint: Arc<CheckpointData>,
54    ) -> anyhow::Result<Self::Message> {
55        let summary = checkpoint.checkpoint_summary.clone();
56        let committee = self.0.get_committee(summary.epoch());
57        // As many workers run as there are cores, so keep their CPU-bound
58        // verification off the runtime's worker threads, where it would hold up
59        // checkpoint execution — which in turn gates how far sync may run
60        // ahead.
61        tokio::task::spawn_blocking(move || {
62            let signatures_verified = match committee {
63                Some(committee) => {
64                    summary
65                        .verify_authority_signatures(&committee)
66                        .map_err(|e| anyhow!("checkpoint signature verification failed: {e}"))?;
67                    true
68                }
69                None => false,
70            };
71            let contents = FullCheckpointContents::from_contents_and_execution_data(
72                checkpoint.checkpoint_contents.clone(),
73                checkpoint.transactions.iter().map(|t| t.execution_data()),
74            )?
75            .verify(summary.contents_digest)?;
76            Ok(VerifiedArchiveCheckpoint {
77                summary,
78                contents,
79                signatures_verified,
80            })
81        })
82        .await?
83    }
84}

86/// Chain-checks and commits checkpoints verified by [`StateSyncWorker`]s.
87///
88/// This is the single sequential stage of archive sync: batches arrive
89/// ordered by sequence number, and each checkpoint is linked to the previous
90/// one before its summary and contents are inserted into the store. It also
91/// verifies the signatures the workers had to defer — the checkpoints at the
92/// head of an epoch whose committee only becomes available here, once the
93/// previous epoch's last checkpoint is committed.
94pub(crate) struct StateSyncReducer<S> {
95    pub(crate) store: S,
96    pub(crate) metrics: Metrics,
97    /// How far the synced watermark may run ahead of the executed one before
98    /// insertion waits for execution to catch up.
99    pub(crate) max_checkpoints_ahead_of_execution: u64,
100}
101
102/// How many checkpoints the reducer commits per store insertion at most,
103/// bounding the size of the underlying write batches.
104const MAX_CHECKPOINTS_PER_COMMIT: usize = 500;
105
106#[async_trait]
107impl<S: WriteStore + Clone + Send + Sync + 'static> Reducer<StateSyncWorker<S>>
108    for StateSyncReducer<S>
109{
110    async fn commit(&self, batch: &[VerifiedArchiveCheckpoint]) -> anyhow::Result<()> {
111        self.verify_deferred_signatures(batch).await?;
112        self.wait_for_execution_to_catch_up(batch).await;
113
114        let mut to_insert = Vec::with_capacity(batch.len());
115        let mut prev_checkpoint = None;
116        for message in batch {
117            let verified_checkpoint =
118                self.verify_against_previous(message, prev_checkpoint.as_ref())?;
119            prev_checkpoint = Some(verified_checkpoint.clone());
120            to_insert.push((verified_checkpoint, message.contents.clone()));
121        }
122
123        self.store
124            .try_insert_synced_checkpoints(to_insert)
125            .map_err(|e| anyhow!("failed to insert synced checkpoints: {e}"))?;
126
127        for _ in batch {
128            self.metrics
129                .update_checkpoints_synced_from_checkpoint_archive();
130        }
131        Ok(())
132    }
133
134    /// Closes a batch at the size cap, and at epoch boundaries so that an
135    /// epoch's last checkpoint — which carries the next committee — is
136    /// committed before any checkpoint of the next epoch needs that committee
137    /// for its deferred signature verification.
138    fn should_close_batch(
139        &self,
140        batch: &[VerifiedArchiveCheckpoint],
141        next_item: Option<&VerifiedArchiveCheckpoint>,
142    ) -> bool {
143        let Some(next) = next_item else {
144            return true;
145        };
146        batch.len() >= MAX_CHECKPOINTS_PER_COMMIT
147            || batch
148                .last()
149                .is_some_and(|last| last.summary.epoch() != next.summary.epoch())
150    }
151}
152
153impl<S: WriteStore + Clone> StateSyncReducer<S> {
154    /// Waits until the whole batch fits within
155    /// `max_checkpoints_ahead_of_execution` of the executed watermark.
156    ///
157    /// Synced contents can only be pruned once executed, so inserting far
158    /// ahead of execution grows disk usage without bound. Waiting here rather
159    /// than bounding what the workers download keeps the disk bound while
160    /// leaving the run's range untouched; execution advances from the
161    /// checkpoints already in the store, so it makes progress in the meantime.
162    async fn wait_for_execution_to_catch_up(&self, batch: &[VerifiedArchiveCheckpoint]) {
163        let (Some(first), Some(last)) = (batch.first(), batch.last()) else {
164            return;
165        };
166        let required = required_executed_checkpoint(
167            last.summary.sequence_number,
168            self.max_checkpoints_ahead_of_execution,
169        )
170        // Execution can only reach checkpoints that are inserted, so never
171        // wait past the batch's predecessor: a cap below the batch size would
172        // otherwise wait for a checkpoint this very batch has to insert.
173        .min(first.summary.sequence_number.saturating_sub(1));
174
175        let highest_executed = self.store.get_highest_executed_checkpoint_seq_number();
176        if highest_executed.unwrap_or(0) >= required {
177            return;
178        }
179        debug!(
180            "checkpoint archive sync paused: inserting checkpoint {} needs executed checkpoint \
181             {required}, which is at {highest_executed:?}",
182            last.summary.sequence_number
183        );
184        self.store.wait_for_executed_checkpoint(required).await;
185        debug!("checkpoint archive sync resumed at executed checkpoint {required}");
186    }
187
188    /// Verifies the authority signatures the workers had to defer, batched per
189    /// epoch. The deferred checkpoints sit at the head of an epoch, so by
190    /// commit time the previous epoch's last checkpoint — which carries their
191    /// committee — is committed and the committee is in the store.
192    async fn verify_deferred_signatures(
193        &self,
194        batch: &[VerifiedArchiveCheckpoint],
195    ) -> anyhow::Result<()> {
196        let mut deferred: BTreeMap<EpochId, Vec<CertifiedCheckpointSummary>> = BTreeMap::new();
197        for message in batch.iter().filter(|message| !message.signatures_verified) {
198            deferred
199                .entry(message.summary.epoch())
200                .or_default()
201                .push(message.summary.clone());
202        }
203
204        for (epoch, summaries) in deferred {
205            let committee = self
206                .store
207                .get_committee(epoch)
208                .context(format!("missing committee for epoch {epoch} in store"))?;
209            tokio::task::spawn_blocking(move || batch_verify_signatures(&summaries, &committee))
210                .await??;
211        }
212        Ok(())
213    }
214
215    /// Chain-checks one checkpoint against its predecessor — the previous
216    /// checkpoint of the batch being committed, or the store's copy at the
217    /// start of a batch.
218    fn verify_against_previous(
219        &self,
220        message: &VerifiedArchiveCheckpoint,
221        prev_in_batch: Option<&VerifiedCheckpoint>,
222    ) -> anyhow::Result<VerifiedCheckpoint> {
223        let sequence_number = message.summary.sequence_number;
224        if let Some(existing) = self
225            .store
226            .get_checkpoint_by_sequence_number(sequence_number)
227        {
228            // The contents will be inserted under the stored summary, and
229            // mismatched contents would only panic after the transactions are
230            // written, so reject archive data that diverges from the store
231            // before any write.
232            ensure!(
233                existing.digest() == message.summary.digest(),
234                "archive checkpoint {sequence_number} does not match the checkpoint already in the store"
235            );
236            return Ok(existing);
237        }
238
239        let prev_checkpoint_seq_num = sequence_number
240            .checked_sub(1)
241            .context("checkpoint seq num underflow")?;
242        let prev_checkpoint = match prev_in_batch {
243            Some(prev) if prev.sequence_number == prev_checkpoint_seq_num => prev.clone(),
244            _ => self
245                .store
246                .get_checkpoint_by_sequence_number(prev_checkpoint_seq_num)
247                .context(format!(
248                    "missing previous checkpoint {prev_checkpoint_seq_num} in store"
249                ))?,
250        };
251
252        verify_checkpoint_linkage(&prev_checkpoint, message.summary.clone())
253            .map(VerifiedCheckpoint::new_unchecked)
254            .map_err(|_| anyhow!("checkpoint linkage verification failed"))
255    }
256}

258/// The executed checkpoint that inserting `sequence_number` waits for, so that
259/// insertion stays within `max_ahead_of_execution` checkpoints of execution.
260pub(crate) fn required_executed_checkpoint(
261    sequence_number: CheckpointSequenceNumber,
262    max_ahead_of_execution: u64,
263) -> CheckpointSequenceNumber {
264    sequence_number.saturating_sub(max_ahead_of_execution)
265}
266
267/// Verifies the authority signatures of `summaries`, all certified by
268/// `committee`, in one batched signature verification. On failure the
269/// summaries are verified one by one, so that the error names the offending
270/// checkpoint.
271fn batch_verify_signatures(
272    summaries: &[CertifiedCheckpointSummary],
273    committee: &Committee,
274) -> anyhow::Result<()> {
275    let Err(batch_error) =
276        CertifiedCheckpointSummary::batch_verify_authority_signatures(summaries, committee)
277    else {
278        return Ok(());
279    };
280    for summary in summaries {
281        summary
282            .verify_authority_signatures(committee)
283            .map_err(|e| {
284                anyhow!(
285                    "checkpoint {} signature verification failed: {e}",
286                    summary.sequence_number
287                )
288            })?;
289    }
290    Err(anyhow!(
291        "checkpoint signature batch verification failed: {batch_error}"
292    ))
