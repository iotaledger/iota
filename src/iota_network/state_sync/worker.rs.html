24/// A checkpoint downloaded from the archive whose content digests — and,
25/// unless deferred, authority signatures — have been verified by a
26/// [`StateSyncWorker`], pending the chain-linkage check and insertion in the
27/// [`StateSyncReducer`].
28pub(crate) struct VerifiedArchiveCheckpoint {
29    summary: CertifiedCheckpointSummary,
30    contents: VerifiedCheckpointContents,
31    /// False when the committee for the checkpoint's epoch was not yet in the
32    /// store, because the previous epoch's last checkpoint had not been
33    /// committed; the reducer verifies the signatures for those instead.
34    signatures_verified: bool,
35}
36
37/// Verifies checkpoints downloaded from the archive.
38///
39/// Multiple workers run concurrently, so this performs only the CPU-heavy
40/// per-checkpoint verification that doesn't depend on the previous checkpoint:
41/// authority signatures and content digests. The [`StateSyncReducer`] receives
42/// the results ordered by sequence number and performs the chain-linkage check
43/// and the store insertion.
44pub(crate) struct StateSyncWorker<S>(pub(crate) S);
45
46#[async_trait]
47impl<S: WriteStore + Clone + Send + Sync + 'static> Worker for StateSyncWorker<S> {
48    type Error = anyhow::Error;
49    type Message = VerifiedArchiveCheckpoint;
50
51    async fn process_checkpoint(
52        &self,
53        checkpoint: Arc<CheckpointData>,
54    ) -> anyhow::Result<Self::Message> {
55        let summary = checkpoint.checkpoint_summary.clone();
56        let committee = self.0.get_committee(summary.epoch());
57        // As many workers run as there are cores, so keep their CPU-bound
58        // verification off the runtime's worker threads, where it would hold up
59        // checkpoint execution — which in turn gates how far sync may run
60        // ahead.
61        tokio::task::spawn_blocking(move || {
62            let signatures_verified = match committee {
63                Some(committee) => {
64                    summary
65                        .verify_authority_signatures(&committee)
66                        .map_err(|e| anyhow!("checkpoint signature verification failed: {e}"))?;
67                    true
68                }
69                None => false,
70            };
71            let full_contents = FullCheckpointContents::from_contents_and_execution_data(
72                checkpoint.checkpoint_contents.clone(),
73                checkpoint.transactions.iter().map(|t| t.execution_data()),
74            );
75            full_contents.verify_digests(summary.contents_digest)?;
76            let contents = VerifiedCheckpointContents::new_unchecked(full_contents);
77            Ok(VerifiedArchiveCheckpoint {
78                summary,
79                contents,
80                signatures_verified,
81            })
82        })
83        .await?
84    }
85}

87/// Chain-checks and commits checkpoints verified by [`StateSyncWorker`]s.
88///
89/// This is the single sequential stage of archive sync: batches arrive
90/// ordered by sequence number, and each checkpoint is linked to the previous
91/// one before its summary and contents are inserted into the store. It also
92/// verifies the signatures the workers had to defer — the checkpoints at the
93/// head of an epoch whose committee only becomes available here, once the
94/// previous epoch's last checkpoint is committed.
95pub(crate) struct StateSyncReducer<S> {
96    pub(crate) store: S,
97    pub(crate) metrics: Metrics,
98    /// How far the synced watermark may run ahead of the executed one before
99    /// insertion waits for execution to catch up.
100    pub(crate) max_checkpoints_ahead_of_execution: u64,
101}
102
103/// How many checkpoints the reducer commits per store insertion at most,
104/// bounding the size of the underlying write batches.
105const MAX_CHECKPOINTS_PER_COMMIT: usize = 500;
106
107#[async_trait]
108impl<S: WriteStore + Clone + Send + Sync + 'static> Reducer<StateSyncWorker<S>>
109    for StateSyncReducer<S>
110{
111    async fn commit(&self, batch: &[VerifiedArchiveCheckpoint]) -> anyhow::Result<()> {
112        self.verify_deferred_signatures(batch).await?;
113        self.wait_for_execution_to_catch_up(batch).await;
114
115        let mut to_insert = Vec::with_capacity(batch.len());
116        let mut prev_checkpoint = None;
117        for message in batch {
118            let verified_checkpoint =
119                self.verify_against_previous(message, prev_checkpoint.as_ref())?;
120            prev_checkpoint = Some(verified_checkpoint.clone());
121            to_insert.push((verified_checkpoint, message.contents.clone()));
122        }
123
124        self.store
125            .try_insert_synced_checkpoints(to_insert)
126            .map_err(|e| anyhow!("failed to insert synced checkpoints: {e}"))?;
127
128        for _ in batch {
129            self.metrics
130                .update_checkpoints_synced_from_checkpoint_archive();
131        }
132        Ok(())
133    }
134
135    /// Closes a batch at the size cap, and at epoch boundaries so that an
136    /// epoch's last checkpoint — which carries the next committee — is
137    /// committed before any checkpoint of the next epoch needs that committee
138    /// for its deferred signature verification.
139    fn should_close_batch(
140        &self,
141        batch: &[VerifiedArchiveCheckpoint],
142        next_item: Option<&VerifiedArchiveCheckpoint>,
143    ) -> bool {
144        let Some(next) = next_item else {
145            return true;
146        };
147        batch.len() >= MAX_CHECKPOINTS_PER_COMMIT
148            || batch
149                .last()
150                .is_some_and(|last| last.summary.epoch() != next.summary.epoch())
151    }
152}
153
154impl<S: WriteStore + Clone> StateSyncReducer<S> {
155    /// Waits until the whole batch fits within
156    /// `max_checkpoints_ahead_of_execution` of the executed watermark.
157    ///
158    /// Synced contents can only be pruned once executed, so inserting far
159    /// ahead of execution grows disk usage without bound. Waiting here rather
160    /// than bounding what the workers download keeps the disk bound while
161    /// leaving the run's range untouched; execution advances from the
162    /// checkpoints already in the store, so it makes progress in the meantime.
163    async fn wait_for_execution_to_catch_up(&self, batch: &[VerifiedArchiveCheckpoint]) {
164        let (Some(first), Some(last)) = (batch.first(), batch.last()) else {
165            return;
166        };
167        let required = required_executed_checkpoint(
168            last.summary.sequence_number,
169            self.max_checkpoints_ahead_of_execution,
170        )
171        // Execution can only reach checkpoints that are inserted, so never
172        // wait past the batch's predecessor: a cap below the batch size would
173        // otherwise wait for a checkpoint this very batch has to insert.
174        .min(first.summary.sequence_number.saturating_sub(1));
175
176        let highest_executed = self.store.get_highest_executed_checkpoint_seq_number();
177        if highest_executed.unwrap_or(0) >= required {
178            return;
179        }
180        debug!(
181            "checkpoint archive sync paused: inserting checkpoint {} needs executed checkpoint \
182             {required}, which is at {highest_executed:?}",
183            last.summary.sequence_number
184        );
185        self.store.wait_for_executed_checkpoint(required).await;
186        debug!("checkpoint archive sync resumed at executed checkpoint {required}");
187    }
188
189    /// Verifies the authority signatures the workers had to defer, batched per
190    /// epoch. The deferred checkpoints sit at the head of an epoch, so by
191    /// commit time the previous epoch's last checkpoint — which carries their
192    /// committee — is committed and the committee is in the store.
193    async fn verify_deferred_signatures(
194        &self,
195        batch: &[VerifiedArchiveCheckpoint],
196    ) -> anyhow::Result<()> {
197        let mut deferred: BTreeMap<EpochId, Vec<CertifiedCheckpointSummary>> = BTreeMap::new();
198        for message in batch.iter().filter(|message| !message.signatures_verified) {
199            deferred
200                .entry(message.summary.epoch())
201                .or_default()
202                .push(message.summary.clone());
203        }
204
205        for (epoch, summaries) in deferred {
206            let committee = self
207                .store
208                .get_committee(epoch)
209                .context(format!("missing committee for epoch {epoch} in store"))?;
210            tokio::task::spawn_blocking(move || batch_verify_signatures(&summaries, &committee))
211                .await??;
212        }
213        Ok(())
214    }
215
216    /// Chain-checks one checkpoint against its predecessor — the previous
217    /// checkpoint of the batch being committed, or the store's copy at the
218    /// start of a batch.
219    fn verify_against_previous(
220        &self,
221        message: &VerifiedArchiveCheckpoint,
222        prev_in_batch: Option<&VerifiedCheckpoint>,
223    ) -> anyhow::Result<VerifiedCheckpoint> {
224        let sequence_number = message.summary.sequence_number;
225        if let Some(existing) = self
226            .store
227            .get_checkpoint_by_sequence_number(sequence_number)
228        {
229            // The contents will be inserted under the stored summary, and
230            // mismatched contents would only panic after the transactions are
231            // written, so reject archive data that diverges from the store
232            // before any write.
233            ensure!(
234                existing.digest() == message.summary.digest(),
235                "archive checkpoint {sequence_number} does not match the checkpoint already in the store"
236            );
237            return Ok(existing);
238        }
239
240        let prev_checkpoint_seq_num = sequence_number
241            .checked_sub(1)
242            .context("checkpoint seq num underflow")?;
243        let prev_checkpoint = match prev_in_batch {
244            Some(prev) if prev.sequence_number == prev_checkpoint_seq_num => prev.clone(),
245            _ => self
246                .store
247                .get_checkpoint_by_sequence_number(prev_checkpoint_seq_num)
248                .context(format!(
249                    "missing previous checkpoint {prev_checkpoint_seq_num} in store"
250                ))?,
251        };
252
253        verify_checkpoint_linkage(&prev_checkpoint, message.summary.clone())
254            .map(VerifiedCheckpoint::new_unchecked)
255            .map_err(|_| anyhow!("checkpoint linkage verification failed"))
256    }
257}

259/// The executed checkpoint that inserting `sequence_number` waits for, so that
260/// insertion stays within `max_ahead_of_execution` checkpoints of execution.
261pub(crate) fn required_executed_checkpoint(
262    sequence_number: CheckpointSequenceNumber,
263    max_ahead_of_execution: u64,
264) -> CheckpointSequenceNumber {
265    sequence_number.saturating_sub(max_ahead_of_execution)
266}
267
268/// Verifies the authority signatures of `summaries`, all certified by
269/// `committee`, in one batched signature verification. On failure the
270/// summaries are verified one by one, so that the error names the offending
271/// checkpoint.
272fn batch_verify_signatures(
273    summaries: &[CertifiedCheckpointSummary],
274    committee: &Committee,
275) -> anyhow::Result<()> {
276    let Err(batch_error) =
277        CertifiedCheckpointSummary::batch_verify_authority_signatures(summaries, committee)
278    else {
279        return Ok(());
280    };
281    for summary in summaries {
282        summary
283            .verify_authority_signatures(committee)
284            .map_err(|e| {
285                anyhow!(
286                    "checkpoint {} signature verification failed: {e}",
287                    summary.sequence_number
288                )
289            })?;
290    }
291    Err(anyhow!(
292        "checkpoint signature batch verification failed: {batch_error}"
293    ))
