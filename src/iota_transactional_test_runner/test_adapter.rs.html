203impl AdapterInitConfig {
204    fn from_args(init_cmd: InitCommand, iota_args: IotaInitArgs) -> Self {
205        let InitCommand { named_addresses } = init_cmd;
206        let IotaInitArgs {
207            accounts,
208            protocol_version,
209            max_gas,
210            move_binary_format_version,
211            simulator,
212            custom_validator_account,
213            reference_gas_price,
214            default_gas_price,
215            move_auth,
216            deny_rule_governance,
217            flavor,
218            epochs_to_keep,
219            data_ingestion_path,
220            grpc_api_url,
221            package_metadata_with_dynamic_module_metadata,
222        } = iota_args;
223
224        let map = verify_and_create_named_address_mapping(named_addresses).unwrap();
225        let accounts = accounts
226            .map(|v| v.into_iter().collect::<BTreeSet<_>>())
227            .unwrap_or_default();
228
229        let mut protocol_config = if let Some(protocol_version) = protocol_version {
230            ProtocolConfig::get_for_version(protocol_version.into(), Chain::Unknown)
231        } else {
232            ProtocolConfig::get_for_max_version_UNSAFE()
233        };
234        if let Some(version) = move_binary_format_version {
235            protocol_config.set_move_binary_format_version_for_testing(version);
236        }
237        if let Some(mx_tx_gas_override) = max_gas {
238            if simulator {
239                panic!("Cannot set max gas in simulator mode");
240            }
241            protocol_config.set_max_tx_gas_for_testing(mx_tx_gas_override)
242        }
243        if let Some(enable) = move_auth {
244            protocol_config.set_enable_move_authentication_for_testing(enable);
245            protocol_config.set_enable_move_authentication_for_sponsor_for_testing(enable);
246            protocol_config.set_pre_consensus_sponsor_only_move_authentication_for_testing(enable);
247        }
248        if let Some(enable) = deny_rule_governance {
249            protocol_config.set_deny_rule_governance_for_testing(enable);
250            protocol_config.set_deny_rule_governance_on_chain_for_testing(enable);
251            if enable {
252                protocol_config.set_deny_rule_update_max_entries_per_tx_for_testing(1000);
253                protocol_config.set_deny_rule_removal_grace_round_floor_for_testing(0);
254            }
255        }
256        if custom_validator_account && !simulator {
257            panic!("Can only set custom validator account in simulator mode");
258        }
259        if reference_gas_price.is_some() && !simulator {
260            panic!("Can only set reference gas price in simulator mode");
261        }
262
263        let offchain_config = if simulator {
264            Some(OffChainConfig {
265                epochs_to_keep,
266                data_ingestion_path: data_ingestion_path.unwrap_or(tempdir().unwrap().keep()),
267                grpc_api_url,
268            })
269        } else {
270            None
271        };
272
273        if let Some(enable) = package_metadata_with_dynamic_module_metadata {
274            if enable {
275                protocol_config.set_publish_package_metadata_for_testing(true);
276            }
277            protocol_config.set_package_metadata_with_dynamic_module_metadata_for_testing(enable);
278        }
279        Self {
280            additional_mapping: map,
281            account_names: accounts,
282            protocol_config,
283            is_simulator: simulator,
284            custom_validator_account,
285            reference_gas_price,
286            default_gas_price,
287            flavor,
288            offchain_config,
289        }
290    }

478    async fn publish_modules(
479        &mut self,
480        mut modules: Vec<MaybeNamedCompiledModule>,
481        gas_budget: Option<u64>,
482        extra: Self::ExtraPublishArgs,
483    ) -> anyhow::Result<(Option<String>, Vec<MaybeNamedCompiledModule>)> {
484        self.next_task();
485        let IotaPublishArgs {
486            sender,
487            upgradeable,
488            dependencies,
489            view_functions,
490            gas_price,
491        } = extra;
492
493        fill_metadata(&mut modules, &view_functions, self.dynamic_module_metadata)?;
494
495        let named_addr_opt = modules.first().unwrap().named_address;
496
497        let first_module_name = modules.first().unwrap().module.self_id().name().to_string();
498
499        let modules_bytes = modules
500            .iter()
501            .map(|m| {
502                let mut module_bytes = vec![];
503                m.module
504                    .serialize_with_version(m.module.version, &mut module_bytes)
505                    .unwrap();
506                Ok(module_bytes)
507            })
508            .collect::<anyhow::Result<_>>()?;
509
510        let gas_budget = gas_budget.unwrap_or(DEFAULT_GAS_BUDGET);
511        let mapping = &self.compiled_state.named_address_mapping;
512
513        let mut dependencies: Vec<_> = dependencies
514            .into_iter()
515            .map(|d| {
516                let Some(addr) = mapping.get(&d) else {
517                    bail!("There is no published module address corresponding to name address {d}");
518                };
519                let id = ObjectId::new(addr.into_bytes());
520                Ok(id)
521            })
522            .collect::<Result<_, _>>()?;
523
524        let gas_price = gas_price.unwrap_or(self.gas_price);
525
526        // we are assuming that all packages depend on Move Stdlib and IOTA Framework,
527        // so these don't have to be provided explicitly as parameters
528        dependencies.extend([ObjectId::STD, ObjectId::FRAMEWORK]);
529
530        let data = |sender, gas| {
531            let mut builder = ProgrammableTransactionBuilder::new();
532            if upgradeable {
533                let cap = builder.publish_upgradeable(modules_bytes, dependencies);
534                builder.transfer_arg(sender, cap);
535            } else {
536                builder.publish_immutable(modules_bytes, dependencies);
537            };
538            let pt = builder.finish();
539            Transaction::new_programmable(sender, gas, pt, gas_budget, gas_price)
540        };
541        let transaction = self.sign_txn(sender, data);
542        let summary = self.execute_txn(transaction).await?;
543
544        let created_package = summary
545            .created
546            .iter()
547            .find_map(|id| {
548                let object = self.get_object(id, None).unwrap();
549                let package = object.data.as_opt_package()?;
550                if package
551                    .serialized_module_map()
552                    .get(&Identifier::new_unchecked(first_module_name.clone()))
553                    .is_some()
554                {
555                    Some(*id)
556                } else {
557                    None
558                }
559            })
560            .unwrap();
561
562        let package_addr = NumericalAddress::new(created_package.into_bytes(), NumberFormat::Hex);
563        if let Some(named_addr) = named_addr_opt {
564            let prev_package = self
565                .compiled_state
566                .named_address_mapping
567                .insert(named_addr.to_string(), package_addr);
568            match prev_package.map(|a| a.into_inner()) {
569                Some(addr) if addr != AccountAddress::ZERO => panic!(
570                    "Cannot reuse named address '{named_addr}' for multiple packages. \
571                It should be set to 0 initially"
572                ),
573                _ => (),
574            }
575        }
576
577        let output = self.object_summary_output(&summary, /* summarize */ false);
578
579        let published_modules = self
580            .get_object(&created_package, None)
581            .unwrap()
582            .data
583            .as_opt_package()
584            .unwrap()
585            .serialized_module_map()
586            .values()
587            .map(|published_module_bytes| MaybeNamedCompiledModule {
588                named_address: named_addr_opt,
589                module: CompiledModule::deserialize_with_defaults(published_module_bytes).unwrap(),
590                source_map: None,
591                function_infos: None,
592            })
593            .collect();
594
595        Ok((output, published_modules))
596    }

631    async fn handle_subcommand(
632        &mut self,
633        task: TaskInput<Self::Subcommand>,
634    ) -> anyhow::Result<Option<String>> {
635        self.next_task();
636        let TaskInput {
637            command,
638            name,
639            number,
640            start_line,
641            command_lines_stop,
642            stop_line,
643            data,
644            task_text,
645        } = task;
646        macro_rules! get_obj {
647            ($fake_id:ident, $version:expr) => {{
648                let id = match self.fake_to_real_object_id($fake_id) {
649                    None => bail!(
650                        "task {}, lines {}-{}\n{}\n. Unbound fake id {}",
651                        number,
652                        start_line,
653                        command_lines_stop,
654                        task_text,
655                        $fake_id
656                    ),
657                    Some(res) => res,
658                };
659                match self.get_object(&id, $version) {
660                    Err(_) => return Ok(Some(format!("No object at id {}", $fake_id))),
661                    Ok(obj) => obj,
662                }
663            }};
664            ($fake_id:ident) => {{ get_obj!($fake_id, None) }};
665        }
666        match command {
667            IotaSubcommand::RunGraphql(RunGraphqlCommand {

1405    /// Process the error string such that it's less dependent on specific
1406    /// addresses or object IDs. Instead, they are replaced by the account
1407    /// names or fake IDs as much as possible. This reduces the effort of
1408    /// updating tests when something changed.
1409    async fn process_error(&self, error: anyhow::Error) -> anyhow::Error {
1410        let mut err = error.to_string();
1411        for (name, account) in &self.accounts {
1412            let addr = account.address.to_string();
1413            let replace = format!("@{name}");
1414            err = err.replace(&addr, &replace);
1415            // Also match without 0x since different error messages may use different
1416            // format.
1417            err = err.replace(&addr[2..], &replace);
1418        }
1419        for (id, fake_id) in &self.object_enumeration {
1420            let id = id.to_string();
1421            let replace = format!("object({fake_id})");
1422            err = err.replace(&id, &replace);
1423            // Also match without 0x since different error messages may use different
1424            // format.
1425            err = err.replace(&id[2..], &replace);
1426        }
1427        anyhow!(err)
1428    }

1590    fn interpolate_contents(
1591        &self,
1592        contents: &str,
1593        variables: &BTreeMap<String, String>,
1594    ) -> anyhow::Result<String> {
1595        let mut interpolated_contents = contents.to_string();
1596
1597        let re = regex::Regex::new(r"@\{([^\}]+)\}").unwrap();
1598
1599        let unique_vars = re
1600            .captures_iter(contents)
1601            .filter_map(|c| c.get(1).map(|m| m.as_str().to_string()))
1602            .collect::<std::collections::HashSet<_>>();
1603
1604        for var_name in unique_vars {
1605            let Some(value) = variables.get(&var_name) else {
1606                bail!("Unknown variable: {var_name}\nAllowed variable mappings are {variables:#?}");
1607            };
1608
1609            let pattern = format!("@{{{var_name}}}");
1610            interpolated_contents = interpolated_contents.replace(&pattern, value);
1611        }
1612
1613        Ok(interpolated_contents)
1614    }

1616    fn encode_cursor(&self, cursor: &str) -> anyhow::Result<String> {
1617        // Cursor format is either bcs(object_id,n1,n2,...) or a json value,
1618        // in which case we just return its base64 encoding.
1619        let Some(args) = cursor
1620            .strip_prefix("bcs(")
1621            .and_then(|c| c.strip_suffix(")"))
1622        else {
1623            // To comply with how `iota-graphql-rpc` decodes the json cursor
1624            // (see `iota_graphql_rpc::types::cursor::JsonCursor`).
1625            //
1626            // This traces back to `async_graphql = 7.0.7` that uses no padding for
1627            // encoding/decoding.
1628            return Ok(base64::Engine::encode(
1629                &base64::engine::general_purpose::URL_SAFE_NO_PAD,
1630                cursor,
1631            ));
1632        };
1633
1634        let mut parts = args.split(",");
1635
1636        let id: ObjectId = parts
1637            .next()
1638            .context("bcs(...) cursors must have at least one argument")?
1639            .trim()
1640            .parse()?;
1641
1642        let mut bytes = bcs::to_bytes(&id.as_bytes().to_vec())?;
1643        for part in parts {
1644            let n: u64 = part.trim().parse()?;
1645            bytes.extend(bcs::to_bytes(&n)?);
1646        }
1647
1648        Ok(Base64::encode(bytes))
1649    }

1652        &self,
1653        contents: &str,
1654        cursors: &[String],
1655        highest_checkpoint: u64,
1656    ) -> anyhow::Result<String> {
1657        // First collect all the variable mappings

1676    async fn upgrade_package(
1677        &mut self,
1678        before_upgrade: NumericalAddress,
1679        modules: &[MaybeNamedCompiledModule],
1680        upgrade_capability: FakeID,
1681        dependencies: Vec<String>,
1682        sender: String,
1683        gas_budget: Option<u64>,
1684        dry_run: bool,
1685        policy: u8,
1686        gas_price: u64,
1687    ) -> anyhow::Result<Option<String>> {
1688        let modules_bytes = modules
1689            .iter()
1690            .map(|m| {
1691                let mut module_bytes = vec![];
1692                m.module
1693                    .serialize_with_version(m.module.version, &mut module_bytes)?;
1694                Ok(module_bytes)
1695            })
1696            .collect::<anyhow::Result<Vec<Vec<u8>>>>()?;
1697        let gas_budget = gas_budget.unwrap_or(DEFAULT_GAS_BUDGET);
1698
1699        let dependencies = self.get_dependency_ids(dependencies, /* include_std */ true)?;
1700
1701        let mut builder = ProgrammableTransactionBuilder::new();
1702
1703        // Argument::Input(0)
1704        IotaValue::Object(upgrade_capability, None).into_argument(&mut builder, self)?;
1705        let upgrade_arg = builder.pure(policy).unwrap();
1706        let digest: Vec<u8> =
1707            MovePackage::compute_digest_for_modules_and_deps(&modules_bytes, &dependencies)
1708                .into_bytes()
1709                .to_vec();
1710        let digest_arg = builder.pure(digest).unwrap();
1711
1712        let upgrade_ticket = builder.programmable_move_call(
1713            ObjectId::FRAMEWORK,
1714            Identifier::PACKAGE_MODULE,
1715            Identifier::from_static("authorize_upgrade"),
1716            vec![],
1717            vec![Argument::Input(0), upgrade_arg, digest_arg],
1718        );
1719
1720        let package_id = ObjectId::new(before_upgrade.into_bytes());
1721        let upgrade_receipt =
1722            builder.upgrade(package_id, upgrade_ticket, dependencies, modules_bytes);
1723
1724        builder.programmable_move_call(
1725            ObjectId::FRAMEWORK,
1726            Identifier::PACKAGE_MODULE,
1727            Identifier::from_static("commit_upgrade"),
1728            vec![],
1729            vec![Argument::Input(0), upgrade_receipt],
1730        );
1731
1732        let pt = builder.finish();
1733
1734        let summary = if dry_run {
1735            let transaction = Transaction::new_programmable(
1736                self.get_sender(Some(sender)).address,
1737                vec![],
1738                pt,
1739                gas_budget,
1740                gas_price,
1741            );
1742            self.dry_run(transaction).await?
1743        } else {
1744            let tx =
1745                |sender, gas| Transaction::new_programmable(sender, gas, pt, gas_budget, gas_price);
1746            let transaction = self.sign_txn(Some(sender), tx);
1747            self.execute_txn(transaction).await?
1748        };
1749        let created_package = summary
1750            .created
1751            .iter()
1752            .find_map(|id| {
1753                let object = self.get_object(id, None).unwrap();
1754                let package = object.data.as_opt_package()?;
1755                Some(package.id())
1756            })
1757            .unwrap();
1758        let package_addr = NumericalAddress::new(created_package.into_bytes(), NumberFormat::Hex);
1759        if let Some(new_package_name) = modules[0].named_address {
1760            let prev_package = self
1761                .compiled_state
1762                .named_address_mapping
1763                .insert(new_package_name.to_string(), package_addr);
1764            match prev_package.map(|a| a.into_inner()) {
1765                Some(addr) if addr != AccountAddress::ZERO => panic!(
1766                    "Cannot reuse named address '{new_package_name}' for multiple packages. \
1767                It should be set to 0 initially"
1768                ),
1769                _ => (),
1770            }
1771        }
1772        let output = self.object_summary_output(&summary, /* summarize */ false);
1773        Ok(output)
1774    }

1898    async fn execute_txn(
1899        &mut self,
1900        transaction: TransactionEnvelope,
1901    ) -> anyhow::Result<TxnSummary> {
1902        let with_shared = transaction.contains_shared_object();
1903        let (effects, error_opt) = self.executor.execute_txn(transaction).await?;
1904        let digest = effects.transaction_digest();
1905
1906        // Try to assign `digest_$task` to this transaction's digest -- panic if a
1907        // transaction has already been set. Currently each task executes at
1908        // most one transaction, and everything is fine. This panic triggering
1909        // will be an early warning that we need to do something
1910        // more sophisticated.
1911        let task = self.next_fake.0;
1912        if let Some(prev) = self.digest_enumeration.insert(task, *digest) {
1913            panic!(
1914                "Task {task} executed two transactions (expected at most one): {prev}, {digest}"
1915            );
1916        }
1917
1918        let mut created_ids: Vec<_> = effects
1919            .created()
1920            .iter()
1921            .map(|created| created.reference.object_id)
1922            .collect();
1923        let mut mutated_ids: Vec<_> = effects
1924            .mutated()
1925            .iter()
1926            .map(|mutated| mutated.reference.object_id)
1927            .collect();
1928        let mut unwrapped_ids: Vec<_> = effects
1929            .unwrapped()
1930            .iter()
1931            .map(|unwrapped| unwrapped.reference.object_id)
1932            .collect();
1933        let mut deleted_ids: Vec<_> = effects
1934            .deleted()
1935            .iter()
1936            .map(|object_ref| object_ref.object_id)
1937            .collect();
1938        let mut unwrapped_then_deleted_ids: Vec<_> = effects
1939            .unwrapped_then_deleted()
1940            .iter()
1941            .map(|object_ref| object_ref.object_id)
1942            .collect();
1943        let mut wrapped_ids: Vec<_> = effects
1944            .wrapped()
1945            .iter()
1946            .map(|object_ref| object_ref.object_id)
1947            .collect();
1948        let gas_summary = effects.gas_cost_summary();
1949
1950        // make sure objects that have previously not been in storage get assigned a
1951        // fake id.
1952        let mut might_need_fake_id: Vec<_> = created_ids
1953            .iter()

2262    fn stabilize_str(&self, input: impl AsRef<str>) -> String {
2263        fn candidate_is_hex(s: &str) -> bool {
2264            const HEX_STR_LENGTH: usize = IOTA_ADDRESS_LENGTH * 2;
2265            let n = s.len();
2266            (s.starts_with("0x") && n >= 3) || n == HEX_STR_LENGTH
2267        }
2268        let mut hex_candidate = String::new();
2269        let mut result = String::new();
2270        let mut chars = input.as_ref().chars().peekable();
2271        let mut cur = chars.next();
2272        while let Some(c) = cur {
2273            match c {
2274                '0' if hex_candidate.is_empty() && matches!(chars.peek(), Some('x')) => {
2275                    let c = chars.next().unwrap();
2276                    assert!(c == 'x');
2277                    hex_candidate.push_str("0x");
2278                }
2279                '0'..='9' | 'a'..='f' | 'A'..='F' => hex_candidate.push(c),
2280                _ => {
2281                    if candidate_is_hex(&hex_candidate) {
2282                        result.push_str(&self.remap_hex_str(hex_candidate));
2283                        hex_candidate = String::new();
2284                    } else {
2285                        result.push_str(&hex_candidate);
2286                        if !hex_candidate.is_empty() {
2287                            hex_candidate = String::new();
2288                        }
2289                    }
2290                    result.push(c);
2291                }
2292            }
2293            cur = chars.next();
2294        }
2295        if candidate_is_hex(&hex_candidate) {
2296            result.push_str(&self.remap_hex_str(hex_candidate));
2297        } else {
2298            result.push_str(&hex_candidate);
2299        }
2300        result
2301    }

2745/// Create an executor using a simulator
2746/// This means we can control the checkpoint, epoch creation process and
2747/// manually advance clock as needed
2748async fn init_sim_executor(
2749    mut rng: StdRng,
2750    account_names: BTreeSet<String>,
2751    additional_mapping: BTreeMap<String, NumericalAddress>,
2752    protocol_config: &ProtocolConfig,
2753    custom_validator_account: bool,
2754    reference_gas_price: Option<u64>,
2755    data_ingestion_path: PathBuf,
2756) -> (
2757    Box<dyn TransactionalAdapter>,
2758    AccountSetup,
2759    Option<Arc<dyn GrpcStateReader + Send + Sync>>,
2760) {
2761    // Initial list of named addresses with specified values
2762    let mut named_address_mapping = NAMED_ADDRESSES.clone();
2763    let mut account_objects = BTreeMap::new();
2764    let mut account_keys = BTreeMap::new();
2765    let mut accounts = BTreeMap::new();
2766    let mut objects = vec![];
2767
2768    // For each named IOTA account without an address value, create a key
2769    for n in account_names {
2770        let test_account = get_key_pair_from_rng(&mut rng);
2771        account_keys.insert(n, test_account);
2772    }
2773
2774    // Make a default account key
2775    let default_account_key = get_key_pair_from_rng(&mut rng);
2776
2777    let (mut validator_addr, mut validator_key, mut key_copy) = (None, None, None);
2778    if custom_validator_account {
2779        // Make a validator account with a gas object
2780        let (a, b): (Address, AccountPrivateKey) = get_key_pair_from_rng(&mut rng);
2781
2782        key_copy = Some(b.clone());
2783        validator_addr = Some(a);
2784        validator_key = Some(b);
2785    }
2786
2787    let mut acc_cfgs = account_keys
2788        .values()
2789        .map(|acc| AccountConfig {
2790            address: Some(acc.0),
2791            gas_amounts: vec![GAS_FOR_TESTING],
2792        })
2793        .collect::<Vec<_>>();
2794    acc_cfgs.push(AccountConfig {
2795        address: Some(default_account_key.0),
2796        gas_amounts: vec![GAS_FOR_TESTING],
2797    });
2798
2799    if let Some(v_addr) = validator_addr {
2800        acc_cfgs.push(AccountConfig {
2801            address: Some(v_addr),
2802            gas_amounts: vec![GAS_FOR_TESTING],
2803        });
2804    }
2805
2806    // Create the simulator with the specific account configs, which also crates
2807    // objects
2808
2809    // The simulator resolves its `ProtocolConfig` from the protocol version alone,
2810    // so feature flags toggled on `protocol_config` would not reach it. Scope a
2811    // thread-local override around the (synchronous) construction so the genesis
2812    // epoch picks the flags up; `Simulacrum::advance_epoch` then carries the
2813    // config over to later epochs.
2814    let config_override = protocol_config.deny_rule_governance().then(|| {
2815        let max_entries = protocol_config.deny_rule_update_max_entries_per_tx();
2816        let grace_floor = protocol_config.deny_rule_removal_grace_round_floor();
2817        ProtocolConfig::apply_overrides_for_testing(move |_, mut config| {
2818            config.set_deny_rule_governance_for_testing(true);
2819            config.set_deny_rule_governance_on_chain_for_testing(true);
2820            config.set_deny_rule_update_max_entries_per_tx_for_testing(max_entries);
2821            config.set_deny_rule_removal_grace_round_floor_for_testing(grace_floor);
2822            config
2823        })
2824    });
2825    let (sim, read_replica) = PersistedStore::new_sim_replica_with_protocol_version_and_accounts(
2826        rng,
2827        DEFAULT_CHAIN_START_TIMESTAMP,
2828        protocol_config.version,
2829        acc_cfgs,
2830        key_copy.map(|q| vec![q]),
2831        reference_gas_price,
2832        None,
2833    );
2834    drop(config_override);
2835
2836    sim.set_data_ingestion_path(data_ingestion_path.clone());
2837
2838    // Get the actual object values from the simulator
2839    let default_account = sim.with_store(|store| {
2840        for (name, (addr, key)) in account_keys {
2841            let o = store.owned_objects(addr).next().unwrap();
2842            objects.push(o.clone());
2843            account_objects.insert(name.clone(), o.id());
2844
2845            accounts.insert(
2846                name.to_owned(),
2847                TestAccount {
2848                    address: addr,
2849                    private_key: Some(key),
2850                    gas: o.id(),
2851                },
2852            );
2853        }
2854        let o = store.owned_objects(default_account_key.0).next().unwrap();
2855        let default_account = TestAccount {
2856            address: default_account_key.0,
2857            private_key: Some(default_account_key.1),
2858            gas: o.id(),
2859        };
2860        objects.push(o);
2861
2862        if let (Some(v_addr), Some(v_key)) = (validator_addr, validator_key) {
2863            let o = store.owned_objects(v_addr).next().unwrap();
2864            let validator_account = TestAccount {
2865                address: v_addr,
2866                private_key: Some(v_key),
2867                gas: o.id(),
2868            };
2869            objects.push(o.clone());
2870            account_objects.insert("validator_0".to_string(), o.id());
2871            accounts.insert("validator_0".to_string(), validator_account);
2872        }
2873
2874        default_account
2875    });
2876
2877    let sim = Box::new(sim);
2878    update_named_address_mapping(
2879        &mut named_address_mapping,
2880        &accounts,
2881        additional_mapping,
2882        &*sim,
2883    )
2884    .await;
2885
2886    (
2887        sim,
2888        AccountSetup {
2889            default_account,
2890            named_address_mapping,
2891            objects,
2892            account_objects,
2893            accounts,
2894        },
2895        Some(Arc::new(read_replica)),
2896    )
2897}

2899async fn update_named_address_mapping(
2900    named_address_mapping: &mut BTreeMap<String, NumericalAddress>,
2901    accounts: &BTreeMap<String, TestAccount>,
2902    additional_mapping: BTreeMap<String, NumericalAddress>,
2903    trans_adapter: &dyn TransactionalAdapter,
2904) {
2905    let active_val_addrs: BTreeMap<_, _> = trans_adapter
2906        .get_active_validator_addresses()
2907        .await
2908        .expect("Failed to get validator addresses")
2909        .iter()
2910        .enumerate()
2911        .map(|(idx, addr)| (format!("validator_{idx}"), *addr))
2912        .collect();
2913
2914    // For mappings where the address is specified, populate the named address
2915    // mapping
2916    let additional_mapping = additional_mapping
2917        .into_iter()
2918        .chain(accounts.iter().map(|(n, test_account)| {
2919            let addr = NumericalAddress::new(test_account.address.into_bytes(), NumberFormat::Hex);
2920            (n.clone(), addr)
2921        }))
2922        .chain(active_val_addrs.iter().map(|(n, addr)| {
2923            let addr = NumericalAddress::new(addr.into_bytes(), NumberFormat::Hex);
2924            (n.clone(), addr)
2925        }));
2926    // Extend the mappings of all named addresses with values
2927    for (name, addr) in additional_mapping {
2928        if (named_address_mapping.contains_key(&name)
2929            && (named_address_mapping.get(&name) != Some(&addr)))
2930            || name == "iota"
2931        {
2932            panic!("Invalid init. The named address '{name}' is reserved or duplicated")
2933        }
2934        named_address_mapping.insert(name, addr);
2935    }
2936}

3070/// Fill the compiled modules with IOTA metadata from compiler function infos
3071/// and transactional-test-only flags.
3072fn fill_metadata(
3073    modules: &mut [MaybeNamedCompiledModule],
3074    view_functions: &[String],
3075    dynamic_module_metadata: bool,
3076) -> anyhow::Result<()> {
3077    // View functions are only representable in V2 (dynamic) runtime metadata, so
3078    // they require the `package_metadata_with_dynamic_module_metadata` feature.
3079    if !dynamic_module_metadata && !view_functions.is_empty() {
3080        bail!(
3081            "--view-functions requires the package_metadata_with_dynamic_module_metadata protocol feature"
3082        );
3083    }
3084
3085    let mut unmatched_view_functions = view_functions.iter().cloned().collect::<BTreeSet<_>>();
3086
3087    for m in modules.iter_mut() {
3088        let module: &mut CompiledModule = &mut m.module;
3089        let mut runtime_metadata = if dynamic_module_metadata {
3090            RuntimeModuleMetadata::v2()
3091        } else {
3092            RuntimeModuleMetadata::v1()
3093        };
3094        let mut view_attributes = BTreeSet::new();
3095
3096        if let Some(fn_infos) = &m.function_infos {
3097            for (_, name, info) in fn_infos.iter() {
3098                // We only need authenticator version here
3099                if let Some(version) = info.attributes.get_authenticator() {
3100                    let attribute = if dynamic_module_metadata {
3101                        IotaAttribute::V2(IotaAttributeV2::authenticator_attribute(version))
3102                    } else {
3103                        IotaAttribute::V1(IotaAttributeV1::authenticator_attribute(version))
3104                    };
3105                    runtime_metadata.add_function_attribute(name.as_str().to_owned(), attribute);
3106                }
3107                if dynamic_module_metadata && info.attributes.is_view() {
3108                    view_attributes.insert(name.as_str().to_owned());
3109                }
3110            }
3111        }
3112
3113        for function_name in view_functions {
3114            if module.find_function_def_by_name(function_name).is_some() {
3115                unmatched_view_functions.remove(function_name);
3116                view_attributes.insert(function_name.clone());
3117            }
3118        }
3119
3120        for function_name in view_attributes {
3121            runtime_metadata.add_function_attribute(
3122                function_name,
3123                IotaAttribute::V2(IotaAttributeV2::view_attribute()),
3124            );
3125        }
3126
3127        if !runtime_metadata.is_empty() {
3128            module.metadata.push(Metadata {
3129                key: IOTA_METADATA_KEY.to_vec(),
3130                value: RuntimeModuleMetadataWrapper::from(runtime_metadata).to_bcs_bytes(),
3131            });
3132        }
3133    }
3134
3135    if let Some(function_name) = unmatched_view_functions.into_iter().next() {
3136        bail!("Could not find function '{function_name}' requested by --view-functions");
3137    }
3138
3139    Ok(())
3140}
