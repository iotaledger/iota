76impl PersistedStore {
77    pub fn new(genesis: &genesis::Genesis, path: PathBuf) -> Self {
78        let samp: SamplingInterval = SamplingInterval::new(Duration::from_secs(60), 0);
79        let read_write = PersistedStoreInner::open_tables_read_write(
80            path.clone(),
81            MetricConf::new("persisted").with_sampling(samp),
82            None,
83            None,
84        );
85
86        let mut res = Self { path, read_write };
87        res.init_with_genesis(genesis);
88
89        res
90    }

92    pub fn read_replica(&self) -> PersistedStoreInnerReadOnlyWrapper {
93        let samp: SamplingInterval = SamplingInterval::new(Duration::from_secs(60), 0);
94        PersistedStoreInnerReadOnlyWrapper {
95            path: self.path.clone(),
96            inner: PersistedStoreInner::get_read_only_handle(
97                self.path.clone(),
98                None,
99                None,
100                MetricConf::new("persisted_readonly").with_sampling(samp),
101            ),
102        }
103    }
104
105    pub fn new_sim_replica_with_protocol_version_and_accounts<R>(
106        mut rng: R,
107        chain_start_timestamp_ms: u64,
108        protocol_version: ProtocolVersion,
109        account_configs: Vec<AccountConfig>,
110        validator_keys: Option<Vec<AccountPrivateKey>>,
111        reference_gas_price: Option<u64>,
112        path: Option<PathBuf>,
113    ) -> (Simulacrum<R, Self>, PersistedStoreInnerReadOnlyWrapper)
114    where
115        R: rand::RngCore + rand::CryptoRng,
116    {
117        let store_directory: PathBuf = path.unwrap_or(iota_common::tempdir().keep());
118
119        let mut builder = ConfigBuilder::new_with_temp_dir()
120            .rng(&mut rng)
121            .with_chain_start_timestamp_ms(chain_start_timestamp_ms)
122            .deterministic_committee_size(NonZeroUsize::new(1).unwrap())
123            .with_protocol_version(protocol_version)
124            .with_accounts(account_configs);
125
126        if let Some(validator_keys) = validator_keys {
127            builder = builder.deterministic_committee_validators(validator_keys)
128        };
129        if let Some(reference_gas_price) = reference_gas_price {
130            builder = builder.with_reference_gas_price(reference_gas_price)
131        };
132
133        let config = builder.build();
134
135        let genesis = &config.genesis;
136
137        let store = PersistedStore::new(genesis, store_directory);
138        let read_only_wrapper = store.read_replica();
139        (
140            Simulacrum::new_with_network_config_store(&config, rng, store),
141            read_only_wrapper,
142        )
143    }

169    fn get_highest_checkpoint(&self) -> Option<VerifiedCheckpoint> {
170        self.read_write
171            .checkpoints
172            .safe_range_iter_reversed(..)
173            .next()
174            .transpose()
175            .expect("failed to fetch highest checkpoint")
176            .map(|(_, checkpoint)| checkpoint.into())

179    fn get_object(&self, id: &ObjectId) -> Option<Object> {
180        let version = self
181            .read_write
182            .live_objects
183            .get(id)
184            .expect("Fatal: DB read failed")?;
185        self.get_object_at_version(id, version)
186    }

188    fn get_object_at_version(&self, id: &ObjectId, version: Version) -> Option<Object> {
189        self.read_write
190            .objects
191            .get(id)
192            .expect("Fatal: DB read failed")
193            .and_then(|versions| versions.get(&version).cloned())
194    }

201        SimulatorStore::get_object(self, &ObjectId::CLOCK)

207    fn get_last_checkpoint_of_epoch(&self, epoch: EpochId) -> Option<CheckpointSequenceNumber> {
208        self.read_write
209            .last_checkpoints_per_epoch
210            .get(&epoch)

214    fn get_system_state_by_epoch(
215        &self,
216        _epoch: EpochId,
217    ) -> Option<&iota_types::iota_system_state::IotaSystemState> {
218        // PersistedStore doesn't currently support historical system state storage
219        None

221
222    fn owned_objects(&self, owner: Address) -> Box<dyn Iterator<Item = Object> + '_> {
223        Box::new(
224            self.read_write
225                .live_objects
226                .safe_iter()
227                .map(|result| result.expect("rocksdb iteration failed"))
228                .flat_map(|(id, version)| self.get_object_at_version(&id, version))
229                .filter(
230                    move |object| matches!(object.owner, Owner::Address(addr) if addr == owner),
231                ),
232        )
233    }
234
235    fn insert_checkpoint(&mut self, checkpoint: VerifiedCheckpoint) {
236        self.read_write
237            .checkpoint_digest_to_sequence_number
238            .insert(checkpoint.digest(), &checkpoint.sequence_number())
239            .expect("Fatal: DB write failed");
240        self.read_write
241            .checkpoints
242            .insert(&checkpoint.sequence_number(), checkpoint.serializable_ref())
243            .expect("Fatal: DB write failed");
244    }
245
246    fn insert_checkpoint_contents(&mut self, contents: CheckpointContents) {
247        self.read_write
248            .checkpoint_contents
249            .insert(&contents.digest(), &contents)
250            .expect("Fatal: DB write failed");
251    }
252
253    fn insert_committee(&mut self, committee: Committee) {
254        let epoch = committee.epoch();
255
256        let committees = self
257            .read_write
258            .epoch_to_committee
259            .get(&epoch)
260            .expect("Fatal: DB read failed");
261
262        if committees.is_some() {
263            return;
264        }
265
266        self.read_write
267            .epoch_to_committee
268            .insert(&epoch, &committee)
269            .expect("Fatal: DB write failed");
270    }
271
272    fn insert_executed_transaction(
273        &mut self,
274        transaction: VerifiedTransaction,
275        effects: TransactionEffects,
276        events: TransactionEvents,
277        written_objects: BTreeMap<ObjectId, Object>,
278    ) {
279        let deleted_objects = effects.deleted();
280        let tx_digest = *effects.transaction_digest();
281        self.insert_transaction(transaction);
282        self.insert_transaction_effects(effects);
283        self.insert_events(&tx_digest, events);
284        self.update_objects(written_objects, deleted_objects);
285    }
286
287    fn insert_transaction(&mut self, transaction: VerifiedTransaction) {
288        self.read_write
289            .transactions
290            .insert(transaction.digest(), transaction.serializable_ref())
291            .expect("Fatal: DB write failed");
292    }
293
294    fn insert_transaction_effects(&mut self, effects: TransactionEffects) {
295        self.read_write
296            .effects
297            .insert(effects.transaction_digest(), &effects)
298            .expect("Fatal: DB write failed");
299    }
300
301    fn insert_events(&mut self, tx_digest: &TransactionDigest, events: TransactionEvents) {
302        self.read_write
303            .events
304            .insert(tx_digest, &events)
305            .expect("Fatal: DB write failed");
306    }
307
308    fn update_objects(
309        &mut self,
310        written_objects: BTreeMap<ObjectId, Object>,
311        deleted_objects: Vec<ObjectReference>,
312    ) {
313        for object_ref in deleted_objects {
314            self.read_write
315                .live_objects
316                .remove(&object_ref.object_id)
317                .expect("Fatal: DB write failed");
318        }
319
320        for (object_id, object) in written_objects {
321            let version = object.version();
322            self.read_write
323                .live_objects
324                .insert(&object_id, &version)
325                .expect("Fatal: DB write failed");
326            let mut q = self
327                .read_write
328                .objects
329                .get(&object_id)
330                .expect("Fatal: DB read failed")
331                .unwrap_or_default();
332            q.insert(version, object);
333            self.read_write
334                .objects
335                .insert(&object_id, &q)
336                .expect("Fatal: DB write failed");
337        }
338    }
339
340    fn backing_store(&self) -> &dyn iota_types::storage::BackingStore {
341        self
342    }
343
344    fn update_last_checkpoint_of_epoch(
345        &mut self,
346        epoch: EpochId,
347        last_checkpoint: CheckpointSequenceNumber,
348    ) {
349        self.read_write
350            .last_checkpoints_per_epoch
351            .insert(&epoch, &last_checkpoint)
352            .expect("Fatal: DB write failed");
353    }
354}

365impl ChildObjectResolver for PersistedStore {
366    fn read_child_object(
367        &self,
368        parent: &ObjectId,
369        child: &ObjectId,
370        child_version_upper_bound: Version,
371    ) -> iota_types::error::IotaResult<Option<Object>> {
372        let child_object = match SimulatorStore::get_object(self, child) {
373            None => return Ok(None),
374            Some(obj) => obj,
375        };
376
377        let parent = *parent;
378        if child_object.owner != Owner::Object(parent) {
379            return Err(IotaError::InvalidChildObjectAccess {
380                object: *child,
381                given_parent: parent,
382                actual_owner: child_object.owner,
383            });
384        }
385
386        if child_object.version() > child_version_upper_bound {
387            return Err(IotaError::UnsupportedFeature {
388                error: "TODO InMemoryStorage::read_child_object does not yet support bounded reads"
389                    .to_owned(),
390            });
391        }
392
393        Ok(Some(child_object))
394    }
395
396    fn get_object_received_at_version(
397        &self,
398        owner: &ObjectId,
399        receiving_object_id: &ObjectId,
400        receive_object_at_version: Version,
401        _epoch_id: EpochId,
402    ) -> iota_types::error::IotaResult<Option<Object>> {
403        let recv_object = match SimulatorStore::get_object(self, receiving_object_id) {
404            None => return Ok(None),
405            Some(obj) => obj,
406        };
407        if recv_object.owner != Owner::Address((*owner).into()) {
408            return Ok(None);
409        }
410
411        if recv_object.version() != receive_object_at_version {
412            return Ok(None);
413        }
414        Ok(Some(recv_object))
415    }
416}

853            0,

861
862        let rng = StdRng::from_seed([9; 32]);
863        let chain2 = PersistedStore::new_sim_with_protocol_version_and_accounts(
864            rng,
865            0,
866            ProtocolVersion::MAX,
867            vec![],
868            None,
869        );
870        let genesis_checkpoint_digest2 = *chain2
871            .with_store(|store| store.get_checkpoint_by_sequence_number(0).unwrap())
872            .digest();

875
876        // Ensure the committees are different when using different seeds
877        let rng = StdRng::from_seed([0; 32]);

879            rng,
880            0,
881            ProtocolVersion::MAX,
882            vec![],
883            None,
884        );
885
886        assert_ne!(
