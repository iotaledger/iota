12/// The paging cursor of the owner-index queries — `getOwnedObjects`,
13/// `getCoins` and `getAllCoins`.
14///
15/// Opaque on the wire: a caller passes back the `nextCursor` a page returned.
16/// It names the position of the row it came from, so the next page can resume
17/// after an object that has since been spent.
18///
19/// - [`Self::ObjectId`] is used by the indexer, whose owner index is ordered by object id, and is
20///   written as the object id.
21/// - [`Self::Position`] is used by a node's own owner index, ordered by object type and balance
22///   before the object id, and is written as base64.
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub enum OwnedObjectCursor {
25    /// A position in an index ordered by object id alone.
26    ObjectId(ObjectId),
27    /// A position in an index ordered by object type and balance before the
28    /// object id.
29    Position(IndexCursor),
30}
31
32impl OwnedObjectCursor {
33    /// A cursor for a store whose owner index is ordered by object id.
34    pub fn from_object_id(object_id: ObjectId) -> Self {
35        Self::ObjectId(object_id)
36    }
37
38    /// A cursor for a store whose owner index is ordered by type and balance.
39    pub fn from_position(cursor: IndexCursor) -> Self {
40        Self::Position(cursor)
41    }
42
43    /// The object the cursor names.
44    pub fn object_id(&self) -> ObjectId {
45        match self {
46            Self::ObjectId(object_id) => *object_id,
47            Self::Position(cursor) => cursor.object_id,
48        }
49    }
50
51    /// The full position, or `None` for a cursor that names only an object.
52    pub fn position(&self) -> Option<&IndexCursor> {
53        match self {
54            Self::ObjectId(_) => None,
55            Self::Position(cursor) => Some(cursor),
56        }
57    }
58}
59
60impl fmt::Display for OwnedObjectCursor {
61    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
62        match self {
63            Self::ObjectId(object_id) => write!(f, "{object_id}"),
64            Self::Position(cursor) => {
65                let bytes = bcs::to_bytes(cursor).map_err(|_| fmt::Error)?;
66                write!(f, "{}", Base64::encode(bytes))
67            }
68        }
69    }
70}
71
72impl FromStr for OwnedObjectCursor {
73    type Err = anyhow::Error;
74
75    fn from_str(s: &str) -> Result<Self, Self::Err> {
76        // An object id is `0x`-prefixed hex and a position is base64, so the
77        // two forms cannot be read as one another.
78        if let Ok(object_id) = s.parse::<ObjectId>() {
79            return Ok(Self::ObjectId(object_id));
80        }
81        let bytes = Base64::decode(s).map_err(|e| anyhow::anyhow!("invalid cursor: {e}"))?;
82        let cursor = bcs::from_bytes(&bytes).map_err(|e| anyhow::anyhow!("invalid cursor: {e}"))?;
83        Ok(Self::Position(cursor))
84    }
85}

87impl Serialize for OwnedObjectCursor {
88    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
89        serializer.collect_str(self)
90    }
91}

93impl<'de> Deserialize<'de> for OwnedObjectCursor {
94    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
95        let encoded = String::deserialize(deserializer)?;
96        encoded.parse().map_err(D::Error::custom)
97    }
98}

100impl JsonSchema for OwnedObjectCursor {
101    fn schema_name() -> String {
102        "OwnedObjectCursor".to_owned()
103    }
104
105    fn json_schema(generator: &mut schemars::r#gen::SchemaGenerator) -> schemars::schema::Schema {
106        let mut schema = String::json_schema(generator).into_object();
107        schema.metadata().description = Some(
108            "An opaque paging cursor. Pass back the `nextCursor` a page returned to read the \
109             page after it; its contents are not part of the API."
110                .to_owned(),
111        );
112        schema.into()
113    }
114}

116#[cfg(test)]
117mod tests {
118    use super::*;
119
120    /// Both forms survive a JSON round trip unchanged.
121    #[test]
122    fn both_forms_survive_the_round_trip_through_the_wire() {
123        let by_id = OwnedObjectCursor::from_object_id(ObjectId::random());
124        for inverted_balance in [None, Some(0), Some(u64::MAX), Some(42)] {
125            let by_position = OwnedObjectCursor::from_position(IndexCursor {
126                object_type_identifier: 7,
127                object_type_params: 9,
128                inverted_balance,
129                object_id: ObjectId::random(),
130            });
131            for cursor in [by_id, by_position] {
132                let json = serde_json::to_string(&cursor).unwrap();
133                assert!(json.starts_with('"'), "must be a JSON string: {json}");
134                assert_eq!(
135                    serde_json::from_str::<OwnedObjectCursor>(&json).unwrap(),
136                    cursor
137                );
138            }
139        }
140    }
141
142    /// The object-id form is written as the bare object id, keeping indexer
143    /// cursors in their existing wire format.
144    #[test]
145    fn the_object_id_form_is_written_as_the_object_id() {
146        let object_id = ObjectId::random();
147        let cursor = OwnedObjectCursor::from_object_id(object_id);
148        assert_eq!(cursor.to_string(), object_id.to_string());
149        assert_eq!(
150            serde_json::to_string(&cursor).unwrap(),
151            serde_json::to_string(&object_id).unwrap()
152        );
153    }
154
155    /// Both forms name an object; only the position form has a position.
156    #[test]
157    fn only_the_position_form_carries_one() {
158        let object_id = ObjectId::random();
159        let by_id = OwnedObjectCursor::from_object_id(object_id);
160        assert_eq!(by_id.object_id(), object_id);
161        assert!(by_id.position().is_none());
162
163        let by_position = OwnedObjectCursor::from_position(IndexCursor {
164            object_type_identifier: 1,
165            object_type_params: 2,
166            inverted_balance: Some(3),
167            object_id,
168        });
169        assert_eq!(by_position.object_id(), object_id);
170        assert!(by_position.position().is_some());
171    }
172
173    #[test]
174    fn a_malformed_cursor_is_refused_rather_than_read_as_a_position() {
175        assert!("not base64!".parse::<OwnedObjectCursor>().is_err());
176        assert!(
177            Base64::encode([0xff_u8; 3])
178                .parse::<OwnedObjectCursor>()
179                .is_err()
180        );
181    }
