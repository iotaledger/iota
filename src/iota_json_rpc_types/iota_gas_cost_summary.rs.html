9/// Summary of gas charges.
10///
11/// Storage is charged independently of computation.
12/// There are 3 parts to the storage charges:
13/// - `storage_cost`: it is the charge of storage at the time the transaction is
14///   executed. The cost of storage is the number of bytes of the objects being
15///   mutated multiplied by a variable storage cost per byte
16/// - `storage_rebate`: this is the amount a user gets back when manipulating an
17///   object. The `storage_rebate` is the `storage_cost` for an object minus
18///   fees.
19/// - `non_refundable_storage_fee`: not all the value of the object storage cost
20///   is given back to user and there is a small fraction that is kept by the
21///   system. This value tracks that charge.
22///
23/// When looking at a gas cost summary the amount charged to the user is
24/// `computation_cost + storage_cost - storage_rebate`
25/// and that is the amount that is deducted from the gas coins.
26/// `non_refundable_storage_fee` is collected from the objects being
27/// mutated/deleted and it is tracked by the system in storage funds.
28///
29/// Objects deleted, including the older versions of objects mutated, have the
30/// storage field on the objects added up to a pool of "potential rebate". This
31/// rebate then is reduced by the "nonrefundable rate" such that:
32/// `potential_rebate(storage cost of deleted/mutated objects) =
33/// storage_rebate + non_refundable_storage_fee`
34#[serde_as]
35#[derive(Serialize, Deserialize, JsonSchema)]
36#[serde(rename_all = "camelCase")]
37#[schemars(rename = "GasCostSummary")]
38pub struct IotaGasCostSummary {
39    /// Cost of computation/execution
40    #[schemars(with = "String")]
41    #[serde_as(as = "DisplayFromStr")]
42    pub computation_cost: u64,
43    /// The burned component of the computation/execution costs
44    #[schemars(with = "String")]
45    #[serde_as(as = "DisplayFromStr")]
46    pub computation_cost_burned: u64,
47    /// Storage cost, it's the sum of all storage cost for all objects created
48    /// or mutated.
49    #[schemars(with = "String")]
50    #[serde_as(as = "DisplayFromStr")]
51    pub storage_cost: u64,
52    /// The amount of storage cost refunded to the user for all objects deleted
53    /// or mutated in the transaction.
54    #[schemars(with = "String")]
55    #[serde_as(as = "DisplayFromStr")]
56    pub storage_rebate: u64,
57    /// The fee for the rebate. The portion of the storage rebate kept by the
58    /// system.
59    #[schemars(with = "String")]
60    #[serde_as(as = "DisplayFromStr")]
61    pub non_refundable_storage_fee: u64,
62}
