112    /// Get the coins for the given address filtered by coin type.
113    /// Returns a stream, which yields each coin once: a coin that comes back
114    /// on a later page, because its balance changed while the pages were read,
115    /// is yielded as first seen. Where the coin's current version matters,
116    /// collect the coins and keep each coin's highest version, as
117    /// [`Self::select_coins`] does.
118    ///
119    /// The coin type defaults to `0x2::iota::IOTA`.
120    ///
121    /// # Examples
122    ///
123    /// ```rust,no_run
124    /// use std::str::FromStr;
125    ///
126    /// use iota_sdk::IotaClientBuilder;
127    /// use iota_sdk_types::Address;
128    ///
129    /// #[tokio::main]
130    /// async fn main() -> Result<(), anyhow::Error> {
131    ///     let iota = IotaClientBuilder::default().build_testnet().await?;
132    ///     let address = Address::from_str("0x0000....0000")?;
133    ///     let coin_type = String::from("0x168da5bf1f48dafc111b0a488fa454aca95e0b5e::usdc::USDC");
134    ///     let coins = iota.coin_read_api().get_coins_stream(address, coin_type);
135    ///     Ok(())
136    /// }
137    /// ```
138    pub fn get_coins_stream(
139        &self,
140        owner: Address,
141        coin_type: impl Into<Option<String>>,
142    ) -> impl Stream<Item = Coin> + '_ {
143        once_each(self.coin_pages(owner, coin_type.into()))
144    }

146    /// The coins of every page `get_coins` returns, in order, including a coin
147    /// that comes back on a later page.
148    fn coin_pages(
149        &self,
150        owner: Address,
151        coin_type: Option<String>,
152    ) -> impl Stream<Item = Coin> + '_ {
153        stream::unfold(
154            (
155                vec![],
156                // cursor
157                None,
158                // has_next_page
159                true,
160                coin_type,
161            ),
162            move |(mut data, cursor, has_next_page, coin_type)| async move {
163                if let Some(item) = data.pop() {
164                    Some((item, (data, cursor, has_next_page, coin_type)))
165                } else if has_next_page {
166                    let page = self
167                        .get_coins(owner, coin_type.clone(), cursor, Some(100))
168                        .await
169                        .ok()?;
170                    let mut data = page.data;
171                    data.reverse();
172                    // Without a cursor the next request would restart the
173                    // stream from the first page.
174                    let has_next_page = page.has_next_page && page.next_cursor.is_some();
175                    data.pop()
176                        .map(|item| (item, (data, page.next_cursor, has_next_page, coin_type)))
177                } else {
178                    None
179                }
180            },
181        )
182    }

184    /// Get a list of coins for the given address filtered by coin type with at
185    /// least `amount` total value.
186    ///
187    /// If it is not possible to select enough coins, this function will return
188    /// an [`Error::InsufficientFunds`].
189    ///
190    /// The coin type defaults to `0x2::iota::IOTA`.
191    ///
192    /// # Examples
193    ///
194    /// ```rust,no_run
195    /// use std::str::FromStr;
196    ///
197    /// use iota_sdk::IotaClientBuilder;
198    /// use iota_sdk_types::Address;
199    ///
200    /// #[tokio::main]
201    /// async fn main() -> Result<(), anyhow::Error> {
202    ///     let iota = IotaClientBuilder::default().build_testnet().await?;
203    ///     let address = Address::from_str("0x0000....0000")?;
204    ///     let coin_type = String::from("0x168da5bf1f48dafc111b0a488fa454aca95e0b5e::usdc::USDC");
205    ///     let coins = iota
206    ///         .coin_read_api()
207    ///         .select_coins(address, coin_type, 5, vec![])
208    ///         .await?;
209    ///     Ok(())
210    /// }
211    /// ```
212    pub async fn select_coins(
213        &self,
214        address: Address,
215        coin_type: impl Into<Option<String>>,
216        amount: u128,
217        exclude: Vec<ObjectId>,
218    ) -> IotaRpcResult<Vec<Coin>> {
219        let (coins, total) =
220            select_from(self.coin_pages(address, coin_type.into()), amount, &exclude).await;
221        if total < amount {
222            return Err(Error::InsufficientFunds { address, amount });
223        }
224        Ok(coins)
225    }

348/// `coins` without the repeats of a coin.
349///
350/// Pages are ordered by balance, so a coin whose balance changes while they
351/// are read can come back on a later page; used twice, it would be a
352/// duplicate input.
353fn once_each(coins: impl Stream<Item = Coin>) -> impl Stream<Item = Coin> {
354    let mut seen = HashSet::new();
355    coins.filter(move |coin: &Coin| future::ready(seen.insert(coin.coin_object_id)))
356}
357
358/// The coins of `coins`, in order, until they add up to `amount`, leaving out
359/// `exclude`; with their total, which falls short of `amount` if `coins` runs
360/// out first.
361///
362/// A coin that comes back with a higher version, because its balance changed
363/// while the pages were read, takes the place of its earlier copy, so the
364/// selection holds the version a transaction can use.
365async fn select_from(
366    coins: impl Stream<Item = Coin>,
367    amount: u128,
368    exclude: &[ObjectId],
369) -> (Vec<Coin>, u128) {
370    let mut selected: Vec<Coin> = Vec::new();
371    let mut positions: HashMap<ObjectId, usize> = HashMap::new();
372    let mut total = 0u128;
373    let mut coins = pin!(coins);
374    while total < amount {
375        let Some(coin) = coins.next().await else {
376            break;
377        };
378        if exclude.contains(&coin.coin_object_id) {
379            continue;
380        }
381        match positions.entry(coin.coin_object_id) {
382            Entry::Occupied(position) => {
383                let held = &mut selected[*position.get()];
384                if coin.version > held.version {
385                    total = total - held.balance as u128 + coin.balance as u128;
386                    *held = coin;
387                }
388            }
389            Entry::Vacant(position) => {
390                position.insert(selected.len());
391                total += coin.balance as u128;
392                selected.push(coin);
393            }
394        }
395    }
396    (selected, total)
397}

403    fn coin(id: ObjectId, version: u64, balance: u64) -> Coin {
404        Coin {
405            coin_type: "0x2::iota::IOTA".to_string(),
406            coin_object_id: id,
407            version: version.into(),
408            digest: iota_sdk_types::ObjectDigest::ZERO,
409            balance,
410            previous_transaction: iota_sdk_types::TransactionDigest::ZERO,
411        }

414    /// The selected coins as `(id, version)`, in selection order.
415    fn picked(coins: &[Coin]) -> Vec<(ObjectId, u64)> {
416        coins
417            .iter()
418            .map(|coin| (coin.coin_object_id, coin.version.as_u64()))
419            .collect()
420    }

422    /// The stream yields a coin that comes back on a later page once, as
423    /// first seen.
424    #[tokio::test]
425    async fn a_coin_repeated_across_pages_is_yielded_once() {
426        let (a, b) = (ObjectId::random(), ObjectId::random());
427        let coins = once_each(stream::iter([coin(a, 1, 5), coin(b, 1, 3), coin(a, 2, 4)]));
428
429        let yielded: Vec<_> = coins.collect().await;
430        assert_eq!(picked(&yielded), [(a, 1), (b, 1)]);
431    }

433    /// A coin that comes back with a higher version replaces its earlier copy,
434    /// and the total follows its new balance.
435    #[tokio::test]
436    async fn a_newer_copy_of_a_selected_coin_replaces_it() {
437        let (a, b, c) = (ObjectId::random(), ObjectId::random(), ObjectId::random());
438        let coins = stream::iter([coin(a, 1, 5), coin(b, 1, 3), coin(a, 2, 4), coin(c, 1, 9)]);
439
440        let (selected, total) = select_from(coins, 15, &[]).await;
441        assert_eq!(picked(&selected), [(a, 2), (b, 1), (c, 1)]);
442        assert_eq!(total, 16);
443    }

445    /// A newer copy that lowers the total below the amount makes the selection
446    /// take further coins.
447    #[tokio::test]
448    async fn a_lower_newer_balance_selects_further_coins() {
449        let (a, b, c) = (ObjectId::random(), ObjectId::random(), ObjectId::random());
450        let coins = stream::iter([coin(a, 1, 5), coin(b, 1, 3), coin(a, 2, 1), coin(c, 1, 9)]);
451
452        let (selected, total) = select_from(coins, 8, &[]).await;
453        assert_eq!(picked(&selected), [(a, 1), (b, 1)]);
454        assert_eq!(total, 8, "the amount is covered before the newer copy");
455
456        let coins = stream::iter([coin(a, 1, 5), coin(b, 1, 3), coin(a, 2, 1), coin(c, 1, 9)]);
457        let (selected, total) = select_from(coins, 9, &[]).await;
458        assert_eq!(picked(&selected), [(a, 2), (b, 1), (c, 1)]);
459        assert_eq!(total, 13);
460    }
