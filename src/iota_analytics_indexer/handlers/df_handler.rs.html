32pub struct DynamicFieldHandler {
33    state: Mutex<State>,
34}
35
36struct State {
37    dynamic_fields: Vec<DynamicFieldEntry>,
38    package_store: LocalDBPackageStore,
39    resolver: Resolver<PackageCache>,
40}
41
42#[async_trait::async_trait]
43impl Worker for DynamicFieldHandler {
44    type Message = ();
45    type Error = anyhow::Error;
46
47    async fn process_checkpoint(
48        &self,
49        checkpoint_data: Arc<CheckpointData>,
50    ) -> Result<Self::Message, Self::Error> {
51        let CheckpointData {
52            checkpoint_summary,
53            transactions: checkpoint_transactions,
54            ..
55        } = checkpoint_data.as_ref();
56        let mut state = self.state.lock().await;
57        for checkpoint_transaction in checkpoint_transactions {
58            for object in checkpoint_transaction.output_objects.iter() {
59                state.package_store.update(object)?;
60            }
61            self.process_transaction(
62                checkpoint_summary.epoch,
63                checkpoint_summary.sequence_number,
64                checkpoint_summary.timestamp_ms,
65                checkpoint_transaction,
66                &mut state,
67            )
68            .await?;
69            if checkpoint_summary.end_of_epoch_data.is_some() {
70                state
71                    .resolver
72                    .package_store()
73                    .evict(SYSTEM_PACKAGE_ADDRESSES);
74            }
75        }
76        Ok(())
77    }
78}

80#[async_trait::async_trait]
81impl AnalyticsHandler<DynamicFieldEntry> for DynamicFieldHandler {
82    async fn read(&self) -> Result<Vec<DynamicFieldEntry>> {
83        let mut state = self.state.lock().await;
84        let cloned = state.dynamic_fields.clone();
85        state.dynamic_fields.clear();
86        Ok(cloned)
87    }
88
89    fn file_type(&self) -> Result<FileType> {
90        Ok(FileType::DynamicField)
91    }
92
93    fn name(&self) -> &str {
94        "dynamic_field"
95    }
96}

98impl DynamicFieldHandler {
99    pub fn new(store_path: &Path, client: iota_grpc_client::Client) -> Self {
100        let package_store = LocalDBPackageStore::new(&store_path.join("dynamic_field"), client);
101        let state = State {
102            dynamic_fields: vec![],
103            package_store: package_store.clone(),
104            resolver: Resolver::new(PackageCache::new(package_store)),
105        };
106        Self {
107            state: Mutex::new(state),
108        }
109    }
110    async fn process_dynamic_field(
111        &self,
112        epoch: u64,
113        checkpoint: u64,
114        timestamp_ms: u64,
115        object: &Object,
116        all_written_objects: &HashMap<ObjectId, Object>,
117        state: &mut State,
118    ) -> Result<()> {
119        let move_obj_opt = object.data.as_opt_struct();
120        // Skip if not a move object
121        let Some(move_object) = move_obj_opt else {
122            return Ok(());
123        };
124        if !move_object.struct_tag().is_dynamic_field() {
125            return Ok(());
126        }
127        let layout = state.resolver.type_layout(move_object.type_tag()).await?;
128        let object_id = object.id();
129
130        let field = DFV::FieldVisitor::deserialize(move_object.contents(), &layout)?;
131
132        let type_ = field.kind;
133        let name_type: TypeTag = type_tag_core_to_sdk(&field.name_layout.into());
134        let bcs_name = field.name_bytes.to_owned();
135
136        let name_value = BoundedVisitor::deserialize_value(field.name_bytes, field.name_layout)
137            .tap_err(|e| {
138                error!("{e}");
139            })?;
140        let name = DynamicFieldName {
141            type_tag: name_type,
142            value: IotaMoveValue::from(name_value).to_json_value(),
143        };
144        let name_json = serde_json::to_string(&name)?;
145        let (_owner_type, owner_id) = owner_to_owner_info(&object.owner);
146        let Some(parent_id) = owner_id else {
147            return Ok(());
148        };
149        let entry = match type_ {
150            DynamicFieldType::DynamicField => DynamicFieldEntry {
151                parent_object_id: parent_id.to_string(),
152                transaction_digest: object.previous_transaction.to_base58(),
153                checkpoint,
154                epoch,
155                timestamp_ms,
156                name: name_json,
157                bcs_name: Base64::encode(bcs_name),
158                type_,
159                object_id: object.id().to_string(),
160                version: object.version().as_u64(),
161                digest: object.digest().to_string(),
162                object_type: move_object.struct_tag().type_params()[1]
163                    .to_canonical_string(/* with_prefix */ true),
164            },
165            DynamicFieldType::DynamicObject => {
166                let object = all_written_objects.get(&object_id).ok_or(
167                    IndexerError::Uncategorized(anyhow!(
168                        "failed to find object_id {object_id} when trying to create dynamic field info",
169
170                    )),
171                )?;
172                let version = object.version().as_u64();
173                let digest = object.digest().to_string();
174                let object_type = object.data.opt_object_type().unwrap().clone();
175                DynamicFieldEntry {
176                    parent_object_id: parent_id.to_string(),
177                    transaction_digest: object.previous_transaction.to_base58(),
178                    checkpoint,
179                    epoch,
180                    timestamp_ms,
181                    name: name_json,
182                    bcs_name: Base64::encode(bcs_name),
183                    type_,
184                    object_id: object.id().to_string(),
185                    digest,
186                    version,
187                    object_type: object_type.to_canonical_string(true),
188                }
189            }
190        };
191        state.dynamic_fields.push(entry);
192        Ok(())
193    }
194
195    async fn process_transaction(
196        &self,
197        epoch: u64,
198        checkpoint: u64,
199        timestamp_ms: u64,
200        checkpoint_transaction: &CheckpointTransaction,
201        state: &mut State,
202    ) -> Result<()> {
203        let all_objects: HashMap<_, _> = checkpoint_transaction
204            .output_objects
205            .iter()
206            .map(|x| (x.id(), x.clone()))
207            .collect();
208        for object in checkpoint_transaction.output_objects.iter() {
209            self.process_dynamic_field(
210                epoch,
211                checkpoint,
212                timestamp_ms,
213                object,
214                &all_objects,
215                state,
216            )
217            .await?;
218        }
219        Ok(())
220    }
221}
