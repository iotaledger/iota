4/// Account authenticator verifier
5///
6/// This module contains the verifier for the authenticator function used
7/// by account objects to verify access to the account. The verifier checks
8/// that the function signature matches the expected signature for an
9/// authenticator function.
10use iota_types::{
11    auth_context::{AuthContext, AuthContextKind},
12    base_types::{TxContext, TxContextKind},
13    error::ExecutionError,
14    is_object_struct, is_primitive_strict,
15    transfer::Receiving,
16};
17use move_binary_format::{
18    CompiledModule,
19    file_format::{AbilitySet, SignatureToken, Visibility},
20};
21use move_bytecode_utils::format_signature_token;
22use move_core_types::identifier::IdentStr;
23
24use crate::verification_failure;

26/// Verify if a given function can be used as an authenticator function
27///
28/// A function is an authenticator function if:
29/// - only has read-only inputs (immutable owned/shared references or pure
30///   types)
31/// - has no return type
32/// - must be a public non-entry function
33/// - the first argument is a reference to the account object type (a Datatype
34///   or a concrete DatatypeInstantiation, i.e., with no template type
35///   parameters but concrete ones, both with `key` ability)
36/// - the last two arguments in order are AuthContext and TxContext
37/// - AuthContext has to be an immutable reference
38/// - TxContext has to be an immutable reference
39pub fn verify_authenticate_func_v1(
40    module: &CompiledModule,
41    function_identifier: &IdentStr,
42) -> Result<(), ExecutionError> {
43    let module_name = module.name();
44
45    let Some((_, function_definition)) =
46        module.find_function_def_by_name(function_identifier.as_str())
47    else {
48        return Err(verification_failure(format!(
49            "Authenticator function '{function_identifier}' not found in '{module_name}'"
50        )));
51    };
52
53    if function_definition.visibility != Visibility::Public {
54        return Err(verification_failure(format!(
55            "Authenticator function '{function_identifier}' must be public"
56        )));
57    }
58
59    if function_definition.is_entry {
60        return Err(verification_failure(format!(
61            "Authenticator function '{function_identifier}' cannot be marked as `entry`"
62        )));
63    }
64
65    let function_handle = module.function_handle_at(function_definition.function);
66    let function_signature = module.signature_at(function_handle.parameters);
67
68    // at least three arguments
69    if function_signature.0.len() < 3 {
70        return Err(verification_failure(format!(
71            "Authenticator function '{function_identifier}' must require at least: a reference to an account object type, &AuthContext and &TxContext arguments."
72        )));
73    }
74
75    // Check param 1st ///////////////////////////////////////
76
77    // The first parameter must match the authenticated account type.
78    // Additional restrictions on the first argument type are enforced in the
79    // following check.
80    let account_parameter = &function_signature.0[0];
81    verify_authenticate_account_type(module, &function_handle.type_parameters, account_parameter)
82        .map_err(verification_failure)?;
83
84    // Check params 2nd to N-2th /////////////////////////////
85
86    // Apart from AuthContext and TxContext we only require that the arguments are
87    // not mutable references. They can be mutable pure values, as their mutability
88    // cannot affect outside state.
89    for param in function_signature
90        .0
91        .iter()
92        .take(function_signature.len() - 2)
93    {
94        verify_authenticate_param_type(module, &function_handle.type_parameters, param)
95            .map_err(verification_failure)?;
96    }
97
98    // Check params N-1th and Nth ////////////////////////////
99
100    // Check type of AuthContext and TxContext, they both must be structs with the
101    // appropriate names, addresses and access
102    let auth_context = &function_signature.0[function_signature.len() - 2];
103    let tx_context = &function_signature.0[function_signature.len() - 1];
104
105    // AuthContext could potentially be passed as value, but that opens up the
106    // possibility for the authenticator function to receive it as mutable
107    // value, from which it could mutate before passing it to further `authenticate`
108    // functions, so similarly to TxContext, it is simply not allowed.
109    if !matches!(
110        AuthContext::kind(module, auth_context),
111        AuthContextKind::Immutable
112    ) {
113        return Err(verification_failure(format!(
114            "Authenticator function '{function_identifier}' can only receive 'AuthContext' as immutable reference"
115        )));
116    }
117
118    // TxContext can only be an immutable reference. Passing it as mutable would
119    // allow authenticator functions to create objects, which would be
120    // problematic.
121    if !matches!(
122        TxContext::kind(module, tx_context),
123        TxContextKind::Immutable
124    ) {
125        return Err(verification_failure(format!(
126            "Authenticator function '{function_identifier}' can only receive 'TxContext' as immutable reference"
127        )));
128    }
129
130    // Check return type (empty) /////////////////////////////
131
132    let return_signature = module.signature_at(function_handle.return_);
133    if !return_signature.is_empty() {
134        return Err(verification_failure(format!(
135            "Authenticator function '{function_identifier}' cannot have a return type"
136        )));
137    }
138
139    Ok(())
140}

142/// Verify that the first parameter type of the authenticator function is an
143/// immutable reference to an Object type, i.e., a Datatype with `key` ability.
144fn verify_authenticate_account_type(
145    module: &CompiledModule,
146    function_type_args: &[AbilitySet],
147    param: &SignatureToken,
148) -> Result<(), String> {
149    use SignatureToken::*;
150
151    // Check that the parameter is an immutable reference
152    if let Reference(ref_param) = param {
153        // Check if a type is a concrete object type (i.e., a Datatype with
154        // `key` ability or a DatatypeInstantiation with `key` ability
155        // and all type arguments being concrete object types).
156        let s = &**ref_param;
157        match s {
158            Datatype(_) => {
159                let abilities = module
160                    .abilities(s, function_type_args)
161                    .map_err(|vm_err| vm_err.to_string())?;
162                if abilities.has_key() {
163                    return Ok(());
164                }
165            }
166            DatatypeInstantiation(struct_inst) => {
167                let (_, type_args) = &**struct_inst;
168                let abilities = module
169                    .abilities(s, function_type_args)
170                    .map_err(|vm_err| vm_err.to_string())?;
171                if abilities.has_key() && type_args.iter().all(is_not_type_parameter) {
172                    return Ok(());
173                }
174            }
175            _ => {}
176        }
177    }
178    Err(format!(
179        "Invalid authenticator function account type: {}. Valid types for the first parameter are immutable references to an object type (with no generics).",
180        format_signature_token(module, param),
181    ))
182}

184/// Verify that the parameter type is a valid type for an authenticator
185/// function. Check that:
186/// - no Receiving objects are passed at all;
187/// - no objects are passed by value or by mutable reference, but only by
188///   immutable reference;
189/// - only primitive types are allowed by value.
190fn verify_authenticate_param_type(
191    module: &CompiledModule,
192    function_type_args: &[AbilitySet],
193    param: &SignatureToken,
194) -> Result<(), String> {
195    use SignatureToken::*;
196
197    // Reject receiving objects even if passed by immutable reference
198    if Receiving::is_receiving(module, param) {
199        return Err(format!(
200            "Invalid authenticator function parameter type: {}. Receiving objects are invalid. Valid types are immutable references to objects or primitive types.",
201            format_signature_token(module, param),
202        ));
203    }
204
205    match param {
206        Reference(inner) => {
207            if is_object_struct(module, function_type_args, inner)? {
208                Ok(())
209            } else {
210                Err(format!(
211                    "Invalid parameter type for authenticator function: {}. Non object immutable references are invalid. Valid types are immutable references to objects or primitive types.",
212                    format_signature_token(module, param)
213                ))
214            }
215        }
216        _ => {
217            if is_primitive_strict(module, function_type_args, param) {
218                Ok(())
219            } else {
220                Err(format!(
221                    "Invalid parameter type for authenticator function: {}. Valid types are immutable references to objects or primitive types.",
222                    format_signature_token(module, param)
223                ))
224            }
225        }
226    }
227}

229/// Check that a type is not a type parameter, recursively
230fn is_not_type_parameter(s: &SignatureToken) -> bool {
231    use SignatureToken as S;
232    match s {
233        S::TypeParameter(_) => false,
234        S::Bool
235        | S::U8
236        | S::U16
237        | S::U32
238        | S::U64
239        | S::U128
240        | S::U256
241        | S::Address
242        | S::Signer
243        | S::Datatype(_) => true,
244        S::DatatypeInstantiation(struct_inst) => {
245            let (_, type_args) = &**struct_inst;
246            type_args.iter().all(is_not_type_parameter)
247        }
248        S::Vector(inner) | S::Reference(inner) | S::MutableReference(inner) => {
249            is_not_type_parameter(inner)
250        }
251    }
