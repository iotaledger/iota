4use ledger_transport::APDUCommand;
5
6use crate::{
7    Transport,
8    api::{
9        bolos, errors, helpers,
10        packable::{Error as PackableError, Packable, Write},
11    },
12};
13
14#[derive(Debug)]
15pub struct Request {
16    pub app: String,
17}
18
19impl Packable for Request {
20    fn packed_len(&self) -> usize {
21        self.app.packed_len()
22    }
23
24    fn pack<W: Write>(&self, buf: &mut W) -> Result<(), PackableError> {
25        self.app.pack(buf)?;
26        Ok(())
27    }
28}
29
30pub fn exec<T: Transport>(transport: &T, app: String) -> Result<(), errors::LedgerError> {
31    let req = Request { app };
32
33    let mut buf = Vec::new();
34    let _ = req.pack(&mut buf);
35
36    // string serializer stores a length byte that is unwanted here because
37    // the p3 parameter will be the length of the string and the data itself
38    // must not contain the length
39    buf.remove(0);
40
41    let cmd = APDUCommand {
42        cla: bolos::APDU_CLA_E0,
43        ins: bolos::APDUInstructions::OpenAppE0 as u8,
44        p1: bolos::APDU_P1,
45        p2: bolos::APDU_P2,
46        data: buf,
47    };
48    helpers::exec::<T, ()>(transport, cmd).map_err(|e| match e {
49        errors::LedgerError::Syscall(errors::SyscallError::InvalidCounter) => {
50            errors::LedgerError::AppNotFound
51        }
52        _ => e,
53    })
54}
