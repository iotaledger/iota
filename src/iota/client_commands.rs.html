586    /// coins are selected automatically to cover the gas budget.

589}

591impl PaymentArgs {
592    /// Output the payment args as a vec of strings for CLI usage.
593    pub fn into_args(self) -> Vec<String> {
594        self.gas
595            .into_iter()
596            .map(|gas_id| format!("--gas {gas_id}"))
597            .collect()
598    }

601/// Arguments related to setting gas data, apart from payment coins.

604    /// An optional gas budget for this transaction (in NANOs). If gas budget is
605    /// not provided, the tool will first perform a dry run to estimate the
606    /// gas cost, and then it will execute the transaction. Please note that
607    /// this incurs a small cost in performance due to the additional
608    /// dry run call.
609    #[arg(long)]
610    pub gas_budget: Option<u64>,
611    /// An optional gas price for this transaction (in NANOs). If gas price is
612    /// not provided, the tool will use the current reference gas price from
613    /// RPC.
614    ///
615    /// Transactions with a gas price lower than the reference will not be
616    /// signed by enough validators to execute. Transactions accessing
617    /// congested shared objects are prioritized by gas price, so setting a
618    /// higher gas price higher than the reference can ensure the
619    /// transaction accesses the shared object sooner.
620    #[arg(long)]
621    pub gas_price: Option<u64>,
622    /// An optional field to specify a gas sponsor address. If provided, the gas
623    /// owner is set to this address, rather than the transaction's sender.
624    ///
625    /// Note that if the CLI does not have access to the sponsor's keys, it will
626    /// not be able to sign and execute transactions that have a sponsor

2099        }
2100    })?;
2101
2102    let resp = read_api
2103        .get_object_with_options(

3235/// Call a dry run with the transaction data to estimate the gas budget.
3236/// The estimated gas budget is computed as following:
3237/// * the maximum between A and B, where:
3238///
3239/// A = computation cost + GAS_SAFE_OVERHEAD * reference gas price
3240/// B = computation cost + storage cost - storage rebate + GAS_SAFE_OVERHEAD *
3241/// reference gas price overhead
3242///
3243/// This gas estimate is computed exactly as in the TypeScript SDK
3244/// <https://github.com/iotaledger/iota/blob/3c4369270605f78a243842098b7029daf8d883d9/sdk/typescript/src/transactions/TransactionBlock.ts#L845-L858>
3245pub async fn estimate_gas_budget(
3246    context: &mut WalletContext,
3247    signer: Address,
3248    kind: TransactionKind,
3249    gas_price: u64,
3250    gas_payment: Vec<ObjectReference>,
3251    sponsor: Option<Address>,
3252) -> Result<u64, anyhow::Error> {
3253    let client = context.get_client().await?;
3254    let dry_run =
3255        execute_dry_run(context, signer, kind, None, gas_price, gas_payment, sponsor).await;
3256
3257    if let Ok(IotaClientCommandResult::DryRun(dry_run)) = dry_run {
3258        let rgp = client.read_api().get_reference_gas_price().await?;
3259        Ok(estimate_gas_budget_from_gas_cost(
3260            dry_run.effects.gas_cost_summary(),
3261            rgp,
3262        ))
3263    } else {
3264        bail!(
3265            "Could not determine the gas budget. Error: {}",
3266            dry_run.unwrap_err()
3267        )
3268    }
3269}

3271pub fn estimate_gas_budget_from_gas_cost(
3272    gas_cost_summary: &GasCostSummary,
3273    reference_gas_price: u64,
3274) -> u64 {
3275    let safe_overhead = GAS_SAFE_OVERHEAD * reference_gas_price;
3276    let computation_cost_with_overhead = gas_cost_summary.computation_cost + safe_overhead;
3277
3278    let gas_usage = gas_cost_summary.net_gas_usage() + safe_overhead as i64;
3279    computation_cost_with_overhead.max(if gas_usage < 0 { 0 } else { gas_usage as u64 })
3280}

3709/// code. The algorithm finds the immediate dependencies of this package, and
3710/// the original ids of each transitive dependencies for all these immediate
3711/// package dependencies. For packages that are not referenced in the source
3712/// code, they will be filtered out from the list of dependencies.
3713pub(crate) async fn pkg_tree_shake(
3714    read_api: &ReadApi,
3715    with_unpublished_dependencies: bool,
3716    compiled_package: &mut CompiledPackage,
3717) -> Result<(), anyhow::Error> {
3718    // these are packages that are immediate dependencies of the root package
3719    let immediate_dep_packages =
3720        compiled_package.find_immediate_deps_pkgs_to_keep(with_unpublished_dependencies)?;
3721
3722    // for every immediate dependency package, we need to use its linkage table to
3723    // determine its transitive dependencies and ensure that we keep the
3724    // required packages, so fetch those tables
3725    let trans_deps_orig_ids = trans_deps_original_ids(read_api, &immediate_dep_packages).await?;
3726    let pkg_name_to_orig_id: BTreeMap<_, _> = compiled_package
3727        .package
3728        .deps_compiled_units
3729        .iter()
3730        .map(|(pkg_name, module)| (*pkg_name, ObjectId::new(module.unit.address.into_bytes())))
3731        .collect();
3732
3733    // for every published package in the original list of published dependencies,
3734    // get its original id and then check if that id exists in the linkage
3735    // table. If it does, then we need to keep this package. Similarly, all
3736    // immediate dep packages must stay
3737    compiled_package.dependency_ids.published.retain(|pkg, _| {
3738        immediate_dep_packages.contains_key(pkg)
3739            || pkg_name_to_orig_id
3740                .get(pkg)
3741                .is_some_and(|id| trans_deps_orig_ids.contains(id))
3742    });
3743
3744    Ok(())
3745}

3797    let call_args = resolve_auth_call_args(
3798        client,
3799        auth_info.value.package,
3800        &auth_info.value.module,
3801        &auth_info.value.function,
3802        &type_args,
3803        json_args,
3804    )
3805    .await?;
3806    Ok(Some((call_args, type_args)))
3807}

3864/// Processes authentication arguments.
3865pub(crate) fn process_auth_args(
3866    auth_call_args: Option<&Vec<String>>,
3867    auth_type_args: Option<&Vec<String>>,
3868    signer: Address,
3869) -> Result<(Vec<TypeTag>, Vec<IotaJsonValue>), anyhow::Error> {
3870    let type_args = auth_type_args
3871        .as_ref()
3872        .map(|args| {
3873            args.iter()
3874                .map(|arg| TypeTag::from_str(arg))
3875                .collect::<Result<Vec<_>, _>>()
3876        })
3877        .transpose()?
3878        .unwrap_or_default();
3879
3880    let mut json_args: Vec<_> = auth_call_args
3881        .as_ref()
3882        .map(|args| {
3883            args.iter()
3884                .map(|arg| {
3885                    // Use bracket-aware parsing only for array-like args
3886                    // (e.g. "[0xAA,0xBBCC]"), so that nested vectors are
3887                    // correctly split. Everything else is kept as a JSON
3888                    // string, which the resolver handles for u64, hex
3889                    // bytes, addresses, object IDs, etc.
3890                    if arg.starts_with('[') {
3891                        IotaJsonValue::from_str(arg).unwrap_or_else(|_| {
3892                            IotaJsonValue::new(serde_json::to_value(arg).unwrap()).unwrap()
3893                        })
3894                    } else {
3895                        IotaJsonValue::new(serde_json::to_value(arg).unwrap()).unwrap()
3896                    }
3897                })
3898                .collect()
3899        })
3900        .unwrap_or_default();
3901
3902    json_args.insert(
3903        0,
3904        IotaJsonValue::new(serde_json::to_value(signer).unwrap()).unwrap(),
3905    );
3906
3907    Ok((type_args, json_args))
3908}
