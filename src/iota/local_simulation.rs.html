4//! Run transaction simulations locally through [`iota_vm_sdk::LocalVm`]
5//! instead of the node's dry-run and dev-inspect endpoints.
6//!
7//! Objects and chain parameters are resolved on demand from the active env's
8//! gRPC endpoint; execution itself happens in-process, against the same Move
9//! engine a node uses. The results are assembled into the same
10//! [`DryRunTransactionBlockResponse`] / [`DevInspectResults`] the node
11//! returns, so they render through the same display code.
12//!
13//! The gas price defaults to the reference gas price the gRPC endpoint
14//! reports, and a dry run's Move abort is rendered from the packages the run
15//! fetched, so the run needs no JSON-RPC.
16//!
17//! Two checks a validator applies are out of reach here, so a transaction the
18//! node rejects can still succeed locally: the operator's transaction
19//! deny-list, and the network's signing verifier limits — this runs with an
20//! empty deny-list and the default limits.
21

50/// Run a dry run locally and return the same
51/// [`DryRunTransactionBlockResponse`] a node returns.
52///
53/// The run blocks the calling thread until it is done. Resolving an object
54/// the run asks for needs a multi-threaded Tokio runtime.
55pub(crate) async fn execute_local_dry_run(
56    context: &WalletContext,
57    signer: Address,
58    kind: TransactionKind,
59    gas_budget: Option<u64>,
60    gas_price: Option<u64>,
61    gas_payment: Vec<ObjectReference>,
62    sponsor: Option<Address>,
63) -> Result<IotaClientCommandResult> {
64    let (store, chain_context) = connect(context).await?;
65    let vm = LocalVm::new(chain_context, store.clone())?;
66    let (mut response, status) = run_dry_run(
67        vm,
68        signer,
69        kind,
70        gas_budget,
71        gas_price,
72        gas_payment,
73        sponsor,
74    )?;
75
76    // Render a Move abort as the node does, from the packages the run fetched.
77    // Like the node, this cannot see a package the run itself published.
78    let resolver = Resolver::new(FetchedPackages(store.store()));
79    let IotaTransactionBlockEffects::V1(effects) = &mut response.effects;
80    effects.status = IotaExecutionStatus::from_native_with_clever_error(status, &resolver).await?;
81    Ok(IotaClientCommandResult::DryRun(DryRunOutput {
82        response,
83        local: true,
84    }))
85}

87/// Run a dev inspect locally and return the same [`DevInspectResults`] a node
88/// returns.
89///
90/// The run blocks the calling thread until it is done. Resolving an object
91/// the run asks for needs a multi-threaded Tokio runtime.
92pub(crate) async fn execute_local_dev_inspect(
93    context: &WalletContext,
94    signer: Address,
95    kind: TransactionKind,
96    gas_budget: Option<u64>,
97    gas_price: Option<u64>,
98    gas_payment: Vec<ObjectReference>,
99    sponsor: Option<Address>,
100) -> Result<IotaClientCommandResult> {
101    let (store, chain_context) = connect(context).await?;
102    let mut vm = LocalVm::new(chain_context, store)?;
103    // A zero budget or price is filled in by the run the same way the node's
104    // dev-inspect fills it in, so neither is resolved here.
105    let tx_data = Transaction::new_with_gas_coins_allow_sponsor(
106        kind,
107        signer,
108        gas_payment,
109        gas_budget.unwrap_or_default(),
110        gas_price.unwrap_or_default(),
111        sponsor.unwrap_or(signer),
112    );
113    let mut result = vm.execute(tx_data, ExecuteOptions::dev_inspect())?;
114
115    let return_values = match result.execution_error.take() {
116        Some(error) => Err(error),
117        None => Ok(std::mem::take(&mut result.command_results)),
118    };
119    let events = result.events.take().unwrap_or_default();
120    let effects = result.effects.clone();
121    let mut layout_resolver = result.layout_resolver(&vm)?;
122    // The node's dev inspect converts the effects as-is, so a Move abort is
123    // reported raw, unlike a dry run. The CLI does not ask for the raw
124    // transaction or effects bytes.
125    let response = DevInspectResults::new(
126        effects,
127        events,
128        return_values,
129        vec![],
130        vec![],
131        layout_resolver.as_mut(),
132    )?;
133    Ok(IotaClientCommandResult::DevInspect(DevInspectOutput {
134        response,
135        local: true,
136    }))
137}

139/// Estimate a gas budget from a local dry run, as
140/// [`estimate_gas_budget`](crate::client_commands::estimate_gas_budget) does
141/// from a node's.
142pub(crate) async fn estimate_local_gas_budget(
143    context: &WalletContext,
144    signer: Address,
145    kind: TransactionKind,
146) -> Result<u64> {
147    let (store, chain_context) = connect(context).await?;
148    let reference_gas_price = chain_context.reference_gas_price;
149    let vm = LocalVm::new(chain_context, store)?;
150    let (response, _) = run_dry_run(vm, signer, kind, None, None, Vec::new(), None)?;
151    Ok(estimate_gas_budget_from_gas_cost(
152        response.effects.gas_cost_summary(),
153        reference_gas_price,
154    ))
155}

157async fn connect(context: &WalletContext) -> Result<(GrpcStore, ChainContext)> {
158    let client = context.get_grpc_client().await.context(
159        "local simulation needs a gRPC endpoint; set `grpc` for the active env in client.yaml",
160    )?;
161    let store = GrpcStore::new(client);
162    let chain_context = store.fetch_chain_context().await?;
163    Ok((store, chain_context))
164}

166struct FetchedPackages(InMemoryStore);
167
168#[async_trait]
169impl PackageStore for FetchedPackages {
170    async fn fetch(&self, address: Address) -> Result<Arc<Package>, PackageResolverError> {
171        let id = ObjectId::from(address);
172        match self.0.get_object(&id, None) {
173            Ok(Some(object)) => Package::read_from_object(&object).map(Arc::new),
174            _ => Err(PackageResolverError::PackageNotFound(id.into())),
175        }
176    }
177}

179fn run_dry_run(
180    mut vm: LocalVm,
181    signer: Address,
182    kind: TransactionKind,
183    gas_budget: Option<u64>,
184    gas_price: Option<u64>,
185    gas_payment: Vec<ObjectReference>,
186    sponsor: Option<Address>,
187) -> Result<(DryRunTransactionBlockResponse, ExecutionStatus)> {
188    let gas_price = gas_price.unwrap_or_else(|| vm.reference_gas_price());
189
190    let tx_data = Transaction::new_with_gas_coins_allow_sponsor(
191        kind,
192        signer,
193        gas_payment,
194        gas_budget.unwrap_or_default(),
195        gas_price,
196        sponsor.unwrap_or(signer),
197    );
198
199    let result = vm.execute(tx_data.clone(), ExecuteOptions::dry_run())?;
200    let status = result.effects.status().clone();
201    Ok((dry_run_response(&vm, tx_data, result)?, status))
202}

204/// Assemble a [`DryRunTransactionBlockResponse`] from a local run, resolving
205/// Move layouts from the packages the run wrote and those in the VM's store.
206fn dry_run_response(
207    vm: &LocalVm,
208    mut tx_data: Transaction,
209    mut result: ExecutionResult,
210) -> Result<DryRunTransactionBlockResponse> {
211    let tx_digest = *result.effects.transaction_digest();
212
213    let execution_error_source = result
214        .execution_error
215        .as_ref()
216        .and_then(|error| error.source().as_ref().map(|source| source.to_string()));
217    let object_changes = result
218        .object_changes()?
219        .into_iter()
220        .map(TryInto::try_into)
221        .collect::<Result<_, _>>()?;
222    let balance_changes = result
223        .balance_changes()?
224        .into_iter()
225        .map(Into::into)
226        .collect();
227    let raw_events = result.events.take().unwrap_or_default();
228
229    // A dry run is not committed, so a package it published is not in the
230    // store; the run's resolver reads it from the run's output first.
231    let resolver = result.module_resolver(vm);
232    let events = IotaTransactionBlockEvents::try_from_using_module_resolver(
233        raw_events, tx_digest, None, &resolver,
234    )?;
235    // Report the gas the run used in place of whatever the caller left unset,
236    // as the node does before it builds the response's input.
237    report_simulation_gas(
238        tx_data.gas_data_mut(),
239        result.transaction.gas_data(),
240        result.effects.gas_cost_summary().gas_used(),
241    );
242    let input =
243        IotaTransactionBlockData::try_from_with_module_cache(tx_data, &resolver, tx_digest)?;
244
245    let ExecutionResult {
246        effects,
247        suggested_gas_price,
248        ..
249    } = result;
250
251    Ok(DryRunTransactionBlockResponse {
252        effects: effects.try_into()?,
253        events,
254        object_changes,
255        balance_changes,
256        input,
257        // Congestion is only observable on a node, so where the SDK withholds
258        // a suggestion report the reference gas price — what a node suggests
259        // when nothing is congested.
260        suggested_gas_price: Some(suggested_gas_price.unwrap_or_else(|| vm.reference_gas_price())),
261        execution_error_source,
262    })
