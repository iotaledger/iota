145/// A parsed PTB command consisting of the command and the parsed arguments to
146/// the command.
147#[derive(Debug, Clone)]
148pub enum ParsedPTBCommand {
149    TransferObjects(Spanned<Vec<Spanned<Argument>>>, Spanned<Argument>),
150    SplitCoins(Spanned<Argument>, Spanned<Vec<Spanned<Argument>>>),
151    MergeCoins(Spanned<Argument>, Spanned<Vec<Spanned<Argument>>>),
152    MakeMoveVec(Spanned<ParsedType>, Spanned<Vec<Spanned<Argument>>>),
153    MoveCall(
154        Spanned<ModuleAccess>,
155        Option<Spanned<Vec<ParsedType>>>,
156        Vec<Spanned<Argument>>,
157    ),
158    Assign(Spanned<String>, Option<Spanned<Argument>>),
159    Publish(Spanned<String>),
160    Upgrade(Spanned<String>, Spanned<Argument>),
161    /// execute-upgrade \<ticket\>
162    /// Executes the system upgrade using previously compiled package data.
163    /// Returns the UpgradeReceipt.
164    ExecuteUpgrade(Spanned<Argument>),
165    /// compile-upgrade \<path\> \<upgrade_cap\>
166    /// Compiles the package and returns the digest as a pure value.
167    /// Stores compiled data for a subsequent execute-upgrade.
168    CompileUpgrade(Spanned<String>, Spanned<Argument>),
169    WarnShadows,
170    Preview,
171}

173/// An enum representing the parsed arguments of a PTB command.
174#[derive(Debug, Clone)]
175pub enum Argument {
176    Bool(bool),
177    U8(u8),
178    U16(u16),
179    U32(u32),
180    U64(u64),
181    U128(u128),
182    U256(move_core_types::u256::U256),
183    InferredNum(move_core_types::u256::U256),
184    Gas,
185    Identifier(String),
186    VariableAccess(Spanned<String>, Vec<Spanned<String>>),
187    Address(NumericalAddress),
188    String(String),
189    Vector(Vec<Spanned<Argument>>),
190    Option(Spanned<Option<Box<Argument>>>),
191}

193impl Argument {
194    /// Resolve an `Argument` into a `MoveValue` if possible. Errors if the
195    /// `Argument` is not convertible to a `MoveValue` of the provided
196    /// `tag`.
197    pub fn checked_to_pure_move_value(&self, loc: Span, tag: &TypeTag) -> PTBResult<MoveValue> {
198        Ok(match (self, tag) {
199            (Argument::Bool(b), TypeTag::Bool) => MoveValue::Bool(*b),
200            (Argument::U8(u), TypeTag::U8) => MoveValue::U8(*u),
201            (Argument::U16(u), TypeTag::U16) => MoveValue::U16(*u),
202            (Argument::U32(u), TypeTag::U32) => MoveValue::U32(*u),
203            (Argument::U64(u), TypeTag::U64) => MoveValue::U64(*u),
204            (Argument::U128(u), TypeTag::U128) => MoveValue::U128(*u),
205            (Argument::U256(u), TypeTag::U256) => MoveValue::U256(*u),
206            // Inferred numbers, unless they need to be used as a specific type default to u64.
207            (Argument::InferredNum(u), tag) => Self::cast_inferred_num(*u, tag, loc)?,
208            (Argument::Address(a), TypeTag::Address) => MoveValue::Address(a.into_inner()),
209            (Argument::Vector(vs), TypeTag::Vector(ty)) => MoveValue::Vector(
210                vs.iter()
211                    .map(|sp!(loc, v)| v.checked_to_pure_move_value(*loc, ty))
212                    .collect::<PTBResult<Vec<_>>>()
213                    .map_err(|e| {
214                        e.with_help("Literal vectors cannot contain object values.".to_string())
215                    })?,
216            ),
217            (Argument::String(s), TypeTag::Vector(ty)) if **ty == TypeTag::U8 => {
218                MoveValue::Vector(s.bytes().map(MoveValue::U8).collect::<Vec<_>>())
219            }
220            (Argument::String(s), TypeTag::Struct(stag))
221                if stag.is_string() || stag.is_ascii_string() =>
222            {
223                MoveValue::Vector(s.bytes().map(MoveValue::U8).collect::<Vec<_>>())
224            }
225            (Argument::Address(a), TypeTag::Struct(stag)) if stag.is_id() => {
226                MoveValue::Address(a.into_inner())
227            }
228            (Argument::Option(sp!(loc, o)), TypeTag::Vector(ty)) => {
229                if let Some(v) = o {
230                    let v = v
231                        .as_ref()
232                        .checked_to_pure_move_value(*loc, ty)
233                        .map_err(|e| {
234                            e.with_help(
235                                "Literal option values cannot contain object values.".to_string(),
236                            )
237                        })?;
238                    MoveValue::Vector(vec![v])
239                } else {
240                    MoveValue::Vector(vec![])
241                }
242            }
243            (Argument::Option(sp!(loc, o)), TypeTag::Struct(stag)) if stag.is_option() => {
244                if let Some(v) = o {
245                    let v = v
246                        .as_ref()
247                        .checked_to_pure_move_value(*loc, &stag.type_params()[0])
248                        .map_err(|e| {
249                            e.with_help(
250                                "Literal option values cannot contain object values.".to_string(),
251                            )
252                        })?;
253                    MoveValue::Vector(vec![v])
254                } else {
255                    MoveValue::Vector(vec![])
256                }
257            }
258            (Argument::Identifier(_) | Argument::VariableAccess(_, _) | Argument::Gas, _) => {
259                error!(loc, "Unable to convert '{self}' to non-object value.")
260            }
261            (arg, tag) => error!(loc, "Unable to serialize '{arg}' as a {tag} value"),
262        })
263    }
264
265    /// Resolve an `Argument` into a `MoveValue` if possible. Errors if the
266    /// `Argument` is not convertible to a `MoveValue`.
267    pub fn to_pure_move_value(&self, loc: Span) -> PTBResult<MoveValue> {
268        Ok(match self {
269            Argument::Bool(b) => MoveValue::Bool(*b),
270            Argument::U8(u) => MoveValue::U8(*u),
271            Argument::U16(u) => MoveValue::U16(*u),
272            Argument::U32(u) => MoveValue::U32(*u),
273            Argument::U64(u) => MoveValue::U64(*u),
274            Argument::U128(u) => MoveValue::U128(*u),
275            Argument::U256(u) => MoveValue::U256(*u),
276            // Inferred numbers, unless they need to be used as a specific type default to u64.
277            Argument::InferredNum(u) => Self::cast_inferred_num(*u, &TypeTag::U64, loc)?,
278            Argument::Address(a) => MoveValue::Address(a.into_inner()),
279            Argument::Vector(vs) => MoveValue::Vector(
280                vs.iter()
281                    .map(|sp!(loc, v)| v.to_pure_move_value(*loc))
282                    .collect::<PTBResult<Vec<_>>>()
283                    .map_err(|e| {
284                        e.with_help("Literal vectors cannot contain object values.".to_string())
285                    })?,
286            ),
287            Argument::String(s) => {
288                MoveValue::Vector(s.bytes().map(MoveValue::U8).collect::<Vec<_>>())
289            }
290            Argument::Option(sp!(loc, o)) => {
291                if let Some(v) = o {
292                    let v = v.as_ref().to_pure_move_value(*loc).map_err(|e| {
293                        e.with_help(
294                            "Literal option values cannot contain object values.".to_string(),
295                        )
296                    })?;
297                    MoveValue::Vector(vec![v])
298                } else {
299                    MoveValue::Vector(vec![])
300                }
301            }
302            Argument::Identifier(_) | Argument::VariableAccess(_, _) | Argument::Gas => {
303                error!(loc, "Unable to convert '{self}' to non-object value.")
304            }
305        })
306    }
307
308    fn cast_inferred_num(
309        val: move_core_types::u256::U256,
310        tag: &TypeTag,
311        loc: Span,
312    ) -> PTBResult<MoveValue> {
313        match tag {
314            TypeTag::U8 => u8::try_from(val)
315                .map(MoveValue::U8)
316                .map_err(|_| err!(loc, "Value {val} is too large to be a u8 value")),
317            TypeTag::U16 => u16::try_from(val)
318                .map(MoveValue::U16)
319                .map_err(|_| err!(loc, "Value {val} is too large to be a u16 value")),
320            TypeTag::U32 => u32::try_from(val)
321                .map(MoveValue::U32)
322                .map_err(|_| err!(loc, "Value {val} is too large to be a u32 value")),
323            TypeTag::U64 => u64::try_from(val)
324                .map(MoveValue::U64)
325                .map_err(|_| err!(loc, "Value {val} is too large to be a u64 value")),
326            TypeTag::U128 => u128::try_from(val)
327                .map(MoveValue::U128)
328                .map_err(|_| err!(loc, "Value {val} is too large to be a u128 value")),
329            TypeTag::U256 => Ok(MoveValue::U256(val)),
330            _ => error!(loc, "Expected an integer type but got {tag} for '{val}'"),
331        }
332    }
333}

335impl fmt::Display for Argument {
336    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
337        match self {
338            Argument::Bool(b) => write!(f, "{b}"),
339            Argument::U8(u) => write!(f, "{u}u8"),
340            Argument::U16(u) => write!(f, "{u}u16"),
341            Argument::U32(u) => write!(f, "{u}u32"),
342            Argument::U64(u) => write!(f, "{u}u64"),
343            Argument::U128(u) => write!(f, "{u}u128"),
344            Argument::U256(u) => write!(f, "{u}u256"),
345            Argument::InferredNum(u) => write!(f, "{u}"),
346            Argument::Gas => write!(f, "gas"),
347            Argument::Identifier(i) => write!(f, "{i}"),
348            Argument::VariableAccess(sp!(_, head), accesses) => {
349                write!(f, "{head}")?;
350                for sp!(_, access) in accesses {
351                    write!(f, ".{access}")?;
352                }
353                Ok(())
354            }
355            Argument::Address(a) => write!(f, "@{a}"),
356            Argument::String(s) => write!(f, "{s:?}"),
357            Argument::Vector(v) => {
358                write!(f, "vector[")?;
359                let mut prefix = "";
360                for sp!(_, arg) in v.iter() {
361                    write!(f, "{prefix}")?;
362                    write!(f, "{arg}")?;
363                    prefix = ", ";
364                }
365                write!(f, "]")
366            }
367            Argument::Option(sp!(_, o)) => match o {
368                Some(v) => write!(f, "some({v})"),
369                None => write!(f, "none"),
370            },
371        }
372    }
373}

375fn delimited_list<T: fmt::Display>(
376    f: &mut std::fmt::Formatter<'_>,
377    sep: &str,
378    items: impl IntoIterator<Item = T>,
379) -> std::fmt::Result {
380    let mut prefix = "";
381    for item in items {
382        write!(f, "{prefix}{item}")?;
383        prefix = sep;
384    }
385    Ok(())
386}

388impl fmt::Display for ParsedPTBCommand {
389    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
390        match self {
391            ParsedPTBCommand::TransferObjects(args, arg) => {
392                write!(f, "{TRANSFER_OBJECTS} [")?;
393                delimited_list(f, ", ", args.value.iter().map(|x| &x.value))?;
394                write!(f, "]")?;
395                write!(f, " {}", arg.value)
396            }
397            ParsedPTBCommand::SplitCoins(arg, args) => {
398                write!(f, "{SPLIT_COINS} {} [", arg.value)?;
399                delimited_list(f, ", ", args.value.iter().map(|x| &x.value))?;
400                write!(f, "]")
401            }
402            ParsedPTBCommand::MergeCoins(arg, args) => {
403                write!(f, "{MERGE_COINS} {} [", arg.value)?;
404                delimited_list(f, ", ", args.value.iter().map(|x| &x.value))?;
405                write!(f, "]")
406            }
407            ParsedPTBCommand::Assign(arg, arg_opt) => write!(
408                f,
409                "{ASSIGN} {}{}",
410                arg.value,
411                if let Some(arg) = arg_opt {
412                    format!(" {}", arg.value)
413                } else {
414                    "".to_string()
415                }
416            ),
417            ParsedPTBCommand::Publish(s) => write!(f, "{PUBLISH} {}", s.value),
418            ParsedPTBCommand::Upgrade(s, a) => write!(f, "{UPGRADE} {} {}", s.value, a.value),
419            ParsedPTBCommand::ExecuteUpgrade(ticket) => {
420                write!(f, "{EXECUTE_UPGRADE} {}", ticket.value)
421            }
422            ParsedPTBCommand::CompileUpgrade(s, cap) => {
423                write!(f, "{COMPILE_UPGRADE} {} {}", s.value, cap.value)
424            }
425            ParsedPTBCommand::WarnShadows => write!(f, "{WARN_SHADOWS}"),
426            ParsedPTBCommand::Preview => write!(f, "{PREVIEW}"),
427            ParsedPTBCommand::MakeMoveVec(ty, args) => {
428                write!(f, "{MAKE_MOVE_VEC} <",)?;
429                write!(f, "{}", TyDisplay(&ty.value))?;
430                write!(f, "> [")?;
431                delimited_list(f, ", ", args.value.iter().map(|x| &x.value))?;
432                write!(f, "]")
433            }
434            ParsedPTBCommand::MoveCall(
435                sp!(
436                    _,
437                    ModuleAccess {
438                        address,
439                        module_name,
440                        function_name
441                    }
442                ),
443                tys,
444                args,
445            ) => {
446                let type_args = |f: &mut std::fmt::Formatter| match tys {
447                    Some(tys) => {
448                        write!(f, "<")?;
449                        delimited_list(f, ", ", tys.value.iter().map(TyDisplay))?;
450                        write!(f, ">")
451                    }
452                    None => Ok(()),
453                };
454                write!(
455                    f,
456                    "{MOVE_CALL} {}::{}::{}",
457                    address.value, module_name.value, function_name.value
458                )?;
459                type_args(f)?;
460
461                if !args.is_empty() {
462                    write!(f, " ")?;
463                }
464
465                delimited_list(f, " ", args.iter().map(|x| x.value.to_string()))
466            }
467        }
468    }
469}

471struct TyDisplay<'a>(&'a ParsedType);
472
473impl fmt::Display for TyDisplay<'_> {
474    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
475        use ParsedType::*;
476        match self.0 {
477            Address => write!(f, "address"),
478            U8 => write!(f, "u8"),
479            U16 => write!(f, "u16"),
480            U32 => write!(f, "u32"),
481            U64 => write!(f, "u64"),
482            U128 => write!(f, "u128"),
483            U256 => write!(f, "u256"),
484            Bool => write!(f, "bool"),
485            Signer => write!(f, "signer"),
486            Vector(ty) => write!(f, "vector<{}>", TyDisplay(ty)),
487            Struct(ParsedStructType {
488                fq_name:
489                    ParsedFqName {
490                        module: ParsedModuleId { address, name },
491                        name: struct_name,
492                    },
493                type_args,
494            }) => {
495                write!(f, "{address}::{name}::{struct_name}")?;
496                if type_args.is_empty() {
497                    Ok(())
498                } else {
499                    write!(f, "<")?;
500                    delimited_list(f, ", ", type_args.iter().map(TyDisplay))?;
501                    write!(f, ">")
502                }
503            }
504        }
505    }
