19pub(crate) async fn load_objects_with_client(
20    client: &IotaClient,
21    transaction: &Transaction,
22) -> Result<Vec<Object>, LedgerSignerError> {
23    let object_ids = object_ids_from_transaction(transaction)?;
24
25    if object_ids.is_empty() {
26        return Ok(vec![]);
27    }
28
29    let responses = client
30        .read_api()
31        .multi_get_object_with_options(object_ids, IotaObjectDataOptions::bcs_lossless())
32        .await?;
33
34    let objects: Vec<Object> = responses
35        .into_iter()
36        .filter_map(object_from_response)
37        .collect();
38
39    Ok(objects)
40}
41
42fn object_ids_from_transaction(
43    transaction: &Transaction,
44) -> Result<Vec<ObjectId>, LedgerSignerError> {
45    let object_ids = transaction
46        .gas_data()
47        .objects
48        .iter()
49        .map(|object| object.object_id);
50
51    let input_objects = transaction
52        .input_objects()?
53        .into_iter()
54        .filter_map(|input| match input {
55            InputObjectKind::ImmOrOwnedMoveObject(id) => Some(id.object_id),
56            _ => None,
57        });
58
59    let mut unique_ids = HashSet::new();
60    unique_ids.extend(object_ids);
61    unique_ids.extend(input_objects);
62
63    Ok(unique_ids.into_iter().collect())
64}

66// Convert IotaObjectResponse to supported clear-sign Objects returning None if
67// the conversion fails
68fn object_from_response(resp: IotaObjectResponse) -> Option<Object> {
69    let data: IotaObjectData = resp.data?;
70
71    let move_object_type = match data.object_type? {
72        ObjectType::Struct(t) => t,
73        _ => return None,
74    };
75
76    let bcs_bytes = match data.bcs? {
77        iota_sdk::rpc_types::IotaRawData::MoveObject(move_obj) => move_obj.bcs_bytes,
78        _ => return None,
79    };
80
81    let move_struct = MoveStruct::new_from_execution_with_limit(
82        move_object_type.into(),
83        data.version,
84        bcs_bytes,
85        250 * 1024, // The limit is not important here, it is copied from the protocol config
86    )
87    .ok()?;
88    let owner = data.owner?;
89    let previous_transaction = data.previous_transaction?;
90
91    let object = Object::new_move(move_struct, owner, previous_transaction);
92
93    // We need to get the inner object to modify the storage rebate
94    let mut inner = object.into_inner();
95    inner.storage_rebate = data.storage_rebate.unwrap_or(0);
96
97    Some(inner.into())
98}
