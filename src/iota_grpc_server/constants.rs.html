4use iota_grpc_types::{google::rpc::bad_request::FieldViolation, v1::error_reason::ErrorReason};
5
6use crate::error::RpcError;
7
8/// Maximum number of nested `All`, `Any` and `Not` filters in a transaction or
9/// event filter. The event filter of a transaction filter adds one level.
10pub const MAX_FILTER_DEPTH: usize = 10;
11
12/// Maximum number of filters in a transaction or event filter, counting `All`,
13/// `Any` and `Not` themselves, every filter inside them, and every transaction
14/// kind of a `TransactionKind` filter.
15pub const MAX_FILTER_NODES: usize = 256;
16
17/// Default maximum message size for chunked responses (4MB)
18pub const DEFAULT_MAX_MESSAGE_SIZE_BYTES: usize = 4 * 1024 * 1024;
19
20/// Minimum allowed message size (1MB)
21pub const MIN_MESSAGE_SIZE_BYTES: usize = 1024 * 1024;
22
23/// Maximum allowed message size (128MB)
24pub const MAX_MESSAGE_SIZE_BYTES: usize = 128 * 1024 * 1024;
25
26/// Validates and converts the max_message_size_bytes parameter.
27///
28/// Accepts `Option<u32>` (the proto field type) and converts internally.
29pub fn validate_max_message_size(max_message_size_bytes: Option<u32>) -> Result<usize, RpcError> {
30    match max_message_size_bytes {
31        Some(size) => {
32            let size = size as usize;
33
34            match size {
35                s if s < MIN_MESSAGE_SIZE_BYTES => {
36                    Err(FieldViolation::new("max_message_size_bytes")
37                        .with_description(format!(
38                            "must be at least {MIN_MESSAGE_SIZE_BYTES} bytes"
39                        ))
40                        .with_reason(ErrorReason::FieldInvalid)
41                        .into())
42                }
43                s if s > MAX_MESSAGE_SIZE_BYTES => {
44                    Err(FieldViolation::new("max_message_size_bytes")
45                        .with_description(format!("must be at most {MAX_MESSAGE_SIZE_BYTES} bytes"))
46                        .with_reason(ErrorReason::FieldInvalid)
47                        .into())
48                }
49                s => Ok(s),
50            }
51        }
52        None => Ok(DEFAULT_MAX_MESSAGE_SIZE_BYTES),
53    }
54}
