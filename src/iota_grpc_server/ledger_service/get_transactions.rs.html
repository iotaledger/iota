31/// The most transactions of one request whose store reads share one call for each kind of data.
32/// Matches JSON-RPC's default limit of digests per request.
33const TRANSACTION_READ_CHUNK_SIZE: usize = 50;
34
35type ValidationResult = Result<(Vec<TransactionDigest>, FieldMaskTree), RpcError>;
36
37pub(crate) fn validate_get_transaction_requests(
38    requests: Vec<Option<Vec<u8>>>,
39    read_mask: Option<FieldMask>,
40) -> ValidationResult {
41    let read_mask =
42        validate_read_mask::<ExecutedTransaction>(read_mask, GET_TRANSACTIONS_READ_MASK)?;
43
44    let requests = requests
45        .into_iter()
46        .enumerate()
47        .map(|(idx, digest_bytes)| {
48            let digest_bytes = digest_bytes.ok_or_else(|| {
49                FieldViolation::new("digest")
50                    .with_reason(ErrorReason::FieldMissing)
51                    .nested_at("requests", idx)
52            })?;
53
54            TransactionDigest::from_bytes(&digest_bytes).map_err(|e| {
55                FieldViolation::new("digest")
56                    .with_description(format!("invalid digest: {e}"))
57                    .with_reason(ErrorReason::FieldInvalid)
58                    .nested_at("requests", idx)
59            })
60        })
61        .collect::<Result<Vec<_>, _>>()?;
62
63    Ok((requests, read_mask))
64}

66/// Available Read Mask Fields
67///
68/// Data the node has pruned is read from the configured key-value store, with one store call for
69/// each kind of data (checkpoint lookup, transactions and effects, checkpoint summaries, events,
70/// objects) for up to [`TRANSACTION_READ_CHUNK_SIZE`] transactions of a request at a time. The
71/// result of a transaction is:
72/// - `UNAVAILABLE` if a store call it needs fails, or once the store calls of the request have
73///   taken 30 s in total. A failed checkpoint lookup counts as a miss while that budget lasts.
74/// - `UNAVAILABLE` if the store returns the checkpoint of a pruned transaction but lacks its
75///   transaction, effects or events.
76/// - `INTERNAL` if the store returns data that does not match the effects.
77///
78/// A store that reports a failure as missing data gives the result of missing
79/// data.
80///
81/// The `get_transactions` function supports the following `read_mask` fields to
82/// control which data is included in the response:
83///
84/// ## Transaction Fields
85/// - `transaction` - includes all transaction fields
86///   - `transaction.digest` - the transaction digest
87///   - `transaction.bcs` - the full BCS-encoded transaction
88/// - `signatures` - includes all signature fields
89///   - `signatures.bcs` - the full BCS-encoded signature
90/// - `effects` - includes all effects fields
91///   - `effects.digest` - the effects digest
92///   - `effects.bcs` - the full BCS-encoded effects
93///
94/// ## Event Fields
95/// If the transaction's events are unavailable (e.g. pruned and not in the
96/// key-value store), the transaction's result is a `FAILED_PRECONDITION` error.
97/// - `events` - includes all event fields
98///   - `events.digest` - the events digest
99///   - `events.events` - includes all event fields
100///     - `events.events.bcs` - the full BCS-encoded event
101///     - `events.events.package_id` - the ID of the package that emitted the event
102///     - `events.events.module` - the module that emitted the event
103///     - `events.events.sender` - the sender that triggered the event
104///     - `events.events.event_type` - the type of the event
105///     - `events.events.bcs_contents` - the full BCS-encoded contents of the event
106///     - `events.events.json_contents` - the JSON-encoded contents of the event
107///
108/// ## Timing Fields
109/// Both are read from the ledger, so a node without the gRPC indexes serves
110/// them too.
111/// - `checkpoint` - the checkpoint that included the transaction
112/// - `timestamp` - the timestamp of the checkpoint that included the transaction; if that
113///   checkpoint's summary is unavailable (e.g. on a node restored from a snapshot),
114///   `FAILED_PRECONDITION` when a key-value store is set and the summary is not in it, and
115///   `INTERNAL` without a store
116///
117/// ## Object Fields
118/// If a required object is unavailable (e.g. pruned and not in the key-value
119/// store), the transaction's result is a `FAILED_PRECONDITION` error rather
120/// than a silently incomplete list; narrow the read mask or fetch objects
121/// individually via `get_objects` for best-effort retrieval.
122/// - `input_objects` - includes all input object fields
123///   - `input_objects.reference` - includes all reference fields
124///     - `input_objects.reference.object_id` - the ID of the input object
125///     - `input_objects.reference.version` - the version of the input object, which can be used to
126///       fetch a specific historical version or the latest version if not provided
127///     - `input_objects.reference.digest` - the digest of the input object contents, which can be
128///       used for integrity verification
129///   - `input_objects.bcs` - the full BCS-encoded object
130/// - `output_objects` - includes all output object fields
131///   - `output_objects.reference` - includes all reference fields
132///     - `output_objects.reference.object_id` - the ID of the output object
133///     - `output_objects.reference.version` - the version of the output object, which can be used
134///       to fetch a specific historical version or the latest version if not provided
135///     - `output_objects.reference.digest` - the digest of the output object contents, which can be
136///       used for integrity verification
137///   - `output_objects.bcs` - the full BCS-encoded object
138///
139/// ## Derived Change Fields
140/// Derived from the transaction's effects and input/output objects. If a
141/// required object is unavailable (e.g. pruned and not in the key-value
142/// store), the transaction's result is a `FAILED_PRECONDITION` error rather
143/// than a silently wrong answer; retry without these fields.
144/// - `balance_changes` - per-owner, per-coin-type balance deltas. For a failed transaction this
145///   contains only the gas charge.
146/// - `object_changes` - structured object changes (created, mutated, deleted, wrapped, unwrapped,
147///   published)
148#[tracing::instrument(
149    skip_all,
150    fields(
151        batch_size = requests.as_ref().map_or(0, |r| r.requests.len()),
152        ?max_message_size_bytes,
153    )
154)]
155pub(crate) fn get_transactions(
156    reader: Arc<GrpcReader>,
157    config: iota_config::node::GrpcApiConfig,
158    metrics: RequestMetrics,
159    GetTransactionsRequest {
160        requests,
161        read_mask,
162        max_message_size_bytes,
163        ..
164    }: GetTransactionsRequest,
165) -> Result<impl Stream<Item = TransactionsStreamResult> + Send, RpcError> {
166    metrics.record_requested_max_message_size(max_message_size_bytes);
167    let requests = requests
168        .map(|r| r.requests)
169        .unwrap_or_default()
170        .into_iter()
171        .map(|req| req.digest.map(|d| d.digest.to_vec()))
172        .collect();
173
174    let (digests, read_mask) = validate_get_transaction_requests(requests, read_mask)?;
175    let max_message_size = validate_max_message_size(max_message_size_bytes)?;
176    let store_budget = StoreReadBudget::default();
177    // Without a store, reading ahead would only hold results in memory.
178    let chunk_size = if reader.has_transaction_fallback() {
179        TRANSACTION_READ_CHUNK_SIZE
180    } else {
181        1
182    };
183
184    Ok(crate::create_batching_stream!(
185        stream: futures::stream::iter(digests.chunks(chunk_size))
186            .then(|digests| get_transactions_impl(&reader, &config, digests, &read_mask, &store_budget))
187            .flat_map(futures::stream::iter),
188        tx_read,
189        {
190            let tx_result = match tx_read {
191                Ok(tx) => TransactionResult::default().with_executed_transaction(tx),
192                Err(error) => TransactionResult::default().with_error(error.into_status_proto()),
193            };
194
195            let tx_size = tx_result.encoded_len();
196            (tx_result, tx_size)
197        },
198        |size| metrics.record_response_item(ResponseItemKind::Transaction, size),
199        max_message_size,
200        GetTransactionsResponse,
201        transaction_results,
202        has_next
203    ))
204}

206async fn get_transactions_impl(
207    reader: &Arc<GrpcReader>,
208    config: &iota_config::node::GrpcApiConfig,
209    digests: &[TransactionDigest],
210    read_mask: &FieldMaskTree,
211    store_budget: &StoreReadBudget,
212) -> Vec<Result<ExecutedTransaction, RpcError>> {
213    // Derive which optional fields to fetch based on the read_mask
214    let fields = TransactionReadFields::from_mask(read_mask);
215
216    // Get transaction data from storage, skipping unrequested fields
217    let tx_reads = reader
218        .get_transaction_reads(digests, &fields, store_budget)
219        .await;
220
221    tx_reads
222        .into_iter()
223        .map(|tx_read| {
224            let tx_read = tx_read?;
225            // Create a source for the merge
226            let source = TransactionReadSource {
227                reader: reader.clone(),
228                config,
229                transaction: tx_read.transaction,
230                signatures: tx_read.signatures,
231                effects: tx_read.effects,
232                events: tx_read.events,
233                checkpoint: tx_read.checkpoint,
234                timestamp_ms: tx_read.timestamp_ms,
235                input_objects: tx_read.input_objects,
236                output_objects: tx_read.output_objects,
237                mocked_coin: None,
238            };
239
240            ExecutedTransaction::merge_from(&source, read_mask)
241                .map_err(|e| e.with_context("failed to merge transaction"))
242        })
243        .collect()
244}
