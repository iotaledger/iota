4//! Implementation of the `get_checkpoint` and `stream_checkpoints` methods
5//! of the LedgerService.
6//!
7//! # Available Read Mask Fields
8//!
9//! All checkpoint query methods support the following `read_mask` fields to
10//! control which data is included in the response:
11//!
12//! ## Checkpoint Fields
13//! - `checkpoint` - includes all checkpoint fields
14//!   - `checkpoint.sequence_number` - the sequence number of the checkpoint
15//!   - `checkpoint.summary` - includes all checkpoint summary fields
16//!     - `checkpoint.summary.digest` - the digest of the checkpoint summary
17//!     - `checkpoint.summary.bcs` - the full BCS-encoded checkpoint summary
18//!   - `checkpoint.contents` - includes all checkpoint contents fields
19//!     - `checkpoint.contents.digest` - the digest of the checkpoint contents
20//!     - `checkpoint.contents.bcs` - the full BCS-encoded checkpoint contents
21//!   - `checkpoint.signature` - the validator aggregated signature for the checkpoint
22//!
23//! ## Transaction Fields
24//! - `transactions` - includes all executed transaction fields
25//!   - `transactions.transaction` - includes all transaction fields
26//!     - `transactions.transaction.digest` - the transaction digest
27//!     - `transactions.transaction.bcs` - the full BCS-encoded transaction
28//!   - `transactions.signatures` - includes all signature fields
29//!     - `transactions.signatures.bcs` - the full BCS-encoded signature
30//!   - `transactions.effects` - includes all effects fields
31//!     - `transactions.effects.digest` - the effects digest
32//!     - `transactions.effects.bcs` - the full BCS-encoded effects
33//!   - `transactions.events` - includes all event fields (all events of the transaction)
34//!     - `transactions.events.digest` - the events digest
35//!     - `transactions.events.events` - includes all event fields
36//!       - `transactions.events.events.bcs` - the full BCS-encoded event
37//!       - `transactions.events.events.package_id` - the ID of the package that emitted the event
38//!       - `transactions.events.events.module` - the module that emitted the event
39//!       - `transactions.events.events.sender` - the sender that triggered the event
40//!       - `transactions.events.events.event_type` - the type of the event
41//!       - `transactions.events.events.bcs_contents` - the full BCS-encoded contents of the event
42//!       - `transactions.events.events.json_contents` - the JSON-encoded contents of the event
43//!   - `transactions.checkpoint` - the checkpoint that included the transaction
44//!   - `transactions.timestamp` - the timestamp of the checkpoint that included the transaction
45//!   - `transactions.input_objects` - includes all input object fields
46//!     - `transactions.input_objects.reference` - includes all reference fields
47//!       - `transactions.input_objects.reference.object_id` - the ID of the input object
48//!       - `transactions.input_objects.reference.version` - the version of the input object
49//!       - `transactions.input_objects.reference.digest` - the digest of the input object contents
50//!     - `transactions.input_objects.bcs` - the full BCS-encoded object
51//!   - `transactions.output_objects` - includes all output object fields
52//!     - `transactions.output_objects.reference` - includes all reference fields
53//!       - `transactions.output_objects.reference.object_id` - the ID of the output object
54//!       - `transactions.output_objects.reference.version` - the version of the output object
55//!       - `transactions.output_objects.reference.digest` - the digest of the output object
56//!         contents
57//!     - `transactions.output_objects.bcs` - the full BCS-encoded object
58//!   - `transactions.balance_changes` - per-owner, per-coin-type balance deltas derived from the
59//!     transaction's effects and input/output objects. For a failed transaction this contains only
60//!     the gas charge.
61//!   - `transactions.object_changes` - structured object changes (created, mutated, deleted,
62//!     wrapped, unwrapped, published) derived from the transaction's effects and input/output
63//!     objects.
64//!
65//! ## Event Fields
66//! - `events` - includes all event fields (all events of all transactions in the checkpoint)
67//!   - `events.bcs` - the full BCS-encoded event
68//!   - `events.package_id` - the ID of the package that emitted the event
69//!   - `events.module` - the module that emitted the event
70//!   - `events.sender` - the sender that triggered the event
71//!   - `events.event_type` - the type of the event
72//!   - `events.bcs_contents` - the full BCS-encoded contents of the event
73//!   - `events.json_contents` - the JSON-encoded contents of the event

95/// Helper function to convert proto filters to internal filters and validate
96/// their complexity
97fn convert_and_validate_filters(
98    transactions_filter: Option<iota_grpc_types::v1::filter::TransactionFilter>,
99    events_filter: Option<iota_grpc_types::v1::filter::EventFilter>,
100) -> Result<(Option<TransactionFilter>, Option<EventFilter>), Status> {
101    // Convert proto filters to internal filters
102    let transaction_filter = transactions_filter
103        .map(TransactionFilter::try_from)
104        .transpose()
105        .map_err(|e| Status::invalid_argument(format!("invalid transaction filter: {e}")))?;
106
107    let event_filter = events_filter
108        .map(EventFilter::try_from)
109        .transpose()
110        .map_err(|e| Status::invalid_argument(format!("invalid event filter: {e}")))?;
111
112    // Validate filter complexity
113    if let Some(ref filter) = transaction_filter {
114        filter
115            .validate_complexity()
116            .map_err(Status::invalid_argument)?;
117    }
118    if let Some(ref filter) = event_filter {
119        filter
120            .validate_complexity()
121            .map_err(Status::invalid_argument)?;
122    }
123
124    Ok((transaction_filter, event_filter))
125}

127fn is_valid_checkpoint_data_path(path: &str) -> bool {
128    if path == FIELD_PATH_WILDCARD {
129        return true;
130    }
131    let (head, rest) = match path.split_once('.') {
132        Some((head, rest)) if !rest.is_empty() => (head, Some(rest)),
133        Some((head, _)) => (head, None),
134        None => (path, None),
135    };
136    let Some(rest) = rest else {
137        return matches!(head, "checkpoint" | "transactions" | "events");
138    };
139    if rest == FIELD_PATH_WILDCARD {
140        return false;
141    }
142    let rest = FieldMask::from_paths([rest]);
143    match head {
144        "checkpoint" => rest.validate::<Checkpoint>().is_ok(),
145        "transactions" => rest.validate::<ExecutedTransaction>().is_ok(),
146        "events" => rest.validate::<Event>().is_ok(),
147        _ => false,
148    }
149}

151/// Parse read_mask from request and extract component masks for checkpoint,
152/// transactions, and events.
153fn parse_checkpoint_read_mask(
154    read_mask: Option<FieldMask>,
155) -> Result<(FieldMaskTree, Option<FieldMaskTree>, Option<FieldMaskTree>), RpcError> {
156    let read_mask = read_mask.unwrap_or_else(|| FieldMask::from_str(GET_CHECKPOINT_READ_MASK));
157    if let Some(path) = read_mask
158        .paths
159        .iter()
160        .find(|path| !is_valid_checkpoint_data_path(path))
161    {
162        Err(FieldViolation::new("read_mask")
163            .with_description(format!("invalid read_mask path: {path}"))
164            .with_reason(ErrorReason::FieldInvalid))?;
165    }
166    let read_mask = FieldMaskTree::from(read_mask);
167
168    // Extract checkpoint-related fields mask
169    let checkpoint_mask = read_mask.subtree("checkpoint").unwrap_or_default();
170
171    // Extract transactions mask if requested
172    let transactions_mask = read_mask.subtree("transactions");
173
174    // Extract events mask if requested
175    let events_mask = read_mask.subtree("events");
176
177    Ok((checkpoint_mask, transactions_mask, events_mask))
178}

180/// Get checkpoint data based on the provided checkpoint ID (sequence number,
181/// digest, or latest) and read mask.
182///
183/// # Request parameters
184/// * `read_mask` - Optional field mask specifying which fields to include. If `None`, uses
185///   [`GET_CHECKPOINT_READ_MASK`] as default. See [module-level
186///   documentation](crate::ledger_service::get_checkpoint) for all available fields.
187/// * `transactions_filter` - Optional filter to apply to transactions included in the checkpoint.
188///   Only transactions matching the filter will be included in the response.
189/// * `events_filter` - Optional filter to apply to events included in the checkpoint. Only events
190///   matching the filter will be included in the response.
191/// * `max_message_size_bytes` - Optional maximum message size in bytes that the client can handle.
192///   The server will use this to limit the size of the response and avoid sending messages that are
193///   too large.
194/// * `checkpoint_id` - The identifier for the checkpoint to fetch. This can be one of:
195///   - `sequence_number` - the sequence number of the checkpoint to fetch
196///   - `digest` - the digest of the checkpoint to fetch
197///   - `latest` - if set, fetches the latest checkpoint
198pub(crate) fn get_checkpoint(
199    service: &LedgerGrpcService,
200    request: Request<grpc_ledger_service::GetCheckpointRequest>,
201) -> Result<impl Stream<Item = CheckpointStreamResult> + Send, RpcError> {
202    let metrics = RequestMetrics::from_extensions(request.extensions());
203    let req = request.into_inner();
204    metrics.record_requested_max_message_size(req.max_message_size_bytes);
205
206    // determine if we need to get the checkpoint based on the sequential number,
207    // digest or the latest one.
208    let sequence_number = match req.checkpoint_id {
209        Some(grpc_ledger_service::get_checkpoint_request::CheckpointId::SequenceNumber(seq)) => seq,
210        Some(grpc_ledger_service::get_checkpoint_request::CheckpointId::Digest(digest)) => {
211            let digest = (&digest)
212                .try_into()
213                .map_err(|e| Status::invalid_argument(format!("invalid checkpoint digest: {e}")))?;
214            service
215                .reader
216                .get_checkpoint_sequence_number_by_digest(&digest)
217                .map_err(|e| Status::internal(format!("failed to get checkpoint by digest: {e}")))?
218                .ok_or(Status::not_found("checkpoint not found"))?
219        }
220        Some(grpc_ledger_service::get_checkpoint_request::CheckpointId::Latest(_)) => service
221            .reader
222            .get_latest_checkpoint_sequence_number()
223            .map_err(|e| Status::internal(format!("failed to get latest checkpoint: {e}")))?
224            .ok_or(Status::not_found("latest checkpoint not found"))?,
225        None => {
226            return Err(Status::invalid_argument("checkpoint_id must be provided").into());
227        }
228        Some(_) => {
229            return Err(Status::invalid_argument("unknown checkpoint_id type").into());
230        }
231    };
232
233    // Check if the requested checkpoint has been pruned
234    let lowest_available = service
235        .reader
236        .get_lowest_available_checkpoint()
237        .map_err(|e| Status::internal(format!("failed to get lowest available checkpoint: {e}")))?;
238    if sequence_number < lowest_available {
239        return Err(Status::not_found(format!(
240            "Requested checkpoint {sequence_number} is below the lowest available checkpoint {lowest_available}"
241        ))
242        .into());
243    }
244
245    let client_max_message_size_bytes = req.max_message_size_bytes;
246
247    debug!(
248        "get_checkpoint called for seq={} with max_size={:?}",
249        sequence_number, client_max_message_size_bytes
250    );
251
252    let max_message_size_bytes = service
253        .config
254        .max_message_size_client_bytes(client_max_message_size_bytes);
255
256    // Parse the read_mask to determine what data to include
257    let (checkpoint_mask, transactions_mask, events_mask) =
258        parse_checkpoint_read_mask(req.read_mask)?;
259
260    debug!(
261        "Parsed read_mask: checkpoint_mask={}, transactions={}, events={}",
262        checkpoint_mask,
263        transactions_mask.is_some(),
264        events_mask.is_some()
265    );
266
267    // Convert proto filters to internal filters and validate complexity
268    let (transaction_filter, event_filter) =
269        convert_and_validate_filters(req.transactions_filter, req.events_filter)?;
270
271    if transaction_filter.is_some() && transactions_mask.is_none() {
272        return Err(Status::invalid_argument(
273            "transactions_filter requires transactions in read_mask",
274        )
275        .into());
276    }
277    if event_filter.is_some() && events_mask.is_none() {
278        return Err(Status::invalid_argument("events_filter requires events in read_mask").into());
279    }
280
281    Ok(service.reader.get_checkpoint_data(
282        sequence_number,
283        checkpoint_mask,
284        transactions_mask,
285        events_mask,
286        max_message_size_bytes,
287        transaction_filter,
288        event_filter,
289        metrics,
290    ))
291}

293/// Stream checkpoint data based on the provided start and end sequence numbers
294/// and read mask. This will continuously stream new checkpoints as they are
295/// produced until the end sequence number is reached (if provided) or the
296/// client disconnects.
297///
298/// # Request parameters
299/// * `start_sequence_number` - Optional sequence number to start streaming from. If not provided,
300///   starts from the next checkpoint produced after the request is received.
301/// * `end_sequence_number` - Optional sequence number to end streaming at. If not provided,
302///   continues streaming indefinitely until the client disconnects.
303/// * `read_mask` - Optional field mask specifying which fields to include. If `None`, uses
304///   [`GET_CHECKPOINT_READ_MASK`] as default. See [module-level
305///   documentation](crate::ledger_service::get_checkpoint) for all available fields.
306/// * `transactions_filter` - Optional filter to apply to transactions included in the streamed
307///   checkpoints. Only transactions matching the filter will be included in the response.
308/// * `events_filter` - Optional filter to apply to events included in the streamed checkpoints.
309///   Only events matching the filter will be included in the response.
310/// * `max_message_size_bytes` - Optional maximum message size in bytes that the client can handle.
311///   The server will use this to limit the size of the response and avoid sending messages that are
312///   too large.
313pub(crate) fn stream_checkpoints(
314    service: &LedgerGrpcService,
315    request: Request<grpc_ledger_service::StreamCheckpointsRequest>,
316) -> Result<impl Stream<Item = CheckpointStreamResult> + Send, RpcError> {
317    let metrics = RequestMetrics::from_extensions(request.extensions());
318    let req = request.into_inner();
319    let start_sequence_number = req.start_sequence_number;
320    let end_sequence_number = req.end_sequence_number;
321    let client_max_message_size_bytes = req.max_message_size_bytes;
322    metrics.record_requested_max_message_size(client_max_message_size_bytes);
323    let filter_checkpoints = req.filter_checkpoints.unwrap_or(false);
324    let progress_interval =
325        std::time::Duration::from_millis(req.progress_interval_ms.unwrap_or(2000).max(500) as u64);
326
327    debug!(
328        "stream_checkpoints called with start={:?}, end={:?}, max_size={:?}, filter_checkpoints={}, progress_interval={:?}",
329        start_sequence_number,
330        end_sequence_number,
331        client_max_message_size_bytes,
332        filter_checkpoints,
333        progress_interval
334    );
335
336    let max_message_size_bytes = service
337        .config
338        .max_message_size_client_bytes(client_max_message_size_bytes);
339
340    // Parse the read_mask to determine what data to include
341    let (checkpoint_mask, transactions_mask, events_mask) =
342        parse_checkpoint_read_mask(req.read_mask)?;
343
344    debug!(
345        "Parsed read_mask: checkpoint_mask={}, transactions={}, events={}",
346        checkpoint_mask,
347        transactions_mask.is_some(),
348        events_mask.is_some()
349    );
350
351    // Check if the requested checkpoint has been pruned
352    if let Some(start) = start_sequence_number {
353        let lowest_available = service
354            .reader
355            .get_lowest_available_checkpoint()
356            .map_err(|e| {
357                Status::internal(format!("failed to get lowest available checkpoint: {e}"))
358            })?;
359        if start < lowest_available {
360            return Err(Status::not_found(format!(
361                "Requested checkpoint {start} is below the lowest available checkpoint {lowest_available}"
362            ))
363            .into());
364        }
365    }
366
367    // Convert proto filters to internal filters and validate complexity
368    let (transaction_filter, event_filter) =
369        convert_and_validate_filters(req.transactions_filter, req.events_filter)?;
370
371    // Validate filter_checkpoints constraints
372    if filter_checkpoints && transaction_filter.is_none() && event_filter.is_none() {
373        return Err(Status::invalid_argument(
374            "filter_checkpoints requires at least one of transactions_filter or events_filter",
375        )
376        .into());
377    }
378
379    if transaction_filter.is_some() && transactions_mask.is_none() {
380        return Err(Status::invalid_argument(
381            "transactions_filter requires transactions in read_mask",
382        )
383        .into());
384    }
385    if event_filter.is_some() && events_mask.is_none() {
386        return Err(Status::invalid_argument("events_filter requires events in read_mask").into());
387    }
388
389    let subscription = service
390        .checkpoint_data_broadcaster
391        .subscribe()
392        .ok_or_else(|| Status::unavailable("maximum concurrent stream subscribers reached"))?;
393    let stream = Box::pin(service.reader.create_checkpoint_data_stream(
394        subscription,
395        start_sequence_number,
396        end_sequence_number,
397        checkpoint_mask,
398        transactions_mask,
399        events_mask,
400        max_message_size_bytes,
401        service.cancellation_token.clone(),
402        transaction_filter,
403        event_filter,
404        filter_checkpoints,
405        progress_interval,
406        metrics,
407    ));
408    Ok(stream)
409}
