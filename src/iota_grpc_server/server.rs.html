34/// Handle to control a running gRPC server
35pub struct GrpcServerHandle {
36    /// Handle to the task that shuts the server down
37    server_handle: tokio::task::JoinHandle<()>,
38    /// Shutdown signal sender
39    shutdown_token: CancellationToken,
40    /// Broadcaster for checkpoint data
41    pub checkpoint_data_broadcaster: GrpcCheckpointDataBroadcaster,
42    /// Actual server address (with resolved port)
43    pub address: SocketAddr,
44}
45
46impl GrpcServerHandle {
47    /// Graceful shutdown of the gRPC server. Calls still running 1 s after the
48    /// shutdown starts are closed.
49    pub async fn shutdown(self) -> Result<()> {
50        self.shutdown_token.cancel();
51        self.server_handle
52            .await
53            .map_err(|join_error| anyhow::anyhow!("server task failed: {join_error}"))
54    }
55
56    /// Get the server address (actual bound address)
57    pub fn address(&self) -> SocketAddr {
58        self.address
59    }
60
61    /// Get a reference to the checkpoint data broadcaster
62    pub fn checkpoint_data_broadcaster(&self) -> &GrpcCheckpointDataBroadcaster {
63        &self.checkpoint_data_broadcaster
64    }
65}

67/// Start a gRPC server with checkpoint and event services
68///
69/// This function creates and starts a gRPC server that hosts checkpoint-related
70/// and event streaming services. Currently includes the checkpoint streaming
71/// and event streaming services, but can be extended to host additional
72/// services in the future.
73pub async fn start_grpc_server(
74    grpc_reader: Arc<GrpcReader>,
75    executor: Option<Arc<dyn TransactionExecutor>>,
76    config: iota_config::node::GrpcApiConfig,
77    listener_limits: &ListenerLimitsConfig,
78    shutdown_token: CancellationToken,
79    chain_id: iota_types::digests::ChainIdentifier,
80    metrics: Option<GrpcServerMetrics>,
81    traffic_controller: Option<Arc<TrafficController>>,
82    client_id_source: Option<ClientIdSource>,
83) -> Result<GrpcServerHandle> {
84    // Create broadcast channels
85    let (checkpoint_data_tx, _) = broadcast::channel(config.broadcast_buffer_size as usize);
86
87    // Create broadcasters
88    let inflight_subscribers = metrics
89        .as_ref()
90        .map(|m| m.inflight_checkpoint_stream_subscribers());
91    let checkpoint_data_broadcaster = GrpcCheckpointDataBroadcaster::new(
92        checkpoint_data_tx,
93        config.max_concurrent_stream_subscribers.max(1) as usize,
94        inflight_subscribers,
95    );
96
97    // Create the gRPC services - get the cancellation token directly from
98    // server level
99    let ledger_service = LedgerGrpcService::new(
100        config.clone(),
101        grpc_reader.clone(),
102        checkpoint_data_broadcaster.clone(),
103        shutdown_token.clone(),
104        chain_id,
105    );
106
107    // Create TransactionExecutionService if executor is provided
108    let tx_service = executor.map(|executor| {
109        TransactionExecutionGrpcService::new(config.clone(), grpc_reader.clone(), executor)
110    });
111
112    // Create StateService and MovePackageService.
113    // Unlike TransactionExecutionService (conditional on executor), these are
114    // always registered: they are read-only and return a clear error
115    // ("indexes are not available") at request time if indexes are absent,
116    // which is more informative than an `Unimplemented` from an unregistered
117    // service.
118    let state_service = StateGrpcService::new(grpc_reader.clone());
119    let move_package_service = MovePackageGrpcService::new(grpc_reader);
120
121    // The listener is public, so its connection count is bounded only by this,
122    // well inside the process fd limit it shares with RocksDB and the other
123    // listeners.
124    const MAX_CONNECTIONS: NonZeroUsize = NonZeroUsize::new(4096).unwrap();
125    // A reverse proxy in front of the node shows as one address, so one
126    // address may hold a quarter of the listener; it then takes four to fill
127    // it.
128    const MAX_CONNECTIONS_PER_CLIENT_IP: NonZeroUsize = NonZeroUsize::new(1024).unwrap();
129    // Frees the connections of clients that stopped sending requests. A
130    // checkpoint subscription counts as a request for as long as it streams.
131    const MAX_CONNECTION_IDLE: Duration = Duration::from_secs(300);
132    // The smallest value RFC 9113 recommends. Subscriptions are streams too,
133    // and are also capped across all connections by
134    // `max_concurrent_stream_subscribers`.
135    const MAX_CONCURRENT_STREAMS: NonZeroU32 = NonZeroU32::new(100).unwrap();
136    // Closes a connection whose peer has gone while a subscription is still
137    // streaming to it, which the idle deadline does not.
138    const HTTP2_KEEPALIVE: Duration = Duration::from_secs(60);
139
140    // gRPC runs over HTTP/2 only, and no route of this API uses the extended
141    // CONNECT protocol.
142    let http_config = iota_http::Config::default()
143        .accept_http1(false)
144        .enable_connect_protocol(false)
145        .handshake_timeout(
146            listener_limits
147                .handshake_timeout
148                .resolve(iota_http::DEFAULT_HANDSHAKE_TIMEOUT),
149        )
150        .max_pending_connections(
151            listener_limits
152                .max_pending_connections
153                .resolve(iota_http::DEFAULT_MAX_PENDING_CONNECTIONS),
154        )
155        .max_connections(listener_limits.max_connections.resolve(MAX_CONNECTIONS))
156        .max_connections_per_client_ip(
157            listener_limits
158                .max_connections_per_client_ip
159                .resolve(MAX_CONNECTIONS_PER_CLIENT_IP),
160        )
161        .ipv6_prefix_len(
162            listener_limits
163                .ipv6_prefix_len
164                .unwrap_or(iota_http::DEFAULT_IPV6_PREFIX_LEN),
165        )
166        .max_connection_idle(
167            listener_limits
168                .max_connection_idle
169                .resolve(MAX_CONNECTION_IDLE),
170        )
171        .max_concurrent_streams(
172            listener_limits
173                .max_concurrent_streams
174                .resolve(MAX_CONCURRENT_STREAMS),
175        )
176        .http2_keepalive_interval(
177            listener_limits
178                .http2_keepalive_interval
179                .resolve(HTTP2_KEEPALIVE),
180        )
181        .http1_header_read_timeout(
182            listener_limits
183                .http1_header_read_timeout
184                .resolve(iota_http::DEFAULT_HTTP1_HEADER_READ_TIMEOUT),
185        )
186        .metrics(metrics.as_ref().map(GrpcServerMetrics::listener_metrics));
187    let mut http_builder = iota_http::Builder::new().config(http_config);
188
189    if let Some(tls_config) = config.tls_config() {
190        http_builder = http_builder
191            .tls_single_cert(tls_config.cert(), tls_config.key())
192            .map_err(|e| {
193                anyhow::anyhow!(
194                    "failed to load the TLS cert file '{}' and key file '{}': {e}",
195                    tls_config.cert(),
196                    tls_config.key()
197                )
198            })?;
199        tracing::info!("gRPC server TLS enabled");
200    }
201
202    let max_message_size = config.max_message_size_bytes() as usize;
203    let mut routes = Routes::new(
204        grpc_ledger_service::ledger_service_server::LedgerServiceServer::new(ledger_service)
205            .max_encoding_message_size(max_message_size),
206    )
207    .add_service(
208        grpc_state_service::state_service_server::StateServiceServer::new(state_service)
209            .max_encoding_message_size(max_message_size),
210    )
211    .add_service(
212        grpc_move_package_service::move_package_service_server::MovePackageServiceServer::new(
213            move_package_service,
214        )
215        .max_encoding_message_size(max_message_size),
216    );
217    if let Some(tx_service) = tx_service {
218        routes = routes.add_service(
219            grpc_tx_service::transaction_execution_service_server::TransactionExecutionServiceServer::new(tx_service)
220                .max_encoding_message_size(max_message_size),
221        );
222    }
223
224    // Order matters: the address copy before traffic control, which reads it;
225    // metrics before traffic control, to count blocked requests; `GrpcTimeout`
226    // inside both, so they see the status of a call that ends at its deadline;
227    // server-timing innermost, to time only the requests that traffic control
228    // lets through.
229    let service = tower::ServiceBuilder::new()
230        .map_request(insert_tonic_connect_info)
231        .option_layer(
232            metrics.map(|m| GrpcMetricsLayer::new(Arc::new(m), &service_methods::ALL_METHOD_PATHS)),
233        )
234        .option_layer(
235            traffic_controller
236                .map(|tc| TrafficControlLayer::new(tc, client_id_source.unwrap_or_default())),
237        )
238        .layer_fn(|service| GrpcTimeout::new(service, None))
239        .layer(ServerTimingLayer)
240        .service(routes.prepare());
241
242    let http_handle = http_builder.serve(config.address, service).map_err(|e| {
243        anyhow::anyhow!("failed to start the gRPC server on {}: {e}", config.address)
244    })?;
245    let actual_addr = *http_handle.local_addr();
246
247    tracing::info!(
248        "Starting gRPC server on {} (bound to {})",
249        config.address,
250        actual_addr
251    );
252
253    let server_handle = tokio::spawn({
254        let shutdown_token = shutdown_token.clone();
255        async move {
256            shutdown_token.cancelled().await;
257            http_handle.shutdown().await;
258            tracing::info!("gRPC server shutdown completed");
259        }
260    });
261
262    Ok(GrpcServerHandle {
263        server_handle,
264        shutdown_token,
265        checkpoint_data_broadcaster,
266        address: actual_addr,
267    })
268}
