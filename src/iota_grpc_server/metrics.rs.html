68/// Metrics for the public-facing gRPC server.
69///
70/// Tracks in-flight requests, total request counts (by method and gRPC status),
71/// request latency per RPC method, checkpoint stream subscribers, the sizes of
72/// response messages and items, the max message size that requests ask for, and
73/// the connections of the listener.
74#[derive(Clone)]
75pub struct GrpcServerMetrics {
76    inflight_requests: IntGaugeVec,
77    num_requests: IntCounterVec,
78    request_latency: HistogramVec,
79    inflight_checkpoint_stream_subscribers: IntGauge,
80    /// The peak of each kind of item, indexed by `ResponseItemKind as usize`.
81    response_item_bytes_peaks: Vec<PeakGauge>,
82    requested_max_message_bytes: Histogram,
83    response_message_bytes: HistogramVec,
84    listener: ListenerMetrics,
85}
86
87impl GrpcServerMetrics {
88    pub fn new(registry: &Registry) -> Self {
89        let response_item_bytes_peak = PeakGaugeVec::register(
90            "node_grpc_response_item_bytes_peak",
91            "The largest item built for a response over the last 2 minutes, by kind. The size \
92             is observed after the field mask is applied, for each item of each response. It \
93             includes an item that is too large for a message and so is rejected, and the \
94             item at the end of a page of a List call that goes to the next page. The kind \
95             checkpoint is the sum of all messages of one checkpoint of GetCheckpoint or \
96             StreamCheckpoints",
97            "kind",
98            module_path!(),
99            registry,
100            MetricLevel::Info,
101        );
102        let response_item_bytes_peaks = ResponseItemKind::iter()
103            .map(|kind| response_item_bytes_peak.with_label_values(&[kind.into()]))
104            .collect();
105        Self {
106            inflight_requests: register_int_gauge_vec_with_registry!(
107                "node_grpc_inflight_requests",
108                "Total in-flight node gRPC requests per method",
109                &["method"],
110                registry;
111                MetricLevel::Warn,
112            )
113            .unwrap(),
114            num_requests: register_int_counter_vec_with_registry!(
115                "node_grpc_requests",
116                "Total node gRPC requests per method and status code",
117                &["method", "status"],
118                registry;
119                MetricLevel::Warn,
120            )
121            .unwrap(),
122            request_latency: register_histogram_vec_with_registry!(
123                "node_grpc_request_latency",
124                "Latency of node gRPC requests per method in seconds",
125                &["method"],
126                LATENCY_SEC_BUCKETS.to_vec(),
127                registry;
128                MetricLevel::Warn,
129            )
130            .unwrap(),
131            inflight_checkpoint_stream_subscribers: register_int_gauge_with_registry!(
132                "node_grpc_inflight_checkpoint_stream_subscribers",
133                "Number of active subscribers to the checkpoint data broadcast",
134                registry;
135                MetricLevel::Warn,
136            )
137            .unwrap(),
138            response_item_bytes_peaks,
139            requested_max_message_bytes: register_histogram_with_registry!(
140                "node_grpc_requested_max_message_bytes",
141                "The max_message_size_bytes a request sets, as sent, before validation. A request \
142                 that leaves it out is not observed",
143                MESSAGE_SIZE_BUCKETS.to_vec(),
144                registry;
145                MetricLevel::Info
146            )
147            .expect("the gRPC server metrics register without collision"),
148            response_message_bytes: register_histogram_vec_with_registry!(
149                "node_grpc_response_message_bytes",
150                "Size of each message of a response per method, in bytes: one message for a \
151                 method that returns one, each message of a stream. A compressed message and a \
152                 response that the tonic encoder rejects for its size are not observed",
153                &["method"],
154                RESPONSE_MESSAGE_SIZE_BUCKETS.to_vec(),
155                registry;
156                MetricLevel::Info
157            )
158            .expect("the gRPC server metrics register without collision"),
159            listener: ListenerMetrics::new("node_grpc", registry, MetricLevel::Warn),
160        }
161    }

175/// Tower [`Layer`] that adds gRPC request metrics to a service.
176///
177/// Only records per-method metrics for paths that exactly match a known gRPC
178/// method. All other requests (e.g. non-gRPC HTTP traffic that reaches
179/// the port) are aggregated under a single `"SPAM"` label to prevent
180/// unbounded cardinality.
181///
182/// It puts the request metrics in the request extensions, for the handlers,
183/// and observes the size of each message of each response.
184#[derive(Clone)]
185pub struct GrpcMetricsLayer {
186    metrics: Arc<GrpcServerMetrics>,
187    /// Exact set of known gRPC method paths (e.g.
188    /// `"/iota.grpc.v1.ledger_service.LedgerService/GetCheckpoint"`).
189    /// Only paths in this set get their own metric label; everything else
190    /// is labelled `"SPAM"`.
191    known_methods: Arc<HashSet<&'static str>>,
192}
193
194impl GrpcMetricsLayer {
195    pub fn new(metrics: Arc<GrpcServerMetrics>, method_paths: &[&'static str]) -> Self {
196        Self {
197            metrics,
198            known_methods: Arc::new(method_paths.iter().copied().collect()),
199        }
200    }
201}
202
203impl<S> Layer<S> for GrpcMetricsLayer {
204    type Service = GrpcMetricsService<S>;
205
206    fn layer(&self, inner: S) -> Self::Service {
207        GrpcMetricsService {
208            inner,
209            metrics: self.metrics.clone(),
210            known_methods: self.known_methods.clone(),
211        }
212    }
213}

215/// Tower [`Service`] wrapper that records gRPC request metrics.
216#[derive(Clone)]
217pub struct GrpcMetricsService<S> {
218    inner: S,
219    metrics: Arc<GrpcServerMetrics>,
220    known_methods: Arc<HashSet<&'static str>>,
221}
222
223impl<S, ReqBody> Service<http::Request<ReqBody>> for GrpcMetricsService<S>
224where
225    S: Service<http::Request<ReqBody>, Response = http::Response<tonic::body::Body>>
226        + Clone
227        + Send
228        + 'static,
229    S::Future: Send + 'static,
230    ReqBody: Send + 'static,
231{
232    type Response = S::Response;
233    type Error = S::Error;
234    type Future = GrpcMetricsFuture<S::Future, S::Response>;
235
236    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
237        self.inner.poll_ready(cx)
238    }
239
240    fn call(&mut self, req: http::Request<ReqBody>) -> Self::Future {
241        let raw_path = req.uri().path();
242
243        if !self.known_methods.contains(raw_path) {
244            // SPAM: bump counter and reject immediately without calling the
245            // inner service, avoiding unnecessary router work.
246            self.metrics
247                .num_requests
248                .with_label_values(&[SPAM_LABEL, "Unimplemented"])
249                .inc();
250
251            let response = Status::unimplemented("").into_http();
252
253            return GrpcMetricsFuture::Rejected {
254                response: Some(response),
255            };
256        }
257
258        let method = raw_path.to_owned();
259        let metrics = self.metrics.clone();
260
261        metrics
262            .inflight_requests
263            .with_label_values(&[&method])
264            .inc();
265
266        let req = RequestMetrics::attach(req, &metrics);
267        let response_message_bytes = metrics.response_message_bytes.with_label_values(&[&method]);
268
269        let guard = InFlightGuard {
270            metrics,
271            method,
272            start: Instant::now(),
273            completed: false,
274        };
275
276        let future = self.inner.call(req);
277
278        GrpcMetricsFuture::Inner {
279            inner: future,
280            guard,
281            response_message_bytes,
282        }
283    }
284}

286/// RAII guard that tracks in-flight requests and records metrics on drop.
287///
288/// When a request completes normally, [`GrpcMetricsFuture::poll`] records the
289/// response status and marks the guard as completed. If the future is dropped
290/// before completion (e.g. client disconnect), the guard records a `"canceled"`
291/// status instead, matching the REST metrics behavior.
292struct InFlightGuard {
293    metrics: Arc<GrpcServerMetrics>,
294    method: String,
295    start: Instant,
296    completed: bool,
297}
298
299impl Drop for InFlightGuard {
300    fn drop(&mut self) {
301        self.metrics
302            .inflight_requests
303            .with_label_values(&[&self.method])
304            .dec();
305
306        let latency = self.start.elapsed().as_secs_f64();
307        self.metrics
308            .request_latency
309            .with_label_values(&[&self.method])
310            .observe(latency);
311
312        if !self.completed {
313            self.metrics
314                .num_requests
315                .with_label_values(&[self.method.as_str(), "canceled"])
316                .inc();
317        }
318    }
319}

321pin_project! {
322    /// Future returned by [`GrpcMetricsService`].
323    ///
324    /// - `Inner`: a real request forwarded to the inner service. Records the
325    ///   gRPC status from the response headers on completion. If dropped before
326    ///   completion (client disconnect), the [`InFlightGuard`] records a
327    ///   `"canceled"` status.
328    ///   `response_message_bytes` observes the size of each message of the
329    ///   response.
330    /// - `Rejected`: a SPAM request that was rejected immediately. Returns the
331    ///   pre-built response on first poll.
332    #[project = GrpcMetricsFutureProj]
333    pub enum GrpcMetricsFuture<F, Res> {
334        Inner {
335            #[pin]
336            inner: F,
337            guard: InFlightGuard,
338            response_message_bytes: Histogram,
339        },
340        Rejected {
341            response: Option<Res>,
342        },
343    }
344}
345
346impl<F, E> Future for GrpcMetricsFuture<F, http::Response<tonic::body::Body>>
347where
348    F: Future<Output = Result<http::Response<tonic::body::Body>, E>>,
349{
350    type Output = F::Output;
351
352    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
353        match self.project() {
354            GrpcMetricsFutureProj::Inner {
355                inner,
356                guard,
357                response_message_bytes,
358            } => match inner.poll(cx) {
359                Poll::Ready(result) => {
360                    let status = match &result {
361                        Ok(response) => grpc_status_from_response(response),
362                        Err(_) => "transport_error",
363                    };
364
365                    guard
366                        .metrics
367                        .num_requests
368                        .with_label_values(&[guard.method.as_str(), status])
369                        .inc();
370
371                    guard.completed = true;
372
373                    Poll::Ready(result.map(|response| {
374                        response_message_sizes::record_sizes(
375                            response,
376                            response_message_bytes.clone(),
377                        )
378                    }))
379                }
380                Poll::Pending => Poll::Pending,
381            },
382            GrpcMetricsFutureProj::Rejected { response } => {
383                Poll::Ready(Ok(response.take().expect("polled after completion")))
384            }
385        }
386    }
387}

389/// Extract the gRPC status code from response headers.
390///
391/// Uses [`tonic::Status::from_header_map`] to parse the `grpc-status` header.
392/// If not present, the response is assumed to be OK. For streaming RPCs, errors
393/// sent via trailers are not visible here and will be reported as OK.
394fn grpc_status_from_response<B>(response: &http::Response<B>) -> &'static str {
395    let code = Status::from_header_map(response.headers()).map_or(Code::Ok, |s| s.code());
396    grpc_code_to_str(code)
397}
398
399pub fn grpc_code_to_str(code: Code) -> &'static str {
400    match code {
401        Code::Ok => "Ok",
402        Code::Cancelled => "Cancelled",
403        Code::Unknown => "Unknown",
404        Code::InvalidArgument => "InvalidArgument",
405        Code::DeadlineExceeded => "DeadlineExceeded",
406        Code::NotFound => "NotFound",
407        Code::AlreadyExists => "AlreadyExists",
408        Code::PermissionDenied => "PermissionDenied",
409        Code::ResourceExhausted => "ResourceExhausted",
410        Code::FailedPrecondition => "FailedPrecondition",
411        Code::Aborted => "Aborted",
412        Code::OutOfRange => "OutOfRange",
413        Code::Unimplemented => "Unimplemented",
414        Code::Internal => "Internal",
415        Code::Unavailable => "Unavailable",
416        Code::DataLoss => "DataLoss",
417        Code::Unauthenticated => "Unauthenticated",
418    }
419}

421/// The metrics a handler can reach: the ones of its request, or none.
422#[derive(Clone, Default)]
423pub(crate) struct RequestMetrics(Option<Arc<GrpcServerMetrics>>);
424
425impl RequestMetrics {
426    /// The metrics that the metrics layer put in the request extensions.
427    pub(crate) fn from_extensions(extensions: &http::Extensions) -> Self {
428        extensions.get::<Self>().cloned().unwrap_or_default()
429    }
430
431    /// Puts the metrics of a request in the extensions of `request`.
432    fn attach<B>(
433        mut request: http::Request<B>,
434        metrics: &Arc<GrpcServerMetrics>,
435    ) -> http::Request<B> {
436        request.extensions_mut().insert(Self(Some(metrics.clone())));
437        request
438    }
439
440    /// Records the size of an item built for a response.
441    pub(crate) fn record_response_item(&self, kind: ResponseItemKind, size: usize) {
442        if let Some(metrics) = &self.0 {
443            metrics.response_item_bytes_peaks[kind as usize].observe(size as u64);
444        }
445    }
446
447    /// Whether the request has metrics.
448    pub(crate) fn is_enabled(&self) -> bool {
449        self.0.is_some()
450    }
451
452    /// Records the `max_message_size_bytes` a request sets.
453    pub(crate) fn record_requested_max_message_size(&self, requested: Option<u32>) {
454        if let (Some(metrics), Some(requested)) = (&self.0, requested) {
455            metrics
456                .requested_max_message_bytes
457                .observe(f64::from(requested));
458        }
459    }
