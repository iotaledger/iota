67/// Metrics for the public-facing gRPC server.
68///
69/// Tracks in-flight requests, total request counts (by method and gRPC status),
70/// request latency per RPC method, the sizes of response messages and items,
71/// and the max message size that requests ask for.
72#[derive(Clone)]
73pub struct GrpcServerMetrics {
74    inflight_requests: IntGaugeVec,
75    num_requests: IntCounterVec,
76    request_latency: HistogramVec,
77    inflight_checkpoint_stream_subscribers: IntGauge,
78    /// The peak of each kind of item, indexed by `ResponseItemKind as usize`.
79    response_item_bytes_peaks: Vec<PeakGauge>,
80    requested_max_message_bytes: Histogram,
81    response_message_bytes: HistogramVec,
82}
83
84impl GrpcServerMetrics {
85    pub fn new(registry: &Registry) -> Self {
86        let response_item_bytes_peak = PeakGaugeVec::register(
87            "node_grpc_response_item_bytes_peak",
88            "The largest item built for a response over the last 2 minutes, by kind. The size \
89             is observed after the field mask is applied, for each item of each response. It \
90             includes an item that is too large for a message and so is rejected, and the \
91             item at the end of a page of a List call that goes to the next page. The kind \
92             checkpoint is the sum of all messages of one checkpoint of GetCheckpoint or \
93             StreamCheckpoints",
94            "kind",
95            module_path!(),
96            registry,
97            MetricLevel::Info,
98        );
99        let response_item_bytes_peaks = ResponseItemKind::iter()
100            .map(|kind| response_item_bytes_peak.with_label_values(&[kind.into()]))
101            .collect();
102        Self {
103            inflight_requests: register_int_gauge_vec_with_registry!(
104                "node_grpc_inflight_requests",
105                "Total in-flight node gRPC requests per method",
106                &["method"],
107                registry,
108            )
109            .unwrap(),
110            num_requests: register_int_counter_vec_with_registry!(
111                "node_grpc_requests",
112                "Total node gRPC requests per method and status code",
113                &["method", "status"],
114                registry,
115            )
116            .unwrap(),
117            request_latency: register_histogram_vec_with_registry!(
118                "node_grpc_request_latency",
119                "Latency of node gRPC requests per method in seconds",
120                &["method"],
121                LATENCY_SEC_BUCKETS.to_vec(),
122                registry,
123            )
124            .unwrap(),
125            inflight_checkpoint_stream_subscribers: register_int_gauge_with_registry!(
126                "node_grpc_inflight_checkpoint_stream_subscribers",
127                "Number of active subscribers to the checkpoint data broadcast",
128                registry,
129            )
130            .unwrap(),
131            response_item_bytes_peaks,
132            requested_max_message_bytes: register_histogram_with_registry!(
133                "node_grpc_requested_max_message_bytes",
134                "The max_message_size_bytes a request sets, as sent, before validation. A request \
135                 that leaves it out is not observed",
136                MESSAGE_SIZE_BUCKETS.to_vec(),
137                registry;
138                MetricLevel::Info
139            )
140            .expect("the gRPC server metrics register without collision"),
141            response_message_bytes: register_histogram_vec_with_registry!(
142                "node_grpc_response_message_bytes",
143                "Size of each message of a response per method, in bytes: one message for a \
144                 method that returns one, each message of a stream. A compressed message and a \
145                 response that the tonic encoder rejects for its size are not observed",
146                &["method"],
147                RESPONSE_MESSAGE_SIZE_BUCKETS.to_vec(),
148                registry;
149                MetricLevel::Info
150            )
151            .expect("the gRPC server metrics register without collision"),
152        }
153    }

157    pub fn inflight_checkpoint_stream_subscribers(&self) -> IntGauge {

162/// Tower [`Layer`] that adds gRPC request metrics to a service.
163///
164/// Only records per-method metrics for paths that exactly match a known gRPC
165/// method. All other requests (e.g. non-gRPC HTTP traffic that reaches
166/// the port) are aggregated under a single `"SPAM"` label to prevent
167/// unbounded cardinality.
168///
169/// It puts the request metrics in the request extensions, for the handlers,
170/// and observes the size of each message of each response.
171#[derive(Clone)]
172pub struct GrpcMetricsLayer {
173    metrics: Arc<GrpcServerMetrics>,
174    /// Exact set of known gRPC method paths (e.g.
175    /// `"/iota.grpc.v1.ledger_service.LedgerService/GetCheckpoint"`).
176    /// Only paths in this set get their own metric label; everything else
177    /// is labelled `"SPAM"`.
178    known_methods: Arc<HashSet<&'static str>>,
179}
180
181impl GrpcMetricsLayer {
182    pub fn new(metrics: Arc<GrpcServerMetrics>, method_paths: &[&'static str]) -> Self {
183        Self {
184            metrics,
185            known_methods: Arc::new(method_paths.iter().copied().collect()),
186        }
187    }
188}

193    fn layer(&self, inner: S) -> Self::Service {
194        GrpcMetricsService {
195            inner,
196            metrics: self.metrics.clone(),
197            known_methods: self.known_methods.clone(),

199    }

203#[derive(Clone)]

210impl<S, ReqBody> Service<http::Request<ReqBody>> for GrpcMetricsService<S>
211where
212    S: Service<http::Request<ReqBody>, Response = http::Response<tonic::body::Body>>
213        + Clone
214        + Send
215        + 'static,
216    S::Future: Send + 'static,
217    ReqBody: Send + 'static,
218{
219    type Response = S::Response;
220    type Error = S::Error;
221    type Future = GrpcMetricsFuture<S::Future, S::Response>;
222
223    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
224        self.inner.poll_ready(cx)
225    }
226
227    fn call(&mut self, req: http::Request<ReqBody>) -> Self::Future {
228        let raw_path = req.uri().path();
229
230        if !self.known_methods.contains(raw_path) {
231            // SPAM: bump counter and reject immediately without calling the
232            // inner service, avoiding unnecessary router work.
233            self.metrics
234                .num_requests
235                .with_label_values(&[SPAM_LABEL, "Unimplemented"])
236                .inc();
237
238            let response = Status::unimplemented("").into_http();
239
240            return GrpcMetricsFuture::Rejected {
241                response: Some(response),
242            };
243        }
244
245        let method = raw_path.to_owned();
246        let metrics = self.metrics.clone();
247
248        metrics
249            .inflight_requests
250            .with_label_values(&[&method])
251            .inc();
252
253        let req = RequestMetrics::attach(req, &metrics);
254        let response_message_bytes = metrics.response_message_bytes.with_label_values(&[&method]);
255
256        let guard = InFlightGuard {
257            metrics,
258            method,
259            start: Instant::now(),
260            completed: false,
261        };
262
263        let future = self.inner.call(req);
264
265        GrpcMetricsFuture::Inner {
266            inner: future,
267            guard,
268            response_message_bytes,
269        }
270    }
271}

273/// RAII guard that tracks in-flight requests and records metrics on drop.
274///
275/// When a request completes normally, [`GrpcMetricsFuture::poll`] records the
276/// response status and marks the guard as completed. If the future is dropped
277/// before completion (e.g. client disconnect), the guard records a `"canceled"`
278/// status instead, matching the REST metrics behavior.
279struct InFlightGuard {
280    metrics: Arc<GrpcServerMetrics>,
281    method: String,
282    start: Instant,
283    completed: bool,
284}
285
286impl Drop for InFlightGuard {
287    fn drop(&mut self) {
288        self.metrics
289            .inflight_requests
290            .with_label_values(&[&self.method])
291            .dec();
292
293        let latency = self.start.elapsed().as_secs_f64();
294        self.metrics
295            .request_latency
296            .with_label_values(&[&self.method])
297            .observe(latency);
298
299        if !self.completed {
300            self.metrics
301                .num_requests
302                .with_label_values(&[self.method.as_str(), "canceled"])
303                .inc();
304        }
305    }
306}

308pin_project! {
309    /// Future returned by [`GrpcMetricsService`].
310    ///
311    /// - `Inner`: a real request forwarded to the inner service. Records the
312    ///   gRPC status from the response headers on completion. If dropped before
313    ///   completion (client disconnect), the [`InFlightGuard`] records a
314    ///   `"canceled"` status.
315    ///   `response_message_bytes` observes the size of each message of the
316    ///   response.
317    /// - `Rejected`: a SPAM request that was rejected immediately. Returns the
318    ///   pre-built response on first poll.
319    #[project = GrpcMetricsFutureProj]
320    pub enum GrpcMetricsFuture<F, Res> {
321        Inner {
322            #[pin]
323            inner: F,
324            guard: InFlightGuard,
325            response_message_bytes: Histogram,
326        },
327        Rejected {
328            response: Option<Res>,
329        },
330    }
331}
332
333impl<F, E> Future for GrpcMetricsFuture<F, http::Response<tonic::body::Body>>
334where
335    F: Future<Output = Result<http::Response<tonic::body::Body>, E>>,
336{
337    type Output = F::Output;
338
339    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
340        match self.project() {
341            GrpcMetricsFutureProj::Inner {
342                inner,
343                guard,
344                response_message_bytes,
345            } => match inner.poll(cx) {
346                Poll::Ready(result) => {
347                    let status = match &result {
348                        Ok(response) => grpc_status_from_response(response),
349                        Err(_) => "transport_error",
350                    };
351
352                    guard
353                        .metrics
354                        .num_requests
355                        .with_label_values(&[guard.method.as_str(), status])
356                        .inc();
357
358                    guard.completed = true;
359
360                    Poll::Ready(result.map(|response| {
361                        response_message_sizes::record_sizes(
362                            response,
363                            response_message_bytes.clone(),
364                        )
365                    }))
366                }
367                Poll::Pending => Poll::Pending,
368            },
369            GrpcMetricsFutureProj::Rejected { response } => {
370                Poll::Ready(Ok(response.take().expect("polled after completion")))
371            }
372        }
373    }
374}

376/// Extract the gRPC status code from response headers.
377///
378/// Uses [`tonic::Status::from_header_map`] to parse the `grpc-status` header.
379/// If not present, the response is assumed to be OK. For streaming RPCs, errors
380/// sent via trailers are not visible here and will be reported as OK.
381fn grpc_status_from_response<B>(response: &http::Response<B>) -> &'static str {
382    let code = Status::from_header_map(response.headers()).map_or(Code::Ok, |s| s.code());
383    grpc_code_to_str(code)
384}
385
386pub fn grpc_code_to_str(code: Code) -> &'static str {
387    match code {
388        Code::Ok => "Ok",
389        Code::Cancelled => "Cancelled",
390        Code::Unknown => "Unknown",
391        Code::InvalidArgument => "InvalidArgument",
392        Code::DeadlineExceeded => "DeadlineExceeded",
393        Code::NotFound => "NotFound",
394        Code::AlreadyExists => "AlreadyExists",
395        Code::PermissionDenied => "PermissionDenied",
396        Code::ResourceExhausted => "ResourceExhausted",
397        Code::FailedPrecondition => "FailedPrecondition",
398        Code::Aborted => "Aborted",
399        Code::OutOfRange => "OutOfRange",
400        Code::Unimplemented => "Unimplemented",
401        Code::Internal => "Internal",
402        Code::Unavailable => "Unavailable",
403        Code::DataLoss => "DataLoss",
404        Code::Unauthenticated => "Unauthenticated",
405    }
406}

408/// The metrics a handler can reach: the ones of its request, or none.
409#[derive(Clone, Default)]
410pub(crate) struct RequestMetrics(Option<Arc<GrpcServerMetrics>>);
411
412impl RequestMetrics {
413    /// The metrics that the metrics layer put in the request extensions.
414    pub(crate) fn from_extensions(extensions: &http::Extensions) -> Self {
415        extensions.get::<Self>().cloned().unwrap_or_default()
416    }
417
418    /// Puts the metrics of a request in the extensions of `request`.
419    fn attach<B>(
420        mut request: http::Request<B>,
421        metrics: &Arc<GrpcServerMetrics>,
422    ) -> http::Request<B> {
423        request.extensions_mut().insert(Self(Some(metrics.clone())));
424        request
425    }
426
427    /// Records the size of an item built for a response.
428    pub(crate) fn record_response_item(&self, kind: ResponseItemKind, size: usize) {
429        if let Some(metrics) = &self.0 {
430            metrics.response_item_bytes_peaks[kind as usize].observe(size as u64);
431        }
432    }
433
434    /// Whether the request has metrics.
435    pub(crate) fn is_enabled(&self) -> bool {
436        self.0.is_some()
437    }
438
439    /// Records the `max_message_size_bytes` a request sets.
440    pub(crate) fn record_requested_max_message_size(&self, requested: Option<u32>) {
441        if let (Some(metrics), Some(requested)) = (&self.0, requested) {
442            metrics
443                .requested_max_message_bytes
444                .observe(f64::from(requested));
445        }
446    }
