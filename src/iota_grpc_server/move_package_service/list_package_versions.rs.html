4use std::sync::Arc;
5
6use iota_grpc_types::v1::move_package_service::{
7    ListPackageVersionsRequest, ListPackageVersionsResponse, PackageVersion,
8};
9use iota_sdk_types::ObjectId;
10use iota_types::move_package::MovePackageExt;
11use prost::Message;
12use serde::{Deserialize, Serialize};
13
14use crate::{
15    constants::validate_max_message_size,
16    error::RpcError,
17    types::GrpcReader,
18    validation::{
19        decode_page_token, encode_page_token, object_id_proto, page_token_mismatch,
20        require_object_id, validate_page_size,
21    },
22};
23
24const DEFAULT_PAGE_SIZE: u32 = 1000;
25const MAX_PAGE_SIZE: u32 = 10000;
26
27#[derive(Serialize, Deserialize)]
28struct PageToken {
29    original_package_id: ObjectId,
30    last_version: u64,
31}
32
33#[tracing::instrument(skip(reader))]
34pub(crate) fn list_package_versions(
35    reader: Arc<GrpcReader>,
36    ListPackageVersionsRequest {
37        package_id,
38        page_size,
39        page_token,
40        max_message_size_bytes,
41        ..
42    }: ListPackageVersionsRequest,
43) -> Result<ListPackageVersionsResponse, RpcError> {
44    let pkg_id = require_object_id(&package_id, "package_id")?;
45    let page_size = validate_page_size(page_size, DEFAULT_PAGE_SIZE, MAX_PAGE_SIZE);
46    let max_message_size = validate_max_message_size(max_message_size_bytes)?;
47
48    // Resolve the original package ID so we can list all versions across
49    // different storage IDs (relevant for upgraded user packages).
50    let original_package_id = match reader.get_object(&pkg_id)? {
51        Some(current_object) => {
52            if !current_object.is_package() {
53                return Err(RpcError::new(
54                    tonic::Code::InvalidArgument,
55                    format!("Object {pkg_id} is not a package"),
56                ));
57            }
58
59            current_object
60                .data
61                .as_opt_package()
62                .ok_or_else(|| {
63                    RpcError::new(
64                        tonic::Code::Internal,
65                        format!("Object {pkg_id} passed is_package() but try_as_package() failed"),
66                    )
67                })?
68                .original_package_id()
69        }
70        None => {
71            // The object may have been pruned from the object store. Fall back
72            // to treating the requested ID as the original package ID and check
73            // whether the version index has any entries for it.
74            pkg_id
75        }
76    };
77
78    let page_token: Option<PageToken> = decode_page_token(&page_token)?;
79    if let Some(ref t) = page_token {
80        if t.original_package_id != original_package_id {
81            return Err(page_token_mismatch());
82        }
83    }
84
85    let cursor_version = page_token.map(|t| t.last_version);
86
87    let mut iter = reader.package_versions_iter(original_package_id, cursor_version)?;
88
89    let mut versions = Vec::with_capacity(page_size);
90    let mut size_bytes = 0usize;
91    let mut last_version: Option<u64> = None;
92
93    for result in iter.by_ref() {
94        let (key, info) = result.map_err(RpcError::from)?;
95
96        let version = PackageVersion::default()
97            .with_original_id(object_id_proto(&key.original_package_id))
98            .with_storage_id(object_id_proto(&info.storage_id))
99            .with_version(key.version);
100
101        let item_size = version.encoded_len();
102
103        if !versions.is_empty() && size_bytes + item_size > max_message_size {
104            let response = ListPackageVersionsResponse::default()
105                .with_versions(versions)
106                .with_next_page_token(encode_page_token(&PageToken {
107                    original_package_id,
108                    last_version: last_version.expect("versions is non-empty"),
109                }));
110            return Ok(response);
111        }
112
113        last_version = Some(key.version);
114        versions.push(version);
115        size_bytes += item_size;
116
117        if versions.len() >= page_size {
118            break;
119        }
120    }
121
122    if versions.is_empty() && cursor_version.is_none() {
123        return Err(RpcError::from(crate::error::ObjectNotFoundError::new(
124            pkg_id,
125        )));
126    }
127
128    // Check if there are more items.
129    let has_more = iter.next().transpose().map_err(RpcError::from)?.is_some();
130
131    let mut response = ListPackageVersionsResponse::default().with_versions(versions);
132    if has_more {
133        if let Some(ver) = last_version {
134            response = response.with_next_page_token(encode_page_token(&PageToken {
135                original_package_id,
136                last_version: ver,
137            }));
138        }
139    }
140
141    Ok(response)
142}
