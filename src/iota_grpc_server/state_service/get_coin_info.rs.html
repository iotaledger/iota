4use std::sync::Arc;

6use iota_grpc_types::{
7    google::rpc::bad_request::FieldViolation,
8    v1::{
9        coin::{
10            CoinMetadata, CoinTreasury, RegulatedCoinMetadata, coin_treasury::SupplyState,
11            regulated_coin_metadata::CoinRegulatedState,
12        },
13        error_reason::ErrorReason,
14        state_service::{GetCoinInfoRequest, GetCoinInfoResponse},
15    },
16};
17use iota_sdk_types::{Address, Owner};
18
19use crate::{error::RpcError, types::GrpcReader, validation::object_id_proto};

21/// Get coin info for a given coin type.
22///
23/// This endpoint does not use the Merge trait / read_mask pattern because the
24/// proto `GetCoinInfoRequest` has no `read_mask` field — all response fields
25/// are always returned. The response is small and fixed-shape, so field
26/// filtering would add complexity without meaningful savings.
27#[tracing::instrument(skip(reader))]
28pub(crate) fn get_coin_info(
29    reader: Arc<GrpcReader>,
30    GetCoinInfoRequest { coin_type, .. }: GetCoinInfoRequest,
31) -> Result<GetCoinInfoResponse, RpcError> {
32    // Validate coin_type
33    let coin_type_str = coin_type.as_deref().ok_or_else(|| {
34        FieldViolation::new("coin_type")
35            .with_description("coin_type is required")
36            .with_reason(ErrorReason::FieldMissing)
37    })?;
38
39    let core_coin_type = iota_types::parse_iota_struct_tag(coin_type_str).map_err(|e| {
40        FieldViolation::new("coin_type")
41            .with_description(format!("invalid coin_type: {e}"))
42            .with_reason(ErrorReason::FieldInvalid)
43    })?;
44
45    let coin_info = reader.get_coin_info(&core_coin_type)?;
46    let coin_info = coin_info.ok_or_else(|| {
47        RpcError::new(
48            tonic::Code::NotFound,
49            format!(
50                "No coin info found for type {coin_type_str} \
51                 — CoinMetadata or TreasuryCap may not be published"
52            ),
53        )
54    })?;
55
56    let mut response = GetCoinInfoResponse::default().with_coin_type(coin_type_str.to_string());
57
58    // Populate metadata if available
59    if let Some(coin_metadata_object_id) = coin_info.coin_metadata_object_id {
60        if let Some(object) = reader.get_object(&coin_metadata_object_id)? {
61            match iota_types::coin::CoinMetadata::try_from(object) {
62                Ok(metadata) => {
63                    let mut grpc_metadata = CoinMetadata::default()
64                        .with_id(object_id_proto(&metadata.id.id.bytes))
65                        .with_decimals(metadata.decimals as u32)
66                        .with_name(metadata.name)
67                        .with_symbol(metadata.symbol)
68                        .with_description(metadata.description);
69                    if let Some(icon_url) = metadata.icon_url {
70                        grpc_metadata = grpc_metadata.with_icon_url(icon_url);
71                    }
72                    response.metadata = Some(grpc_metadata);
73                }
74                Err(e) => {
75                    tracing::error!(
76                        "Unable to read object {coin_metadata_object_id} as CoinMetadata \
77                         for coin type {coin_type_str}: {e}"
78                    );
79                }
80            }
81        }
82    }
83
84    // Populate treasury if available
85    if let Some(treasury_object_id) = coin_info.treasury_object_id {
86        if let Some(object) = reader.get_object(&treasury_object_id)? {
87            // Determine supply_state from ownership before consuming the object.
88            // Immutable or owned by 0x0 means the TreasuryCap can never be used
89            // to mint, so supply is fixed.
90            let supply_state = match &object.owner {
91                Owner::Immutable => SupplyState::Fixed,
92                Owner::Address(addr) if *addr == Address::ZERO => SupplyState::Fixed,
93                _ => SupplyState::Unknown,
94            };
95
96            match iota_types::coin::TreasuryCap::try_from(object) {
97                Ok(treasury) => {
98                    response.treasury = Some(
99                        CoinTreasury::default()
100                            .with_id(object_id_proto(&treasury.id.id.bytes))
101                            .with_total_supply(treasury.total_supply.value)
102                            .with_supply_state(supply_state),
103                    );
104                }
105                Err(e) => {
106                    tracing::error!(
107                        "Unable to read object {treasury_object_id} as TreasuryCap \
108                         for coin type {coin_type_str}: {e}"
109                    );
110                }
111            }
112        }
113    } else if core_coin_type.is_gas() {
114        // Special case for native GAS coin: get treasury from system state.
115        // The native gas coin's total supply is fixed (no TreasuryCap can mint more).
116        let summary = reader.get_system_state_summary()?;
117        response.treasury = Some(
118            CoinTreasury::default()
119                .with_id(object_id_proto(&summary.iota_treasury_cap_id()))
120                .with_total_supply(summary.iota_total_supply())
121                .with_supply_state(SupplyState::Fixed),
122        );
123    }
124
125    // Populate regulated metadata.
126    //
127    // NOTE: Unlike `CoinMetadata` and `TreasuryCap` above which use the
128    // type-safe `TryFrom<Object>` conversion, `RegulatedCoinMetadata` is
129    // deserialized via raw `bcs::from_bytes`. This is because
130    // `iota_types::deny_list_v1::RegulatedCoinMetadata` does not implement
131    // `TryFrom<Object>`. If that impl is added upstream, this should be
132    // updated to use it for consistency and better error handling.
133    if let Some(regulated_object_id) = coin_info.regulated_coin_metadata_object_id {
134        if let Some(object) = reader.get_object(&regulated_object_id)? {
135            if let Some(move_obj) = object.data.as_opt_struct() {
136                match bcs::from_bytes::<iota_types::deny_list_v1::RegulatedCoinMetadata>(
137                    move_obj.contents(),
138                ) {
139                    Ok(regulated) => {
140                        // Proto fields `allow_global_pause` and `variant` are not populated
141                        // because the on-chain `RegulatedCoinMetadata` struct does not store
142                        // these values — it only contains `id`, `coin_metadata_object`, and
143                        // `deny_cap_object`.
144                        response.regulated_metadata = Some(
145                            RegulatedCoinMetadata::default()
146                                .with_id(object_id_proto(&regulated.id.id.bytes))
147                                .with_coin_metadata_object(object_id_proto(
148                                    &regulated.coin_metadata_object.bytes,
149                                ))
150                                .with_deny_cap_object(object_id_proto(
151                                    &regulated.deny_cap_object.bytes,
152                                ))
153                                .with_coin_regulated_state(CoinRegulatedState::Regulated),
154                        );
155                    }
156                    Err(e) => {
157                        tracing::error!(
158                            "Unable to read object {regulated_object_id} as \
159                             RegulatedCoinMetadata for coin type {coin_type_str}: {e}"
160                        );
161                    }
162                }
163            } else {
164                tracing::error!("Object {regulated_object_id} is not a Move object");
165            }
166        }
167    } else {
168        // No RegulatedCoinMetadata object exists for this coin type — explicitly
169        // mark as unregulated so clients can distinguish "not regulated" from
170        // "field not populated".
171        response.regulated_metadata = Some(
172            RegulatedCoinMetadata::default()
173                .with_coin_regulated_state(CoinRegulatedState::Unregulated),
174        );
175    }
176
177    Ok(response)
178}
