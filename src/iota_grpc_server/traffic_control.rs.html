4//! Tower [`Layer`] that wires the shared [`TrafficController`] into the
5//! gRPC server.
6//!
7//! The layer extracts the client IP and tallies each request against the
8//! traffic controller, deriving the error weight from the response's gRPC
9//! status code. A batch API's items are invisible at this level: their results
10//! ride in the response body — embedded in a unary response or produced lazily
11//! by a stream — while the layer only inspects the response status, so a batch
12//! would count as a single request regardless of how many items it carries.
13//! Handlers report each item through the [`TallyHandle`] request extension: an
14//! item counts as one request for the spam policy, and a per-item client error
15//! additionally feeds the error policy. Unary batch handlers report their items
16//! eagerly once the response is built; streaming handlers tally each item as
17//! its response stream is consumed, calling [`TallyHandle::mark_accounted`] up
18//! front so the layer still skips its default per-request tally.
19//!
20//! [check]: iota_traffic_controller::TrafficController::check
21//! [tally]: iota_traffic_controller::TrafficController::tally
22

73/// Tower [`Service`] wrapper that gates requests through the traffic
74/// controller.
75#[derive(Clone)]
76pub struct TrafficControlService<S> {
77    inner: S,
78    traffic_controller: Arc<TrafficController>,
79    client_id_source: ClientIdSource,
80}
81
82impl<S, ReqBody, ResBody> Service<http::Request<ReqBody>> for TrafficControlService<S>
83where
84    S: Service<http::Request<ReqBody>, Response = http::Response<ResBody>> + Clone + Send + 'static,
85    S::Future: Send + 'static,
86    ReqBody: Send + 'static,
87    ResBody: Default + Send + 'static,
88{
89    type Response = S::Response;
90    type Error = S::Error;
91    type Future = Pin<Box<dyn Future<Output = Result<S::Response, S::Error>> + Send>>;
92
93    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
94        self.inner.poll_ready(cx)
95    }
96
97    fn call(&mut self, req: http::Request<ReqBody>) -> Self::Future {
98        let traffic_controller = self.traffic_controller.clone();
99        let remote_addr = req
100            .extensions()
101            .get::<TcpConnectInfo>()
102            .and_then(|info| info.remote_addr());
103        let client = match get_client_ip(req.headers(), remote_addr, &self.client_id_source) {
104            ClientIpStatus::Ok(ip) => Some(ip),
105            other => {
106                warn!(
107                    "Skipping traffic controller request handling for client {remote_addr:?}: {other:?}"
108                );
109                None
110            }
111        };
112
113        // A batch handler sets this through its `TallyHandle` once it has
114        // accounted for each item, so the layer skips its default
115        // one-tally-per-request accounting below.
116        let tally_handle = TallyHandle::new(traffic_controller.clone(), client);
117        let mut req = req;
118        req.extensions_mut().insert(tally_handle.clone());
119
120        // Tower contract: `self.inner` is the ready one (poll_ready set it up),
121        // so call it and leave the clone for the next request.
122        let cloned = self.inner.clone();
123        let mut inner = std::mem::replace(&mut self.inner, cloned);
124
125        Box::pin(async move {
126            if !traffic_controller.check(&client, &None) {
127                return Ok(Status::resource_exhausted("Too many requests").into_http());
128            }
129
130            let result = inner.call(req).await;
131            if let Ok(response) = &result {
132                // Batch handlers account for each embedded item themselves; only
133                // tally the request here when a handler didn't.
134                if !tally_handle.per_item_accounted() {
135                    let code =
136                        Status::from_header_map(response.headers()).map_or(Code::Ok, |s| s.code());
137                    tally(&traffic_controller, client, code);
138                }
139            }
140            result
141        })
142    }
143}

145/// Request extension through which a batch handler reports each per-item result
146/// to the traffic controller.
147///
148/// A batch API's items are invisible to the transport-level layer: their
149/// per-item errors are embedded in an otherwise successful response, and a
150/// batch would otherwise count as a single request no matter how many items it
151/// carries. Reporting each item makes it count as one request for the spam
152/// policy and, on a client error, feeds the error policy — so batching cannot
153/// dilute a client's request rate or hide its errors.
154#[derive(Clone)]
155pub struct TallyHandle {
156    traffic_controller: Arc<TrafficController>,
157    client: Option<IpAddr>,
158    per_item_accounted: Arc<AtomicBool>,
159}
160
161impl TallyHandle {
162    fn new(traffic_controller: Arc<TrafficController>, client: Option<IpAddr>) -> Self {
163        Self {
164            traffic_controller,
165            client,
166            per_item_accounted: Arc::new(AtomicBool::new(false)),
167        }
168    }
169
170    /// Whether a handler has accounted for this request's items itself, in
171    /// which case the layer skips its default one-tally-per-request
172    /// accounting.
173    fn per_item_accounted(&self) -> bool {
174        self.per_item_accounted.load(Ordering::Relaxed)
175    }
176
177    /// Signal that the handler will account for this request's items itself, so
178    /// the layer skips its default one-tally-per-request accounting. Call this
179    /// when the per-item tallies happen later than the handler returns — e.g. a
180    /// streaming handler that tallies each item as its response stream is
181    /// consumed, after the layer has already checked the flag.
182    pub fn mark_accounted(&self) {
183        self.per_item_accounted.store(true, Ordering::Relaxed);
184    }
185
186    /// Account for one item of a batch response: count it as one request for
187    /// the spam policy and, if `code` is a client error, feed the error policy.
188    /// Passing `Code::Ok` contributes to the spam policy only.
189    ///
190    /// Also signals the layer to skip its default per-request tally, so a batch
191    /// is charged per item rather than once.
192    pub fn tally_item(&self, code: Code) {
193        self.per_item_accounted.store(true, Ordering::Relaxed);
194        tally(&self.traffic_controller, self.client, code);
195    }
196
197    /// Account for each item of a batch response by its status code (see
198    /// [`Self::tally_item`]).
199    pub fn tally_items(&self, codes: impl IntoIterator<Item = Code>) {
200        for code in codes {
201            self.tally_item(code);
202        }
203    }
204}
205
206fn tally(traffic_controller: &TrafficController, client: Option<IpAddr>, code: Code) {
207    let error_info = if matches!(code, Code::Ok) {
208        None
209    } else {
210        Some((normalize(code), format!("{code:?}")))
211    };
212    traffic_controller.tally(TrafficTally {
213        direct: client,
214        through_fullnode: None,
215        error_info,
216        spam_weight: Weight::one(),
217    });
218}

220/// Map a gRPC status code to a tally weight.
221///
222/// Mirrors the conservative shape of the JSON-RPC `normalize`: only obvious
223/// client-side mistakes contribute to the error-policy budget. Server-side
224/// failures (`Internal`, `Unavailable`, `DataLoss`) must not count, or a
225/// flaky backend would auto-block legitimate clients.
226fn normalize(code: Code) -> Weight {
227    match code {
228        Code::InvalidArgument
229        | Code::FailedPrecondition
230        | Code::OutOfRange
231        | Code::Unauthenticated
232        | Code::PermissionDenied => Weight::one(),
233        _ => Weight::zero(),
234    }
