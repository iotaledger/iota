4//! The sizes of the items built for a response.
5
6use crate::{metrics::RequestMetrics, utils::checkpoint_data_wrapper_overhead};
7
8/// What an item of a response is, for the item size metrics. `kind as usize`
9/// indexes the peak of a kind, so the variants keep their implicit discriminants.
10#[derive(Clone, Copy, strum::IntoStaticStr, strum::EnumIter)]
11#[strum(serialize_all = "snake_case")]
12pub(crate) enum ResponseItemKind {
13    Object,
14    Transaction,
15    Event,
16    Checkpoint,
17    DynamicField,
18    OwnedObject,
19}
20
21/// Adds up the sizes of the messages of one checkpoint while the stream builds
22/// them. Does nothing when the request has no metrics.
23pub(crate) struct CheckpointSizeTracker {
24    metrics: RequestMetrics,
25    bytes: usize,
26}
27
28impl CheckpointSizeTracker {
29    pub(crate) fn new(metrics: &RequestMetrics) -> Self {
30        Self {
31            metrics: metrics.clone(),
32            bytes: 0,
33        }
34    }
35
36    /// Adds a message that holds a batch of items. `batch_size` is the size of
37    /// the items in the batch, with the overhead of each item, as the stream
38    /// counts them. The overhead of the message itself is added here.
39    pub(crate) fn add_batch_size(&mut self, batch_size: usize) {
40        if self.metrics.is_enabled() {
41            self.bytes += batch_size + checkpoint_data_wrapper_overhead(batch_size);
42        }
43    }
44
45    /// Adds the encoded size of `message`.
46    pub(crate) fn add_message_size(&mut self, message: &impl prost::Message) {
47        if self.metrics.is_enabled() {
48            self.bytes += message.encoded_len();
49        }
50    }
51
52    /// Records the sum of the added sizes.
53    pub(crate) fn record(self) {
54        self.metrics
55            .record_response_item(ResponseItemKind::Checkpoint, self.bytes);
56    }
57}
