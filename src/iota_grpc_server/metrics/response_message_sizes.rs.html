16/// One byte of compression flag and four bytes of message length.
17const PREFIX_LEN: usize = 5;
18
19/// Reads the messages of a gRPC response body as the body passes. Each message
20/// starts with a prefix of a compression flag and the length of the message.
21#[derive(Default)]
22struct MessageReader {
23    prefix: [u8; PREFIX_LEN],
24    filled: usize,
25    /// The bytes of the current message still to come after its prefix.
26    remaining: usize,
27}
28
29impl MessageReader {
30    /// Reads the next `chunk` of the body and calls `on_message` with the
31    /// length of each message whose prefix ends in it. A compressed message is
32    /// skipped: the length in its prefix is its compressed size.
33    fn read(&mut self, mut chunk: &[u8], mut on_message: impl FnMut(u32)) {
34        while !chunk.is_empty() {
35            if self.remaining > 0 {
36                let skipped = chunk.len().min(self.remaining);
37                self.remaining -= skipped;
38                chunk = &chunk[skipped..];
39                continue;
40            }
41            let copied = chunk.len().min(PREFIX_LEN - self.filled);
42            self.prefix[self.filled..self.filled + copied].copy_from_slice(&chunk[..copied]);
43            self.filled += copied;
44            chunk = &chunk[copied..];
45            if self.filled == PREFIX_LEN {
46                self.filled = 0;
47                let mut prefix = &self.prefix[..];
48                let compressed = prefix.get_u8() != 0;
49                let length = prefix.get_u32();
50                self.remaining = length as usize;
51                if !compressed {
52                    on_message(length);
53                }
54            }
55        }
56    }
57}

59/// Wraps the body of `response` so that the size of each of its messages is
60/// recorded in `sizes`. See [`SizeRecordingBody`].
61pub(super) fn record_sizes(
62    response: http::Response<tonic::body::Body>,
63    sizes: Histogram,
64) -> http::Response<tonic::body::Body> {
65    response.map(|body| tonic::body::Body::new(SizeRecordingBody::new(body, sizes)))
66}
67
68pin_project! {
69    /// The body of a response. It records the size of each message, which it
70    /// reads from the prefix of the message, and passes all frames on
71    /// unchanged.
72    struct SizeRecordingBody<B> {
73        #[pin]
74        inner: B,
75        sizes: Histogram,
76        reader: MessageReader,
77    }
78}
79
80impl<B> SizeRecordingBody<B> {
81    fn new(inner: B, sizes: Histogram) -> Self {
82        Self {
83            inner,
84            sizes,
85            reader: MessageReader::default(),
86        }
87    }
88}
89
90impl<B> Body for SizeRecordingBody<B>
91where
92    B: Body<Data = Bytes>,
93{
94    type Data = Bytes;
95    type Error = B::Error;
96
97    fn poll_frame(
98        self: Pin<&mut Self>,
99        cx: &mut Context<'_>,
100    ) -> Poll<Option<Result<Frame<Bytes>, Self::Error>>> {
101        let this = self.project();
102        let poll = this.inner.poll_frame(cx);
103        if let Poll::Ready(Some(Ok(frame))) = &poll {
104            if let Some(data) = frame.data_ref() {
105                let sizes = &*this.sizes;
106                this.reader
107                    .read(data, |length| sizes.observe(f64::from(length)));
108            }
109        }
110        poll
111    }
112
113    fn is_end_stream(&self) -> bool {
114        self.inner.is_end_stream()
115    }
116
117    fn size_hint(&self) -> SizeHint {
118        self.inner.size_hint()
119    }
120}

129    fn prefixed(length: u32, compressed: bool) -> Vec<u8> {
130        let mut message = vec![u8::from(compressed)];
131        message.extend_from_slice(&length.to_be_bytes());
132        message.extend(std::iter::repeat_n(7, length as usize));
133        message
134    }
135
136    type Frames =
137        futures::stream::Iter<std::vec::IntoIter<Result<Frame<Bytes>, std::convert::Infallible>>>;
138
139    /// A body that sends `body` in chunks of `sizes`, and the rest in one more
140    /// chunk.
141    fn chunked(body: &[u8], sizes: &[usize]) -> http_body_util::StreamBody<Frames> {
142        let mut frames = Vec::new();
143        let mut rest = body;
144        for size in sizes {
145            let (head, tail) = rest.split_at((*size).min(rest.len()));
146            frames.push(Ok(Frame::data(Bytes::copy_from_slice(head))));
147            rest = tail;
148        }
149        if !rest.is_empty() {
150            frames.push(Ok(Frame::data(Bytes::copy_from_slice(rest))));
151        }
152        http_body_util::StreamBody::new(futures::stream::iter(frames))
153    }
154
155    /// Passes `body` in chunks of `sizes` through a [`SizeRecordingBody`] and
156    /// returns the count and the sum of the observed sizes.
157    async fn observe(body: &[u8], sizes: &[usize]) -> (u64, f64) {
158        use std::future::poll_fn;
159
160        let registry = Registry::new();
161        let histogram = GrpcServerMetrics::new(&registry)
162            .response_message_bytes
163            .with_label_values(&["method"]);
164        let mut recording = SizeRecordingBody::new(chunked(body, sizes), histogram);
165        while poll_fn(|cx| Pin::new(&mut recording).poll_frame(cx))
166            .await
167            .is_some()
168        {}
169        let totals = iota_metrics::test_utils::MetricsReader::new(&registry)
170            .histogram_totals("node_grpc_response_message_bytes", &[]);
171        (totals.count, totals.sum)
172    }

174    #[tokio::test]
175    async fn the_size_is_the_big_endian_length_in_the_prefix() {
176        assert_eq!(
177            observe(&prefixed(0x0001_0203, false), &[]).await,
178            (1, 66_051.0)
179        );
180    }

185        assert_eq!(observe(&message, &[40_000, 40_000]).await, (1, 100_000.0));
186    }

191    }

193    #[tokio::test]
194    async fn each_message_of_a_stream_is_observed() {
195        let mut body = prefixed(10, false);
196        body.extend(prefixed(0, false));
197        body.extend(prefixed(20, false));
198        assert_eq!(observe(&body, &[]).await, (3, 30.0));
199        assert_eq!(observe(&body, &[3, 9, 1, 5, 4]).await, (3, 30.0));
