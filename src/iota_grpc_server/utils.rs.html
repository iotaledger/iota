11/// Per-item overhead when an item is placed in a protobuf `repeated`
12/// length-delimited field (field number 1-15):
13///   field_tag (1 byte) + length_delimiter_varint
14pub(crate) fn repeated_field_item_overhead(item_encoded_len: usize) -> usize {
15    1 + prost::length_delimiter_len(item_encoded_len)
16}
17
18/// Wrapper overhead for a `CheckpointData` oneof message wrapping an inner
19/// message (e.g. `ExecutedTransactions` or `Events`).
20///
21/// Encoding: field_tag (1 byte) + length_delimiter(inner_encoded_len).
22pub(crate) fn checkpoint_data_wrapper_overhead(inner_encoded_len: usize) -> usize {
23    1 + prost::length_delimiter_len(inner_encoded_len)
24}
25
26/// Exact overhead for `has_next: true` in `GetObjectsResponse` /
27/// `GetTransactionsResponse`. A proto3 bool field set to `true` encodes as
28/// field_tag (1 byte) + varint(1) (1 byte) = 2 bytes.
29pub(crate) const HAS_NEXT_TRUE_OVERHEAD: usize = 2;

31pub(crate) fn render_json(
32    grpc_reader: Arc<GrpcReader>,
33    max_json_move_value_size: usize,
34    type_tag: &TypeTag,
35    contents: &[u8],
36) -> Option<prost_types::Value> {
37    // JSON rendering is best-effort - log errors but don't fail the request
38    let layout = grpc_reader
39        .get_type_layout(type_tag)
40        .map_err(|e| tracing::debug!("unable to get type layout for JSON rendering: {e}"))
41        .ok()
42        .flatten()?;
43
44    iota_types::proto_value::ProtoVisitorBuilder::new(max_json_move_value_size)
45        .deserialize_value(contents, &layout)
46        .map_err(|e| tracing::debug!("unable to convert move value to JSON: {e}"))
47        .ok()
48}
