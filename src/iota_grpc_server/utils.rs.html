14/// Parses the optional module or function name of a filter. `field` names it
15/// in the error.
16pub(crate) fn parse_filter_identifier(
17    name: Option<String>,
18    field: &str,
19) -> Result<Option<Identifier>, String> {
20    name.map(|name| {
21        // The move-core check rejects `<SELF>`, which the SDK parser accepts,
22        // and keeps invalid names away from the SDK parser, whose errors copy
23        // the rest of the input.
24        if name.len() <= MAX_IDENTIFIER_LENGTH
25            && move_core_types::identifier::Identifier::is_valid(&name)
26        {
27            Ok(Identifier::new_unchecked(name))
28        } else {
29            Err(format!(
30                "invalid {field} name: {}",
31                quote_filter_input(&name)
32            ))
33        }
34    })
35    .transpose()
36}

38/// Parses the struct tag of an event filter. For a syntax error, the error
39/// gives the byte offset and the input where parsing stopped.
40pub(crate) fn parse_filter_struct_tag(struct_tag: &str) -> Result<StructTag, String> {
41    let tag: StructTag = struct_tag.parse().map_err(|error| match error {
42        // `input` is the rest of the struct tag from where parsing stopped.
43        TypeParseError::Parse { input, .. } => format!(
44            "invalid struct tag: unexpected input at byte {}: {}",
45            struct_tag.len().saturating_sub(input.len()),
46            quote_filter_input(&input)
47        ),
48        error => format!("invalid struct tag: {error}"),
49    })?;
50    // The SDK parser accepts `<SELF>`, which never names a module or type.
51    move_core_identifier_validity_check(tag.module())
52        .and_then(|()| move_core_identifier_validity_check(tag.name()))
53        .and_then(|()| type_tags_identifiers_validity_check(tag.type_params()))
54        .map_err(|_| format!("invalid struct tag: {}", quote_filter_input(struct_tag)))?;
55    Ok(tag)
56}

58/// Quotes a filter value for an error message. Values longer than
59/// [`MAX_IDENTIFIER_LENGTH`] bytes are cut to that length before quoting,
60/// which escapes control characters.
61pub(crate) fn quote_filter_input(value: &str) -> String {
62    if value.len() <= MAX_IDENTIFIER_LENGTH {
63        return format!("{value:?}");
64    }
65    let end = value.floor_char_boundary(MAX_IDENTIFIER_LENGTH);
66    format!("{:?}... ({} bytes)", &value[..end], value.len())
67}

69/// Per-item overhead when an item is placed in a protobuf `repeated`
70/// length-delimited field (field number 1-15):
71///   field_tag (1 byte) + length_delimiter_varint
72pub(crate) fn repeated_field_item_overhead(item_encoded_len: usize) -> usize {
73    1 + prost::length_delimiter_len(item_encoded_len)
74}
75
76/// Wrapper overhead for a `CheckpointData` oneof message wrapping an inner
77/// message (e.g. `ExecutedTransactions` or `Events`).
78///
79/// Encoding: field_tag (1 byte) + length_delimiter(inner_encoded_len).
80pub(crate) fn checkpoint_data_wrapper_overhead(inner_encoded_len: usize) -> usize {
81    1 + prost::length_delimiter_len(inner_encoded_len)
82}

84/// Exact overhead for `has_next: true` in `GetObjectsResponse` /
85/// `GetTransactionsResponse`. A proto3 bool field set to `true` encodes as
86/// field_tag (1 byte) + varint(1) (1 byte) = 2 bytes.
87pub(crate) const HAS_NEXT_TRUE_OVERHEAD: usize = 2;

89pub(crate) fn render_json(
90    grpc_reader: Arc<GrpcReader>,
91    max_json_move_value_size: usize,
92    type_tag: &TypeTag,
93    contents: &[u8],
94) -> Option<prost_types::Value> {
95    // JSON rendering is best-effort - log errors but don't fail the request
96    let layout = grpc_reader
97        .get_type_layout(type_tag)
98        .map_err(|e| tracing::debug!("unable to get type layout for JSON rendering: {e}"))
99        .ok()
100        .flatten()?;
101
102    iota_types::proto_value::ProtoVisitorBuilder::new(max_json_move_value_size)
103        .deserialize_value(contents, &layout)
104        .map_err(|e| tracing::debug!("unable to convert move value to JSON: {e}"))
105        .ok()
106}
