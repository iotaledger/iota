44/// Call each requested `#[view]` Move function and return its results, in
45/// request order.
46///
47/// Each call is resolved against the callee's on-chain signature and run in its
48/// own dev-inspect transaction, so one call aborting or being rejected leaves
49/// the others unaffected. A call's outcome is one of:
50///
51/// - `outputs.return_values` — the call ran and returned;
52/// - `outputs.execution_error` — the call ran but aborted;
53/// - `error` — the server rejected the call before running it (not a `#[view]`
54///   function, unknown package/module/function, or an argument that could not
55///   be resolved).
56///
57/// The request's `read_mask` selects which fields of each result to populate,
58/// defaulting to [`VIEW_FUNCTION_CALLS_READ_MASK`].
59#[tracing::instrument(skip_all, fields(batch_size = request.view_function_calls.len()))]
60pub async fn view_function_calls(
61    reader: &Arc<GrpcReader>,
62    executor: &Arc<dyn TransactionExecutor>,
63    build_client: &NodeTransactionBuilderLedgerClient,
64    config: &iota_config::node::GrpcApiConfig,
65    request: ViewFunctionCallsRequest,
66) -> Result<ViewFunctionCallsResponse, RpcError> {
67    super::validate_batch_size(
68        request.view_function_calls.len(),
69        config.max_view_function_call_batch_size as usize,
70    )?;
71    let read_mask = validate_read_mask::<ViewFunctionCallOutputs>(
72        request.read_mask,
73        VIEW_FUNCTION_CALLS_READ_MASK,
74    )?;
75
76    let mut call_results = Vec::with_capacity(request.view_function_calls.len());
77    for item in &request.view_function_calls {
78        let result = match view_function_call(
79            reader,
80            executor,
81            build_client,
82            config,
83            item,
84            &read_mask,
85        )
86        .await
87        {
88            Ok(r) => r,
89            Err(error) => ViewFunctionCallResult::default().with_error(error.into_status_proto()),
90        };
91        call_results.push(result);
92    }
93
94    Ok(ViewFunctionCallsResponse::default().with_call_results(call_results))
95}

97/// Run a single view function call in its own dev-inspect transaction and build
98/// its [`ViewFunctionCallResult`].
99///
100/// Returns `Err` only for calls the server rejects before execution (bad
101/// arguments, missing or non-`#[view]` function); the caller records that as
102/// the result's `error`. A call that runs — whether it returns or aborts — is
103/// reported through the `Ok` result's `outputs`.
104async fn view_function_call(
105    reader: &Arc<GrpcReader>,
106    executor: &Arc<dyn TransactionExecutor>,
107    build_client: &NodeTransactionBuilderLedgerClient,
108    config: &iota_config::node::GrpcApiConfig,
109    item: &ViewFunctionCallItem,
110    read_mask: &FieldMaskTree,
111) -> Result<ViewFunctionCallResult, RpcError> {
112    let mut builder = TransactionBuilder::new(Address::ZERO).with_client(build_client.clone());
113    add_view_function_call(reader, &mut builder, item)?;
114
115    let kind = builder.finish_kind().await.map_err(|err| {
116        RpcError::new(
117            Code::InvalidArgument,
118            format!("Failed to build transaction kind from arguments: {err}"),
119        )
120    })?;
121    // A view call needs no gas: the sender is the zero address with no gas
122    // coins, and a zero gas price and budget signal the executor to mock a gas
123    // coin for the dev-inspect run.
124    let transaction = TransactionV1 {
125        kind,
126        sender: Address::ZERO,
127        gas_payment: GasPayment {
128            objects: vec![],
129            owner: Address::ZERO,
130            price: 0,
131            budget: 0,
132        },
133        expiration: TransactionExpiration::None,
134    };
135
136    // Dev-inspect the transaction
137    let simulation = executor
138        .simulate_transaction(transaction.into(), VmChecks::Disabled)
139        .map_err(|e| {
140            RpcError::new(
141                if matches!(e, IotaError::UserInput { .. }) {
142                    Code::InvalidArgument
143                } else {
144                    Code::Internal
145                },
146                format!("transaction simulation failed: {e}"),
147            )
148        })?;
149
150    // Build the response
151    let mut response = ViewFunctionCallResult::default();
152
153    // Only include the result if requested
154    if let Some(outputs_mask) = read_mask.subtree(ViewFunctionCallOutputs::EXECUTION_RESULT_ONEOF) {
155        // The call ran either way; report its return values, or the reason it
156        // aborted, as the `execution_result` of the outputs.
157        let outputs = match simulation.execution_result {
158            Ok(command_results) => {
159                // The transaction holds exactly one `MoveCall` command, so the
160                // executor returns exactly one set of command results.
161                let [(mutable_ref_outputs, return_values)] = command_results.as_slice() else {
162                    return Err(RpcError::new(
163                        Code::Internal,
164                        format!(
165                            "expected exactly one command result, got {}",
166                            command_results.len()
167                        ),
168                    ));
169                };
170                if !mutable_ref_outputs.is_empty() {
171                    return Err(RpcError::new(
172                        Code::Internal,
173                        format!(
174                            "expected no mutable ref outputs in command result, got {}",
175                            mutable_ref_outputs.len()
176                        ),
177                    ));
178                }
179                build_return_values(reader, config, &outputs_mask, return_values)?
180            }
181            Err(execution_error) => build_execution_error(&outputs_mask, &execution_error)?,
182        };
183        response.result = Some(super::view_function_call_result::Result::CallOutputs(
184            outputs,
185        ));
186    }
187
188    Ok(response)
189}

191/// Build the `return_values` of a call that ran to completion, honoring the
192/// read mask (a subtree of the `outputs` field).
193fn build_return_values(
194    reader: &Arc<GrpcReader>,
195    config: &iota_config::node::GrpcApiConfig,
196    outputs_mask: &FieldMaskTree,
197    return_values: &[(Vec<u8>, TypeTag)],
198) -> Result<ViewFunctionCallOutputs, RpcError> {
199    let Some(return_values_mask) =
200        outputs_mask.subtree(ViewFunctionCallOutputs::RETURN_VALUES_FIELD.name)
201    else {
202        return Ok(ViewFunctionCallOutputs::default());
203    };
204    let source = CommandOutputsReadSource {
205        reader,
206        config,
207        outputs: return_values
208            .iter()
209            .map(|(bcs, ty)| (None, bcs.as_slice(), ty))
210            .collect(),
211    };
212    Ok(ViewFunctionCallOutputs::default()
213        .with_return_values(CommandOutputs::merge_from(&source, &return_values_mask)?))
214}

216/// Build the `execution_error` of a call that aborted, honoring the read mask
217/// (a subtree of the `outputs` field). Mirrors the error handling in
218/// `simulate_transactions`.
219fn build_execution_error(
220    outputs_mask: &FieldMaskTree,
221    execution_error: &iota_types::error::ExecutionError,
222) -> Result<ViewFunctionCallOutputs, RpcError> {
223    let Some(error_mask) =
224        outputs_mask.subtree(ViewFunctionCallOutputs::EXECUTION_ERROR_FIELD.name)
225    else {
226        return Ok(ViewFunctionCallOutputs::default());
227    };
228    let mut exec_error = ExecutionError::default();
229    if error_mask.contains(ExecutionError::BCS_KIND_FIELD.name) {
230        exec_error.bcs_kind = Some(BcsData::serialize(execution_error.kind()).map_err(|e| {
231            RpcError::new(
232                Code::Internal,
233                format!("failed to serialize execution error kind: {e}"),
234            )
235        })?);
236    }
237    if error_mask.contains(ExecutionError::SOURCE_FIELD.name) {
238        exec_error.source = execution_error
239            .source()
240            .as_ref()
241            .map(|source| source.to_string());
242    }
243    if error_mask.contains(ExecutionError::COMMAND_INDEX_FIELD.name) {
244        if let Some(command_idx) = execution_error.command() {
245            exec_error.command_index = Some(command_idx);
246        }
247    }
248    Ok(ViewFunctionCallOutputs::default().with_execution_error(exec_error))
249}

251/// Resolve one call's arguments against its `#[view]` signature and append a
252/// `MoveCall` command to `builder`.
253fn add_view_function_call(
254    reader: &Arc<GrpcReader>,
255    builder: &mut TransactionBuilder<NodeTransactionBuilderLedgerClient>,
256    item: &ViewFunctionCallItem,
257) -> Result<(), RpcError> {
258    let (package_id, module, function) = parse_fq_function_name(&item.fq_function_name)?;
259    let type_args = convert_type_args(&item.type_args)?;
260    let args = item
261        .inputs
262        .iter()
263        .map(proto_arg_to_call_arg)
264        .collect::<Result<Vec<_>, _>>()?;
265
266    let object = reader
267        .get_object(&package_id)
268        .map_err(|e| RpcError::new(Code::Internal, format!("failed to read package: {e}")))?
269        .ok_or_else(|| RpcError::new(Code::NotFound, format!("package {package_id} not found")))?;
270    let package = object.as_opt_package().ok_or_else(|| {
271        RpcError::new(
272            Code::InvalidArgument,
273            format!("{package_id} is not a package"),
274        )
275    })?;
276
277    let compiled_module = package
278        .deserialize_module(&module, &BinaryConfig::standard())
279        .map_err(|e| {
280            RpcError::new(
281                Code::InvalidArgument,
282                format!("failed to deserialize module {module}: {e}"),
283            )
284        })?;
285    let resolved = resolve_move_function_args(&compiled_module, &function, &type_args, args, true)
286        .map_err(|e| RpcError::new(Code::InvalidArgument, format!("{e}")))?;
287
288    let mut args = Vec::with_capacity(resolved.len());
289    for (resolved_arg, _) in resolved {
290        let arg = match resolved_arg {
291            ResolvedCallArg::Pure(bytes) => {
292                builder.apply_argument(iota_sdk_types::Input::Pure(bytes))
293            }
294            ResolvedCallArg::Object(object_id) => builder.apply_argument(object_id),
295            ResolvedCallArg::ObjVec(_) => {
296                return Err(RpcError::new(
297                    Code::InvalidArgument,
298                    "vector of objects argument to view functions not supported".to_string(),
299                ));
300            }
301        };
302        args.push(arg);
303    }
304
305    builder
306        .move_call(package_id, module.as_str(), function.as_str())
307        .type_tags(type_args)
308        .arguments(args);
309
310    Ok(())
311}

313/// Split `<package>::<module>::<function>` into its parts.
314fn parse_fq_function_name(name: &str) -> Result<(ObjectId, Identifier, Identifier), RpcError> {
315    let (module_id, function) = parse_iota_fq_name(name).map_err(|e| {
316        RpcError::new(
317            Code::InvalidArgument,
318            format!("invalid function name `{name}`: {e}"),
319        )
320    })?;
321    let package = ObjectId::new((*module_id.address()).into_bytes());
322    let module = Identifier::new(module_id.name().as_str())
323        .map_err(|e| RpcError::new(Code::InvalidArgument, format!("invalid module name: {e}")))?;
324    let function = Identifier::new(function)
325        .map_err(|e| RpcError::new(Code::InvalidArgument, format!("invalid function name: {e}")))?;
326    Ok((package, module, function))
327}

329/// Convert the request's protobuf type arguments into internal `TypeTag`s.
330fn convert_type_args(type_args: &[ProtoTypeTag]) -> Result<Vec<TypeTag>, RpcError> {
331    type_args
332        .iter()
333        .map(|tt| {
334            TypeTag::try_from(tt).map_err(|e| {
335                RpcError::new(Code::InvalidArgument, format!("invalid type argument: {e}"))
336            })
337        })
338        .collect()
339}

341/// Convert one request argument into a [`IotaMoveCallInputValue`]. Exactly one
342/// of `bcs` / `json` must be set.
343fn proto_arg_to_call_arg(arg: &InputArgument) -> Result<IotaMoveCallInputValue, RpcError> {
344    match &arg.input {
345        Some(input_argument::Input::Bcs(bcs)) => Ok(IotaMoveCallInputValue::Bcs(bcs.data.to_vec())),
346        Some(input_argument::Input::Json(value)) => IotaJsonValue::new(prost_to_json(value))
347            .map(IotaMoveCallInputValue::Json)
348            .map_err(|e| {
349                RpcError::new(Code::InvalidArgument, format!("invalid json argument: {e}"))
350            }),
351        Some(_) => Err(RpcError::new(
352            Code::InvalidArgument,
353            "argument is neither bcs nor json",
354        )),
355        None => Err(RpcError::new(Code::InvalidArgument, "argument is not set")),
356    }
357}

362    use iota_sdk_types::ObjectId;
363    use prost_types::{Value, value::Kind};

367    #[test]
368    fn json_arg_becomes_json_view_arg() {
369        let arg = InputArgument::default().with_json(Value {
370            kind: Some(Kind::StringValue("42".into())),
371        });
372        assert!(matches!(
373            proto_arg_to_call_arg(&arg).unwrap(),
374            IotaMoveCallInputValue::Json(_)
375        ));
376    }
377
378    #[test]
379    fn bcs_arg_becomes_bcs_view_arg() {
380        let arg = InputArgument::default().with_bcs(BcsData::default().with_data(vec![1, 2, 3]));
381        match proto_arg_to_call_arg(&arg).unwrap() {
382            IotaMoveCallInputValue::Bcs(bytes) => assert_eq!(bytes, vec![1, 2, 3]),
383            _ => panic!("expected bcs"),
384        }
385    }
386
387    #[test]
388    fn arg_with_neither_field_is_rejected() {
389        let arg = InputArgument::default();
390        assert!(proto_arg_to_call_arg(&arg).is_err());
391    }
392
393    #[test]
394    fn parse_fq_name_splits_package_module_function() {
395        let (pkg, module, function) = parse_fq_function_name("0x2::coin::value").unwrap();
396        assert_eq!(pkg, ObjectId::from_short_hex("0x2").unwrap());
397        assert_eq!(module.as_str(), "coin");
398        assert_eq!(function.as_str(), "value");
399    }
