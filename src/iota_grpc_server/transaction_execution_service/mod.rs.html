165/// Validate that a batch is non-empty and within the size limit.
166fn validate_batch_size(items_len: usize, max_batch: usize) -> Result<(), RpcError> {
167    if items_len == 0 {
168        return Err(RpcError::new(
169            tonic::Code::InvalidArgument,
170            "batch must not be empty",
171        ));
172    }
173    if items_len > max_batch {
174        return Err(RpcError::new(
175            tonic::Code::InvalidArgument,
176            format!("batch size {items_len} exceeds maximum allowed ({max_batch})"),
177        ));
178    }
179    Ok(())
180}

182/// Extract, deserialize, and validate a transaction from its proto
183/// representation.
184///
185/// This performs the common validation steps shared by both execute and
186/// simulate:
187/// 1. Ensure the transaction field is present
188/// 2. Extract and deserialize the BCS data
189/// 3. Validate the digest if provided
190fn parse_transaction_proto(
191    transaction: Option<&ProtoTransaction>,
192) -> Result<iota_sdk_types::transaction::Transaction, RpcError> {
193    let transaction_proto = transaction
194        .ok_or_else(|| FieldViolation::new("transaction").with_reason(ErrorReason::FieldMissing))?;
195
196    let transaction_bcs = transaction_proto.bcs.as_ref().ok_or_else(|| {
197        FieldViolation::new("transaction.bcs")
198            .with_description("transaction BCS is required")
199            .with_reason(ErrorReason::FieldMissing)
200    })?;
201
202    let sdk_transaction: iota_sdk_types::transaction::Transaction =
203        bcs::from_bytes(&transaction_bcs.data).map_err(|e| {
204            FieldViolation::new("transaction.bcs")
205                .with_description(format!("invalid transaction BCS: {e}"))
206                .with_reason(ErrorReason::FieldInvalid)
207        })?;
208
209    if let Some(provided_digest) = &transaction_proto.digest {
210        let computed_digest = sdk_transaction.digest();
211        let provided_digest_bytes: [u8; 32] =
212            provided_digest.digest.as_ref().try_into().map_err(|_| {
213                FieldViolation::new("transaction.digest")
214                    .with_description("digest must be exactly 32 bytes")
215                    .with_reason(ErrorReason::FieldInvalid)
216            })?;
217
218        if computed_digest.bytes() != &provided_digest_bytes {
219            let provided_digest_typed = iota_sdk_types::Digest::new(provided_digest_bytes);
220            return Err(FieldViolation::new("transaction.digest")
221                .with_description(format!(
222                    "provided digest does not match computed digest: provided={provided_digest_typed}, computed={computed_digest}"
223                ))
224                .with_reason(ErrorReason::FieldInvalid)
225                .into());
226        }
227    }
228
229    Ok(sdk_transaction)
230}

232/// Execute a batch of transactions sequentially.
233///
234/// Each transaction is executed independently — failure of one does not abort
235/// the rest. Results are returned in the same order as the input.
236///
237/// ## Checkpoint Inclusion
238///
239/// If `checkpoint_inclusion_timeout_ms` is set in the request, the server will
240/// wait (after executing all transactions) for them to be included in a
241/// checkpoint. On success the `checkpoint` and `timestamp` fields of each
242/// `ExecutedTransaction` are populated. If the timeout expires, partial results
243/// are returned: transactions that were already checkpointed will have the
244/// fields set, while the rest will have them unset.
245///
246/// Note: `checkpoint` and `timestamp` must also be included in the `read_mask`
247/// for them to appear in the response.
248///
249/// ## Read Mask
250///
251/// The read mask paths apply directly to
252/// [`ExecutedTransaction`](iota_grpc_types::v1::transaction::ExecutedTransaction)
253/// fields (e.g. `"effects"`, not `"executed_transaction.effects"`).
254///
255/// ## Available Read Mask Fields
256///
257/// The `execute_transactions` function supports the following `read_mask`
258/// fields to control which data is included in each `ExecutedTransaction`
259/// result:
260///
261/// ## Transaction Fields
262/// - `transaction` - includes all transaction fields
263///   - `transaction.digest` - the transaction digest
264///   - `transaction.bcs` - the full BCS-encoded transaction
265/// - `signatures` - includes all signature fields
266///   - `signatures.bcs` - the full BCS-encoded signature
267/// - `effects` - includes all effects fields
268///   - `effects.digest` - the effects digest
269///   - `effects.bcs` - the full BCS-encoded effects
270///
271/// ## Event Fields
272/// - `events` - includes all event fields (all events of the transaction)
273///   - `events.digest` - the events digest
274///   - `events.events.bcs` - the full BCS-encoded event
275///   - `events.events.package_id` - the ID of the package that emitted the
276///     event
277///   - `events.events.module` - the module that emitted the event
278///   - `events.events.sender` - the sender that triggered the event
279///   - `events.events.event_type` - the type of the event
280///   - `events.events.bcs_contents` - the full BCS-encoded contents of the
281///     event
282///   - `events.events.json_contents` - the JSON-encoded contents of the event
283///
284/// ## Checkpoint Fields
285/// - `checkpoint` - the checkpoint that included the transaction. Requires
286///   `checkpoint_inclusion_timeout_ms` to be set.
287/// - `timestamp` - the timestamp of the checkpoint. Requires
288///   `checkpoint_inclusion_timeout_ms` to be set.
289///
290/// ## Object Fields
291/// - `input_objects` - includes all input object fields
292///   - `input_objects.reference` - includes all reference fields
293///     - `input_objects.reference.object_id` - the ID of the input object
294///     - `input_objects.reference.version` - the version of the input object
295///     - `input_objects.reference.digest` - the digest of the input object
296///       contents
297///   - `input_objects.bcs` - the full BCS-encoded object
298/// - `output_objects` - includes all output object fields
299///   - `output_objects.reference` - includes all reference fields
300///     - `output_objects.reference.object_id` - the ID of the output object
301///     - `output_objects.reference.version` - the version of the output object
302///     - `output_objects.reference.digest` - the digest of the output object
303///       contents
304///   - `output_objects.bcs` - the full BCS-encoded object
305///
306/// ## Derived Change Fields
307/// Derived from the transaction's effects and input/output objects.
308/// - `balance_changes` - per-owner, per-coin-type balance deltas. For a failed
309///   transaction this contains only the gas charge.
310/// - `object_changes` - structured object changes (created, mutated, deleted,
311///   wrapped, unwrapped, published)
312#[tracing::instrument(skip_all, fields(batch_size = request.transactions.len()))]
313pub async fn execute_transactions(
314    reader: &Arc<GrpcReader>,
315    executor: &Arc<dyn TransactionExecutor>,
316    config: &iota_config::node::GrpcApiConfig,
317    request: ExecuteTransactionsRequest,
318) -> Result<ExecuteTransactionsResponse, RpcError> {
319    validate_batch_size(
320        request.transactions.len(),
321        config.max_execute_transaction_batch_size as usize,
322    )?;
323    let read_mask = validate_read_mask::<ExecutedTransaction>(
324        request.read_mask,
325        EXECUTE_TRANSACTIONS_READ_MASK,
326    )?;
327
328    // Parse and clamp checkpoint inclusion timeout.
329    // If None or 0 is provided, we won't wait for checkpoint inclusion and the
330    // response will be returned immediately after execution with
331    // checkpoint/timestamp fields unset. If a positive value is provided, the
332    // server will wait up to the specified duration for the transaction to be
333    // included in a checkpoint before returning. The timeout is clamped by the
334    // server's max_checkpoint_inclusion_timeout_ms config to prevent excessively
335    // long waits.
336    let checkpoint_timeout = request
337        .checkpoint_inclusion_timeout_ms
338        .filter(|&ms| ms > 0)
339        .map(|ms| Duration::from_millis(ms.min(config.max_checkpoint_inclusion_timeout_ms)));
340
341    // If the client opted into waiting for checkpoint inclusion, the
342    // subsequent `wait_for_checkpoint_inclusion` call below provides
343    // finality, so we instruct the executor to skip its 2f+1 effects
344    // certification. Otherwise the executor performs the full certification.
345    let skip_certification = checkpoint_timeout.is_some();
346
347    // Drive the per-tx executions concurrently so a slow validator submission
348    // on one item doesn't stall the rest of the batch. Results are collected
349    // back into per-index slots so the response order matches the request
350    // order regardless of completion order. The post-batch
351    // `wait_for_checkpoint_inclusion` + `finalize_item` gate still applies
352    // unchanged — uncertified single-validator data is never returned to the
353    // client without checkpoint corroboration (or a `DeadlineExceeded` error
354    // in its place).
355    let mut transaction_results: Vec<ExecuteTransactionResult> = (0..request.transactions.len())
356        .map(|_| ExecuteTransactionResult::default())
357        .collect();
358    // For each successful execution: (index, digest, rebuild_ctx). `rebuild_ctx`
359    // is `Some` only when the response carried uncertified single-validator
360    // data and must be rebuilt from cache after checkpoint inclusion.
361    let mut successful_digests: Vec<(usize, TransactionDigest, Option<RebuildCtx>)> = Vec::new();
362    let read_mask_ref = &read_mask;
363    let mut driver_futs: FuturesUnordered<_> = request
364        .transactions
365        .iter()
366        .enumerate()
367        .map(|(i, item)| async move {
368            let result = execute_single_transaction(
369                reader,
370                executor,
371                config,
372                item,
373                read_mask_ref,
374                skip_certification,
375            )
376            .await;
377            (i, result)
378        })
379        .collect();
380    while let Some((i, result)) = driver_futs.next().await {
381        transaction_results[i] = match result {
382            Ok((digest, tx, rebuild_ctx)) => {
383                successful_digests.push((i, digest, rebuild_ctx));
384                ExecuteTransactionResult::default().with_executed_transaction(tx)
385            }
386            Err(error) => ExecuteTransactionResult::default().with_error(error.into_status_proto()),
387        };
388    }
389
390    // Optionally wait for checkpoint inclusion, then finalize each successful
391    // item: rebuild from cache for skip-effect-cert items (so we don't return
392    // uncertified single-validator data) or patch checkpoint/timestamp on the
393    // already-built response for the cert path.
394    if let (Some(timeout), false) = (checkpoint_timeout, successful_digests.is_empty()) {
395        let digests: Vec<_> = successful_digests.iter().map(|(_, d, _)| *d).collect();
396        let (checkpoint_map, wait_error) = match executor
397            .wait_for_checkpoint_inclusion(&digests, timeout)
398            .await
399        {
400            Ok(m) => (m, None),
401            Err(e) => {
402                tracing::warn!("wait_for_checkpoint_inclusion failed: {e}");
403                (std::collections::BTreeMap::new(), Some(format!("{e}")))
404            }
405        };
406        let flags = ReadFlags::from_mask(&read_mask);
407        let flags_ref = &flags;
408        let read_mask_ref = &read_mask;
409        let wait_error_ref = wait_error.as_deref();
410
411        // Run per-tx finalization (which calls into `rebuild_from_cache` →
412        // cache reads + object derivation) in parallel; results land back
413        // in per-index slots so the response order matches the request.
414        let mut finalize_futs: FuturesUnordered<_> = successful_digests
415            .into_iter()
416            .map(|(i, digest, rebuild_ctx)| {
417                let original = std::mem::take(&mut transaction_results[i]);
418                let checkpoint_entry = checkpoint_map.get(&digest).copied();
419                async move {
420                    let result = finalize_item(
421                        reader,
422                        executor,
423                        config,
424                        read_mask_ref,
425                        flags_ref,
426                        original,
427                        &digest,
428                        rebuild_ctx,
429                        checkpoint_entry,
430                        wait_error_ref,
431                    )
432                    .await;
433                    (i, result)
434                }
435            })
436            .collect();
437        while let Some((i, result)) = finalize_futs.next().await {
438            transaction_results[i] = result;
439        }
440    }
441
442    Ok(ExecuteTransactionsResponse::default().with_transaction_results(transaction_results))
443}

445/// Read-mask flags extracted once and passed into the per-item finalizer.
446struct ReadFlags {
447    include_events: bool,
448    include_input_objects: bool,
449    include_output_objects: bool,
450    needs_checkpoint: bool,
451    needs_timestamp: bool,
452}
453
454impl ReadFlags {
455    fn from_mask(mask: &FieldMaskTree) -> Self {
456        // Balance/object changes are derived from the input/output objects, so
457        // the executor must return them whenever the derived fields are
458        // requested.
459        let derives_changes = mask.contains(ExecutedTransaction::BALANCE_CHANGES_FIELD.name)
460            || mask.contains(ExecutedTransaction::OBJECT_CHANGES_FIELD.name);
461
462        Self {
463            include_events: mask.contains(ExecutedTransaction::EVENTS_FIELD.name),
464            include_input_objects: mask.contains(ExecutedTransaction::INPUT_OBJECTS_FIELD.name)
465                || derives_changes,
466            include_output_objects: mask.contains(ExecutedTransaction::OUTPUT_OBJECTS_FIELD.name)
467                || derives_changes,
468            needs_checkpoint: mask.contains(ExecutedTransaction::CHECKPOINT_FIELD.name),
469            needs_timestamp: mask.contains(ExecutedTransaction::TIMESTAMP_FIELD.name),
470        }
471    }
472}

474/// Patch `checkpoint` / `timestamp` on an already-built executed transaction
475/// result when requested by the read mask. No-op if the result isn't an
476/// `ExecutedTransaction`.
477fn patch_checkpoint_timestamp(
478    result: &mut ExecuteTransactionResult,
479    seq: u64,
480    ts_ms: u64,
481    flags: &ReadFlags,
482) {
483    if let Some(execute_transaction_result::Result::ExecutedTransaction(ref mut tx)) = result.result
484    {
485        if flags.needs_checkpoint {
486            tx.checkpoint = Some(seq);
487        }
488        if flags.needs_timestamp && ts_ms > 0 {
489            tx.timestamp = Some(timestamp_ms_to_proto(ts_ms));
490        }
491    }
492}

498/// Produce the final result for one item after the checkpoint wait. Handles
499/// all four combinations of (cert vs skip-cert) × (checkpointed vs not).
500#[allow(clippy::too_many_arguments)]
501async fn finalize_item(
502    reader: &Arc<GrpcReader>,
503    executor: &Arc<dyn TransactionExecutor>,
504    config: &iota_config::node::GrpcApiConfig,
505    read_mask: &FieldMaskTree,
506    flags: &ReadFlags,
507    mut original: ExecuteTransactionResult,
508    digest: &TransactionDigest,
509    rebuild_ctx: Option<RebuildCtx>,
510    checkpoint_entry: Option<(u64, u64)>,
511    wait_error: Option<&str>,
512) -> ExecuteTransactionResult {
513    let Some((seq, ts)) = checkpoint_entry else {
514        // Not checkpointed within the timeout (or the wait call itself failed).
515        // Cert path: leave the 2f+1-safe response as-is; the client can poll
516        // for the checkpoint/timestamp fields later. Skip-effect-cert path:
517        // the response holds uncertified single-validator data — return an
518        // error instead of leaking it.
519        if rebuild_ctx.is_none() {
520            return original;
521        }
522        return match wait_error {
523            Some(e) => error_result(
524                tonic::Code::Internal,
525                format!("wait_for_checkpoint_inclusion failed for tx {digest:?}: {e}"),
526            ),
527            None => error_result(
528                tonic::Code::DeadlineExceeded,
529                format!(
530                    "transaction {digest:?} was submitted but not included in a checkpoint \
531                     within the timeout"
532                ),
533            ),
534        };
535    };
536
537    let Some(ctx) = rebuild_ctx else {
538        // Cert path, checkpointed: just patch checkpoint/timestamp.
539        patch_checkpoint_timestamp(&mut original, seq, ts, flags);
540        return original;
541    };
542
543    // Was the original (single-validator) response claiming events? Captured
544    // here so we can detect a submitter-vs-cache events disagreement after
545    // the rebuild (mirrors the orchestrator's `skip_effect_cert_events_cache_miss`
546    // invariant).
547    let submitter_claimed_events = matches!(
548        &original.result,
549        Some(execute_transaction_result::Result::ExecutedTransaction(tx)) if tx.events.is_some(),
550    );
551
552    // Skip-effect-cert path, checkpointed: rebuild the response from the
553    // authoritative local cache to replace the single-validator data.
554    match rebuild_from_cache(
555        reader, executor, config, read_mask, ctx, digest, seq, ts, flags,
556    )
557    .await
558    {
559        Ok(Some(tx)) => {
560            if submitter_claimed_events && flags.include_events && tx.events.is_none() {
561                tracing::warn!(
562                    ?digest,
563                    "submitter claimed events but local cache has none — discarding \
564                     (possible byzantine submitter)"
565                );
566            }
567            ExecuteTransactionResult::default().with_executed_transaction(tx)
568        }
569        Ok(None) => {
570            // Executor has no cache (e.g. simulacrum). The skip-cert path
571            // here means `original` still carries uncertified single-validator
572            // data — never return that to the client. In production the
573            // orchestrator impl always returns `Some`; an executor that opts
574            // into `skip_certification` without providing a cache is a
575            // configuration error.
576            tracing::error!(
577                ?digest,
578                "skip-cert finalize: executor returned no cached data despite checkpoint \
579                 inclusion — refusing to return uncertified response"
580            );
581            error_result(
582                tonic::Code::Internal,
583                format!(
584                    "executor cannot rebuild tx {digest:?} from local cache; \
585                     skip-certification requires a cache-backed executor"
586                ),
587            )
588        }
589        Err(e) => {
590            tracing::warn!(?digest, "failed to rebuild executed tx from cache: {e}");
591            error_result(
592                tonic::Code::Internal,
593                format!(
594                    "failed to rebuild tx {digest:?} from local cache after checkpoint \
595                     inclusion: {e}"
596                ),
597            )
598        }
599    }
600}

610/// Rebuild an `ExecutedTransaction` from the local cache for a tx that has
611/// just been observed in a checkpoint. Returns `Ok(None)` if the executor
612/// does not have cache data for this tx (e.g. simulacrum).
613async fn rebuild_from_cache(
614    reader: &Arc<GrpcReader>,
615    executor: &Arc<dyn TransactionExecutor>,
616    config: &iota_config::node::GrpcApiConfig,
617    read_mask: &FieldMaskTree,
618    ctx: RebuildCtx,
619    digest: &TransactionDigest,
620    checkpoint_seq: u64,
621    checkpoint_ts_ms: u64,
622    flags: &ReadFlags,
623) -> Result<Option<ExecutedTransaction>, RpcError> {
624    let Some(cached) = executor
625        .read_transaction_from_cache(
626            digest,
627            flags.include_events,
628            flags.include_input_objects,
629            flags.include_output_objects,
630        )
631        .map_err(|e| {
632            RpcError::new(
633                tonic::Code::Internal,
634                format!("failed to read tx {digest:?} from cache: {e:?}"),
635            )
636        })?
637    else {
638        return Ok(None);
639    };
640
641    let source = TransactionReadSource {
642        reader: reader.clone(),
643        config,
644        transaction: Some(ctx.transaction),
645        signatures: Some(ctx.signatures),
646        effects: Some(cached.effects),
647        events: cached.events,
648        checkpoint: Some(checkpoint_seq),
649        timestamp_ms: if checkpoint_ts_ms > 0 {
650            Some(checkpoint_ts_ms)
651        } else {
652            None
653        },
654        input_objects: cached.input_objects,
655        output_objects: cached.output_objects,
656        mocked_coin: None,
657    };
658
659    let executed = ExecutedTransaction::merge_from(&source, read_mask)
660        .map_err(|e| e.with_context("failed to merge executed transaction from cache"))?;
661
662    Ok(Some(executed))
663}

665/// Validate, execute, and merge a single transaction item.
666async fn execute_single_transaction(
667    reader: &Arc<GrpcReader>,
668    executor: &Arc<dyn TransactionExecutor>,
669    config: &iota_config::node::GrpcApiConfig,
670    item: &ExecuteTransactionItem,
671    read_mask: &FieldMaskTree,
672    skip_certification: bool,
673) -> Result<(TransactionDigest, ExecutedTransaction, Option<RebuildCtx>), RpcError> {
674    let sdk_transaction = parse_transaction_proto(item.transaction.as_ref())?;
675
676    // Extract and validate signatures
677    let signatures_proto = item
678        .signatures
679        .as_ref()
680        .ok_or_else(|| FieldViolation::new("signatures").with_reason(ErrorReason::FieldMissing))?;
681
682    let sdk_signatures = signatures_proto
683        .signatures
684        .iter()
685        .enumerate()
686        .map(|(i, sig)| {
687            let bcs_data = sig.bcs.as_ref().ok_or_else(|| {
688                FieldViolation::new_at("signatures", i)
689                    .with_description("signature BCS is required")
690                    .with_reason(ErrorReason::FieldMissing)
691            })?;
692
693            bcs::from_bytes::<iota_sdk_types::UserSignature>(&bcs_data.data).map_err(|e| {
694                FieldViolation::new_at("signatures", i)
695                    .with_description(format!("invalid signature: {e}"))
696                    .with_reason(ErrorReason::FieldInvalid)
697            })
698        })
699        .collect::<std::result::Result<Vec<_>, _>>()?;
700
701    // Create signed transaction
702    let sdk_signed_transaction = iota_sdk_types::SignedTransaction {
703        transaction: sdk_transaction,
704        signatures: sdk_signatures,
705    };
706
707    let transaction = iota_types::transaction::TransactionEnvelope::from(sdk_signed_transaction);
708
709    // Determine what to include in the request based on read mask.
710    // Balance/object changes are derived from the input/output objects, so the
711    // executor must return them whenever the derived fields are requested.
712    let derives_changes = read_mask.contains(ExecutedTransaction::BALANCE_CHANGES_FIELD.name)
713        || read_mask.contains(ExecutedTransaction::OBJECT_CHANGES_FIELD.name);
714    let include_events = read_mask.contains(ExecutedTransaction::EVENTS_FIELD.name);
715    let include_input_objects =
716        read_mask.contains(ExecutedTransaction::INPUT_OBJECTS_FIELD.name) || derives_changes;
717    let include_output_objects =
718        read_mask.contains(ExecutedTransaction::OUTPUT_OBJECTS_FIELD.name) || derives_changes;
719
720    // Create execution request
721    let exec_request = ExecuteTransactionRequestV1 {
722        transaction: transaction.clone(),
723        include_events,
724        include_input_objects,
725        include_output_objects,
726        include_auxiliary_data: false,
727    };
728
729    // Execute the transaction
730    let ExecuteTransactionResponseV1 {
731        effects,
732        events,
733        input_objects,
734        output_objects,
735        auxiliary_data: _,
736    } = executor
737        .execute_transaction(exec_request, skip_certification, None)
738        .await
739        .map_err(RpcError::from)?;
740
741    let digest = *effects.effects.transaction_digest();
742
743    // Build the merged response
744    let sdk_transaction: iota_sdk_types::Transaction = transaction.transaction().clone();
745    let signatures = transaction.signatures().to_owned();
746
747    // Keep a pre-parsed copy for the rebuild-from-cache path so it doesn't
748    // have to re-parse the proto request. Only materialised when the response
749    // carries uncertified single-validator data (skip-effect-cert path).
750    let rebuild_ctx = matches!(
751        effects.finality_info,
752        iota_types::quorum_driver_types::EffectsFinalityInfo::UncertifiedSingleValidator(_)
753    )
754    .then(|| RebuildCtx {
755        transaction: sdk_transaction.clone(),
756        signatures: signatures.clone(),
757    });
758
759    let source = TransactionReadSource {
760        reader: reader.clone(),
761        config,
762        transaction: Some(sdk_transaction),
763        signatures: Some(signatures),
764        effects: Some(effects.effects),
765        events,
766        checkpoint: None,
767        timestamp_ms: None,
768        input_objects,
769        output_objects,
770        mocked_coin: None,
771    };
772
773    let executed = ExecutedTransaction::merge_from(&source, read_mask)
774        .map_err(|e| e.with_context("failed to merge executed transaction"))?;
775
776    Ok((digest, executed, rebuild_ctx))
777}
