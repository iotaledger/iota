36/// Simulate a batch of transactions sequentially.
37///
38/// Each transaction is simulated independently — failure of one does not abort
39/// the rest. Results are returned in the same order as the input.
40///
41/// ## Available Read Mask Fields
42///
43/// The `simulate_transactions` function supports the following `read_mask`
44/// fields to control which data is included in each simulated transaction
45/// result:
46///
47/// ## Transaction Fields
48/// - `executed_transaction` - includes all executed transaction fields
49///   - `executed_transaction.transaction` - includes all transaction fields
50///     - `executed_transaction.transaction.digest` - the transaction digest
51///     - `executed_transaction.transaction.bcs` - the full BCS-encoded
52///       transaction
53///   - `executed_transaction.signatures` - includes all signature fields
54///     - `executed_transaction.signatures.bcs` - the full BCS-encoded signature
55///   - `executed_transaction.effects` - includes all effects fields
56///     - `executed_transaction.effects.digest` - the effects digest
57///     - `executed_transaction.effects.bcs` - the full BCS-encoded effects
58///   - `executed_transaction.events` - includes all event fields
59///     - `executed_transaction.events.digest` - the events digest
60///     - `executed_transaction.events.events` - includes all event fields (all
61///       events of the transaction)
62///       - `executed_transaction.events.events.bcs` - the full BCS-encoded
63///         event
64///       - `executed_transaction.events.events.package_id` - the ID of the
65///         package that emitted the event
66///       - `executed_transaction.events.events.module` - the module that
67///         emitted the event
68///       - `executed_transaction.events.events.sender` - the sender that
69///         triggered the event
70///       - `executed_transaction.events.events.event_type` - the type of the
71///         event
72///       - `executed_transaction.events.events.bcs_contents` - the full
73///         BCS-encoded contents of the event
74///       - `executed_transaction.events.events.json_contents` - the
75///         JSON-encoded contents of the event
76///   - `executed_transaction.checkpoint` - the checkpoint that included the
77///     transaction (not available for just-executed transactions)
78///   - `executed_transaction.timestamp` - the timestamp of the checkpoint (not
79///     available for just-executed transactions)
80///   - `executed_transaction.input_objects` - includes all input object fields
81///     - `executed_transaction.input_objects.reference` - includes all
82///       reference fields
83///       - `executed_transaction.input_objects.reference.object_id` - the ID of
84///         the input object
85///       - `executed_transaction.input_objects.reference.version` - the version
86///         of the input object
87///       - `executed_transaction.input_objects.reference.digest` - the digest
88///         of the input object contents
89///     - `executed_transaction.input_objects.bcs` - the full BCS-encoded object
90///   - `executed_transaction.output_objects` - includes all output object
91///     fields
92///     - `executed_transaction.output_objects.reference` - includes all
93///       reference fields
94///       - `executed_transaction.output_objects.reference.object_id` - the ID
95///         of the output object
96///       - `executed_transaction.output_objects.reference.version` - the
97///         version of the output object
98///       - `executed_transaction.output_objects.reference.digest` - the digest
99///         of the output object contents
100///     - `executed_transaction.output_objects.bcs` - the full BCS-encoded
101///       object
102///   - `executed_transaction.balance_changes` - per-owner, per-coin-type
103///     balance deltas derived from the simulated effects and input/output
104///     objects; a mocked gas coin is excluded. For a failed transaction this
105///     contains only the gas charge.
106///   - `executed_transaction.object_changes` - structured object changes
107///     (created, mutated, deleted, wrapped, unwrapped, published) derived from
108///     the simulated effects and input/output objects.
109///
110/// ## Gas Fields
111/// - `suggested_gas_price` - the suggested gas price for the transaction,
112///   denominated in NANOS
113///
114/// ## Execution Result Fields
115/// - `execution_result` - the execution result (oneof: command_results on
116///   success, execution_error on failure)
117///   - `execution_result.command_results` - includes all fields of per-command
118///     results if execution succeeded
119///     - `execution_result.command_results.mutated_by_ref` - includes all
120///       fields of objects mutated by reference
121///       - `execution_result.command_results.mutated_by_ref.argument` - the
122///         argument reference
123///       - `execution_result.command_results.mutated_by_ref.type_tag` - the
124///         Move type tag
125///       - `execution_result.command_results.mutated_by_ref.bcs` - the
126///         BCS-encoded value
127///       - `execution_result.command_results.mutated_by_ref.json` - the
128///         JSON-encoded value
129///     - `execution_result.command_results.return_values` - includes all fields
130///       of return values returned by the command
131///       - `execution_result.command_results.return_values.argument` - the
132///         argument reference
133///       - `execution_result.command_results.return_values.type_tag` - the Move
134///         type tag
135///       - `execution_result.command_results.return_values.bcs` - the
136///         BCS-encoded value
137///       - `execution_result.command_results.return_values.json` - the
138///         JSON-encoded value
139///   - `execution_result.execution_error` - includes all fields of the
140///     execution error if execution failed
141///     - `execution_result.execution_error.bcs_kind` - the BCS-encoded error
142///       kind
143///     - `execution_result.execution_error.source` - the error source
144///       description
145///     - `execution_result.execution_error.command_index` - the index of the
146///       command that failed
147#[tracing::instrument(skip_all, fields(batch_size = request.transactions.len()))]
148pub async fn simulate_transactions(
149    reader: &Arc<GrpcReader>,
150    executor: &Arc<dyn TransactionExecutor>,
151    config: &iota_config::node::GrpcApiConfig,
152    request: SimulateTransactionsRequest,
153) -> Result<SimulateTransactionsResponse, RpcError> {
154    super::validate_batch_size(
155        request.transactions.len(),
156        config.max_simulate_transaction_batch_size as usize,
157    )?;
158    let read_mask = validate_read_mask::<SimulatedTransaction>(
159        request.read_mask,
160        SIMULATE_TRANSACTIONS_READ_MASK,
161    )?;
162
163    // Simulate each transaction sequentially, collecting per-item results
164    let mut transaction_results = Vec::with_capacity(request.transactions.len());
165    for item in &request.transactions {
166        let result =
167            match simulate_single_transaction(reader, executor, config, item, &read_mask).await {
168                Ok(tx) => SimulateTransactionResult::default().with_simulated_transaction(tx),
169                Err(error) => {
170                    SimulateTransactionResult::default().with_error(error.into_status_proto())
171                }
172            };
173        transaction_results.push(result);
174    }
175
176    Ok(SimulateTransactionsResponse::default().with_transaction_results(transaction_results))
177}

179/// Simulate a single transaction item.
180async fn simulate_single_transaction(
181    reader: &Arc<GrpcReader>,
182    executor: &Arc<dyn TransactionExecutor>,
183    config: &iota_config::node::GrpcApiConfig,
184    item: &SimulateTransactionItem,
185    read_mask: &FieldMaskTree,
186) -> Result<SimulatedTransaction, RpcError> {
187    let mut transaction_data = super::parse_transaction_proto(item.transaction.as_ref())?;
188
189    // Determine VM checks from request
190    let vm_checks = if item
191        .tx_checks
192        .contains(&(TransactionCheckModes::DisableVmChecks as i32))
193    {
194        VmChecks::Disabled
195    } else {
196        VmChecks::Enabled
197    };
198
199    let system_state = if read_mask.contains(SimulatedTransaction::SUGGESTED_GAS_PRICE_FIELD.name) {
200        Some(reader.get_system_state_summary().map_err(|e| {
201            RpcError::new(
202                tonic::Code::Internal,
203                format!("failed to get system state: {e}"),
204            )
205        })?)
206    } else {
207        None
208    };
209
210    // Simulate the transaction
211    let InternalSimulateResult {
212        effects,
213        events,
214        input_objects,
215        output_objects,
216        execution_result,
217        mock_gas_id,
218        suggested_gas_price,
219        gas_data,
220    } = executor
221        .simulate_transaction(transaction_data.clone(), vm_checks)
222        .map_err(|e| {
223            RpcError::new(
224                if matches!(e, IotaError::UserInput { .. }) {
225                    tonic::Code::InvalidArgument
226                } else {
227                    tonic::Code::Internal
228                },
229                format!("transaction simulation failed: {e}"),
230            )
231        })?;
232
233    // Build the response
234    let mut response = SimulatedTransaction::default();
235
236    // Only include transaction if requested
237    if let Some(tx_mask) = read_mask.subtree(SimulatedTransaction::EXECUTED_TRANSACTION_FIELD.name)
238    {
239        // Report the transaction the simulation ran, gas payment included, so it
240        // hashes to the digest the effects below are keyed by. Same rule as the
241        // JSON-RPC dry run, which shares the helper.
242        iota_types::gas::report_simulation_gas(
243            transaction_data.gas_data_mut(),
244            &gas_data,
245            effects.gas_cost_summary().gas_used(),
246        );
247
248        // Create a source for the merge
249        let source = TransactionReadSource {
250            reader: reader.clone(),
251            config,
252            transaction: Some(transaction_data),
253            signatures: None,
254            effects: Some(effects),
255            events,
256            checkpoint: None,
257            timestamp_ms: None,
258            input_objects: Some(input_objects.into_values().collect()),
259            output_objects: Some(output_objects.into_values().collect()),
260            mocked_coin: mock_gas_id,
261        };
262
263        response.executed_transaction = Some(
264            ExecutedTransaction::merge_from(&source, &tx_mask)
265                .map_err(|e| e.with_context("failed to merge executed transaction"))?,
266        );
267    }
268
269    // Only include suggested gas price if requested
270    if read_mask.contains(SimulatedTransaction::SUGGESTED_GAS_PRICE_FIELD.name) {
271        response.suggested_gas_price = Some(suggested_gas_price.unwrap_or_else(|| {
272            system_state
273                .as_ref()
274                .expect("system state should be available")
275                .reference_gas_price()
276        }));
277    }
278
279    // Only include the result if requested
280    if let Some(result_mask) = read_mask.subtree(SimulatedTransaction::EXECUTION_RESULT_ONEOF) {
281        match execution_result {
282            Ok(ref execution_results) => {
283                if let Some(command_results_mask) =
284                    result_mask.subtree(SimulatedTransaction::COMMAND_RESULTS_FIELD.name)
285                {
286                    // Only build command results if the execution was successful
287                    let cmd_source = CommandResultsReadSource {
288                        reader: reader.clone(),
289                        config,
290                        execution_results: execution_results.clone(),
291                    };
292
293                    let command_results =
294                        CommandResults::merge_from(&cmd_source, &command_results_mask)
295                            .map_err(|e| e.with_context("failed to merge command results"))?;
296
297                    response.execution_result = Some(
298                        iota_grpc_types::v1::transaction_execution_service::simulated_transaction::ExecutionResult::CommandResults(command_results),
299                    );
300                }
301            }
302            Err(ref execution_error) => {
303                if let Some(error_mask) =
304                    result_mask.subtree(SimulatedTransaction::EXECUTION_ERROR_FIELD.name)
305                {
306                    let mut exec_error = ExecutionError::default();
307
308                    // Serialize the execution error kind as BCS
309                    if error_mask.contains(ExecutionError::BCS_KIND_FIELD.name) {
310                        exec_error.bcs_kind = Some(
311                            grpc_bcs::BcsData::serialize(execution_error.kind()).map_err(|e| {
312                                RpcError::new(
313                                    tonic::Code::Internal,
314                                    format!("failed to serialize execution error kind: {e}"),
315                                )
316                            })?,
317                        );
318                    }
319
320                    if error_mask.contains(ExecutionError::SOURCE_FIELD.name) {
321                        exec_error.source = execution_error
322                            .source()
323                            .as_ref()
324                            .map(|source| source.to_string());
325                    }
326
327                    // Set the command index if available
328                    if error_mask.contains(ExecutionError::COMMAND_INDEX_FIELD.name) {
329                        if let Some(command_idx) = execution_error.command() {
330                            exec_error.command_index = Some(command_idx);
331                        }
332                    }
333
334                    response.execution_result = Some(
335                        iota_grpc_types::v1::transaction_execution_service::simulated_transaction::ExecutionResult::ExecutionError(exec_error),
336                    );
337                }
338            }
339        }
340    }
341
342    Ok(response)
