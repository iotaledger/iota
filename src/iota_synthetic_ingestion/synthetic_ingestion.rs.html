20/// Configuration for generating synthetic checkpoint data for for benchmarking
21/// or testing ingestion processes. Defines the behavior of synthetic data
22/// generation, such as checkpoint size, number of checkpoints, and the
23/// directory for storing generated checkpoint data.
24#[derive(Parser, Debug, Clone)]
25pub struct Config {
26    /// Directory to write the ingestion data to.
27    #[arg(long)]
28    pub ingestion_dir: PathBuf,
29    /// Starting checkpoint sequence number for workload generation.
30    /// Useful for benchmarking or testing against a non-empty database.
31    #[arg(long, default_value_t = Self::DEFAULT_STARTING_CHECKPOINT)]
32    pub starting_checkpoint: u64,
33    /// Number of checkpoints to generate.
34    /// If `starting_checkpoint` is 0 (default), two additional initial
35    /// checkpoints are generated:
36    /// - `0.chk`: genesis state
37    /// - `1.chk`: initial gas provisioning
38    ///
39    /// Thus, the total number of generated checkpoint files will be
40    /// `num_checkpoints + 2`. Otherwise, exactly `num_checkpoints`
41    /// checkpoints are generated.
42    #[arg(long, default_value_t = Self::DEFAULT_NUM_CHECKPOINTS)]
43    pub num_checkpoints: u64,
44    /// Number of transactions in a checkpoint.
45    #[arg(long, default_value_t = Self::DEFAULT_CHECKPOINT_SIZE)]
46    pub checkpoint_size: u64,
47}
48
49impl Config {
50    const DEFAULT_STARTING_CHECKPOINT: u64 = 0;
51    const DEFAULT_NUM_CHECKPOINTS: u64 = 2000;
52    const DEFAULT_CHECKPOINT_SIZE: u64 = 200;
53}

55/// Generates synthetic checkpoint data based on the provided configuration.
56// TODO: Simulacrum does serial execution which could be slow if
57// we need to generate a large number of transactions.
58// We may want to make Simulacrum support parallel execution.
59pub async fn generate_ingestion(config: Config) -> Result<()> {
60    info!("Generating synthetic ingestion data. config: {:?}", config);
61    let timer = std::time::Instant::now();
62
63    let Config {
64        ingestion_dir,
65        checkpoint_size,
66        num_checkpoints,
67        starting_checkpoint,
68    } = config;
69
70    // Simulacrum will generate `0.chk` as the genesis checkpoint.
71    let sim = Simulacrum::new();
72    sim.set_data_ingestion_path(ingestion_dir.clone());
73
74    let gas_price = sim.reference_gas_price();
75    let (sender, sender_key) = get_account_private_key();
76    let mut gas_object = {
77        let effects = sim.request_gas(sender, NANOS_PER_IOTA * 1000000)?;
78        // Generate `1.chk` and includes the gas request transaction.
79        sim.create_checkpoint();
80        effects.created()[0].reference
81    };
82
83    // When generating a workload that includes the genesis state, retain the
84    // initial checkpoints `0.chk` (genesis state) and `1.chk` (initial gas
85    // provisioning) to accurately represent the state history.
86    // For starting_checkpoint > 0, remove existing checkpoints (0 and 1)
87    // and generate a consistent workload.
88    if starting_checkpoint > 0 {
89        fs::remove_file(ingestion_dir.join("0.chk")).await?;
90        fs::remove_file(ingestion_dir.join("1.chk")).await?;
91        sim.override_next_checkpoint_number(starting_checkpoint);
92    }
93
94    let mut tx_count = 0;
95    for i in 0..num_checkpoints {
96        for _ in 0..checkpoint_size {
97            let tx_data = TestTransactionBuilder::new(sender, gas_object, gas_price)
98                .transfer_iota(Some(1), sender)
99                .build();
100            let tx = to_sender_signed_transaction(tx_data, &sender_key);
101            let (effects, _) = sim.execute_transaction(tx)?;
102            gas_object = effects.gas_object().reference;
103            tx_count += 1;
104        }
105
106        let checkpoint = sim.create_checkpoint();
107
108        let expected_checkpoint_number = if starting_checkpoint == 0 {
109            i + 2 // offset by 2 because of the auto-generated `0.chk` and `1.chk` files
110        } else {
111            i + starting_checkpoint
112        };
113
114        assert_eq!(checkpoint.sequence_number, expected_checkpoint_number);
115
116        if (i + 1) % 100 == 0 {
117            info!("Generated {} checkpoints, {tx_count} transactions", i + 1);
118        }
119    }
120
121    info!(
122        "Synthetic ingestion generation completed: {num_checkpoints} checkpoints generated (excluding genesis and gas funding), {tx_count} transactions in {:.2?}.",
123        timer.elapsed()
124    );
125
126    Ok(())
127}

129/// Reads serialized, synthetic checkpoint data from disk into memory.
130pub async fn read_ingestion_data(path: &PathBuf) -> anyhow::Result<BTreeMap<u64, CheckpointData>> {
131    let mut data = BTreeMap::new();
132    let mut dir = fs::read_dir(path).await?;
133    while let Some(entry) = dir.next_entry().await? {
134        let path = entry.path();
135        let bytes = fs::read(path).await?;
136        let checkpoint_data: CheckpointData = Blob::from_bytes(&bytes)?;
137        data.insert(
138            checkpoint_data.checkpoint_summary.sequence_number,
139            checkpoint_data,
140        );
141    }
142    Ok(data)
143}

145#[cfg(test)]
146mod tests {
147    use std::path::Path;
148
149    use iota_storage::blob::Blob;
150    use iota_types::full_checkpoint_content::CheckpointData;
151
152    use crate::synthetic_ingestion::generate_ingestion;
153
154    #[tokio::test]
155    async fn test_ingestion_from_zero() {
156        let tmp_dir = iota_common::tempdir();
157        let ingestion_dir = tmp_dir.path().to_path_buf();
158
159        let config = super::Config {
160            ingestion_dir: ingestion_dir.clone(),
161            starting_checkpoint: 0,
162            num_checkpoints: 10,
163            checkpoint_size: 2,
164        };
165
166        generate_ingestion(config).await.unwrap();
167
168        // Check for the genesis checkpoint (0.chk)
169        check_checkpoint_data(&ingestion_dir, 0, 1, 1).await;
170        // Check for the gas funding checkpoint (1.chk)
171        check_checkpoint_data(&ingestion_dir, 1, 1, 1).await;
172        // Rest of the checkpoints
173        check_checkpoint_data(&ingestion_dir, 2, 10, 2).await;
174    }
175
176    #[tokio::test]
177    async fn test_ingestion_from_non_zero() {
178        let tmp_dir = iota_common::tempdir();
179        let ingestion_dir = tmp_dir.path().to_path_buf();
180
181        let config = super::Config {
182            ingestion_dir: ingestion_dir.clone(),
183            starting_checkpoint: 10,
184            num_checkpoints: 10,
185            checkpoint_size: 2,
186        };
187
188        generate_ingestion(config).await.unwrap();
189        check_checkpoint_data(&ingestion_dir, 10, 10, 2).await;
190    }
191
192    async fn check_checkpoint_data(
193        ingestion_dir: &Path,
194        first_checkpoint: u64,
195        num_checkpoints: u64,
196        checkpoint_size: u64,
197    ) {
198        for checkpoint in first_checkpoint..first_checkpoint + num_checkpoints {
199            let path = ingestion_dir.join(format!("{checkpoint}.chk"));
200            let bytes = tokio::fs::read(&path).await.unwrap();
201            let checkpoint_data: CheckpointData = Blob::from_bytes(&bytes).unwrap();
202
203            assert_eq!(
204                checkpoint_data.checkpoint_summary.sequence_number,
205                checkpoint
206            );
207            assert_eq!(checkpoint_data.transactions.len(), checkpoint_size as usize);
208        }
209    }
