4//! Sampling routines whose results depend only on the bytes an RNG produces,
5//! never on which version of `rand` is in the dependency graph.
6//!
7//! `rand`'s own `shuffle`, `choose`, `choose_multiple` and
8//! `choose_multiple_weighted` are free to change between releases, and they
9//! have: 0.9 replaced the range sampler and the weighted-selection wrapper, so
10//! the same seed produces a different permutation. That is fine for a test
11//! fixture and unacceptable for a permutation every validator has to agree on.
12//!
13//! The routines here reproduce `rand` 0.8's algorithms and are fixed by
14//! tests against captured vectors, so callers that need a permutation to
15//! stay the same forever can bump `rand` freely.
16//!
17//! Two rules for callers:
18//!
19//! - Seed [`rand::rngs::ChaCha12Rng`], not `StdRng`. `StdRng` is documented as
20//!   non-portable — `rand` reserves the right to swap the algorithm under it —
21//!   whereas `ChaCha12Rng` names the generator and is what `StdRng` currently
22//!   happens to be.
23//! - Draw floats through [`f64_unit`] rather than `rand`'s `random::<f64>()`,
24//!   which is another conversion `rand` owns and could change.
25
26use rand::Rng;
27
28/// Draws a `f64` uniformly from `[0, 1)`, using the top 53 bits of one 64-bit
29/// word.
30pub fn f64_unit<R: Rng + ?Sized>(rng: &mut R) -> f64 {
31    const SCALE: f64 = 1.0 / ((1u64 << 53) as f64);
32    SCALE * ((rng.next_u64() >> 11) as f64)
33}
34
35/// Draws a `u32` uniformly from `low..=high`.
36///
37/// Lemire's multiply-shift with rejection: multiply a random word by the range
38/// width and keep the high half, rejecting the low half's biased tail.
39fn uniform_u32_inclusive<R: Rng + ?Sized>(rng: &mut R, low: u32, high: u32) -> u32 {
40    let range = high.wrapping_sub(low).wrapping_add(1);
41    if range == 0 {
42        // `low..=high` spans the whole of u32, so every word is in range.
43        return rng.next_u32();
44    }
45    let zone = (range << range.leading_zeros()).wrapping_sub(1);
46    loop {
47        let product = u64::from(rng.next_u32()) * u64::from(range);
48        let (hi, lo) = ((product >> 32) as u32, product as u32);
49        if lo <= zone {
50            return low.wrapping_add(hi);
51        }
52    }
53}

55/// Draws an index uniformly from `0..ubound`.
56///
57/// Sampling is done in 32 bits so that the result does not depend on the width
58/// of `usize`. Panics if `ubound` is 0 or exceeds `u32::MAX`.
59pub fn index<R: Rng + ?Sized>(rng: &mut R, ubound: usize) -> usize {
60    assert!(ubound > 0, "cannot draw an index from an empty range");
61    assert!(
62        ubound <= u32::MAX as usize,
63        "index sampling is limited to u32::MAX to stay independent of usize width"
64    );
65    uniform_u32_inclusive(rng, 0, ubound as u32 - 1) as usize
66}

68/// Permutes `slice` in place (Fisher-Yates, from the back).
69pub fn shuffle<T, R: Rng + ?Sized>(rng: &mut R, slice: &mut [T]) {
70    for i in (1..slice.len()).rev() {
71        slice.swap(i, index(rng, i + 1));
72    }
73}
74
75/// Returns a uniformly drawn element, or `None` if `slice` is empty.
76pub fn choose<'a, T, R: Rng + ?Sized>(rng: &mut R, slice: &'a [T]) -> Option<&'a T> {
77    if slice.is_empty() {
78        None
79    } else {
80        Some(&slice[index(rng, slice.len())])
81    }
82}

84/// Draws `amount` distinct indices from `0..length`, in random order.
85///
86/// Which of the two algorithms runs is decided by `length` and `amount` alone,
87/// so it is as reproducible as the draws themselves. Panics if `amount`
88/// exceeds `length`, or if the pair falls in the range where `rand` 0.8 used a
89/// third algorithm — unreachable for `amount == length`, which is how every
90/// caller here uses it, and better to fail loudly than to diverge quietly.
91pub fn sample_indices<R: Rng + ?Sized>(rng: &mut R, length: usize, amount: usize) -> Vec<usize> {
92    assert!(
93        amount <= length,
94        "cannot sample more indices than are available"
95    );
96    assert!(
97        length <= u32::MAX as usize,
98        "index sampling is limited to u32::MAX to stay independent of usize width"
99    );
100    let (amount, length) = (amount as u32, length as u32);
101
102    // Thresholds carried over verbatim from `rand` 0.8; they trade set-up cost
103    // against draw count and have no meaning beyond picking the cheaper
104    // algorithm.
105    if amount < 163 {
106        let scale = if length < 500_000 {
107            (1.6, 10.0)
108        } else {
109            (8.0 / 45.0, 70.0 / 9.0)
110        };
111        let amount_fp = amount as f32;
112        if amount > 11 && (length as f32) < (scale.1 + scale.0 * amount_fp) * amount_fp {
113            sample_in_place(rng, length, amount)
114        } else {
115            sample_floyd(rng, length, amount)
116        }
117    } else {
118        let scale = if length < 500_000 { 270.0 } else { 330.0 / 9.0 };
119        assert!(
120            (length as f32) < scale * (amount as f32),
121            "sampling {amount} of {length} indices needs the algorithm this module does not carry"
122        );
123        sample_in_place(rng, length, amount)
124    }
125}

127/// Floyd's combination algorithm, fully shuffled.
128fn sample_floyd<R: Rng + ?Sized>(rng: &mut R, length: u32, amount: u32) -> Vec<usize> {
129    // Floyd's insert keeps the result shuffled as it goes, but `Vec::insert`
130    // gets expensive, so past this size the shuffle is done in one pass after.
131    let shuffle_while_inserting = amount < 50;
132
133    let mut indices: Vec<u32> = Vec::with_capacity(amount as usize);
134    for j in length - amount..length {
135        let t = uniform_u32_inclusive(rng, 0, j);
136        if shuffle_while_inserting {
137            if let Some(pos) = indices.iter().position(|&x| x == t) {
138                indices.insert(pos, j);
139                continue;
140            }
141        } else if indices.contains(&t) {
142            indices.push(j);
143            continue;
144        }
145        indices.push(t);
146    }
147    if !shuffle_while_inserting {
148        for i in (1..amount).rev() {
149            let j = uniform_u32_inclusive(rng, 0, i) as usize;
150            indices.swap(i as usize, j);
151        }
152    }
153    indices.into_iter().map(|i| i as usize).collect()
154}

156/// Partial Fisher-Yates over the whole index range.
157fn sample_in_place<R: Rng + ?Sized>(rng: &mut R, length: u32, amount: u32) -> Vec<usize> {
158    let mut indices: Vec<u32> = (0..length).collect();
159    for i in 0..amount {
160        let j = uniform_u32_inclusive(rng, i, length - 1);
161        indices.swap(i as usize, j as usize);
162    }
163    indices.truncate(amount as usize);
164    indices.into_iter().map(|i| i as usize).collect()
165}

167/// Draws `amount` distinct indices from `0..length` with probability
168/// proportional to weight, in descending order of the keys drawn.
169///
170/// Efraimidis and Spirakis, <https://doi.org/10.1016/j.ipl.2005.11.003>: give
171/// index `i` the key `u^(1/w_i)` for a uniform `u`, then take the largest keys.
172/// Passing `amount == length` therefore yields a weighted permutation of every
173/// index.
174///
175/// Panics if any weight is negative or not a number.
176pub fn sample_weighted<R, F>(rng: &mut R, length: usize, weight: F, amount: usize) -> Vec<usize>
177where
178    R: Rng + ?Sized,
179    F: Fn(usize) -> f64,
180{
181    let amount = amount.min(length);
182    let mut keyed: Vec<(f64, usize)> = (0..length)
183        .map(|i| {
184            let w = weight(i);
185            assert!(
186                w >= 0.0,
187                "weights must be non-negative and not NaN, got {w}"
188            );
189            (f64_unit(rng).powf(1.0 / w), i)
190        })
191        .collect();
192    // Keys are drawn from a CSPRNG, so ties do not arise in practice; comparing
193    // the index as well keeps the order total regardless.
194    keyed.sort_by(|a, b| b.0.partial_cmp(&a.0).unwrap().then(a.1.cmp(&b.1)));
195    keyed.into_iter().take(amount).map(|(_, i)| i).collect()
196}

198#[cfg(test)]
199mod tests {
200    use rand::{SeedableRng, rngs::ChaCha12Rng};
201
202    use super::*;
203
204    fn rng(seed: u8) -> ChaCha12Rng {
205        ChaCha12Rng::from_seed([seed; 32])
206    }
207
208    /// Vectors captured from `rand` 0.8, whose algorithms these routines
209    /// reproduce. They are the permutations consensus already agreed on, so a
210    /// change here is a change to the protocol, not a test to re-record.
211    #[test]
212    fn matches_captured_vectors() {
213        let mut v: Vec<u32> = (0..10).collect();
214        shuffle(&mut rng(0), &mut v);
215        assert_eq!(v, [6, 8, 3, 0, 1, 9, 2, 5, 7, 4]);
216
217        let mut v: Vec<u32> = (0..40).collect();
218        shuffle(&mut rng(7), &mut v);
219        assert_eq!(v[..8], [13, 36, 32, 11, 39, 30, 6, 24]);
220
221        let items: Vec<u32> = (0..10).collect();
222        assert_eq!(choose(&mut rng(0), &items), Some(&4));
223        assert_eq!(choose(&mut rng(1), &items), Some(&8));
224
225        // Under 12 indices draws Floyd's, over it draws partial Fisher-Yates,
226        // and 50 crosses Floyd's shuffle-as-you-go cutoff.
227        assert_eq!(
228            sample_indices(&mut rng(0), 10, 10),
229            [1, 8, 4, 9, 7, 6, 3, 0, 5, 2]
230        );
231        assert_eq!(
232            sample_indices(&mut rng(0), 20, 20),
233            [
234                8, 7, 3, 17, 5, 14, 11, 13, 2, 6, 0, 1, 18, 12, 4, 16, 19, 15, 10, 9
235            ]
236        );
237        assert_eq!(
238            sample_indices(&mut rng(3), 63, 63)[..8],
239            [32, 62, 58, 27, 54, 2, 56, 26]
240        );
241
242        let weights = [1u64, 5, 2, 8, 3];
243        assert_eq!(
244            sample_weighted(
245                &mut rng(0),
246                weights.len(),
247                |i| weights[i] as f64,
248                weights.len()
249            ),
250            [1, 4, 3, 0, 2]
251        );
252        assert_eq!(
253            sample_weighted(&mut rng(9), weights.len(), |i| weights[i] as f64, 2),
254            [1, 0]
255        );
256    }

258    #[test]
259    fn f64_unit_lies_in_the_unit_interval() {
260        let mut gen = rng(0);
261        for _ in 0..10_000 {
262            let x = f64_unit(&mut gen);
263            assert!((0.0..1.0).contains(&x), "{x} outside [0, 1)");
264        }
265        // The conversion `rand` performs, for as long as both exist.
266        assert_eq!(
267            f64_unit(&mut rng(4)),
268            rand::RngExt::random::<f64>(&mut rng(4))
269        );
270    }

272    #[test]
273    fn sampling_every_index_yields_a_permutation() {
274        for len in [1usize, 2, 11, 12, 13, 49, 50, 51, 162, 163, 200] {
275            for seed in 0..4 {
276                let mut got = sample_indices(&mut rng(seed), len, len);
277                got.sort_unstable();
278                assert_eq!(got, (0..len).collect::<Vec<_>>(), "len {len} seed {seed}");
279
280                let mut got = sample_weighted(&mut rng(seed), len, |i| (i % 7 + 1) as f64, len);
281                got.sort_unstable();
282                assert_eq!(got, (0..len).collect::<Vec<_>>(), "len {len} seed {seed}");
283            }
284        }
285    }

287    #[test]
288    fn zero_weight_sorts_last() {
289        let weights = [0u64, 4, 4, 4];
290        let order = sample_weighted(&mut rng(2), weights.len(), |i| weights[i] as f64, 4);
291        assert_eq!(order.last(), Some(&0));
