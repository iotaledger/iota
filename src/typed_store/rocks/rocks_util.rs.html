5use crate::{
6    rocks::options::ReadWriteOptions,
7    util::{big_endian_add_one, is_max},

10/// Read options for an iterator over `[lower_bound, upper_bound)`, both
11/// already serialized. A reverse iterator gets only the lower bound, since
12/// its initial seek to the upper bound enforces that one.
13///
14/// With a prefix extractor installed, the iterator uses the prefix bloom
15/// filters only when the scan provably stays within one prefix, and seeks in
16/// total order otherwise.
17pub(crate) fn iterator_read_options(
18    opts: &ReadWriteOptions,
19    lower_bound: Option<Vec<u8>>,
20    upper_bound: Option<Vec<u8>>,
21    reverse: bool,
22) -> rocksdb::ReadOptions {
23    let mut read_options = opts.readopts();
24    if let Some(prefix_len) = opts.prefix_len {
25        let within_prefix = scan_stays_within_prefix(
26            prefix_len,
27            lower_bound.as_deref(),
28            upper_bound.as_deref(),
29            reverse,
30        );
31        read_options.set_total_order_seek(!within_prefix);
32    }
33    let upper_bound = if reverse { None } else { upper_bound };
34    apply_range_bounds(read_options, lower_bound, upper_bound)
35}
36
37/// Whether every key in `[lower_bound, upper_bound)` shares its first
38/// `prefix_len` bytes with the key the scan seeks to. An upper bound that is
39/// exactly the next prefix ends a forward scan at the end of the prefix, so
40/// it qualifies there, but not for a reverse scan, which seeks to it.
41pub(crate) fn scan_stays_within_prefix(
42    prefix_len: usize,
43    lower_bound: Option<&[u8]>,
44    upper_bound: Option<&[u8]>,
45    reverse: bool,
46) -> bool {
47    let (Some(lower), Some(upper)) = (lower_bound, upper_bound) else {
48        return false;
49    };
50    if lower.len() < prefix_len {
51        return false;
52    }
53    let prefix = &lower[..prefix_len];
54    if upper.len() >= prefix_len && &upper[..prefix_len] == prefix {
55        return true;
56    }
57    if reverse || is_max(prefix) {
58        return false;
59    }
60    let mut next_prefix = prefix.to_vec();
61    big_endian_add_one(&mut next_prefix);
62    upper == next_prefix.as_slice()
63}
64
65pub(crate) fn apply_range_bounds(
66    mut read_options: rocksdb::ReadOptions,
67    lower_bound: Option<Vec<u8>>,
68    upper_bound: Option<Vec<u8>>,
69) -> rocksdb::ReadOptions {
70    if let Some(lower_bound) = lower_bound {
71        read_options.set_iterate_lower_bound(lower_bound);
72    }
73    if let Some(upper_bound) = upper_bound {
74        read_options.set_iterate_upper_bound(upper_bound);
75    }
76    read_options
77}
