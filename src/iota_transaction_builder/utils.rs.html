34    /// Select a gas coin for the provided gas budget.
35    pub async fn select_gas(
36        &self,
37        signer: Address,
38        input_gas: impl Into<Option<ObjectId>>,
39        gas_budget: u64,
40        input_objects: Vec<ObjectId>,
41        gas_price: u64,
42    ) -> Result<ObjectReference, anyhow::Error> {
43        if gas_budget < gas_price {
44            bail!(
45                "Gas budget {gas_budget} is less than the reference gas price {gas_price}. The gas budget must be at least the current reference gas price of {gas_price}."
46            )
47        }
48        if let Some(gas) = input_gas.into() {
49            self.get_object_ref(gas).await
50        } else {
51            if let Some(gas) = self
52                .0
53                .select_gas_coin(signer, gas_budget, &input_objects)
54                .await?
55            {
56                return Ok(gas);
57            }
58            Err(anyhow!(
59                "Cannot find gas coin for signer address {signer} with amount sufficient for the required gas budget {gas_budget}. If you are using the pay or transfer commands, you can use the pay-iota command instead, which will use the only object as gas payment."
60            ))
61        }
62    }

64    /// Get the object references for a list of object IDs
65    pub async fn input_refs(
66        &self,
67        obj_ids: &[ObjectId],
68    ) -> Result<Vec<ObjectReference>, anyhow::Error> {
69        let handles: Vec<_> = obj_ids.iter().map(|id| self.get_object_ref(*id)).collect();
70        let obj_refs = join_all(handles)
71            .await
72            .into_iter()
73            .collect::<anyhow::Result<Vec<ObjectReference>>>()?;
74        Ok(obj_refs)
75    }

77    /// Resolve a provided [`ObjectId`] to the required [`CallArg`] for a
78    /// given move module.
79    async fn get_object_arg(
80        &self,
81        id: ObjectId,
82        is_mutable_ref: bool,
83        view: &CompiledModule,
84        arg_type: &SignatureToken,
85    ) -> Result<CallArg, anyhow::Error> {
86        let response = self
87            .0
88            .get_object_with_options(id, IotaObjectDataOptions::bcs_lossless())
89            .await?;
90
91        let obj: Object = response.into_object()?.try_into()?;
92        let obj_ref = obj.object_ref();
93        let owner = obj.owner;
94        if is_receiving_argument(view, arg_type) {
95            return Ok(CallArg::Receiving(obj_ref));
96        }
97        Ok(match owner {
98            Owner::Shared(initial_shared_version) => CallArg::Shared(SharedObjectReference::new(
99                id,
100                initial_shared_version,
101                is_mutable_ref,
102            )),
103            Owner::Address(_) | Owner::Object(_) | Owner::Immutable => {
104                CallArg::ImmutableOrOwned(obj_ref)
105            }
106            _ => unimplemented!("a new Owner enum variant was added and needs to be handled"),
107        })
108    }

110    /// Resolve a [`ResolvedCallArg`] to a [`CallArg`] or a list of
111    /// [`CallArg`] for object vectors.
112    async fn resolved_call_arg_to_call_arg(
113        &self,
114        resolved_arg: ResolvedCallArg,
115        param: &SignatureToken,
116        module: &CompiledModule,
117    ) -> Result<ResolvedCallArgResult, anyhow::Error> {
118        match resolved_arg {
119            ResolvedCallArg::Pure(bytes) => {
120                Ok(ResolvedCallArgResult::CallArg(CallArg::Pure(bytes)))
121            }
122            ResolvedCallArg::Object(id) => {
123                let is_mutable =
124                    matches!(param, SignatureToken::MutableReference(_)) || !param.is_reference();
125                let object_arg = self.get_object_arg(id, is_mutable, module, param).await?;
126                Ok(ResolvedCallArgResult::CallArg(object_arg))
127            }
128            ResolvedCallArg::ObjVec(vec_ids) => {
129                let mut object_args = Vec::with_capacity(vec_ids.len());
130                for id in vec_ids {
131                    object_args.push(self.get_object_arg(id, false, module, param).await?);
132                }
133                Ok(ResolvedCallArgResult::ObjVec(object_args))
134            }
135        }
136    }

138    /// Resolve a single JSON value to a [`ResolvedCallArgResult`].
139    async fn resolve_json_value_to_call_arg(
140        &self,
141        module: &CompiledModule,
142        type_args: &[TypeTag],
143        json_arg: IotaJsonValue,
144        param: &SignatureToken,
145        idx: usize,
146    ) -> Result<ResolvedCallArgResult, anyhow::Error> {
147        let arg = json_arg.into();
148        let arg_slice = std::slice::from_ref(&arg);
149        let param_slice = std::slice::from_ref(param);
150        let resolved = resolve_call_args(module, type_args, arg_slice, param_slice)?;
151        let resolved_arg = resolved
152            .into_iter()
153            .next()
154            .ok_or_else(|| anyhow!("Unable to resolve argument at index {idx}"))?;
155        self.resolved_call_arg_to_call_arg(resolved_arg, param, module)
156            .await
157    }

159    /// Convert provided JSON arguments for a move function to their
160    /// [`Argument`] representation and check their validity.
161    pub async fn resolve_and_checks_json_args(
162        &self,
163        builder: &mut ProgrammableTransactionBuilder,
164        package_id: ObjectId,
165        module_ident: &Identifier,
166        function_ident: &Identifier,
167        type_args: &[TypeTag],
168        json_args: Vec<IotaJsonValue>,
169    ) -> Result<Vec<Argument>, anyhow::Error> {
170        // Fetch the move package for the given package ID.
171        let package = self.fetch_move_package(package_id).await?;
172        let module = package.deserialize_module(module_ident, &BinaryConfig::standard())?;
173
174        // Then resolve the function parameters type.
175        let json_args_and_tokens = resolve_move_function_args(
176            &module,
177            function_ident,
178            type_args,
179            json_args.into_iter().map(Into::into).collect(),
180            false,
181        )
182        .map_err(|err| {
183            let package_id = &package.id;
184            anyhow!(
185                "Failed to resolve {package_id}::{module_ident}::{function_ident} move function: {err}"
186            )
187        })?;
188
189        // Finally construct the input arguments for the builder.
190        let mut args = Vec::new();
191        for (arg, expected_type) in json_args_and_tokens {
192            let result = self
193                .resolved_call_arg_to_call_arg(arg, &expected_type, &module)
194                .await?;
195            args.push(match result {
196                ResolvedCallArgResult::CallArg(call_arg) => builder.input(call_arg)?,
197                ResolvedCallArgResult::ObjVec(object_args) => builder.make_obj_vec(object_args)?,
198            });
199        }
200
201        Ok(args)
202    }

204    /// Convert provided PtbInput's for a move function to their
205    /// [`Argument`] representation and check their validity.
206    pub async fn resolve_and_check_call_args(
207        &self,
208        builder: &mut ProgrammableTransactionBuilder,
209        package_id: ObjectId,
210        module: &Identifier,
211        function: &Identifier,
212        type_args: &[TypeTag],
213        call_args: Vec<PtbInput>,
214    ) -> Result<Vec<Argument>, anyhow::Error> {
215        let package = self.fetch_move_package(package_id).await?;
216        let module_compiled = package.deserialize_module(module, &BinaryConfig::standard())?;
217        let parameters = get_function_parameters(&module_compiled, function)?;
218        let expected_len = expected_arg_count(&module_compiled, parameters);
219
220        if call_args.len() != expected_len {
221            bail!("Expected {expected_len} args, found {}", call_args.len());
222        }
223
224        let mut arguments = Vec::with_capacity(expected_len);
225
226        for (idx, (arg, param)) in call_args
227            .into_iter()
228            .zip(parameters.iter().take(expected_len))
229            .enumerate()
230        {
231            let argument = match arg {
232                PtbInput::CallArg(value) => {
233                    let resolved_arg = self
234                        .resolve_json_value_to_call_arg(
235                            &module_compiled,
236                            type_args,
237                            value,
238                            param,
239                            idx,
240                        )
241                        .await?;
242                    match resolved_arg {
243                        ResolvedCallArgResult::CallArg(call_arg) => builder.input(call_arg)?,
244                        ResolvedCallArgResult::ObjVec(object_args) => {
245                            builder.make_obj_vec(object_args)?
246                        }
247                    }
248                }
249                PtbInput::PtbRef(iota_arg) => match iota_arg {
250                    IotaArgument::GasCoin => Argument::Gas,
251                    IotaArgument::Input(idx) => Argument::Input(idx),
252                    IotaArgument::Result(idx) => Argument::Result(idx),
253                    IotaArgument::NestedResult(idx, nested_idx) => {
254                        Argument::NestedResult(idx, nested_idx)
255                    }
256                },
257            };
258
259            arguments.push(argument);
260        }
261
262        Ok(arguments)
263    }

265    /// Convert provided JSON arguments for a move function to their
266    /// [`Argument`] representation and check their validity. Also, check that
267    /// the passed function is declared as a `#[view]` function in the
268    /// module's runtime metadata.
269    pub async fn resolve_and_checks_json_view_args(
270        &self,
271        builder: &mut ProgrammableTransactionBuilder,
272        package_id: ObjectId,
273        module_ident: &Identifier,
274        function_ident: &Identifier,
275        type_args: &[TypeTag],
276        json_args: Vec<IotaJsonValue>,
277    ) -> Result<Vec<Argument>, anyhow::Error> {
278        // Fetch the move package for the given package ID.
279        let package = self.fetch_move_package(package_id).await?;
280        let module = package.deserialize_module(module_ident, &BinaryConfig::standard())?;
281
282        // Then resolve the function parameters type.
283        let json_args_and_tokens = resolve_move_function_args(
284            &module,
285            function_ident,
286            type_args,
287            json_args.into_iter().map(Into::into).collect(),
288            true,
289        )
290        .map_err(|err| {
291            let package_id = &package.id;
292            anyhow!(
293                "Failed to resolve {package_id}::{module_ident}::{function_ident} move function: {err}"
294            )
295        })?;
296
297        // Finally construct the input arguments for the builder.
298        let mut args = Vec::new();
299        for (arg, expected_type) in json_args_and_tokens {
300            args.push(match arg {
301                // Move View Functions can accept pure arguments.
302                // `p` is already BCS-encoded for the expected Move type.
303                ResolvedCallArg::Pure(p) => Ok(builder.pure_bytes(p, false)),
304                // Move View Functions can accept only immutable object references.
305                ResolvedCallArg::Object(id) => {
306                    fp_ensure!(
307                            matches!(expected_type, SignatureToken::Reference(_)),
308                            UserInputError::InvalidMoveViewFunction {
309                                error: format!("Found a function parameter which is not an immutable reference {expected_type:?}")
310                                    ,
311                            }
312                            .into()
313                        );
314                    builder.input(
315                        self.get_object_arg(
316                            id,
317                            // Setting false is safe because of fp_ensure! above
318                            false,
319                            &module,
320                            &expected_type,
321                        )
322                        .await?,
323                    )
324                }
325                // Move View Functions can not accept vector of object by value (this case).
326                ResolvedCallArg::ObjVec(_) => Err(UserInputError::InvalidMoveViewFunction {
327                    error: "Found a function parameter which is a vector of objects".to_owned(),
328                }
329                .into()),
330            }?);
331        }
332
333        Ok(args)
334    }

336    /// Convert provided JSON arguments for a move function to their
337    /// [`CallArg`] representation and check their validity.
338    ///
339    /// Note: For object vectors, each object is added as a separate
340    /// `CallArg::Object` entry.
341    pub async fn resolve_and_check_json_args_to_call_args(
342        &self,
343        package_id: ObjectId,
344        module: &Identifier,
345        function: &Identifier,
346        type_args: &[TypeTag],
347        call_args: Vec<IotaJsonValue>,
348    ) -> Result<Vec<CallArg>, anyhow::Error> {
349        let package = self.fetch_move_package(package_id).await?;
350        let module_compiled = package.deserialize_module(module, &BinaryConfig::standard())?;
351        let parameters = get_function_parameters(&module_compiled, function)?;
352        let expected_len = expected_arg_count(&module_compiled, parameters);
353
354        let mut arguments = Vec::with_capacity(expected_len);
355
356        for (idx, (value, param)) in call_args
357            .into_iter()
358            .zip(parameters.iter().take(expected_len))
359            .enumerate()
360        {
361            let resolved_arg = self
362                .resolve_json_value_to_call_arg(&module_compiled, type_args, value, param, idx)
363                .await?;
364
365            match resolved_arg {
366                ResolvedCallArgResult::CallArg(call_arg) => arguments.push(call_arg),
367                ResolvedCallArgResult::ObjVec(object_args) => {
368                    // For object vectors, add each object as a separate CallArg entry
369                    for obj_arg in object_args {
370                        arguments.push(obj_arg);
371                    }
372                }
373            }
374        }
375
376        Ok(arguments)
377    }

379    /// Get the latest object ref for an object.
380    pub async fn get_object_ref(&self, object_id: ObjectId) -> anyhow::Result<ObjectReference> {

391        &self,
392        object_id: ObjectId,

394        let object = self
395            .0
396            .get_object_with_options(object_id, IotaObjectDataOptions::new().with_type())
397            .await?
398            .into_object()?;
399
400        Ok((object.object_ref(), object.object_type()?))
401    }

403    /// Helper function to get a Move Package for a provided ObjectId.
404    async fn fetch_move_package(&self, package_id: ObjectId) -> Result<MovePackage, anyhow::Error> {
405        let object = self
406            .0
407            .get_object_with_options(package_id, IotaObjectDataOptions::bcs_lossless())
408            .await?
409            .into_object()?;
410        let Some(IotaRawData::Package(package)) = object.bcs else {
411            bail!("Bcs field in object [{package_id}] is missing or not a package.");
412        };
413
414        // Not size-checked: this package is already on-chain, and the limit the
415        // network held it to is not knowable from here.
416        Ok(MovePackage::new(
417            package.id,
418            object.version,
419            package
420                .module_map
421                .iter()
422                .map(|(k, v)| (Identifier::new_unchecked(k.as_str()), v.clone()))
423                .collect(),
424            package.type_origin_table,
425            package
426                .linkage_table
427                .into_iter()
428                .map(|(k, v)| (k, v.into()))
429                .collect(),
430        ))
431    }

441/// Get function parameters from a compiled module, excluding TxContext.
442fn get_function_parameters<'a>(
443    module: &'a CompiledModule,
444    function: &Identifier,
445) -> Result<&'a [SignatureToken], anyhow::Error> {
446    let function_str = function.as_str();
447    let function_def = module
448        .function_defs
449        .iter()
450        .find(|function_def| {
451            module
452                .identifier_at(module.function_handle_at(function_def.function).name)
453                .as_str()
454                == function_str
455        })
456        .ok_or_else(|| {
457            anyhow!(
458                "Could not resolve function {function} in module {}",
459                module.self_id()
460            )
461        })?;
462    let function_signature = module.function_handle_at(function_def.function);
463    Ok(&module.signature_at(function_signature.parameters).0)
464}

466/// Calculate expected argument count, excluding TxContext if present.
467fn expected_arg_count(module: &CompiledModule, parameters: &[SignatureToken]) -> usize {
468    match parameters.last() {
469        Some(param) if TxContext::kind(module, param) != TxContextKind::None => {
470            parameters.len() - 1
471        }
472        _ => parameters.len(),
473    }
