16const SYSTEM_PACKAGE_PUBLISH_ORDER: &[ObjectId] = &[
17    ObjectId::STD,
18    ObjectId::FRAMEWORK,
19    ObjectId::SYSTEM,
20    ObjectId::STARDUST,
21];
22
23/// Returns the list of system packages in the order they should be published.
24/// If the protocol version is < 9 then include also the bridge package.
25pub fn get_system_package_publish_order(protocol_version: u64) -> Vec<ObjectId> {
26    let mut publish_order = SYSTEM_PACKAGE_PUBLISH_ORDER.to_vec();
27    if protocol_version < 9 {
28        publish_order.insert(3, ObjectId::GENESIS_BRIDGE);
29    }
30    publish_order
31}

33pub fn load_bytecode_snapshot(protocol_version: u64) -> anyhow::Result<Vec<SystemPackage>> {
34    let snapshot_path = snapshot_path_for_version(protocol_version)?;
35    let mut snapshots: BTreeMap<ObjectId, SystemPackage> = fs::read_dir(&snapshot_path)?
36        .flatten()
37        .map(|entry| {
38            let file_name = entry.file_name().to_str().unwrap().to_string();
39            let mut file = fs::File::open(snapshot_path.clone().join(file_name))?;
40            let mut buffer = Vec::new();
41            file.read_to_end(&mut buffer)?;
42            let package: SystemPackage = bcs::from_bytes(&buffer)?;
43            Ok((package.id, package))
44        })
45        .collect::<anyhow::Result<_>>()?;
46
47    // system packages need to be restored in a specific order
48    let snapshots_publish_order = get_system_package_publish_order(protocol_version);
49    assert!(snapshots.len() <= snapshots_publish_order.len());
50    let mut snapshot_objects = Vec::new();
51    for package_id in &snapshots_publish_order {
52        if let Some(object) = snapshots.remove(package_id) {
53            snapshot_objects.push(object);
54        }
55    }
56    Ok(snapshot_objects)
57}

59/// Given a protocol version:
60/// * The path to the snapshot directory for that version is returned, if it
61///   exists.
62/// * If the version is greater than the latest snapshot version, then
63///   `Ok(None)` is returned.
64/// * If the version does not exist, but there are snapshots present with
65///   versions greater than `version`, then the smallest snapshot number greater
66///   than `version` is returned.
67fn snapshot_path_for_version(version: u64) -> anyhow::Result<PathBuf> {
68    let snapshot_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("bytecode_snapshot");
69    let mut snapshots = BTreeSet::new();
70
71    for entry in fs::read_dir(&snapshot_dir)? {
72        let entry = entry?;
73        let path = entry.path();
74        if path.is_dir() {
75            if let Some(snapshot_number) = path
76                .file_name()
77                .and_then(|n| n.to_str())
78                .and_then(|n| n.parse::<u64>().ok())
79            {
80                snapshots.insert(snapshot_number);
81            }
82        }
83    }
84
85    snapshots
86        .range(version..)
87        .next()
88        .map(|v| snapshot_dir.join(v.to_string()))
89        .ok_or_else(|| anyhow::anyhow!("No snapshot found for version {version}"))
90}
