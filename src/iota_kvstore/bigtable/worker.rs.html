25/// How often the worker logs progress at info level: one line every this many
26/// checkpoints.
27const PROGRESS_LOG_INTERVAL: CheckpointSequenceNumber = 1000;

29/// Represents the available BigTable tables the Client and the KvWorker can
30/// interact with.
31
32// Variants are declared in write order; the BTreeSet<Table> field
33// iterates by derived Ord, which follows declaration order.
34//
35// Order matters: data tables (Objects, Transactions, Checkpoints)
36// are written before their indexes (TransactionsByAddress,
37// CheckpointsByDigest), so a reader that sees an index entry can
38// always resolve the underlying row. Reordering variants will
39// reorder writes and may break that read-after-write invariant.
40#[derive(
41    Debug,
42    Clone,
43    Copy,
44    PartialEq,
45    Eq,
46    Hash,
47    PartialOrd,
48    Ord,
49    serde::Serialize,
50    serde::Deserialize,
51    strum::Display,
52    strum::AsRefStr,
53    strum::EnumIter,
54)]
55#[serde(rename_all = "kebab-case")]
56#[strum(serialize_all = "snake_case")]
57#[non_exhaustive]
58pub enum Table {
59    /// Stores a mapping of [`ObjectKey`](iota_types::storage::ObjectKey) to
60    /// [`Object`](iota_types::object::Object) for every object.
61    Objects,
62    /// Stores a mapping of
63    /// [`TransactionDigest`](iota_sdk_types::TransactionDigest) to
64    /// [`TransactionEnvelope`](iota_types::transaction::TransactionEnvelope),
65    /// [`TransactionEffects`](iota_sdk_types::TransactionEffects),
66    /// [`TransactionEvents`](iota_sdk_types::TransactionEvents) and
67    /// [`CheckpointSequenceNumber`](iota_types::messages_checkpoint::CheckpointSequenceNumber) for every transaction.
68    Transactions,
69    /// Stores a mapping of ( [`Address`], [`TransactionSequenceNumber`] ) to
70    /// [`TransactionDigest`] for every affected address.
71    ///
72    /// An address is considered "affected" if it appears as the sender, a
73    /// recipient, or the gas payer.
74    TransactionsByAddress,
75    /// Stores a mapping of
76    /// [`CheckpointSequenceNumber`](iota_types::messages_checkpoint::CheckpointSequenceNumber)
77    /// to [`CheckpointContents`](iota_sdk_types::checkpoint::CheckpointContents) and [`CertifiedCheckpointSummary`](iota_types::messages_checkpoint::CertifiedCheckpointSummary) for
78    /// every checkpoint.
79    Checkpoints,
80    /// Stores a mapping of
81    /// [`CheckpointDigest`](iota_sdk_types::CheckpointDigest) to
82    /// [`CheckpointSequenceNumber`](iota_types::messages_checkpoint::CheckpointSequenceNumber)
83    /// for every checkpoint.
84    CheckpointsByDigest,
85}

87/// This worker implementation is responsible for processing checkpoints by
88/// storing its data as Key-Value pairs. The Key-Value pairs are stored in a
89/// BigTableDB.
90pub struct KvWorker {
91    pub client: BigTableClient,
92    /// The tables enabled for writing by this worker.
93    pub enabled_tables: BTreeSet<Table>,
94}
95
96impl KvWorker {
97    /// Creates a new KvWorker that writes data to all tables.
98    ///
99    /// For selective writing, use [`KvWorker::new_selective`].
100    ///
101    /// # Examples
102    ///
103    /// ```rust
104    /// # use iota_kvstore::KvWorker;
105    /// # use iota_kvstore::{Table, BigTableClient};
106    /// # #[tokio::main]
107    /// # async fn main() {
108    /// let client =
109    ///     BigTableClient::new_local("localhost", "local", "instance_id", "column_family").unwrap();
110    ///
111    /// // Write all available tables to BigTable.
112    /// let worker = KvWorker::new(client);
113    /// # drop(worker);
114    /// # }
115    /// ```
116    pub fn new(client: BigTableClient) -> Self {
117        Self {
118            client,
119            // All tables are enabled by default.
120            enabled_tables: Table::iter().collect(),
121        }
122    }
123
124    /// Creates a new KvWorker that writes only to the specified tables.
125    ///
126    /// # NOTE
127    /// Passing an empty iterator yields a worker that performs no writes.
128    /// Use [`KvWorker::new`] to write to all tables.
129    ///
130    /// # Examples
131    ///
132    /// ```rust
133    /// # use iota_kvstore::KvWorker;
134    /// # use iota_kvstore::{Table, BigTableClient};
135    /// # #[tokio::main]
136    /// # async fn main() {
137    /// let client =
138    ///     BigTableClient::new_local("localhost", "local", "instance_id", "column_family").unwrap();
139    ///
140    /// // Write only the `Objects` table to BigTable.
141    /// let worker = KvWorker::new_selective(client, [Table::Objects]);
142    /// # drop(worker);
143    /// # }
144    /// ```
145    pub fn new_selective(client: BigTableClient, tables: impl IntoIterator<Item = Table>) -> Self {
146        Self {
147            client,
148            enabled_tables: BTreeSet::from_iter(tables),
149        }
150    }
151}

153#[async_trait]
154impl Worker for KvWorker {
155    type Message = ();
156    type Error = anyhow::Error;
157
158    async fn process_checkpoint(&self, checkpoint: Arc<CheckpointData>) -> anyhow::Result<()> {
159        let mut client = self.client.clone();
160
161        for table in &self.enabled_tables {
162            match table {
163                Table::Objects => {
164                    let objects = checkpoint
165                        .transactions
166                        .iter()
167                        .flat_map(|t| &t.output_objects)
168                        .collect::<Vec<_>>();
169                    client.save_objects(&objects).await?;
170                }
171                Table::Transactions => {
172                    let transactions = checkpoint
173                        .transactions
174                        .iter()
175                        .map(|t| {
176                            TransactionData::new(t, checkpoint.checkpoint_summary.sequence_number)
177                        })
178                        .collect::<Vec<_>>();
179                    client.save_transactions(&transactions).await?;
180                }
181                Table::TransactionsByAddress => {
182                    let entries_by_address = transactions_by_address(&checkpoint);
183                    client
184                        .save_transactions_by_address(entries_by_address)
185                        .await?;
186                }
187                Table::Checkpoints => {
188                    client.save_checkpoint(&checkpoint).await?;
189                }
190                Table::CheckpointsByDigest => {
191                    client.save_checkpoint_by_digest(&checkpoint).await?;
192                }
193            }
194        }
195
196        let sequence_number = checkpoint.checkpoint_summary.sequence_number;
197        if sequence_number.is_multiple_of(PROGRESS_LOG_INTERVAL) {
198            tracing::info!("stored checkpoint {sequence_number} in the KV store");
199        }
200
201        Ok(())
202    }
203}

205/// Uses [`CheckpointData`] to map addresses affected by a transaction to the
206/// respective transaction identifiers.
207///
208/// Affected addresses include the sender, and payer of the transaction,
209/// plus the owners of all changed objects.
210pub fn transactions_by_address<'a>(
211    checkpoint: &'a CheckpointData,
212) -> impl Iterator<Item = (Address, TransactionSequenceNumber, TransactionDigest)> + 'a {
213    checkpoint
214        .checkpoint_contents
215        .enumerate_transactions(&checkpoint.checkpoint_summary)
216        .zip(&checkpoint.transactions)
217        .flat_map(|((seq, exec_digest), tx)| {
218            let txn = tx.transaction.transaction();
219            let affected: HashSet<Address> =
220                std::iter::once(txn.sender())
221                    .chain(std::iter::once(txn.gas_owner()))
222                    .chain(tx.effects.all_changed_objects().into_iter().filter_map(
223                        |(changed, _)| match changed.owner {
224                            Owner::Address(a) => Some(a),
225                            _ => None,
226                        },
227                    ))
228                    .collect();
229            let digest = exec_digest.transaction;
230            affected.into_iter().map(move |addr| (addr, seq, digest))
231        })
232}
