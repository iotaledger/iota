36/// Read key-value data from a persistent store, such as objects, transactions,
37/// and checkpoints.
38#[async_trait]
39pub trait KeyValueStoreReader {
40    type Error;
41
42    /// Fetches a list of objects by their keys.
43    ///
44    /// Not found objects are omitted from the output list.
45    async fn get_objects(&mut self, objects: &[ObjectKey]) -> Result<Vec<Object>, Self::Error>;
46
47    /// Fetches a list of transactions by their digests.
48    ///
49    /// Not found transactions are omitted from the output list.
50    async fn get_transactions(
51        &mut self,
52        transactions: &[TransactionDigest],
53    ) -> Result<Vec<TransactionData>, Self::Error>;
54
55    /// Fetches a list `(sequence number, digest)` pairs for transactions
56    /// affecting the given address, ordered by [`TransactionsOrder`] and capped
57    /// at `limit`.
58    ///
59    /// # Pagination
60    /// `cursor` is **exclusive**.
61    ///
62    /// - [`TransactionsOrder::NewestFirst`]: returns entries with `tx_seq <
63    ///   cursor`.
64    /// - [`TransactionsOrder::OldestFirst`]: returns entries with `tx_seq >
65    ///   cursor`.
66    ///
67    /// When `None`, the scan starts from the beginning of the requested
68    /// [`TransactionsOrder`].
69    async fn get_transaction_digests_by_address(
70        &mut self,
71        address: Address,
72        cursor: impl Into<Option<TransactionSequenceNumber>> + Send,
73        limit: impl TryInto<NonZeroUsize> + Send,
74        order: TransactionsOrder,
75    ) -> Result<Vec<(TransactionSequenceNumber, TransactionDigest)>, Self::Error>;
76
77    /// Fetches a list of checkpoints by their sequence numbers.
78    ///
79    /// Not found checkpoints are omitted from the output list.
80    async fn get_checkpoints(
81        &mut self,
82        sequence_numbers: &[CheckpointSequenceNumber],
83    ) -> Result<Vec<Checkpoint>, Self::Error>;
84
85    /// Fetches a list of checkpoints by their digests.
86    ///
87    /// Not found checkpoints are omitted from the output list.
88    async fn get_checkpoints_by_digest<I>(
89        &mut self,
90        digests: I,
91    ) -> Result<Vec<Checkpoint>, Self::Error>
92    where
93        I: IntoIterator<Item = CheckpointDigest> + Send,
94        I::IntoIter: Send;
95
96    /// Fetches a list of checkpoint sequence numbers by their digests.
97    ///
98    /// Not found checkpoints are omitted from the output list.
99    async fn get_checkpoint_sequence_numbers<I>(
100        &mut self,
101        digests: I,
102    ) -> Result<Vec<CheckpointSequenceNumber>, Self::Error>
103    where
104        I: IntoIterator<Item = CheckpointDigest> + Send,
105        I::IntoIter: Send;
106}

108/// Writing key-value data to a persistent store, such as objects, transactions,
109/// and checkpoints.
110#[async_trait]
111pub trait KeyValueStoreWriter {
112    type Error;
113
114    /// Persists a list of objects to the store.
115    async fn save_objects(&mut self, objects: &[&Object]) -> Result<(), Self::Error>;
116
117    /// Persists a list of transactions to the store.
118    async fn save_transactions(
119        &mut self,
120        transactions: &[TransactionData],
121    ) -> Result<(), Self::Error>;
122
123    /// Persists a mapping of `(` [`Address`], `transaction_sequence_number`
124    /// `)` to `TransactionDigest` for every affected address.
125    ///
126    /// An address is considered "affected" if it appears as the sender, a
127    /// recipient, or the gas payer.
128    async fn save_transactions_by_address<I>(&mut self, entries: I) -> Result<(), Self::Error>
129    where
130        I: IntoIterator<Item = (Address, u64, TransactionDigest)> + Send,
131        I::IntoIter: Send;
132
133    /// Persists a checkpoint to the store.
134    async fn save_checkpoint(&mut self, checkpoint: &CheckpointData) -> Result<(), Self::Error>;
135
136    /// Persists a checkpoint digest to its corresponding sequence number to the
137    /// store.
138    async fn save_checkpoint_by_digest(
139        &mut self,
140        checkpoint: &CheckpointData,
141    ) -> Result<(), Self::Error>;
142}

144/// Represents all stored Key-Value data associated to a checkpoint containing
145/// both the summary and the full contents.
146#[derive(Clone, Debug, Serialize, Deserialize)]
147pub struct Checkpoint {
148    pub summary: CertifiedCheckpointSummary,
149    pub contents: CheckpointContents,
150}
151
152/// Represents all stored Key-Value data associated with a transaction,
153/// including its effects, events, and the checkpoint number it belongs to.
154#[derive(Clone, Debug, Serialize, Deserialize)]
155pub struct TransactionData {
156    pub transaction: TransactionEnvelope,
157    pub effects: TransactionEffects,
158    pub events: Option<TransactionEvents>,
159    pub checkpoint_number: CheckpointSequenceNumber,
160}
161
162impl TransactionData {
163    pub fn new(
164        checkpoint_transaction: &CheckpointTransaction,
165        checkpoint_sequence_number: CheckpointSequenceNumber,
166    ) -> Self {
167        Self {
168            transaction: checkpoint_transaction.transaction.clone(),
169            effects: checkpoint_transaction.effects.clone(),
170            events: checkpoint_transaction.events.clone(),
171            checkpoint_number: checkpoint_sequence_number,
172        }
173    }
174}
