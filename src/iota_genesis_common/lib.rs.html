25/// Gets a `ProtocolConfig` for genesis based on a `ProtocolVersion`.
26pub fn get_genesis_protocol_config(version: ProtocolVersion) -> ProtocolConfig {
27    // We have a circular dependency here. Protocol config depends on chain ID,
28    // which depends on genesis checkpoint (digest), which depends on genesis
29    // transaction, which depends on protocol config.
30    //
31    // ChainIdentifier::default().chain() which can be overridden by the
32    // IOTA_PROTOCOL_CONFIG_CHAIN_OVERRIDE if necessary
33    ProtocolConfig::get_for_version(version, ChainIdentifier::default().chain())
34}

36/// Prepares the data necessary and then invokes `execute_genesis_transaction`.
37/// The `chain_start_timestamp_ms` is used to construct a genesis `EpochData`.
38/// The `protocol_version` is used to create a `ProtocolConfig` for genesis.
39/// The `genesis_transaction` is the transaction to be executed.
40pub fn prepare_and_execute_genesis_transaction(
41    chain_start_timestamp_ms: CheckpointTimestamp,
42    protocol_version: ProtocolVersion,
43    genesis_transaction: &TransactionEnvelope,
44) -> (TransactionEffects, TransactionEvents, Vec<Object>) {
45    let registry = Registry::new();
46    let metrics = Arc::new(LimitsMetrics::new(&registry));
47    let epoch_data = EpochData::new_genesis(chain_start_timestamp_ms);
48    let protocol_config = get_genesis_protocol_config(protocol_version);
49
50    execute_genesis_transaction(&epoch_data, &protocol_config, metrics, genesis_transaction)
51}

53/// Takes as input a transaction of kind Genesis and executes it. This can be
54/// used to obtain the `TransactionEffects`, `TransactionEvents` and
55/// `Vec<Object>` (vector of created objects) that result from the execution.
56/// Some `EpochData`, `ProtocolConfig` and `LimitsMetrics` are needed as input
57/// to be passed to the transaction executor.
58pub fn execute_genesis_transaction(
59    epoch_data: &EpochData,
60    protocol_config: &ProtocolConfig,
61    metrics: Arc<LimitsMetrics>,
62    genesis_transaction: &TransactionEnvelope,
63) -> (TransactionEffects, TransactionEvents, Vec<Object>) {
64    assert!(
65        matches!(
66            genesis_transaction.transaction().kind(),
67            TransactionKind::Genesis(_)
68        ),
69        "wrong transaction type to execute"
70    );
71    let genesis_digest = *genesis_transaction.digest();
72    // execute txn to effects
73    let silent = true;
74
75    let executor =
76        executor(protocol_config, silent, None).expect("Creating an executor should not fail here");
77
78    let expensive_checks = false;
79    let certificate_deny_set = HashSet::new();
80    let transaction = genesis_transaction.data().transaction();
81    let (kind, signer, mut gas_data) = transaction.execution_parts();
82    gas_data.objects = vec![];
83    let input_objects = CheckedInputObjects::new_for_genesis(vec![]);
84    let (inner_temp_store, _, effects, _execution_error) = executor.execute_transaction_to_effects(
85        &InMemoryStorage::new(Vec::new()),
86        protocol_config,
87        metrics,
88        expensive_checks,
89        &certificate_deny_set,
90        &epoch_data.epoch_id(),
91        epoch_data.epoch_start_timestamp(),
92        input_objects,
93        gas_data,
94        IotaGasStatus::new_unmetered(),
95        kind,
96        signer,
97        genesis_digest,
98        &mut None,
99    );
100    assert!(inner_temp_store.input_objects.is_empty());
101    assert!(inner_temp_store.mutable_inputs.is_empty());
102    assert!(effects.mutated().is_empty());
103    assert!(effects.unwrapped().is_empty());
104    assert!(effects.deleted().is_empty());
105    assert!(effects.wrapped().is_empty());
106    assert!(effects.unwrapped_then_deleted().is_empty());
107
108    let objects = inner_temp_store.written.into_values().collect();
109    (effects, inner_temp_store.events, objects)
110}
