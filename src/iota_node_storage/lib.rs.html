4//! Full-node-specific storage traits.
5//!
6//! This crate defines the [`GrpcStateReader`] and [`GrpcIndexes`] traits, which
7//! extend the core [`ObjectStore`] and [`ReadStore`] interfaces from
8//! `iota-types` with queries that are only available on full nodes (e.g. epoch
9//! info, transaction-to-checkpoint mapping, type layout resolution).
10//!
11//! These traits live in a dedicated lightweight crate so that:
12//! - `iota-core` can implement them without depending on `iota-grpc-server`
13//! - `iota-grpc-server` can consume them without depending on `iota-core`
14//! - `simulacrum` and other test harnesses can implement them freely
15
16use iota_sdk_types::{Address, ObjectId, StructTag, TransactionDigest, TypeTag};
17use iota_types::{
18    base_types::EpochId,
19    digests::ChainIdentifier,
20    messages_checkpoint::{CheckpointSequenceNumber, VerifiedCheckpoint},
21    storage::{
22        CoinInfo, DynamicFieldIteratorItem, EpochInfoV2, ObjectStore, OwnedObjectCursor,
23        OwnedObjectIteratorItem, PackageVersionIteratorItem, ReadStore, TransactionInfo,
24        error::Result,
25    },
26};
27
28/// Trait extending [`ReadStore`] with full-node-specific queries that may
29/// require richer databases or indexes to support.
30pub trait GrpcStateReader: ObjectStore + ReadStore + Send + Sync {
31    /// Lowest available checkpoint for which object data can be requested.
32    ///
33    /// Specifically this is the lowest checkpoint for which input/output object
34    /// data will be available.
35    fn get_lowest_available_checkpoint_objects(&self) -> Result<CheckpointSequenceNumber>;
36
37    fn get_chain_identifier(&self) -> Result<ChainIdentifier>;
38
39    fn get_epoch_last_checkpoint(&self, epoch_id: EpochId) -> Result<Option<VerifiedCheckpoint>>;
40
41    /// Per-epoch verified metadata, held by every node in the CheckpointStore's
42    /// `epoch_info` table (served by `GetEpoch`).
43    fn get_epoch_info(&self, epoch: EpochId) -> Result<Option<EpochInfoV2>>;
44
45    /// Get a handle to an instance of the gRPC indexes.
46    fn grpc_indexes(&self) -> Option<&dyn GrpcIndexes>;
47
48    fn get_type_layout(
49        &self,
50        type_tag: &TypeTag,
51    ) -> Result<Option<move_core_types::annotated_value::MoveTypeLayout>> {
52        use move_core_types::annotated_value::MoveTypeLayout;
53        match type_tag {
54            TypeTag::Bool => Ok(Some(MoveTypeLayout::Bool)),
55            TypeTag::U8 => Ok(Some(MoveTypeLayout::U8)),
56            TypeTag::U64 => Ok(Some(MoveTypeLayout::U64)),
57            TypeTag::U128 => Ok(Some(MoveTypeLayout::U128)),
58            TypeTag::Address => Ok(Some(MoveTypeLayout::Address)),
59            TypeTag::Signer => Ok(Some(MoveTypeLayout::Signer)),
60            TypeTag::Vector(type_tag) => Ok(self
61                .get_type_layout(type_tag)?
62                .map(|layout| MoveTypeLayout::Vector(Box::new(layout)))),
63            TypeTag::Struct(struct_tag) => self.get_struct_layout(struct_tag),
64            TypeTag::U16 => Ok(Some(MoveTypeLayout::U16)),
65            TypeTag::U32 => Ok(Some(MoveTypeLayout::U32)),
66            TypeTag::U256 => Ok(Some(MoveTypeLayout::U256)),
67        }
68    }
69
70    fn get_struct_layout(
71        &self,
72        type_tag: &StructTag,
73    ) -> Result<Option<move_core_types::annotated_value::MoveTypeLayout>>;
74}

76/// GRPC-index-specific queries.
77///
78/// Provides access to the on-disk indexes needed by the gRPC server:
79/// transaction-to-checkpoint mapping, owned objects, dynamic fields, coin
80/// info, and package versions.
81pub trait GrpcIndexes: Send + Sync {
82    fn get_transaction_info(&self, digest: &TransactionDigest) -> Result<Option<TransactionInfo>>;
83
84    /// Iterate over objects owned by `owner`, optionally filtered by
85    /// `object_type`.
86    ///
87    /// Each item includes an [`OwnedObjectCursor`] for seek-based pagination.
88    /// The `cursor` bound is **inclusive**.
89    fn account_owned_objects_info_iter(
90        &self,
91        owner: Address,
92        cursor: Option<&OwnedObjectCursor>,
93        object_type: Option<StructTag>,
94    ) -> Result<Box<dyn Iterator<Item = OwnedObjectIteratorItem> + '_>>;
95
96    /// Iterate over the dynamic fields of `parent`.
97    ///
98    /// The `cursor` bound is **inclusive**.  Only the `DynamicFieldKey` is
99    /// returned; field metadata is loaded on demand from the object store.
100    fn dynamic_field_iter(
101        &self,
102        parent: ObjectId,
103        cursor: Option<ObjectId>,
104    ) -> Result<Box<dyn Iterator<Item = DynamicFieldIteratorItem> + '_>>;
105
106    /// Get unified coin info.
107    fn get_coin_info(&self, coin_type: &StructTag) -> Result<Option<CoinInfo>>;
108
109    /// Iterate over all versions of a package by its original package ID.
110    fn package_versions_iter(
111        &self,
112        original_package_id: ObjectId,
113        cursor: Option<u64>,
114    ) -> Result<Box<dyn Iterator<Item = PackageVersionIteratorItem> + '_>>;
115}
