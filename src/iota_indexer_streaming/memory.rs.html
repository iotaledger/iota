4//! Contains the implementation of in memory streaming of
5//! transactions and events to subscribers.
6//!
7//! It leverages PostgreSQL NOTIFY channel for receiving committed checkpoints
8//! notifications on which it fetches transactions by sequence number ranges,
9//! extracts events from them, and forwards all data to subscribers through
10//! [`tokio::sync::broadcast`]. Supports backfill of historical data from
11//! indexer database enabling stream recovery after a disconnection.

65/// Notification received from PostgreSQL NOTIFY channel when a checkpoint is
66/// committed.
67///
68/// It implies that the [`iota_indexer`] has applied the migrations which
69/// enables the Postgres database to send notification through the channel.
70///
71/// The [`CHANNEL_NAME`] should reflect the same name used in the migrations.
72#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Deserialize)]
73struct CheckpointCommitNotification {
74    /// The sequence number of the committed checkpoint.
75    checkpoint_sequence_number: i64,
76    /// The minimum transaction sequence number in this checkpoint.
77    min_tx_sequence_number: i64,
78    /// The maximum transaction sequence number in this checkpoint.
79    max_tx_sequence_number: i64,
80}

82/// Represents the possible configuration of the [`InMemory`] streaming of
83/// transactions and events data.
84#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord, Deserialize, Serialize)]
85pub struct Config {
86    /// The buffer size of the [`tokio::sync::broadcast`] channel used for
87    /// broadcasting transactions and events data to subscribers.
88    ///
89    /// - default: 1000
90    pub channel_buffer_size: NonZeroUsize,
91    /// The maximum number of checkpoint notifications to batch together for
92    /// processing.
93    ///
94    /// This controls how many PostgreSQL NOTIFY messages are collected before
95    /// resolving transaction bounds and fetching data from the database. Each
96    /// notification represents a committed checkpoint containing one or more
97    /// transactions.
98    ///
99    /// **Performance Trade-offs:**
100    /// - **Higher values**: Reduce database query frequency but increase
101    ///   latency and memory usage per batch
102    /// - **Lower values**: Increase responsiveness but may cause more frequent
103    ///   database queries for small checkpoints
104    ///
105    /// The value of 10 provides a good balance between throughput and latency
106    /// for typical checkpoint sizes.
107    pub notification_chunk_size: NonZeroUsize,
108    /// The maximum number of transactions to send to subscribers in a single
109    /// batch.
110    ///
111    /// This controls how many transactions are processed and broadcast together
112    /// when streaming data to subscribers. Large checkpoints (e.g., genesis
113    /// with thousands of transactions) are automatically split into
114    /// multiple batches of this size to maintain consistent performance.
115    ///
116    /// **Performance Trade-offs:**
117    /// - **Too small**: May fall behind the indexer commit rate, causing the
118    ///   streaming service to lag behind real-time data ingestion
119    /// - **Too large**: May overwhelm subscribers with large batches, causing
120    ///   them to lag or drop messages due to slow processing
121    ///
122    /// The value of 50 provides good balance between indexer synchronization
123    /// and subscriber responsiveness for typical workloads.
124    pub transaction_batch_size: NonZeroI64,
125}
126
127impl Default for Config {
128    fn default() -> Self {
129        Self {
130            channel_buffer_size: NonZeroUsize::new(1000).expect("value should be greater than 0"),
131            transaction_batch_size: NonZeroI64::new(50).expect("value should be greater than 0"),
132            notification_chunk_size: NonZeroUsize::new(10).expect("value should be greater than 0"),
133        }
134    }

137/// Where to start a recovery stream relative to a known transaction digest.
138///
139/// - [`Inclusive`](Self::Inclusive): yield the identified transaction, then
140///   everything after it.
141/// - [`Exclusive`](Self::Exclusive): skip the identified transaction; start
142///   from the next one.
143#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord, Deserialize, Serialize)]
144pub enum RecoveryPoint {
145    /// Include the transaction identified by the digest.
146    Inclusive(TransactionDigest),
147    /// Start after the transaction identified by the digest.
148    Exclusive(TransactionDigest),
149}
150
151impl RecoveryPoint {
152    /// Returns the contained transaction digest.
153    fn digest(&self) -> TransactionDigest {
154        match self {
155            Self::Inclusive(d) | Self::Exclusive(d) => *d,
156        }
157    }
158
159    /// Checks if the starting transaction should be included in the stream.
160    fn is_inclusive(&self) -> bool {
161        matches!(self, Self::Inclusive(_))
162    }
163}

165/// Provides real-time streaming of transactions and events from the IOTA
166/// Indexer by listening to PostgreSQL NOTIFY messages triggered when new
167/// checkpoints are committed to the indexer database.
168///
169/// Also supports backfill of historical data from indexer database enabling
170/// stream recovery after a disconnection.
171///
172/// The streamer consists of:
173/// - A PostgreSQL connection listening for notifications after every committed
174///   checkpoint.
175/// - Internal broadcasters that fan-out data to multiple subscribers using a
176///   [`tokio::sync::broadcast`] channels.
177///
178/// # Usage
179///
180/// ```rust,ignore
181/// use iota_indexer_streaming::{memory::InMemory, metrics::InMemoryStreamMetrics};
182///
183/// // create a new streamer
184/// let streamer = InMemory::new(
185///     Default::default(),
186///     indexer_reader,
187///     InMemoryStreamMetrics::new(registry),
188/// )
189/// .await?;
190///
191/// // subscribe to all events
192/// let events = streamer.subscribe_events(None);
193/// tokio::spawn(async move {
194///     use futures::StreamExt;
195///     while let Some(event) = events.next().await {
196///         println!("New event: {event:?}");
197///     }
198/// });
199/// ```
200pub struct InMemory {
201    event_tx: broadcast::Sender<IndexerStreamingResult<StoredEvent>>,
202    transaction_tx: broadcast::Sender<IndexerStreamingResult<StoredTransaction>>,
203    reader: IndexerReader,
204    metrics: Arc<InMemoryStreamMetrics>,
205    config: Config,
206}

208impl InMemory {
209    /// Creates a new `InMemory` instance.
210    ///
211    /// It performs the following steps:
212    /// - establishes a connection to PostgreSQL.
213    /// - sets up the notification listener.
214    /// - spawns the background task that processes checkpoint notifications.
215    /// - handles automatically reconnecting to PostgreSQL if the connection is
216    ///   lost.
217    pub async fn new(
218        config: Config,
219        indexer_reader: IndexerReader,
220        metrics: impl Into<Arc<InMemoryStreamMetrics>>,
221    ) -> IndexerStreamingResult<Self> {
222        let metrics = metrics.into();
223
224        let (event_tx, _) = broadcast::channel(config.channel_buffer_size.get());
225        let (transaction_tx, _) = broadcast::channel(config.channel_buffer_size.get());
226
227        // task responsible for establishing a permanent postgres connection for
228        // listening to notifications and broadcasting events and transactions to
229        // subscribers.
230        tokio::spawn({
231            let event_tx = event_tx.clone();
232            let transaction_tx = transaction_tx.clone();
233            let metrics = metrics.clone();
234            let indexer_reader = indexer_reader.clone();
235            let span = tracing::info_span!("live_broker");
236
237            async move {
238                loop {
239                    let mut connection = match indexer_reader.get_pool().get() {
240                        Ok(value) => value,
241                        Err(e) => {
242                            error!(
243                                "failed to get connection from postgres connection pool with error: {e:?}"
244                            );
245                            Self::publish_error(
246                                IndexerStreamingError::Postgres(e.to_string()),
247                                &event_tx,
248                                &transaction_tx,
249                            );
250                            tokio::time::sleep(RETRY_POSTGRES_CONNECTION_DELAY).await;
251                            continue;
252                        }
253                    };
254
255                    if let Err(e) =
256                        diesel::sql_query(format!("LISTEN {CHANNEL_NAME}")).execute(&mut connection)
257                    {
258                        error!("failed listening to postgres notify channel: {e}");
259                        Self::publish_error(e.into(), &event_tx, &transaction_tx);
260                        continue;
261                    }
262
263                    if let Err(e) = Self::process_checkpoint_notifications(
264                        &metrics,
265                        &config,
266                        &mut connection,
267                        &indexer_reader,
268                        &event_tx,
269                        &transaction_tx,
270                    )
271                    .await
272                    {
273                        error!("processing checkpoint notifications failed: {e}");
274                        Self::publish_error(e, &event_tx, &transaction_tx);
275                    }
276                }
277            }.instrument(span)
278        });
279
280        Ok(Self {
281            event_tx,
282            transaction_tx,
283            reader: indexer_reader,
284            metrics,
285            config,
286        })
287    }

289    /// Subscribes to a stream of [`StoredEvent`].
290    ///
291    /// By default all events are received, the client shall handle the
292    /// filtering.
293    ///
294    /// When `start_from` is `None`, subscribes to live events only. When
295    /// `start_from` is `Some(recovery_point)`, the stream first backfills
296    /// historical events from the transaction identified by the recovery
297    /// point's digest up to the tip of the network, then seamlessly transitions
298    /// to live events. This enables stream recovery after a disconnection.
299    ///
300    /// The recovery point variant controls whether events of the starting
301    /// transaction are included:
302    /// - [`RecoveryPoint::Inclusive`]: events of the starting transaction are
303    ///   yielded.
304    /// - [`RecoveryPoint::Exclusive`]: events of the starting transaction are
305    ///   skipped; streaming begins from the next transaction onward. Useful for
306    ///   reconnection, where the client already received that transaction's
307    ///   events in the previous session.
308    ///
309    /// # Note
310    /// Since under the hood a [`tokio::sync::broadcast`] channel is used for
311    /// live events, the slow subscriber problem will be handled according to [documentation](https://docs.rs/tokio/latest/tokio/sync/broadcast/index.html#lagging)
312    ///
313    /// # Example
314    ///
315    /// ```rust,ignore
316    /// use iota_indexer_streaming::{error::IndexerStreamingError, memory::RecoveryPoint};
317    /// use tokio_stream::wrappers::errors::BroadcastStreamRecvError;
318    ///
319    /// // live events only.
320    /// let event_stream = streamer.subscribe_events(None);
321    /// tokio::spawn(async move {
322    ///     use futures::StreamExt;
323    ///     while let Some(ev) = event_stream.next().await {
324    ///         if let Ok(ev) = ev.inspect_err(
325    ///             |IndexerStreamingError::Lagged(BroadcastStreamRecvError::Lagged(num))| {
326    ///                 println!("Lagged by {num} events")
327    ///             },
328    ///         ) {
329    ///             println!("Received event: {ev:?}");
330    ///         }
331    ///     }
332    /// });
333    /// ```
334    /// # Example with a starting transaction digest
335    ///
336    /// ```rust,ignore
337    /// use iota_indexer_streaming::{error::IndexerStreamingError, memory::RecoveryPoint};
338    /// use tokio_stream::wrappers::errors::BroadcastStreamRecvError;
339    ///
340    /// // recover events from a specific transaction digest onwards.
341    /// let event_stream = streamer.subscribe_events(Some(RecoveryPoint::Inclusive(tx_digest)));
342    /// tokio::spawn(async move {
343    ///     use futures::StreamExt;
344    ///     while let Some(ev) = event_stream.next().await {
345    ///         if let Ok(ev) = ev.inspect_err(
346    ///             |IndexerStreamingError::Lagged(BroadcastStreamRecvError::Lagged(num))| {
347    ///                 println!("Lagged by {num} events")
348    ///             },
349    ///         ) {
350    ///             println!("Received event: {ev:?}");
351    ///         }
352    ///     }
353    /// });
354    /// ```
355    pub fn subscribe_events(
356        &self,
357        start_from: Option<RecoveryPoint>,
358    ) -> Pin<Box<dyn Stream<Item = IndexerStreamingResult<StoredEvent>> + Send>> {
359        let stream = BroadcastStream::new(self.event_tx.subscribe());
360        let live_stream = SubscriberStream::new(
361            stream,
362            METRICS_EVENT_LABEL,
363            self.metrics.clone(),
364            start_from.is_some(),
365        )
366        .map(Self::flatten_error);
367        let Some(recovery_point) = start_from else {
368            return Box::pin(live_stream);
369        };
370
371        let historical = HistoricalFetch::new(
372            METRICS_EVENT_LABEL,
373            recovery_point.digest(),
374            self.reader.clone(),
375            self.config.transaction_batch_size.get(),
376            self.metrics.clone(),
377        );
378
379        let cursor = historical.cursor_handle();
380
381        let historical_events = historical
382            .into_stream()
383            .skip_while({
384                let mut include_starting_tx = recovery_point.is_inclusive();
385                move |result| {
386                    future::ready(match result {
387                        Ok(_) if !include_starting_tx => {
388                            include_starting_tx = true;
389                            true
390                        }
391                        _ => false,
392                    })
393                }
394            })
395            // we use Either represented by left/right stream methods to unify the two stream types
396            // returned by the closure: stream::iter (multiple events) and stream::once (single
397            // error). This avoids an extra heap allocation from Box::pin.
398            .map(|result| match result {
399                Ok(tx) => match Self::stored_events_from_transaction(&tx) {
400                    Ok(events) => stream::iter(events.into_iter().map(Ok)).left_stream(),
401                    Err(e) => stream::once(future::err(e)).right_stream(),
402                },
403                Err(e) => stream::once(future::err(e)).right_stream(),
404            })
405            .flatten();
406
407        Box::pin(
408            historical_events
409                // when switching from historical to live, the broadcast buffer may
410                // contain transactions already delivered by the historical backfill.
411                // Filter them out using the cursor to prevent duplicates.
412                .chain({
413                    let mut initial_lagged_skipped = false;
414                    live_stream.skip_while(move |result| {
415                        future::ready(match result {
416                            Ok(tx) => tx.tx_sequence_number < cursor.load(Ordering::Relaxed),
417                            Err(IndexerStreamingError::Lagged(_)) if !initial_lagged_skipped => {
418                                initial_lagged_skipped = true;
419                                true // skip this one
420                            }
421                            // surface any other error or second Lagged
422                            Err(_) => false,
423                        })
424                    })
425                }),
426        )
427    }

554    /// Flattens nested `Result` types from the broadcast stream.
555    fn flatten_error<T>(
556        result: Result<IndexerStreamingResult<T>, BroadcastStreamRecvError>,
557    ) -> IndexerStreamingResult<T> {
558        match result {
559            Ok(Ok(ev)) => Ok(ev),

565    /// Listens for database notifications and processes them.
566    ///
567    /// - resolves from received notifications the transaction sequence number
568    ///   bounds.
569    /// - splits the transaction sequence number bounds into batches if
570    ///   exceeded.
571    /// - fetches the transactions within the batch bounds and sends them to
572    ///   subscribers alongside extracted events.
573    /// - leverages exponential backoff retries for transient errors when
574    ///   processing notifications.
575    async fn process_checkpoint_notifications(
576        metrics: &InMemoryStreamMetrics,
577        config: &Config,
578        connection: &mut PoolConnection,
579        indexer_reader: &IndexerReader,
580        event_tx: &broadcast::Sender<IndexerStreamingResult<StoredEvent>>,
581        transaction_tx: &broadcast::Sender<IndexerStreamingResult<StoredTransaction>>,
582    ) -> IndexerStreamingResult<()> {
583        let mut backoff = ExponentialBackoff::default();
584        backoff.max_elapsed_time = Some(Duration::from_secs(5));
585        backoff.initial_interval = Duration::from_millis(100);
586        backoff.current_interval = backoff.initial_interval;
587        backoff.multiplier = 1.0;
588
589        loop {
590            // Poll the PostgreSQL NOTIFY channel for new checkpoint commit
591            // notifications. The iterator is non-blocking, it drains whatever
592            // is currently buffered and returns None when empty. We re-poll
593            // after a short sleep since new notifications can arrive at any time.
594            let messages = connection
595                .notifications_iter()
596                .take(config.notification_chunk_size.get())
597                .collect::<Vec<QueryResult<PgNotification>>>();
598
599            if messages.is_empty() {
600                tokio::time::sleep(PG_NOTIFICATION_POLL_INTERVAL).await;
601                continue;
602            }
603
604            // auto-records duration on drop (after each iteration).
605            let _record_processed_checkpoint_notifications =
606                metrics.process_notification_batch_latency.start_timer();
607
608            if let Some((min_tx_sequence_number, max_tx_sequence_number)) =
609                Self::resolve_tx_bounds(metrics, &messages)?
610            {
611                metrics
612                    .notified_tx_seq_num_start_range
613                    .set(min_tx_sequence_number);
614                metrics
615                    .notified_tx_seq_num_end_range
616                    .set(max_tx_sequence_number);
617
618                let mut start = min_tx_sequence_number;
619
620                while start <= max_tx_sequence_number {
621                    let end = (start + config.transaction_batch_size.get().saturating_sub(1))
622                        .min(max_tx_sequence_number);
623
624                    if let Err(e) = backoff::future::retry(backoff.clone(), || {
625                        Self::process_transaction_batch(
626                            metrics,
627                            start,
628                            end,
629                            indexer_reader,
630                            event_tx,
631                            transaction_tx,
632                        )
633                        .map_err(backoff::Error::transient)
634                        .inspect_err(|e| {
635                            error!("transient error processing transaction batch: {e}")
636                        })
637                    })
638                    .await
639                    {
640                        // once we exhaust all backoff retries, we publish the error and move on to
641                        // the next batch. The client can decide how to handle the error
642                        // accordingly.
643                        error!(
644                            batch_start = start,
645                            batch_end = end,
646                            error = ?e,
647                            "batch processing failed after retries, publishing error to clients"
648                        );
649                        Self::publish_error(e, event_tx, transaction_tx);
650                    }
651
652                    start = end + 1;
653                }
654            }
655        }
656    }

658    /// Resolves the transaction sequence number bounds from the given messages
659    /// batch.
660    fn resolve_tx_bounds(
661        metrics: &InMemoryStreamMetrics,
662        messages: &[QueryResult<PgNotification>],
663    ) -> IndexerStreamingResult<Option<(i64, i64)>> {
664        let mut filtered_messages = Self::filter_checkpoint_notifications(metrics, messages);
665
666        let first = filtered_messages.next().transpose()?;
667        let last = filtered_messages.last().transpose()?;
668
669        Ok(first.map(|f| {
670            (
671                f.min_tx_sequence_number,
672                last.unwrap_or(f).max_tx_sequence_number,
673            )
674        }))
675    }
676
677    /// Fetches transactions from the database within the given range and
678    /// publish them to subscribers alongside extracted events from every
679    /// transaction.
680    async fn process_transaction_batch(
681        metrics: &InMemoryStreamMetrics,
682        start: i64,
683        end: i64,
684        indexer_reader: &IndexerReader,
685        event_tx: &broadcast::Sender<IndexerStreamingResult<StoredEvent>>,
686        transaction_tx: &broadcast::Sender<IndexerStreamingResult<StoredTransaction>>,
687    ) -> IndexerStreamingResult<()> {
688        // auto-records duration on drop (function return).
689        let _record_function_execution_latency =
690            metrics.process_transaction_batch_latency.start_timer();
691        let db_query_timer = metrics.query_tx_from_indexer_db_latency.start_timer();
692
693        let transactions: Vec<StoredTransaction> = indexer_reader
694            .spawn_blocking(move |this| {
695                this.multi_get_transactions_by_sequence_numbers_range(start, end)
696            })
697            .await?;
698
699        let elapsed = db_query_timer.stop_and_record();
700        debug!(
701            "transactions query took: {:?}, tx: {}",
702            Duration::from_secs_f64(elapsed),
703            transactions.len()
704        );
705
706        let publish_data_to_subscribers_timer = metrics
707            .broadcast_tx_and_ev_to_subscribers_latency
708            .start_timer();
709
710        Self::publish_tx_and_events(metrics, transactions, event_tx, transaction_tx).await?;
711
712        let elapsed = publish_data_to_subscribers_timer.stop_and_record();
713        debug!(
714            "broadcast data took: {:?}",
715            Duration::from_secs_f64(elapsed)
716        );
717
718        metrics
719            .broadcasted_to_subscribers_tx_seq_num_start_range
720            .set(start);
721        metrics
722            .broadcasted_to_subscribers_tx_seq_num_end_range
723            .set(end);
724        Ok(())
725    }
726
727    /// Publishes transactions and extracted events from them to subscribers.
728    async fn publish_tx_and_events(
729        metrics: &InMemoryStreamMetrics,
730        transactions: Vec<StoredTransaction>,
731        event_tx: &broadcast::Sender<IndexerStreamingResult<StoredEvent>>,
732        transaction_tx: &broadcast::Sender<IndexerStreamingResult<StoredTransaction>>,
733    ) -> IndexerStreamingResult<()> {
734        // we ignore errors here because we may receive an error if no subscribers are
735        // registered which may happen.
736        for tx in transactions {
737            for event in Self::stored_events_from_transaction(&tx)? {
738                _ = event_tx.send(Ok(event));
739            }
740            _ = transaction_tx.send(Ok(tx));
741        }
742
743        // we sacrifice per-event/transaction granularity to avoid degrading
744        // performance from frequent metric updates in a hot path.
745        metrics
746            .channel_pending_messages
747            .with_label_values(&[METRICS_EVENT_LABEL])
748            .set(event_tx.len() as i64);
749
750        metrics
751            .channel_pending_messages
752            .with_label_values(&[METRICS_TRANSACTION_LABEL])
753            .set(transaction_tx.len() as i64);
754        Ok(())
755    }
756
757    /// Relay an irrecoverable error to all subscribers.
758    ///
759    /// Providing transparency on what happen on the broker side, the client can
760    /// decide how to handle the error accordingly.
761    fn publish_error(
762        error: IndexerStreamingError,
763        event_tx: &broadcast::Sender<IndexerStreamingResult<StoredEvent>>,
764        transaction_tx: &broadcast::Sender<IndexerStreamingResult<StoredTransaction>>,
765    ) {
766        _ = event_tx.send(Err(error.clone()));
767        _ = transaction_tx.send(Err(error));
768    }

770    /// Filters and parses database notifications into
771    /// [`CheckpointCommitNotification`] from PostgreSQL messages.
772    fn filter_checkpoint_notifications<'a>(
773        metrics: &'a InMemoryStreamMetrics,
774        messages: &'a [QueryResult<PgNotification>],
775    ) -> impl Iterator<Item = IndexerStreamingResult<CheckpointCommitNotification>> + 'a {
776        messages.iter().filter_map(|msg_result| match msg_result {
777            Ok(PgNotification { payload, .. }) => {
778                match serde_json::from_str::<CheckpointCommitNotification>(payload) {
779                    Ok(notification) => {
780                        metrics
781                            .notified_checkpoint_sequence_number
782                            .set(notification.checkpoint_sequence_number);
783
784                        Some(Ok(notification))
785                    }
786                    Err(_) => None,
787                }
788            }
789            Err(e) => Some(Err(IndexerStreamingError::Postgres(format!(
790                "database connection error: {e}"
791            )))),
792        })
793    }
794
795    /// Extract [`StoredEvent`]'s from [`StoredTransaction`].
796    fn stored_events_from_transaction(
797        tx: &StoredTransaction,
798    ) -> IndexerStreamingResult<Vec<StoredEvent>> {
799        let with_prefix = true;
800        let native_events: Vec<Event> = stored_events_to_events(tx.events.clone())?;
801        let stored = native_events
802            .into_iter()
803            .enumerate()
804            .map(|(idx, native)| StoredEvent {
805                tx_sequence_number: tx.tx_sequence_number,
806                event_sequence_number: idx as i64,
807                transaction_digest: tx.transaction_digest.clone(),
808                senders: vec![Some(native.sender.as_bytes().to_vec())],
809                package: native.package_id.as_bytes().to_vec(),
810                module: native.module.to_string(),
811                event_type: native.struct_tag.to_canonical_string(with_prefix),
812                timestamp_ms: tx.timestamp_ms,
813                bcs: native.contents,
814            })
815            .collect();
816        Ok(stored)
817    }

820/// A [`Stream`] wrapper that provides metrics capabilities for the
821/// [`BroadcastStream`].
822///
823/// It counts internally the total numbers of subscribers by incrementing the
824/// value every time the [`new`](Self::new) constructor is invoked and
825/// decrementing it when the stream is dropped.
826///
827/// Also the provides a way to track the lagging status of
828/// the subscriber.
829struct SubscriberStream<T> {
830    /// The inner stream implementation we want to wrap.
831    inner: BroadcastStream<T>,
832    /// Tracks if the subscriber is active.
833    active_subscriber_number: IntGauge,
834    /// Tracks if the subscriber is lagging.
835    lagging_subscribers: IntGauge,
836    /// Tracks if this subscriber is currently lagged.
837    is_lagging: bool,
838    /// Timer to track lag duration.
839    lag_start: Option<Instant>,
840}
841
842impl<T> SubscriberStream<T> {
843    /// Represents the duration of the lag state of the subscriber, mostly to
844    /// help Prometheus scrape the metric.
845    const LAG_STATE: Duration = Duration::from_secs(1);
846
847    pub fn new(
848        inner: BroadcastStream<T>,
849        label: &'static str,
850        metrics: Arc<InMemoryStreamMetrics>,
851        is_historical_used_before: bool,
852    ) -> Self {
853        let active_subscriber_number = metrics.active_subscriber_number.with_label_values(&[label]);
854        let lagging_subscribers = metrics.lagging_subscribers.with_label_values(&[label]);
855
856        if !is_historical_used_before {
857            active_subscriber_number.inc();
858        }
859
860        Self {
861            inner,
862            active_subscriber_number,
863            lagging_subscribers,
864            is_lagging: false,
865            lag_start: None,
866        }
867    }
868
869    /// Marks the subscriber as lagging.
870    fn mark_as_lagging(&mut self) {
871        if !self.is_lagging {
872            self.is_lagging = true;
873            self.lag_start = Some(Instant::now());
874            self.lagging_subscribers.inc();
875        }
876    }
877
878    /// Clears the lag flag if the subscriber has been lagging.
879    ///
880    /// It holds the lagging state for at least [`LAG_STATE`](Self::LAG_STATE)
881    /// second in order for Prometheus to scrape the metric.
882    fn clear_lagging(&mut self) {
883        if self.is_lagging
884            && self
885                .lag_start
886                .as_ref()
887                .is_some_and(|instant| instant.elapsed() >= Self::LAG_STATE)
888        {
889            self.lagging_subscribers.dec();
890            self.is_lagging = false;
891            self.lag_start = None;
892        }
893    }
894}
895
896impl<T> Drop for SubscriberStream<T> {
897    fn drop(&mut self) {
898        self.active_subscriber_number.dec();
899        if self.is_lagging {
900            self.lagging_subscribers.dec();
901        }
902    }
903}
904impl<T: Clone + Send + 'static> Stream for SubscriberStream<T> {
905    type Item = Result<T, BroadcastStreamRecvError>;
906
907    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
908        // check if we should clear the lag flag (independent of message arrival)
909        self.clear_lagging();
910
911        let poll = Pin::new(&mut self.inner).poll_next(cx);
912
913        if let Poll::Ready(Some(Err(BroadcastStreamRecvError::Lagged(_)))) = poll {
914            self.mark_as_lagging();
915        }
916
917        poll
918    }
919}

921/// Fetches historical transactions from the indexer database starting from a
922/// given transaction digest up to the latest available sequence number.
923///
924/// This stream is used for stream recovery, the caller chains this stream with
925/// a live broadcast stream to provide gap-free delivery from a recovery point.
926struct HistoricalFetch {
927    /// Transaction digest to start fetching from.
928    start_from_digest: TransactionDigest,
929    /// Next transaction sequence number to fetch.
930    cursor: Arc<AtomicI64>,
931    /// Whether [`Self::cursor`] has been resolved from
932    /// [`Self::start_from_digest`].
933    ///
934    /// Needed because `0` is a valid cursor value (the genesis transaction),
935    /// so it cannot be used as a starting point for the unresolved state.
936    cursor_resolved: bool,
937    /// Latest committed transaction sequence number in the database.
938    latest_tx_in_db: i64,
939    /// Database reader for fetching transactions.
940    reader: IndexerReader,
941    /// Buffered transactions from the last database batch fetch.
942    buffer: VecDeque<StoredTransaction>,
943    /// Number of transactions to fetch per database query.
944    batch_size: i64,
945    /// Upon an unrecoverable error, the stream should be closed.
946    should_close_stream: bool,
947    /// Tracks the latency of querying transaction batch from the indexer
948    /// database.
949    query_tx_from_indexer_db_latency: Histogram,
950}
951
952impl HistoricalFetch {
953    fn new(
954        label: &'static str,
955        start_from_digest: TransactionDigest,
956        reader: IndexerReader,
957        batch_size: i64,
958        metrics: Arc<InMemoryStreamMetrics>,
959    ) -> Self {
960        let active_subscriber_number = metrics.active_subscriber_number.with_label_values(&[label]);
961        active_subscriber_number.inc();
962
963        Self {
964            start_from_digest,
965            cursor: Arc::new(AtomicI64::new(0)),
966            cursor_resolved: false,
967            latest_tx_in_db: 0,
968            reader,
969            buffer: VecDeque::new(),
970            batch_size,
971            should_close_stream: false,
972            query_tx_from_indexer_db_latency: metrics.query_tx_from_indexer_db_latency.clone(),
973        }
974    }
975
976    /// Returns the current value of the cursor.
977    fn cursor_value(&self) -> i64 {
978        self.cursor.load(Ordering::Relaxed)
979    }
980
981    /// Updates the cursor to the given value.
982    fn update_cursor(&self, value: i64) {
983        self.cursor.store(value, Ordering::Relaxed);
984    }
985
986    /// Returns a shared reference to the cursor.
987    ///
988    /// The cursor tracks the next expected transaction sequence number and
989    /// is updated as the historical stream progresses. Used by the live
990    /// stream filter to skip already-delivered transactions.
991    fn cursor_handle(&self) -> Arc<AtomicI64> {
992        self.cursor.clone()
993    }
994
995    /// Retries the provided closure using an [`ExponentialBackoff`]
996    async fn with_retry<F, Fut, T, E>(f: F) -> Result<T, E>
997    where
998        F: Fn() -> Fut,
999        Fut: Future<Output = Result<T, backoff::Error<E>>>,
1000    {
1001        let backoff = ExponentialBackoff {
1002            max_elapsed_time: Some(Duration::from_secs(5)),
1003            initial_interval: Duration::from_millis(100),
1004            multiplier: 1.0,
1005            ..Default::default()
1006        };
1007
1008        backoff::future::retry(backoff, f).await
1009    }
1010
1011    /// Converts into a stream that yields historical transactions from the
1012    /// database in batches. The stream ends once all transactions up to
1013    /// the latest committed transaction to database have been yielded, at which
1014    /// point the caller can chain a live stream.
1015    fn into_stream(self) -> impl Stream<Item = IndexerStreamingResult<StoredTransaction>> {
1016        stream::unfold(self, |mut state| {
1017            let start_from = state.start_from_digest;
1018            let span = tracing::info_span!("historical_backfill", %start_from);
1019            async move {
1020                loop {
1021                    if state.should_close_stream {
1022                        return None;
1023                    }
1024
1025                    // drain the buffer before fetching new transactions from the database.
1026                    // This is the main point of sending transactions to the caller.
1027                    if let Some(tx) = state.buffer.pop_front() {
1028                        return Some((Ok(tx), state));
1029                    }
1030
1031                    // resolve the cursor from the digest if it hasn't been set yet.
1032                    if !state.cursor_resolved {
1033                        match Self::with_retry(|| async {
1034                            state
1035                                .reader
1036                                .db()
1037                                .resolve_cursor_tx_digest_to_seq_num(state.start_from_digest)
1038                                .await
1039                                .map_err(backoff::Error::transient)
1040                        })
1041                        .await
1042                        {
1043                            Ok(cursor) => {
1044                                state.cursor_resolved = true;
1045                                state.update_cursor(cursor);
1046                            },
1047                            Err(e) => {
1048                                state.should_close_stream = true;
1049                                let e = IndexerStreamingError::NotFound(format!(
1050                                    "unable to resolve transaction, may not exist or has been pruned: {e}"
1051                                ));
1052                                error!("{e}");
1053                                return Some((Err(e), state));
1054                            }
1055                        }
1056                    }
1057
1058                    // check if cursor is ahead of latest available transaction on database, if so,
1059                    // refresh latest, so the stream can continue.
1060                    if state.cursor_value() > state.latest_tx_in_db {
1061                        state.latest_tx_in_db = match Self::with_retry(|| async {
1062                            state
1063                                .reader
1064                                .db()
1065                                .latest_tx_sequence_number()
1066                                .await
1067                                .map_err(backoff::Error::transient)
1068                        })
1069                        .await
1070                        {
1071                            Ok(Some(latest)) => latest,
1072                            // this case is very unlikely, but we should handle it gracefully. This
1073                            // is mostly because when we resolve the provided tx digest to a
1074                            // sequence number (in the previous step), this implies that we already
1075                            // have data in transactions table.
1076                            Ok(None) => {
1077                                state.should_close_stream = true;
1078                                let e = IndexerStreamingError::Postgres(
1079                                    "unable to fetch latest tx sequence number".into(),
1080                                );
1081                                error!("{e}");
1082                                return Some((Err(e), state));
1083                            }
1084                            Err(e) => {
1085                                state.should_close_stream = true;
1086                                let e = IndexerStreamingError::Postgres(format!(
1087                                    "unable to fetch latest tx sequence number: {e}"
1088                                ));
1089                                error!("{e}");
1090                                return Some((Err(e), state));
1091                            }
1092                        };
1093
1094                        debug!(
1095                            cursor = state.cursor_value(),
1096                            latest = state.latest_tx_in_db,
1097                            "current state"
1098                        );
1099
1100                        // if latest did not advance, we're at the tip and can close the historical
1101                        // backfill stream and move to the live one.
1102                        if state.cursor_value() > state.latest_tx_in_db {
1103                            debug!("reached the tip and are in sync with live data");
1104                            return None;
1105                        }
1106                    }
1107
1108                    // fetch transaction batch from database and update the cursor.
1109                    let start = state.cursor_value();
1110                    let end = (start + state.batch_size.saturating_sub(1)).min(state.latest_tx_in_db);
1111
1112                    let db_query_timer = state.query_tx_from_indexer_db_latency.start_timer();
1113                    match Self::with_retry(|| async {
1114                        state
1115                            .reader
1116                            .spawn_blocking(move |this| {
1117                                this.multi_get_transactions_by_sequence_numbers_range(start, end)
1118                            })
1119                            .await
1120                            .map_err(backoff::Error::transient)
1121                    })
1122                    .await
1123                    {
1124                        Ok(batch) => {
1125                            let elapsed = db_query_timer.stop_and_record();
1126                            debug!(
1127                                "transactions query took: {:?}, tx: {}",
1128                                Duration::from_secs_f64(elapsed),
1129                                batch.len()
1130                            );
1131                            state.buffer.extend(batch);
1132                            state.update_cursor(end + 1);
1133                        }
1134                        Err(e) => {
1135                            state.should_close_stream = true;
1136                            let e = IndexerStreamingError::Postgres(e.to_string());
1137                            error!(
1138                                batch_start = start,
1139                                batch_end = end,
1140                                error = ?e,
1141                                "batch processing failed after retries, publishing error to clients"
1142                            );
1143                            return Some((Err(e), state));
1144                        }
1145                    }
1146                }
1147            }
1148            .instrument(span)
1149        })
1150    }
