4//! Node-internal implementation of the SDK's
5//! [`TransactionBuilderLedgerClient`].
6//!
7//! [`NodeTransactionBuilderLedgerClient`] backs the SDK
8//! [`TransactionBuilder`](iota_sdk_transaction_builder::TransactionBuilder)
9//! with a node's local state instead of a remote gRPC or GraphQL endpoint.
10//! All reads go through [`GrpcStateReader`] — the same interface the gRPC
11//! server uses — so the client is available wherever that store is
12//! (fullnodes via `GrpcReadStore`, simulacrum in tests).
13//!
14//! The client is ledger-only: it resolves objects, gas, and protocol
15//! parameters, but does not simulate or execute, so transactions are built
16//! with an explicit gas budget via
17//! [`TransactionBuilder::finish_with_budget`](iota_sdk_transaction_builder::TransactionBuilder::finish_with_budget).
18

36/// Default number of objects returned by
37/// [`TransactionBuilderLedgerClient::objects`] when no limit (or a zero
38/// limit) is given.
39const DEFAULT_OBJECTS_PAGE_SIZE: usize = 50;
40
41/// Upper bound on the number of objects returned by
42/// [`TransactionBuilderLedgerClient::objects`].
43const MAX_OBJECTS_PAGE_SIZE: usize = 1000;
44
45/// Payload of the opaque [`TransactionBuilderLedgerClient::objects`] cursor.
46///
47/// Binds the index position to the request that produced it, mirroring the
48/// gRPC server's page token, so a cursor replayed with a different `owner`
49/// or `struct_tag` is rejected instead of silently resuming from a
50/// meaningless position.
51#[derive(Serialize, Deserialize)]
52struct PageToken {
53    owner: Address,
54    struct_tag: Option<StructTag>,
55    cursor: OwnedObjectCursor,
56}
57
58impl PageToken {
59    /// Decodes a cursor and checks that it was produced by a request with the
60    /// same `owner` and `struct_tag`.
61    fn decode(
62        bytes: &[u8],
63        owner: Address,
64        struct_tag: Option<&StructTag>,
65    ) -> Result<OwnedObjectCursor, Error> {
66        let token: PageToken = bcs::from_bytes(bytes).map_err(|e| Error::Cursor(e.to_string()))?;
67        if token.owner != owner || token.struct_tag.as_ref() != struct_tag {
68            return Err(Error::Cursor(
69                "does not match the request parameters".to_string(),
70            ));
71        }
72        Ok(token.cursor)
73    }
74
75    fn encode(&self) -> Vec<u8> {
76        bcs::to_bytes(self).expect("page token serialization cannot fail")
77    }
78}

80/// Error type for [`NodeTransactionBuilderLedgerClient`].
81#[derive(Debug, thiserror::Error)]
82pub enum Error {
83    #[error(transparent)]
84    Storage(#[from] iota_types::storage::error::Error),
85    #[error(transparent)]
86    Store(#[from] TypedStoreError),
87    #[error(transparent)]
88    Conversion(#[from] SdkTypeConversionError),
89    #[error(transparent)]
90    Node(#[from] IotaError),
91    #[error("invalid objects page cursor: {0}")]
92    Cursor(String),
93    #[error("gRPC indexes are disabled on this node")]
94    IndexesDisabled,
95    #[error("protocol version {0} is not supported by this node")]
96    UnsupportedProtocolVersion(u64),
97}

99/// A [`TransactionBuilderLedgerClient`] backed directly by a node's local
100/// state.
101///
102/// [`objects`](TransactionBuilderLedgerClient::objects) requires the gRPC
103/// indexes (the owned-object index) to be enabled and returns
104/// [`Error::IndexesDisabled`] otherwise.
105#[derive(Clone)]
106pub struct NodeTransactionBuilderLedgerClient {
107    reader: Arc<dyn GrpcStateReader>,
108}
109
110impl NodeTransactionBuilderLedgerClient {
111    pub fn new(reader: Arc<dyn GrpcStateReader>) -> Self {
112        Self { reader }
113    }
114
115    /// Protocol config of the current epoch.
116    pub fn node_protocol_config(&self) -> Result<NodeProtocolConfig, Error> {
117        let system_state = get_iota_system_state(self.reader.as_ref())?;
118        let chain = self.reader.get_chain_identifier()?.chain();
119        let protocol_version = system_state.protocol_version();
120        NodeProtocolConfig::get_for_version_if_supported(
121            ProtocolVersion::new(protocol_version),
122            chain,
123        )
124        .ok_or(Error::UnsupportedProtocolVersion(protocol_version))
125    }
126}
127
128impl TransactionBuilderClientBase for NodeTransactionBuilderLedgerClient {
129    type Error = Error;
130}
131
132impl TransactionBuilderLedgerClient for NodeTransactionBuilderLedgerClient {
133    async fn object(
134        &self,
135        object_id: ObjectId,
136        version: impl Into<Option<Version>>,
137    ) -> Result<Option<Object>, Self::Error> {
138        let object = match version.into() {
139            Some(version) => self.reader.try_get_object_by_key(&object_id, version)?,
140            None => self.reader.try_get_object(&object_id)?,
141        };
142        object.map(Object::try_from).transpose().map_err(Into::into)
143    }
144
145    async fn objects(
146        &self,
147        struct_tag: Option<StructTag>,
148        owner: Address,
149        cursor: Option<Vec<u8>>,
150        limit: Option<usize>,
151    ) -> Result<ObjectsPage, Self::Error> {
152        let limit = match limit {
153            None | Some(0) => DEFAULT_OBJECTS_PAGE_SIZE,
154            Some(limit) => limit.min(MAX_OBJECTS_PAGE_SIZE),
155        };
156        let cursor = cursor
157            .map(|bytes| PageToken::decode(&bytes, owner, struct_tag.as_ref()))
158            .transpose()?;
159
160        let indexes = self.reader.grpc_indexes().ok_or(Error::IndexesDisabled)?;
161        // The index iterator's cursor bound is inclusive, so skip the cursor
162        // item itself to advance past the previous page.
163        let skip = usize::from(cursor.is_some());
164        let mut iter = indexes
165            .account_owned_objects_info_iter(owner, cursor.as_ref(), struct_tag.clone())?
166            .skip(skip);
167
168        let mut data = Vec::with_capacity(limit);
169        let mut last_cursor = None;
170        for item in iter.by_ref() {
171            let (info, item_cursor) = item?;
172            let Some(object) = self
173                .reader
174                .try_get_object_by_key(&info.object_id, info.version)?
175            else {
176                // The object is no longer at the indexed version (e.g. mutated
177                // between the index scan and the fetch).
178                tracing::debug!(
179                    object_id = %info.object_id,
180                    version = %info.version,
181                    "object not found while iterating owned objects, skipping",
182                );
183                continue;
184            };
185            data.push(Object::try_from(object)?);
186            last_cursor = Some(item_cursor);
187            if data.len() >= limit {
188                break;
189            }
190        }
191
192        let has_more = iter.next().transpose()?.is_some();
193        let next_cursor = last_cursor.filter(|_| has_more).map(|cursor| {
194            PageToken {
195                owner,
196                struct_tag,
197                cursor,
198            }
199            .encode()
200        });
201
202        Ok(ObjectsPage { data, next_cursor })
203    }
204
205    async fn protocol_config(&self) -> Result<ProtocolConfig, Self::Error> {
206        let attributes = self
207            .node_protocol_config()?
208            .attr_map()
209            .into_iter()
210            .filter_map(|(name, value)| value.map(|value| (name, value.to_string())))
211            .collect();
212        Ok(ProtocolConfig { attributes })
213    }
214
215    async fn reference_gas_price(
216        &self,
217        epoch: impl Into<Option<u64>>,
218    ) -> Result<Option<u64>, Self::Error> {
219        match epoch.into() {
220            None => {
221                let system_state = get_iota_system_state(self.reader.as_ref())?;
222                Ok(Some(system_state.reference_gas_price()))
223            }
224            Some(epoch) => Ok(self
225                .reader
226                .get_epoch_info(epoch)?
227                .map(|info| info.reference_gas_price())),
228        }
229    }
230}
