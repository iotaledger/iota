56    /// Inject a random unused pure input so that two otherwise-identical
57    /// transactions build to distinct digests.
58    ///
59    /// Use this for workloads that repeatedly submit logically identical
60    /// transactions (same sender, gas object and arguments) and must avoid
61    /// colliding on an already-executed digest.

63        self.nonce = Some(rand::thread_rng().gen());

315    pub fn build(self) -> Transaction {
316        let nonce = self.nonce;
317        let mut data = self.build_inner();
318        if let Some(nonce) = nonce {
319            // A trailing pure input that no command references leaves execution
320            // unchanged but alters the serialized transaction, and hence its
321            // digest.
322            if let TransactionKind::Programmable(pt) = data.kind_mut() {
323                pt.inputs.push(Input::Pure(nonce.to_le_bytes().to_vec()));
324            }
325        }
326        data
327    }
328
329    fn build_inner(self) -> Transaction {
330        match self.test_data {
331            TestTransactionData::Move(data) => Transaction::new_move_call(
332                self.sender,
333                data.package_id,
334                data.module,
335                data.function,
336                data.type_args,
337                self.gas_object,
338                data.args,
339                self.gas_budget
340                    .unwrap_or(self.gas_price * TEST_ONLY_GAS_UNIT_FOR_HEAVY_COMPUTATION_STORAGE),
341                self.gas_price,
342            )
343            .unwrap(),
344            TestTransactionData::Transfer(data) => Transaction::new_transfer(
345                data.recipient,
346                data.object,
347                self.sender,
348                self.gas_object,
349                self.gas_budget
350                    .unwrap_or(self.gas_price * TEST_ONLY_GAS_UNIT_FOR_TRANSFER),
351                self.gas_price,
352            ),
353            TestTransactionData::TransferIota(data) => Transaction::new_transfer_iota(
354                data.recipient,
355                self.sender,
356                data.amount,
357                self.gas_object,
358                self.gas_budget
359                    .unwrap_or(self.gas_price * TEST_ONLY_GAS_UNIT_FOR_TRANSFER),
360                self.gas_price,
361            ),
362            TestTransactionData::SplitCoin(data) => Transaction::new_split_coin(
363                self.sender,
364                data.coin,
365                data.amounts,
366                self.gas_object,
367                self.gas_budget
368                    .unwrap_or(self.gas_price * TEST_ONLY_GAS_UNIT_FOR_TRANSFER),
369                self.gas_price,
370            ),
371            TestTransactionData::Publish(data) => {
372                let (all_module_bytes, dependencies) = match data {
373                    PublishData::Source(path, with_unpublished_deps) => {
374                        let compiled_package = BuildConfig::new_for_testing().build(&path).unwrap();
375                        let all_module_bytes =
376                            compiled_package.get_package_bytes(with_unpublished_deps);
377                        let dependencies = compiled_package.get_dependency_storage_package_ids();
378                        (all_module_bytes, dependencies)
379                    }
380                    PublishData::ModuleBytes(bytecode) => (bytecode, vec![]),
381                    PublishData::CompiledPackage(compiled_package) => {
382                        let all_module_bytes = compiled_package.get_package_bytes(false);
383                        let dependencies = compiled_package.get_dependency_storage_package_ids();
384                        (all_module_bytes, dependencies)
385                    }
386                };
387
388                Transaction::new_module(
389                    self.sender,
390                    self.gas_object,
391                    all_module_bytes,
392                    dependencies,
393                    self.gas_budget.unwrap_or(
394                        self.gas_price * TEST_ONLY_GAS_UNIT_FOR_HEAVY_COMPUTATION_STORAGE,
395                    ),
396                    self.gas_price,
397                )
398            }
399            TestTransactionData::Programmable(pt) => Transaction::new_programmable(
400                self.sender,
401                vec![self.gas_object],
402                pt,
403                self.gas_budget
404                    .unwrap_or(self.gas_price * TEST_ONLY_GAS_UNIT_FOR_HEAVY_COMPUTATION_STORAGE),
405                self.gas_price,
406            ),
407            TestTransactionData::Empty => {
408                panic!("Cannot build empty transaction");
409            }
410        }
411    }

413    pub fn build_and_sign(self, signer: &impl Signer<SimpleSignature>) -> TransactionEnvelope {
414        TransactionEnvelope::from_data_and_signer(self.build(), vec![signer])
415    }
416
417    pub fn build_and_sign_multisig(
418        self,
419        multisig_pk: MultisigCommittee,
420        signers: &[&dyn Signer<SimpleSignature>],
421        bitmap: BitmapUnit,
422    ) -> TransactionEnvelope {
423        let data = self.build();
424        let digest = data.signing_digest();
425        let signatures = signers.iter().map(|s| s.sign(&digest).into()).collect();
426        let multisig = UserSignature::Multisig(MultisigAggregatedSignature::new_unchecked(
427            signatures,
428            bitmap,
429            multisig_pk,
430        ));
431
432        TransactionEnvelope::from_user_sig_data(data, vec![multisig])
433    }

480/// A helper function to make Transactions with controlled accounts in
481/// WalletContext. Particularly, the wallet needs to own gas objects for
482/// transactions. However, if this function is called multiple times without any
483/// "sync" actions on gas object management, txns may fail and objects may be
484/// locked.
485///
486/// The param is called `max_txn_num` because it does not always return the
487/// exact same amount of Transactions, for example when there are not enough gas
488/// objects controlled by the WalletContext. Caller should rely on the return
489/// value to check the count.
490pub async fn batch_make_transfer_transactions(
491    context: &WalletContext,
492    max_txn_num: usize,
493) -> Vec<TransactionEnvelope> {
494    let recipient = Address::random();
495    let result = context.get_all_accounts_and_gas_objects().await;
496    let accounts_and_objs = result.unwrap();
497    let mut res = Vec::with_capacity(max_txn_num);
498
499    let gas_price = context.get_reference_gas_price().await.unwrap();
500    for (address, objs) in accounts_and_objs {
501        for obj in objs {
502            if res.len() >= max_txn_num {
503                return res;
504            }
505            let tx = Transaction::new_transfer_iota(
506                recipient,
507                address,
508                Some(2),
509                obj,
510                gas_price * TEST_ONLY_GAS_UNIT_FOR_TRANSFER,
511                gas_price,
512            );
513            let tx = context.sign_transaction(&tx);
514            res.push(tx);
515        }
516    }
517    res
518}

602/// Executes a transaction to publish the `basics` package and another one to
603/// create a counter. Returns the package object ref and the counter object ref.
604pub async fn publish_basics_package_and_make_counter(
605    context: &WalletContext,
606) -> (ObjectReference, ObjectReference) {
607    let package_ref = publish_basics_package(context).await;
608    let (sender, gas_object) = context.get_one_gas_object().await.unwrap().unwrap();
609    let gas_price = context.get_reference_gas_price().await.unwrap();
610    let counter_creation_txn = context.sign_transaction(
611        &TestTransactionBuilder::new(sender, gas_object, gas_price)
612            .call_counter_create(package_ref.object_id)
613            .build(),
614    );
615    let resp = context
616        .execute_transaction_must_succeed(counter_creation_txn)
617        .await;
618    let counter_ref = resp
619        .effects
620        .unwrap()
621        .created()
622        .iter()
623        .find(|obj_ref| matches!(obj_ref.owner, Owner::Shared(_)))
624        .unwrap()
625        .reference;
626    (package_ref, counter_ref)
627}

655}

657/// Executes a transaction that generates a new random u128 using Random and
658/// emits it as an event.
659pub async fn emit_new_random_u128(
660    context: &WalletContext,
661    package_id: ObjectId,
662) -> IotaTransactionBlockResponse {
663    let (sender, gas_object) = context.get_one_gas_object().await.unwrap().unwrap();
664    let rgp = context.get_reference_gas_price().await.unwrap();
665
666    let client = context.get_client().await.unwrap();
667    let random_obj = client
668        .read_api()
669        .get_object_with_options(
670            ObjectId::RANDOMNESS_STATE,
671            IotaObjectDataOptions::new().with_owner(),
672        )
673        .await
674        .unwrap()
675        .into_object()
676        .unwrap();
677    let random_obj_owner = random_obj
678        .owner
679        .expect("Expect Randomness object to have an owner");
680
681    let Owner::Shared(initial_shared_version) = random_obj_owner else {
682        panic!("Expect Randomness to be shared object")
683    };

800/// Fetch one IOTA coin owned by `sender` to use as an explicit gas coin.
801///
802/// Without an explicit gas coin,
803/// [`TransactionBuilder::finish`](iota_sdk_transaction_builder::TransactionBuilder::finish) auto-adds
804/// every IOTA coin the sender owns as gas inputs and merges the leftover into
805/// one output coin, breaking tests that observe the sender's coin count.
806pub async fn select_gas_coin(grpc_client: &iota_grpc_client::Client, sender: Address) -> ObjectId {
807    let gas_coin = grpc_client
808        .list_owned_objects(
809            sender,
810            Some(StructTag::new_gas_coin()),
811            Some(1),
812            None,
813            OwnedObjectReadMask::default(),
814        )
815        .collect(Some(1))
816        .await
817        .expect("failed to fetch gas coin")
818        .into_inner()
819        .into_iter()
820        .next()
821        .expect("sender has no gas coin");
822    *gas_coin
823        .object_reference()
824        .expect("gas coin missing object reference")
825        .object_id()
826}

828/// Build a Move-call transaction ready to be signed, paying gas from a single
829/// coin picked with [`select_gas_coin`].
830///
831/// `args` is anything the builder accepts as an argument list: a tuple of
832/// mixed argument types (`ObjectId` for owned objects, `SharedMut(id)` for
833/// shared mutable objects, `u64`/`Address` and other pure values), or an
834/// array/`Vec` of a single argument type.
835pub async fn move_call_tx<A: PTBArgumentList>(
836    grpc_client: &iota_grpc_client::Client,
837    sender: Address,
838    package_id: ObjectId,
839    module: &str,
840    function: &str,
841    args: A,
842    gas_budget: u64,
843) -> Transaction {
844    let mut builder = grpc_client.transaction_builder(sender);
845
846    builder
847        .move_call(package_id, module, function)
848        .arguments(args);
849
850    builder.gas(vec![select_gas_coin(grpc_client, sender).await]);
851    builder.gas_budget(gas_budget);
852
853    builder
854        .finish()
855        .await
856        .expect("failed to construct move call transaction")
857}

859/// Build a transaction splitting `coin_to_split` into `num_coins` coins of
860/// equal value, ready to be signed. The original coin keeps the remainder.
861///
862/// `gas_coin` must differ from `coin_to_split`; when `None`, the builder
863/// selects gas automatically from the sender's IOTA coins.
864pub async fn split_coin_equal_tx(
865    grpc_client: &iota_grpc_client::Client,
866    sender: Address,
867    coin_to_split: ObjectId,
868    num_coins: u64,
869    gas_coin: Option<ObjectId>,
870    gas_budget: u64,
871) -> Transaction {
872    let coin_object = grpc_client
873        .get_objects([coin_to_split], ObjectReadMask::default())
874        .await
875        .expect("failed to fetch coin")
876        .into_inner()
877        .into_iter()
878        .next()
879        .expect("no result for the requested coin")
880        .expect("coin not found")
881        .object()
882        .expect("invalid coin object");
883    let coin_balance = iota_sdk_types::Coin::try_from_object(&coin_object)
884        .expect("object is not a coin")
885        .balance();
886
887    // Create `num_coins - 1` new coins of equal value; the original keeps the
888    // remainder.
889    let amount_per_split = coin_balance / num_coins;
890    let split_amounts: Vec<u64> = vec![amount_per_split; (num_coins - 1) as usize];
891
892    let mut builder = grpc_client.transaction_builder(sender);
893
894    // Split off the new coin and transfer it back to the sender; an untransferred
895    // `Coin` would be an unused PTB value (coins have no `drop`) and the
896    // transaction would be rejected.
897    let new_coin = builder.split_coins(coin_to_split, split_amounts).result();
898    builder.transfer_objects(sender, [new_coin]);
899
900    if let Some(gas) = gas_coin {
901        builder.gas([gas]);
902    }
903    builder.gas_budget(gas_budget);
904
905    builder
906        .finish()
907        .await
908        .expect("failed to construct split coin transaction")
909}

917    #[test]

922        let build =

927

929        let unique = build().ensure_unique().build();
930
931        // ensure_unique() perturbs the digest by appending one trailing pure
932        // input that no command references.
933        assert_ne!(base.digest(), unique.digest());
934        match (base.kind(), unique.kind()) {
935            (TransactionKind::Programmable(base), TransactionKind::Programmable(unique)) => {
936                assert_eq!(unique.inputs.len(), base.inputs.len() + 1);
937                assert!(matches!(unique.inputs.last(), Some(Input::Pure(_))));
938            }
939            _ => panic!("expected programmable transactions"),
940        }
941
942        // Two independent unique builds differ from each other.
943        assert_ne!(
944            build().ensure_unique().build().digest(),
945            build().ensure_unique().build().digest()
946        );
947    }
