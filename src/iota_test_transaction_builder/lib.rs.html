56    /// transactions build to distinct digests.
57    ///
58    /// Use this for workloads that repeatedly submit logically identical
59    /// transactions (same sender, gas object and arguments) and must avoid
60    /// colliding on an already-executed digest.

314    pub fn build(self) -> Transaction {
315        let nonce = self.nonce;
316        let mut data = self.build_inner();
317        if let Some(nonce) = nonce {
318            // A trailing pure input that no command references leaves execution
319            // unchanged but alters the serialized transaction, and hence its
320            // digest.
321            if let TransactionKind::Programmable(pt) = data.kind_mut() {
322                pt.inputs.push(Input::Pure(nonce.to_le_bytes().to_vec()));
323            }
324        }
325        data
326    }
327
328    fn build_inner(self) -> Transaction {
329        match self.test_data {
330            TestTransactionData::Move(data) => Transaction::new_move_call(
331                self.sender,
332                data.package_id,
333                data.module,
334                data.function,
335                data.type_args,
336                self.gas_object,
337                data.args,
338                self.gas_budget
339                    .unwrap_or(self.gas_price * TEST_ONLY_GAS_UNIT_FOR_HEAVY_COMPUTATION_STORAGE),
340                self.gas_price,
341            )
342            .unwrap(),
343            TestTransactionData::Transfer(data) => Transaction::new_transfer(
344                data.recipient,
345                data.object,
346                self.sender,
347                self.gas_object,
348                self.gas_budget
349                    .unwrap_or(self.gas_price * TEST_ONLY_GAS_UNIT_FOR_TRANSFER),
350                self.gas_price,
351            ),
352            TestTransactionData::TransferIota(data) => Transaction::new_transfer_iota(
353                data.recipient,
354                self.sender,
355                data.amount,
356                self.gas_object,
357                self.gas_budget
358                    .unwrap_or(self.gas_price * TEST_ONLY_GAS_UNIT_FOR_TRANSFER),
359                self.gas_price,
360            ),
361            TestTransactionData::SplitCoin(data) => Transaction::new_split_coin(
362                self.sender,
363                data.coin,
364                data.amounts,
365                self.gas_object,
366                self.gas_budget
367                    .unwrap_or(self.gas_price * TEST_ONLY_GAS_UNIT_FOR_TRANSFER),
368                self.gas_price,
369            ),
370            TestTransactionData::Publish(data) => {
371                let (all_module_bytes, dependencies) = match data {
372                    PublishData::Source(path, with_unpublished_deps) => {
373                        let compiled_package = BuildConfig::new_for_testing().build(&path).unwrap();
374                        let all_module_bytes =
375                            compiled_package.get_package_bytes(with_unpublished_deps);
376                        let dependencies = compiled_package.get_dependency_storage_package_ids();
377                        (all_module_bytes, dependencies)
378                    }
379                    PublishData::ModuleBytes(bytecode) => (bytecode, vec![]),
380                    PublishData::CompiledPackage(compiled_package) => {
381                        let all_module_bytes = compiled_package.get_package_bytes(false);
382                        let dependencies = compiled_package.get_dependency_storage_package_ids();
383                        (all_module_bytes, dependencies)
384                    }
385                };
386
387                Transaction::new_module(
388                    self.sender,
389                    self.gas_object,
390                    all_module_bytes,
391                    dependencies,
392                    self.gas_budget.unwrap_or(
393                        self.gas_price * TEST_ONLY_GAS_UNIT_FOR_HEAVY_COMPUTATION_STORAGE,
394                    ),
395                    self.gas_price,
396                )
397            }
398            TestTransactionData::Programmable(pt) => Transaction::new_programmable(
399                self.sender,
400                vec![self.gas_object],
401                pt,
402                self.gas_budget
403                    .unwrap_or(self.gas_price * TEST_ONLY_GAS_UNIT_FOR_HEAVY_COMPUTATION_STORAGE),
404                self.gas_price,
405            ),
406            TestTransactionData::Empty => {
407                panic!("Cannot build empty transaction");
408            }
409        }
410    }

412    pub fn build_and_sign(self, signer: &impl Signer<SimpleSignature>) -> TransactionEnvelope {
413        TransactionEnvelope::from_data_and_signer(self.build(), vec![signer])
414    }
415
416    pub fn build_and_sign_multisig(
417        self,
418        multisig_pk: MultisigCommittee,
419        signers: &[&dyn Signer<SimpleSignature>],
420        bitmap: BitmapUnit,
421    ) -> TransactionEnvelope {
422        let data = self.build();
423        let digest = data.signing_digest();
424        let signatures = signers.iter().map(|s| s.sign(&digest).into()).collect();
425        let multisig = UserSignature::Multisig(MultisigAggregatedSignature::new_unchecked(
426            signatures,
427            bitmap,
428            multisig_pk,
429        ));
430
431        TransactionEnvelope::from_user_sig_data(data, vec![multisig])
432    }

479/// A helper function to make Transactions with controlled accounts in
480/// WalletContext. Particularly, the wallet needs to own gas objects for
481/// transactions. However, if this function is called multiple times without any
482/// "sync" actions on gas object management, txns may fail and objects may be
483/// locked.
484///
485/// The param is called `max_txn_num` because it does not always return the
486/// exact same amount of Transactions, for example when there are not enough gas
487/// objects controlled by the WalletContext. Caller should rely on the return
488/// value to check the count.
489pub async fn batch_make_transfer_transactions(
490    context: &WalletContext,
491    max_txn_num: usize,
492) -> Vec<TransactionEnvelope> {
493    let recipient = Address::random();
494    let result = context.get_all_accounts_and_gas_objects().await;
495    let accounts_and_objs = result.unwrap();
496    let mut res = Vec::with_capacity(max_txn_num);
497
498    let gas_price = context.get_reference_gas_price().await.unwrap();
499    for (address, objs) in accounts_and_objs {
500        for obj in objs {
501            if res.len() >= max_txn_num {
502                return res;
503            }
504            let tx = Transaction::new_transfer_iota(
505                recipient,
506                address,
507                Some(2),
508                obj,
509                gas_price * TEST_ONLY_GAS_UNIT_FOR_TRANSFER,
510                gas_price,
511            );
512            let tx = context.sign_transaction(&tx);
513            res.push(tx);
514        }
515    }
516    res
517}

601/// Executes a transaction to publish the `basics` package and another one to
602/// create a counter. Returns the package object ref and the counter object ref.
603pub async fn publish_basics_package_and_make_counter(
604    context: &WalletContext,
605) -> (ObjectReference, ObjectReference) {
606    let package_ref = publish_basics_package(context).await;
607    let (sender, gas_object) = context.get_one_gas_object().await.unwrap().unwrap();
608    let gas_price = context.get_reference_gas_price().await.unwrap();
609    let counter_creation_txn = context.sign_transaction(
610        &TestTransactionBuilder::new(sender, gas_object, gas_price)
611            .call_counter_create(package_ref.object_id)
612            .build(),
613    );
614    let resp = context
615        .execute_transaction_must_succeed(counter_creation_txn)
616        .await;
617    let counter_ref = resp
618        .effects
619        .unwrap()
620        .created()
621        .iter()
622        .find(|obj_ref| matches!(obj_ref.owner, Owner::Shared(_)))
623        .unwrap()
624        .reference;
625    (package_ref, counter_ref)
626}

628/// Executes a transaction to increment a counter object.
629/// Must be called after calling `publish_basics_package_and_make_counter`.

634    package_id: ObjectId,

645            .unwrap()

647    let rgp = context.get_reference_gas_price().await.unwrap();
648    let txn = context.sign_transaction(
649        &TestTransactionBuilder::new(sender, gas_object, rgp)
650            .call_counter_increment(package_id, counter_id, initial_shared_version)
651            .build(),
652    );
653    context.execute_transaction_must_succeed(txn).await
654}

656/// Executes a transaction that generates a new random u128 using Random and
657/// emits it as an event.
658pub async fn emit_new_random_u128(
659    context: &WalletContext,
660    package_id: ObjectId,
661) -> IotaTransactionBlockResponse {
662    let (sender, gas_object) = context.get_one_gas_object().await.unwrap().unwrap();
663    let rgp = context.get_reference_gas_price().await.unwrap();
664
665    let client = context.get_client().await.unwrap();
666    let random_obj = client
667        .read_api()
668        .get_object_with_options(
669            ObjectId::RANDOMNESS_STATE,
670            IotaObjectDataOptions::new().with_owner(),
671        )
672        .await
673        .unwrap()
674        .into_object()
675        .unwrap();
676    let random_obj_owner = random_obj
677        .owner
678        .expect("Expect Randomness object to have an owner");
679
680    let Owner::Shared(initial_shared_version) = random_obj_owner else {
681        panic!("Expect Randomness to be shared object")
682    };

799/// Fetch one IOTA coin owned by `sender` to use as an explicit gas coin.
800///
801/// Without an explicit gas coin,
802/// [`TransactionBuilder::finish`](iota_sdk_transaction_builder::TransactionBuilder::finish)
803/// auto-adds every IOTA coin the sender owns as gas inputs and merges the leftover into
804/// one output coin, breaking tests that observe the sender's coin count.
805pub async fn select_gas_coin(
806    grpc_client: &iota_grpc_client::GrpcClient,
807    sender: Address,
808) -> ObjectId {
809    let gas_coin = grpc_client
810        .owned_objects(sender)
811        .object_type(StructTag::new_gas_coin())
812        .page_size(1)
813        .collect(Some(1))
814        .await
815        .expect("failed to fetch gas coin")
816        .into_inner()
817        .into_iter()
818        .next()
819        .expect("sender has no gas coin");
820    *gas_coin
821        .object_reference()
822        .expect("gas coin missing object reference")
823        .object_id()
824}

826/// Build a Move-call transaction ready to be signed, paying gas from a single
827/// coin picked with [`select_gas_coin`].
828///
829/// `args` is anything the builder accepts as an argument list: a tuple of
830/// mixed argument types (`ObjectId` for owned objects, `SharedMut(id)` for
831/// shared mutable objects, `u64`/`Address` and other pure values), or an
832/// array/`Vec` of a single argument type.
833pub async fn move_call_tx<A: PTBArgumentList>(
834    grpc_client: &iota_grpc_client::GrpcClient,
835    sender: Address,
836    package_id: ObjectId,
837    module: &str,
838    function: &str,
839    args: A,
840    gas_budget: u64,
841) -> Transaction {
842    let mut builder = grpc_client.transaction_builder(sender);
843
844    builder
845        .move_call(package_id, module, function)
846        .arguments(args);
847
848    builder.gas(vec![select_gas_coin(grpc_client, sender).await]);
849    builder.gas_budget(gas_budget);
850
851    builder
852        .finish()
853        .await
854        .expect("failed to construct move call transaction")
855}

857/// Build a transaction splitting `coin_to_split` into `num_coins` coins of
858/// equal value, ready to be signed. The original coin keeps the remainder.
859///
860/// `gas_coin` must differ from `coin_to_split`; when `None`, the builder
861/// selects gas automatically from the sender's IOTA coins.
862pub async fn split_coin_equal_tx(
863    grpc_client: &iota_grpc_client::GrpcClient,
864    sender: Address,
865    coin_to_split: ObjectId,
866    num_coins: u64,
867    gas_coin: Option<ObjectId>,
868    gas_budget: u64,
869) -> Transaction {
870    let coin_object = grpc_client
871        .objects([coin_to_split])
872        .await
873        .expect("failed to fetch coin")
874        .into_inner()
875        .into_iter()
876        .next()
877        .expect("no result for the requested coin")
878        .expect("coin not found")
879        .object()
880        .expect("invalid coin object");
881    let coin_balance = iota_sdk_types::Coin::try_from_object(&coin_object)
882        .expect("object is not a coin")
883        .balance();
884
885    // Create `num_coins - 1` new coins of equal value; the original keeps the
886    // remainder.
887    let amount_per_split = coin_balance / num_coins;
888    let split_amounts: Vec<u64> = vec![amount_per_split; (num_coins - 1) as usize];
889
890    let mut builder = grpc_client.transaction_builder(sender);
891
892    // Split off the new coin and transfer it back to the sender; an untransferred
893    // `Coin` would be an unused PTB value (coins have no `drop`) and the
894    // transaction would be rejected.
895    let new_coin = builder.split_coins(coin_to_split, split_amounts).result();
896    builder.transfer_objects(sender, [new_coin]);
897
898    if let Some(gas) = gas_coin {
899        builder.gas([gas]);
900    }
901    builder.gas_budget(gas_budget);
902
903    builder
904        .finish()
905        .await
906        .expect("failed to construct split coin transaction")
907}

913    use super::*;
914
915    #[test]
916    fn ensure_unique_changes_digest() {
917        let sender = dbg_addr(1);
918        let recipient = dbg_addr(2);
919        let gas = random_object_ref();
920        let build =
921            || TestTransactionBuilder::new(sender, gas, 1000).transfer_iota(Some(1), recipient);
922
923        // Identical inputs build to the same digest.
924        assert_eq!(build().build().digest(), build().build().digest());
925
926        let base = build().build();
927        let unique = build().ensure_unique().build();
928
929        // ensure_unique() perturbs the digest by appending one trailing pure
930        // input that no command references.
931        assert_ne!(base.digest(), unique.digest());
932        match (base.kind(), unique.kind()) {
933            (TransactionKind::Programmable(base), TransactionKind::Programmable(unique)) => {
934                assert_eq!(unique.inputs.len(), base.inputs.len() + 1);
935                assert!(matches!(unique.inputs.last(), Some(Input::Pure(_))));
936            }
937            _ => panic!("expected programmable transactions"),
938        }
939
940        // Two independent unique builds differ from each other.
941        assert_ne!(
942            build().ensure_unique().build().digest(),
943            build().ensure_unique().build().digest()
944        );
945    }
