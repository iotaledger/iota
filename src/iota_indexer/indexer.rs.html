32/// Maximum timeout for resolving the remote checkpoint source.
33const MAX_URL_RESOLUTION_TIMEOUT: Duration = Duration::from_secs(30);
34
35pub struct Indexer;

37impl Indexer {
38    pub async fn start_writer_with_config(
39        config: &IngestionConfig,
40        store: PgIndexerStore,
41        metrics: IndexerMetrics,
42        retention_config: Option<RetentionConfig>,
43        pruning_delay_ms: u64,
44        pruning_batch_size: u64,
45        cancel: CancellationToken,
46    ) -> Result<(), IndexerError> {
47        info!(
48            "IOTA Indexer Writer (version {:?}) started...",
49            env!("CARGO_PKG_VERSION")
50        );
51
52        info!("IOTA Indexer Writer config: {config:?}",);
53        let extra_reader_options = ReaderOptions {
54            batch_size: config.checkpoint_download_queue_size,
55            timeout_secs: config.checkpoint_download_timeout,
56            data_limit: config.checkpoint_download_queue_size_bytes,
57            ..Default::default()
58        };
59
60        let remote_store_url =
61            resolve_remote_url(&config.sources, MAX_URL_RESOLUTION_TIMEOUT).await?;
62
63        if let Some(retention_config) = retention_config {
64            let pruner = Pruner::new(
65                store.clone(),
66                retention_config,
67                pruning_delay_ms,
68                pruning_batch_size,
69                metrics.clone(),
70            )?;
71            let cancel_clone = cancel.clone();
72            spawn_monitored_task!(pruner.start(cancel_clone));
73        }
74
75        // If we already have chain identifier indexed (i.e. the first checkpoint has
76        // been indexed), then we persist protocol configs for protocol versions
77        // not yet in the db. Otherwise, we would do the persisting in
78        // `commit_checkpoint` while the first cp is being indexed.
79        if let Some(chain_id) = IndexerStore::get_chain_identifier(&store).await? {
80            store.persist_protocol_configs_and_feature_flags(chain_id)?;
81        }
82
83        // Restore metric from the DB so it doesn't stay as 0 until next epoch change.
84        let (_, latest_epoch) = IndexerStore::get_available_epoch_range(&store).await?;
85        metrics.last_committed_epoch.set(latest_epoch as i64);
86
87        let primary_pipeline = PrimaryPipeline::setup(
88            store.clone(),
89            metrics.clone(),
90            config.checkpoint_download_queue_size,
91            cancel.clone(),
92        )
93        .await?;
94
95        info!("Starting data ingestion executor...");
96        let primary_pipeline_handle = primary_pipeline
97            .run(
98                config.sources.data_ingestion_path.clone(),
99                remote_store_url,
100                extra_reader_options,
101            )
102            .await;
103
104        let result = primary_pipeline_handle
105            .await
106            .context("failed to join primary pipeline")?
107            .context("primary pipeline failed");
108        info!("Primary pipeline finished");
109        // Tell other tasks (e.g. the pruner) to stop.
110        cancel.cancel();
111        result?;
112
113        Ok(())
114    }

116    pub async fn start_reader(
117        config: &JsonRpcConfig,
118        store: PgIndexerStore,
119        registry: &Registry,
120        connection_pool: ConnectionPool,
121        metrics: IndexerMetrics,
122        cancel: CancellationToken,
123    ) -> Result<(), IndexerError> {
124        info!(
125            "IOTA Indexer Reader (version {:?}) started...",
126            env!("CARGO_PKG_VERSION")
127        );
128
129        // Create the watermark cache that will track pruning state
130        let watermark_cache = WatermarkCache::new();
131        let mut read = IndexerReader::new(connection_pool.clone(), watermark_cache.clone());
132
133        if let HistoricFallbackOptions {
134            fallback_kv_url: Some(ref url),
135            fallback_kv_multi_fetch_batch_size,
136            fallback_kv_concurrent_fetches,
137            fallback_kv_cache_size,
138        } = config.historic_fallback_options
139        {
140            let historic_fallback_reader = HistoricalFallbackReader::new(
141                url.as_str(),
142                fallback_kv_cache_size,
143                read.package_resolver().clone(),
144                fallback_kv_multi_fetch_batch_size,
145                fallback_kv_concurrent_fetches,
146                registry,
147            )?;
148            info!("HistoricalFallbackReader initialized with URL: {url}");
149            read.with_fallback_reader(historic_fallback_reader);
150        } else {
151            info!("No config for HistoricalFallbackReader provided, skipping...");
152        }
153
154        let handle = build_json_rpc_server(
155            store.clone(),
156            registry,
157            read.clone(),
158            config,
159            metrics,
160            cancel.clone(),
161        )
162        .await
163        .expect("json rpc server should not run into errors upon start.");
164
165        tracing::info!("Starting watermark background task to track pruning state");
166        let watermark_task = WatermarkTask::new(store, watermark_cache);
167        watermark_task.start(cancel.clone());
168
169        tracing::info!("Starting system package task");
170        let system_package_task =
171            SystemPackageTask::new(read, cancel, std::time::Duration::from_secs(10));
172        spawn_monitored_task!(async move { system_package_task.run().await });
173
174        tokio::spawn(async move { handle.stopped().await })
175            .await
176            .expect("rpc server task failed");
177
178        Ok(())
179    }

181    pub async fn start_analytical_worker<
182        S: IndexerAnalyticalStore + Clone + Send + Sync + 'static,
183    >(
184        store: S,
185        metrics: IndexerMetrics,
186    ) -> Result<(), IndexerError> {
187        info!(
188            "IOTA Indexer Analytical Worker (version {:?}) started...",
189            env!("CARGO_PKG_VERSION")
190        );
191        let mut processor_orchestrator = ProcessorOrchestrator::new(store, metrics);
192        processor_orchestrator.run_forever().await;
193        Ok(())
194    }
