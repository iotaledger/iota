204        &self,
205        start_checkpoint: i64,
206        end_checkpoint: i64,

208        info!(
209            "Persisting tx count metrics for checkpoints [{start_checkpoint}-{}]",
210            end_checkpoint - 1

213            &self.blocking_cp,

224        info!(
225            "Persisted tx count metrics for checkpoints [{start_checkpoint}-{}]",
226            end_checkpoint - 1
227        );
228        Ok(())
229    }

304    async fn calculate_and_persist_address_metrics(&self, checkpoint: i64) -> IndexerResult<()> {
305        let mut checkpoint_opt = self
306            .get_checkpoints_in_range(checkpoint, checkpoint + 1)
307            .await?
308            .pop();
309        while checkpoint_opt.is_none() {
310            tokio::time::sleep(std::time::Duration::from_secs(1)).await;
311            checkpoint_opt = self
312                .get_checkpoints_in_range(checkpoint, checkpoint + 1)
313                .await?
314                .pop();
315        }
316        let checkpoint = checkpoint_opt.unwrap();
317        let cp_timestamp_ms = checkpoint.timestamp_ms;
318        let addr_count = read_only_blocking!(&self.blocking_cp, |conn| {
319            addresses::dsl::addresses
320                .filter(addresses::first_appearance_time.le(cp_timestamp_ms))
321                .count()
322                .get_result::<i64>(conn)
323        })?;
324        let active_addr_count = read_only_blocking!(&self.blocking_cp, |conn| {
325            active_addresses::dsl::active_addresses
326                .filter(active_addresses::first_appearance_time.le(cp_timestamp_ms))
327                .count()
328                .get_result::<i64>(conn)
329        })?;
330        let time_one_day_ago = cp_timestamp_ms - 1000 * 60 * 60 * 24;
331        let daily_active_addresses = read_only_blocking!(&self.blocking_cp, |conn| {
332            active_addresses::dsl::active_addresses
333                .filter(active_addresses::first_appearance_time.le(cp_timestamp_ms))
334                .filter(active_addresses::last_appearance_time.gt(time_one_day_ago))
335                .select(count(active_addresses::address))
336                .first(conn)
337        })?;
338        let address_metrics_to_commit = StoredAddressMetrics {
339            checkpoint: checkpoint.sequence_number,
340            epoch: checkpoint.epoch,
341            timestamp_ms: checkpoint.timestamp_ms,
342            cumulative_addresses: addr_count,
343            cumulative_active_addresses: active_addr_count,
344            daily_active_addresses,
345        };
346        transactional_blocking_with_retry!(
347            &self.blocking_cp,
348            |conn| {
349                diesel::insert_into(address_metrics::table)
350                    .values(address_metrics_to_commit.clone())
351                    .on_conflict_do_nothing()
352                    .execute(conn)
353            },
354            Duration::from_secs(60)
355        )
356        .context("failed persisting address metrics to PostgresDB")?;
357        Ok(())
358    }

401    async fn calculate_and_persist_move_call_metrics(&self, epoch: i64) -> IndexerResult<()> {
402        let move_call_query_3d = build_move_call_metric_query(epoch, 3);
403        let move_call_query_7d = build_move_call_metric_query(epoch, 7);
404        let move_call_query_30d = build_move_call_metric_query(epoch, 30);
405
406        let mut calculate_tasks = vec![];
407        let blocking_cp_3d = self.blocking_cp.clone();
408        calculate_tasks.push(tokio::task::spawn_blocking(move || {
409            read_only_blocking!(&blocking_cp_3d, |conn| {
410                diesel::sql_query(move_call_query_3d).get_results::<QueriedMoveMetrics>(conn)
411            })
412        }));
413        let blocking_cp_7d = self.blocking_cp.clone();
414        calculate_tasks.push(tokio::task::spawn_blocking(move || {
415            read_only_blocking!(&blocking_cp_7d, |conn| {
416                diesel::sql_query(move_call_query_7d).get_results::<QueriedMoveMetrics>(conn)
417            })
418        }));
419        let blocking_cp_30d = self.blocking_cp.clone();
420        calculate_tasks.push(tokio::task::spawn_blocking(move || {
421            read_only_blocking!(&blocking_cp_30d, |conn| {
422                diesel::sql_query(move_call_query_30d).get_results::<QueriedMoveMetrics>(conn)
423            })
424        }));
425        let chained = futures::future::join_all(calculate_tasks)
426            .await
427            .into_iter()
428            .collect::<Result<Vec<_>, _>>()
429            .tap_err(|e| {
430                error!("error joining move call calculation tasks: {e:?}");
431            })?
432            .into_iter()
433            .collect::<Result<Vec<_>, _>>()
434            .tap_err(|e| {
435                error!("error calculating move call metrics: {e:?}");
436            })?
437            .into_iter()
438            .flatten()
439            .collect::<Vec<_>>();
440
441        let move_call_metrics: Vec<StoredMoveCallMetrics> = chained
442            .into_iter()
443            .filter_map(|queried_move_metrics| {
444                let package = ObjectId::from_bytes(queried_move_metrics.move_package.clone()).ok();
445                let package_str = match package {
446                    Some(p) => p.to_canonical_string(/* with_prefix */ true),
447                    None => {
448                        tracing::error!(
449                            "failed to parse move package ID: {:?}",
450                            queried_move_metrics.move_package
451                        );
452                        return None;
453                    }
454                };
455                Some(StoredMoveCallMetrics {
456                    id: None,
457                    epoch,
458                    day: queried_move_metrics.day,
459                    move_package: package_str,
460                    move_module: queried_move_metrics.move_module,
461                    move_function: queried_move_metrics.move_function,
462                    count: queried_move_metrics.count,
463                })
464            })
465            .collect();
466
467        transactional_blocking_with_retry!(
468            &self.blocking_cp,
469            |conn| {
470                diesel::insert_into(move_call_metrics::table)
471                    .values(move_call_metrics.clone())
472                    .on_conflict_do_nothing()
473                    .execute(conn)
474            },
475            Duration::from_secs(60)
476        )
477        .context("failed persisting move call metrics to PostgresDB")?;
478        Ok(())
479    }

482fn construct_peak_tps_query(epoch: i64, offset: i64) -> String {
483    format!(
484        "WITH filtered_checkpoints AS (
485            SELECT
486              MAX(checkpoint_sequence_number) AS checkpoint_sequence_number,
487              SUM(total_successful_transactions) AS total_successful_transactions,
488              timestamp_ms
489            FROM
490              tx_count_metrics
491              WHERE epoch > ({epoch} - {offset}) AND epoch <= {epoch}
492            GROUP BY
493              timestamp_ms
494          ),
495          tps_data AS (
496            SELECT
497              checkpoint_sequence_number,
498              total_successful_transactions,
499              timestamp_ms - LAG(timestamp_ms) OVER (ORDER BY timestamp_ms) AS time_diff
500            FROM
501              filtered_checkpoints
502          )
503          SELECT
504            MAX(total_successful_transactions * 1000.0 / time_diff)::float8 as peak_tps
505          FROM
506            tps_data
507          WHERE
508            time_diff IS NOT NULL;
509        "
510    )
511}

513fn construct_move_call_persist_query(start_tx_seq: i64, end_tx_seq: i64) -> String {
514    format!(
515        "INSERT INTO move_calls
516    SELECT
517        m.tx_sequence_number AS transaction_sequence_number,
518        c.sequence_number AS checkpoint_sequence_number,
519        c.epoch AS epoch,
520        m.package AS move_package,
521        m.module AS move_module,
522        m.func AS move_function
523    FROM tx_calls_fun m
524    INNER JOIN transactions t
525        ON m.tx_sequence_number = t.tx_sequence_number
526    INNER JOIN checkpoints c
527        ON t.checkpoint_sequence_number = c.sequence_number
528    -- Ensure partition pruning
529    WHERE t.tx_sequence_number >= {start_tx_seq} AND t.tx_sequence_number < {end_tx_seq}
530    ON CONFLICT (transaction_sequence_number, move_package, move_module, move_function) DO NOTHING;
531    "
532    )
