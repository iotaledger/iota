20const PG_DB_COMMIT_SLEEP_DURATION: Duration = Duration::from_secs(3600);
21
22pub(crate) struct ObjectChangesUnwrappedBackfill;
23
24#[async_trait::async_trait]
25impl IngestionBackfill for ObjectChangesUnwrappedBackfill {
26    type ProcessedType = StoredTransaction;
27
28    async fn process_checkpoint(
29        checkpoint: Arc<CheckpointData>,
30    ) -> Result<Vec<Self::ProcessedType>, IndexerError> {
31        let checkpoint_summary = &checkpoint.checkpoint_summary;
32        let checkpoint_contents = &checkpoint.checkpoint_contents;
33        let transactions = &checkpoint.transactions;
34        let checkpoint_seq = checkpoint_summary.sequence_number;
35
36        if checkpoint_contents.len() != transactions.len() {
37            return Err(IndexerError::FullNodeReading(format!(
38                "checkpoint content size mismatch at checkpoint {checkpoint_seq}: expected {}, found {}",
39                checkpoint_contents.len(),
40                transactions.len()
41            )));
42        }
43
44        let tx_seq_numbers = checkpoint_contents
45            .enumerate_transactions(checkpoint_summary)
46            .map(|(seq, digest)| (digest.transaction, seq));
47
48        let mut results = Vec::new();
49        let dummy_metrics = IndexerMetrics::new(&Registry::new());
50
51        // Only transactions with unwrapped objects need to be backfilled
52        for (tx, (expected_digest, tx_sequence_number)) in transactions
53            .iter()
54            .zip(tx_seq_numbers)
55            .filter(|(tx, _)| !tx.effects.unwrapped().is_empty())
56        {
57            let actual_digest = tx.transaction.digest();
58
59            if expected_digest != *actual_digest {
60                return Err(IndexerError::FullNodeReading(format!(
61                    "digest mismatch at checkpoint {checkpoint_seq}: expected {expected_digest}, found {actual_digest}",
62                )));
63            }
64
65            let indexed_tx = PrimaryWorker::index_transaction(
66                tx,
67                tx_sequence_number,
68                checkpoint_seq,
69                checkpoint_summary.timestamp_ms,
70                &dummy_metrics,
71            )
72            .await?;
73
74            results.push(StoredTransaction::from(&indexed_tx));
75        }
76
77        Ok(results)
78    }
79
80    async fn persist_chunk(
81        pool: ConnectionPool,
82        processed_data: Vec<Self::ProcessedType>,
83    ) -> Result<(), IndexerError> {
84        if processed_data.is_empty() {
85            return Ok(());
86        }
87
88        let (tx_sequence_numbers, object_changes): (Vec<i64>, Vec<Vec<Option<Vec<u8>>>>) =
89            processed_data
90                .into_iter()
91                .map(|tx| (tx.tx_sequence_number, tx.object_changes))
92                .unzip();
93
94        // The UPDATE only affects rows that exist in the database. Update for
95        // non-existing rows is silently skipped.
96        transactional_blocking_with_retry!(
97            &pool,
98            |conn| {
99                for (tx_seq, obj_changes) in tx_sequence_numbers.iter().zip(object_changes.iter()) {
100                    diesel::update(transactions::table)
101                        .filter(transactions::tx_sequence_number.eq(tx_seq))
102                        .set(transactions::object_changes.eq(obj_changes))
103                        .execute(conn)?;
104                }
105
106                Ok::<(), IndexerError>(())
107            },
108            PG_DB_COMMIT_SLEEP_DURATION
109        )
110    }
111}
