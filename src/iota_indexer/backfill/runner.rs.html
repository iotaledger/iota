26/// Entry point for orchestrating backfills.
27///
28/// `BackfillRunner` selects the appropriate backfill implementation, splits the
29/// requested checkpoint range into manageable chunks, and dispatches them in
30/// parallel.
31pub struct BackfillRunner;
32
33impl BackfillRunner {
34    /// Execute a backfill over `total_range` using the specified task kind.
35    pub async fn run(
36        runner_kind: BackfillKind,
37        pool: ConnectionPool,
38        backfill_config: BackfillConfig,
39        total_range: RangeInclusive<usize>,
40    ) -> Result<(), IndexerError> {
41        let backfill = get_backfill(runner_kind, *total_range.start()).await?;
42        Self::run_impl(pool, backfill_config, total_range, backfill).await
43    }
44
45    async fn run_impl(
46        pool: ConnectionPool,
47        config: BackfillConfig,
48        total_range: RangeInclusive<usize>,
49        backfill: Arc<dyn Backfill>,
50    ) -> Result<(), IndexerError> {
51        let timer = Instant::now();
52        let processed_counter = Arc::new(AtomicUsize::new(0));
53
54        // Generate chunks
55        let chunk_stream = chunk_range_stream(total_range, config.chunk_size);
56
57        // Process chunks in parallel, fail-fast on error
58        chunk_stream
59            .map(|range| {
60                let pool = pool.clone();
61                let backfill = backfill.clone();
62                let counter = processed_counter.clone();
63
64                async move {
65                    let start = *range.start();
66                    let end = *range.end();
67
68                    // Execute backfill for the range
69                    if let Err(e) = backfill.backfill_range(pool, &range).await {
70                        error!("chunk {start}-{end} failed. Error: {e}");
71                        return Err(e);
72                    }
73
74                    // Update metrics
75                    let count = end - start + 1;
76                    let total = counter.fetch_add(count, Ordering::Relaxed) + count;
77                    let elapsed = timer.elapsed().as_secs_f64();
78                    let avg_rate = total as f64 / elapsed;
79                    info!(
80                        processed = total,
81                        secs = elapsed,
82                        rate = avg_rate,
83                        "Avg backfill speed"
84                    );
85
86                    Ok(())
87                }
88            })
89            .buffer_unordered(config.max_concurrency)
90            .try_for_each(|_| async { Ok(()) })
91            .await?;
92
93        let total = processed_counter.load(Ordering::Relaxed);
94        let elapsed = timer.elapsed().as_secs_f64();
95        let final_rate = total as f64 / elapsed;
96        info!(
97            total,
98            secs = elapsed,
99            rate = final_rate,
100            "Completed backfill"
101        );
102
103        Ok(())
104    }
105}

107/// Returns an asynchronous stream that yields consecutive, non-overlapping
108/// subranges ("chunks") from the given inclusive range, each with a maximum
109/// length of `chunk_size`.
110///
111/// This is useful for processing a large range in smaller, manageable pieces,
112/// such as batching database queries or parallelizing work.
113fn chunk_range_stream(
114    total: RangeInclusive<usize>,
115    chunk_size: usize,
116) -> impl Stream<Item = RangeInclusive<usize>> {
117    let end = *total.end();
118    let start = *total.start();
119    unfold(start, move |state| {
120        let end = end;
121        let size = chunk_size;
122        async move {
123            if state > end {
124                None
125            } else {
126                let chunk_end = cmp::min(state + size - 1, end);
127                let next = state + size;
128                Some((state..=chunk_end, next))
129            }
130        }
131    })
132}

134#[cfg(test)]
135mod tests {
136    use futures::{StreamExt, pin_mut};
137
138    use super::*;
139
140    #[tokio::test]
141    async fn test_chunk_range_stream() {
142        let range = 0..=10;
143        let chunk_size = 3;
144        let stream = chunk_range_stream(range, chunk_size);
145        pin_mut!(stream);
146
147        let mut results = vec![];
148        while let Some(chunk) = stream.next().await {
149            results.push(chunk);
150        }
151
152        assert_eq!(results, vec![0..=2, 3..=5, 6..=8, 9..=10]);
153    }
