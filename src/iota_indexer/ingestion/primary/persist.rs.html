28#[derive(Debug)]
29pub(crate) struct CheckpointDataToCommit {
30    pub(crate) checkpoint: IndexedCheckpoint,
31    pub(crate) transactions: Vec<IndexedTransaction>,
32    pub(crate) events: Vec<IndexedEvent>,
33    pub(crate) event_indices: Vec<EventIndex>,
34    pub(crate) tx_indices: Vec<TxIndex>,
35    pub(crate) displays: BTreeMap<String, StoredDisplay>,
36    pub(crate) object_changes: CheckpointObjectChanges,
37    pub(crate) backward_history_changes: Vec<StoredBackwardHistoryObject>,
38    pub(crate) object_versions: Vec<StoredObjectVersion>,
39    pub(crate) packages: Vec<IndexedPackage>,
40    pub(crate) epoch: Option<EpochToCommit>,
41}
42
43#[derive(Clone, Debug, Default)]
44pub struct TransactionObjectChangesToCommit {
45    pub changed_objects: Vec<IndexedObject>,
46    pub deleted_objects: Vec<IndexedDeletedObject>,
47}
48
49#[derive(Clone, Debug)]
50pub struct EpochToCommit {
51    pub(crate) last_epoch: Option<EndOfEpochUpdate>,
52    pub(crate) new_epoch: StartOfEpochUpdate,
53}
54
55pub(crate) struct PrimaryWriter {
56    state: PgIndexerStore,
57    metrics: IndexerMetrics,
58    pub stream: ReadyChunks<ReceiverStream<CheckpointDataToCommit>>,
59    pub checkpoint_commit_batch_size: usize,
60}
61
62impl PrimaryWriter {
63    pub fn new(
64        state: PgIndexerStore,
65        metrics: IndexerMetrics,
66        tx_indexing_receiver: iota_metrics::metered_channel::Receiver<CheckpointDataToCommit>,
67    ) -> Self {
68        let checkpoint_commit_batch_size = std::env::var("CHECKPOINT_COMMIT_BATCH_SIZE")
69            .unwrap_or(CHECKPOINT_COMMIT_BATCH_SIZE.to_string())
70            .parse::<usize>()
71            .unwrap();
72        info!("Using checkpoint commit batch size {checkpoint_commit_batch_size}");
73
74        let stream =
75            ReceiverStream::new(tx_indexing_receiver).ready_chunks(checkpoint_commit_batch_size);
76
77        Self {
78            state,
79            metrics,
80            stream,
81            checkpoint_commit_batch_size,
82        }
83    }
84
85    /// Writes indexed checkpoint data to the database, and then update
86    /// watermark upper bounds and metrics. Expects
87    /// `indexed_checkpoint_batch` to be non-empty, and contain contiguous
88    /// checkpoints. There can be at most one epoch boundary at the end. If
89    /// an epoch boundary is detected, epoch-partitioned tables must be
90    /// advanced.
91    // Unwrap: Caller needs to make sure indexed_checkpoint_batch is not empty
92    #[instrument(skip_all, fields(
93        first = indexed_checkpoint_batch.first().as_ref().unwrap().checkpoint.sequence_number,
94        last = indexed_checkpoint_batch.last().as_ref().unwrap().checkpoint.sequence_number
95    ))]
96    pub(crate) async fn commit_checkpoints(
97        &self,
98        indexed_checkpoint_batch: Vec<CheckpointDataToCommit>,
99        epoch: Option<EpochToCommit>,
100    ) {
101        let batch_len = indexed_checkpoint_batch.len();
102        let mut checkpoint_batch = Vec::with_capacity(batch_len);
103        let mut tx_batch = Vec::with_capacity(batch_len);
104        let mut events_batch = Vec::with_capacity(batch_len);
105        let mut tx_indices_batch = Vec::with_capacity(batch_len);
106        let mut event_indices_batch = Vec::with_capacity(batch_len);
107        let mut displays_batch = Vec::with_capacity(batch_len);
108        let mut object_changes_batch = Vec::with_capacity(batch_len);
109        let mut backward_history_batch = Vec::new();
110        let mut object_versions_batch = Vec::with_capacity(batch_len);
111        let mut packages_batch = Vec::with_capacity(batch_len);
112
113        for indexed_checkpoint in indexed_checkpoint_batch {
114            let CheckpointDataToCommit {
115                checkpoint,
116                transactions,
117                events,
118                event_indices,
119                tx_indices,
120                displays,
121                object_changes,
122                backward_history_changes,
123                object_versions,
124                packages,
125                ..
126            } = indexed_checkpoint;
127            checkpoint_batch.push(checkpoint);
128            tx_batch.push(transactions);
129            events_batch.push(events);
130            tx_indices_batch.push(tx_indices);
131            event_indices_batch.push(event_indices);
132            displays_batch.extend(displays.into_values());
133            object_changes_batch.push(object_changes);
134            backward_history_batch.extend(backward_history_changes);
135            object_versions_batch.push(object_versions);
136            packages_batch.push(packages);
137        }
138
139        let first_checkpoint_seq = checkpoint_batch.first().as_ref().unwrap().sequence_number;
140        let committer_watermark = CommitterWatermark::from(checkpoint_batch.last().unwrap());
141
142        let guard = self.metrics.checkpoint_db_commit_latency.start_timer();
143        let tx_batch = tx_batch.into_iter().flatten().collect::<Vec<_>>();
144
145        let tx_global_order_batch: Vec<_> = tx_batch.iter().map(Into::into).collect();
146        let tx_indices_batch = tx_indices_batch.into_iter().flatten().collect::<Vec<_>>();
147        let events_batch = events_batch.into_iter().flatten().collect::<Vec<_>>();
148        let event_indices_batch = event_indices_batch
149            .into_iter()
150            .flatten()
151            .collect::<Vec<_>>();
152        let object_versions_batch = object_versions_batch
153            .into_iter()
154            .flatten()
155            .collect::<Vec<_>>();
156        let packages_batch = packages_batch.into_iter().flatten().collect::<Vec<_>>();
157        let checkpoint_num = checkpoint_batch.len();
158        let tx_count = tx_batch.len();
159
160        {
161            let _step_1_guard = self
162                .metrics
163                .checkpoint_db_commit_latency_step_1
164                .start_timer();
165            let mut persist_tasks = vec![
166                self.state.persist_transactions(tx_batch),
167                self.state.persist_tx_indices(tx_indices_batch),
168                self.state.persist_events(events_batch),
169                self.state.persist_event_indices(event_indices_batch),
170                self.state.persist_displays(displays_batch),
171                self.state
172                    .persist_packages(packages_batch.into_iter().map(Into::into).collect()),
173                self.state
174                    .persist_object_versions(object_versions_batch.clone()),
175                Box::pin({
176                    let object_changes_batch = object_changes_batch.clone();
177                    async move {
178                        // We need to persist global order before writing objects, so that
179                        // optimistic indexing is blocked from overwriting
180                        // objects table with old tx data reference: https://github.com/iotaledger/iota/issues/10250
181                        self.state
182                            .persist_tx_global_order(tx_global_order_batch.clone())
183                            .await?;
184                        self.state.persist_objects(object_changes_batch).await
185                    }
186                }),
187                // Backward history must be persisted before checkpointed_objects
188                // to prevent read races during consistent view queries.
189                Box::pin({
190                    let object_changes_batch = object_changes_batch.clone();
191                    async move {
192                        self.state
193                            .persist_object_backward_history(backward_history_batch)
194                            .await?;
195                        self.state
196                            .persist_checkpointed_objects(object_changes_batch)
197                            .await
198                    }
199                }),
200            ];
201            if let Some(epoch_data) = epoch.clone() {
202                persist_tasks.push(self.state.persist_epoch(epoch_data));
203            }
204            futures::future::join_all(persist_tasks)
205                .await
206                .into_iter()
207                .map(|res| {
208                    if res.is_err() {
209                        error!("failed to persist data with error: {:?}", res);
210                    }
211                    res
212                })
213                .collect::<IndexerResult<Vec<_>>>()
214                .expect("persisting data into DB should not fail.");
215        }
216
217        let is_epoch_end = epoch.is_some();
218
219        // On epoch boundary, we need to modify the existing partitions' upper bound,
220        // and introduce a new partition for incoming data for the upcoming epoch.
221        if let Some(epoch_data) = epoch {
222            let new_epoch_id = epoch_data.new_epoch.epoch;
223            self.state
224                .advance_epoch(epoch_data)
225                .await
226                .tap_err(|e| {
227                    error!("failed to advance epoch with error: {}", e.to_string());
228                })
229                .expect("advancing epochs in DB should not fail.");
230            self.metrics.last_committed_epoch.set(new_epoch_id);
231
232            // Refresh participation metrics after advancing epoch
233            self.state
234                .refresh_participation_metrics()
235                .await
236                .tap_err(|e| {
237                    error!("failed to update participation metrics: {e}");
238                })
239                .expect("updating participation metrics should not fail.");
240        }
241
242        self.state
243            .persist_checkpoints(checkpoint_batch)
244            .await
245            .tap_err(|e| {
246                error!(
247                    "failed to persist checkpoint data with error: {}",
248                    e.to_string()
249                );
250            })
251            .expect("persisting data into DB should not fail.");
252
253        if is_epoch_end {
254            // The epoch has advanced so we update the configs for the new protocol version,
255            // if it has changed.
256            let chain_id = <PgIndexerStore as IndexerStore>::get_chain_identifier(&self.state)
257                .await
258                .expect("failed to get chain identifier")
259                .expect("chain identifier should have been indexed at this point");
260            let _ = self
261                .state
262                .persist_protocol_configs_and_feature_flags(chain_id);
263        }
264
265        self.state
266            .update_watermarks_upper_bound::<CommitterTables>(committer_watermark)
267            .await
268            .tap_err(|e| {
269                error!(
270                    "Failed to update watermark upper bound with error: {}",
271                    e.to_string()
272                );
273            })
274            .expect("Updating watermark upper bound in DB should not fail.");
275
276        let elapsed = guard.stop_and_record();
277
278        info!(
279            elapsed,
280            "Checkpoint {}-{} committed with {} transactions.",
281            first_checkpoint_seq,
282            committer_watermark.max_committed_cp,
283            tx_count,
284        );
285        self.metrics
286            .latest_tx_checkpoint_sequence_number
287            .set(committer_watermark.max_committed_cp as i64);
288        self.metrics
289            .total_tx_checkpoint_committed
290            .inc_by(checkpoint_num as u64);
291        self.metrics
292            .total_transaction_committed
293            .inc_by(tx_count as u64);
294        self.metrics.transaction_per_checkpoint.observe(
295            tx_count as f64
296                / (committer_watermark.max_committed_cp - first_checkpoint_seq + 1) as f64,
297        );
298        // 1000.0 is not necessarily the batch size, it's to roughly map average tx
299        // commit latency to [0.1, 1] seconds, which is well covered by
300        // DB_COMMIT_LATENCY_SEC_BUCKETS.
301        self.metrics
302            .thousand_transaction_avg_db_commit_latency
303            .observe(elapsed * 1000.0 / tx_count as f64);
304    }
305}
