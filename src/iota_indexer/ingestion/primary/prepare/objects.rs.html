17/// The builder of all object data to commit to the database.
18#[derive(Debug, Clone, Copy)]
19pub(super) struct ObjectsTransformer<'chk> {
20    checkpoint: ValidatedCheckpoint<'chk>,
21}
22
23impl<'chk> ObjectsTransformer<'chk> {
24    pub(super) fn new(checkpoint: ValidatedCheckpoint<'chk>) -> Self {
25        Self { checkpoint }
26    }
27
28    pub(super) fn transform(self, metrics: &IndexerMetrics) -> ObjectData {
29        let checkpoint_objects = {
30            let _timer = metrics.indexing_objects_latency.start_timer();
31            self.checkpoint.into()
32        };
33
34        ObjectData {
35            checkpoint_objects,
36            object_versions: self.build_object_versions(),
37            packages: self.build_packages(metrics),
38            history_objects: self.build_history_objects(),
39        }
40    }
41
42    fn build_packages(self, metrics: &IndexerMetrics) -> Vec<IndexedPackage> {
43        let _timer = metrics.indexing_packages_latency.start_timer();
44        let checkpoint_sequence_number = self.checkpoint.sequence_number();
45        self.checkpoint
46            .iter_transactions()
47            .flat_map(|tx| &tx.output_objects)
48            .filter_map(|object| {
49                let iota_sdk_types::ObjectData::Package(package) = object.data() else {
50                    return None;
51                };
52                Some(IndexedPackage::new(
53                    package.clone(),
54                    checkpoint_sequence_number,
55                ))
56            })
57            .collect()
58    }
59
60    /// Builds backward history entries for a checkpoint.
61    ///
62    /// For each transaction, records the *previous* state of every object that
63    /// was superseded, so that a consistent view at an earlier checkpoint can
64    /// be reconstructed by applying backward diffs to the current
65    /// `checkpointed_objects` snapshot.
66    ///
67    /// The logic is split into three categories:
68    ///
69    /// 1. **Input objects that were mutated or removed** (mutate, delete, wrap): these had an
70    ///    active prior state available in `input_objects` → `ACTIVE` entries with full data.
71    ///
72    /// 2. **Created objects**: did not exist before → `NOT_YET_CREATED`.
73    ///
74    /// 3. **Unwrapped / unwrapped-then-deleted objects**: were previously wrapped so no prior data
75    ///    is available → `WRAPPED_OR_DELETED` with a lamport version approximation.
76    fn build_history_objects(self) -> Vec<StoredBackwardHistoryObject> {
77        let checkpoint_seq = self.checkpoint.sequence_number() as i64;
78        let mut history_objects = Vec::new();
79
80        for tx in self.checkpoint.iter_transactions() {
81            let effects = &tx.effects;
82
83            // 1. Input objects that were mutated or removed (deleted/wrapped) had an active prior
84            //    state — record it from input_objects. Collect the affected IDs so we can iterate
85            //    input_objects once.
86            let superseded_ids: HashSet<ObjectId> = effects
87                .mutated()
88                .into_iter()
89                .map(|mutated| mutated.reference().object_id)
90                .chain(
91                    effects
92                        .all_removed_objects()
93                        .into_iter()
94                        .map(|(r, _)| r.object_id),
95                )
96                .collect();
97
98            for input_obj in &tx.input_objects {
99                if superseded_ids.contains(&input_obj.id()) {
100                    let df_kind = extract_df_kind(input_obj);
101                    let indexed = IndexedObject::from_object(
102                        Some(checkpoint_seq as u64),
103                        input_obj.clone(),
104                        df_kind,
105                    );
106                    history_objects.push(StoredBackwardHistoryObject::try_from(indexed).expect(
107                        "backward history conversion should not fail for active input objects",
108                    ));
109                }
110            }
111
112            // 2. Created objects did not exist before this transaction. Use lamport version
113            //    - 1 so the version is monotonic with other backward-history rows for the
114            //    same object.
115            for created in effects.created() {
116                history_objects.push(StoredBackwardHistoryObject::from_empty(
117                    created.reference().object_id,
118                    created.reference().version.as_u64() as i64 - 1,
119                    ObjectStatus::NotYetCreated,
120                    checkpoint_seq,
121                ));
122            }
123
124            // 3. Unwrapped and unwrapped-then-deleted objects were previously wrapped — no data
125            //    available. Use lamport version - 1 as approximation.
126            let unwrapped_refs = effects
127                .unwrapped()
128                .into_iter()
129                .map(|unwrapped| *unwrapped.reference());
130            let unwrapped_then_deleted_refs = effects.unwrapped_then_deleted().into_iter();
131            for r in unwrapped_refs.chain(unwrapped_then_deleted_refs) {
132                history_objects.push(StoredBackwardHistoryObject::from_empty(
133                    r.object_id,
134                    r.version.as_u64() as i64 - 1,
135                    ObjectStatus::WrappedOrDeleted,
136                    checkpoint_seq,
137                ));
138            }
139        }
140
141        history_objects
142    }
143
144    fn build_object_versions(self) -> Vec<StoredObjectVersion> {
145        let cp_sequence_number = self.checkpoint.sequence_number() as i64;
146        let removed = self
147            .checkpoint
148            .iter_transactions()
149            .flat_map(|tx| tx.removed_object_refs_post_version())
150            .map(|obj_ref| StoredObjectVersion {
151                object_id: obj_ref.object_id.as_bytes().to_vec(),
152                object_version: obj_ref.version.as_u64() as i64,
153                cp_sequence_number,
154            });
155        let output = self
156            .checkpoint
157            .iter_transactions()
158            .flat_map(|tx| &tx.output_objects)
159            .map(|o| StoredObjectVersion {
160                object_id: o.id().as_bytes().to_vec(),
161                object_version: o.version().as_u64() as i64,
162                cp_sequence_number,
163            });
164        removed.chain(output).collect()
165    }
166}

168#[derive(Debug)]
169pub(super) struct ObjectData {
170    pub(super) checkpoint_objects: CheckpointObjectChanges,
171    pub(super) object_versions: Vec<StoredObjectVersion>,
172    pub(super) history_objects: Vec<StoredBackwardHistoryObject>,
173    pub(super) packages: Vec<IndexedPackage>,
174}
