74    async fn process_checkpoint(
75        &self,
76        checkpoint: Arc<CheckpointData>,
77    ) -> Result<Self::Message, Self::Error> {
78        self.metrics
79            .latest_fullnode_checkpoint_sequence_number
80            .set(checkpoint.checkpoint_summary.sequence_number as i64);
81        let time_now_ms = chrono::Utc::now().timestamp_millis();
82        let cp_download_lag = time_now_ms - checkpoint.checkpoint_summary.timestamp_ms as i64;
83        info!(
84            "checkpoint download lag for cp {}: {} ms",
85            checkpoint.checkpoint_summary.sequence_number, cp_download_lag
86        );
87        self.metrics.download_lag_ms.set(cp_download_lag);
88        self.metrics
89            .max_downloaded_checkpoint_sequence_number
90            .set(checkpoint.checkpoint_summary.sequence_number as i64);
91        self.metrics
92            .downloaded_checkpoint_timestamp_ms
93            .set(checkpoint.checkpoint_summary.timestamp_ms as i64);
94        info!(
95            "Indexer lag: downloaded checkpoint {} with time now {} and checkpoint time {}",
96            checkpoint.checkpoint_summary.sequence_number,
97            time_now_ms,
98            checkpoint.checkpoint_summary.timestamp_ms
99        );
100
101        let checkpoint_data = Self::index_checkpoint(
102            &checkpoint,
103            Arc::new(self.metrics.clone()),
104            Self::index_packages(slice::from_ref(&checkpoint), &self.metrics),
105        )
106        .await?;
107        self.indexed_checkpoint_sender
108            .send(checkpoint_data)
109            .await
110            .map_err(|_| {
111                IndexerError::MpscChannel(
112                    "failed to send checkpoint data, receiver half closed".into(),
113                )
114            })?;
115        Ok(())
116    }
117}

130    async fn index_epoch(data: &CheckpointData) -> Result<Option<EpochToCommit>, IndexerError> {
131        let checkpoint_object_store = EpochEndIndexingObjectStore::new(data);
132
133        let CheckpointData {
134            transactions,
135            checkpoint_summary,
136            checkpoint_contents: _,
137        } = data;
138
139        // Genesis epoch
140        if checkpoint_summary.sequence_number() == 0 {
141            info!("Processing genesis epoch");
142            let system_state =
143                get_iota_system_state(&checkpoint_object_store)?.into_iota_system_state_summary();
144            return Ok(Some(EpochToCommit {
145                last_epoch: None,
146                new_epoch: StartOfEpochUpdate::new(
147                    &system_state,
148                    0, // first_checkpoint_id
149                    0, // first_tx_sequence_number
150                    None,
151                ),
152            }));
153        }
154
155        // If not end of epoch, return
156        if checkpoint_summary.end_of_epoch_data.is_none() {
157            return Ok(None);
158        }
159
160        let event = transactions
161            .iter()
162            .find_map(|t| t.events.as_ref().and_then(extract_epoch_info_event));
163
164        let system_state = get_iota_system_state(&checkpoint_object_store)?;
165        if event.is_none() {
166            warn!(
167                "no SystemEpochInfoEvent found at end of epoch {}, some epoch data will be set to default.",
168                checkpoint_summary.epoch,
169            );
170            assert!(
171                system_state.safe_mode(),
172                "iota is not in safe mode but no SystemEpochInfoEvent found at end of epoch {}",
173                checkpoint_summary.epoch
174            );
175        }
176
177        let event = event.unwrap_or_default();
178        let new_epoch_first_checkpoint_id = checkpoint_summary.sequence_number + 1;
179        let new_epoch_first_tx_sequence_number = checkpoint_summary.network_total_transactions;
180        Ok(Some(EpochToCommit {
181            last_epoch: Some(EndOfEpochUpdate::new(checkpoint_summary, &event)),
182            new_epoch: StartOfEpochUpdate::new(
183                &system_state.into_iota_system_state_summary(),
184                new_epoch_first_checkpoint_id,
185                new_epoch_first_tx_sequence_number,
186                Some(&event),
187            ),
188        }))
189    }

191    /// Builds one `objects_version` row per object touched (modified / deleted
192    /// / wrapped / unwrapped-then-deleted) in this checkpoint
193    fn index_object_versions(data: &CheckpointData) -> Vec<StoredObjectVersion> {
194        let cp_sequence_number = data.checkpoint_summary.sequence_number as i64;
195        let removed = data
196            .transactions
197            .iter()
198            .flat_map(|tx| tx.removed_object_refs_post_version())
199            .map(|obj_ref| StoredObjectVersion {
200                object_id: obj_ref.object_id.as_bytes().to_vec(),
201                object_version: obj_ref.version.as_u64() as i64,
202                cp_sequence_number,
203            });
204        let output = data
205            .transactions
206            .iter()
207            .flat_map(|tx| &tx.output_objects)
208            .map(|o| StoredObjectVersion {
209                object_id: o.id().as_bytes().to_vec(),
210                object_version: o.version().as_u64() as i64,
211                cp_sequence_number,
212            });
213        removed.chain(output).collect()
214    }

216    async fn index_checkpoint(
217        data: &CheckpointData,
218        metrics: Arc<IndexerMetrics>,
219        packages: Vec<IndexedPackage>,
220    ) -> Result<CheckpointDataToCommit, IndexerError> {
221        let checkpoint_seq = data.checkpoint_summary.sequence_number;
222        info!(checkpoint_seq, "Indexing checkpoint data blob");
223
224        // Index epoch
225        let epoch = Self::index_epoch(data).await?;
226
227        // Index Objects
228        let object_changes = Self::index_checkpoint_objects(data, &metrics).await?;
229        let object_versions = Self::index_object_versions(data);
230        let backward_history_changes = Self::index_objects_backward_history(data)?;
231
232        let (checkpoint, db_transactions, db_events, db_tx_indices, db_event_indices, db_displays) = {
233            let CheckpointData {
234                transactions,
235                checkpoint_summary,
236                checkpoint_contents,
237            } = data;
238
239            let (db_transactions, db_events, db_tx_indices, db_event_indices, db_displays) =
240                Self::index_transactions(
241                    transactions,
242                    checkpoint_summary,
243                    checkpoint_contents,
244                    &metrics,
245                )
246                .await?;
247
248            let successful_tx_num: u64 = db_transactions.iter().map(|t| t.successful_tx_num).sum();
249            (
250                IndexedCheckpoint::from_iota_checkpoint(
251                    checkpoint_summary,
252                    checkpoint_contents,
253                    successful_tx_num as usize,
254                ),
255                db_transactions,
256                db_events,
257                db_tx_indices,
258                db_event_indices,
259                db_displays,
260            )
261        };
262        let time_now_ms = chrono::Utc::now().timestamp_millis();
263        metrics
264            .index_lag_ms
265            .set(time_now_ms - checkpoint.timestamp_ms as i64);
266        metrics
267            .max_indexed_checkpoint_sequence_number
268            .set(checkpoint.sequence_number as i64);
269        metrics
270            .indexed_checkpoint_timestamp_ms
271            .set(checkpoint.timestamp_ms as i64);
272        info!(
273            "Indexer lag: indexed checkpoint {} with time now {} and checkpoint time {}",
274            checkpoint.sequence_number, time_now_ms, checkpoint.timestamp_ms
275        );
276
277        Ok(CheckpointDataToCommit {
278            checkpoint,
279            transactions: db_transactions,
280            events: db_events,
281            tx_indices: db_tx_indices,
282            event_indices: db_event_indices,
283            displays: db_displays,
284            object_changes,
285            backward_history_changes,
286            object_versions,
287            packages,
288            epoch,
289        })
290    }

292    async fn index_transactions(
293        transactions: &[CheckpointTransaction],
294        checkpoint_summary: &CertifiedCheckpointSummary,
295        checkpoint_contents: &CheckpointContents,
296        metrics: &IndexerMetrics,
297    ) -> IndexerResult<(
298        Vec<IndexedTransaction>,
299        Vec<IndexedEvent>,
300        Vec<TxIndex>,
301        Vec<EventIndex>,
302        BTreeMap<String, StoredDisplay>,
303    )> {
304        let checkpoint_seq = checkpoint_summary.sequence_number();
305
306        let mut tx_seq_num_iter = checkpoint_contents
307            .enumerate_transactions(checkpoint_summary)
308            .map(|(seq, execution_digest)| (execution_digest.transaction, seq));
309
310        if checkpoint_contents.len() != transactions.len() {
311            return Err(IndexerError::FullNodeReading(format!(
312                "checkpointContents has different size {} compared to Transactions {} for checkpoint {checkpoint_seq}",
313                checkpoint_contents.len(),
314                transactions.len()
315            )));
316        }
317
318        let mut db_transactions = Vec::new();
319        let mut db_events = Vec::new();
320        let mut db_displays = BTreeMap::new();
321        let mut db_tx_indices = Vec::new();
322        let mut db_event_indices = Vec::new();
323
324        for tx in transactions {
325            // Unwrap safe - we checked they have equal length above
326            let (tx_digest, tx_sequence_number) = tx_seq_num_iter.next().unwrap();
327            let actual_tx_digest = tx.transaction.digest();
328            if tx_digest != *actual_tx_digest {
329                return Err(IndexerError::FullNodeReading(format!(
330                    "transactions has different ordering from CheckpointContents, for checkpoint {checkpoint_seq}, Mismatch found at {tx_digest} v.s. {actual_tx_digest}",
331                )));
332            }
333
334            let (indexed_tx, tx_indices, indexed_events, events_indices, stored_displays) =
335                Self::index_transaction_components(
336                    tx,
337                    tx_sequence_number,
338                    checkpoint_seq,
339                    checkpoint_summary.timestamp_ms,
340                    metrics,
341                )
342                .await?;
343            db_transactions.push(indexed_tx);
344            db_tx_indices.push(tx_indices);
345            db_events.extend(indexed_events);
346            db_event_indices.extend(events_indices);
347            db_displays.extend(stored_displays);
348        }
349        Ok((
350            db_transactions,
351            db_events,
352            db_tx_indices,
353            db_event_indices,
354            db_displays,
355        ))
356    }

358    pub(crate) async fn index_transaction_components(
359        tx: &CheckpointTransaction,
360        tx_sequence_number: u64,
361        checkpoint_seq: CheckpointSequenceNumber,
362        checkpoint_timestamp_ms: u64,
363        metrics: &IndexerMetrics,
364    ) -> IndexerResult<IndexedTransactionComponents> {
365        let db_txn = Self::index_transaction(
366            tx,
367            tx_sequence_number,
368            checkpoint_seq,
369            checkpoint_timestamp_ms,
370            metrics,
371        )
372        .await?;
373
374        let CheckpointTransaction {
375            transaction: sender_signed_data,
376            effects: fx,
377            events,
378            output_objects,
379            ..
380        } = tx;
381
382        let tx_digest = sender_signed_data.digest();
383        let tx = sender_signed_data.transaction();
384        let events = events.clone().unwrap_or_default();
385
386        let transaction_kind = IotaTransactionKind::from(tx.kind());
387
388        let db_events = events
389            .iter()
390            .enumerate()
391            .map(|(idx, event)| {
392                IndexedEvent::from_event(
393                    tx_sequence_number,
394                    idx as u64,
395                    checkpoint_seq,
396                    *tx_digest,
397                    event,
398                    checkpoint_timestamp_ms,
399                )
400            })
401            .collect();
402
403        let db_event_indices = events
404            .iter()
405            .enumerate()
406            .map(|(idx, event)| EventIndex::from_event(tx_sequence_number, idx as u64, event))
407            .collect();
408
409        let mut db_displays: BTreeMap<String, StoredDisplay> = events
410            .iter()
411            .flat_map(StoredDisplay::try_from_event)
412            .map(|display| (display.object_type.clone(), display))
413            .collect();
414
415        // `display::new` only emits a `DisplayCreated` event with no fields. If an
416        // author creates a `Display<T>` but never calls `update_version`, it wouldn't
417        // get indexed normally. To prevent this, we fall back to the contents of the
418        // created object itself for types with no `VersionUpdated` event in this
419        // transaction.
420        for event in events.iter() {
421            let Some(object_type) = displayed_type_from_created_event(event) else {
422                continue;
423            };
424            if db_displays.contains_key(&object_type) {
425                continue;
426            }
427            if let Some(display) = display_id_from_created_event(event)
428                .and_then(|display_id| {
429                    output_objects
430                        .iter()
431                        .find(|object| object.id() == display_id)
432                })
433                .and_then(StoredDisplay::try_from_object)
434            {
435                db_displays.insert(object_type, display);
436            }
437        }
438
439        // Input Objects
440        let input_objects = tx
441            .input_objects()
442            .expect("committed txns have been validated")
443            .into_iter()
444            .map(|obj_kind| obj_kind.object_id())
445            .collect::<Vec<_>>();
446
447        // Changed Objects
448        let changed_objects = fx
449            .all_changed_objects()
450            .into_iter()
451            .map(
452                |(
453                    OwnedObjectReference {
454                        reference: object_ref,
455                        owner: _owner,
456                    },
457                    _write_kind,
458                )| object_ref.object_id,
459            )
460            .collect::<Vec<_>>();
461
462        // Wrapped or deleted objects
463        let wrapped_or_deleted_objects = fx
464            .all_tombstones()
465            .into_iter()
466            .map(|(object_id, _)| object_id)
467            .chain(fx.created_then_wrapped_objects())
468            .collect::<Vec<_>>();
469
470        // Payers
471        let payers = vec![tx.gas_owner()];
472
473        // Sender
474        let sender = tx.sender();
475
476        // Recipients
477        let recipients = fx
478            .all_changed_objects()
479            .into_iter()
480            .filter_map(
481                |(
482                    OwnedObjectReference {
483                        reference: _object_ref,
484                        owner,
485                    },
486                    _write_kind,
487                )| match owner {
488                    Owner::Address(address) => Some(address),
489                    _ => None,
490                },
491            )
492            .unique()
493            .collect::<Vec<_>>();
494
495        // Move Calls
496        let move_calls = tx
497            .move_calls()
498            .iter()
499            .map(|(p, m, f)| (*<&ObjectId>::clone(p), m.to_string(), f.to_string()))
500            .collect();
501
502        let db_tx_indices = TxIndex {
503            tx_sequence_number,
504            transaction_digest: *tx_digest,
505            checkpoint_sequence_number: checkpoint_seq,
506            input_objects,
507            changed_objects,
508            sender,
509            payers,
510            recipients,
511            move_calls,
512            tx_kind: transaction_kind,
513            wrapped_or_deleted_objects,
514        };
515
516        Ok((
517            db_txn,
518            db_tx_indices,
519            db_events,
520            db_event_indices,
521            db_displays,
522        ))
523    }

525    /// Creates a new [`IndexedTransaction`]
526    pub(crate) async fn index_transaction(
527        tx: &CheckpointTransaction,
528        tx_sequence_number: u64,
529        checkpoint_seq: CheckpointSequenceNumber,
530        checkpoint_timestamp_ms: u64,
531        metrics: &IndexerMetrics,
532    ) -> IndexerResult<IndexedTransaction> {
533        let tx_digest = tx.transaction.digest();
534        let txn = tx.transaction.transaction();
535
536        let events = tx
537            .events
538            .as_ref()
539            .map(|TransactionEvents(events)| events.clone())
540            .unwrap_or_default();
541
542        let transaction_kind = IotaTransactionKind::from(txn.kind());
543
544        let objects = tx
545            .input_objects
546            .iter()
547            .chain(tx.output_objects.iter())
548            .collect::<Vec<_>>();
549
550        let (balance_change, object_changes) = InMemTxChanges::new(&objects, metrics.clone())
551            .get_changes(txn, &tx.effects, tx_digest)
552            .await?;
553
554        Ok(IndexedTransaction {
555            tx_sequence_number,
556            tx_digest: *tx_digest,
557            checkpoint_sequence_number: checkpoint_seq,
558            timestamp_ms: checkpoint_timestamp_ms,
559            sender_signed_data: tx.transaction.data().clone(),
560            successful_tx_num: if tx.effects.status().is_success() {
561                txn.kind().num_transactions() as u64
562            } else {
563                0
564            },
565            effects: tx.effects.clone(),
566            object_changes,
567            balance_change,
568            events,
569            transaction_kind,
570        })
571    }

581    /// Builds backward history entries for a checkpoint.
582    ///
583    /// For each transaction, records the *previous* state of every object that
584    /// was superseded, so that a consistent view at an earlier checkpoint can
585    /// be reconstructed by applying backward diffs to the current
586    /// `checkpointed_objects` snapshot.
587    ///
588    /// The logic is split into three categories:
589    ///
590    /// 1. **Input objects that were mutated or removed** (mutate, delete,
591    ///    wrap): these had an active prior state available in `input_objects` →
592    ///    `ACTIVE` entries with full data.
593    ///
594    /// 2. **Created objects**: did not exist before → `NOT_YET_CREATED`.
595    ///
596    /// 3. **Unwrapped / unwrapped-then-deleted objects**: were previously
597    ///    wrapped so no prior data is available → `WRAPPED_OR_DELETED` with a
598    ///    lamport version approximation.
599    fn index_objects_backward_history(
600        data: &CheckpointData,
601    ) -> Result<Vec<StoredBackwardHistoryObject>, IndexerError> {
602        let checkpoint_seq = data.checkpoint_summary.sequence_number as i64;
603        let mut result = Vec::new();
604
605        for tx in &data.transactions {
606            let effects = &tx.effects;
607
608            // 1. Input objects that were mutated or removed (deleted/wrapped) had an active
609            //    prior state — record it from input_objects. Collect the affected IDs so we
610            //    can iterate input_objects once.
611            let superseded_ids: HashSet<ObjectId> = effects
612                .mutated()
613                .into_iter()
614                .map(|mutated| mutated.reference.object_id)
615                .chain(
616                    effects
617                        .all_removed_objects()
618                        .into_iter()
619                        .map(|(r, _)| r.object_id),
620                )
621                .collect();
622
623            for input_obj in &tx.input_objects {
624                if superseded_ids.contains(&input_obj.id()) {
625                    let df_kind = extract_df_kind(input_obj);
626                    let indexed = IndexedObject::from_object(
627                        Some(checkpoint_seq as u64),
628                        input_obj.clone(),
629                        df_kind,
630                    );
631                    result.push(StoredBackwardHistoryObject::try_from(indexed).expect(
632                        "backward history conversion should not fail for active input objects",
633                    ));
634                }
635            }
636
637            // 2. Created objects did not exist before this transaction. Use lamport version
638            //    - 1 so the version is monotonic with other backward-history rows for the
639            //    same object.
640            for created in effects.created() {
641                result.push(StoredBackwardHistoryObject::from_empty(
642                    created.reference.object_id,
643                    created.reference.version.as_u64() as i64 - 1,
644                    ObjectStatus::NotYetCreated,
645                    checkpoint_seq,
646                ));
647            }
648
649            // 3. Unwrapped and unwrapped-then-deleted objects were previously wrapped — no
650            //    data available. Use lamport version - 1 as approximation.
651            let unwrapped_refs = effects
652                .unwrapped()
653                .into_iter()
654                .map(|unwrapped| unwrapped.reference);
655            let unwrapped_then_deleted_refs = effects.unwrapped_then_deleted().into_iter();
656            for r in unwrapped_refs.chain(unwrapped_then_deleted_refs) {
657                result.push(StoredBackwardHistoryObject::from_empty(
658                    r.object_id,
659                    r.version.as_u64() as i64 - 1,
660                    ObjectStatus::WrappedOrDeleted,
661                    checkpoint_seq,
662                ));
663            }
664        }
665
666        Ok(result)
667    }

669    fn index_packages(
670        checkpoint_data: &[CheckpointData],
671        metrics: &IndexerMetrics,
672    ) -> Vec<IndexedPackage> {
673        let _timer = metrics.indexing_packages_latency.start_timer();
674        checkpoint_data
675            .iter()
676            .flat_map(|data| {
677                let checkpoint_sequence_number = data.checkpoint_summary.sequence_number;
678                data.transactions
679                    .iter()
680                    .flat_map(|tx| &tx.output_objects)
681                    .filter_map(|object| {
682                        let iota_sdk_types::ObjectData::Package(package) = object.data() else {
683                            return None;
684                        };
685                        Some(IndexedPackage::new(
686                            package.clone(),
687                            checkpoint_sequence_number,
688                        ))
689                    })
690                    .collect::<Vec<_>>()
691            })
692            .collect()
693    }

695    pub(crate) fn pg_blocking_cp(state: PgIndexerStore) -> Result<ConnectionPool, IndexerError> {
696        let state_as_any = state.as_any();
697        if let Some(pg_state) = state_as_any.downcast_ref::<PgIndexerStore>() {
698            return Ok(pg_state.blocking_cp());
699        }
700        Err(IndexerError::Uncategorized(anyhow::anyhow!(
701            "failed to downcast state to PgIndexerStore"
702        )))
703    }

706pub struct InMemObjectCache {
707    id_map: HashMap<ObjectId, Object>,
708    seq_map: HashMap<(ObjectId, Version), Object>,
709}
710
711impl InMemObjectCache {
712    pub fn new() -> Self {
713        Self {
714            id_map: HashMap::new(),
715            seq_map: HashMap::new(),
716        }
717    }
718
719    pub fn insert_object(&mut self, obj: Object) {
720        self.id_map.insert(obj.id(), obj.clone());
721        self.seq_map.insert((obj.id(), obj.version()), obj);
722    }
723
724    pub fn get(&self, id: &ObjectId, version: Option<&Version>) -> Option<&Object> {
725        if let Some(version) = version {
726            self.seq_map.get(&(*id, *version))
727        } else {
728            self.id_map.get(id)
729        }
730    }
731}
732
733impl Default for InMemObjectCache {
734    fn default() -> Self {
735        Self::new()
736    }
737}

739/// Along with InMemObjectCache, TxChangesProcessor implements ObjectProvider
740/// so it can be used in indexing write path to get object/balance changes.
741/// Its lifetime is per checkpoint.
742pub struct InMemTxChanges {
743    object_cache: InMemObjectCache,
744    metrics: IndexerMetrics,
745}
746
747impl InMemTxChanges {
748    pub fn new(objects: &[&Object], metrics: IndexerMetrics) -> Self {
749        let mut object_cache = InMemObjectCache::new();
750        for obj in objects {
751            object_cache.insert_object(<&Object>::clone(obj).clone());
752        }
753        Self {
754            object_cache,
755            metrics,
756        }
757    }
758
759    pub(crate) async fn get_changes(
760        &self,
761        tx: &Transaction,
762        effects: &TransactionEffects,
763        tx_digest: &TransactionDigest,
764    ) -> IndexerResult<(Vec<IndexedBalanceChange>, Vec<IndexedObjectChange>)> {
765        let _timer = self
766            .metrics
767            .indexing_tx_object_changes_latency
768            .start_timer();
769        let object_change: Vec<_> = get_object_changes(
770            self,
771            tx.sender(),
772            effects.modified_at_versions(),
773            effects.all_changed_objects(),
774            effects.all_removed_objects(),
775        )
776        .await?
777        .into_iter()
778        .map(IndexedObjectChange::from)
779        .collect();
780        let balance_change = get_balance_changes_from_effect(
781            self,
782            effects,
783            tx.input_objects().unwrap_or_else(|e| {
784                panic!("checkpointed tx {tx_digest} has invalid input objects: {e}")
785            }),
786            None,
787        )
788        .await?
789        .into_iter()
790        .map(IndexedBalanceChange::from)
791        .collect();
792        Ok((balance_change, object_change))
793    }
794}

796#[async_trait]
797impl ObjectProvider for InMemTxChanges {
798    type Error = IndexerError;
799
800    async fn get_object(&self, id: &ObjectId, version: &Version) -> Result<Object, Self::Error> {
801        let object = self
802            .object_cache
803            .get(id, Some(version))
804            .as_ref()
805            .map(|o| <&Object>::clone(o).clone());
806        if let Some(o) = object {
807            self.metrics.indexing_get_object_in_mem_hit.inc();
808            return Ok(o);
809        }
810
811        panic!(
812            "object {id} is not found in TxChangesProcessor as an ObjectProvider (fn get_object)"
813        );
814    }
815
816    async fn find_object_lt_or_eq_version(
817        &self,
818        id: &ObjectId,
819        version: &Version,
820    ) -> Result<Option<Object>, Self::Error> {
821        // First look up the exact version in object_cache.
822        let object = self
823            .object_cache
824            .get(id, Some(version))
825            .as_ref()
826            .map(|o| <&Object>::clone(o).clone());
827        if let Some(o) = object {
828            self.metrics.indexing_get_object_in_mem_hit.inc();
829            return Ok(Some(o));
830        }
831
832        // Second look up the latest version in object_cache. This may be
833        // called when the object is deleted hence the version at deletion
834        // is given.
835        let object = self
836            .object_cache
837            .get(id, None)
838            .as_ref()
839            .map(|o| <&Object>::clone(o).clone());
840        if let Some(o) = object {
841            if o.version() > *version {
842                panic!(
843                    "found a higher version {} for object {id}, expected lt_or_eq {version}",
844                    o.version(),
845                );
846            }
847            if o.version() <= *version {
848                self.metrics.indexing_get_object_in_mem_hit.inc();
849                return Ok(Some(o));
850            }
851        }
852
853        panic!(
854            "object {id} is not found in TxChangesProcessor as an ObjectProvider (fn find_object_lt_or_eq_version)"
855        );
856    }
857}

859/// Represents objects for end-of-epoch indexing.
860/// Used to extract IotaSystemState and its dynamic children for end-of-epoch
861/// indexing.
862pub(crate) struct EpochEndIndexingObjectStore<'a> {
863    objects: Vec<&'a Object>,
864}
865
866impl<'a> EpochEndIndexingObjectStore<'a> {
867    pub fn new(data: &'a CheckpointData) -> Self {
868        Self {
869            objects: data.latest_live_output_objects(),
870        }
871    }
872}
873
874impl iota_types::storage::ObjectStore for EpochEndIndexingObjectStore<'_> {
875    fn try_get_object(
876        &self,
877        object_id: &ObjectId,
878    ) -> Result<Option<Object>, iota_types::storage::error::Error> {
879        Ok(self
880            .objects
881            .iter()
882            .find(|o| o.id() == *object_id)
883            .cloned()
884            .cloned())
885    }
886
887    fn try_get_object_by_key(
888        &self,
889        object_id: &ObjectId,
890        version: iota_types::base_types::VersionNumber,
891    ) -> Result<Option<Object>, iota_types::storage::error::Error> {
892        Ok(self
893            .objects
894            .iter()
895            .find(|o| o.id() == *object_id && o.version() == version)
896            .cloned()
897            .cloned())
898    }
