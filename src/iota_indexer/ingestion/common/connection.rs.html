19/// Resolves the remote checkpoint source from the provided
20/// [`remote_store_url`](IngestionSources::remote_store_url).
21///
22/// Since `remote_store_url` accepts either a fullnode gRPC endpoint or an
23/// object store URL for historical checkpoint data, this function probes the
24/// URL to determine which type it is:
25///
26/// 1. **gRPC health check**: attempts to connect and call `GetHealth`. If
27///    successful, the URL is treated as a fullnode gRPC endpoint.
28/// 2. **Historical manifest fetch**: if gRPC fails, attempts to fetch the
29///    MANIFEST file from the URL as an S3-compatible object store. If
30///    successful, the URL is treated as a historical checkpoint store.
31///
32/// Both probes are retried with exponential backoff within the given timeout.
33/// If neither succeeds, returns an error.
34///
35/// When `live_checkpoints_store_url` is provided, the URL is assumed to be a
36/// historical store and the live URL is included in the
37/// [`RemoteUrl::HybridHistoricalStore`].
38pub async fn resolve_remote_url(
39    ingestion_sources: &IngestionSources,
40    timeout: Duration,
41) -> IndexerResult<Option<RemoteUrl>> {
42    let Some(url) = ingestion_sources
43        .remote_store_url
44        .as_ref()
45        .map(ToString::to_string)
46    else {
47        return Ok(None);
48    };
49
50    let live_url = ingestion_sources
51        .live_checkpoints_store_url
52        .as_ref()
53        .map(ToString::to_string);
54
55    // if live URL is provided, remote-store-url can be assumed as a historical
56    // store
57    if live_url.is_some() {
58        return Ok(Some(RemoteUrl::HybridHistoricalStore {
59            historical_url: url,
60            live_url,
61        }));
62    }
63
64    let backoff = ExponentialBackoff {
65        max_elapsed_time: Some(timeout),
66        multiplier: 2.0,
67        ..Default::default()
68    };
69
70    backoff::future::retry(backoff, || {
71        let url = url.clone();
72        async move {
73            let grpc_result = async {
74                let client = GrpcClient::new(url.clone())?;
75                client.get_health(None).await
76            }
77            .await
78            .inspect_err(|e| debug!("gRPC health check failed: {e}"));
79
80            if grpc_result.is_ok() {
81                info!("resolved remote store as fullnode gRPC: {url}");
82                return Ok(Some(RemoteUrl::Fullnode(url)));
83            }
84
85            // we use a lightweight S3 client to check if the MANIFEST file exists.
86            // we avoid HistoricalReader here as its internal manifest fetch retries
87            // with a 15-minute default backoff and does not have a timeout.
88            let store =
89                create_remote_store_client(url.clone(), Default::default(), timeout.as_secs())
90                    .map_err(|e| {
91                        debug!("failed to create historical store client: {e}");
92                        backoff::Error::transient(IndexerError::Generic(format!(
93                            "remote store not reachable: {url}"
94                        )))
95                    })?;
96
97            store.head(&Manifest::file_path()).await.map_err(|e| {
98                debug!("historical store MANIFEST not found: {e}");
99                backoff::Error::transient(IndexerError::Generic(format!(
100                    "remote store not reachable: {url}"
101                )))
102            })?;
103
104            info!("resolved remote store as historical object store: {url}");
105            Ok(Some(RemoteUrl::HybridHistoricalStore {
106                historical_url: url,
107                live_url: None,
108            }))
109        }
110    })
111    .await
112    .map_err(|_: IndexerError| IndexerError::Generic(format!(
113        "failed to resolve remote store '{url}' after {timeout:?}: not reachable as gRPC or historical store"
114    )))
115}
