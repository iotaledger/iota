39/// Data derived from the live-object set included in the snapshot.
40#[derive(Default)]
41struct ObjectDerivedData {
42    hasher: Sha3_256,
43    displays: BTreeMap<String, StoredDisplay>,
44    packages: Vec<StoredPackage>,
45    object_versions: Vec<StoredObjectVersion>,
46}
47
48impl ObjectDerivedData {
49    fn extend(&mut self, object: &Object, checkpoint_sequence_number: u64) {
50        self.hasher.update(object.object_ref().digest.bytes());
51        if let Some(display) = StoredDisplay::try_from_object(object) {
52            self.displays.insert(display.object_type.clone(), display);
53        }
54        if let iota_sdk_types::ObjectData::Package(package) = object.data() {
55            self.packages
56                .push(IndexedPackage::new(package.clone(), checkpoint_sequence_number).into());
57        }
58        self.object_versions.push(StoredObjectVersion {
59            object_id: object.id().as_bytes().to_vec(),
60            object_version: object.version().as_u64() as i64,
61            cp_sequence_number: checkpoint_sequence_number as i64,
62        })
63    }
64}

66impl Restore for PgIndexerStore {
67    async fn insert_partition(
68        &self,
69        file_metadata: FileMetadata,
70        bytes: Bytes,
71        expected_checksum: &[u8; SHA3_BYTES],
72    ) -> anyhow::Result<()> {
73        let mut derived_data = ObjectDerivedData::default();
74        let partition = LiveObjectIter::new(&file_metadata, bytes)?.scan(
75            &mut derived_data,
76            |derived_data, snapshot_object| {
77                let SnapshotObject {
78                    object,
79                    previous_transaction_checkpoint,
80                } = snapshot_object;
81                let checkpoint_sequence_number =
82                    previous_transaction_checkpoint.unwrap_or_default();
83                derived_data.extend(&object, checkpoint_sequence_number);
84                Some(LiveObject::new(checkpoint_sequence_number, object))
85            },
86        );
87        let chunks = chunk!(partition, self.config.parallel_objects_chunk_size);
88        let sha3_digest = derived_data.hasher.finalize().digest;
89        if *expected_checksum != sha3_digest {
90            tracing::error!(
91                "sha does not match! expected: {expected_checksum:?}, actual: {sha3_digest:?}",
92            );
93            anyhow::bail!(
94                "checksum verification failed for bucket/partition: {}/{}",
95                file_metadata.bucket_num,
96                file_metadata.part_num
97            );
98        }
99
100        let (live_chunks, checkpointed_chunks): (Vec<_>, Vec<_>) = chunks
101            .into_iter()
102            .map(|c| {
103                let checkpointed = c
104                    .iter()
105                    .map(|live| StoredCheckpointedObject::try_from(live.indexed_object.clone()))
106                    .collect::<Result<Vec<_>, IndexerError>>()?;
107                Ok::<_, IndexerError>((c, checkpointed))
108            })
109            .collect::<Result<Vec<_>, _>>()?
110            .into_iter()
111            .unzip();
112
113        let live_tasks = live_chunks
114            .into_iter()
115            .map(|c| self.spawn_blocking_task(move |this| this.persist_live_objects(c)));
116        futures::future::try_join_all(live_tasks)
117            .await
118            .map_err(|e| {
119                tracing::error!(
120                    "failed to join futures for persisting formal snapshot partition: {e}"
121                );
122                IndexerError::from(e)
123            })?
124            .into_iter()
125            .collect::<Result<Vec<_>, _>>()
126            .map_err(|e| {
127                IndexerError::PostgresWrite(format!(
128                    "failed to persist all formal snapshot object chunks: {e:?}",
129                ))
130            })?;
131
132        let checkpointed_tasks = checkpointed_chunks.into_iter().map(|c| {
133            self.spawn_blocking_task(move |this| this.persist_checkpointed_objects_chunk(c))
134        });
135        futures::future::try_join_all(checkpointed_tasks)
136            .await
137            .map_err(|e| {
138                tracing::error!(
139                    "failed to join futures for persisting formal snapshot partition to checkpointed_objects: {e}"
140                );
141                IndexerError::from(e)
142            })?
143            .into_iter()
144            .collect::<Result<Vec<_>, _>>()
145            .map_err(|e| {
146                IndexerError::PostgresWrite(format!(
147                    "failed to persist all formal snapshot checkpointed object chunks: {e:?}",
148                ))
149            })?;
150        self.persist_displays(derived_data.displays.into_values().collect())
151            .await?;
152        self.persist_packages(derived_data.packages).await?;
153        self.persist_object_versions(derived_data.object_versions)
154            .await?;
155        Ok(())
156    }
157}

159impl EpochToCommit {
160    /// Builds the epoch commits for a verified snapshot.
161    ///
162    /// Returns one commit per epoch from 0 through `snapshot_epoch + 1`, in
163    /// epoch order. Each entry closes its own epoch and opens the next, exactly
164    /// as live ingestion does at an end-of-epoch checkpoint. The leading commit
165    /// opens the genesis epoch.
166    pub(crate) fn batch_from_verified_epoch_info(
167        verified_epoch_info: VerifiedEpochInfo,
168    ) -> Vec<Self> {
169        let (epoch_info, _committees, start_system_states) = verified_epoch_info.into_parts();
170        let epoch_info_entries = epoch_info.into_entries();
171
172        // Initialize with the genesis epoch to commit.
173        let genesis_system_state = start_system_states[0]
174            .clone()
175            .into_iota_system_state_summary();
176        let mut epochs_to_commit = vec![EpochToCommit {
177            last_epoch: None,
178            new_epoch: StartOfEpochUpdate::new(&genesis_system_state, 0, 0, None),
179        }];
180
181        for (epoch, epoch_info_entry) in epoch_info_entries.iter().enumerate() {
182            let event = extract_epoch_info_event(&epoch_info_entry.end_of_epoch_tx_events)
183                .unwrap_or_default();
184            let last_checkpoint_summary = &epoch_info_entry.last_checkpoint_summary;
185            let new_epoch_system_state = start_system_states[epoch + 1]
186                .clone()
187                .into_iota_system_state_summary();
188            let new_epoch_first_checkpoint_id = last_checkpoint_summary.sequence_number() + 1;
189            let new_epoch_first_tx_sequence_number =
190                last_checkpoint_summary.network_total_transactions;
191            epochs_to_commit.push(EpochToCommit {
192                last_epoch: Some(EndOfEpochUpdate::new(last_checkpoint_summary, &event)),
193                new_epoch: StartOfEpochUpdate::new(
194                    &new_epoch_system_state,
195                    new_epoch_first_checkpoint_id,
196                    new_epoch_first_tx_sequence_number,
197                    Some(&event),
198                ),
199            });
200        }
201        epochs_to_commit
202    }
203}

205async fn populate_chain_id(store: &PgIndexerStore, chain_id: ChainIdentifier) -> IndexerResult<()> {
206    let checkpoint_digest = chain_id.digest().into_bytes().to_vec();
207    store
208        .execute_in_blocking_worker(|this| {
209            this.persist_chain_identifier(StoredChainIdentifier { checkpoint_digest })
210        })
211        .await
212}

214async fn populate_protocol_and_feature_flags(
215    store: &PgIndexerStore,
216    chain_id: ChainIdentifier,
217) -> IndexerResult<()> {
218    let checkpoint_digest = chain_id.digest().into_bytes().to_vec();
219    store

221            this.persist_protocol_configs_and_feature_flags(checkpoint_digest)
222        })
223        .await
224}

226async fn populate_epochs(
227    store: &PgIndexerStore,
228    verified_epoch_info: VerifiedEpochInfo,
229) -> IndexerResult<()> {
230    let epochs_to_commit = EpochToCommit::batch_from_verified_epoch_info(verified_epoch_info);
231    store
232        .execute_in_blocking_worker(move |this| this.persist_epochs(epochs_to_commit))
233        .await
234}

236const EPOCH_PARTITIONED_TABLES: [&str; 2] = ["transactions", "events"];
237
238/// Moves the initial partition of `transactions`, and `events` to `epoch`.
239///
240/// The migrations create a single `<table>_partition_0` that eludes the pruner,
241/// and grows indefinitely.
242///
243/// Herein, that partition is dropped, and the partition of the restore epoch
244/// is created instead with the respective lower bound.
245///
246/// # Errors
247///
248/// Returns an error if the database rejects the partition restore.
249async fn restore_partitions(
250    store: &PgIndexerStore,
251    epoch: u64,
252    epoch_start_tx: u64,
253) -> IndexerResult<()> {
254    store
255        .execute_in_blocking_worker(move |this| {
256            let pool = this.blocking_cp();
257            transactional_blocking_with_retry!(
258                &pool,
259                |conn| {
260                    let query = EPOCH_PARTITIONED_TABLES
261                        .iter()
262                        .map(|table| {
263                            format!(
264                                "DROP TABLE {table}_partition_0;
265                                 CREATE TABLE {table}_partition_{epoch} PARTITION OF {table}
266                                     FOR VALUES FROM ({epoch_start_tx}) TO (MAXVALUE);"
267                            )
268                        })
269                        .join("\n");
270                    conn.batch_execute(&query)?;
271                    Ok::<(), IndexerError>(())
272                },
273                Duration::from_secs(10)
274            )?;
275            tracing::info!("Moved the initial epoch partitions to epoch {epoch}");
276            Ok(())
277        })
278        .await
279}

281/// We populate the remaining tables after the objects.
282///
283/// This includes `epochs`, `chain_identifier` which we populate in parallel
284/// with setting the checkpoint watermark to the last checkpoint of the snapshot
285/// epoch.
286///
287/// We also move the initial partition of `transactions`, and `events` to the
288/// epoch that follows the snapshot.
289///
290/// Finally we populate `protocol_configs`, `feature_flags` up to the
291/// protocol version that the snapshot epoch corresponds to, and the
292/// `watermarks` table with the lower bounds associated with the epoch following
293/// the snapshot.
294pub(crate) async fn populate_remaining_tables(
295    store: &PgIndexerStore,
296    verified_epoch_info: VerifiedEpochInfo,
297    snapshot_chain_id: ChainIdentifier,
298) -> IndexerResult<()> {
299    let snapshot_epoch_boundary = &verified_epoch_info
300        .entries()
301        .last()
302        .expect("there should be an entry for the snapshot epoch");
303    let sync_watermark = IndexedCheckpoint::from_iota_checkpoint(
304        &snapshot_epoch_boundary.last_checkpoint_summary,
305        &snapshot_epoch_boundary.last_checkpoint_contents,
306        Default::default(), // We don't store this as part of the checkpoint so it's ok to set to 0
307    );
308    let next_epoch = sync_watermark.epoch + 1;
309    let next_epoch_start_tx = sync_watermark.network_total_transactions;
310    let pruning_watermarks: Vec<_> = PrunableTable::iter()
311        .map(|table| (table, next_epoch))
312        .collect();
313    tokio::try_join!(
314        populate_epochs(store, verified_epoch_info),
315        populate_chain_id(store, snapshot_chain_id),
316        store.persist_checkpoints(vec![sync_watermark]),
317        restore_partitions(store, next_epoch, next_epoch_start_tx),
318    )?;
319    tokio::try_join!(
320        populate_protocol_and_feature_flags(store, snapshot_chain_id),
321        store.update_watermarks_lower_bound(pruning_watermarks.clone()),
322        // The restore effectively folds the object_versions table to the
323        // set of objects that were live at the end of the target epoch.
324        //
325        // The next epoch is the one where the table starts getting filled with historical
326        // versions again. Hence we use the respective lower bounds to represent that point in the
327        // history of the table.
328        store.update_watermarks_lower_bound(vec![(CommitterTables::ObjectsVersion, next_epoch)]),
329    )?;
330    // Finally align the lowest unpruned key with the lower bounds
331    // to let the pruner know the pruning range start after restoring
332    let (stored_watermarks, _) = store.get_watermarks().await?;
333    let lowest_unpruned_keys = stored_watermarks
334        .iter()
335        .filter_map(|watermark| {
336            let table = PrunableTable::try_from(watermark).ok()?;
337            Some((table, table.pruning_strategy().range_end(watermark)))
338        })
339        .collect::<Vec<_>>();
340    tokio::try_join!(
341        store.update_watermarks_lowest_unpruned_key(lowest_unpruned_keys),
342        // Same as for the lower bounds, we set the lowest_unpruned_key to
343        // the next epoch.
344        store.update_watermarks_lowest_unpruned_key(vec![(
345            CommitterTables::ObjectsVersion,
346            next_epoch
347        )]),
348    )?;
349    Ok(())
