15pub(crate) async fn verify_epoch_info(
16    epoch_info: EpochInfo,
17    genesis: Genesis,
18    snapshot_chain_id: ChainIdentifier,
19) -> IndexerResult<VerifiedEpochInfo> {
20    let genesis_committee = genesis.committee().expect("genesis committee");
21    let genesis_system_state = genesis.iota_system_object();
22    let genesis_chain_id = ChainIdentifier::from(*genesis.checkpoint().digest());
23    tokio::task::spawn_blocking(move || {
24        verify_epoch_info_chain(
25            epoch_info,
26            genesis_committee,
27            genesis_system_state,
28            snapshot_chain_id,
29            genesis_chain_id,
30        )
31    })
32    .await?
33    .map_err(|e| IndexerError::Restore(format!("snapshot verification failed: {e}")))
34}

36/// Verifies the root state hash evaluated from the formal snapshot.
37///
38/// This is done by comparing the value against the verified commitment of the
39/// snapshot epoch.
40///
41/// Returns the number of live objects accumulated.
42///
43/// # Errors
44///
45/// Returns an error if:
46///
47/// - State hash verification fails.
48/// - The restore stops before accumulation completes.
49/// - The verified checkpoint carries no end-of-epoch commitment.
50/// - The accumulated root state hash does not match that commitment.
51pub(crate) async fn verify_state_hash(
52    state_hash_rx: mpsc::Receiver<(GlobalStateHash, u64)>,
53    accumulation_done_rx: oneshot::Receiver<()>,
54    verified_epoch_info: &VerifiedEpochInfo,
55) -> IndexerResult<u64> {
56    // The partial hashes must be drained before the completion signal is
57    // awaited: the reader's accumulation tasks block on the bounded channel
58    // and only signal completion after the last of them, so awaiting
59    // completion first would deadlock the restore.
60    let (root_state_hash, num_objects) = accumulate_state_hash(state_hash_rx).await;
61    // A dropped completion sender means the restore failed before every
62    // partial hash was sent — the accumulated hash is incomplete, and
63    // comparing it below would report a misleading mismatch.
64    if accumulation_done_rx.await.is_err() {
65        return Err(IndexerError::Restore(
66            "snapshot accumulation did not complete".to_string(),
67        ));
68    }
69
70    let last_checkpoint_summary = &verified_epoch_info
71        .entries()
72        .last()
73        .expect("there should be an entry for the associated epoch")
74        .last_checkpoint_summary;
75    let commitment = last_checkpoint_summary
76        .end_of_epoch_data
77        .as_ref()
78        .and_then(|end_of_epoch| end_of_epoch.epoch_commitments.last())
79        .ok_or_else(|| {
80            IndexerError::Restore("verified checkpoint has no end-of-epoch commitment".to_string())
81        })?;
82    let verified_digest = commitment.as_ecmh_live_object_set_digest();
83    let root_state_hash = ECMHLiveObjectSetDigest::from(root_state_hash.digest()).digest;
84    if *verified_digest != root_state_hash {
85        return Err(IndexerError::Restore(format!(
86            "root state hash {root_state_hash:?} does not match the verified commitment \
87             {verified_digest:?}"
88        )));
89    }
90    Ok(num_objects)
91}

93/// Evaluates the root state hash of the live object set stored in a formal
94/// snapshot.
95///
96/// This is done by accumulating the partial state hashes received during a
97/// restore.
98///
99/// The total number of objects found in the live set is returned along with the
100/// root state hash.
101async fn accumulate_state_hash(
102    mut state_hash_rx: mpsc::Receiver<(GlobalStateHash, u64)>,
103) -> (GlobalStateHash, u64) {
104    let mut root_state_hash = GlobalStateHash::default();
105    let mut num_objects = 0u64;
106    while let Some((partial_hash, count)) = state_hash_rx.recv().await {
107        root_state_hash.union(&partial_hash);
108        num_objects += count;
109    }
110    (root_state_hash, num_objects)
111}
