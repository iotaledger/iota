4//! Conversion utilities for historical fallback data.
5//!
6//! This module provides wrapper types that enable conversion from raw data
7//! fetched from historical fallback storage into the `Stored*` or JSON-RPC
8//! compatible types used by the Indexer's JSON-RPC API layer.

37/// Alias for an [`Object`] fetched from historical fallback storage.
38///
39/// Contains all data needed to reconstruct a [`StoredObject`].
40pub(crate) type HistoricalFallbackObject = Object;
41
42/// Alias for [`CertifiedCheckpointSummary`] with its [`CheckpointContents`]
43/// data fetched from historical fallback storage.
44///
45/// Contains all data needed to reconstruct a [`StoredCheckpoint`].
46pub(crate) type HistoricalFallbackCheckpoint = (CertifiedCheckpointSummary, CheckpointContents);
47
48impl From<HistoricalFallbackObject> for StoredObject {
49    fn from(object: HistoricalFallbackObject) -> Self {
50        let df_kind = extract_df_kind(&object);
51        let indexed = IndexedObject::from_object(None, object, df_kind);
52        StoredObject::from(indexed)
53    }
54}
55
56impl From<HistoricalFallbackCheckpoint> for StoredCheckpoint {
57    fn from(checkpoint: HistoricalFallbackCheckpoint) -> Self {
58        let (checkpoint_summary, checkpoint_contents) = checkpoint;
59        // StoredCheckpoint::from implementation does not use the `successful_tx_num`
60        // param in IndexedCheckpoint::from_iota_checkpoint, in this regard it is safe
61        // to hardcode to 0.
62        let indexed =
63            IndexedCheckpoint::from_iota_checkpoint(&checkpoint_summary, &checkpoint_contents, 0);
64        StoredCheckpoint::from(&indexed)
65    }
66}

68/// Wrapper for [`TransactionEvents`] and additional data fetched from
69/// historical fallback storage.
70///
71/// Contains all data needed to reconstruct [`IotaEvent`]s or [`StoredEvent`]s.
72#[derive(Debug, Clone)]
73pub struct HistoricalFallbackEvents {
74    /// Events emitted during transaction execution.
75    events: TransactionEvents,
76    /// Digest of the transaction that emitted the events.
77    tx_digest: TransactionDigest,
78    /// Sequence number of the transaction that emitted the events.
79    tx_sequence_number: u64,
80    /// Sequence number of the checkpoint the transaction is part of.
81    checkpoint_sequence_number: CheckpointSequenceNumber,
82    /// Checkpoint timestamp.
83    timestamp: u64,
84}
85
86impl HistoricalFallbackEvents {
87    /// Creates the wrapper from the events of the `tx_digest` transaction and
88    /// the checkpoint the transaction is part of.
89    ///
90    /// # Errors
91    ///
92    /// Returns [`IndexerError::HistoricalFallbackStorageError`] when
93    /// transaction is not part of the provided checkpoint.
94    pub fn new(
95        events: TransactionEvents,
96        tx_digest: TransactionDigest,
97        checkpoint_summary: &CertifiedCheckpointSummary,
98        checkpoint_contents: &CheckpointContents,
99    ) -> IndexerResult<Self> {
100        let Some(tx_sequence_number) = checkpoint_contents
101            .enumerate_transactions(checkpoint_summary)
102            .find(|(_, execution_digest)| execution_digest.transaction == tx_digest)
103            .map(|(seq, _)| seq)
104        else {
105            return Err(IndexerError::HistoricalFallbackStorageError(format!(
106                "cannot find transaction sequence number to transaction: {tx_digest}"
107            )));
108        };
109
110        Ok(Self {
111            events,
112            tx_digest,
113            tx_sequence_number,
114            checkpoint_sequence_number: checkpoint_summary.sequence_number,
115            timestamp: checkpoint_summary.timestamp_ms,
116        })
117    }
118
119    /// Converts the raw [`TransactionEvents`] into JSON RPC compatible
120    /// [`IotaEvent`]s.
121    pub(crate) async fn into_iota_events(
122        self,
123        package_resolver: &Arc<Resolver<impl PackageStore>>,
124    ) -> IndexerResult<Vec<IotaEvent>> {
125        tx_events_to_iota_tx_events(
126            self.events,
127            package_resolver,
128            self.tx_digest,
129            Some(self.timestamp),
130        )
131        .await
132        .map(|tx_block_event| tx_block_event.data)
133    }
134
135    /// Converts the raw [`TransactionEvents`] into [`StoredEvent`]s.
136    pub(crate) fn into_stored_events(self) -> Vec<StoredEvent> {
137        self.events
138            .iter()
139            .enumerate()
140            .map(|(idx, event)| {
141                StoredEvent::from(IndexedEvent::from_event(
142                    self.tx_sequence_number,
143                    idx as u64,
144                    self.checkpoint_sequence_number,
145                    self.tx_digest,
146                    event,
147                    self.timestamp,
148                ))
149            })
150            .collect()
151    }
152}

154/// Wrapper for a complete transaction fetched from historical fallback storage.
155///
156/// Contains all data needed to reconstruct a [`StoredTransaction`].
157#[derive(Debug, Clone)]
158pub struct HistoricalFallbackTransaction {
159    /// Checkpointed transaction data.
160    checkpoint_transaction: CheckpointTransaction,
161    /// Checkpoint sequence number the transaction is part of.
162    historical_checkpoint: HistoricalFallbackCheckpoint,
163}
164
165impl HistoricalFallbackTransaction {
166    pub fn new(
167        checkpoint_transaction: CheckpointTransaction,
168        historical_checkpoint: HistoricalFallbackCheckpoint,
169    ) -> Self {
170        Self {
171            checkpoint_transaction,
172            historical_checkpoint,
173        }
174    }
175
176    /// Converts the historical fallback transaction into a
177    /// [`StoredTransaction`].
178    pub(crate) async fn into_stored_transaction(self) -> IndexerResult<StoredTransaction> {
179        let tx_digest = self.checkpoint_transaction.transaction.digest();
180        let (summary, contents) = self.historical_checkpoint;
181
182        let Some(tx_sequence_number) = contents
183            .enumerate_transactions(&summary)
184            .find(|(_seq, execution_digest)| &execution_digest.transaction == tx_digest)
185            .map(|(seq, _execution_digest)| seq)
186        else {
187            return Err(IndexerError::HistoricalFallbackStorageError(format!(
188                "cannot find transaction sequence number to transaction: {tx_digest}"
189            )));
190        };
191
192        let indexed_tx = PrimaryWorker::index_transaction(
193            &self.checkpoint_transaction,
194            tx_sequence_number,
195            summary.sequence_number,
196            summary.timestamp_ms,
197            &IndexerMetrics::new(&Registry::new()),
198        )
199        .await?;
200
201        Ok(StoredTransaction::from(&indexed_tx))
202    }
203}
