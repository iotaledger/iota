69    async fn object_read_to_object_response(
70        &self,
71        object_read: ObjectRead,
72        options: IotaObjectDataOptions,
73    ) -> RpcResult<IotaObjectResponse> {
74        match object_read {
75            ObjectRead::NotExists(id) => Ok(IotaObjectResponse::new_with_error(
76                IotaObjectResponseError::NotExists { object_id: id },
77            )),
78            ObjectRead::Exists(object_ref, o, layout) => {
79                let mut display_fields = None;
80                if options.show_display {
81                    match self.inner.get_rendered_display_fields(&o, &layout).await {
82                        Ok(rendered_fields) => display_fields = Some(rendered_fields),
83                        Err(e) => {
84                            return Ok(IotaObjectResponse::new(
85                                Some(
86                                    IotaObjectData::new(object_ref, o, layout, &options, None)
87                                        .map_err(internal_error)?,
88                                ),
89                                Some(IotaObjectResponseError::Display {
90                                    error: e.to_string(),
91                                }),
92                            ));
93                        }
94                    }
95                }
96                Ok(IotaObjectResponse::new_with_data(
97                    IotaObjectData::new(object_ref, o, layout, &options, display_fields)
98                        .map_err(internal_error)?,
99                ))
100            }
101            ObjectRead::Deleted(object_ref) => Ok(IotaObjectResponse::new_with_error(
102                IotaObjectResponseError::Deleted {
103                    object_id: object_ref.object_id,
104                    version: object_ref.version.into(),
105                    digest: object_ref.digest,
106                },
107            )),
108        }
109    }

111    async fn past_object_read_to_response(
112        &self,
113        options: Option<IotaObjectDataOptions>,
114        past_object_read: PastObjectRead,
115    ) -> RpcResult<IotaPastObjectResponse> {
116        let options = options.unwrap_or_default();
117
118        match past_object_read {
119            PastObjectRead::ObjectNotExists(id) => Ok(IotaPastObjectResponse::ObjectNotExists(id)),
120
121            PastObjectRead::ObjectDeleted(object_ref) => {
122                Ok(IotaPastObjectResponse::ObjectDeleted(object_ref))
123            }
124
125            PastObjectRead::VersionFound(object_ref, object, layout) => {
126                let display_fields = if options.show_display {
127                    let rendered_fields = self
128                        .inner
129                        .get_rendered_display_fields(&object, &layout)
130                        .await
131                        .map_err(internal_error)?;
132
133                    Some(rendered_fields)
134                } else {
135                    None
136                };
137
138                Ok(IotaPastObjectResponse::VersionFound(
139                    IotaObjectData::new(object_ref, object, layout, &options, display_fields)
140                        .map_err(internal_error)?,
141                ))
142            }
143
144            PastObjectRead::VersionNotFound(object_id, version) => Ok(
145                IotaPastObjectResponse::VersionNotFound(object_id, version.into()),
146            ),
147
148            PastObjectRead::VersionTooHigh {
149                object_id,
150                asked_version,
151                latest_version,
152            } => Ok(IotaPastObjectResponse::VersionTooHigh {
153                object_id,
154                asked_version: asked_version.into(),
155                latest_version: latest_version.into(),
156            }),
157        }
158    }

160    /// Checks if the transaction is indexed on node through fullnode gRPC API.
161    async fn is_transaction_indexed_on_node(
162        &self,
163        digest: TransactionDigest,
164    ) -> IndexerResult<bool> {
165        let result = self
166            .fullnode_grpc_client
167            .get_transactions([digest], TransactionField::TRANSACTION_DIGEST)
168            .await?
169            .into_inner()
170            .pop()
171            .ok_or_else(|| IndexerError::Grpc("there should be one tx lookup response".into()))?;
172
173        match result {
174            Ok(executed_tx) => Ok(executed_tx.transaction()?.digest()? == digest),
175            // The node reports a transaction it has not indexed as a per-digest
176            // `NOT_FOUND`.
177            Err(e) if e.is_not_found() => Ok(false),
178            Err(e) => Err(IndexerError::from(e)),
179        }
180    }

184impl ReadApiServer for ReadApi {
185    async fn get_object(
186        &self,
187        object_id: ObjectId,
188        options: Option<IotaObjectDataOptions>,
189    ) -> RpcResult<IotaObjectResponse> {
190        let object_read = self
191            .inner
192            .get_object_read_in_blocking_task(object_id)
193            .await?;
194        self.object_read_to_object_response(object_read, options.unwrap_or_default())
195            .await
196    }

198    async fn multi_get_objects(
199        &self,
200        object_ids: Vec<ObjectId>,
201        options: Option<IotaObjectDataOptions>,
202    ) -> RpcResult<Vec<IotaObjectResponse>> {
203        if object_ids.len() > *QUERY_MAX_RESULT_LIMIT {
204            return Err(
205                IotaRpcInputError::SizeLimitExceeded(QUERY_MAX_RESULT_LIMIT.to_string()).into(),
206            );
207        }
208
209        // Doesn't take care of missing objects.
210        let stored_objects = self
211            .inner
212            .multi_get_objects_in_blocking_task(object_ids.clone())
213            .await?;
214
215        // Map the returned `StoredObject`s to `ObjectId`
216        let object_map: Arc<HashMap<ObjectId, StoredObject>> = Arc::new(
217            stored_objects
218                .into_iter()
219                .map(|obj| {
220                    let object_id = ObjectId::from_bytes(obj.object_id.clone()).map_err(|_| {
221                        IndexerError::PersistentStorageDataCorruption(format!(
222                            "failed to parse ObjectId: {:?}",
223                            obj.object_id
224                        ))
225                    })?;
226                    Ok::<(ObjectId, StoredObject), IndexerError>((object_id, obj))
227                })
228                .collect::<Result<_, IndexerError>>()?,
229        );
230
231        let options = options.unwrap_or_default();
232        let resolver = self.inner.package_resolver();
233
234        // Create a future for each requested object id
235        let futures = object_ids.into_iter().map(|object_id| {
236            let options = options.clone();
237            let maybe_stored = object_map.get(&object_id).cloned();
238            async move {
239                match maybe_stored {
240                    Some(stored) => {
241                        let object_read = stored.try_into_object_read(resolver).await?;
242                        self.object_read_to_object_response(object_read, options)
243                            .await
244                    }
245                    None => {
246                        self.object_read_to_object_response(
247                            ObjectRead::NotExists(object_id),
248                            options,
249                        )
250                        .await
251                    }
252                }
253            }
254        });
255
256        futures::future::try_join_all(futures).await
257    }

341    async fn try_multi_get_past_objects(
342        &self,
343        past_objects: Vec<IotaGetPastObjectRequest>,
344        options: Option<IotaObjectDataOptions>,
345    ) -> RpcResult<Vec<IotaPastObjectResponse>> {
346        let mut responses = Vec::with_capacity(past_objects.len());
347
348        for request in past_objects {
349            let past_object_read = self
350                .inner
351                .get_past_object_read_with_fallback(request.object_id, request.version, false)
352                .await?;
353
354            responses.push(
355                self.past_object_read_to_response(options.clone(), past_object_read)
356                    .await?,
357            );
358        }
359
360        Ok(responses)
361    }

372    async fn get_checkpoints(
373        &self,
374        cursor: Option<BigInt<u64>>,
375        limit: Option<usize>,
376        descending_order: bool,
377    ) -> RpcResult<CheckpointPage> {
378        let cursor = cursor.map(BigInt::into_inner);
379        let limit = iota_json_rpc_api::validate_limit(
380            limit,
381            iota_json_rpc_api::QUERY_MAX_RESULT_LIMIT_CHECKPOINTS,
382        )
383        .map_err(IotaRpcInputError::from)?;
384        let limit = NonZeroUsize::new(limit).expect("limit should be validated as positive");
385
386        let mut checkpoints = self
387            .inner
388            .get_checkpoints_with_fallback(cursor, limit.saturating_add(1), descending_order)
389            .await?;
390
391        let has_next_page = checkpoints.len() > limit.get();
392        checkpoints.truncate(limit.get());
393
394        let next_cursor = checkpoints.last().map(|d| d.sequence_number.into());
395
396        Ok(CheckpointPage {
397            data: checkpoints,
398            next_cursor,
399            has_next_page,
400        })
401    }

410    async fn get_protocol_config(
411        &self,
412        version: Option<BigInt<u64>>,
413    ) -> RpcResult<ProtocolConfigResponse> {
414        let chain = self.get_chain_identifier().await?.chain();
415        let version = if let Some(version) = version {
416            (*version).into()
417        } else {
418            let latest_epoch = self
419                .inner
420                .spawn_blocking(|this| this.get_latest_epoch_info_from_db())
421                .await?;
422            (latest_epoch.protocol_version as u64).into()
423        };
424
425        ProtocolConfig::get_for_version_if_supported(version, chain)
426            .ok_or(IotaRpcInputError::ProtocolVersionUnsupported(
427                ProtocolVersion::MIN.as_u64(),
428                ProtocolVersion::MAX.as_u64(),
429            ))
430            .map_err(Into::into)
431            .map(ProtocolConfigResponse::from)
432    }
