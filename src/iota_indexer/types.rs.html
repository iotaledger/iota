59impl IndexedCheckpoint {
60    pub fn from_iota_checkpoint(
61        checkpoint: &iota_types::messages_checkpoint::CertifiedCheckpointSummary,
62        contents: &CheckpointContents,
63        successful_tx_num: usize,
64    ) -> Self {
65        let total_gas_cost = checkpoint.epoch_rolling_gas_cost_summary.computation_cost as i64
66            + checkpoint.epoch_rolling_gas_cost_summary.storage_cost as i64
67            - checkpoint.epoch_rolling_gas_cost_summary.storage_rebate as i64;
68        let tx_digests = contents.iter().map(|t| t.transaction).collect::<Vec<_>>();
69        let max_tx_sequence_number = checkpoint.network_total_transactions - 1;
70        // NOTE: + 1u64 first to avoid subtraction with overflow
71        let min_tx_sequence_number = max_tx_sequence_number + 1u64 - tx_digests.len() as u64;
72        let auth_sig = &checkpoint.auth_sig().signature;
73        Self {
74            sequence_number: checkpoint.sequence_number,
75            checkpoint_digest: *checkpoint.digest(),
76            epoch: checkpoint.epoch,
77            tx_digests,
78            previous_checkpoint_digest: checkpoint.previous_digest,
79            end_of_epoch_data: checkpoint.end_of_epoch_data.clone(),
80            end_of_epoch: checkpoint.end_of_epoch_data.clone().is_some(),
81            total_gas_cost,
82            computation_cost: checkpoint.epoch_rolling_gas_cost_summary.computation_cost,
83            computation_cost_burned: checkpoint
84                .epoch_rolling_gas_cost_summary
85                .computation_cost_burned,
86            storage_cost: checkpoint.epoch_rolling_gas_cost_summary.storage_cost,
87            storage_rebate: checkpoint.epoch_rolling_gas_cost_summary.storage_rebate,
88            non_refundable_storage_fee: checkpoint
89                .epoch_rolling_gas_cost_summary
90                .non_refundable_storage_fee,
91            successful_tx_num,
92            network_total_transactions: checkpoint.network_total_transactions,
93            timestamp_ms: checkpoint.timestamp_ms,
94            validator_signature: auth_sig.clone(),
95            checkpoint_commitments: checkpoint.checkpoint_commitments.clone(),
96            content_digest: checkpoint.contents_digest,
97            version_specific_data: checkpoint.version_specific_data.clone(),
98            min_tx_sequence_number,
99            max_tx_sequence_number,
100        }
101    }
102}

271pub enum ObjectStatus {
272    Active = 0,
273    WrappedOrDeleted = 1,
274    /// Only `objects_backward_history` stores this status, to mark that the
275    /// object did not yet exist at the recorded version. Encountering it when
276    /// reading any other table is data corruption.
277    NotYetCreated = 2,
278}
279
280impl TryFrom<i16> for ObjectStatus {
281    type Error = IndexerError;
282
283    fn try_from(value: i16) -> Result<Self, Self::Error> {
284        Ok(match value {
285            0 => ObjectStatus::Active,
286            1 => ObjectStatus::WrappedOrDeleted,
287            2 => ObjectStatus::NotYetCreated,
288            value => {
289                return Err(IndexerError::PersistentStorageDataCorruption(format!(
290                    "{value} as ObjectStatus"
291                )));
292            }
293        })
294    }
295}
296
297impl TryFrom<i16> for OwnerType {
298    type Error = IndexerError;
299
300    fn try_from(value: i16) -> Result<Self, Self::Error> {
301        Ok(match value {
302            0 => OwnerType::Immutable,
303            1 => OwnerType::Address,
304            2 => OwnerType::Object,
305            3 => OwnerType::Shared,
306            value => {
307                return Err(IndexerError::PersistentStorageDataCorruption(format!(
308                    "{value} as OwnerType"
309                )));
310            }
311        })
312    }
313}

315// Returns owner_type, owner_address
316pub fn owner_to_owner_info(owner: &Owner) -> (OwnerType, Option<Address>) {
317    match owner {
318        Owner::Address(address) => (OwnerType::Address, Some(*address)),
319        Owner::Object(object_id) => (OwnerType::Object, Some(*object_id.as_address())),
320        Owner::Shared(_) => (OwnerType::Shared, None),
321        Owner::Immutable => (OwnerType::Immutable, None),
322        _ => unimplemented!("a new Owner enum variant was added and needs to be handled"),
323    }

477        let wrapped_or_deleted_objects = repeat_with(ObjectId::random).take(MAX_OBJECTS).collect();

479        TxIndex {
480            tx_sequence_number: rng.gen(),
481            tx_kind,
482            transaction_digest: TransactionDigest::random(),
483            checkpoint_sequence_number: rng.gen(),

839/// Provides conversion methods from gRPC types to iota core types.
840pub(crate) mod grpc_conversion {
841
842    use iota_grpc_types::v1::{
843        command::{CommandOutputs as GrpcCommandOutputs, CommandResults as GrpcCommandResults},
844        object::Objects as GrpcObjects,
845        transaction::{
846            BalanceChanges as GrpcBalanceChanges, ObjectChanges as GrpcObjectChanges,
847            object_change::Kind as GrpcObjectChangeKind,
848        },
849    };
850    use iota_json_rpc_types::{
851        BalanceChange, IotaArgument, IotaExecutionResult, IotaTypeTag, ObjectChange,
852    };
853    use iota_sdk_types::{StructTag, TypeTag, Version};
854    use iota_types::object::Object;
855
856    use crate::{errors::IndexerError, types::IndexerResult};
857
858    /// Converts [`GrpcObjects`] into [`Vec<Object>`]
859    pub(crate) fn objects(objects: &GrpcObjects) -> IndexerResult<Vec<Object>> {
860        objects
861            .objects
862            .iter()
863            .map(|o| -> IndexerResult<_> { Ok(Object::from(o.object()?)) })
864            .collect()
865    }
866
867    /// Converts [`GrpcBalanceChanges`] into [`Vec<BalanceChange>`]
868    pub(crate) fn balance_changes(
869        balance_changes: &GrpcBalanceChanges,
870    ) -> IndexerResult<Vec<BalanceChange>> {
871        balance_changes
872            .balance_changes
873            .iter()
874            .map(|change| -> IndexerResult<_> {
875                Ok(BalanceChange {
876                    owner: change.owner()?,
877                    coin_type: change.coin_type()?,
878                    amount: change.amount_i128()?,
879                })
880            })
881            .collect()
882    }
883
884    fn object_change_version(version: Option<u64>) -> IndexerResult<Version> {
885        version
886            .map(Version::from_u64)
887            .ok_or_else(|| IndexerError::Grpc("object change version should be present".into()))
888    }
889
890    fn object_change_struct_tag(object_type: TypeTag) -> IndexerResult<StructTag> {
891        object_type.into_opt_struct_tag().ok_or_else(|| {
892            IndexerError::Grpc("object change object type should be a struct tag".into())
893        })
894    }
895
896    /// Converts [`GrpcObjectChanges`] into [`Vec<ObjectChange>`]
897    pub(crate) fn object_changes(
898        object_changes: &GrpcObjectChanges,
899    ) -> IndexerResult<Vec<ObjectChange>> {
900        object_changes
901            .object_changes
902            .iter()
903            .map(|change| -> IndexerResult<_> {
904                let kind = change.kind.as_ref().ok_or_else(|| {
905                    IndexerError::Grpc("object change kind should be present".into())
906                })?;
907                Ok(match kind {
908                    GrpcObjectChangeKind::Published(published) => ObjectChange::Published {
909                        package_id: published.package_id()?,
910                        version: object_change_version(published.version)?,
911                        digest: published.digest()?,
912                        modules: published.modules.clone(),
913                    },
914                    GrpcObjectChangeKind::Mutated(mutated) => ObjectChange::Mutated {
915                        sender: mutated.sender()?,
916                        owner: mutated.owner()?,
917                        object_type: object_change_struct_tag(mutated.object_type()?)?,
918                        object_id: mutated.object_id()?,
919                        version: object_change_version(mutated.version)?,
920                        previous_version: object_change_version(mutated.previous_version)?,
921                        digest: mutated.digest()?,
922                    },
923                    GrpcObjectChangeKind::Deleted(deleted) => ObjectChange::Deleted {
924                        sender: deleted.sender()?,
925                        object_type: object_change_struct_tag(deleted.object_type()?)?,
926                        object_id: deleted.object_id()?,
927                        version: object_change_version(deleted.version)?,
928                    },
929                    GrpcObjectChangeKind::Wrapped(wrapped) => ObjectChange::Wrapped {
930                        sender: wrapped.sender()?,
931                        object_type: object_change_struct_tag(wrapped.object_type()?)?,
932                        object_id: wrapped.object_id()?,
933                        version: object_change_version(wrapped.version)?,
934                    },
935                    GrpcObjectChangeKind::Unwrapped(unwrapped) => ObjectChange::Unwrapped {
936                        sender: unwrapped.sender()?,
937                        owner: unwrapped.owner()?,
938                        object_type: object_change_struct_tag(unwrapped.object_type()?)?,
939                        object_id: unwrapped.object_id()?,
940                        version: object_change_version(unwrapped.version)?,
941                        digest: unwrapped.digest()?,
942                    },
943                    GrpcObjectChangeKind::Created(created) => ObjectChange::Created {
944                        sender: created.sender()?,
945                        owner: created.owner()?,
946                        object_type: object_change_struct_tag(created.object_type()?)?,
947                        object_id: created.object_id()?,
948                        version: object_change_version(created.version)?,
949                        digest: created.digest()?,
950                    },
951                    kind => {
952                        return Err(IndexerError::Grpc(format!(
953                            "unknown object change kind: {kind:?}"
954                        )));
955                    }
956                })
957            })
958            .collect()
959    }
960
961    fn convert_command_outputs_into_mutated_by_ref(
962        command_outputs: GrpcCommandOutputs,
963    ) -> IndexerResult<Vec<(IotaArgument, Vec<u8>, IotaTypeTag)>> {
964        command_outputs
965            .outputs
966            .into_iter()
967            .map(|command_output| -> IndexerResult<_> {
968                Ok((
969                    IotaArgument::from(command_output.argument()?),
970                    command_output.output_bcs()?.to_vec(),
971                    command_output.type_tag()?.into(),
972                ))
973            })
974            .collect()
975    }
976
977    fn convert_command_outputs_into_return_values(
978        command_outputs: GrpcCommandOutputs,
979    ) -> IndexerResult<Vec<(Vec<u8>, IotaTypeTag)>> {
980        command_outputs
981            .outputs
982            .into_iter()
983            .map(|command_output| -> IndexerResult<_> {
984                Ok((
985                    command_output.output_bcs()?.to_vec(),
986                    command_output.type_tag()?.into(),
987                ))
988            })
989            .collect()
990    }
991
992    /// Converts [`GrpcCommandResults`] into [`IotaExecutionResult`]
993    pub(crate) fn command_results(
994        command_results: GrpcCommandResults,
995    ) -> IndexerResult<Vec<IotaExecutionResult>> {
996        command_results
997            .results
998            .into_iter()
999            .map(|command_result| -> IndexerResult<_> {
1000                let mutable_reference_outputs = command_result
1001                    .mutated_by_ref()
1002                    .map_err(Into::into)
1003                    .and_then(|c| convert_command_outputs_into_mutated_by_ref(c.clone()))?;
1004                let return_values = command_result
1005                    .return_values()
1006                    .map_err(Into::into)
1007                    .and_then(|c| convert_command_outputs_into_return_values(c.clone()))?;
1008
1009                Ok(IotaExecutionResult {
1010                    mutable_reference_outputs,
1011                    return_values,
1012                })
1013            })
1014            .collect()
1015    }
