15/// Represents a row in the `watermarks` table.
16#[derive(Queryable, Insertable, Default, QueryableByName, Clone)]
17#[diesel(table_name = watermarks, primary_key(entity))]
18pub struct StoredWatermark {
19    /// The table governed by this watermark, i.e `epochs`, `checkpoints`,
20    /// `transactions`.
21    pub entity: String,
22    /// Current epoch for this entity's data. Committer updates this field.
23    /// Pruner uses this to determine if pruning is necessary based on the
24    /// retention policy.
25    pub current_epoch: i64,
26    /// Maximum committed checkpoint for this entity's data. Committer
27    /// updates this field. All data of this entity in the checkpoint must
28    /// be persisted before advancing this watermark. The committer refers
29    /// to this on disaster recovery to resume writing.
30    pub max_committed_cp: i64,
31    /// Maximum committed transaction sequence number for this entity's data.
32    /// Committer updates this field. This is inclusive.
33    pub max_committed_tx: i64,
34    /// Minimum available epoch for this entity's data. Pruner updates this
35    /// field when the epoch range exceeds the retention policy.
36    pub min_available_epoch: i64,
37    /// Timestamp in milliseconds of the last update to the min_available_*
38    /// columns. The pruner uses this to determine whether to prune or wait
39    /// long enough that all in-flight reads complete or timeout before it
40    /// acts on an updated watermark.
41    pub min_bounds_updated_at_timestamp_ms: i64,
42    /// Lowest key (epoch, checkpoint, or tx) that has not been pruned yet.
43    /// The pruner uses this to track its progress.
44    pub lowest_unpruned_key: i64,
45    /// Minimum available transaction sequence number for this entity's data.
46    /// Pruner updates this field based on min_available_epoch.
47    pub min_available_tx: i64,
48    /// Minimum available checkpoint sequence number for this entity's data.
49    /// Pruner updates this field based on min_available_epoch.
50    pub min_available_cp: i64,
51}
52
53impl StoredWatermark {
54    pub fn from_upper_bound_update(entity: &str, watermark: CommitterWatermark) -> Self {
55        StoredWatermark {
56            entity: entity.to_string(),
57            current_epoch: watermark.current_epoch as i64,
58            max_committed_cp: watermark.max_committed_cp as i64,
59            max_committed_tx: watermark.max_committed_tx as i64,
60            ..StoredWatermark::default()
61        }
62    }
63
64    pub fn from_lower_bound_update(
65        entity: &str,
66        min_available_epoch: u64,
67        min_available_cp: u64,
68        min_available_tx: u64,
69    ) -> Self {
70        StoredWatermark {
71            entity: entity.to_string(),
72            min_available_epoch: min_available_epoch as i64,
73            min_available_cp: min_available_cp as i64,
74            min_available_tx: min_available_tx as i64,
75            ..StoredWatermark::default()
76        }
77    }
78
79    pub fn entity(&self) -> Option<PrunableTable> {
80        PrunableTable::from_str(&self.entity).ok()
81    }
82
83    /// Determine whether to set a new minimum available epoch based on the
84    /// retention policy.
85    pub fn new_min_available_epoch(&self, retention: u64) -> Option<u64> {
86        if self.min_available_epoch as u64 + retention <= self.current_epoch as u64 {
87            Some((self.current_epoch as u64).saturating_sub(retention - 1))
88        } else {
89            None
90        }
91    }
92}
