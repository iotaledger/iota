25#[serde_as]
26#[derive(Debug, Serialize, Deserialize, Clone)]
27#[serde(rename_all = "camelCase")]
28pub struct StoredSystemStateV1 {
29    /// The current epoch ID, starting from 0.
30    #[serde_as(as = "Readable<BigInt<u64>, _>")]
31    pub epoch: u64,
32    /// The current protocol version, starting from 1.
33    #[serde_as(as = "Readable<BigInt<u64>, _>")]
34    pub protocol_version: u64,
35    /// The current version of the system state data structure type.
36    #[serde_as(as = "Readable<BigInt<u64>, _>")]
37    pub system_state_version: u64,
38    /// The current IOTA supply.
39    #[serde_as(as = "Readable<BigInt<u64>, _>")]
40    pub iota_total_supply: u64,
41    /// The `TreasuryCap<IOTA>` object ID.
42    pub iota_treasury_cap_id: ObjectId,
43    /// The storage rebates of all the objects on-chain stored in the storage
44    /// fund.
45    #[serde_as(as = "Readable<BigInt<u64>, _>")]
46    pub storage_fund_total_object_storage_rebates: u64,
47    /// The non-refundable portion of the storage fund coming from
48    /// non-refundable storage rebates and any leftover
49    /// staking rewards.
50    #[serde_as(as = "Readable<BigInt<u64>, _>")]
51    pub storage_fund_non_refundable_balance: u64,
52    /// The reference gas price for the current epoch.
53    #[serde_as(as = "Readable<BigInt<u64>, _>")]
54    pub reference_gas_price: u64,
55    /// Whether the system is running in a downgraded safe mode due to a
56    /// non-recoverable bug. This is set whenever we failed to execute
57    /// advance_epoch, and ended up executing advance_epoch_safe_mode.
58    /// It can be reset once we are able to successfully execute advance_epoch.
59    pub safe_mode: bool,
60    /// Amount of storage charges accumulated (and not yet distributed) during
61    /// safe mode.
62    #[serde_as(as = "Readable<BigInt<u64>, _>")]
63    pub safe_mode_storage_charges: u64,
64    /// Amount of computation rewards accumulated (and not yet distributed)
65    /// during safe mode.
66    #[serde_as(as = "Readable<BigInt<u64>, _>")]
67    pub safe_mode_computation_rewards: u64,
68    /// Amount of storage rebates accumulated (and not yet burned) during safe
69    /// mode.
70    #[serde_as(as = "Readable<BigInt<u64>, _>")]
71    pub safe_mode_storage_rebates: u64,
72    /// Amount of non-refundable storage fee accumulated during safe mode.
73    #[serde_as(as = "Readable<BigInt<u64>, _>")]
74    pub safe_mode_non_refundable_storage_fee: u64,
75    /// Unix timestamp of the current epoch start
76    #[serde_as(as = "Readable<BigInt<u64>, _>")]
77    pub epoch_start_timestamp_ms: u64,
78
79    // System parameters
80    /// The duration of an epoch, in milliseconds.
81    #[serde_as(as = "Readable<BigInt<u64>, _>")]
82    pub epoch_duration_ms: u64,
83
84    /// Minimum number of active validators at any moment.
85    /// We do not allow the number of validators in any epoch to go under this.
86    #[serde_as(as = "Readable<BigInt<u64>, _>")]
87    pub min_validator_count: u64,
88
89    /// Maximum number of active validators at any moment.
90    /// We do not allow the number of validators in any epoch to go above this.
91    #[serde_as(as = "Readable<BigInt<u64>, _>")]
92    pub max_validator_count: u64,
93
94    /// Lower-bound on the amount of stake required to become a validator.
95    #[serde_as(as = "Readable<BigInt<u64>, _>")]
96    pub min_validator_joining_stake: u64,
97
98    /// Validators with stake amount below `validator_low_stake_threshold` are
99    /// considered to have low stake and will be escorted out of the
100    /// validator set after being below this threshold for more than
101    /// `validator_low_stake_grace_period` number of epochs.
102    #[serde_as(as = "Readable<BigInt<u64>, _>")]
103    pub validator_low_stake_threshold: u64,
104
105    /// Validators with stake below `validator_very_low_stake_threshold` will be
106    /// removed immediately at epoch change, no grace period.
107    #[serde_as(as = "Readable<BigInt<u64>, _>")]
108    pub validator_very_low_stake_threshold: u64,
109
110    /// A validator can have stake below `validator_low_stake_threshold`
111    /// for this many epochs before being kicked out.
112    #[serde_as(as = "Readable<BigInt<u64>, _>")]
113    pub validator_low_stake_grace_period: u64,
114
115    // Validator set
116    /// Total amount of stake from all active validators at the beginning of the
117    /// epoch.
118    #[serde_as(as = "Readable<BigInt<u64>, _>")]
119    pub total_stake: u64,
120    /// The list of active validators in the current epoch.
121    pub active_validators: Vec<StoredValidator>,
122    /// ID of the object that contains the list of new validators that will join
123    /// at the end of the epoch.
124    pub pending_active_validators_id: ObjectId,
125    /// Number of new validators that will join at the end of the epoch.
126    #[serde_as(as = "Readable<BigInt<u64>, _>")]
127    pub pending_active_validators_size: u64,
128    /// Removal requests from the validators. Each element is an index
129    /// pointing to `active_validators`.
130    #[serde_as(as = "Vec<Readable<BigInt<u64>, _>>")]
131    pub pending_removals: Vec<u64>,
132    /// ID of the object that maps from staking pool's ID to the iota address of
133    /// a validator.
134    pub staking_pool_mappings_id: ObjectId,
135    /// Number of staking pool mappings.
136    #[serde_as(as = "Readable<BigInt<u64>, _>")]
137    pub staking_pool_mappings_size: u64,
138    /// ID of the object that maps from a staking pool ID to the inactive
139    /// validator that has that pool as its staking pool.
140    pub inactive_pools_id: ObjectId,
141    /// Number of inactive staking pools.
142    #[serde_as(as = "Readable<BigInt<u64>, _>")]
143    pub inactive_pools_size: u64,
144    /// ID of the object that stores preactive validators, mapping their
145    /// addresses to their `Validator` structs.
146    pub validator_candidates_id: ObjectId,
147    /// Number of preactive validators.
148    #[serde_as(as = "Readable<BigInt<u64>, _>")]
149    pub validator_candidates_size: u64,
150    /// Map storing the number of epochs for which each validator has been below
151    /// the low stake threshold.
152    #[serde_as(as = "Vec<(_, Readable<BigInt<u64>, _>)>")]
153    pub at_risk_validators: Vec<(Address, u64)>,
154    /// A map storing the records of validator reporting each other.
155    pub validator_report_records: Vec<(Address, Vec<Address>)>,
156}

315    pub image_url: String,
316    pub project_url: String,
317    pub net_address: String,
318    pub p2p_address: String,
319    pub primary_address: String,
320    #[serde_as(as = "Option<Base64>")]
321    pub next_epoch_authority_pubkey_bytes: Option<Vec<u8>>,
322    #[serde_as(as = "Option<Base64>")]
323    pub next_epoch_proof_of_possession: Option<Vec<u8>>,
324    #[serde_as(as = "Option<Base64>")]

335    #[serde_as(as = "Readable<BigInt<u64>, _>")]

356    /// The total number of IOTA tokens in this pool.
357    #[serde_as(as = "Readable<BigInt<u64>, _>")]
358    pub staking_pool_iota_balance: u64,
359    /// The epoch stake rewards will be added here at the end of each epoch.

365    /// Pending stake amount for this epoch.
366    #[serde_as(as = "Readable<BigInt<u64>, _>")]

376    /// ID of the exchange rate table object.

378    /// Number of exchange rates in the table.
379    #[serde_as(as = "Readable<BigInt<u64>, _>")]
380    pub exchange_rates_size: u64,
381}
