44/// Using transaction messages we update the state of shard reconstructor
45/// Two types of messages are supported: full transaction and shard
46#[derive(Clone, Debug)]
47pub(crate) enum TransactionMessage {
48    FullTransaction(TransactionRef),
49    Shard(ShardMessage),
50}
51
52/// Shard message contains shard with index and the reference to the
53/// transactions the shard was erasure-coded from, plus an optional block digest
54/// (present for V1 shards, absent for V2 shards that use TransactionRef).
55#[derive(Clone, Debug)]
56pub(crate) struct ShardMessage {
57    transaction_ref: TransactionRef,
58    block_digest: Option<BlockHeaderDigest>,
59    shard: Shard,
60    shard_index: usize,
61}
62
63impl TransactionMessage {
64    pub fn transaction_ref(&self) -> TransactionRef {
65        match self {
66            TransactionMessage::FullTransaction(tx_ref) => *tx_ref,
67            TransactionMessage::Shard(msg) => msg.transaction_ref,
68        }
69    }
70
71    /// Create transaction messages (full, shards) for a given block
72    /// bundle
73    pub fn create_transaction_messages(
74        block: &VerifiedBlock,
75        shards: &[ShardWithProof],
76        shard_index: usize,
77    ) -> Vec<TransactionMessage> {
78        let full = TransactionMessage::FullTransaction(block.transaction_ref());
79
80        let shard_msgs = shards.iter().map(|swp| {
81            TransactionMessage::Shard(ShardMessage {
82                transaction_ref: TransactionRef {
83                    round: swp.round(),
84                    author: swp.author(),
85                    transactions_commitment: swp.transaction_commitment(),
86                },
87                block_digest: swp.block_digest(),
88                shard: swp.shard().clone(),
89                shard_index,
90            })
91        });
92
93        std::iter::once(full).chain(shard_msgs).collect()
94    }
95}

97/// A basic structure that represents the collection of shards for a given
98/// transaction reference. We track the number of shards and the shards
99/// themselves.
100#[derive(Clone)]
101pub struct ShardAccumulator {
102    /// Reference to the transactions these shards were erasure-coded from
103    transaction_ref: TransactionRef,
104    /// Block digest of the source block (present for V1, absent for V2)
105    block_digest: Option<BlockHeaderDigest>,
106    /// Collected shards, one slot per authority, indexed by the authority index
107    /// of the peer that relayed the shard.
108    collected_shards: Vec<Option<Shard>>,
109    /// Number of collected data shards
110    number_shards: usize,
111}
112
113impl ShardAccumulator {
114    /// Create a new accumulator initialized with the first shard
115    fn new_with_shard(msg: ShardMessage, total_length: usize) -> Self {
116        let ShardMessage {
117            transaction_ref,
118            block_digest,
119            shard,
120            shard_index,
121        } = msg;
122        let mut collected_shards = vec![None; total_length];
123        collected_shards[shard_index] = Some(shard);
124        Self {
125            transaction_ref,
126            block_digest,
127            collected_shards,
128            number_shards: 1,
129        }
130    }
131
132    /// Update the accumulator with a new shard
133    fn update_with_shard(&mut self, msg: ShardMessage) {
134        let ShardMessage {
135            shard, shard_index, ..
136        } = msg;
137        if self.collected_shards[shard_index].is_none() {
138            self.collected_shards[shard_index] = Some(shard);
139            self.number_shards += 1;
140        }
141    }
142
143    /// Ready once enough shards for decoding are collected and the relayers'
144    /// combined stake reaches the validity threshold (f+1), which guarantees
145    /// an honest relayer and thus a genuinely authored commitment.
146    fn is_ready_to_reconstruct(&self, info_length: usize, committee: &Committee) -> bool {
147        if self.number_shards < info_length {
148            return false;
149        }
150        let relayer_stake: Stake = self
151            .collected_shard_indices()
152            .filter_map(|i| committee.to_authority_index(i))
153            .map(|i| committee.stake(i))
154            .sum();
155        committee.reached_validity(relayer_stake)
156    }
157
158    /// Indices of the shards collected so far. A shard at index `i` is
159    /// authenticated as authority `i`'s (its Merkle proof is verified at
160    /// position `i`, which is the relaying peer's index), so a collected
161    /// shard identifies the peer that relayed it.
162    fn collected_shard_indices(&self) -> impl Iterator<Item = usize> + '_ {
163        self.collected_shards
164            .iter()
165            .enumerate()
166            .filter_map(|(i, shard)| shard.as_ref().map(|_| i))
167    }
168
169    /// Decodes the transaction data from the collected shards and verifies
170    /// the reconstructed bytes against the transactions commitment in the
171    /// ref. Consumes the accumulator, so the collected shards move into the
172    /// decoder rather than being copied.
173    fn decode_and_verify_commitment(
174        self,
175        codec: &mut Codec,
176    ) -> ConsensusResult<CommitmentVerifiedTransactions> {
177        let Self {
178            transaction_ref,
179            block_digest,
180            collected_shards,
181            ..
182        } = self;
183
184        let transactions = codec.decoder.decode_shards(
185            codec.info_length,
186            codec.parity_length,
187            collected_shards,
188        )?;
189
190        let serialized =
191            Transaction::serialize(&transactions).expect("We should expect serialization to work");
192
193        // Verify the commitment
194        let computed_commitment = TransactionsCommitment::compute_transactions_commitment(
195            &serialized,
196            &codec.context.clone(),
197            &mut codec.encoder,
198        )?;
199        if computed_commitment != transaction_ref.transactions_commitment {
200            return Err(ConsensusError::TransactionCommitmentMismatch { transaction_ref });
201        }
202
203        Ok(CommitmentVerifiedTransactions::new(
204            transactions,
205            transaction_ref,
206            block_digest,
207            serialized,
208        ))
209    }
210
211    /// Whether a shard is already collected at the given index.
212    fn contains_shard_at_index(&self, shard_index: usize) -> bool {
213        self.collected_shards[shard_index].is_some()
214    }
215
216    /// Remove the shard collected at the given index, if any.
217    fn remove_shard_at_index(&mut self, shard_index: usize) {
218        if self.collected_shards[shard_index].take().is_some() {
219            self.number_shards -= 1;
220        }
221    }
222}

224/// Attributes a reconstructed payload that failed the transaction validity
225/// check.
226///
227/// A peer must hold the full payload to erasure-code its shard, so every peer
228/// that contributed a shard could have checked the transactions' validity and
229/// is charged an unprovable fault for relaying invalid bytes. The commitment
230/// here is peer-supplied, so without a verified author-signed header for this
231/// ref a coalition of peers could reconstruct invalid transactions under a
232/// fabricated commitment to frame the author; charge the author a provable
233/// fault only when such a header exists, as he then committed to the invalid
234/// transactions.
235///
236/// Attribution is one-shot: a header arriving only after this failure does not
237/// retroactively charge the author (the failed ref is marked processed and not
238/// revisited until garbage collection). Such an author is instead charged on
239/// the direct primary-block route, where the full payload is verified against
240/// the author.
241fn record_reconstruction_validity_failure(
242    dag_state: &RwLock<DagState>,
243    misbehavior_store: &MisbehaviorStore,
244    tx_ref: TransactionRef,
245    relayers: Vec<AuthorityIndex>,
246    err: &ConsensusError,
247) {
248    let author = tx_ref.author;
249    let authored = dag_state
250        .read()
251        .contains_verified_block_headers_for_transaction_refs(&[tx_ref])[0];
252    misbehavior_store.record_faulty_transactions(author, authored, relayers);
253    error!(
254        "Reconstructed transactions for {:?} failed the validity check: {:?}",
255        tx_ref, err
256    );
257}
258
259/// Charges the peers that relayed shards for a reconstructed payload whose
260/// commitment doesn't match the one its ref commits to. The mismatch comes from
261/// the peer-supplied shards, not the author, so no author fault is recorded
262/// even when a verified header for the ref exists.
263fn record_reconstruction_commitment_mismatch(
264    misbehavior_store: &MisbehaviorStore,
265    tx_ref: TransactionRef,
266    relayers: Vec<AuthorityIndex>,
267) {
268    misbehavior_store.record_faulty_transactions(tx_ref.author, false, relayers);
269}

292/// By keeping this handle, we continue running ShardCollector, responsible for
293/// shard collection, and given number of shard reconstructor workers.
294/// One field, transaction_message_sender, can be cloned to send transaction
295/// messages to the internal ShardReconstructor
296pub struct ShardReconstructorHandle {
297    transaction_message_sender: Sender<Vec<TransactionMessage>>,
298    join_handle: Mutex<Option<JoinHandle<()>>>,
299}
300
301impl ShardReconstructorHandle {
302    /// Access the transaction sender
303    pub fn transaction_message_sender(&self) -> Sender<Vec<TransactionMessage>> {
304        self.transaction_message_sender.clone()
305    }
306
307    /// Gracefully stop the shard reconstructor.
308    pub async fn stop(&self) -> Result<(), JoinError> {
309        let mut guard = self.join_handle.lock().await;
310
311        if let Some(handle) = guard.take() {
312            handle.abort();
313            match handle.await {
314                Ok(_) => Ok(()),
315                Err(e) if e.is_cancelled() => Ok(()), // expected cancellation
316                Err(e) => Err(e),                     // propagate panic or other errors
317            }
318        } else {
319            Ok(()) // already stopped
320        }
321    }
322}

346/// Result of a reconstruction job: the verified transactions on success, or
347/// the failed job's transaction reference so its queue entry can be dropped.
348type ReconstructionResult = Result<CommitmentVerifiedTransactions, TransactionRef>;

350/// The main structure responsible for collecting shards and reconstructing
351/// transaction data once enough shards are collected. Keeps track of already
352/// locally available transaction data. The transaction is reconstructed only
353/// when it is still not locally available and enough shards are reconstructed.
354/// The structure periodically sends data to the core. In addition, eviction
355/// mechanism is implemented by relying on the transaction GC round.
356pub struct ShardReconstructor<C: CoreThreadDispatcher> {
357    /// Shards below this round will not be collected
358    transaction_gc_round: Round,
359    /// Minimum number of shards the decoder needs to reconstruct the data
360    info_length: usize,
361    /// The total number of shards
362    total_length: usize,
363    context: Arc<Context>,
364    /// Already processed transaction either by authority service or by shard
365    /// reconstructor
366    processed_transactions: BTreeSet<TransactionRef>,
367    /// A cache of reconstructed transactions that will be periodically sent in
368    /// the core
369    reconstructed_transactions: BTreeMap<TransactionRef, CommitmentVerifiedTransactions>,
370    /// A map of all shard accumulators. Periodically evicted. Keyed by
371    /// TransactionRef which uniquely identifies transactions via
372    /// transactions_commitment
373    shard_accumulators: BTreeMap<TransactionRef, ShardAccumulator>,
374    /// Use only read access to the dag state to read the transaction GC round
375    /// and check whether the respected headers are available
376    dag_state: Arc<RwLock<DagState>>,
377    /// The receiver for transaction message sent from the authority service
378    transaction_message_receiver: Receiver<Vec<TransactionMessage>>,
379    /// After full reconstruction and verification, send data to the core
380    core_dispatcher: Arc<C>,
381    /// Applies the same transaction limit and batch verification checks to
382    /// reconstructed payloads as the direct block-bundle route
383    block_verifier: Arc<dyn BlockVerifier>,
384    /// Charges faults for payloads that fail verification: the author, when a
385    /// verified header proves the payload is theirs, and the peers that relayed
386    /// the shards.
387    misbehavior_store: Arc<MisbehaviorStore>,
388    /// Queue is used to not reconstruct the same data twice
389    reconstruction_queue: BTreeSet<TransactionRef>,
390    /// Once enough shards are collected, they are sent to reconstructor workers
391    ready_to_reconstruct_sender: Sender<ShardAccumulator>,
392    /// Channel to receive accumulated shard for reconstruction by workers
393    ready_to_reconstruct_receiver: Arc<Mutex<Receiver<ShardAccumulator>>>,
394    /// Reconstruction workers report each job's result through this channel
395    reconstruction_result_sender: Sender<ReconstructionResult>,
396    /// Job results are received by this channel
397    reconstruction_result_receiver: Receiver<ReconstructionResult>,
398    /// For each authority, the accumulators currently retaining a shard it
399    /// relayed. Enforces the per-authority shard budget: at the budget, the
400    /// authority's oldest retained shard is evicted to admit a new one.
401    retained_shards_by_authority: Vec<BTreeSet<TransactionRef>>,
402    /// Slots whose genuine payload is already known, from a successful decode
403    /// or a directly received full payload. Their accumulators are purged and
404    /// further shards for them are dropped. Periodically evicted by round.
405    resolved_slots: BTreeSet<(Round, AuthorityIndex)>,
406    /// Maximum number of shards from one relaying authority retained across
407    /// all accumulators.
408    shard_budget_per_authority: usize,
409}

411impl<C: CoreThreadDispatcher> ShardReconstructor<C> {
412    /// Create a new ShardReconstructor and its associated Sender
413    pub fn new(
414        context: Arc<Context>,
415        dag_state: Arc<RwLock<DagState>>,
416        core_dispatcher: Arc<C>,
417        block_verifier: Arc<dyn BlockVerifier>,
418    ) -> (Self, Sender<Vec<TransactionMessage>>) {
419        let info_length = context.committee.info_length();
420        let total_length = context.committee.size();
421        let shard_budget_per_authority = context.parameters.shard_budget_per_authority as usize;
422
423        let (transaction_message_sender, transaction_message_receiver) = mpsc::channel(1000);
424        let (ready_sender, ready_receiver) = mpsc::channel(1000);
425        let (result_sender, result_receiver) = mpsc::channel(1000);
426
427        let misbehavior_store = dag_state.read().misbehavior_store().clone();
428        let reconstructor = Self {
429            info_length,
430            total_length,
431            context,
432            core_dispatcher,
433            dag_state,
434            block_verifier,
435            misbehavior_store,
436            transaction_gc_round: GENESIS_ROUND,
437            reconstruction_queue: BTreeSet::new(),
438            ready_to_reconstruct_sender: ready_sender,
439            ready_to_reconstruct_receiver: Arc::new(Mutex::new(ready_receiver)),
440            reconstruction_result_sender: result_sender,
441            reconstruction_result_receiver: result_receiver,
442            processed_transactions: BTreeSet::new(),
443            reconstructed_transactions: BTreeMap::new(),
444            shard_accumulators: BTreeMap::new(),
445            transaction_message_receiver,
446            retained_shards_by_authority: vec![BTreeSet::new(); total_length],
447            resolved_slots: BTreeSet::new(),
448            shard_budget_per_authority,
449        };
450
451        (reconstructor, transaction_message_sender)
452    }

454    pub fn start_reconstruction_workers(&self) {
455        for _ in 0..NUMBER_OF_RECONSTRUCTION_WORKERS {
456            let mut codec = Codec::new(&self.context);
457            let ready_rx = Arc::clone(&self.ready_to_reconstruct_receiver);
458            let result_tx = self.reconstruction_result_sender.clone();
459            let context = self.context.clone();
460            let dag_state = self.dag_state.clone();
461            let block_verifier = self.block_verifier.clone();
462            let misbehavior_store = self.misbehavior_store.clone();
463            tokio::spawn(async move {
464                let metrics = &context.metrics;
465                // Receive a job from the ready to reconstruct channel until it closes.
466                while let Some(shard_accumulator) = {
467                    let mut rx = ready_rx.lock().await;
468                    rx.recv().await
469                } {
470                    metrics.node_metrics.reconstruction_jobs_started.inc();
471                    // Read what the failure paths attribute with before decoding
472                    // consumes the accumulator.
473                    let tx_ref = shard_accumulator.transaction_ref;
474                    let relayers: Vec<_> = shard_accumulator
475                        .collected_shard_indices()
476                        .filter_map(|i| context.committee.to_authority_index(i))
477                        .collect();
478                    let result = match shard_accumulator.decode_and_verify_commitment(&mut codec) {
479                        // Validity-threshold relayer stake guarantees an honest
480                        // relayer and thus a genuine commitment, so a decode
481                        // failure indicates a codec bug or Byzantine stake
482                        // beyond the fault model.
483                        Err(err) => {
484                            error!("Failed to reconstruct transactions for {tx_ref:?}: {err:?}");
485                            // A commitment mismatch means the reconstructed bytes
486                            // aren't the ones the author committed to; the shards,
487                            // and thus the mismatch, come from peers, so charge
488                            // only the peers that relayed shards, never the author.
489                            if matches!(err, ConsensusError::TransactionCommitmentMismatch { .. }) {
490                                record_reconstruction_commitment_mismatch(
491                                    &misbehavior_store,
492                                    tx_ref,
493                                    relayers,
494                                );
495                            }
496                            Err(tx_ref)
497                        }
498                        Ok(verified_transactions) => match block_verifier
499                            .verify_transactions_validity(&verified_transactions)
500                        {
501                            Ok(()) => {
502                                debug!("Successfully reconstructed transactions for {tx_ref:?}");
503                                Ok(verified_transactions)
504                            }
505                            Err(err) => {
506                                record_reconstruction_validity_failure(
507                                    &dag_state,
508                                    &misbehavior_store,
509                                    tx_ref,
510                                    relayers,
511                                    &err,
512                                );
513                                Err(tx_ref)
514                            }
515                        },
516                    };
517                    if let Err(err) = result_tx.send(result).await {
518                        warn!("Failed to send the result to shard accumulator {err}");
519                    }
520                    metrics.node_metrics.reconstruction_jobs_finished.inc();
521                }
522                debug!("Ready to reconstruct channel closed, workers exiting");
523            });
524        }
525    }

527    /// Run the main loop, consuming TransactionMessages from the channel
528    async fn run(&mut self) {
529        self.start_reconstruction_workers();
530
531        let send_to_core_timeout =
532            sleep_until(Instant::now() + SEND_TO_CORE_RECONSTRUCTED_TXS_TIMEOUT);
533        tokio::pin!(send_to_core_timeout);
534
535        let eviction_timeout = sleep_until(Instant::now() + EVICTION_TIMEOUT);
536        tokio::pin!(eviction_timeout);
537
538        loop {
539            tokio::select! {
540                    // Receive new shard/header/full-transaction
541                    transaction_msgs = self.transaction_message_receiver.recv() => {
542                        match transaction_msgs {
543                            Some(msgs) => {
544                                for msg in msgs {
545                                    // Handle the message and update internal state
546                                    if let Err(e) = self.handle_transaction_message(msg.clone()).await {
547                                        warn!("Error when handling transaction message{:?}: {:?}", msg, e);
548                                    }
549                                }
550                            }
551                            None => {
552                                debug!("Transaction channel is closed, shutting down");
553                                break;
554                            }
555                        }
556                    }
557                    // A reconstruction job finished in one of the reconstruction workers
558                    Some(result) = self.reconstruction_result_receiver.recv() => {
559                        // Success and failure are both final for the ref: the shards
560                        // proved membership against the commitment inside the ref, so
561                        // a failed job would fail identically on retry.
562                        let tx_ref = match &result {
563                            Ok(verified_transactions) => verified_transactions.transaction_ref(),
564                            Err(tx_ref) => *tx_ref,
565                        };
566                        self.processed_transactions.insert(tx_ref);
567                        self.reconstruction_queue.remove(&tx_ref);
568                        if let Ok(verified_transactions) = result {
569                            self.reconstructed_transactions.insert(tx_ref, verified_transactions);
570                            self.resolve_slot(tx_ref);
571                        }
572                    }
573
574                 () = &mut send_to_core_timeout => {
575
576                    // Grab reconstructed transactions and send them to core to add to the DAG state
577                    if let Err(e) = self.send_to_core().await {
578                        debug!("Error when sending reconstructed transactions to core: {:?}", e);
579                    }
580
581                    send_to_core_timeout
582                        .as_mut()
583                        .reset(Instant::now() + SEND_TO_CORE_RECONSTRUCTED_TXS_TIMEOUT);
584                        }
585
586                 () = &mut eviction_timeout => {
587
588                    // Clean accumulators and processed transaction from memory
589                    self.evict_memory();
590
591                    eviction_timeout
592                        .as_mut()
593                        .reset(Instant::now() + EVICTION_TIMEOUT);
594                }
595
596            }
597        }
598    }

600    /// Evict old accumulators and processed transactions to free memory. We
601    /// read the dag state to find the transaction garbage collection round
602    /// and evict all accumulators and processed transactions below that
603    /// round.
604    fn evict_memory(&mut self) {
605        self.context
606            .metrics
607            .node_metrics
608            .shard_accumulators
609            .set(self.shard_accumulators.len() as i64);
610        self.context
611            .metrics
612            .node_metrics
613            .reconstruction_queue
614            .set(self.reconstruction_queue.len() as i64);
615        self.context
616            .metrics
617            .node_metrics
618            .shard_reconstructor_processed_transactions
619            .set(self.processed_transactions.len() as i64);
620
621        let transaction_gc_round = self.dag_state.read().gc_round_for_last_solid_commit();
622        self.evict_below(transaction_gc_round);
623    }
624
625    /// Evict accumulators, processed and reconstructed transactions, resolved
626    /// slots, and retained-shard bookkeeping below the given round.
627    fn evict_below(&mut self, transaction_gc_round: Round) {
628        // Update the internal transaction_gc_round
629        self.transaction_gc_round = transaction_gc_round;
630
631        let lower_bound = TransactionRef {
632            round: transaction_gc_round,
633            author: AuthorityIndex::ZERO,
634            transactions_commitment: TransactionsCommitment::MIN,
635        };
636
637        self.processed_transactions = self.processed_transactions.split_off(&lower_bound);
638        self.reconstructed_transactions = self.reconstructed_transactions.split_off(&lower_bound);
639        self.shard_accumulators = self.shard_accumulators.split_off(&lower_bound);
640        self.reconstruction_queue = self.reconstruction_queue.split_off(&lower_bound);

644        for retained in &mut self.retained_shards_by_authority {
645            *retained = retained.split_off(&lower_bound);
646        }

649    fn get_transactions_with_headers_in_dag_state(
650        &mut self,
651    ) -> Vec<CommitmentVerifiedTransactions> {
652        let transactions_map = std::mem::take(&mut self.reconstructed_transactions);
653        // In most cases, all reconstructed transactions will go to the core
654        let mut ready_to_be_sent_transactions = Vec::new();
655
656        // We introduce a check about the existence of block headers to ensure that for
657        // every transaction, we have the respected header in the dag state
658        self.reconstructed_transactions = {
659            #[cfg(not(test))]
660            {
661                let mut to_stay_transactions = BTreeMap::new();
662                let block_headers_exist = {
663                    let tx_refs: Vec<TransactionRef> = transactions_map.keys().copied().collect();
664                    self.dag_state
665                        .read()
666                        .contains_verified_block_headers_for_transaction_refs(&tx_refs)
667                };
668                for (exists, (tx_ref, transactions)) in
669                    block_headers_exist.into_iter().zip(transactions_map)
670                {
671                    if exists {
672                        ready_to_be_sent_transactions.push(transactions);
673                    } else {
674                        to_stay_transactions.insert(tx_ref, transactions);
675                    }
676                }
677                to_stay_transactions
678            }
679            #[cfg(test)]
680            {
681                for transactions in transactions_map.values() {
682                    ready_to_be_sent_transactions.push(transactions.clone());
683                }
684                BTreeMap::new()
685            }
686        };
687        self.context
688            .metrics
689            .node_metrics
690            .reconstructed_transactions_unknown
691            .set(self.reconstructed_transactions.len() as i64);
692

694    }

696    /// Send reconstructed transactions to the core
697    async fn send_to_core(&mut self) -> ConsensusResult<()> {
698        let transactions = self.get_transactions_with_headers_in_dag_state();
699        if !transactions.is_empty() {
700            let highest_accepted_round = self.dag_state.read().highest_accepted_round();
701            for transaction in &transactions {
702                let difference = highest_accepted_round.saturating_sub(transaction.round());
703                self.context
704                    .metrics
705                    .node_metrics
706                    .reconstruction_lag
707                    .observe(difference as f64);
708            }
709
710            // Add the transactions to the core
711            self.core_dispatcher
712                .add_transactions(transactions, DataSource::ShardReconstructor)
713                .await
714                .map_err(|_| ConsensusError::Shutdown)?;
715        }
716        Ok(())
717    }

719    /// Handle a message and update internal state
720    async fn handle_transaction_message(&mut self, msg: TransactionMessage) -> ConsensusResult<()> {
721        let tx_ref = msg.transaction_ref();
722
723        if self.processed_transactions.contains(&tx_ref)
724            || self.reconstruction_queue.contains(&tx_ref)
725            || tx_ref.round < self.transaction_gc_round
726        {
727            return Ok(());
728        }
729
730        let total_length = self.total_length;
731
732        let shard_msg = match msg {
733            TransactionMessage::FullTransaction(tx_ref) => {
734                self.processed_transactions.insert(tx_ref);
735                // The full payload arrived through the direct path, so no
736                // accumulator in the slot can ever be needed again — release
737                // them instead of waiting for round eviction.
738                self.resolve_slot(tx_ref);
739                return Ok(());
740            }
741            TransactionMessage::Shard(shard_msg) => shard_msg,
742        };
743
744        // The slot's genuine payload is already known, so this shard
745        // can only re-grow an accumulator the resolution purged.
746        if self.resolved_slots.contains(&(tx_ref.round, tx_ref.author)) {
747            self.context
748                .metrics
749                .node_metrics
750                .shard_reconstructor_dropped_shards
751                .with_label_values(&["slot_resolved"])
752                .inc();
753            debug!("Dropping shard for {tx_ref:?}: its slot is already resolved");
754            return Ok(());
755        }
756
757        // Relaying two shards for one slot is the peer's own fault and is
758        // charged; exceeding the accumulator limit is not, as others may
759        // have filled the slot.
760        let slot_start = TransactionRef {
761            round: tx_ref.round,
762            author: tx_ref.author,
763            transactions_commitment: TransactionsCommitment::MIN,
764        };
765        let slot_end = TransactionRef {
766            round: tx_ref.round,
767            author: tx_ref.author,
768            transactions_commitment: TransactionsCommitment::MAX,
769        };
770        let mut accumulators_in_slot = 0usize;
771        let mut peer_in_other_accumulator = false;
772        for (existing_ref, accumulator) in self.shard_accumulators.range(slot_start..=slot_end) {
773            accumulators_in_slot += 1;
774            if *existing_ref != tx_ref && accumulator.contains_shard_at_index(shard_msg.shard_index)
775            {
776                peer_in_other_accumulator = true;
777            }
778        }
779
780        // One shard per (relaying peer, slot), across all accumulators: the
781        // per-slot header cap leaves an honest peer holding exactly one own
782        // shard per slot, so a peer whose index already appears in another
783        // accumulator of the slot has relayed a shard it could not have held
784        // honestly. The shard carries no relayer signature, so the evidence
785        // stays local.
786        if peer_in_other_accumulator {
787            self.context
788                .metrics
789                .node_metrics
790                .shard_reconstructor_dropped_shards
791                .with_label_values(&["peer_already_in_slot"])
792                .inc();
793            debug!(
794                "Dropping shard for {tx_ref:?}: peer index {} already contributed a shard in this slot",
795                shard_msg.shard_index
796            );
797            if let Some(peer) = self
798                .context
799                .committee
800                .to_authority_index(shard_msg.shard_index)
801            {
802                self.misbehavior_store.record_faulty_block(
803                    peer,
804                    peer,
805                    &ConsensusError::SecondShardForSlot {
806                        peer,
807                        author: tx_ref.author,
808                        round: tx_ref.round,
809                    },
810                );
811            }
812            return Ok(());
813        }
814
815        let shard_index = shard_msg.shard_index;
816        let occupies_new_position = match self.shard_accumulators.get(&tx_ref) {
817            Some(accumulator) => !accumulator.contains_shard_at_index(shard_index),
818            None => {
819                // With one shard per (peer, slot), a slot holding
820                // `parity_length + 1` accumulators has too few
821                // uncommitted peers left for any further commitment to
822                // ever gather `info_length` contributors — a new
823                // accumulator would be dead weight by construction.
824                let max_accumulators_per_slot = self.context.committee.parity_length() + 1;
825                if accumulators_in_slot >= max_accumulators_per_slot {
826                    self.context
827                        .metrics
828                        .node_metrics
829                        .shard_reconstructor_dropped_shards
830                        .with_label_values(&["slot_full"])
831                        .inc();
832                    debug!(
833                        "Dropping shard for {tx_ref:?}: slot already holds {accumulators_in_slot} accumulators"
834                    );
835                    return Ok(());
836                }
837                true
838            }
839        };
840        if occupies_new_position {
841            self.make_room_in_peer_budget(shard_index);
842            self.retained_shards_by_authority[shard_index].insert(tx_ref);
843        }
844        match self.shard_accumulators.entry(tx_ref) {
845            Entry::Vacant(v) => {
846                v.insert(ShardAccumulator::new_with_shard(shard_msg, total_length));
847            }
848            Entry::Occupied(mut o) => {
849                o.get_mut().update_with_shard(shard_msg);
850            }
851        }
852
853        // Check if we can reconstruct the block now and enqueue it if so
854        Self::enqueue_if_ready(
855            &mut self.shard_accumulators,
856            &mut self.reconstruction_queue,
857            &self.ready_to_reconstruct_sender,
858            self.info_length,
859            &self.context.committee,
860            &tx_ref,
861            &mut self.retained_shards_by_authority,
862        )
863        .await?;
864
865        Ok(())
866    }

868    /// If the accumulator for the given key is ready to reconstruct, remove it
869    /// from the map and enqueue it for reconstruction
870    async fn enqueue_if_ready(
871        accumulators: &mut BTreeMap<TransactionRef, ShardAccumulator>,
872        reconstruction_queue: &mut BTreeSet<TransactionRef>,
873        sender: &Sender<ShardAccumulator>,
874        info_length: usize,
875        committee: &Committee,
876        tx_ref: &TransactionRef,
877        retained_shards_by_authority: &mut [BTreeSet<TransactionRef>],
878    ) -> ConsensusResult<()> {
879        if let Some(acc) = accumulators.get(tx_ref) {
880            if acc.is_ready_to_reconstruct(info_length, committee) {
881                // take ownership out of map
882                let acc = accumulators
883                    .remove(tx_ref)
884                    .expect("We should expect the shard accumulator to be present");
885                for shard_index in acc.collected_shard_indices() {
886                    retained_shards_by_authority[shard_index].remove(tx_ref);
887                }
888                sender
889                    .send(acc)
890                    .await
891                    .map_err(|_| ConsensusError::AccumulatorSenderClosed)?;
892                reconstruction_queue.insert(*tx_ref);
893            }
894        }
895        Ok(())
896    }

898    /// At the authority's shard budget, evicts its oldest (lowest-round)
899    /// retained shard to admit the new one. Whatever eviction costs stays
900    /// fetchable via the transaction synchronizer, since a decodable
901    /// payload's relayers hold it in full.
902    fn make_room_in_peer_budget(&mut self, shard_index: usize) {
903        let retained = &self.retained_shards_by_authority[shard_index];
904        if retained.len() < self.shard_budget_per_authority {
905            return;
906        }
907        let Some(victim_ref) = retained.first().copied() else {
908            return;
909        };
910        self.retained_shards_by_authority[shard_index].remove(&victim_ref);
911        if let Entry::Occupied(mut occupied) = self.shard_accumulators.entry(victim_ref) {
912            occupied.get_mut().remove_shard_at_index(shard_index);
913            if occupied.get().number_shards == 0 {
914                occupied.remove();
915            }
916        }
917        self.context
918            .metrics
919            .node_metrics
920            .shard_reconstructor_dropped_shards
921            .with_label_values(&["peer_budget_evicted"])
922            .inc();
923        debug!(
924            "Evicted the oldest retained shard of peer index {shard_index} ({victim_ref:?}) to admit a new one"
925        );
926    }
927
928    /// Marks the slot of `tx_ref` resolved and releases every accumulator in
929    /// it. Both triggers pin an author-signed payload (a decode reached f+1
930    /// relayer stake, so an honest relayer coded it from the signed payload);
931    /// every other commitment in the slot is fabricated, except an
932    /// equivocating twin, which the transaction synchronizer can still fetch.
933    fn resolve_slot(&mut self, tx_ref: TransactionRef) {
934        self.resolved_slots.insert((tx_ref.round, tx_ref.author));
935        let slot_start = TransactionRef {
936            round: tx_ref.round,
937            author: tx_ref.author,
938            transactions_commitment: TransactionsCommitment::MIN,
939        };
940        let slot_end = TransactionRef {
941            round: tx_ref.round,
942            author: tx_ref.author,
943            transactions_commitment: TransactionsCommitment::MAX,
944        };
945        let purged_refs: Vec<TransactionRef> = self
946            .shard_accumulators
947            .range(slot_start..=slot_end)
948            .map(|(purged_ref, _)| *purged_ref)
949            .collect();
950        for purged_ref in purged_refs {
951            if let Some(accumulator) = self.shard_accumulators.remove(&purged_ref) {
952                for shard_index in accumulator.collected_shard_indices() {
953                    self.retained_shards_by_authority[shard_index].remove(&purged_ref);
954                }
955            }
956        }
957    }
