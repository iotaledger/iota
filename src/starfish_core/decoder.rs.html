10/// Trait for decoding shard collections using systematic Reed-Solomon decoding
11/// and reconstructing the original transactions.
12pub trait ShardsDecoder {
13    /// Attempts to decode a collection of arbitrary >= info_length shards into
14    /// a vector of Transactions.
15    fn decode_shards(
16        &mut self,
17        info_length: usize,
18        parity_length: usize,
19        shards_collection: Vec<Option<Shard>>,
20    ) -> Result<Vec<Transaction>, ConsensusError>;
21
22    /// Reconstructs the original vector of Transactions from a vector of first
23    /// info_length shards.
24    fn reconstruct_transactions(
25        shards: Vec<Shard>,
26        info_length: usize,
27    ) -> Result<Vec<Transaction>, ConsensusError>
28    where
29        Self: Sized;
30}
31
32impl ShardsDecoder for ReedSolomonDecoder {
33    fn decode_shards(
34        &mut self,
35        info_length: usize,
36        parity_length: usize,
37        shards_collection: Vec<Option<Shard>>,
38    ) -> Result<Vec<Transaction>, ConsensusError> {
39        let total_length = info_length + parity_length;
40
41        assert_eq!(
42            shards_collection.len(),
43            total_length,
44            "Shards collection length must match total_length"
45        );
46
47        let shards_count = shards_collection.iter().filter(|x| x.is_some()).count();
48        if shards_count < info_length {
49            return Err(ConsensusError::InsufficientShardsInDecoder(
50                shards_count,
51                info_length,
52            ));
53        }
54
55        let position = shards_collection
56            .iter()
57            .position(|x| x.is_some())
58            .ok_or(ConsensusError::InsufficientShardsInDecoder(0, info_length))?;
59        let shard_size = shards_collection[position].as_ref().unwrap().len();
60
61        self.reset(info_length, parity_length, shard_size)
62            .map_err(|e| ConsensusError::EncoderResetFailed(e.to_string()))?;
63
64        for (i, maybe_shard) in shards_collection.iter().take(info_length).enumerate() {
65            if let Some(shard) = maybe_shard {
66                self.add_original_shard(i, shard)
67                    .map_err(|e| ConsensusError::AddShardFailed(e.to_string()))?;
68            }
69        }
70
71        for (i, maybe_shard) in shards_collection.iter().enumerate().skip(info_length) {
72            if let Some(shard) = maybe_shard {
73                self.add_recovery_shard(i - info_length, shard)
74                    .map_err(|e| ConsensusError::AddShardFailed(e.to_string()))?;
75            }
76        }
77
78        let mut data: Vec<Shard> = vec![vec![]; info_length];
79        for (i, item) in data.iter_mut().enumerate().take(info_length) {
80            if shards_collection[i].is_some() {
81                *item = shards_collection[i].clone().unwrap();
82            }
83        }
84
85        let result = self
86            .decode()
87            .map_err(|e| ConsensusError::ShardsDecodingFailed(e.to_string()))?;
88
89        let restored: HashMap<_, _> = result.restored_original_iter().collect();
90        for el in restored {
91            data[el.0] = Shard::from(el.1);
92        }
93        drop(result);
94
95        Self::reconstruct_transactions(data, info_length)
96    }
97
98    fn reconstruct_transactions(
99        shards: Vec<Shard>,
100        info_length: usize,
101    ) -> Result<Vec<Transaction>, ConsensusError> {
102        reconstruct_transactions_impl(shards, info_length)
103    }
104}

106/// Trivial decoder: assumes no redundancy, just reconstructs transactions
107/// directly from the provided shards.
108pub(crate) struct TrivialDecoder {}
109
110impl ShardsDecoder for TrivialDecoder {
111    fn decode_shards(
112        &mut self,
113        info_length: usize,
114        parity_length: usize,
115        shards_collection: Vec<Option<Shard>>,
116    ) -> Result<Vec<Transaction>, ConsensusError> {
117        let total_length = info_length + parity_length;
118        assert_eq!(
119            shards_collection.len(),
120            total_length,
121            "Shards collection length must match total_length"
122        );
123
124        let mut data: Vec<Shard> = Vec::with_capacity(info_length);
125        for maybe_shard in shards_collection.into_iter().take(info_length) {
126            match maybe_shard {
127                Some(shard) => data.push(shard),
128                None => {
129                    return Err(ConsensusError::InsufficientShardsInDecoder(
130                        data.len(),
131                        info_length,
132                    ));
133                }
134            }
135        }
136
137        Self::reconstruct_transactions(data, info_length)
138    }
139
140    fn reconstruct_transactions(
141        shards: Vec<Shard>,
142        info_length: usize,
143    ) -> Result<Vec<Transaction>, ConsensusError> {
144        reconstruct_transactions_impl(shards, info_length)
145    }
146}

148/// Common logic for reconstructing transactions from data shards.
149fn reconstruct_transactions_impl(
150    shards: Vec<Shard>,
151    info_length: usize,
152) -> Result<Vec<Transaction>, ConsensusError> {
153    assert_eq!(
154        shards.len(),
155        info_length,
156        "Data length must match info length"
157    );
158    assert!(info_length > 0, "Info length must be greater than 0");
159
160    let mut reconstructed_data = Vec::new();
161    for shard in shards.iter().take(info_length) {
162        reconstructed_data.extend(shard);
163    }
164
165    // Read the first 4 bytes for `bytes_length` to get the size of the original
166    // serialized block
167    if reconstructed_data.len() < 4 {
168        return Err(ConsensusError::ShardsVecIsTooSmall(
169            reconstructed_data.len(),
170            4,
171        ));
172    }
173
174    let bytes_length = u32::from_le_bytes(
175        reconstructed_data[0..4]
176            .try_into()
177            .map_err(|_| ConsensusError::ShardsVecIsTooSmall(reconstructed_data.len(), 4))?,
178    ) as usize;
179
180    if reconstructed_data.len() < 4 + bytes_length {
181        return Err(ConsensusError::ShardsVecIsTooSmall(
182            reconstructed_data.len(),
183            4 + bytes_length,
184        ));
185    }
186
187    tracing::debug!(
188        "Reconstructed data length {}, bytes_length {}",
189        reconstructed_data.len(),
190        bytes_length
191    );
192
193    let serialized_block = &reconstructed_data[4..4 + bytes_length];
194    let reconstructed_transactions: Vec<Transaction> =
195        bcs::from_bytes(serialized_block).map_err(ConsensusError::DeserializationFailure)?;
196    Ok(reconstructed_transactions)
197}

199/// Creates a decoder from the context, using ReedSolomonDecoder if redundancy
200/// is present, otherwise falling back to TrivialDecoder.
201pub(crate) fn create_decoder(context: &Arc<Context>) -> Box<dyn ShardsDecoder + Send + Sync> {
202    let info_length = context.committee.info_length();
203    let parity_length = context.committee.size() - info_length;
204
205    if info_length > 0 && parity_length > 0 {
206        Box::new(
207            ReedSolomonDecoder::new(info_length, parity_length, 2)
208                .expect("We should expect correct creation of the ReedSolomonDecoder"),
209        )
210    } else {
211        Box::new(TrivialDecoder {})
212    }
213}
214#[cfg(test)]
215mod tests {
216    use std::sync::Arc;
217
218    use crate::{
219        Transaction, block_header::Shard, context::Context, decoder::create_decoder,
220        encoder::create_encoder,
221    };
222
223    #[tokio::test]
224    async fn decode_should_fail_cases() {
225        let (context, _) = Context::new_for_test(4); // info=2, parity=2
226        let context = Arc::new(context);
227        let mut encoder = create_encoder(&context);
228        let mut decoder = create_decoder(&context);
229
230        let transactions = Transaction::random_transactions(3, 64);
231        let serialized = Transaction::serialize(&transactions).expect("serialization should work");
232
233        let shards = encoder
234            .encode_serialized_data(&serialized, context.committee.info_length(), 2)
235            .expect("encode should succeed");
236
237        // Case 1: too few shards (< info_length)
238        let shards_collection: Vec<Option<Shard>> = vec![Some(shards[0].clone()), None, None, None];
239        assert!(
240            decoder
241                .decode_shards(context.committee.info_length(), 2, shards_collection)
242                .is_err()
243        );
244
245        // Case 2: corrupted shard length
246        let mut shards_collection: Vec<Option<Shard>> =
247            shards.clone().into_iter().map(Some).collect();
248        shards_collection[1] = Some(vec![1, 2, 3]); // wrong size
249        assert!(
250            decoder
251                .decode_shards(context.committee.info_length(), 2, shards_collection)
252                .is_err()
253        );
254
255        // Case 3: missing too many shards (drop 3 shards, parity=2)
256        let mut shards_collection: Vec<Option<Shard>> = shards.into_iter().map(Some).collect();
257        shards_collection[0] = None;
258        shards_collection[1] = None;
259        shards_collection[2] = None;
260        assert!(
261            decoder
262                .decode_shards(context.committee.info_length(), 2, shards_collection)
263                .is_err()
264        );
265    }
