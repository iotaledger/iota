21/// Per-authority misbehavior counters.
22///
23/// Two buckets track different lifecycle stages:
24/// - `in_memory`: from blocks currently in the DAG cache (volatile, recomputed
25///   on each flush from blocks still in cache).
26/// - `persisted`: cumulative counts from blocks evicted from cache (restored
27///   from storage on restart).
28pub(crate) struct MisbehaviorStore {
29    in_memory: CommitteeMisbehaviorCounts,
30    persisted: CommitteeMisbehaviorCounts,
31    /// Per-authority rounds in which a second verified header for an occupied
32    /// slot was dropped at ingest. Counted into the in-memory equivocations on
33    /// each flush and folded into the persisted count when the round is
34    /// evicted.
35    equivocating_rounds: Vec<Mutex<BTreeSet<Round>>>,
36}
37
38impl MisbehaviorStore {
39    pub(crate) fn new(context: &Context) -> Self {
40        let metrics = &context.metrics.node_metrics;
41        Self {
42            in_memory: CommitteeMisbehaviorCounts::new(&context.committee, metrics, "in_memory"),
43            persisted: CommitteeMisbehaviorCounts::new(&context.committee, metrics, "persisted"),
44            equivocating_rounds: (0..context.committee.size())
45                .map(|_| Mutex::new(BTreeSet::new()))
46                .collect(),
47        }
48    }
49
50    /// Resets all counters to zero. Used during fast sync reinitialization.
51    pub(crate) fn reset(&self) {
52        self.in_memory.reset();
53        self.persisted.reset();
54        for rounds in &self.equivocating_rounds {
55            rounds.lock().unwrap().clear();
56        }
57    }
58
59    /// Records that `author` signed more than one header for `round`: a second
60    /// verified header for an occupied slot was dropped at ingest. Idempotent —
61    /// repeated arrivals and further distinct headers for the same slot
62    /// collapse into one equivocation. Callers pass the author of a verified
63    /// header, so an author outside the committee is a verification bug and is
64    /// reported rather than recorded.
65    pub(crate) fn record_equivocating_slot(&self, author: AuthorityIndex, round: Round) {
66        let Some(rounds) = self.equivocating_rounds.get(author.value()) else {
67            debug_fatal!("equivocating author {author} is outside the committee");
68            return;
69        };
70        rounds.lock().unwrap().insert(round);
71    }

73    /// Restores persisted counts from storage and computes in-memory counts
74    /// from the block refs already loaded into the DAG cache.
75    pub(crate) fn initialize_misbehavior_counts(
76        &self,
77        recovered: BTreeMap<AuthorityIndex, MisbehaviorCounts>,
78        recent_refs_by_authority: &[BTreeSet<BlockRef>],
79        evicted_rounds: &[Round],
80        threshold_clock_round: Round,
81        context: &Arc<Context>,
82    ) {
83        for (authority_index, _) in context.committee.authorities() {
84            let idx = authority_index.value();
85
86            // Restore persisted counts from storage. Match exhaustively so a
87            // future MisbehaviorCounts variant trips the compiler here.
88            let storage_metrics = recovered.get(&authority_index).cloned().unwrap_or_default();
89            match storage_metrics {
90                // V1 rows carry no bundle-part count; restore it as zero.
91                MisbehaviorCounts::V1(inner) => {
92                    self.persisted.set_block_faults(
93                        idx,
94                        inner.faulty_blocks_provable,
95                        inner.faulty_blocks_unprovable,
96                        0,
97                    );
98                    self.persisted.set_dag_faults(
99                        idx,
100                        inner.missing_proposals,
101                        inner.equivocations,
102                    );
103                }
104                MisbehaviorCounts::V2(inner) => {
105                    self.persisted.set_block_faults(
106                        idx,
107                        inner.faulty_blocks_provable,
108                        inner.faulty_blocks_unprovable,
109                        inner.invalid_bundle_parts,
110                    );
111                    self.persisted.set_dag_faults(
112                        idx,
113                        inner.missing_proposals,
114                        inner.equivocations,
115                    );
116                }
117            }
118
119            // Compute in-memory counts from cached block refs.
120            let eviction_round = evicted_rounds[idx];
121            if threshold_clock_round > 0 {
122                let cached_rounds: Vec<Round> = recent_refs_by_authority[idx]
123                    .iter()
124                    .map(|r| r.round)
125                    .collect();
126                // Flags are not persisted, so only rounds with more than one
127                // cached header count after a restart.
128                let (eq, missing) = calculate_misbehavior_counts_for_range(
129                    cached_rounds,
130                    &BTreeSet::new(),
131                    eviction_round + 1,
132                    threshold_clock_round.saturating_sub(1),
133                );
134                self.in_memory.set_dag_faults(idx, missing, eq);
135            }
136        }
137    }

139    /// Updates misbehavior counts for one authority during flush.
140    /// Must be called before cache eviction is applied, while evicted block
141    /// refs are still in `recent_refs`.
142    /// Returns `Some(MisbehaviorCounts)` if persisted data changed and needs
143    /// to be written to storage.
144    pub(crate) fn update_misbehavior_counts_on_eviction(
145        &self,
146        authority_index: AuthorityIndex,
147        recent_refs: &BTreeSet<BlockRef>,
148        eviction_round: Round,
149        last_eviction_round: Round,
150        threshold_clock_round: Round,
151        context: &Arc<Context>,
152    ) -> Option<MisbehaviorCounts> {
153        if threshold_clock_round == 0 {
154            return None;
155        }
156        let idx = authority_index.value();
157        if idx >= context.committee.size() {
158            debug_fatal!("authority {authority_index} is outside the committee");
159            return None;
160        }
161
162        // Move buffered faulty block counts to persisted.
163        let had_faulty = self.flush_faulty_block_buffer(idx);
164
165        let mut equivocating_rounds = self.equivocating_rounds[idx].lock().unwrap();
166
167        // Recompute in-memory window from blocks still in cache.
168        let in_memory_block_rounds: Vec<Round> = recent_refs
169            .iter()
170            .map(|b| b.round)
171            .filter(|&r| r > eviction_round && r < threshold_clock_round)
172            .collect();
173        let (in_memory_eq, in_memory_missing) = calculate_misbehavior_counts_for_range(
174            in_memory_block_rounds,
175            &equivocating_rounds,
176            eviction_round + 1,
177            threshold_clock_round.saturating_sub(1),
178        );
179        self.in_memory
180            .set_dag_faults(idx, in_memory_missing, in_memory_eq);
181
182        let eviction_advanced = eviction_round != last_eviction_round;
183        if eviction_advanced {
184            // Accumulate newly-evicted rounds into persisted.
185            let evicted_block_rounds: Vec<Round> = recent_refs
186                .iter()
187                .map(|b| b.round)
188                .filter(|&r| r <= eviction_round)
189                .collect();
190            let (evicted_eq, evicted_missing) = calculate_misbehavior_counts_for_range(
191                evicted_block_rounds,
192                &equivocating_rounds,
193                last_eviction_round + 1,
194                eviction_round,
195            );
196            self.persisted
197                .add_dag_faults(idx, evicted_missing, evicted_eq);
198            // Evicted rounds are folded into persisted above; drop their flags.
199            let live_rounds = equivocating_rounds.split_off(&(eviction_round + 1));
200            *equivocating_rounds = live_rounds;
201        }
202
203        if eviction_advanced || had_faulty {
204            let counts = self.persisted.snapshot(idx);
205            let persisted = match context.protocol_config.scorer_version_as_option() {
206                Some(2) => MisbehaviorCounts::V2(counts),
207                _ => MisbehaviorCounts::V1(counts.fold_into_v1()),
208            };
209            Some(persisted)
210        } else {
211            None
212        }
213    }

215    /// Flush buffered faulty block counts from in_memory to persisted
216    /// for one authority. Returns true if any counts were moved.
217    fn flush_faulty_block_buffer(&self, idx: usize) -> bool {
218        let (prov, unprov, bundle) = self.in_memory.drain_block_faults(idx);
219        if prov == 0 && unprov == 0 && bundle == 0 {
220            return false;
221        }
222        self.persisted.add_block_faults(idx, prov, unprov, bundle);
223        true
224    }

226    /// Returns an absolute per-authority snapshot of `persisted + in_memory`
227    /// counts for emission with `CommittedSubDag`. Locks the two buckets
228    /// independently per authority; callers must hold `dag_state.read()` so
229    /// concurrent flush (which writes both buckets under `dag_state.write()`)
230    /// is excluded.
231    pub(crate) fn snapshot_totals(&self) -> Vec<MisbehaviorCounts> {
232        (0..self.in_memory.authorities.len())
233            .map(|i| {
234                let persisted = self.persisted.snapshot(i);
235                let in_memory = self.in_memory.snapshot(i);
236                MisbehaviorCounts::V2(MisbehaviorCountsV2 {
237                    faulty_blocks_provable: persisted.faulty_blocks_provable
238                        + in_memory.faulty_blocks_provable,
239                    faulty_blocks_unprovable: persisted.faulty_blocks_unprovable
240                        + in_memory.faulty_blocks_unprovable,
241                    missing_proposals: persisted.missing_proposals + in_memory.missing_proposals,
242                    equivocations: persisted.equivocations + in_memory.equivocations,
243                    invalid_bundle_parts: persisted.invalid_bundle_parts
244                        + in_memory.invalid_bundle_parts,
245                })
246            })
247            .collect()
248    }

250    /// Records a faulty block event detected during ingest validation — a
251    /// rejected block header, or a corrupt part of a relayed bundle (framing,
252    /// metadata, or shard). Events are buffered in the in_memory bucket and
253    /// moved to persisted on the next flush.
254    ///
255    /// `peer` is the authority that sent us the block (always known from the
256    /// network connection). `author` is the claimed block author (from the
257    /// header's author field, only trustworthy if the signature is valid).
258    ///
259    /// Attribution rules:
260    /// - Provable faults (valid signature, protocol violation): charged to
261    ///   `author` — cryptographic proof they produced an invalid block. If
262    ///   `peer != author`, the peer is also charged (unprovable) for
263    ///   distributing a block they could have verified themselves.
264    /// - Unprovable faults (bad/missing signature): charged to `peer` only — we
265    ///   can't verify the author field, but we know who sent it to us.
266    /// - Bundle-part faults (corrupt framing, metadata, or shard, or a shard
267    ///   the peer had no right to relay): charged to `peer` only, under the
268    ///   dedicated bundle-part counter.
269    pub(crate) fn record_faulty_block(
270        &self,
271        peer: AuthorityIndex,
272        author: AuthorityIndex,
273        error: &ConsensusError,
274    ) {
275        // `peer` is trusted (authenticated connection); `author` is
276        // attacker-controlled and may be out of committee range.
277        let committee_size = self.in_memory.authorities.len();
278        let peer_idx = peer.value();
279        let author_idx = author.value();
280        if peer_idx >= committee_size {
281            debug_fatal!("peer {peer} is outside the committee");
282            return;
283        }
284        match classify_block_error(error) {
285            FaultType::Provable => {
286                if author_idx >= committee_size {
287                    // Can't credit a bogus author; charge the serving peer instead.
288                    self.in_memory.record_block_fault_unprovable(peer_idx);
289                    return;
290                }
291                self.in_memory.record_block_fault_provable(author_idx);
292                if peer != author {
293                    self.in_memory.record_block_fault_unprovable(peer_idx);
294                }
295            }
296            FaultType::Unprovable => {
297                self.in_memory.record_block_fault_unprovable(peer_idx);
298            }
299            FaultType::BundlePart => {
300                self.in_memory.record_bundle_part_fault(peer_idx);
301            }
302            FaultType::Untracked => {}
303        }
304    }
305
306    /// Attributes a transaction payload that failed the commitment check or
307    /// the transaction validity check.
308    ///
309    /// `author` is charged with a provable fault only when `authored` is true,
310    /// i.e. a verified, author-signed block header commits to this payload, so
311    /// the invalid transactions are provably the author's. Each peer in
312    /// `relayers` is charged an unprovable fault for handing us a payload it
313    /// should not have relayed; a relayer that is also the author is not
314    /// double-charged when the author is already charged provably.
315    pub(crate) fn record_faulty_transactions(
316        &self,
317        author: AuthorityIndex,
318        authored: bool,
319        relayers: impl IntoIterator<Item = AuthorityIndex>,
320    ) {
321        let committee_size = self.in_memory.authorities.len();
322        if authored {
323            if author.value() < committee_size {
324                self.in_memory.record_block_fault_provable(author.value());
325            } else {
326                debug_fatal!("author {author} of a verified header is outside the committee");
327            }
328        }
329        for peer in relayers {
330            let idx = peer.value();
331            if idx >= committee_size {
332                debug_fatal!("relaying peer {peer} is outside the committee");
333                continue;
334            }
335            if authored && peer == author {
336                continue;
337            }
338            self.in_memory.record_block_fault_unprovable(idx);
339        }
340    }

343/// Whether a block fault can be cryptographically proven.
344enum FaultType {
345    /// Block has a valid author signature but violates protocol rules.
346    /// The signed block header itself is proof of misbehavior.
347    Provable,
348    /// Can't prove the fault from the signed data — either because the
349    /// signature is bad or missing, or because the header is rejected by a
350    /// pre-signature check (epoch / genesis / author-vs-peer mismatch) so its
351    /// `author` field can't be trusted, or because the fault is in how the
352    /// peer sent the block (stream order) rather than in the block itself.
353    /// Charged to the sending peer, not the claimed author.
354    Unprovable,
355    /// Relayed bundle part (framing, metadata, or shard) that is corrupt,
356    /// invalid, or one the peer had no right to relay, and isn't tied to a
357    /// verified author. Charged to the sending peer.
358    BundlePart,
359    /// Not counted as misbehavior.
360    Untracked,
361}
362
363fn classify_block_error(error: &ConsensusError) -> FaultType {
364    // Exhaustive on purpose: a new `ConsensusError` variant must be assigned a
365    // fault type here rather than silently falling through to `Untracked`.
366    match error {
367        // Pre-signature / parsing errors — the header's author field can't
368        // be trusted, so charge the sender, not the claimed author. A streamed
369        // block that repeats or lowers the peer's own round is charged the
370        // same way: the signed header proves authorship, not the send order.
371        ConsensusError::WrongEpoch { .. }
372        | ConsensusError::UnexpectedGenesisHeader
373        | ConsensusError::UnexpectedAuthority(..)
374        | ConsensusError::InvalidAuthorityIndex { .. }
375        | ConsensusError::MalformedHeader(_)
376        | ConsensusError::MalformedTransactions(_)
377        | ConsensusError::MalformedSignature(_)
378        | ConsensusError::SignatureVerificationFailure(_)
379        | ConsensusError::SerializationFailure(_)
380        | ConsensusError::DeserializationFailure(_)
381        | ConsensusError::SerializedTransactionsTooLarge { .. }
382        | ConsensusError::TransactionCommitmentFailure { .. }
383        | ConsensusError::UnexpectedBlockHeaderForCommit { .. }
384        | ConsensusError::TooManyFetchedHeadersReturned { .. }
385        | ConsensusError::StreamedBlockRoundNotIncreasing { .. } => FaultType::Unprovable,
386
387        // Relayed bundle parts that are corrupt or invalid (framing,
388        // additional-header round, shard structure/proof) or that the peer had
389        // no right to relay (a second shard for one slot). We know which peer
390        // relayed them but can't tie them to a verified author.
391        ConsensusError::MalformedShard(_)
392        | ConsensusError::TooBigHeaderRoundInABundle { .. }
393        | ConsensusError::TooBigShardRoundInABundle { .. }
394        | ConsensusError::IncorrectShardProof { .. }
395        | ConsensusError::UnrequestedHeaderOutOfWindow { .. }
396        | ConsensusError::SerializedShardTooLarge { .. }
397        | ConsensusError::SecondShardForSlot { .. } => FaultType::BundlePart,
398
399        // Checks that run only after the author's signature is verified, so the
400        // signed header itself proves the author produced a block that violates
401        // protocol rules.
402        ConsensusError::TooManyAncestors(..)
403        | ConsensusError::InsufficientParentStakes { .. }
404        | ConsensusError::InvalidAncestorPosition { .. }
405        | ConsensusError::InvalidAncestorRound { .. }
406        | ConsensusError::AncestorRoundTooOld { .. }
407        | ConsensusError::AcknowledgmentRoundTooOld { .. }
408        | ConsensusError::InvalidGenesisAncestor(_)
409        | ConsensusError::DuplicatedAncestorsAuthority(_)
410        | ConsensusError::InvalidOverlapIndices { .. }
411        | ConsensusError::TooManyAcknowledgments { .. }
412        | ConsensusError::TooManyCommitVotes { .. }
413        | ConsensusError::InvalidAcknowledgmentRound { .. }
414        | ConsensusError::InvalidStrongVoteAuthority { .. }
415        | ConsensusError::StrongVoteLeaderNotInAncestors { .. }
416        | ConsensusError::TransactionTooLarge { .. }
417        | ConsensusError::TooManyTransactions { .. }
418        | ConsensusError::TooManyTransactionBytes { .. }
419        | ConsensusError::InvalidTransaction(_) => FaultType::Provable,
420
421        // Equivocation is counted once per slot when the header is dropped, so
422        // the error that rejects the delivery carrying it is not charged again.
423        ConsensusError::BlockHeaderEquivocation { .. }
424
425        // Not attributable as misbehavior from a signed block alone: subjective
426        // rejections, commit-chain and commit-sync inconsistencies, fetch-shape
427        // mismatches, reconstruction and encoder failures, storage, network
428        // transport, request-shape checks, local lifecycle signals, and version
429        // mismatches that may stem from a benign upgrade misconfiguration rather
430        // than a faulty peer.
431        | ConsensusError::MalformedCommit(_)
432        | ConsensusError::UnexpectedGenesisRequested { .. }
433        | ConsensusError::NotEnoughHeadersFetched { .. }
434        | ConsensusError::UnexpectedLastOwnHeader { .. }
435        // Transaction fetch faults, recorded against the serving peer at the
436        // fetch sites via `record_faulty_transactions` — deliberately not
437        // tracked again here.
438        | ConsensusError::TooManyFetchedTransactionsReturned(_)
439        | ConsensusError::UnrequestedTransactionFetched { .. }
440        | ConsensusError::UnexpectedTransactionForCommit { .. }
441        | ConsensusError::TooManyAuthoritiesProvided(_)
442        | ConsensusError::InvalidSizeOfHighestAcceptedRounds(..)
443        | ConsensusError::InvalidAuthorityIndexRequested { .. }
444        | ConsensusError::TransactionCommitmentMismatch { .. }
445        | ConsensusError::SynchronizerSaturated(_)
446        | ConsensusError::TransactionSynchronizerSaturated
447        | ConsensusError::BlockRejected { .. }
448        | ConsensusError::EmptyMerkleTree
449        | ConsensusError::MissingBlockHeader { .. }
450        | ConsensusError::CommitRangeExceededAfterScanning { .. }
451        | ConsensusError::TooManyCommitsFromPeer { .. }
452        | ConsensusError::NoCommitReceived { .. }
453        | ConsensusError::UnexpectedStartCommit { .. }
454        | ConsensusError::UnexpectedCommitSequence { .. }
455        | ConsensusError::NotEnoughCommitVotes { .. }
456        | ConsensusError::TooManyCommitVoteHeaders { .. }
457        | ConsensusError::SerializedCommitTooLarge { .. }
458        | ConsensusError::SerializedBlockHeaderTooLarge { .. }
459        | ConsensusError::InvalidCommitRange { .. }
460        | ConsensusError::FetchedTransactionsMismatch { .. }
461        | ConsensusError::RocksDBFailure(_)
462        | ConsensusError::NetworkConfig(_)
463        | ConsensusError::NetworkClientConnection(_)
464        | ConsensusError::NetworkRequest(_)
465        | ConsensusError::NetworkRequestTimeout(_)
466        | ConsensusError::AccumulatorSenderClosed
467        | ConsensusError::Shutdown
468        | ConsensusError::EncoderResetFailed(_)
469        | ConsensusError::AddShardFailed(_)
470        | ConsensusError::ShardsEncodingFailed(_)
471        | ConsensusError::ShardsDecodingFailed(_)
472        | ConsensusError::InsufficientShardsInDecoder(..)
473        | ConsensusError::ShardsVecIsTooSmall(..)
474        | ConsensusError::InconsistentTransactionRefVariants
475        | ConsensusError::TransactionRefVariantMismatch { .. }
476        | ConsensusError::FailedToFetchBlockHeaders { .. }
477        | ConsensusError::MissingVotingBlockHeaderInStorage { .. }
478        | ConsensusError::WrongShardVersion { .. }
479        | ConsensusError::WrongCommitVersionForFlags { .. }
480        | ConsensusError::WrongBlockHeaderVersionForFlag { .. } => FaultType::Untracked,
481    }
482}

484/// The five Prometheus gauges that mirror `MisbehaviorCountsV2` for one
485/// `(authority, source)` pair, pre-resolved at construction so per-call
486/// methods don't repeat `with_label_values` lookups and don't have to
487/// thread `&NodeMetrics` and `hostname` through every call site.
488struct AuthorityGauges {
489    block_fault_provable: IntGauge,
490    block_fault_unprovable: IntGauge,
491    missing_proposals: IntGauge,
492    equivocations: IntGauge,
493    invalid_bundle_parts: IntGauge,
494}
495
496/// Per-authority misbehavior counters and matching Prometheus gauges.
497/// State mutations and gauge updates always happen together inside the same
498/// per-authority `Mutex` lock, so concurrent observers see counter and gauge
499/// agree.
500struct CommitteeMisbehaviorCounts {
501    authorities: Vec<Mutex<MisbehaviorCountsV2>>,
502    gauges: Vec<AuthorityGauges>,
503}
504
505impl CommitteeMisbehaviorCounts {
506    /// `source` is the `source` label baked into all gauge updates for this
507    /// instance (`"in_memory"` or `"persisted"`).
508    fn new(committee: &Committee, metrics: &NodeMetrics, source: &'static str) -> Self {
509        let gauges = committee
510            .authorities()
511            .map(|(_, a)| {
512                let labels = &[a.hostname.as_str(), source];
513                AuthorityGauges {
514                    block_fault_provable: metrics
515                        .faulty_blocks_provable_by_authority
516                        .with_label_values(labels),
517                    block_fault_unprovable: metrics
518                        .faulty_blocks_unprovable_by_peer
519                        .with_label_values(labels),
520                    missing_proposals: metrics
521                        .missing_proposals_by_authority
522                        .with_label_values(labels),
523                    equivocations: metrics.equivocations_by_authority.with_label_values(labels),
524                    invalid_bundle_parts: metrics
525                        .invalid_bundle_parts_by_peer
526                        .with_label_values(labels),
527                }
528            })
529            .collect();
530        Self {
531            authorities: (0..committee.size())
532                .map(|_| Mutex::new(MisbehaviorCountsV2::default()))
533                .collect(),
534            gauges,
535        }
536    }
537
538    fn record_block_fault_provable(&self, idx: usize) {
539        let mut c = self.authorities[idx].lock().unwrap();
540        c.faulty_blocks_provable += 1;
541        self.gauges[idx].block_fault_provable.inc();
542    }
543
544    fn record_block_fault_unprovable(&self, idx: usize) {
545        let mut c = self.authorities[idx].lock().unwrap();
546        c.faulty_blocks_unprovable += 1;
547        self.gauges[idx].block_fault_unprovable.inc();
548    }
549
550    fn record_bundle_part_fault(&self, idx: usize) {
551        let mut c = self.authorities[idx].lock().unwrap();
552        c.invalid_bundle_parts += 1;
553        self.gauges[idx].invalid_bundle_parts.inc();
554    }
555
556    /// Atomically take all block-fault counters AND zero the matching gauges.
557    /// The caller transfers the returned `(provable, unprovable,
558    /// bundle_parts)` into the persisted bucket via `add_block_faults`.
559    fn drain_block_faults(&self, idx: usize) -> (u64, u64, u64) {
560        let mut c = self.authorities[idx].lock().unwrap();
561        let prov = std::mem::take(&mut c.faulty_blocks_provable);
562        let unprov = std::mem::take(&mut c.faulty_blocks_unprovable);
563        let bundle = std::mem::take(&mut c.invalid_bundle_parts);
564        self.gauges[idx].block_fault_provable.set(0);
565        self.gauges[idx].block_fault_unprovable.set(0);
566        self.gauges[idx].invalid_bundle_parts.set(0);
567        (prov, unprov, bundle)
568    }
569
570    /// Receive a `drain_block_faults` payload — additive on counters and
571    /// gauges.
572    fn add_block_faults(&self, idx: usize, prov: u64, unprov: u64, bundle: u64) {
573        let mut c = self.authorities[idx].lock().unwrap();
574        c.faulty_blocks_provable += prov;
575        c.faulty_blocks_unprovable += unprov;
576        c.invalid_bundle_parts += bundle;
577        self.gauges[idx].block_fault_provable.add(prov as i64);
578        self.gauges[idx].block_fault_unprovable.add(unprov as i64);
579        self.gauges[idx].invalid_bundle_parts.add(bundle as i64);
580    }

582    /// Overwrite all block-fault counters and gauges. Used to restore from
583    /// storage on startup.
584    fn set_block_faults(&self, idx: usize, prov: u64, unprov: u64, bundle: u64) {
585        let mut c = self.authorities[idx].lock().unwrap();
586        c.faulty_blocks_provable = prov;
587        c.faulty_blocks_unprovable = unprov;
588        c.invalid_bundle_parts = bundle;
589        self.gauges[idx].block_fault_provable.set(prov as i64);
590        self.gauges[idx].block_fault_unprovable.set(unprov as i64);
591        self.gauges[idx].invalid_bundle_parts.set(bundle as i64);
592    }

594    /// Overwrite the DAG-observed faults (`missing_proposals` +
595    /// `equivocations`) and their gauges. Used on `in_memory` where the
596    /// window is recomputed wholesale on every flush.
597    fn set_dag_faults(&self, idx: usize, missing: u64, equiv: u64) {
598        let mut c = self.authorities[idx].lock().unwrap();
599        c.missing_proposals = missing;
600        c.equivocations = equiv;
601        self.gauges[idx].missing_proposals.set(missing as i64);
602        self.gauges[idx].equivocations.set(equiv as i64);
603    }
604
605    /// Accumulate DAG-observed faults — additive on counters and gauges.
606    /// Used on `persisted` to fold in newly-evicted rounds.
607    fn add_dag_faults(&self, idx: usize, missing: u64, equiv: u64) {
608        let mut c = self.authorities[idx].lock().unwrap();
609        c.missing_proposals += missing;
610        c.equivocations += equiv;
611        self.gauges[idx].missing_proposals.add(missing as i64);
612        self.gauges[idx].equivocations.add(equiv as i64);
613    }

615    /// Stable read clone of one authority's counters.
616    fn snapshot(&self, idx: usize) -> MisbehaviorCountsV2 {
617        self.authorities[idx].lock().unwrap().clone()
618    }

620    fn reset(&self) {
621        for (m, g) in self.authorities.iter().zip(self.gauges.iter()) {
622            let mut c = m.lock().unwrap();
623            *c = MisbehaviorCountsV2::default();
624            g.block_fault_provable.set(0);
625            g.block_fault_unprovable.set(0);
626            g.missing_proposals.set(0);
627            g.equivocations.set(0);
628            g.invalid_bundle_parts.set(0);
629        }
630    }

641/// Given one authority's cached block rounds in [start, end] and the rounds
642/// flagged as equivocating at ingest, returns (equivocations,
643/// missing_proposals). An equivocation is a round with more than one signed
644/// header observed: more than one cached header, or a flag recorded when a
645/// second header for the slot was dropped. A flagged round with no cached
646/// header still counts as missing — none of its headers was accepted.
647fn calculate_misbehavior_counts_for_range(
648    mut block_rounds: Vec<Round>,
649    flagged_rounds: &BTreeSet<Round>,
650    start: Round,
651    end: Round,
652) -> (u64, u64) {
653    if start > end {
654        return (0, 0);
655    }
656    block_rounds.retain(|&round| round >= start && round <= end);
657    block_rounds.sort();
658    let mut equivocating_rounds: BTreeSet<Round> = block_rounds
659        .windows(2)
660        .filter_map(|pair| (pair[0] == pair[1]).then_some(pair[0]))
661        .collect();
662    equivocating_rounds.extend(flagged_rounds.range(start..=end).copied());
663    block_rounds.dedup();
664    let unique_block_rounds = block_rounds.len();
665    let number_of_missing_blocks =
666        (end + 1).saturating_sub(start + unique_block_rounds as u32) as u64;
667    (equivocating_rounds.len() as u64, number_of_missing_blocks)
668}
