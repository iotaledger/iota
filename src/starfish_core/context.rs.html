26/// Context contains per-epoch configuration and metrics shared by all
27/// components of this authority.
28#[derive(Clone)]
29pub(crate) struct Context {
30    /// Timestamp of the start of the current epoch.
31    pub epoch_start_timestamp_ms: u64,
32    /// Index of this authority in the committee.
33    pub own_index: AuthorityIndex,
34    /// Committee of the current epoch.
35    pub committee: Committee,
36    /// Parameters of this authority.
37    pub parameters: Parameters,
38    /// Protocol configuration of current epoch.
39    pub protocol_config: ProtocolConfig,
40    /// Metrics of this authority.
41    pub metrics: Arc<Metrics>,
42    /// Access to local clock
43    pub clock: Arc<Clock>,
44    /// Tracks per-peer responsiveness and ranks candidates for synchronizer
45    /// peer selection. Shared per epoch.
46    pub peer_responsiveness: Arc<PeerResponsiveness>,
47    /// Commitment over an empty transaction list for this committee.
48    pub(crate) empty_transactions_commitment: TransactionsCommitment,
49}
50
51impl Context {
52    pub(crate) fn new(
53        epoch_start_timestamp_ms: u64,
54        own_index: AuthorityIndex,
55        committee: Committee,
56        parameters: Parameters,
57        protocol_config: ProtocolConfig,
58        metrics: Arc<Metrics>,
59        clock: Arc<Clock>,
60    ) -> Self {
61        let peer_responsiveness = PeerResponsiveness::new(&committee, metrics.clone());
62        let empty_transactions_commitment =
63            TransactionsCommitment::compute_empty_transactions_commitment(&committee);
64        Self {
65            epoch_start_timestamp_ms,
66            own_index,
67            committee,
68            parameters,
69            protocol_config,
70            metrics,
71            clock,
72            peer_responsiveness,
73            empty_transactions_commitment,
74        }
75    }
76
77    pub(crate) fn authority_hostname(&self, authority: AuthorityIndex) -> &str {
78        self.committee.authority(authority).hostname.as_str()
79    }
80
81    pub(crate) fn empty_transactions_for_ref(
82        &self,
83        transaction_ref: GenericTransactionRef,
84    ) -> Option<CommitmentVerifiedTransactions> {
85        let GenericTransactionRef::TransactionRef(transaction_ref) = transaction_ref else {
86            return None;
87        };
88        (transaction_ref.transactions_commitment == self.empty_transactions_commitment)
89            .then(|| CommitmentVerifiedTransactions::new_empty_from_ref(transaction_ref, None))
90    }
91
92    /// Lower bound (inclusive) on the round of an acknowledgment or non-own
93    /// ancestor of a block at `block_round`.
94    pub(crate) fn min_ref_round(&self, block_round: Round) -> Round {
95        block_round.saturating_sub(self.protocol_config.gc_depth())
96    }

98    /// Create a test context with a committee of given size and even stake
99    #[cfg(test)]
100    pub(crate) fn new_for_test(
101        committee_size: usize,
102    ) -> (Self, Vec<(NetworkKeyPair, ProtocolKeyPair)>) {
103        let (committee, keypairs) =
104            starfish_config::local_committee_and_keys(0, vec![1; committee_size]);
105        let metrics = test_metrics();
106        let temp_dir = TempDir::new().unwrap();
107        let clock = Arc::new(Clock::default());
108
109        let context = Context::new(
110            0,
111            AuthorityIndex::new_for_test(0),
112            committee,
113            Parameters {
114                db_path: temp_dir.keep(),
115                commit_recovery_batch_size: 3,
116                // Keep peer selection deterministic (no responsiveness ranking)
117                // for the many tests that rely on stable peer order; tests that
118                // exercise ranking opt in explicitly.
119                enable_peer_responsiveness_ranking: false,
120                // Tests enable bounded peer selection explicitly when needed.
121                enable_bounded_header_advertisement: false,
122                ..Default::default()
123            },
124            ProtocolConfig::get_for_max_version_UNSAFE(),
125            metrics,
126            clock,
127        );
128        (context, keypairs)
129    }

158/// A clock that allows to derive the current UNIX system timestamp while
159/// guaranteeing that timestamp will be monotonically incremented, tolerating
160/// ntp and system clock changes and corrections. Explicitly avoid to make
161/// `[Clock]` cloneable to ensure that a single instance is shared behind an
162/// `[Arc]` wherever is needed in order to make sure that consecutive calls to
163/// receive the system timestamp will remain monotonically increasing.
164pub struct Clock {
165    initial_instant: Instant,
166    initial_system_time: SystemTime,
167    // `clock_drift` should be used only for testing
168    #[cfg(any(test, msim))]
169    clock_drift: BlockTimestampMs,
170}
171
172impl Default for Clock {
173    fn default() -> Self {
174        Self {
175            initial_instant: Instant::now(),
176            initial_system_time: SystemTime::now(),
177            #[cfg(any(test, msim))]
178            clock_drift: 0,
179        }
180    }
181}
182
183impl Clock {
184    #[cfg(any(test, msim))]
185    pub fn new_for_test(clock_drift: BlockTimestampMs) -> Self {
186        Self {
187            initial_instant: Instant::now(),
188            initial_system_time: SystemTime::now(),
189            clock_drift,
190        }
191    }
192
193    // Returns the current time expressed as a UNIX timestamp in milliseconds.
194    // Calculated with Tokio Instant to ensure monotonicity
195    // and to allow testing with a tokio clock.
196    pub(crate) fn timestamp_utc_ms(&self) -> BlockTimestampMs {
197        let now: Instant = Instant::now();
198        let monotonic_system_time = self
199            .initial_system_time
200            .checked_add(
201                now.checked_duration_since(self.initial_instant)
202                    .unwrap_or_else(|| {
203                        panic!(
204                            "current instant ({:?}) < initial instant ({:?})",
205                            now, self.initial_instant
206                        )
207                    }),
208            )
209            .expect("Computing system time should not overflow");
210        let timestamp = monotonic_system_time
211            .duration_since(SystemTime::UNIX_EPOCH)
212            .unwrap_or_else(|_| {
213                panic!(
214                    "system time ({:?}) < UNIX_EPOCH ({:?})",
215                    monotonic_system_time,
216                    SystemTime::UNIX_EPOCH,
217                )
218            })
219            .as_millis() as BlockTimestampMs;
220        // Apply clock drift only in test/msim environments to simulate clock skew
221        // between nodes. In production builds, the clock_drift field doesn't exist,
222        // and this returns the timestamp directly.
223        #[cfg(any(test, msim))]
224        {
225            timestamp + self.clock_drift
226        }
227        #[cfg(not(any(test, msim)))]
228        {
229            timestamp
230        }
231    }
