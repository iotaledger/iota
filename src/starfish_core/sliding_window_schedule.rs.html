4//! Sliding-window reputation scoring for the Starfish leader schedule.
5//!
6//! Maintains a running aggregate of per-commit scoring contributions (computed
7//! by [`crate::leader_scoring::compute_per_commit_contribution`]) over the most
8//! recent `window_size` scored commits. The aggregate updates on every commit
9//! (O(1) amortized over committee size): each commit's contribution is added,
10//! and once the window is full the oldest is evicted and subtracted.
11//!
12//! [`SlidingWindowSchedule::reputation_scores`] exposes the aggregate as
13//! [`ReputationScores`] — the same type the V2 `LeaderSwapTable` is built from.
14//! The sliding window is an alternative *score source* for the existing swap
15//! table; it does not select leaders itself.
16//!
17//! # Usage
18//!
19//! When the sliding-window leader schedule is enabled (protocol flag
20//! `consensus_enable_sliding_window_leader_schedule`), the leader schedule
21//! feeds each committed subdag to [`SlidingWindowSchedule::add_commit`] in
22//! consecutive index order and rebuilds the `LeaderSwapTable` from
23//! [`SlidingWindowSchedule::reputation_scores`] every `commits_per_schedule`
24//! commits.
25//!
26//! On a restart or fast-sync the in-memory window starts empty and is rebuilt
27//! from storage: [`SlidingWindowSchedule::replay_start`] gives the earliest
28//! commit index to replay, and the committed subdags from that index up to the
29//! last persisted commit are fed back through
30//! [`SlidingWindowSchedule::add_commit`] to repopulate the window before any
31//! scores are read.
32
33use std::{collections::VecDeque, sync::Arc};
34
35use crate::{
36    commit::{CommitIndex, CommitRange, SubDagBase},
37    context::Context,
38    leader_scoring::{ReputationScores, compute_per_commit_contribution},
39};
40
41/// Number of pending committed subdags retained for scoring lookback.
42/// `compute_per_commit_contribution` scores `c_minus_3` using `c_minus_2..=c`
43/// as the lookback window.
44const MAX_PENDING_COMMITS: usize = 3;
45
46/// Sliding-window reputation scorer.
47pub(crate) struct SlidingWindowSchedule {
48    context: Arc<Context>,
49    /// Running window length (number of scored commits aggregated).
50    window_size: u32,
51    /// Running aggregate, indexed by `AuthorityIndex`.
52    total_scores_per_authority: Vec<u64>,
53    /// Last [`MAX_PENDING_COMMITS`] committed subdags, retained for scoring
54    /// lookback.
55    pending_commits: VecDeque<SubDagBase>,
56    /// Per-commit score contributions currently in the running window, each
57    /// tagged with the index of the commit it scored. Retained so eviction can
58    /// subtract them from the aggregate and so the window's commit range can be
59    /// reported alongside the scores.
60    scores_entries: VecDeque<(CommitIndex, Vec<u64>)>,
61}
62
63impl SlidingWindowSchedule {
64    /// Creates a fresh scorer. `window_size` must be at least 1.
65    pub(crate) fn new(context: Arc<Context>, window_size: u32) -> Self {
66        assert!(
67            window_size >= 1,
68            "window_size ({window_size}) must be at least 1"
69        );
70        let committee_size = context.committee.size();
71        Self {
72            context,
73            window_size,
74            total_scores_per_authority: vec![0u64; committee_size],
75            pending_commits: VecDeque::with_capacity(MAX_PENDING_COMMITS),
76            scores_entries: VecDeque::with_capacity(window_size as usize),
77        }
78    }
79
80    /// Earliest commit index a caller must replay through [`Self::add_commit`]
81    /// to rebuild the window state as of `last_commit_index`.
82    ///
83    /// Replays a full window: the window slides by adding new and subtracting
84    /// evicted per-commit contributions, so recovery must rebuild those
85    /// per-commit entries — the aggregate sum alone cannot evict. The extra
86    /// `MAX_PENDING_COMMITS` starts the replay three commits before the window
87    /// so the lookback buffer (`pending_commits`) is refilled and the first
88    /// in-window commit scores identically to live operation. The swap table in
89    /// effect at restart is not recovered by this replay.
90    pub(crate) fn replay_start(last_commit_index: CommitIndex, window_size: u32) -> CommitIndex {
91        last_commit_index
92            .saturating_sub(window_size + MAX_PENDING_COMMITS as u32)
93            .max(1)
94    }
95
96    /// Ingest a new committed subdag, updating the running aggregate. Commits
97    /// must be supplied in consecutive index order.
98    pub(crate) fn add_commit(&mut self, c: SubDagBase) {
99        let _s = self
100            .context
101            .metrics
102            .node_metrics
103            .scope_processing_time
104            .with_label_values(&["SlidingWindowSchedule::add_commit"])
105            .start_timer();
106
107        if let Some(last) = self.pending_commits.back() {
108            assert_eq!(c.commit_ref.index, last.commit_ref.index + 1);
109        }
110
111        if self.pending_commits.len() < MAX_PENDING_COMMITS {
112            self.pending_commits.push_back(c);
113            return;
114        }
115
116        // Steady state: the window now holds four consecutive commits. We score
117        // the OLDEST of the four, c_minus_3: its leader is at round r, and the
118        // r+1 votes and r+2 certifiers that score it live in the three newer
119        // commits. So a commit is scored only once the third commit after it is
120        // committed; the just-fed c merely unlocks scoring c_minus_3.
121        let c_minus_3 = &self.pending_commits[0];
122        let c_minus_2 = &self.pending_commits[1];
123        let c_minus_1 = &self.pending_commits[2];
124        let scored_index = c_minus_3.commit_ref.index;
125
126        let contribution =
127            compute_per_commit_contribution(&self.context, c_minus_3, c_minus_2, c_minus_1, &c);
128
129        // Evict if the window is full. saturating_sub is defensive — under normal
130        // operation contributions are added and later subtracted symmetrically.
131        while self.scores_entries.len() >= self.window_size as usize {
132            let (_, evicted) = self
133                .scores_entries
134                .pop_front()
135                .expect("scores_entries is non-empty when len >= window_size >= 1");
136            for (i, &delta) in evicted.iter().enumerate() {
137                self.total_scores_per_authority[i] =
138                    self.total_scores_per_authority[i].saturating_sub(delta);
139            }
140        }
141
142        // Add the new contribution.
143        for (i, &delta) in contribution.iter().enumerate() {
144            self.total_scores_per_authority[i] =
145                self.total_scores_per_authority[i].saturating_add(delta);
146        }
147        self.scores_entries.push_back((scored_index, contribution));
148
149        // Rotate pending commits: drop the now-scored c_minus_3, append c.
150        self.pending_commits.pop_front();
151        self.pending_commits.push_back(c);
152    }
153
154    /// Current running aggregate as [`ReputationScores`], tagged with the
155    /// commit range of the scored commits in the window. The range ends at
156    /// the last *scored* commit, which trails the last fed commit by
157    /// `MAX_PENDING_COMMITS` (a commit is scored only once the third commit
158    /// after it arrives). Returns all-zero scores over the empty range
159    /// until the first commit is scored.
160    pub(crate) fn reputation_scores(&self) -> ReputationScores {
161        let commit_range = match (self.scores_entries.front(), self.scores_entries.back()) {
162            (Some((first, _)), Some((last, _))) => CommitRange::new(*first..=*last),
163            _ => CommitRange::default(),
164        };
165        ReputationScores::new(commit_range, self.total_scores_per_authority.clone())
166    }
167}

176    fn build_full_commits(context: Arc<Context>, n: u32) -> Vec<SubDagBase> {
177        let mut dag_builder = DagBuilder::new(context);
178        dag_builder.layers(1..=n).build();
179        dag_builder
180            .get_sub_dag_and_commits(1..=n)
181            .into_iter()
182            .map(|(sub_dag, _)| sub_dag.base)
183            .collect()
184    }

186    #[tokio::test]
187    async fn test_warmup_no_scoring_before_three_commits() {
188        let context = Arc::new(Context::new_for_test(4).0);
189        let mut schedule = SlidingWindowSchedule::new(context.clone(), 10);
190        let subdags = build_full_commits(context, 2);
191        for s in &subdags {
192            schedule.add_commit(s.clone());
193        }
194        assert_eq!(schedule.total_scores_per_authority, vec![0u64; 4]);
195        assert_eq!(schedule.scores_entries.len(), 0);
196    }

198    #[rstest]
199    #[tokio::test]
200    async fn test_eviction_subtracts_from_aggregate(#[values(false, true)] starfish_speed: bool) {
201        // Small window so we can saturate it within a small DAG.
202        let window_size: u32 = 2;
203        let mut context = Context::new_for_test(4).0;
204        context
205            .protocol_config
206            .set_consensus_starfish_speed_for_testing(starfish_speed);
207        let context = Arc::new(context);
208        let mut schedule = SlidingWindowSchedule::new(context.clone(), window_size);
209        // 6 commits → 3 scored (commits 1..=3 scored as c_minus_3 in iterations 4..=6).
210        let subdags = build_full_commits(context.clone(), 6);
211        for s in &subdags {
212            schedule.add_commit(s.clone());
213        }
214        // Window holds at most 2 entries; 3 were added so 1 was evicted.
215        assert_eq!(schedule.scores_entries.len(), window_size as usize);
216
217        // In a fully-connected DAG each scored commit contributes
218        // committee.total_stake() to every authority — doubled under
219        // StarfishSpeed, where every vote is a strong vote. The aggregate
220        // equals `window_size * per_commit` because earlier contributions were
221        // subtracted on eviction.
222        let vote_weight: u64 = if starfish_speed { 2 } else { 1 };
223        let per_commit = context.committee.total_stake() * vote_weight;
224        let expected = per_commit * (window_size as u64);
225        assert_eq!(
226            schedule.total_scores_per_authority,
227            vec![expected; context.committee.size()]
228        );
229    }

231    #[rstest]
232    #[tokio::test]
233    async fn test_reputation_scores_reports_window_aggregate_and_range(
234        #[values(false, true)] starfish_speed: bool,
235    ) {
236        let mut context = Context::new_for_test(4).0;
237        context
238            .protocol_config
239            .set_consensus_starfish_speed_for_testing(starfish_speed);
240        let context = Arc::new(context);
241        let mut schedule = SlidingWindowSchedule::new(context.clone(), 10);
242
243        // Before anything is scored: all-zero scores over the empty range.
244        let empty = schedule.reputation_scores();
245        assert_eq!(
246            empty.scores_per_authority,
247            vec![0u64; context.committee.size()]
248        );
249        assert_eq!(empty.commit_range, CommitRange::default());
250
251        // 6 commits → commits 1..=3 scored (as c_minus_3 in iterations 4..=6).
252        let subdags = build_full_commits(context.clone(), 6);
253        for s in &subdags {
254            schedule.add_commit(s.clone());
255        }
256        let scores = schedule.reputation_scores();
257        assert_eq!(scores.commit_range, (1..=3).into());
258        // Under StarfishSpeed every vote is a strong vote, which scores double.
259        let vote_weight: u64 = if starfish_speed { 2 } else { 1 };
260        let per_commit = context.committee.total_stake() * vote_weight;
261        assert_eq!(
262            scores.scores_per_authority,
263            vec![per_commit * 3; context.committee.size()]
264        );
265    }

268    fn test_replay_start_basic() {
269        // window=10, last_commit=23 → replay_start = max(1, 23 - (10 + 3)) = 10.
270        assert_eq!(SlidingWindowSchedule::replay_start(23, 10), 10);
271        // Small last_commit_index → clamps to 1.
272        assert_eq!(SlidingWindowSchedule::replay_start(2, 10), 1);
