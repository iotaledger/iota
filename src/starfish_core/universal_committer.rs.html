26/// The first round of each wave has one leader by default. We leave this as a
27/// parameter of UniversalCommitter as we might change this in future
28pub(crate) const DEFAULT_NUMBER_OF_LEADERS_IN_ROUND: usize = 1;
29
30/// A universal committer uses a collection of committers to commit a sequence
31/// of leaders. It can be configured to use a combination of different commit
32/// strategies, including multi-leaders, backup leaders, and pipelines.
33pub(crate) struct UniversalCommitter {
34    /// The per-epoch configuration of this authority.
35    context: Arc<Context>,
36    /// In memory block store representing the dag state
37    dag_state: Arc<RwLock<DagState>>,
38    /// The list of committers for multi-leader or pipelining
39    committers: Vec<BaseCommitter>,
40}
41
42impl UniversalCommitter {
43    /// Try to decide part of the dag. This function is idempotent and returns
44    /// an ordered list of decided leaders.
45    #[tracing::instrument(skip_all, fields(last_finalized = %last_finalized))]
46    pub(crate) fn try_decide(&self, last_finalized: Slot) -> Vec<DecidedLeader> {
47        let highest_accepted_round = self.dag_state.read().highest_accepted_round();
48
49        // Try to decide as many leaders as possible, starting with the highest round.
50        let mut leaders: VecDeque<(LeaderStatus, Decision)> = VecDeque::new();
51
52        let last_round = last_finalized.round + 1;
53
54        // try to commit a leader up to the highest_accepted_round - 2. There is no
55        // reason to try and iterate on higher rounds as in order to make a direct
56        // decision for a leader at round R we need blocks from round R+2 to figure
57        // out that enough certificates and support exist to commit a leader.
58        'outer: for round in (last_round..=highest_accepted_round.saturating_sub(2)).rev() {
59            for committer in self.committers.iter().rev() {
60                // Skip committers that don't have a leader for this round.
61                let Some(slot) = committer.elect_leader(round) else {
62                    tracing::debug!("No leader for round {round}, skipping");
63                    continue;
64                };
65
66                // now that we reached the last finalized leader we can stop the commit rule
67                if slot == last_finalized {
68                    tracing::debug!("Reached last finalized {slot}, now exit");
69                    break 'outer;
70                }
71
72                tracing::trace!("Trying to decide {slot} with {committer}",);
73
74                let mut status = committer.try_direct_decide(slot);
75                let mut decision = Decision::Direct;
76                tracing::debug!("Outcome of direct rule: {status} with {committer}");
77
78                // If the direct result is not resolved (Commit(Pending) or
79                // Undecided), run the indirect rule. For Pending, a
80                // committed anchor's path can upgrade the metastate; for
81                // Undecided, indirect may resolve the slot entirely.
82                if !status.is_resolved() {
83                    let indirect = committer
84                        .try_indirect_decide(status.clone(), leaders.iter().map(|(x, _)| x));
85                    if indirect != status {
86                        tracing::debug!("Outcome of indirect rule: {indirect} with {committer}");
87                        decision = match (&status, &indirect) {
88                            (
89                                LeaderStatus::Commit(_, Some(CommitMetastate::Pending), _),
90                                LeaderStatus::Commit(_, Some(m), _),
91                            ) if *m != CommitMetastate::Pending => Decision::Upgraded,
92                            _ => Decision::Indirect,
93                        };
94                        status = indirect;
95                    }
96                }
97                leaders.push_front((status, decision));
98            }
99        }
100
101        // The decided sequence is the longest prefix where every leader is
102        // resolved. Commit(Pending) is decided but not resolved — its metastate
103        // is still pending — so sequencing stops there until a later commit pass
104        // upgrades it.
105        let mut decided_leaders = Vec::new();
106        for (leader, decision) in leaders {
107            if leader.round() == GENESIS_ROUND {
108                continue;
109            }
110            if !leader.is_resolved() {
111                break;
112            }
113            let decided_leader = leader
114                .into_decided_leader()
115                .expect("is_resolved implies a DecidedLeader");
116            Self::update_metrics(&self.context, &decided_leader, decision);
117            decided_leaders.push(decided_leader);
118        }
119        if !decided_leaders.is_empty() {
120            tracing::debug!("Decided {decided_leaders:?}");
121        }
122        decided_leaders
123    }
124
125    /// Return list of leaders for the round.
126    /// Can return empty vec if round does not have a designated leader.
127    pub(crate) fn get_leaders(&self, round: Round) -> Vec<AuthorityIndex> {
128        self.committers
129            .iter()
130            .filter_map(|committer| committer.elect_leader(round))
131            .map(|l| l.authority)
132            .collect()
133    }
134
135    /// Update metrics.
136    pub(crate) fn update_metrics(
137        context: &Context,
138        decided_leader: &DecidedLeader,
139        decision: Decision,
140    ) {
141        let decision_str = match decision {
142            Decision::Direct => "direct",
143            Decision::Upgraded => "upgraded",
144            Decision::Indirect => "indirect",
145        };
146        let status = match decided_leader {
147            DecidedLeader::Commit(_, None, _) => format!("{decision_str}-commit"),
148            DecidedLeader::Commit(_, Some(metastate), _) => {
149                format!("{decision_str}-commit-{metastate}")
150            }
151            DecidedLeader::Skip(..) => format!("{decision_str}-skip"),
152        };
153        let leader_host = &context
154            .committee
155            .authority(decided_leader.slot().authority)
156            .hostname;
157        context
158            .metrics
159            .node_metrics
160            .decided_leaders_total
161            .with_label_values(&[leader_host, &status])
162            .inc();
163    }
164}

166/// A builder for a universal committer. By default, the builder creates a
167/// single base committer, that is, a single leader and no pipeline.
168pub(crate) mod universal_committer_builder {
169    use super::*;
170    use crate::{
171        base_committer::BaseCommitterOptions, commit::WAVE_LENGTH, leader_schedule::LeaderSchedule,
172    };
173
174    pub(crate) struct UniversalCommitterBuilder {
175        context: Arc<Context>,
176        leader_schedule: Arc<LeaderSchedule>,
177        dag_state: Arc<RwLock<DagState>>,
178        number_of_leaders: usize,
179    }
180
181    impl UniversalCommitterBuilder {
182        // We use hardcoded parameters: the wave length is 3 and we use one leader in
183        // each round. Support for multi-leader option might be enabled at a
184        // later stage.
185        pub(crate) fn new(
186            context: Arc<Context>,
187            leader_schedule: Arc<LeaderSchedule>,
188            dag_state: Arc<RwLock<DagState>>,
189        ) -> Self {
190            Self {
191                context,
192                leader_schedule,
193                dag_state,
194                number_of_leaders: DEFAULT_NUMBER_OF_LEADERS_IN_ROUND,
195            }
196        }
197
198        #[expect(unused)]
199        pub(crate) fn with_number_of_leaders(mut self, number_of_leaders: usize) -> Self {
200            self.number_of_leaders = number_of_leaders;
201            self
202        }
203
204        pub(crate) fn build(self) -> UniversalCommitter {
205            let mut committers = Vec::new();
206            // We use pipelined scheduler by default, thereby there should be 3 intersecting
207            // waves corresponding to each committer
208            let pipeline_stages = WAVE_LENGTH;
209            for round_offset in 0..pipeline_stages {
210                for leader_offset in 0..self.number_of_leaders {
211                    let options = BaseCommitterOptions {
212                        round_offset,
213                        leader_offset: leader_offset as Round,
214                    };
215                    let committer = BaseCommitter::new(
216                        self.context.clone(),
217                        self.leader_schedule.clone(),
218                        self.dag_state.clone(),
219                        options,
220                    );
221                    committers.push(committer);
222                }
223            }
224
225            UniversalCommitter {
226                context: self.context,
227                dag_state: self.dag_state,
228                committers,
229            }
230        }
231    }
