4//! Per-peer responsiveness tracking and ranking for synchronizer peer
5//! selection.
6//!
7//! The transactions synchronizer, the commit syncer and the header
8//! synchronizer pick peers to fetch from. On a node with slow or asymmetric
9//! inbound links a uniform choice regularly draws slow-but-not-failing peers,
10//! adding avoidable latency to payload, commit and header retrieval. Each fetch
11//! source in
12//! [`DataSource::RESPONSIVENESS_SOURCES`] is tracked separately, so a peer's
13//! record on one kind of fetch does not decide its rank on another. Until a
14//! source has a sample of its own for a peer, it places that peer by what the
15//! nearest source in [`DataSource::responsiveness_fallbacks`] that has measured
16//! it saw.
17//!
18//! [`PeerResponsiveness`] tracks a per-peer smoothed "effective latency" fed by
19//! those existing latency/outcome signals, and exposes
20//! [`PeerResponsiveness::prioritize`] which reorders the candidate set to
21//! prefer responsive peers. It is a preference, never a change of membership:
22//! the output is always a permutation of the input, and every fetched response
23//! is still fully verified, so a peer that merely appears fast is never trusted
24//! to deliver. A peer that fails or stalls is demoted by a timeout-scale
25//! penalty and moved behind every healthy candidate rather than excluded, so
26//! liveness is preserved and it recovers with its next success.
27//!
28//! Selection sorts healthy candidates by effective latency, permutes the
29//! window of the best peers by draws with probability proportional to the
30//! inverse latency — the selection shape of the checkpoint state-sync's peer
31//! balancer, with a weighted draw inside the window — and keeps the remaining
32//! healthy candidates in latency order. Peers whose most recent fetch failed
33//! always come last, outside the randomization. So fast peers lead in
34//! proportion to how fast they are, the window keeps a large slow tail from
35//! diluting the weights, and a just-failed peer cannot displace a healthy
36//! one. An exploration fraction of selections ignores
37//! ranking (including the failure state) and shuffles uniformly, so every peer
38//! keeps being sampled and a failed peer re-enters the ranked draw as soon as
39//! a fetch succeeds again.
40//!
41//! The block-bundle streaming path is tracked separately, per (peer, author)
42//! pair; see [`PeerResponsiveness::record_streaming_header_deliveries`].
43

52impl DataSource {
53    /// Fetch sources ranked by peer responsiveness. Sources not listed here are
54    /// not ranked, and every [`PeerResponsiveness`] call for them is a no-op.
55    pub(crate) const RESPONSIVENESS_SOURCES: [DataSource; 4] = [
56        DataSource::TransactionSynchronizer,
57        DataSource::CommitSyncer,
58        DataSource::FastCommitSyncer,
59        DataSource::HeaderSynchronizerRequested,
60    ];
61
62    /// Sources whose measurements order the peers this source has no sample
63    /// for yet, tried in this order until one of them has measured the peer.
64    ///
65    /// A commit syncer reads the other flavor first (same fetches, same
66    /// scale), then the header synchronizer (same endpoint, and seeded for
67    /// every peer by the startup probe), then the transactions synchronizer
68    /// (always populated, but its fetches are the lightest). The transactions
69    /// and header synchronizers read each other, the closest scale either has.
70    fn responsiveness_fallbacks(self) -> &'static [DataSource] {
71        match self {
72            DataSource::CommitSyncer => &[
73                DataSource::FastCommitSyncer,
74                DataSource::HeaderSynchronizerRequested,
75                DataSource::TransactionSynchronizer,
76            ],
77            DataSource::FastCommitSyncer => &[
78                DataSource::CommitSyncer,
79                DataSource::HeaderSynchronizerRequested,
80                DataSource::TransactionSynchronizer,
81            ],
82            DataSource::TransactionSynchronizer => &[DataSource::HeaderSynchronizerRequested],
83            DataSource::HeaderSynchronizerRequested => &[DataSource::TransactionSynchronizer],
84            _ => &[],
85        }
86    }
87
88    /// Latency assumed for a peer that neither this source nor any of its
89    /// fallbacks has measured, on the scale of this source's own fetches so
90    /// that an untried peer is not placed ahead of the peers it has measured.
91    fn neutral_latency_ms(self) -> f64 {
92        match self {
93            DataSource::CommitSyncer => COMMIT_SYNC_NEUTRAL_LATENCY_MS,
94            DataSource::FastCommitSyncer => FAST_COMMIT_SYNC_NEUTRAL_LATENCY_MS,
95            DataSource::HeaderSynchronizerRequested => HEADER_SYNC_NEUTRAL_LATENCY_MS,
96            _ => TRANSACTIONS_SYNC_NEUTRAL_LATENCY_MS,
97        }
98    }
99}

101/// Probability that a `prioritize` call ignores ranking and returns a uniform
102/// shuffle, applied uniformly to every fetch kind. Guarantees every eligible
103/// peer keeps a floor probability of being tried early regardless of its rank,
104/// which bounds monopolization, prevents starvation of the latency tail, and
105/// keeps every peer's measurement fresh.
106const EXPLORE_PROBABILITY: f64 = 0.05;
107
108/// The weighted draw permutes only this many of the best-ranked healthy peers;
109/// peers beyond the window follow in plain latency order. Bounding the draw to
110/// a small window keeps the inverse-latency weights meaningful (a large slow
111/// tail cannot dilute them); the sorted-then-sample-from-a-window shape
112/// follows the checkpoint state-sync's `PEER_BALANCER_SELECTION_WINDOW`
113/// selection, which is proven in production.
114const SELECTION_WINDOW: usize = 5;
115
116/// Floor applied to every latency sample (ms). Keeps effective latency strictly
117/// positive so a selection weight can never divide-by-zero or let a near-zero
118/// sample produce an unbounded weight.
119const MIN_LATENCY_MS: f64 = 1.0;
120
121/// Effective latency (ms) assigned to a peer nothing is known about yet: no
122/// sample for the source being ranked, and none for any of its fallbacks
123/// either. Each source has its own, on the scale its fetches measure, so that
124/// an untried peer is placed among the peers that source has measured rather
125/// than ahead of or behind all of them.
126///
127/// This one is the transactions synchronizer's, whose fetches are small and
128/// frequent.
129const TRANSACTIONS_SYNC_NEUTRAL_LATENCY_MS: f64 = 250.0;
130
131/// Neutral prior for regular commit sync, which fetches the commits, their
132/// headers and their transactions in requests of their own. Taken from what
133/// these fetches measure on testnet and mainnet.
134const COMMIT_SYNC_NEUTRAL_LATENCY_MS: f64 = 4_000.0;
135
136/// Neutral prior for fast commit sync, which serves a range in one request
137/// and so measures quicker than regular sync on the same networks.
138const FAST_COMMIT_SYNC_NEUTRAL_LATENCY_MS: f64 = 2_000.0;
139
140/// Neutral prior for the header synchronizer, rarely reached: the startup
141/// probe seeds this track for every reachable peer before the first header
142/// goes missing.
143const HEADER_SYNC_NEUTRAL_LATENCY_MS: f64 = 500.0;
144
145/// EWMA weight for a successful sample. Small, so the score is "slow to
146/// trust".
147const ALPHA_SUCCESS: f64 = 0.3;
148
149/// First sample seeds the value; later ones are EWMA-blended with weight
150/// `alpha`.
151fn blend_latency_ms(prev: Option<f64>, sample_ms: f64, alpha: f64) -> f64 {
152    match prev {
153        None => sample_ms,
154        Some(prev) => (1.0 - alpha) * prev + alpha * sample_ms,
155    }
156}

158#[derive(Clone, Default)]
159struct PeerStat {
160    /// Smoothed effective latency in milliseconds; `None` until the first
161    /// sample.
162    effective_latency_ms: Option<f64>,
163    /// Whether the most recent fetch from this peer failed; cleared by the
164    /// next success.
165    last_fetch_failed: bool,
166}
167
168/// Per-fetch-source per-peer statistics, keyed by the fetch source. Each vector
169/// is indexed by [`AuthorityIndex`] and sized to the committee, so the slot for
170/// `own_index` simply stays unused. Only the sources in
171/// [`DataSource::RESPONSIVENESS_SOURCES`] are present.
172struct Tracks {
173    per_kind: HashMap<DataSource, Vec<PeerStat>>,
174}
175
176/// Tracks per-peer responsiveness and ranks candidates for synchronizer peer
177/// selection. Shared per epoch.
178pub(crate) struct PeerResponsiveness {
179    metrics: Arc<Metrics>,
180    committee_size: usize,
181    inner: Mutex<Tracks>,
182    /// Smoothed streaming-delivery latency, indexed `[peer][author]`.
183    streaming_headers: Mutex<Vec<Vec<Option<f64>>>>,
184}
185
186impl PeerResponsiveness {
187    pub(crate) fn new(committee: &Committee, metrics: Arc<Metrics>) -> Arc<Self> {
188        let size = committee.size();
189        Arc::new(Self {
190            metrics,
191            committee_size: size,
192            inner: Mutex::new(Tracks {
193                per_kind: DataSource::RESPONSIVENESS_SOURCES
194                    .into_iter()
195                    .map(|source| (source, vec![PeerStat::default(); size]))
196                    .collect(),
197            }),
198            streaming_headers: Mutex::new(vec![vec![None; size]; size]),
199        })
200    }
201
202    /// Records a successful fetch from `peer` for `source` that took `latency`.
203    ///
204    /// Callers must only report latency for fetches that delivered useful data;
205    /// a response that returned nothing (or a small fraction of what was
206    /// requested) should be reported via [`Self::record_failure_with_timeout`]
207    /// (or with a latency scaled up by the shortfall), so a peer cannot look
208    /// fast by replying quickly with little. A shortfall-scaled latency must be
209    /// capped at the operation's failure penalty: the failure is the ceiling of
210    /// the scale, and a peer that delivered something must never record worse
211    /// than one that failed.
212    pub(crate) fn record_success(
213        &self,
214        source: DataSource,
215        peer: AuthorityIndex,
216        latency: Duration,
217    ) {
218        let sample = (latency.as_secs_f64() * 1_000.0).max(MIN_LATENCY_MS);
219        self.update(source, peer, sample, ALPHA_SUCCESS);
220    }
221
222    /// Records a failed or timed-out fetch from `peer` for `source`, demoting
223    /// it to at least the operation's timeout, and never below the neutral
224    /// prior so a failure is never recorded as a fast sample. Until its next
225    /// success the peer is also ordered behind every non-failed candidate
226    /// (see [`Self::prioritize`]).
227    pub(crate) fn record_failure_with_timeout(
228        &self,
229        source: DataSource,
230        peer: AuthorityIndex,
231        timeout: Duration,
232    ) {
233        let sample = (timeout.as_secs_f64() * 1_000.0).max(source.neutral_latency_ms());
234        self.update_failure(source, peer, sample);
235    }
236
237    /// Records one bundle's header deliveries from `peer`, duplicates
238    /// included — sampling only first deliveries would leave a consistently
239    /// second peer unmeasured. One sample per author, from its newest header,
240    /// measured from the header's own timestamp to `now_ms`: the author's
241    /// clock bias cancels across peers and a re-delivery cannot read fast.
242    /// The lock is taken once per bundle.
243    pub(crate) fn record_streaming_header_deliveries(
244        &self,
245        peer: AuthorityIndex,
246        now_ms: BlockTimestampMs,
247        headers: impl IntoIterator<Item = (AuthorityIndex, Round, BlockTimestampMs)>,
248    ) {
249        let mut newest_per_author: Vec<Option<(Round, BlockTimestampMs)>> =
250            vec![None; self.committee_size];
251        for (author, round, timestamp_ms) in headers {
252            let Some(newest) = newest_per_author.get_mut(author.value()) else {
253                continue;
254            };
255            if newest.is_none_or(|(current_round, _)| current_round < round) {
256                *newest = Some((round, timestamp_ms));
257            }
258        }
259        let mut table = self.streaming_headers.lock();
260        let Some(row) = table.get_mut(peer.value()) else {
261            return;
262        };
263        for (author, newest) in newest_per_author.into_iter().enumerate() {
264            let Some((_, timestamp_ms)) = newest else {
265                continue;
266            };
267            let sample = (now_ms.saturating_sub(timestamp_ms) as f64).max(MIN_LATENCY_MS);
268            row[author] = Some(blend_latency_ms(row[author], sample, ALPHA_SUCCESS));
269        }
270    }
271
272    fn update(&self, source: DataSource, peer: AuthorityIndex, sample: f64, alpha: f64) {
273        let snapshot = {
274            let mut tracks = self.inner.lock();
275            let Some(track) = tracks.per_kind.get_mut(&source) else {
276                return;
277            };
278            let Some(stat) = track.get_mut(peer.value()) else {
279                return;
280            };
281            stat.effective_latency_ms =
282                Some(blend_latency_ms(stat.effective_latency_ms, sample, alpha));
283            stat.last_fetch_failed = false;
284            Self::snapshot(track)
285        };
286
287        self.publish_expected_latencies(source, &snapshot);
288    }
289
290    fn update_failure(&self, source: DataSource, peer: AuthorityIndex, sample: f64) {
291        let snapshot = {
292            let mut tracks = self.inner.lock();
293            let Some(track) = tracks.per_kind.get_mut(&source) else {
294                return;
295            };
296            let Some(stat) = track.get_mut(peer.value()) else {
297                return;
298            };
299            let new = match stat.effective_latency_ms {
300                None => sample,
301                Some(prev) => prev.max(sample),
302            };
303            stat.effective_latency_ms = Some(new);
304            stat.last_fetch_failed = true;
305            Self::snapshot(track)
306        };
307
308        self.publish_expected_latencies(source, &snapshot);
309    }

311    /// Reorders `candidates` in place to prefer peers that have been more
312    /// responsive for `source`, keeping the set itself unchanged (the output is
313    /// a permutation of the input: never adds or drops a peer).
314    ///
315    /// Non-failed candidates are sorted by effective latency; the
316    /// [`SELECTION_WINDOW`] best are then permuted by repeated draws with
317    /// probability proportional to `1 / effective_latency`, and the remaining
318    /// non-failed candidates follow in latency order. A peer whose most recent
319    /// fetch failed does not take part in the randomization at all: it is
320    /// ordered behind every non-failed candidate (fastest of the failed first)
321    /// until its next success. A fraction of calls (see
322    /// [`EXPLORE_PROBABILITY`]) ignore ranking and return a uniform shuffle.
323    /// `rng` is consumed fresh on every call.
324    ///
325    /// A candidate with no sample for `source` is placed by its latency from
326    /// the nearest source in [`DataSource::responsiveness_fallbacks`] that has
327    /// measured it, and by [`DataSource::neutral_latency_ms`] only when none of
328    /// them has. A fallback that measures lighter fetches than `source` can put
329    /// an untried peer ahead of one this source has already measured; that is
330    /// the intended trade, since it is the reachable-and-quick signal available
331    /// before this source has any sample of its own. Only the latency is taken
332    /// from a fallback, never the failure state, so a peer that failed on
333    /// another source ranks as healthy here - carrying the timeout-scale
334    /// latency that failure left behind.
335    pub(crate) fn prioritize<R: Rng>(
336        &self,
337        source: DataSource,
338        candidates: &mut [AuthorityIndex],
339        rng: &mut R,
340    ) {
341        if candidates.len() <= 1 {
342            return;
343        }
344
345        // Exploration round: ignore ranking entirely. This is the floor that
346        // keeps every peer reachable early and prevents any peer from
347        // monopolizing the head of the list across rounds.
348        if self.should_explore(rng) {
349            candidates.shuffle(rng);
350            return;
351        }
352
353        // Snapshot each candidate's own sample, the first fallback sample that
354        // exists for it and its failure state under the lock, then release it
355        // before ordering (parking_lot::Mutex must not be held across the work).
356        let samples: Vec<(Option<f64>, Option<f64>, bool)> = {
357            let tracks = self.inner.lock();
358            let Some(track) = tracks.per_kind.get(&source) else {
359                return;
360            };
361            let fallback_tracks: Vec<_> = source
362                .responsiveness_fallbacks()
363                .iter()
364                .filter_map(|fallback| tracks.per_kind.get(fallback))
365                .collect();
366            candidates
367                .iter()
368                .map(|peer| {
369                    let stat = track.get(peer.value());
370                    let fallback_latency = fallback_tracks.iter().find_map(|track| {
371                        track
372                            .get(peer.value())
373                            .and_then(|stat| stat.effective_latency_ms)
374                    });
375                    (
376                        stat.and_then(|stat| stat.effective_latency_ms),
377                        fallback_latency,
378                        stat.is_some_and(|stat| stat.last_fetch_failed),
379                    )
380                })
381                .collect()
382        };
383
384        // A peer with no sample for this source is placed by what the closest
385        // fallback that has measured it saw, and only by the neutral prior when
386        // none of them has. Every sample is floored so it stays strictly
387        // positive.
388        let neutral = source.neutral_latency_ms();
389        let stats: Vec<(f64, bool)> = samples
390            .iter()
391            .map(|(own, fallback, last_fetch_failed)| {
392                (
393                    own.or(*fallback).unwrap_or(neutral).max(MIN_LATENCY_MS),
394                    *last_fetch_failed,
395                )
396            })
397            .collect();
398
399        Self::reorder_by_latency(candidates, &stats, rng);
400    }

402    /// Returns whether this selection should explore instead of using ranking.
403    pub(crate) fn should_explore<R: Rng>(&self, rng: &mut R) -> bool {
404        rng.gen::<f64>() < EXPLORE_PROBABILITY
405    }

407    /// Returns the uniquely fastest peer measured delivering `author`'s
408    /// headers.
409    pub(crate) fn fastest_peer_for_header_delivery(
410        &self,
411        author: AuthorityIndex,
412        candidates: &[AuthorityIndex],
413    ) -> Option<(AuthorityIndex, f64)> {
414        let latencies = self.header_delivery_latencies(author, candidates);
415        let fastest_latency = latencies.iter().flatten().copied().min_by(f64::total_cmp)?;
416        let mut fastest = candidates
417            .iter()
418            .copied()
419            .zip(latencies)
420            .filter(|(_, latency)| {
421                latency.is_some_and(|latency| latency.total_cmp(&fastest_latency).is_eq())
422            })
423            .map(|(peer, _)| peer);
424        let peer = fastest.next()?;
425        fastest.next().is_none().then_some((peer, fastest_latency))
426    }

428    /// Returns the slowest peer, treating an unmeasured peer as slower.
429    pub(crate) fn slowest_peer_for_header_delivery(
430        &self,
431        author: AuthorityIndex,
432        candidates: &[AuthorityIndex],
433    ) -> Option<(AuthorityIndex, Option<f64>)> {
434        candidates
435            .iter()
436            .copied()
437            .zip(self.header_delivery_latencies(author, candidates))
438            .max_by(|(left_peer, left_latency), (right_peer, right_latency)| {
439                left_latency
440                    .unwrap_or(f64::INFINITY)
441                    .total_cmp(&right_latency.unwrap_or(f64::INFINITY))
442                    .then_with(|| left_peer.cmp(right_peer))
443            })
444    }

446    fn header_delivery_latencies(
447        &self,
448        author: AuthorityIndex,
449        candidates: &[AuthorityIndex],
450    ) -> Vec<Option<f64>> {
451        let table = self.streaming_headers.lock();
452        candidates
453            .iter()
454            .map(|peer| {
455                table
456                    .get(peer.value())
457                    .and_then(|row| row.get(author.value()).copied().flatten())
458            })
459            .collect()
460    }

462    /// Writes the ranked order back into `candidates` from per-candidate
463    /// `(effective latency, last fetch failed)` stats; the ordering rules are
464    /// described on [`Self::prioritize`].
465    fn reorder_by_latency<R: Rng>(
466        candidates: &mut [AuthorityIndex],
467        stats: &[(f64, bool)],
468        rng: &mut R,
469    ) {
470        let mut healthy: Vec<(f64, f64, AuthorityIndex)> = Vec::new();
471        let mut failed: Vec<(f64, f64, AuthorityIndex)> = Vec::new();
472        for (peer, (latency, last_fetch_failed)) in candidates.iter().zip(stats) {
473            let entry = (*latency, rng.gen::<f64>(), *peer);
474            if *last_fetch_failed {
475                failed.push(entry);
476            } else {
477                healthy.push(entry);
478            }
479        }
480        let by_latency_then_key =
481            |a: &(f64, f64, AuthorityIndex), b: &(f64, f64, AuthorityIndex)| {
482                a.0.partial_cmp(&b.0)
483                    .unwrap_or(std::cmp::Ordering::Equal)
484                    .then(a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal))
485            };
486        healthy.sort_by(by_latency_then_key);
487        failed.sort_by(by_latency_then_key);
488
489        let mut ordered: Vec<AuthorityIndex> = Vec::with_capacity(candidates.len());
490        let tail = healthy.split_off(healthy.len().min(SELECTION_WINDOW));
491        while !healthy.is_empty() {
492            let total: f64 = healthy.iter().map(|(latency, _, _)| 1.0 / latency).sum();
493            let mut draw = rng.gen::<f64>() * total;
494            let mut chosen = healthy.len() - 1;
495            for (i, (latency, _, _)) in healthy.iter().enumerate() {
496                draw -= 1.0 / latency;
497                if draw <= 0.0 {
498                    chosen = i;
499                    break;
500                }
501            }
502            ordered.push(healthy.remove(chosen).2);
503        }
504        ordered.extend(tail.into_iter().map(|(_, _, peer)| peer));
505        ordered.extend(failed.into_iter().map(|(_, _, peer)| peer));
506
507        for (slot, peer) in candidates.iter_mut().zip(ordered) {
508            *slot = peer;
509        }
510    }

512    fn snapshot(track: &[PeerStat]) -> Vec<PeerStat> {
513        track.to_vec()
514    }

516    /// Recomputes and publishes, for `source`, the expected per-fetch latency
517    /// under two selection strategies over the peers measured so far: `uniform`
518    /// (a random peer, the behaviour without ranking) and `weighted` (this
519    /// module's ranking, including its exploration fraction). `uniform /
520    /// weighted` is the latency improvement the ranking buys; equal values mean
521    /// peers are homogeneous and ranking is not helping. Untried peers carry no
522    /// sample and are excluded.
523    fn publish_expected_latencies(&self, source: DataSource, stats: &[PeerStat]) {
524        let measured: Vec<(f64, bool)> = stats
525            .iter()
526            .filter_map(|stat| {
527                stat.effective_latency_ms
528                    .map(|latency| (latency.max(MIN_LATENCY_MS), stat.last_fetch_failed))
529            })
530            .collect();
531        if measured.is_empty() {
532            return;
533        }
534
535        let uniform =
536            measured.iter().map(|(latency, _)| latency).sum::<f64>() / measured.len() as f64;
537
538        // The first pick is drawn from the window of the fastest healthy
539        // measured peers with weight `1 / latency`; the expected latency of
540        // such a draw is the harmonic mean of the window. The exploration
541        // fraction falls back to a uniform draw. With every measured peer
542        // failed, the draw degenerates to the failed set.
543        let mut pool: Vec<f64> = measured
544            .iter()
545            .filter(|(_, last_fetch_failed)| !*last_fetch_failed)
546            .map(|(latency, _)| *latency)
547            .collect();
548        if pool.is_empty() {
549            pool = measured.iter().map(|(latency, _)| *latency).collect();
550        }
551        pool.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
552        let window = pool.len().min(SELECTION_WINDOW);
553        let harmonic_mean = window as f64
554            / pool[..window]
555                .iter()
556                .map(|latency| 1.0 / latency)
557                .sum::<f64>();
558        let weighted = (1.0 - EXPLORE_PROBABILITY) * harmonic_mean + EXPLORE_PROBABILITY * uniform;
559
560        let gauge = &self
561            .metrics
562            .node_metrics
563            .peer_responsiveness_expected_latency_ms;
564        gauge
565            .with_label_values(&[source.as_str(), "uniform"])
566            .set(uniform);
567        gauge
568            .with_label_values(&[source.as_str(), "weighted"])
569            .set(weighted);
570    }

620    /// Counts, over `trials` seeded calls, how often each candidate ends up
621    /// first. Deterministic for a fixed seed.
622    fn lead_counts(
623        pr: &PeerResponsiveness,
624        source: DataSource,
625        candidates: &[AuthorityIndex],
626        trials: usize,
627        seed: u64,
628    ) -> std::collections::BTreeMap<AuthorityIndex, usize> {
629        let mut rng = StdRng::seed_from_u64(seed);
630        let mut counts = std::collections::BTreeMap::new();
631        for _ in 0..trials {
632            let mut c = candidates.to_vec();
633            pr.prioritize(source, &mut c, &mut rng);
634            *counts.entry(c[0]).or_insert(0) += 1;
635        }
636        counts
637    }

1180    #[test]
1181    fn last_failed_peer_goes_last_until_it_succeeds() {
1182        let pr = responsiveness(4);
1183        // idx(1) has a fast history but its most recent fetch failed;
1184        // idx(2) is a measured slow-but-healthy peer; idx(3) is untried.
1185        pr.record_success(DataSource::TransactionSynchronizer, idx(1), ms(10));
1186        pr.record_failure_with_timeout(DataSource::TransactionSynchronizer, idx(1), ms(2_000));
1187        pr.record_success(DataSource::TransactionSynchronizer, idx(2), ms(1_000));
1188        let candidates = vec![idx(1), idx(2), idx(3)];
1189
1190        let trials = 10_000;
1191        let count_positions = |seed: u64| {
1192            let mut rng = StdRng::seed_from_u64(seed);
1193            let mut last = 0usize;
1194            let mut first = 0usize;
1195            for _ in 0..trials {
1196                let mut c = candidates.clone();
1197                pr.prioritize(DataSource::TransactionSynchronizer, &mut c, &mut rng);
1198                if c[2] == idx(1) {
1199                    last += 1;
1200                }
1201                if c[0] == idx(1) {
1202                    first += 1;
1203                }
1204            }
1205            (last, first)
1206        };
1207
1208        // Outside the exploration shuffle the failed peer is always last, and
1209        // only exploration can put it first.
1210        let (last, first) = count_positions(23);
1211        assert!(
1212            last as f64 / trials as f64 > 0.93,
1213            "failed peer should be pinned last: {last}"
1214        );
1215        assert!(
1216            (first as f64) / (trials as f64) < 0.05,
1217            "failed peer should lead only via exploration: {first}"
1218        );
1219
1220        // A single success clears the failure state: the peer competes in the
1221        // randomized tiers again and is no longer pinned last.
1222        pr.record_success(DataSource::TransactionSynchronizer, idx(1), ms(10));
1223        let (last_after, _) = count_positions(29);
1224        assert!(
1225            (last_after as f64) / (trials as f64) < 0.8,
1226            "recovered peer must not stay pinned last: {last_after}"
1227        );
1228    }
