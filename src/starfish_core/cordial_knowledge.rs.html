35/// Maximum round gap to consider a peer's useful shards/headers as still
36/// relevant. 40 rounds correspond to at least 2 second due to the minimum block
37/// delay

39/// Capacity of the cordial knowledge channel. For normal operation with
40/// 100 authorities, this allows buffering up to 5 seconds of headers at 20
41/// blocks/sec. When the channel is full, the sender will skip sending new
42/// messages.
43const CORDIAL_KNOWLEDGE_CHANNEL_CAPACITY: usize = 10_000;
44/// Eviction is performed every EVICTION_CHECK_INTERVAL processed messages.
45/// This allows batching eviction checks instead of checking on every
46/// message. For this operation, we don't need high precision, but we don't
47/// skip evictions for too long either.
48const EVICTION_CHECK_INTERVAL: usize = 10_000;

52/// Tracks one author for which recent peer block bundles contained an accepted
53/// header or referenced a missing ancestor, and the peers asked to keep
54/// including the author's headers in block bundles.

56struct MissingAuthor {
57    /// Latest local own-block round when a peer's block bundle supplied one of
58    /// this author's headers or exposed one as a missing ancestor.
59    last_useful_round: Round,

65    selected_peers: BTreeSet<AuthorityIndex>,

68/// Manages the global cordial knowledge state.
69/// Receives high-level updates from DagState and AuthorityService and
70/// notifies per-connection tasks.
71pub(crate) struct CordialKnowledge {
72    context: Arc<Context>,
73    /// Receives high-level updates from DAG state (new headers, new own shards)
74    /// and AuthorityService
75    cordial_knowledge_receiver: Receiver<CordialKnowledgeMessage>,
76    /// Receives eviction rounds from DagState (latest-only).
77    eviction_rounds_receiver: tokio::sync::watch::Receiver<Vec<Round>>,
78    /// Latest round per author whose shards are useful to fetch from peers:
79    /// we know of the block header but lack its payload. Global input from
80    /// which each peer's shard requests are selected.
81    last_useful_shards_from_peer_round: Vec<Option<Round>>,
82    /// Keeps track of the most recent DAG cordial
83    /// knowledge (who knows which blocks) for each authority. This is a helper
84    /// structure that is used primarily for traversing the recent DAG. This
85    /// struct is evicted after flushing the dag state to storage and is not
86    /// persisted. To access the cordial knowledge of a given block_ref, one
87    /// shall retrieve it from `cordial_knowledge[block_ref.
88    /// author][block_ref.round][block_ref.digest]`. The provided value is a
89    /// tuple of (ancestors, who knows the block header).
90    cordial_knowledge: Vec<BTreeMap<Round, AHashMap<BlockHeaderDigest, (Ancestors, AuthoritySet)>>>,
91    /// Each Connection Knowledge corresponds to one peer. Upon reception of a
92    /// message from CordialKnowledge, we propagate the respected
93    /// information for each connection.
94    connection_knowledges: Vec<Arc<RwLock<ConnectionKnowledge>>>,
95    /// Authors for which recent peer block bundles contained an accepted header
96    /// or referenced a missing ancestor, indexed by author.
97    missing_authors: Vec<Option<MissingAuthor>>,
98    /// Highest local own-block round seen.
99    latest_own_block_round: Round,
100    /// Whether headers are currently requested from each peer. This ensures an
101    /// empty set is sent once when the last request is removed.
102    has_useful_headers_from_peer: Vec<bool>,
103    rng: StdRng,
104    /// Shard authors currently requested from each peer.
105    requested_shard_authors: Vec<BTreeMap<AuthorityIndex, Round>>,
106}

108/// High-level messages sent to the CordialKnowledge task.
109/// NewHeader, NewShard are received from DAG state.
110/// UsefulShardsFromPeers is received from AuthorityService.
111#[derive(Debug)]
112pub enum CordialKnowledgeMessage {
113    /// A new verified block header to integrate into cordial knowledge.
114    /// Includes transaction commitments of all blocks acknowledged by this
115    /// header.
116    NewHeader {
117        header: VerifiedBlockHeader,
118        ack_transactions_commitments: Vec<Option<TransactionsCommitment>>,
119    },
120    /// A new verified own shard to integrate into cordial knowledge.
121    NewShard(GenericTransactionRef),
122    /// Update internal state about shards from which authorities are useful for
123    /// the local node
124    UsefulShardsFromPeers(BTreeMap<AuthorityIndex, Round>),
125    /// Useful header information from one peer's block bundle.
126    UsefulHeadersFromPeer {
127        peer: AuthorityIndex,
128        useful_header_authors: BTreeSet<AuthorityIndex>,
129    },
130}

132impl CordialKnowledgeMessage {
133    /// Outputs the type of CordialKnowledgeMessage in a string slice format
134    fn type_label(&self) -> &'static str {
135        match self {
136            CordialKnowledgeMessage::NewHeader { .. } => "New header",
137            CordialKnowledgeMessage::NewShard(_) => "New shard",
138            CordialKnowledgeMessage::UsefulShardsFromPeers(_) => "Useful authors for shards",
139            CordialKnowledgeMessage::UsefulHeadersFromPeer { .. } => "Useful headers from peer",
140        }
141    }

162    /// Gracefully stop the CordialKnowledge background task and all connection
163    /// tasks.
164    pub async fn stop(&self) -> Result<(), JoinError> {
165        // Stop main CordialKnowledge loop
166        let mut guard = self.cordial_knowledge_handle.lock().await;
167
168        if let Some(main_handle) = guard.take() {
169            main_handle.abort();
170            match main_handle.await {
171                Ok(_) => (),
172                Err(e) if e.is_cancelled() => (),
173                Err(e) => return Err(e),
174            }
175        }
176        Ok(())
177    }
178    /// Report from AuthorityService useful information about headers and
179    /// shards to global knowledge and connection knowledge.
180    pub fn report_useful_authors(
181        &self,
182        peer: AuthorityIndex,
183        serialized_block_bundle_parts: &SerializedBlockBundleParts,
184        additional_block_headers: &[VerifiedBlockHeader],
185        missing_ancestors: &BTreeSet<BlockRef>,
186        block_round: Round,
187    ) -> ConsensusResult<()> {
188        let cordial_knowledge_sender = &self.cordial_knowledge_sender;
189        // Accepted headers arrive before their full blocks and missing ancestors
190        // lack header and payload alike, so both authors' shards are useful.
191        let mut useful_shard_authors: BTreeMap<AuthorityIndex, Round> = BTreeMap::new();
192        let useful_shard_refs = additional_block_headers
193            .iter()
194            .map(|header| (header.author(), header.round()))
195            .chain(
196                missing_ancestors
197                    .iter()
198                    .map(|block_ref| (block_ref.author, block_ref.round)),
199            );
200        for (author, round) in useful_shard_refs {
201            useful_shard_authors
202                .entry(author)
203                .and_modify(|was_round| *was_round = (*was_round).max(round))
204                .or_insert(round);
205        }
206
207        // Extract authorities this peer finds useful for cordial dissemination from our
208        // side
209        let useful_headers_to_peer = serialized_block_bundle_parts.useful_headers_authors();
210        let useful_headers_to_peer = useful_headers_to_peer
211            .iter()
212            .map(|&a| (a, block_round))
213            .collect::<BTreeMap<_, _>>();
214        // Extract authorities this peer finds useful shards from our side
215        let useful_shards_to_peer = serialized_block_bundle_parts.useful_shards_authors();
216        let useful_shards_to_peer = useful_shards_to_peer
217            .iter()
218            .map(|&a| (a, block_round))
219            .collect::<BTreeMap<_, _>>();
220
221        // Notify connection knowledge about useful headers and shards to/from this peer
222        let connection_knowledge_message = ConnectionKnowledgeMessage::UsefulAuthors {
223            useful_headers_to_peer,
224            useful_shards_to_peer,
225        };
226        {
227            let mut connection_knowledge_guard = self.connection_knowledges[peer].write();
228            connection_knowledge_guard.process_one_message(connection_knowledge_message);
229        }
230        // Notify global cordial knowledge about useful shards from this peer
231        if !useful_shard_authors.is_empty() {
232            let cordial_knowledge_message =
233                CordialKnowledgeMessage::UsefulShardsFromPeers(useful_shard_authors);
234            if let Err(TrySendError::Closed(_)) =
235                cordial_knowledge_sender.try_send(cordial_knowledge_message)
236            {
237                return Err(ConsensusError::Shutdown);
238            }
239        }
240
241        // Report what this peer's bundle said about headers, so the task can
242        // pick who to ask for the missing authors.
243        // Accepted headers remain useful input even when they arrive before the
244        // block that references them and therefore prevent a missing ancestor.
245        let useful_header_authors = additional_block_headers
246            .iter()
247            .map(|header| header.author())
248            .chain(missing_ancestors.iter().map(|block_ref| block_ref.author))
249            .collect::<BTreeSet<_>>();
250        if !useful_header_authors.is_empty() {
251            let cordial_knowledge_message = CordialKnowledgeMessage::UsefulHeadersFromPeer {
252                peer,
253                useful_header_authors,
254            };
255            if let Err(TrySendError::Closed(_)) =
256                cordial_knowledge_sender.try_send(cordial_knowledge_message)
257            {
258                return Err(ConsensusError::Shutdown);
259            }
260        }
261
262        Ok(())
263    }
264}

267    /// Create a new CordialKnowledge instance along with its associated
268    /// channels.
269    fn new(
270        context: Arc<Context>,
271        dag_state: Arc<RwLock<DagState>>,
272    ) -> (
273        Self,
274        Vec<Arc<RwLock<ConnectionKnowledge>>>,
275        Sender<CordialKnowledgeMessage>,
276        tokio::sync::watch::Sender<Vec<Round>>,
277    ) {
278        let num_authorities = context.committee.size();
279
280        // Main bounded channel for high-level DAG updates (monitored for metrics)
281        let (cordial_knowledge_sender, cordial_knowledge_receiver): (
282            Sender<CordialKnowledgeMessage>,
283            Receiver<CordialKnowledgeMessage>,
284        ) = monitored_mpsc::channel("cordial_knowledge", CORDIAL_KNOWLEDGE_CHANNEL_CAPACITY);
285        let (eviction_rounds_sender, eviction_rounds_receiver) =
286            tokio::sync::watch::channel(Vec::new());
287
288        let mut connection_knowledges = Vec::with_capacity(num_authorities);
289
290        for _ in 0..num_authorities {
291            let connection_knowledge = ConnectionKnowledge::new(context.clone(), dag_state.clone());
292
293            let connection_knowledge = Arc::new(RwLock::new(connection_knowledge));
294
295            connection_knowledges.push(connection_knowledge);
296        }
297
298        (
299            Self {
300                context,
301                cordial_knowledge_receiver,
302                eviction_rounds_receiver,
303                cordial_knowledge: vec![BTreeMap::new(); num_authorities],
304                last_useful_shards_from_peer_round: vec![None; num_authorities],
305                connection_knowledges: connection_knowledges.clone(),
306                missing_authors: vec![None; num_authorities],
307                latest_own_block_round: Round::MIN,
308                has_useful_headers_from_peer: vec![false; num_authorities],
309                rng: StdRng::from_entropy(),
310                requested_shard_authors: vec![BTreeMap::new(); num_authorities],
311            },
312            connection_knowledges,
313            cordial_knowledge_sender,
314            eviction_rounds_sender,
315        )
316    }

331    /// Start the CordialKnowledge task and all ConnectionKnowledge tasks.
332    /// Updates the DAG state with the sender to the CordialKnowledge task.
333    /// Return a handle to these tasks.
334    pub fn start(
335        context: Arc<Context>,
336        dag_state: Arc<RwLock<DagState>>,
337    ) -> Arc<CordialKnowledgeHandle> {
338        // Build main CordialKnowledge and associated channels
339        let (
340            cordial_knowledge,
341            connection_knowledges,
342            cordial_knowledge_sender,
343            eviction_rounds_sender,
344        ) = CordialKnowledge::new(context, dag_state.clone());
345        // Spawn the main CordialKnowledge loop
346        let cordial_knowledge_handle = tokio::spawn(async move {
347            cordial_knowledge.run().await;
348        });
349
350        dag_state.write().set_cordial_knowledge_senders(
351            cordial_knowledge_sender.clone(),
352            eviction_rounds_sender,
353        );
354
355        // Return handle with all pieces assembled
356        Arc::new(CordialKnowledgeHandle {
357            cordial_knowledge_sender,
358            connection_knowledges,
359            cordial_knowledge_handle: Mutex::new(Some(cordial_knowledge_handle)),
360        })
361    }

363    /// Main async loop: receives high-level updates (headers, shards)
364    /// from DAG state and updates global knowledge + notifies per-connection
365    /// tasks. Evictions are checked periodically via a watch channel.
366    async fn run(mut self) {
367        debug!("Cordial Knowledge main loop started");
368        let mut processed_since_eviction = 0usize;
369
370        loop {
371            match self.cordial_knowledge_receiver.recv().await {
372                Some(msg) => {
373                    let mut batch = vec![msg];
374                    while let Ok(msg) = self.cordial_knowledge_receiver.try_recv() {
375                        batch.push(msg);
376                    }
377                    processed_since_eviction = processed_since_eviction.saturating_add(batch.len());
378                    // Report the buffer size
379                    self.context
380                        .metrics
381                        .node_metrics
382                        .cordial_knowledge_message_batch_size
383                        .observe(batch.len() as f64);
384                    let mut vec_connection_knowledge_msgs_batch: Vec<Vec<_>> =
385                        (0..self.context.committee.size())
386                            .map(|_| Vec::new())
387                            .collect();
388
389                    let own_block_round_before_batch = self.latest_own_block_round;
390                    for msg in batch {
391                        if let Some(vec_connection_knowledge_msgs) = self.process_message(msg) {
392                            for (index, msgs) in
393                                vec_connection_knowledge_msgs.into_iter().enumerate()
394                            {
395                                vec_connection_knowledge_msgs_batch[index].extend(msgs);
396                            }
397                        }
398                    }
399
400                    // Requested authors can be sent only with an own block, so
401                    // refresh each peer's requests after processing a newer one.
402                    if self.latest_own_block_round > own_block_round_before_batch {
403                        self.append_useful_headers_from_peer_msgs(
404                            &mut vec_connection_knowledge_msgs_batch,
405                        );
406                        self.append_useful_shards_from_peer_msgs(
407                            &mut vec_connection_knowledge_msgs_batch,
408                        );
409                    }
410
411                    if processed_since_eviction >= EVICTION_CHECK_INTERVAL {
412                        self.append_eviction_msgs_if_changed(
413                            &mut vec_connection_knowledge_msgs_batch,
414                        );
415                        self.report_sizes();
416                        processed_since_eviction = 0;
417                    }
418
419                    for (index, msgs) in vec_connection_knowledge_msgs_batch.into_iter().enumerate()
420                    {
421                        if !msgs.is_empty() {
422                            let mut guard = self.connection_knowledges[index].write();
423                            guard.process_vec_messages(msgs);
424                        }
425                    }
426                }
427                None => {
428                    debug!("Cordial Knowledge channel closed; exiting loop");
429                    break;
430                }
431            }
432        }
433
434        debug!("Cordial Knowledge main loop finished");
435    }

437    fn append_eviction_msgs_if_changed(
438        &mut self,
439        vec_connection_knowledge_msgs_batch: &mut [Vec<ConnectionKnowledgeMessage>],
440    ) {
441        if !self.eviction_rounds_receiver.has_changed().unwrap_or(false) {
442            return;
443        }
444        let evicted_rounds = self.eviction_rounds_receiver.borrow_and_update().clone();
445        if evicted_rounds.len() != self.context.committee.size() {
446            warn!(
447                "Eviction rounds length {} does not match committee size {}; skipping eviction",
448                evicted_rounds.len(),
449                self.context.committee.size()
450            );
451            return;
452        }
453        if let Some(vec_connection_knowledge_msgs) = self.handle_evict_below(evicted_rounds) {
454            for (index, msgs) in vec_connection_knowledge_msgs.into_iter().enumerate() {
455                vec_connection_knowledge_msgs_batch[index].extend(msgs);
456            }
457        }
458    }

460    /// Processes a single high-level cordial knowledge message.
461    fn process_message(
462        &mut self,
463        cordial_knowledge_message: CordialKnowledgeMessage,
464    ) -> Option<Vec<Vec<ConnectionKnowledgeMessage>>> {
465        // Report the type of message
466        self.context
467            .metrics
468            .node_metrics
469            .cordial_knowledge_processed_messages
470            .with_label_values(&[cordial_knowledge_message.type_label()])
471            .inc();
472
473        // Handle the cordial knowledge message depending on its type
474
475        match cordial_knowledge_message {
476            CordialKnowledgeMessage::NewHeader {
477                header,
478                ack_transactions_commitments,
479            } => self.update_cordial_knowledge(&header, &ack_transactions_commitments),
480            CordialKnowledgeMessage::NewShard(gen_tx_ref) => {
481                self.prepare_new_shard_msgs(gen_tx_ref)
482            }
483            CordialKnowledgeMessage::UsefulShardsFromPeers(useful_shards_from_peer) => {
484                self.handle_useful_shards_from(useful_shards_from_peer);
485                None
486            }
487            CordialKnowledgeMessage::UsefulHeadersFromPeer {
488                peer,
489                useful_header_authors,
490            } => {
491                self.handle_useful_headers_from_peer(peer, useful_header_authors);
492                None
493            }
494        }
495    }

497    // Helper function to update authority rounds if the new round is greater
498    fn update_authority_rounds_if_greater(
499        target: &mut [Option<Round>],
500        updates: BTreeMap<AuthorityIndex, Round>,
501    ) -> bool {
502        let mut changed = false;
503        for (authority, new_round) in updates {
504            if let Some(existing_round) = &mut target[authority.value()] {
505                if new_round > *existing_round {
506                    *existing_round = new_round;
507                    changed = true;
508                }
509            } else {
510                target[authority.value()] = Some(new_round);
511                changed = true;
512            }
513        }
514        changed
515    }

528    /// Record useful header information from one peer's block bundle.
529    fn handle_useful_headers_from_peer(
530        &mut self,
531        peer: AuthorityIndex,
532        useful_header_authors: BTreeSet<AuthorityIndex>,
533    ) {
534        let own_index = self.context.own_index;
535        let latest_own_block_round = self.latest_own_block_round;
536        for author in useful_header_authors {
537            if author == own_index {
538                continue;
539            }
540            let missing_author = self.missing_authors[author].get_or_insert_default();
541            missing_author.last_useful_round = latest_own_block_round;
542            if peer != author && peer != own_index {
543                missing_author
544                    .useful_peers
545                    .insert(peer, latest_own_block_round);
546            }
547        }
548    }

550    /// Refresh which authors each peer is asked to supply. Accepted headers and
551    /// references to missing headers both keep a request active.
552    fn append_useful_headers_from_peer_msgs(
553        &mut self,
554        vec_connection_knowledge_msgs_batch: &mut [Vec<ConnectionKnowledgeMessage>],
555    ) {
556        self.refresh_missing_authors_and_peers();
557
558        let bounded = self.context.parameters.enable_bounded_header_advertisement;
559        let latest_own_block_round = self.latest_own_block_round;
560        let mut missing_authors = 0;
561        let mut useful_headers_from_peer: Vec<BTreeMap<AuthorityIndex, Round>> =
562            vec![BTreeMap::new(); self.context.committee.size()];
563        for (author_index, missing_author) in self.missing_authors.iter().enumerate() {
564            let Some(missing_author) = missing_author else {
565                continue;
566            };
567            missing_authors += 1;
568            let asked_count = if bounded {
569                missing_author.selected_peers.len()
570            } else {
571                missing_author.useful_peers.len()
572            };
573            let unselected_useful_count = if bounded {
574                missing_author
575                    .useful_peers
576                    .len()
577                    .saturating_sub(missing_author.selected_peers.len())
578            } else {
579                0
580            };
581            let author = AuthorityIndex::from(author_index as u8);
582            self.context
583                .metrics
584                .node_metrics
585                .cordial_knowledge_selected_peers
586                .with_label_values(&[self.context.authority_hostname(author)])
587                .set(asked_count as i64);
588            self.context
589                .metrics
590                .node_metrics
591                .cordial_knowledge_unselected_useful_peers
592                .with_label_values(&[self.context.authority_hostname(author)])
593                .set(unselected_useful_count as i64);
594            // Bounding asks the selected subset; otherwise every useful peer is asked.
595            if bounded {
596                for peer in &missing_author.selected_peers {
597                    useful_headers_from_peer[peer.value()].insert(author, latest_own_block_round);
598                }
599            } else {
600                for peer in missing_author.useful_peers.keys() {
601                    useful_headers_from_peer[peer.value()].insert(author, latest_own_block_round);
602                }
603            }
604        }
605        for (peer_index, authors) in useful_headers_from_peer.into_iter().enumerate() {
606            if peer_index == self.context.own_index.value() {
607                continue;
608            }
609            let non_empty = !authors.is_empty();
610            if non_empty || self.has_useful_headers_from_peer[peer_index] {
611                vec_connection_knowledge_msgs_batch[peer_index].push(
612                    ConnectionKnowledgeMessage::SetUsefulHeadersFromPeer(authors),
613                );
614            }
615            self.has_useful_headers_from_peer[peer_index] = non_empty;
616        }
617
618        self.context
619            .metrics
620            .node_metrics
621            .cordial_knowledge_missing_authors
622            .set(missing_authors);

625    /// Remove stale authors and useful peers. When bounded, retain selected
626    /// useful peers and fill free places. With ranking enabled, keep the
627    /// fastest measured useful peer selected and occasionally try another
628    /// useful peer.
629    fn refresh_missing_authors_and_peers(&mut self) {
630        let bounded = self.context.parameters.enable_bounded_header_advertisement;
631        let latest_own_block_round = self.latest_own_block_round;
632        let committee_size = self.context.committee.size();
633        let is_stale = |round: Round| {
634            latest_own_block_round.saturating_sub(round) > MAX_ROUND_GAP_FOR_USEFUL_PARTS
635        };
636
637        for author_index in 0..committee_size {
638            let author = AuthorityIndex::from(author_index as u8);
639            let Some(missing_author) = self.missing_authors[author_index].as_mut() else {
640                continue;
641            };
642
643            if is_stale(missing_author.last_useful_round) {
644                self.missing_authors[author_index] = None;
645                self.context
646                    .metrics
647                    .node_metrics
648                    .cordial_knowledge_selected_peers
649                    .with_label_values(&[self.context.authority_hostname(author)])
650                    .set(0);
651                self.context
652                    .metrics
653                    .node_metrics
654                    .cordial_knowledge_unselected_useful_peers
655                    .with_label_values(&[self.context.authority_hostname(author)])
656                    .set(0);
657                continue;
658            }
659
660            missing_author
661                .useful_peers
662                .retain(|_, round| !is_stale(*round));
663            if !bounded {
664                continue;
665            }
666
667            missing_author
668                .selected_peers
669                .retain(|peer| missing_author.useful_peers.contains_key(peer));
670            for peer in missing_author.useful_peers.keys() {
671                if missing_author.selected_peers.len() == MAX_AUTHORITIES_TO_FETCH_PER_BLOCK_HEADER
672                {
673                    break;
674                }
675                missing_author.selected_peers.insert(*peer);
676            }
677            if self.context.parameters.enable_peer_responsiveness_ranking {
678                Self::select_fastest_useful_peer(&self.context, author, missing_author);
679                Self::explore_header_peer(&self.context, &mut self.rng, author, missing_author);
680            }
681        }
682    }

684    fn select_fastest_useful_peer(
685        context: &Context,
686        author: AuthorityIndex,
687        missing_author: &mut MissingAuthor,
688    ) {
689        let useful_peers: Vec<_> = missing_author.useful_peers.keys().copied().collect();
690        let Some((fastest_peer, fastest_latency)) = context
691            .peer_responsiveness
692            .fastest_peer_for_header_delivery(author, &useful_peers)
693        else {
694            return;
695        };
696        if missing_author.selected_peers.contains(&fastest_peer) {
697            return;
698        }
699
700        if missing_author.selected_peers.len() == MAX_AUTHORITIES_TO_FETCH_PER_BLOCK_HEADER {
701            let selected_peers: Vec<_> = missing_author.selected_peers.iter().copied().collect();
702            let Some((slowest_peer, slowest_latency)) = context
703                .peer_responsiveness
704                .slowest_peer_for_header_delivery(author, &selected_peers)
705            else {
706                return;
707            };
708            if slowest_latency.is_some_and(|latency| fastest_latency >= latency) {
709                return;
710            }
711            missing_author.selected_peers.remove(&slowest_peer);
712        }
713
714        missing_author.selected_peers.insert(fastest_peer);
715    }

717    fn explore_header_peer(
718        context: &Context,
719        rng: &mut StdRng,
720        author: AuthorityIndex,
721        missing_author: &mut MissingAuthor,
722    ) {
723        if missing_author.selected_peers.len() != MAX_AUTHORITIES_TO_FETCH_PER_BLOCK_HEADER
724            || !context.peer_responsiveness.should_explore(rng)
725        {
726            return;
727        }
728
729        let candidates: Vec<_> = missing_author
730            .useful_peers
731            .keys()
732            .filter(|peer| !missing_author.selected_peers.contains(peer))
733            .copied()
734            .collect();
735        if candidates.is_empty() {
736            return;
737        }
738        let peer = candidates[rng.gen_range(0..candidates.len())];
739        let selected_peers: Vec<_> = missing_author.selected_peers.iter().copied().collect();
740        let Some((slowest_peer, _)) = context
741            .peer_responsiveness
742            .slowest_peer_for_header_delivery(author, &selected_peers)
743        else {
744            return;
745        };
746
747        missing_author.selected_peers.remove(&slowest_peer);
748        missing_author.selected_peers.insert(peer);
749    }

751    /// Refresh which authors each peer is asked to push shards of: only the
752    /// fastest peers that together provide enough shards and stake for
753    /// reconstruction, or every peer when selection is disabled.
754    fn append_useful_shards_from_peer_msgs(
755        &mut self,
756        vec_connection_knowledge_msgs_batch: &mut [Vec<ConnectionKnowledgeMessage>],
757    ) {
758        let own_index = self.context.own_index;
759        let latest_own_block_round = self.latest_own_block_round;
760        let selection_enabled = self.context.parameters.enable_shard_peer_selection;
761        let mut next_requested_shard_authors = vec![BTreeMap::new(); self.context.committee.size()];
762        let mut latest_selection = None;
763
764        for author_index in 0..self.last_useful_shards_from_peer_round.len() {
765            let author = AuthorityIndex::from(author_index as u8);
766            let Some(round) = self.last_useful_shards_from_peer_round[author_index] else {
767                continue;
768            };
769            if round.saturating_add(MAX_ROUND_GAP_FOR_USEFUL_PARTS) < latest_own_block_round {
770                continue;
771            }
772
773            if !selection_enabled {
774                for (peer_index, authors) in next_requested_shard_authors.iter_mut().enumerate() {
775                    if peer_index != own_index.value() {
776                        authors.insert(author, round);
777                    }
778                }
779                continue;
780            }
781
782            let Some(selection) = self.context.peer_responsiveness.select_shard_peers(
783                &self.context.committee,
784                own_index,
785                author,
786            ) else {
787                continue;
788            };
789            for peer in &selection.peers {
790                next_requested_shard_authors[*peer].insert(author, round);
791            }
792            latest_selection = Some(selection);
793        }
794
795        if selection_enabled {
796            self.set_shard_selection_metrics(latest_selection.as_ref());
797        }
798
799        for (peer_index, authors) in next_requested_shard_authors.into_iter().enumerate() {
800            if peer_index == own_index.value()
801                || authors == self.requested_shard_authors[peer_index]
802            {
803                continue;
804            }
805            vec_connection_knowledge_msgs_batch[peer_index].push(
806                ConnectionKnowledgeMessage::SetUsefulShardsFromPeer(authors.clone()),
807            );
808            self.requested_shard_authors[peer_index] = authors;
809        }
810    }

903    /// Update cordial knowledge for exactly one new header.
904    /// Assumes all parents are already stored somewhere in
905    /// `recent_dag_cordial_knowledge` (if not, they will be skipped).
906    /// We traverse back the causal past of the new header and mark all
907    /// ancestors as known by the block author. All acknowledged blocks are
908    /// marked as known by the block author as well.
909    /// At the end, we notify all connections about new
910    /// knowledge changes.
911    fn update_cordial_knowledge(
912        &mut self,
913        header: &VerifiedBlockHeader,
914        ack_transactions_commitments: &[Option<TransactionsCommitment>],
915    ) -> Option<Vec<Vec<ConnectionKnowledgeMessage>>> {
916        let block_ref = header.reference();
917        let block_author = block_ref.author.value();
918        let block_round = block_ref.round;
919        let block_digest = block_ref.digest;
920        let own_index = self.context.own_index.value();
921
922        if block_author == own_index {
923            self.latest_own_block_round = max(self.latest_own_block_round, block_round);
924        }
925
926        // Pre-allocate message buffers
927        let mut vec_knowledge_msgs: Vec<Vec<ConnectionKnowledgeMessage>> =
928            (0..self.context.committee.size())
929                .map(|_| Vec::new())
930                .collect();
931
932        // 1) Ensure we have a round map for this author and insert the block if new
933        let btree_map = &mut self.cordial_knowledge[block_author];
934        let round_map = btree_map.entry(block_round).or_default();
935
936        // Already recorded — nothing else to do.
937        if round_map.contains_key(&block_digest) {
938            return None;
939        }
940
941        // Insert block into cordial knowledge
942        let ancestors: Ancestors = Arc::from(header.ancestors());
943        let who_knows_this_block = AuthoritySet::new_with(block_ref.author, self.context.own_index);
944        round_map.insert(block_digest, (ancestors, who_knows_this_block));
945
946        // 2) Notify all *other* authorities (except self and block_author) about new
947        //    header
948        for (authority, msgs) in vec_knowledge_msgs.iter_mut().enumerate() {
949            // don't send shard to self nor to the author of the block
950            if authority == block_author || authority == own_index {
951                continue;
952            }
953            msgs.push(ConnectionKnowledgeMessage::NewHeader { block_ref });
954        }
955
956        // 3) The block_author now acknowledges previously known transactions,
957        // using the provided transaction commitments.
958        for (acknowledgment, &transactions_commitment) in header
959            .acknowledgments()
960            .iter()
961            .zip(ack_transactions_commitments.iter())
962        {
963            let Some(transactions_commitment) = transactions_commitment else {
964                continue;
965            };
966            let gen_tx_ref =
967                GenericTransactionRef::TransactionRef(crate::transaction_ref::TransactionRef {
968                    round: acknowledgment.round,
969                    author: acknowledgment.author,
970                    transactions_commitment,
971                });
972
973            vec_knowledge_msgs[block_author]
974                .push(ConnectionKnowledgeMessage::RemoveShard { gen_tx_ref });
975        }
976
977        // 4) Traversing back and marking the causal past as known by block_author
978        let mut stack = vec![block_ref];
979        while let Some(current_ref) = stack.pop() {
980            let current_author = current_ref.author.value();
981            let current_round = current_ref.round;
982            let current_digest = current_ref.digest;
983
984            // ---- Get parents of current block ----
985            let parents_buf: Ancestors = {
986                let author_map = &self.cordial_knowledge[current_author];
987                let Some(current_round_map) = author_map.get(&current_round) else {
988                    continue;
989                };
990                let Some((parents, _)) = current_round_map.get(&current_digest) else {
991                    continue;
992                };
993                parents.clone()
994            };
995
996            // Traverse parents
997            for parent_ref in parents_buf.iter() {
998                let parent_author = parent_ref.author.value();
999                let parent_round = parent_ref.round;
1000                let parent_digest = parent_ref.digest;
1001
1002                let parent_author_map = &mut self.cordial_knowledge[parent_author];
1003
1004                if let Some(parent_round_map) = parent_author_map.get_mut(&parent_round) {
1005                    if let Some((_, who_knows_parent)) = parent_round_map.get_mut(&parent_digest) {
1006                        // Mark that block_author now knows this parent
1007                        if who_knows_parent.insert(block_ref.author) {
1008                            vec_knowledge_msgs[block_author].push(
1009                                ConnectionKnowledgeMessage::RemoveHeader {
1010                                    block_ref: *parent_ref,
1011                                },
1012                            );
1013                            stack.push(*parent_ref);
1014                        }
1015                    }
1016                }
1017            }
1018        }
1019        Some(vec_knowledge_msgs)
1020    }
1021}

1093    /// Take useful refs (headers or shards) for the given authorities
1094    /// up to the given round (exclusive), up to max_take total.
1095    /// Generic function that works with both BlockRef and
1096    /// GenericTransactionRef.
1097    fn take_useful_refs_round<T>(
1098        maps: &mut [BTreeMap<Round, AHashSet<T>>],
1099        round_upper_bound_exclusive: Round,
1100        useful_authorities: &[usize],
1101        max_take: usize,
1102        get_author: impl Fn(&T) -> usize,
1103        get_round: impl Fn(&T) -> Round,
1104    ) -> Vec<T>
1105    where
1106        T: Copy + Eq + std::hash::Hash,
1107    {
1108        if useful_authorities.is_empty() || max_take == 0 {
1109            return Vec::new();
1110        }
1111
1112        // Find the smallest existing round among all useful authorities.
1113        let min_round = useful_authorities
1114            .iter()
1115            .filter_map(|&auth| maps[auth].keys().next().copied())
1116            .min();
1117
1118        let Some(mut current_round) = min_round else {
1119            return Vec::new();
1120        };
1121
1122        let mut taken = Vec::with_capacity(max_take);
1123
1124        'outer: while current_round < round_upper_bound_exclusive {
1125            for &authority in useful_authorities {
1126                let map = &maps[authority];
1127                if let Some(refs_from_authority_in_round) = map.get(&current_round) {
1128                    for &item_ref in refs_from_authority_in_round {
1129                        taken.push(item_ref);
1130                        if taken.len() >= max_take {
1131                            break 'outer;
1132                        }
1133                    }
1134                }
1135            }
1136            current_round += 1;
1137        }
1138
1139        // Remove the taken refs from the corresponding authorities
1140        for item_ref in &taken {
1141            let authority = get_author(item_ref);
1142            let round = get_round(item_ref);
1143            if let Some(refs_from_authority_in_round) = maps[authority].get_mut(&round) {
1144                refs_from_authority_in_round.remove(item_ref);
1145                // Remove empty rounds to keep map small
1146                if refs_from_authority_in_round.is_empty() {
1147                    maps[authority].remove(&round);
1148                }
1149            }
1150        }
1151
1152        taken
1153    }

1290    /// Used by AuthorityService to create a block bundle
1291    /// to send to the peer.
1292    pub fn create_bundle(&mut self, block: VerifiedBlock) -> BlockBundle {
1293        let block_round = block.round();
1294        // The global knowledge updates for these ancestors may still be in flight,
1295        // so record them here to have them available for this bundle: each becomes
1296        // the header disseminated as an additional header for its slot. For an
1297        // equivocating author this overrides a header recorded for that slot
1298        // earlier, which is fine, as that one is still obtainable by reference.
1299        for ancestor_block_ref in block.ancestors() {
1300            self.set_referenced_header(*ancestor_block_ref);
1301        }
1302        // 1. Own headers and shards for round up to round_upper_bound_exclusive should
1303        //    be marked as known
1304        let own_index = self.context.own_index;
1305        let mut rounds = vec![Round::MIN; self.context.committee.size()];
1306        rounds[own_index] = block_round; // We are supposed to send own block of this round in a bundle when calling this function with this parameter
1307
1308        self.evict_below(rounds);
1309
1310        // 2. Identify useful authorities for headers and take the corresponding headers
1311        //    from the DAG state
1312        let useful_headers_authors_to_peer: Vec<usize> = self
1313            .last_useful_headers_to_peer_round
1314            .iter()
1315            .enumerate()
1316            .filter(|(_authority_index, &opt_round)| {
1317                if let Some(round) = opt_round {
1318                    round.saturating_add(MAX_ROUND_GAP_FOR_USEFUL_PARTS) >= block_round
1319                } else {
1320                    false
1321                }
1322            })
1323            .map(|(authority_index, _opt_round)| authority_index)
1324            .collect();
1325
1326        let useful_headers_block_refs_to_peer =
1327            self.take_useful_header_block_refs_round(block_round, &useful_headers_authors_to_peer);
1328
1329        let useful_headers_to_peer: Vec<VerifiedBlockHeader> = {
1330            let dag_state_read = self.dag_state.read();
1331            dag_state_read
1332                .get_cached_block_headers(&useful_headers_block_refs_to_peer)
1333                .into_iter()
1334                .flatten() // Filter out None values
1335                .collect()
1336        };
1337
1338        // 3. Identify useful authorities for shards and take the corresponding shards
1339        //    from the DAG state
1340        let useful_shards_authors_to_peer: Vec<usize> = self
1341            .last_useful_shards_to_peer_round
1342            .iter()
1343            .enumerate()
1344            .filter(|(_authority_index, &opt_round)| {
1345                if let Some(round) = opt_round {
1346                    round.saturating_add(MAX_ROUND_GAP_FOR_USEFUL_PARTS) >= block_round
1347                } else {
1348                    false
1349                }
1350            })
1351            .map(|(authority_index, _opt_round)| authority_index)
1352            .collect();
1353
1354        let useful_shards_block_refs_to_peer =
1355            self.take_useful_shard_block_refs_round(block_round, &useful_shards_authors_to_peer);
1356        let useful_shards_to_peer: Vec<Bytes> = {
1357            let dag_state_read = self.dag_state.read();
1358            dag_state_read
1359                .get_cached_shards(&useful_shards_block_refs_to_peer)
1360                .into_iter()
1361                .flatten() // Filter out None values
1362                .collect()
1363        };
1364
1365        // 4. Get useful header authors from peer.
1366        // Authority is (potentially) useful if the
1367        // last known useful round + MAX_ROUND_GAP_FOR_USEFUL_PARTS >=
1368        // round_upper_bound_exclusive
1369        let useful_headers_authors_from_peer = self
1370            .last_useful_headers_from_peer_round
1371            .iter()
1372            .enumerate()
1373            .filter(|(_authority_index, &opt_round)| {
1374                if let Some(round) = opt_round {
1375                    round.saturating_add(MAX_ROUND_GAP_FOR_USEFUL_PARTS) >= block_round
1376                } else {
1377                    false
1378                }
1379            })
1380            .map(|(authority_index, _opt_round)| AuthorityIndex::from(authority_index as u8))
1381            .collect::<BTreeSet<AuthorityIndex>>();
1382
1383        // 5. Get useful shard authors from peer
1384        let useful_shards_authors_from_peer = self
1385            .last_useful_shards_from_peer_round
1386            .iter()
1387            .enumerate()
1388            .filter(|(_authority_index, &opt_round)| {
1389                if let Some(round) = opt_round {
1390                    round.saturating_add(MAX_ROUND_GAP_FOR_USEFUL_PARTS) >= block_round
1391                } else {
1392                    false
1393                }
1394            })
1395            .map(|(authority_index, _opt_round)| AuthorityIndex::from(authority_index as u8))
1396            .collect::<BTreeSet<AuthorityIndex>>();
1397
1398        BlockBundle {
1399            verified_block: block,
1400            verified_headers: useful_headers_to_peer,
1401            serialized_shards: useful_shards_to_peer,
1402            useful_headers_authors: useful_headers_authors_from_peer,
1403            useful_shards_authors: useful_shards_authors_from_peer,
1404        }
1405    }

1407    /// Handles adding a new header to the set of potentially unknown headers.
1408    /// Keeps at most one header per slot, since a bundle carries only one: the
1409    /// receiver drops additional headers of a slot as spam protection. An
1410    /// equivocating header a peer needs is obtained by reference instead.
1411    fn handle_new_header(&mut self, block_ref: BlockRef) {
1412        let round = block_ref.round;
1413        let authority = block_ref.author.value();
1414
1415        let refs_at_slot = self.headers_not_known[authority].entry(round).or_default();
1416        if refs_at_slot.is_empty() {
1417            refs_at_slot.insert(block_ref);
1418        }
1419    }
1420
1421    /// Records `block_ref` as the header to offer for its slot, replacing any
1422    /// header tracked there. Used for the ancestors of a block we are about to
1423    /// send: the peer needs them to accept it.
1424    fn set_referenced_header(&mut self, block_ref: BlockRef) {
1425        let refs_at_slot = self.headers_not_known[block_ref.author.value()]
1426            .entry(block_ref.round)
1427            .or_default();
1428        refs_at_slot.clear();
1429        refs_at_slot.insert(block_ref);
1430    }

1432    /// Handles adding a new shard to the set of potentially unknown shards.
1433    fn handle_new_shard(&mut self, gen_tx_ref: GenericTransactionRef) {
1434        let round = gen_tx_ref.round();
1435        let authority = gen_tx_ref.author().value();
1436
1437        self.shards_not_known[authority]
1438            .entry(round)
1439            .or_default()
1440            .insert(gen_tx_ref);
1441    }

1443    /// Returns (total_headers_not_known, total_shards_not_known) entry counts.
1444    fn sizes(&self) -> (usize, usize) {
1445        let headers: usize = self
1446            .headers_not_known
1447            .iter()
1448            .map(|m| m.values().map(|s| s.len()).sum::<usize>())
1449            .sum();
1450        let shards: usize = self
1451            .shards_not_known
1452            .iter()
1453            .map(|m| m.values().map(|s| s.len()).sum::<usize>())
1454            .sum();
1455        (headers, shards)
1456    }

1458    /// Handles removing a header that this peer now knows.
1459    fn handle_remove_header(&mut self, block_ref: BlockRef) {
1460        let authority = block_ref.author.value();
1461        let round = block_ref.round;
1462
1463        if let Some(set) = self.headers_not_known[authority].get_mut(&round) {
1464            set.remove(&block_ref);
1465            // Optional: remove empty round entries to keep map clean
1466            if set.is_empty() {
1467                self.headers_not_known[authority].remove(&round);
1468            }
1469        }
1470    }
1471
1472    /// Handles removing a shard that this peer now knows.
1473    fn handle_remove_shard(&mut self, gen_tx_ref: GenericTransactionRef) {
1474        let authority = gen_tx_ref.author().value();
1475        let round = gen_tx_ref.round();
1476
1477        if let Some(set) = self.shards_not_known[authority].get_mut(&round) {
1478            set.remove(&gen_tx_ref);
1479            if set.is_empty() {
1480                self.shards_not_known[authority].remove(&round);
1481            }
1482        }
1483    }

2335                    A -> [-D3],
2336                    B -> [-D3],
2337                    C -> [-D3],
2338                    D -> [*],
2339                },
2340                Round 5 : {
2341                    A -> [-D4],
2342                    B -> [-D4],
2343                    C -> [-D4],

2347                    A -> [*],
2348                    B -> [-D5],

2350                    D -> [*],

2367        let connection_knowledge = cordial_knowledge.connection_knowledges[to_whom_index].clone();
2368        // Inject useful info for connection knowledge of peer 1 (B)
2369        // A says that C and D are useful for headers and shards when receiving from B

2371        let msg = ConnectionKnowledgeMessage::UsefulAuthors {
2372            useful_headers_to_peer: BTreeMap::from([
2373                (AuthorityIndex::new_for_test(2), GENESIS_ROUND),

2379            ]),

2387                    (AuthorityIndex::new_for_test(3), GENESIS_ROUND),
2388                ])),
2389            );
2390            connection_knowledge.process_one_message(

2392                    (AuthorityIndex::new_for_test(1), GENESIS_ROUND),
2393                    (AuthorityIndex::new_for_test(3), GENESIS_ROUND),
2394                ])),
2395            );
2396        }
2397
2398        // get all blocks of D. They will be injected to dag state at final_round
2399        let d_blocks = all_blocks
2400            .iter()
2401            .flat_map(|blocks| blocks.iter().filter(|b| b.author() == byzantine_index))
2402            .cloned()
2403            .collect::<Vec<VerifiedBlock>>();

2405        for round in 1..=final_round - 1 {
2406            if round == final_round - 1 {
2407                // Add D's blocks to DAG state only at final_round-1
2408                for block in d_blocks.iter() {
2409                    let VerifiedBlock {
2410                        verified_block_header,
2411                        verified_transactions,
2412                    } = block.clone();
2413                    dag_state
2414                        .write()
2415                        .accept_block_header(verified_block_header, DataSource::Test);
2416                    let gen_transaction_ref = GenericTransactionRef::TransactionRef(
2417                        verified_transactions.transaction_ref(),
2418                    );
2419                    let shard_for_core = VerifiedOwnShard {
2420                        serialized_shard: Bytes::from([0u8; 32].to_vec()), /* put some dummy
2421                                                                            * shard data */
2422                        gen_transaction_ref,
2423                    };
2424                    dag_state.write().add_shard(shard_for_core);
2425                }
2426            }
2427            // add all blocks of this round and our block of next round to dag state
2428            for block in all_blocks[round as usize]
2429                .iter()
2430                .filter(|b| b.author() != our_index && b.author() != byzantine_index)
2431                .chain(std::iter::once(&all_blocks[round as usize + 1][our_index]))
2432            {
2433                let VerifiedBlock {
2434                    verified_block_header,
2435                    verified_transactions,
2436                } = block.clone();
2437                dag_state
2438                    .write()
2439                    .accept_block_header(verified_block_header, DataSource::Test);
2440                let gen_transaction_ref =
2441                    GenericTransactionRef::TransactionRef(verified_transactions.transaction_ref());
2442                let shard_for_core = VerifiedOwnShard {
2443                    serialized_shard: Bytes::from([0u8; 32].to_vec()), // put some dummy shard data
2444                    gen_transaction_ref,
2445                };
2446                dag_state.write().add_shard(shard_for_core);
2447            }
2448            sleep(std::time::Duration::from_millis(10)).await; // give some time for cordial knowledge to update
2449            // By default, for MAX_ROUND_GAP_FOR_USEFUL_PARTS rounds, all unknown
2450            // shards/headers are useful
2451            let block_bundle = {
2452                connection_knowledge
2453                    .write()
2454                    .create_bundle(all_blocks[round as usize + 1][our_index].clone())
2455            };
2456            let BlockBundle {
2457                verified_headers: headers,
2458                serialized_shards: shards,
2459                ..
2460            } = block_bundle;
2461            // In rounds 1..final_round, A should not know any of D's blocks, so no headers
2462            // or shards should be sent to B.
2463            if round < final_round - 1 {
2464                // Only headers of C's block of previous round should be sent
2465                assert_eq!(
2466                    headers.len(),
2467                    1,
2468                    "In round {round}, unexpected headers found: {headers:?}",
2469                );
2470                assert_eq!(
2471                    headers[0].digest(),
2472                    all_blocks[round as usize][2].verified_block_header.digest()
2473                );
2474                assert_eq!(
2475                    shards.len(),
2476                    1,
2477                    "In round {round}, unexpected shards found: {shards:?}",
2478                );
2479            } else {
2480                // In round 6, A should know about D's blocks and send them all to B
2481                let d_headers_in_bundle: Vec<&VerifiedBlockHeader> = headers
2482                    .iter()
2483                    .filter(|h| h.author() == byzantine_index)
2484                    .collect();
2485                assert_eq!(d_headers_in_bundle.len(), final_round as usize - 1); // All 5 headers of D's blocks
2486                // Validator A sends to B all 5 shards of D's blocks and 1 header/shard of C's
2487                // block of round 5
2488                assert_eq!(
2489                    headers.len(),
2490                    final_round as usize,
2491                    "In round {round}, unexpected headers found: {headers:?}",
2492                );
2493                assert_eq!(shards.len(), final_round as usize);
2494            }
2495        }
