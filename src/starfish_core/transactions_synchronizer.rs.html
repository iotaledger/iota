44/// The number of concurrent live transaction fetch requests
45/// Set to the maximum number of rounds per second as it can be called by newly
46/// produced commits only
47const LIVE_FETCH_TRANSACTIONS_CONCURRENCY: usize = 20;
48
49/// The number of concurrent periodic transaction fetch requests
50const PERIODIC_FETCH_TRANSACTIONS_CONCURRENCY: usize = 4;
51
52/// The maximum number of concurrent request per authority for fetching
53/// transactions. It is used separately for live and periodic fetches
54const MAX_CONCURRENT_REQUESTS_PER_AUTHORITY: usize = 2;
55
56/// The maximum number of assigned peers per one call of transaction fetch
57/// It allows to globally limit the number of spawned tasks by
58/// (LIVE_FETCH_TRANSACTIONS_CONCURRENCY +
59/// PERIODIC_FETCH_TRANSACTIONS_CONCURRENCY) *
60/// MAX_ASSIGNED_AUTHORITIES_PER_TRANSACTION_FETCH
61const MAX_ASSIGNED_AUTHORITIES_PER_TRANSACTION_FETCH: usize = 4;
62
63/// Timeout for the transactions synchronizer to run periodically and fetch
64/// missing transactions.
65const TRANSACTIONS_SYNCHRONIZER_TIMEOUT: Duration = Duration::from_millis(500);
66
67/// Timeout that is given to fetch transactions from a given peer.
68pub(crate) const FETCH_REQUEST_TIMEOUT: Duration = Duration::from_millis(2000);

70/// Maximum number of authorities that can concurrently fetch transactions for a
71/// given block ref.

96/// Bounds the concurrent fetches: at most
97/// `MAX_AUTHORITIES_TO_FETCH_PER_TRANSACTION` peers fetch the same transaction
98/// at a time, and each peer serves at most
99/// `MAX_CONCURRENT_REQUESTS_PER_AUTHORITY` requests per sync method.
100#[derive(Default)]
101struct InflightTransactionsMap {
102    inner: Mutex<InflightState>,
103}
104
105#[derive(Default)]
106struct InflightState {
107    /// The peers currently fetching each transaction.
108    fetching_peers: HashMap<TransactionRef, BTreeSet<AuthorityIndex>>,
109    /// The number of requests in flight per peer and sync method.
110    active_requests: BTreeMap<(AuthorityIndex, SyncMethod), usize>,
111}
112
113impl InflightTransactionsMap {
114    /// Locks up to `max_transactions` of `missing_transaction_refs` for
115    /// `peer`, skipping those that `MAX_AUTHORITIES_TO_FETCH_PER_TRANSACTION`
116    /// peers are already fetching. Returns `None` when `peer` has too many
117    /// requests in flight or nothing could be locked.
118    fn lock_transactions(
119        self: &Arc<Self>,
120        missing_transaction_refs: BTreeSet<TransactionRef>,
121        peer: AuthorityIndex,
122        max_transactions: usize,
123        sync_method: SyncMethod,
124    ) -> Option<TransactionsGuard> {
125        let mut guard = self.inner.lock();
126        let state = &mut *guard;
127        let active_requests = state
128            .active_requests
129            .entry((peer, sync_method))
130            .or_insert(0);
131        if *active_requests >= MAX_CONCURRENT_REQUESTS_PER_AUTHORITY {
132            return None;
133        }
134
135        let mut transactions_refs = BTreeSet::new();
136        for tx_ref in missing_transaction_refs {
137            let peers = state.fetching_peers.entry(tx_ref).or_default();
138            if peers.len() < MAX_AUTHORITIES_TO_FETCH_PER_TRANSACTION && peers.insert(peer) {
139                transactions_refs.insert(tx_ref);
140                if transactions_refs.len() >= max_transactions {
141                    break;
142                }
143            }
144        }
145        if transactions_refs.is_empty() {
146            return None;
147        }
148        *active_requests += 1;
149
150        Some(TransactionsGuard {
151            map: self.clone(),
152            transactions_refs,
153            peer,
154            sync_method,
155        })
156    }
157
158    #[cfg(test)]
159    fn num_of_locked_transactions(&self) -> usize {
160        self.inner.lock().fetching_peers.len()
161    }
162}
163
164/// Holds the transactions locked for one request to `peer`; releases them and
165/// the peer's request slot on drop.
166struct TransactionsGuard {
167    map: Arc<InflightTransactionsMap>,
168    transactions_refs: BTreeSet<TransactionRef>,
169    peer: AuthorityIndex,
170    sync_method: SyncMethod,
171}
172
173impl Drop for TransactionsGuard {
174    fn drop(&mut self) {
175        let mut state = self.map.inner.lock();
176        for tx_ref in &self.transactions_refs {
177            let peers = state
178                .fetching_peers
179                .get_mut(tx_ref)
180                .expect("a locked transaction is tracked");
181            assert!(
182                peers.remove(&self.peer),
183                "a locked transaction is tracked for its peer"
184            );
185            if peers.is_empty() {
186                state.fetching_peers.remove(tx_ref);
187            }
188        }
189        *state
190            .active_requests
191            .get_mut(&(self.peer, self.sync_method))
192            .expect("a locked request is counted for its peer") -= 1;
193    }
194}

196pub(crate) struct TransactionsSynchronizerHandle {
197    live_fetch_requests: Sender<BTreeMap<TransactionRef, BTreeSet<AuthorityIndex>>>,
198    tasks: tokio::sync::Mutex<JoinSet<()>>,
199}
200
201impl TransactionsSynchronizerHandle {
202    /// Queues a live fetch of the missing transactions from the authorities
203    /// that acknowledged them. Fails with `TransactionSynchronizerSaturated`
204    /// when the live queue is full; the periodic scheduler picks the
205    /// transactions up later.
206    pub(crate) fn fetch_transactions(
207        &self,
208        missing_transaction_refs: BTreeMap<TransactionRef, BTreeSet<AuthorityIndex>>,
209    ) -> ConsensusResult<()> {
210        self.live_fetch_requests
211            .try_send(missing_transaction_refs)
212            .map_err(|err| match err {
213                TrySendError::Full(_) => ConsensusError::TransactionSynchronizerSaturated,
214                TrySendError::Closed(_) => ConsensusError::Shutdown,
215            })
216    }
217
218    pub(crate) async fn stop(&self) -> Result<(), JoinError> {
219        let mut tasks = self.tasks.lock().await;
220        tasks.abort_all();
221        while let Some(result) = tasks.join_next().await {
222            match result {
223                // task finished successfully
224                Ok(_) => (),
225                // task was cancelled, which is expected on shutdown
226                Err(e) if e.is_cancelled() => (),
227                // propagate other errors (e.g. panics)
228                Err(e) => return Err(e),
229            }
230        }
231
232        Ok(())
233    }
234}

236/// `TransactionsSynchronizer` oversees live transaction synchronization,
237/// crucial for node progress. Live synchronization refers to the process of
238/// retrieving missing transactions, particularly those essential for advancing
239/// a node when transactions from the committed blocks is absent.
240/// `TransactionsSynchronizer` aims for swift catch-up employing two mechanisms:
241///
242/// 1. Explicitly requesting missing transactions from authorities that have acknowledged them in
243///    their blocks that were committed. A locking mechanism allows concurrent requests for missing
244///    transactions from a limited number of authorities simultaneously, enhancing the chances of
245///    timely retrieval.
246///
247/// 2. Periodically requesting from the core the transactions that are still missing and fetching
248///    them via a scheduler. This retrieves the missing transactions that were not fetched via the
249///    live synchronization.
250pub(crate) struct TransactionsSynchronizer<C: NetworkClient, D: CoreThreadDispatcher> {
251    context: Arc<Context>,
252    network_client: Arc<C>,
253    core_dispatcher: Arc<D>,
254    dag_state: Arc<RwLock<DagState>>,
255    /// Applies the same transaction limit and batch verification checks to
256    /// fetched payloads as the direct block-bundle route.
257    block_verifier: Arc<dyn BlockVerifier>,
258    /// Charges faults for fetched payloads that fail verification: the author,
259    /// when the payload is provably theirs, and the peer that served it.
260    misbehavior_store: Arc<MisbehaviorStore>,
261    inflight_transactions_map: Arc<InflightTransactionsMap>,
262}
263
264impl<C: NetworkClient, D: CoreThreadDispatcher> TransactionsSynchronizer<C, D> {
265    /// Starts the transactions synchronizer, which is responsible for fetching
266    /// transactions from other authorities and managing transaction
267    /// synchronization tasks.
268    pub fn start(
269        network_client: Arc<C>,
270        context: Arc<Context>,
271        core_dispatcher: Arc<D>,
272        dag_state: Arc<RwLock<DagState>>,
273        block_verifier: Arc<dyn BlockVerifier>,
274    ) -> Arc<TransactionsSynchronizerHandle> {
275        let misbehavior_store = dag_state.read().misbehavior_store().clone();
276        let synchronizer = Arc::new(Self {
277            context,
278            network_client,
279            core_dispatcher,
280            dag_state,
281            block_verifier,
282            misbehavior_store,
283            inflight_transactions_map: Arc::default(),
284        });
285
286        let (live_fetch_sender, live_fetch_receiver) = channel(
287            "consensus_transactions_synchronizer_live_fetches",
288            LIVE_FETCH_TRANSACTIONS_CONCURRENCY,
289        );
290        let mut tasks = JoinSet::new();
291        let live_fetcher = synchronizer.clone().live_fetcher(live_fetch_receiver);
292        tasks.spawn(monitored_future!(live_fetcher));
293        tasks.spawn(monitored_future!(synchronizer.run()));
294
295        Arc::new(TransactionsSynchronizerHandle {
296            live_fetch_requests: live_fetch_sender,
297            tasks: tokio::sync::Mutex::new(tasks),
298        })
299    }
300
301    /// Runs the periodic scheduler: on every tick, while fewer than
302    /// `PERIODIC_FETCH_TRANSACTIONS_CONCURRENCY` periodic fetches are
303    /// running, asks the core for the missing transactions and fetches them.
304    /// Returns when the core shuts down.
305    #[cfg_attr(test, tracing::instrument(skip_all, name = "", fields(authority = %self.context.own_index
306    )))]
307    async fn run(self: Arc<Self>) {
308        let mut tasks = JoinSet::new();
309        let scheduler_timeout = sleep_until(Instant::now() + TRANSACTIONS_SYNCHRONIZER_TIMEOUT);
310        tokio::pin!(scheduler_timeout);
311
312        loop {
313            tokio::select! {
314                Some(result) = tasks.join_next(), if !tasks.is_empty() => {
315                    resume_if_panicked(result);
316                },
317                () = &mut scheduler_timeout => {
318                    let mut next_tick = TRANSACTIONS_SYNCHRONIZER_TIMEOUT;
319                    if tasks.len() < PERIODIC_FETCH_TRANSACTIONS_CONCURRENCY {
320                        let missing_transactions = match self.missing_transactions().await {
321                            Ok(missing_transactions) => missing_transactions,
322                            Err(err) => {
323                                debug!("Core is shutting down, transactions synchronizer is shutting down: {err:?}");
324                                return;
325                            }
326                        };
327                        if !missing_transactions.is_empty() {
328                            // Retry sooner while there is something to fetch.
329                            next_tick /= 2;
330                            let fetch = self.clone().periodic_fetch(missing_transactions);
331                            tasks.spawn(monitored_future!(fetch));
332                        }
333                    }
334                    scheduler_timeout.as_mut().reset(Instant::now() + next_tick);
335                }
336            }
337        }
338    }

340    /// Fetches the missing transactions found by one periodic tick.
341    async fn periodic_fetch(
342        self: Arc<Self>,
343        missing_transactions: BTreeMap<TransactionRef, BTreeSet<AuthorityIndex>>,
344    ) {
345        let _scope = monitored_scope("FetchMissingTransactionsScheduler");
346        fail_point_async!("consensus-delay");
347        let metrics = &self.context.metrics.node_metrics;
348        let _inflight = GaugeGuard::acquire(&metrics.transactions_synchronizer_periodic_inflight);
349        self.fetch_from_authorities(missing_transactions, SyncMethod::Periodic)
350            .await;
351    }

353    /// Serves the live fetch requests, at most
354    /// `LIVE_FETCH_TRANSACTIONS_CONCURRENCY` at a time. Returns once the
355    /// handle is dropped.
356    async fn live_fetcher(
357        self: Arc<Self>,
358        mut receiver: Receiver<BTreeMap<TransactionRef, BTreeSet<AuthorityIndex>>>,
359    ) {
360        // Dropped with this task on shutdown, which aborts the running fetches.
361        let mut tasks = JoinSet::new();
362        loop {
363            tokio::select! {
364                request = receiver.recv(), if tasks.len() < LIVE_FETCH_TRANSACTIONS_CONCURRENCY => {
365                    let Some(missing_transactions) = request else {
366                        info!("Live fetcher task will now abort.");
367                        return;
368                    };
369                    let synchronizer = self.clone();
370                    tasks.spawn(async move {
371                        synchronizer
372                            .fetch_from_authorities(missing_transactions, SyncMethod::Live)
373                            .await;
374                    });
375                },
376                Some(result) = tasks.join_next(), if !tasks.is_empty() => {
377                    resume_if_panicked(result);
378                },
379            }
380        }
381    }

383    /// Asks the core for the missing transactions and publishes the gap to
384    /// the earliest unavailable one and the per-authority counts, so they
385    /// read zero once nothing is missing. Fails only when the core is
386    /// shutting down; a ref of the wrong kind skips this tick.
387    async fn missing_transactions(
388        &self,
389    ) -> ConsensusResult<BTreeMap<TransactionRef, BTreeSet<AuthorityIndex>>> {
390        let missing_transactions = self
391            .core_dispatcher
392            .get_missing_transaction_data()
393            .await
394            .map_err(|_err| ConsensusError::Shutdown)?;
395
396        let metrics = &self.context.metrics.node_metrics;
397        let accepted_round = self.dag_state.read().highest_accepted_round();
398        let earliest_unavailable_round = missing_transactions
399            .first_key_value()
400            .map(|(tx_ref, _)| tx_ref.round)
401            .unwrap_or(accepted_round);
402        metrics
403            .gap_to_unavailable_transactions
404            .set(accepted_round.saturating_sub(earliest_unavailable_round) as i64);
405
406        let mut missing_per_authority = vec![0u64; self.context.committee.size()];
407        for tx_ref in missing_transactions.keys() {
408            missing_per_authority[tx_ref.author] += 1;
409        }
410        for (missing, (_, authority)) in missing_per_authority
411            .into_iter()
412            .zip(self.context.committee.authorities())
413        {
414            let hostname = authority.hostname.as_str();
415            metrics
416                .transactions_synchronizer_missing_transactions_by_authority
417                .with_label_values(&[hostname])
418                .inc_by(missing);
419            metrics
420                .transactions_synchronizer_current_missing_transactions_by_authority
421                .with_label_values(&[hostname])
422                .set(missing as i64);
423        }
424        Ok(missing_transactions)
425    }

427    /// Fetches the missing transactions from the authorities that acknowledged
428    /// them. Up to `MAX_ASSIGNED_AUTHORITIES_PER_TRANSACTION_FETCH` peers are
429    /// asked concurrently, each for the transactions it could lock; a
430    /// transaction that `MAX_AUTHORITIES_TO_FETCH_PER_TRANSACTION` peers are
431    /// already fetching is skipped. Every response feeds the peer's
432    /// responsiveness ranking.
433    async fn fetch_from_authorities(
434        self: &Arc<Self>,
435        missing_transactions: BTreeMap<TransactionRef, BTreeSet<AuthorityIndex>>,
436        sync_method: SyncMethod,
437    ) {
438        let context = &self.context;
439        // Build a mapping from authority -> set of transaction refs it has
440        // acknowledged. A committed payload was acknowledged by a quorum, so
441        // when no acknowledger is known, every peer is asked for it.
442        let committee: BTreeSet<AuthorityIndex> = context
443            .committee
444            .authorities()
445            .map(|(index, _)| index)
446            .collect();
447        let mut transaction_refs_by_authority: BTreeMap<AuthorityIndex, BTreeSet<TransactionRef>> =
448            BTreeMap::new();
449        for (tx_ref, authorities) in &missing_transactions {
450            let authorities = if authorities.is_empty() {
451                &committee
452            } else {
453                authorities
454            };
455            for authority in authorities {
456                if *authority != context.own_index {
457                    transaction_refs_by_authority
458                        .entry(*authority)
459                        .or_default()
460                        .insert(*tx_ref);
461                }
462            }
463        }
464
465        // Responsive peers are tried first when ranking is enabled. Without
466        // ranking the order is uniformly random, or the committee order under
467        // test so that scenarios stay deterministic.
468        let mut rng = StdRng::from_rng(&mut rng());
469        let mut order: Vec<AuthorityIndex> =
470            transaction_refs_by_authority.keys().copied().collect();
471        if context.parameters.enable_peer_responsiveness_ranking {
472            context.peer_responsiveness.prioritize(
473                DataSource::TransactionSynchronizer,
474                &mut order,
475                &mut rng,
476            );
477        } else {
478            #[cfg(not(test))]
479            order.shuffle(&mut rng);
480        }
481
482        let mut request_futures = FuturesUnordered::new();
483        for authority in order {
484            let authority_transaction_refs = transaction_refs_by_authority
485                .remove(&authority)
486                .expect("order should be a permutation of the candidate set");
487            let Some(transactions_guard) = self.inflight_transactions_map.lock_transactions(
488                authority_transaction_refs,
489                authority,
490                context
491                    .parameters
492                    .max_transactions_per_transaction_sync_fetch,
493                sync_method,
494            ) else {
495                continue;
496            };
497            request_futures.push(async move {
498                let result = self.fetch_from_authority(transactions_guard).await;
499                (authority, result)
500            });
501            if request_futures.len() == MAX_ASSIGNED_AUTHORITIES_PER_TRANSACTION_FETCH {
502                break;
503            }
504        }
505
506        // The recorded latency spans fetch and verification. A successful
507        // fetch records that latency scaled up by the fraction of requested
508        // transactions it returned, so unrelated or partial responses cannot
509        // improve a peer's rank; the scaled sample is capped at the failure
510        // penalty, so a partial delivery never records worse than a failed
511        // fetch. An empty response, an error or a timeout demote the peer to
512        // a timeout-scale latency, ordering it behind healthy peers in later
513        // selections without dropping it from the set.
514        let record_failure = |peer| {
515            context.peer_responsiveness.record_failure_with_timeout(
516                DataSource::TransactionSynchronizer,
517                peer,
518                FETCH_REQUEST_TIMEOUT,
519            )
520        };
521        while let Some((peer, result)) = request_futures.next().await {
522            match result {
523                Ok(stats) if stats.matched_requested > 0 => {
524                    let latency = stats
525                        .latency
526                        .mul_f64(shortfall_factor(stats.requested, stats.matched_requested))
527                        .min(FETCH_REQUEST_TIMEOUT);
528                    context.peer_responsiveness.record_success(
529                        DataSource::TransactionSynchronizer,
530                        peer,
531                        latency,
532                    );
533                }
534                Ok(_) => record_failure(peer),
535                Err(err) => {
536                    record_failure(peer);
537                    warn!(
538                        "[{}] Error when fetching and processing transactions from authority {peer}: {err}",
539                        sync_method.as_str(),
540                    );
541                }
542            }
543        }
544    }

546    /// Fetches the locked transactions from their peer, verifies them and
547    /// hands them to the core. The reported latency spans fetch and
548    /// verification, consistent with the commit syncer.
549    async fn fetch_from_authority(
550        self: &Arc<Self>,
551        transactions_guard: TransactionsGuard,
552    ) -> ConsensusResult<FetchStats> {
553        let peer = transactions_guard.peer;
554        let sync_method = transactions_guard.sync_method;
555        let peer_hostname = &self.context.committee.authority(peer).hostname;
556        let requested = transactions_guard.transactions_refs.len();
557        debug!(
558            "[{}] Syncing {requested} missing committed transactions from authority {peer} {peer_hostname}",
559            sync_method.as_str(),
560        );
561
562        let started = Instant::now();
563        let serialized_transactions = self
564            .fetch_transactions_request(peer, &transactions_guard.transactions_refs, sync_method)
565            .await?;
566        debug!(
567            "Transactions from {requested} blocks requested, fetched from {} blocks",
568            serialized_transactions.len()
569        );
570        let matched_requested = self
571            .process_fetched_transactions(serialized_transactions, transactions_guard)
572            .await?;
573
574        Ok(FetchStats {
575            latency: started.elapsed(),
576            requested,
577            matched_requested,
578        })
579    }

581    /// Requests the transactions from `peer` and records the outcome.
582    async fn fetch_transactions_request(
583        &self,
584        peer: AuthorityIndex,
585        transactions_refs: &BTreeSet<TransactionRef>,
586        sync_method: SyncMethod,
587    ) -> ConsensusResult<Vec<Bytes>> {
588        let metrics = &self.context.metrics.node_metrics;
589        let _inflight = GaugeGuard::acquire(&metrics.transactions_synchronizer_inflight_requests);
590        let peer_hostname = self.context.committee.authority(peer).hostname.as_str();
591
592        let start_time = Instant::now();
593        let result = timeout(
594            FETCH_REQUEST_TIMEOUT,
595            self.network_client.fetch_transactions(
596                peer,
597                transactions_refs.iter().copied().collect(),
598                FETCH_REQUEST_TIMEOUT,
599            ),
600        )
601        .await;
602
603        fail_point_async!("consensus-delay");
604
605        let fetch_duration = start_time.elapsed().as_secs_f64();
606        metrics
607            .transactions_synchronizer_fetch_latency
608            .observe(fetch_duration);
609        metrics
610            .transactions_synchronizer_fetch_latency_by_peer
611            .with_label_values(&[peer_hostname, sync_method.as_str()])
612            .observe(fetch_duration);
613
614        let record_failure = |reason: &str| {
615            metrics
616                .transactions_synchronizer_failure_by_peer
617                .with_label_values(&[peer_hostname, sync_method.as_str(), reason])
618                .inc();
619        };
620        match result {
621            Ok(Ok(serialized_transactions)) => {
622                metrics
623                    .transactions_synchronizer_success_by_peer
624                    .with_label_values(&[peer_hostname, sync_method.as_str()])
625                    .inc();
626                Ok(serialized_transactions)
627            }
628            Ok(Err(err)) => {
629                self.misbehavior_store.record_fetch_fault(peer, &err);
630                record_failure(err.name());
631                Err(err)
632            }
633            Err(elapsed) => {
634                record_failure("timeout");
635                Err(ConsensusError::NetworkRequestTimeout(elapsed.to_string()))
636            }
637        }
638    }

640    /// Verifies the fetched payloads and hands them to the core; the locked
641    /// transactions are released only once the core accepted them. Returns
642    /// how many requested transactions were delivered. Fails if the response
643    /// does not verify or the core is shutting down.
644    async fn process_fetched_transactions(
645        self: &Arc<Self>,
646        serialized_transactions_vec: Vec<Bytes>,
647        transactions_guard: TransactionsGuard,
648    ) -> ConsensusResult<usize> {
649        let peer = transactions_guard.peer;
650        let sync_method = transactions_guard.sync_method;
651        let metrics = &self.context.metrics.node_metrics;
652        let _timer = metrics
653            .scope_processing_time
654            .with_label_values(&["Synchronizer::process_fetched_transactions"])
655            .start_timer();
656        // Ensure that all the returned transactions do not go over the total max
657        // allowed returned transactions
658        if serialized_transactions_vec.len() > transactions_guard.transactions_refs.len() {
659            self.misbehavior_store
660                .record_faulty_transactions(None, [peer]);
661            return Err(ConsensusError::TooManyFetchedTransactionsReturned(peer));
662        }
663
664        // Deserialization, commitment checks and the transaction batch
665        // verification run on the blocking pool.
666        let transactions = spawn_blocking({
667            let synchronizer = self.clone();
668            let requested_transactions_refs = transactions_guard.transactions_refs.clone();
669            move || {
670                synchronizer.verify_fetched_transactions(
671                    serialized_transactions_vec,
672                    &requested_transactions_refs,
673                    peer,
674                )
675            }
676        })
677        .await??;
678
679        let peer_hostname = self.context.committee.authority(peer).hostname.as_str();
680        metrics
681            .transactions_synchronizer_fetched_transactions_by_peer
682            .with_label_values(&[peer_hostname, sync_method.as_str()])
683            .inc_by(transactions.len() as u64);
684        for transactions in &transactions {
685            let block_hostname = &self
686                .context
687                .committee
688                .authority(transactions.author())
689                .hostname;
690            metrics
691                .transactions_synchronizer_fetched_transactions_by_authority
692                .with_label_values(&[block_hostname.as_str(), sync_method.as_str()])
693                .inc();
694        }
695
696        let matched_requested = transactions.len();
697        self.core_dispatcher
698            .add_transactions(transactions, DataSource::TransactionSynchronizer)
699            .await
700            .map_err(|_| ConsensusError::Shutdown)?;
701        drop(transactions_guard);
702
703        Ok(matched_requested)
704    }
705
706    /// Deserializes the fetched payloads, checks each against the commitment
707    /// in its transaction reference and runs the same validity checks as the
708    /// block-bundle route. Records metrics and misbehavior for a failure.
709    fn verify_fetched_transactions(
710        &self,
711        serialized_transactions_vec: Vec<Bytes>,
712        requested_transactions_refs: &BTreeSet<TransactionRef>,
713        peer_index: AuthorityIndex,
714    ) -> ConsensusResult<Vec<CommitmentVerifiedTransactions>> {
715        let context = &self.context;
716        let misbehavior_store = &self.misbehavior_store;
717        let metrics = &context.metrics.node_metrics;
718        let peer_hostname = &context.committee.authority(peer_index).hostname;
719
720        let mut serialized_transactions_map: BTreeMap<TransactionRef, Bytes> = BTreeMap::new();
721        for serialized_transaction_bytes in &serialized_transactions_vec {
722            let serialized_transactions: SerializedTransactionsV2 =
723                bcs::from_bytes(serialized_transaction_bytes)
724                    .inspect_err(|_| {
725                        misbehavior_store.record_faulty_transactions(None, [peer_index])
726                    })
727                    .map_err(ConsensusError::MalformedTransactions)?;
728            let committed_transaction_ref = serialized_transactions.transaction_ref;
729            // The commitment check below only proves each payload matches
730            // its own claimed ref; it does not tie the ref to anything we
731            // asked for. Reject a ref outside the requested set so a peer
732            // cannot serve correctly-committed transactions we never
733            // requested (which need not correspond to any real header).
734            if !requested_transactions_refs.contains(&committed_transaction_ref) {
735                misbehavior_store.record_faulty_transactions(None, [peer_index]);
736                return Err(ConsensusError::UnrequestedTransactionFetched {
737                    peer: peer_index,
738                    transaction_ref: serialized_transactions.transaction_ref,
739                });
740            }
741            serialized_transactions_map.insert(
742                committed_transaction_ref,
743                serialized_transactions.serialized_transactions,
744            );
745        }
746
747        let transactions: Vec<_> = match verify_transactions_commitments(
748            context,
749            misbehavior_store,
750            peer_index,
751            serialized_transactions_map,
752        ) {
753            Ok(transactions) => transactions.into_values().collect(),
754            Err(err) => {
755                metrics
756                    .invalid_transactions
757                    .with_label_values(&[
758                        peer_hostname.as_str(),
759                        "transaction_synchronizer",
760                        err.name(),
761                    ])
762                    .inc();
763                return Err(err);
764            }
765        };
766
767        // The commitment check above only proves the fetched bytes match what
768        // the author committed to; it does not enforce the per-transaction
769        // limits or the application-level `verify_batch` checks the direct
770        // block-bundle route applies before a payload can be acknowledged.
771        // Run the same checks here so a payload violating them can't be
772        // acknowledged and become committable via this route either.
773        for verified_transactions in &transactions {
774            if let Err(err) = self
775                .block_verifier
776                .verify_transactions_validity(verified_transactions)
777            {
778                let author = verified_transactions.author();
779                metrics
780                    .invalid_transactions
781                    .with_label_values(&[
782                        peer_hostname.as_str(),
783                        "transaction_synchronizer",
784                        err.name(),
785                    ])
786                    .inc();
787                // The recomputed commitment (checked above) ties this payload to
788                // the author's signed transactions_commitment, so the invalid
789                // payload is provably the author's. `peer_index` served a full
790                // payload it could have verified before relaying, so it is also
791                // charged an unprovable fault.
792                misbehavior_store.record_faulty_transactions(
793                    Some(Slot::new(verified_transactions.round(), author)),
794                    [peer_index],
795                );
796                return Err(err);
797            }
798        }
799        Ok(transactions)
800    }
801}

803/// Re-raises a panic from a fetch task; a cancelled task is expected on
804/// shutdown.
805fn resume_if_panicked(result: Result<(), JoinError>) {
806    if let Err(err) = result {
807        if err.is_panic() {
808            std::panic::resume_unwind(err.into_panic());
809        }
810    }

1288    #[tokio::test]
1289    async fn inflight_transactions_map_with_active_requests() {
1290        telemetry_subscribers::init_for_testing();
1291
1292        // GIVEN
1293        let map = Arc::new(InflightTransactionsMap::default());
1294        let sync_method = SyncMethod::Periodic;
1295
1296        let context = Context::new_for_test(10).0;
1297        let max_transactions = context
1298            .parameters
1299            .max_transactions_per_transaction_sync_fetch;
1300        let missing_transactions_refs = [
1301            (1, AuthorityIndex::new_for_test(0)),
1302            (10, AuthorityIndex::new_for_test(0)),
1303            (12, AuthorityIndex::new_for_test(3)),
1304            (15, AuthorityIndex::new_for_test(2)),
1305        ]
1306        .into_iter()
1307        .map(|(round, author)| TransactionRef {
1308            round,
1309            author,
1310            transactions_commitment: TransactionsCommitment::MIN,
1311        })
1312        .collect::<BTreeSet<_>>();
1313        // We keep the guards so that drops happen at the end
1314        let mut all_guards: Vec<TransactionsGuard> = Vec::new();
1315
1316        // Try to acquire the transaction locks for authorities
1317        // 0..MAX_AUTHORITIES_TO_FETCH_PER_TRANSACTION
1318        for i in 0..=MAX_AUTHORITIES_TO_FETCH_PER_TRANSACTION {
1319            let authority = AuthorityIndex::new_for_test(i as u8);
1320
1321            let guard = map.lock_transactions(
1322                missing_transactions_refs.clone(),
1323                authority,
1324                max_transactions,
1325                sync_method,
1326            );
1327
1328            if i == MAX_AUTHORITIES_TO_FETCH_PER_TRANSACTION {
1329                // Trying to acquire for authority MAX_AUTHORITIES_TO_FETCH_PER_TRANSACTION will
1330                // fail - as we have maxed out the number of allowed peers for
1331                // each transaction
1332                assert!(guard.is_none());
1333                break;
1334            }
1335            let guard = guard.expect("Guard should be created");
1336            assert_eq!(guard.transactions_refs.len(), 4);
1337
1338            all_guards.push(guard);
1339
1340            // trying to acquire any of them again for the *same* authority should not
1341            // succeed
1342            let guard = map.lock_transactions(
1343                missing_transactions_refs.clone(),
1344                authority,
1345                max_transactions,
1346                sync_method,
1347            );
1348            assert!(guard.is_none());
1349        }
1350
1351        // Explicitly drop the guard of authority 0 (the first we stored) and try for
1352        // authority MAX_AUTHORITIES_TO_FETCH_PER_TRANSACTION again - it will now
1353        // succeed because one slot per transaction got freed
1354        drop(all_guards.remove(0));
1355
1356        let guard = map.lock_transactions(
1357            missing_transactions_refs.clone(),
1358            AuthorityIndex::new_for_test(MAX_AUTHORITIES_TO_FETCH_PER_TRANSACTION as u8),
1359            max_transactions,
1360            sync_method,
1361        );
1362        let guard = guard.expect("Guard should be successfully acquired");
1363        assert_eq!(guard.transactions_refs, missing_transactions_refs);
1364
1365        // Dropping all guards should unlock all transaction refs
1366        drop(guard);
1367        drop(all_guards);
1368
1369        assert_eq!(map.num_of_locked_transactions(), 0);
1370        assert!(
1371            map.inner
1372                .lock()
1373                .active_requests
1374                .values()
1375                .all(|&count| count == 0)
1376        );
1377
1378        // A peer holding `MAX_CONCURRENT_REQUESTS_PER_AUTHORITY` requests for a
1379        // sync method gets no more, even for transactions nobody is fetching.
1380        let peer = AuthorityIndex::new_for_test(0);
1381        let lock_one =
1382            || map.lock_transactions(missing_transactions_refs.clone(), peer, 1, sync_method);
1383        let _held: Vec<_> = (0..MAX_CONCURRENT_REQUESTS_PER_AUTHORITY)
1384            .map(|_| lock_one().expect("the peer has a free request slot"))
1385            .collect();
1386        assert!(lock_one().is_none());
1387    }
