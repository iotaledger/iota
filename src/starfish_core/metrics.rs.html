76/// A pair of counters exposed as `<name>_sum` / `<name>_count` — the same
77/// series a histogram of that name would produce, minus the buckets. Use it
78/// instead of a histogram when only the average
79/// (`rate(<name>_sum) / rate(<name>_count)`) is consumed, so the per-bucket
80/// series cost is avoided while existing average queries keep working
81/// unchanged.
82#[derive(Clone)]
83pub(crate) struct SumCount {
84    sum: Counter,
85    count: IntCounter,
86}
87
88impl SumCount {
89    fn register(name: &str, help: &str, registry: &Registry, level: MetricLevel) -> Self {
90        Self {
91            sum: register_counter_with_registry!(format!("{name}_sum"), help, registry; level)
92                .unwrap(),
93            count: register_int_counter_with_registry!(
94                format!("{name}_count"),
95                help,
96                registry;
97                level
98            )
99            .unwrap(),
100        }
101    }
102
103    pub(crate) fn observe(&self, value: f64) {
104        self.sum.inc_by(value);
105        self.count.inc();
106    }
107
108    /// Observes the elapsed time in seconds when the returned guard drops.
109    #[must_use]
110    pub(crate) fn start_timer(&self) -> SumCountTimer {
111        SumCountTimer {
112            pair: self.clone(),
113            start: Instant::now(),
114        }
115    }
116}

118pub(crate) struct SumCountTimer {
119    pair: SumCount,

123impl Drop for SumCountTimer {
124    fn drop(&mut self) {
125        self.pair.observe(self.start.elapsed().as_secs_f64());
