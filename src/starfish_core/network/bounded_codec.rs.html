4//! A prost codec that refuses a message carrying more entries than its type
5//! allows, before decoding the entries past that count.
6//!
7//! Protobuf writes no count ahead of a repeated field: each entry is a field of
8//! its own on the wire. An empty `bytes` entry is two bytes there and a 32-byte
9//! `Bytes` once decoded, so the message size limit alone lets a peer make the
10//! decoder build 16 times the message in `Bytes`.
11
12use std::marker::PhantomData;
13
14use bytes::{Buf, BufMut};
15use prost::{
16    DecodeError, Message,
17    encoding::{DecodeContext, WireType},
18};
19use tonic::{
20    Status,
21    codec::{BufferSettings, Codec, DecodeBuf, Decoder},
22};
23use tonic_prost::{ProstCodec, ProstEncoder};
24
25/// A message whose decoding refuses more than `MAX_ENTRIES` entries.
26pub(crate) trait EntryLimit {
27    /// Maximum number of entries one encoded message may contain. Each element
28    /// of a `Vec<Bytes>` field counts as one entry, and a whole `Vec<u32>` field
29    /// counts as one, because protobuf encodes the numbers together.
30    const MAX_ENTRIES: usize;
31    /// Message of the status the decoder returns when it refuses a message of
32    /// this type: `TOO_MANY_ENTRIES_IN_REQUEST` or
33    /// `TOO_MANY_ENTRIES_IN_RESPONSE`.
34    const REFUSAL: &'static str;
35}
36
37/// Refusal message for a request carrying more entries than its type allows.
38pub(crate) const TOO_MANY_ENTRIES_IN_REQUEST: &str =
39    "request carries more entries than its type allows";
40
41/// Refusal message for a response carrying more entries than its type allows.
42pub(crate) const TOO_MANY_ENTRIES_IN_RESPONSE: &str =
43    "response carries more entries than its type allows";
44
45/// `MAX_ENTRIES` of a request whose fields are all numbers. Decoding keeps at
46/// most one `u32` per byte of the request, so at the default 1 MiB request size
47/// limit it builds at most 4 MiB.
48pub(crate) const NO_ENTRY_LIMIT: usize = usize::MAX;
49
50/// A prost codec whose decoder refuses a message carrying more entries than
51/// its type allows.
52#[derive(Debug, Clone)]
53pub(crate) struct BoundedProstCodec<T, U> {
54    _pd: PhantomData<(T, U)>,
55}
56
57impl<T, U> Default for BoundedProstCodec<T, U> {
58    fn default() -> Self {
59        Self { _pd: PhantomData }
60    }
61}
62
63impl<T, U> Codec for BoundedProstCodec<T, U>
64where
65    T: Message + Send + 'static,
66    U: Message + Default + EntryLimit + Send + 'static,
67{
68    type Encode = T;
69    type Decode = U;
70
71    type Encoder = ProstEncoder<T>;
72    type Decoder = BoundedProstDecoder<U>;
73
74    fn encoder(&mut self) -> Self::Encoder {
75        ProstCodec::<T, U>::raw_encoder(BufferSettings::default())
76    }
77
78    fn decoder(&mut self) -> Self::Decoder {
79        BoundedProstDecoder { _pd: PhantomData }
80    }
81}

83/// Decodes `U`, refusing a message carrying more than `U::MAX_ENTRIES`
84/// entries.
85#[derive(Debug)]
86pub(crate) struct BoundedProstDecoder<U> {
87    _pd: PhantomData<U>,
88}
89
90impl<U: Message + Default + EntryLimit> Decoder for BoundedProstDecoder<U> {
91    type Item = U;
92    type Error = Status;
93
94    fn decode(&mut self, buf: &mut DecodeBuf<'_>) -> Result<Option<U>, Status> {
95        decode_bounded(buf).map(Some)
96    }
97}
98
99fn decode_bounded<U: Message + Default + EntryLimit>(buf: impl Buf) -> Result<U, Status> {
100    let mut bounded = Bounded::<U>::default();
101    // Parse errors map to INTERNAL, as in tonic-prost.
102    bounded
103        .merge(buf)
104        .map_err(|e| Status::internal(e.to_string()))?;
105    if bounded.entries > U::MAX_ENTRIES {
106        // OUT_OF_RANGE, as tonic answers a message over its size limit.
107        return Err(Status::out_of_range(U::REFUSAL));
108    }
109    Ok(bounded.message)
110}

112/// `M` decoded up to its first `M::MAX_ENTRIES` entries; an entry past them
113/// drops the rest of the message unread.
114#[derive(Debug, Default)]
115struct Bounded<M> {
116    message: M,
117    entries: usize,
118}
119
120impl<M: Message + EntryLimit> Message for Bounded<M> {
121    fn encode_raw(&self, buf: &mut impl BufMut) {
122        self.message.encode_raw(buf);
123    }
124
125    fn merge_field(
126        &mut self,
127        tag: u32,
128        wire_type: WireType,
129        buf: &mut impl Buf,
130        ctx: DecodeContext,
131    ) -> Result<(), DecodeError> {
132        self.entries += 1;
133        if self.entries > M::MAX_ENTRIES {
134            buf.advance(buf.remaining());
135            return Ok(());
136        }
137        self.message.merge_field(tag, wire_type, buf, ctx)
138    }
139
140    fn encoded_len(&self) -> usize {
141        self.message.encoded_len()
142    }
143
144    fn clear(&mut self) {
145        self.message.clear();
146        self.entries = 0;
147    }
148}

150#[cfg(test)]
151mod tests {
152    use bytes::Bytes;
153
154    use super::*;
155
156    #[derive(Clone, PartialEq, prost::Message)]
157    struct Entries {
158        #[prost(bytes = "bytes", repeated, tag = "1")]
159        entries: Vec<Bytes>,
160    }
161
162    impl EntryLimit for Entries {
163        const MAX_ENTRIES: usize = 4;
164        const REFUSAL: &'static str = TOO_MANY_ENTRIES_IN_RESPONSE;
165    }
166
167    fn encoded(entries: Vec<Bytes>) -> Bytes {
168        Entries { entries }.encode_to_vec().into()
169    }
170
171    #[test]
172    fn a_message_at_the_ceiling_decodes_whole() {
173        let entries: Vec<_> = (0..4u8).map(|i| Bytes::from(vec![i; 3])).collect();
174
175        let decoded: Entries = decode_bounded(encoded(entries.clone())).unwrap();
176
177        assert_eq!(decoded.entries, entries);
178    }
179
180    #[test]
181    fn a_message_past_the_ceiling_is_refused() {
182        let status = decode_bounded::<Entries>(encoded(vec![Bytes::new(); 5])).unwrap_err();
183
184        assert_eq!(status.code(), tonic::Code::OutOfRange);
185        assert_eq!(status.message(), TOO_MANY_ENTRIES_IN_RESPONSE);
186    }
187
188    /// However many entries a message carries, decoding builds at most the
189    /// ceiling's worth and reads no further.
190    #[test]
191    fn decoding_stops_at_the_ceiling() {
192        let mut buf = encoded(vec![Bytes::new(); 1 << 16]);
193        let mut bounded = Bounded::<Entries>::default();
194
195        bounded.merge(&mut buf).unwrap();
196
197        assert_eq!(bounded.message.entries.len(), 4);
198        assert_eq!(bounded.entries, 5);
199        assert!(!buf.has_remaining());
200    }
201
202    /// A malformed message is a parse error, not a refusal.
203    #[test]
204    fn a_truncated_entry_is_not_a_refusal() {
205        // Field 1 declares five bytes and carries none.
206        let status = decode_bounded::<Entries>(Bytes::from_static(&[0x0a, 0x05])).unwrap_err();
207
208        assert_eq!(status.code(), tonic::Code::Internal);
209        assert_ne!(status.message(), TOO_MANY_ENTRIES_IN_RESPONSE);
210    }
211}
