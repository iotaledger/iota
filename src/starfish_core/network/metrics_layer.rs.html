5//! Tower layer adapters that allow specifying callbacks for request and
6//! response handling can be implemented for different networking stacks.
7
8use std::sync::Arc;
9
10use prometheus_filtered::HistogramTimer;
11
12use super::metrics::NetworkRouteMetrics;
13
14/// Length of the prefix gRPC puts in front of every message: a compression
15/// flag byte followed by the big-endian `u32` payload length.
16const GRPC_MESSAGE_PREFIX_LEN: usize = 5;
17
18#[derive(Clone)]
19pub(crate) struct MetricsCallbackMaker {
20    metrics: Arc<NetworkRouteMetrics>,
21    /// Size in bytes above which a response message is counted as excessively
22    /// large.
23    excessive_message_size: usize,
24}
25
26impl MetricsCallbackMaker {
27    pub(crate) fn new(metrics: Arc<NetworkRouteMetrics>, excessive_message_size: usize) -> Self {
28        Self {
29            metrics,
30            excessive_message_size,
31        }
32    }
33
34    // Update request metrics. And create a callback that should be called on
35    // response.
36    pub(crate) fn handle_request(&self, route: &'static str) -> MetricsResponseCallback {
37        self.metrics.requests.with_label_values(&[route]).inc();
38        self.metrics
39            .inflight_requests
40            .with_label_values(&[route])
41            .inc();
42        let timer = self
43            .metrics
44            .request_latency
45            .with_label_values(&[route])
46            .start_timer();
47
48        MetricsResponseCallback {
49            metrics: self.metrics.clone(),
50            timer,
51            route,
52            excessive_message_size: self.excessive_message_size,
53            messages: GrpcMessageParser::default(),
54        }
55    }
56}
57
58pub(crate) struct MetricsResponseCallback {
59    metrics: Arc<NetworkRouteMetrics>,
60    // The timer is held on to and "observed" once dropped
61    #[expect(unused)]
62    timer: HistogramTimer,
63    route: &'static str,
64    excessive_message_size: usize,
65    /// Splits the response body into gRPC messages.
66    messages: GrpcMessageParser,
67}
68
69impl MetricsResponseCallback {
70    /// Records a failed response status. Sizes are recorded per message as the
71    /// body streams through `on_chunk`.
72    pub(crate) fn on_response(&mut self, error_type: Option<&str>) {
73        if let Some(err) = error_type {
74            self.metrics
75                .errors
76                .with_label_values(&[self.route, err])
77                .inc();
78        }
79    }
80
81    pub(crate) fn on_error<E>(&mut self, _error: &E) {
82        self.metrics
83            .errors
84            .with_label_values(&[self.route, "unknown"])
85            .inc();
86    }
87
88    /// Records the wire size of every message that completes inside `chunk`.
89    pub(crate) fn on_chunk(&mut self, chunk: &[u8]) {
90        let metrics = &self.metrics;
91        let route = self.route;
92        let excessive_message_size = self.excessive_message_size;
93        self.messages.feed(chunk, |message_size| {
94            metrics
95                .response_size
96                .with_label_values(&[route])
97                .observe(message_size as f64);
98            if message_size > excessive_message_size {
99                metrics
100                    .excessive_size_responses
101                    .with_label_values(&[route])
102                    .inc();
103            }
104        });
105    }
106
107    /// Counts a body that ended inside a message as a `truncated` error.
108    pub(crate) fn on_end_of_stream(&mut self) {
109        if self.messages.finish() {
110            self.metrics
111                .errors
112                .with_label_values(&[self.route, "truncated"])
113                .inc();
114        }
115    }
116}
117
118impl Drop for MetricsResponseCallback {
119    fn drop(&mut self) {
120        self.metrics
121            .inflight_requests
122            .with_label_values(&[self.route])
123            .dec();
124    }
125}
126
127/// Splits a gRPC body into its length-prefixed messages and reports the wire
128/// size of each: the prefix plus the payload, which is compressed when the
129/// channel compresses.
130///
131/// Body chunks carry no message boundaries of their own: a chunk can hold
132/// several messages, part of one, or even part of a prefix.
133enum GrpcMessageParser {
134    /// Reading the prefix; `len` bytes of it have arrived.
135    Prefix {
136        bytes: [u8; GRPC_MESSAGE_PREFIX_LEN],
137        len: usize,
138    },
139    /// Reading the payload of a `size`-byte message; `remaining` payload
140    /// bytes are still to come.
141    Payload { remaining: usize, size: usize },
142}
143
144impl Default for GrpcMessageParser {
145    fn default() -> Self {
146        Self::Prefix {
147            bytes: [0; GRPC_MESSAGE_PREFIX_LEN],
148            len: 0,
149        }
150    }
151}
152
153impl GrpcMessageParser {
154    /// Consumes `chunk`, calling `on_message` with the wire size of every
155    /// message that completes in it.
156    fn feed(&mut self, mut chunk: &[u8], mut on_message: impl FnMut(usize)) {
157        while !chunk.is_empty() {
158            match self {
159                Self::Prefix { bytes, len } => {
160                    let take = (GRPC_MESSAGE_PREFIX_LEN - *len).min(chunk.len());
161                    bytes[*len..*len + take].copy_from_slice(&chunk[..take]);
162                    *len += take;
163                    chunk = &chunk[take..];
164                    if *len < GRPC_MESSAGE_PREFIX_LEN {
165                        continue;
166                    }
167                    let payload_len =
168                        u32::from_be_bytes([bytes[1], bytes[2], bytes[3], bytes[4]]) as usize;
169                    // The length is peer-supplied; saturate rather than trust it.
170                    let size = GRPC_MESSAGE_PREFIX_LEN.saturating_add(payload_len);
171                    if payload_len == 0 {
172                        on_message(size);
173                        *self = Self::default();
174                    } else {
175                        *self = Self::Payload {
176                            remaining: payload_len,
177                            size,
178                        };
179                    }
180                }
181                Self::Payload { remaining, size } => {
182                    let take = (*remaining).min(chunk.len());
183                    *remaining -= take;
184                    chunk = &chunk[take..];
185                    if *remaining == 0 {
186                        on_message(*size);
187                        *self = Self::default();
188                    }
189                }
190            }
191        }
192    }
193
194    /// Resets the parser, returning whether the body ended inside a message.
195    fn finish(&mut self) -> bool {
196        let mid_message = !matches!(self, Self::Prefix { len: 0, .. });
197        *self = Self::default();
198        mid_message
199    }
200}

202#[cfg(test)]
203mod tests {
204    use prometheus_filtered::Registry;
205
206    use super::*;
207
208    const ROUTE: &str = "route";
209    const EXCESSIVE_MESSAGE_SIZE: usize = 64;
210
211    fn callback() -> (Arc<NetworkRouteMetrics>, MetricsResponseCallback) {
212        let metrics = Arc::new(NetworkRouteMetrics::new("test", &Registry::new()));
213        let callback = MetricsCallbackMaker::new(metrics.clone(), EXCESSIVE_MESSAGE_SIZE)
214            .handle_request(ROUTE);
215        (metrics, callback)
216    }
217
218    /// A gRPC message with an uncompressed payload of `payload_len` bytes.
219    fn message(payload_len: usize) -> Vec<u8> {
220        let mut message = vec![0];
221        message.extend((payload_len as u32).to_be_bytes());
222        message.extend(std::iter::repeat_n(0xAB, payload_len));
223        message
224    }
225
226    fn sizes(metrics: &NetworkRouteMetrics) -> (u64, f64) {
227        let histogram = metrics.response_size.with_label_values(&[ROUTE]);
228        (histogram.get_sample_count(), histogram.get_sample_sum())
229    }
230
231    fn excessive(metrics: &NetworkRouteMetrics) -> u64 {
232        metrics
233            .excessive_size_responses
234            .with_label_values(&[ROUTE])
235            .get()
236    }
237
238    fn truncated(metrics: &NetworkRouteMetrics) -> u64 {
239        metrics
240            .errors
241            .with_label_values(&[ROUTE, "truncated"])
242            .get()
243    }
244
245    #[test]
246    fn message_split_across_chunks_is_one_sample() {
247        let (metrics, mut callback) = callback();
248        let message = message(100);
249
250        // Prefix split at byte 3, payload split in two.
251        callback.on_chunk(&message[..3]);
252        callback.on_chunk(&message[3..40]);
253        assert_eq!(sizes(&metrics), (0, 0.0));
254        callback.on_chunk(&message[40..]);
255        callback.on_end_of_stream();
256
257        assert_eq!(sizes(&metrics), (1, 105.0));
258        assert_eq!(excessive(&metrics), 1);
259        assert_eq!(truncated(&metrics), 0);
260    }
261
262    #[test]
263    fn several_messages_in_one_chunk_are_separate_samples() {
264        let (metrics, mut callback) = callback();
265        let mut chunk = message(10);
266        chunk.extend(message(0));
267        chunk.extend(message(20));
268
269        callback.on_chunk(&chunk);
270        callback.on_end_of_stream();
271
272        assert_eq!(sizes(&metrics), (3, 45.0));
273        assert_eq!(excessive(&metrics), 0);
274        assert_eq!(truncated(&metrics), 0);
275    }
276
277    #[test]
278    fn body_ending_inside_a_message_is_truncated_once() {
279        let (metrics, mut callback) = callback();
280        let message = message(100);
281
282        callback.on_chunk(&message[..50]);
283        callback.on_end_of_stream();
284        // Trailers and the final poll both end the stream.
285        callback.on_end_of_stream();
286
287        assert_eq!(sizes(&metrics), (0, 0.0));
288        assert_eq!(truncated(&metrics), 1);
289    }
