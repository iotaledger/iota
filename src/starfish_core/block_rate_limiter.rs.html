4use std::time::Duration;
5
6/// Rate limiter for own block proposals (GCRA / token-bucket): sustained
7/// production of at most one block per `min_block_delay`, with bursts of up to
8/// `block_rate_window / min_block_delay` back-to-back blocks after idle
9/// periods. A burst capacity of 1 degenerates to a fixed minimum delay
10/// between consecutive blocks.
11///
12/// State is a single instant: the time at which an ideal schedule emitting
13/// exactly one block per interval would emit the next block. Idle time lets it
14/// fall back toward `now` (accruing budget); each proposal advances it by one
15/// interval.
16pub(crate) struct BlockRateLimiter {
17    /// UTC ms at which the next block would be emitted under an ideal schedule
18    /// of one block per `interval_ms` (the GCRA theoretical arrival time). It
19    /// runs ahead of `now` after recent proposals and falls back toward `now`
20    /// while idle; the gap below `now` is the unspent burst budget.
21    next_block_ms: u64,
22    /// Sustained spacing between blocks (`min_block_delay`), in ms.
23    interval_ms: u64,
24    /// Maximum number of back-to-back blocks allowed after an idle period.
25    burst: u64,
26}
27
28impl BlockRateLimiter {
29    pub(crate) fn new(min_block_delay: Duration, burst: u64) -> Self {
30        Self {
31            next_block_ms: 0,
32            interval_ms: min_block_delay.as_millis().max(1) as u64,
33            burst: burst.max(1),
34        }
35    }
36
37    /// Whether a proposal at `now_ms` fits the rate envelope.
38    pub(crate) fn is_conforming(&self, now_ms: u64) -> bool {
39        self.next_block_ms.saturating_sub(now_ms) <= (self.burst - 1) * self.interval_ms
40    }
41
42    /// Records a proposal at `now_ms`. Called for every own block, including
43    /// forced proposals that bypass the conformance check; the cap bounds
44    /// their overdraft so the next non-forced proposal waits at most one
45    /// interval after the last block.
46    pub(crate) fn record(&mut self, now_ms: u64) {
47        self.next_block_ms = (self.next_block_ms.max(now_ms) + self.interval_ms)
48            .min(now_ms + self.burst * self.interval_ms);
49    }
50}

52#[cfg(test)]
53mod tests {
54    use std::time::Duration;
55
56    use starfish_config::Parameters;
57
58    use crate::block_rate_limiter::BlockRateLimiter;
59
60    /// Drive the tests from the default consensus parameters rather than magic
61    /// numbers, so they track the production config (interval and burst).
62    fn interval_ms_and_burst() -> (u64, u64) {
63        let params = Parameters::default();
64        (
65            params.min_block_delay.as_millis() as u64,
66            params.block_rate_burst(),
67        )
68    }
69
70    fn limiter() -> BlockRateLimiter {
71        let params = Parameters::default();
72        BlockRateLimiter::new(params.min_block_delay, params.block_rate_burst())
73    }
74
75    #[test]
76    fn burst_after_idle_then_reject() {
77        let (interval_ms, burst) = interval_ms_and_burst();
78        let mut l = limiter();
79        let now = 1_000_000;
80        // Exactly `burst` back-to-back proposals conform, the next does not.
81        for _ in 0..burst {
82            assert!(l.is_conforming(now));
83            l.record(now);
84        }
85        assert!(!l.is_conforming(now));
86        // Budget for one more block regenerates after one interval.
87        assert!(!l.is_conforming(now + interval_ms - 1));
88        assert!(l.is_conforming(now + interval_ms));
89    }
90
91    #[test]
92    fn sustained_rate_is_one_per_interval() {
93        let (interval_ms, burst) = interval_ms_and_burst();
94        let mut l = limiter();
95        let start = 1_000_000;
96        // Drain the burst budget.
97        for _ in 0..burst {
98            l.record(start);
99        }
100        // Under continuous attempts, conforming instants are spaced exactly
101        // one interval apart.
102        let mut now = start;
103        for _ in 0..10 {
104            assert!(!l.is_conforming(now + interval_ms - 1));
105            now += interval_ms;
106            assert!(l.is_conforming(now));
107            l.record(now);
108        }
109    }
110
111    #[test]
112    fn burst_one_degenerates_to_fixed_delay() {
113        let (interval_ms, _) = interval_ms_and_burst();
114        let mut l = BlockRateLimiter::new(Duration::from_millis(interval_ms), 1);
115        let now = 1_000_000;
116        assert!(l.is_conforming(now));
117        l.record(now);
118        // Identical to the old rule: blocked until exactly one interval elapses.
119        assert!(!l.is_conforming(now));
120        assert!(!l.is_conforming(now + interval_ms - 1));
121        assert!(l.is_conforming(now + interval_ms));
122    }
123
124    #[test]
125    fn forced_overdraft_is_capped() {
126        let (interval_ms, burst) = interval_ms_and_burst();
127        let mut l = limiter();
128        let now = 1_000_000;
129        // Forced proposals record without a conformance check; the cap keeps
130        // the next non-forced eligibility within one interval of the last.
131        for _ in 0..(3 * burst) {
132            l.record(now);
133        }
134        assert!(!l.is_conforming(now));
135        assert!(l.is_conforming(now + interval_ms));
136    }
137
138    #[test]
139    fn seeding_by_replay_restores_budget_spent() {
140        let (interval_ms, burst) = interval_ms_and_burst();
141        let mut l = limiter();
142        let now = 1_000_000;
143        // Replaying k recent block timestamps leaves budget for `burst - k`
144        // more (no time elapses here, so no budget regenerates in between).
145        let k = burst / 2;
146        for _ in 0..k {
147            l.record(now);
148        }
149        for _ in 0..(burst - k) {
150            assert!(l.is_conforming(now));
151            l.record(now);
152        }
153        assert!(!l.is_conforming(now));
154        // Timestamps older than the whole window leave the budget full.
155        let mut l = limiter();
156        l.record(now - burst * interval_ms);
157        for _ in 0..burst {
158            assert!(l.is_conforming(now));
159            l.record(now);
160        }
161        assert!(!l.is_conforming(now));
162    }
