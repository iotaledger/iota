37#[derive(Default)]
38pub(crate) struct BaseCommitterOptions {
39    /// The offset used in the leader-election protocol. This is used by the
40    /// multi-committer to ensure that each [`BaseCommitter`] instance elects
41    /// a different leader.
42    pub leader_offset: u32,
43    /// The offset of the first wave. This is used by the pipelined committer to
44    /// ensure that each[`BaseCommitter`] instances operates on a different
45    /// view of the dag.
46    pub round_offset: u32,
47}
48
49/// The [`BaseCommitter`] contains the bare bone commit logic. Once
50/// instantiated, the method `try_direct_decide` and `try_indirect_decide` can
51/// be called at any time and any number of times (it is idempotent) to
52/// determine whether a leader can be committed or skipped.
53pub(crate) struct BaseCommitter {
54    /// The per-epoch configuration of this authority.
55    context: Arc<Context>,
56    /// The consensus leader schedule to be used to resolve the leader for a
57    /// given round.
58    leader_schedule: Arc<LeaderSchedule>,
59    /// In memory block store representing the dag state
60    dag_state: Arc<RwLock<DagState>>,
61    /// The options used by this committer
62    options: BaseCommitterOptions,
63    /// Certificate classification per (certifying block, leader block), keyed
64    /// by the certifying block's round. The answer depends only on accepted
65    /// blocks (the certifier's ancestors and, through them, the voting-round
66    /// blocks accepted before it), which never change, so an entry stays
67    /// correct until its round is evicted.
68    certificates: BTreeMap<Round, AHashMap<(BlockRef, BlockRef), (bool, bool)>>,
69}

94    /// Apply the direct decision rule to the specified leader to see whether we
95    /// can direct-commit or direct-skip it.
96    #[tracing::instrument(skip_all, fields(leader = %leader))]
97    pub fn try_direct_decide(&mut self, leader: Slot) -> LeaderStatus {
98        // Check whether the leader has enough blame. That is, whether there are 2f+1
99        // non-votes for that leader (which ensure there will never be a
100        // certificate for that leader).
101        let voting_round = leader.round + 1;
102        let voting_blocks = self
103            .dag_state
104            .read()
105            .get_block_headers_at_round_above_last_commit(voting_round);
106        if self.enough_leader_blame(voting_round, &voting_blocks, leader.authority) {
107            return LeaderStatus::Skip(leader);
108        }
109
110        // Check whether the leader(s) has enough support. That is, whether there are
111        // 2f+1 certificates over the leader. Note that there could be more than
112        // one leader block (created by Byzantine leaders).
113        let leader_blocks = self
114            .dag_state
115            .read()
116            .get_block_headers_at_slot_above_last_commit(leader);
117        if leader_blocks.is_empty() {
118            return LeaderStatus::Undecided(leader);
119        }
120
121        let wave = self.wave_number(leader.round);
122        let certifying_round = self.certifying_round(wave);
123        let decision_blocks = self
124            .dag_state
125            .read()
126            .get_block_headers_at_round_above_last_commit(certifying_round);
127
128        // Quickly reject if there isn't enough stake at the certifying round to
129        // support any leader block.
130        let total_stake: Stake = decision_blocks
131            .iter()
132            .map(|b| self.context.committee.stake(b.author()))
133            .sum();
134        if !self.context.committee.reached_quorum(total_stake) {
135            tracing::trace!(
136                "Not enough support for {leader}. Stake not enough: {total_stake} < {}",
137                self.context.committee.quorum_threshold()
138            );
139            return LeaderStatus::Undecided(leader);
140        }
141
142        let mut leaders_with_enough_support: Vec<_> = leader_blocks
143            .into_iter()
144            .filter_map(|leader_block| {
145                self.direct_decide_leader_block(&leader_block, &voting_blocks, &decision_blocks)
146                    .map(|(metastate, strong_voters)| {
147                        LeaderStatus::Commit(leader_block, metastate, strong_voters)
148                    })
149            })
150            .collect();
151
152        // There can be at most one leader with enough support for each round, otherwise
153        // it means the BFT assumption is broken.
154        if leaders_with_enough_support.len() > 1 {
155            panic!("[{self}] More than one certified block for {leader}")
156        }
157
158        leaders_with_enough_support
159            .pop()
160            .unwrap_or(LeaderStatus::Undecided(leader))
161    }

163    /// Apply the indirect rule; return `current` unchanged if no committed
164    /// anchor is reachable.
165    #[tracing::instrument(skip_all, fields(leader = %current))]
166    pub fn try_indirect_decide<'a>(
167        &mut self,
168        current: LeaderStatus,
169        leaders: impl Iterator<Item = &'a LeaderStatus>,
170    ) -> LeaderStatus {
171        let slot = current.slot();
172        let anchors = leaders.filter(|x| slot.round + WAVE_LENGTH <= x.round());
173        for anchor in anchors {
174            tracing::trace!("[{self}] Trying to indirect-decide {slot} using anchor {anchor}",);
175            match anchor {
176                LeaderStatus::Commit(anchor, _, _) => {
177                    return self.decide_leader_from_anchor(anchor, slot);
178                }
179                LeaderStatus::Skip(..) => (),
180                LeaderStatus::Undecided(..) => return current,
181            }
182        }
183        current
184    }

186    pub fn elect_leader(&self, round: Round) -> Option<Slot> {
187        let wave = self.wave_number(round);
188        tracing::trace!(
189            "elect_leader: round={}, wave={}, leader_round={}, leader_offset={}",
190            round,
191            wave,
192            self.leader_round(wave),
193            self.options.leader_offset
194        );
195        if self.leader_round(wave) != round {
196            return None;
197        }
198
199        Some(Slot::new(
200            round,
201            self.leader_schedule
202                .elect_leader(round, self.options.leader_offset),
203        ))
204    }

206    /// Return the leader round of the specified wave. The leader round is
207    /// always the first round of the wave. This takes into account round
208    /// offset for when pipelining is enabled.
209    pub(crate) fn leader_round(&self, wave_number: WaveNumber) -> Round {
210        (wave_number * WAVE_LENGTH) + self.options.round_offset
211    }
212
213    /// Return the certifying round of the specified wave. The certifying round
214    /// is always the last round of the wave. This takes into account round
215    /// offset for when pipelining is enabled.
216    pub(crate) fn certifying_round(&self, wave_number: WaveNumber) -> Round {
217        (wave_number * WAVE_LENGTH) + WAVE_LENGTH - 1 + self.options.round_offset
218    }
219
220    /// Return the wave in which the specified round belongs. This takes into
221    /// account the round offset for when pipelining is enabled.
222    pub(crate) fn wave_number(&self, round: Round) -> WaveNumber {
223        round.saturating_sub(self.options.round_offset) / WAVE_LENGTH
224    }

226    /// Check whether the specified block (`potential_vote`) is a vote for
227    /// the specified leader (`leader_block`).
228    fn is_vote(potential_vote: &VerifiedBlockHeader, leader_block: &VerifiedBlockHeader) -> bool {
229        potential_vote
230            .ancestors()
231            .contains(&leader_block.reference())
232    }
233
234    /// Return the refs of the given voting-round (`leader_block.round() + 1`)
235    /// blocks that directly include `leader_block` as an ancestor — i.e. the
236    /// set of votes for this specific leader block.
237    fn vote_refs_for_leader(
238        &self,
239        leader_block: &VerifiedBlockHeader,
240        voting_blocks: &[VerifiedBlockHeader],
241    ) -> HashSet<BlockRef> {
242        voting_blocks
243            .iter()
244            .filter(|voting_block| Self::is_vote(voting_block, leader_block))
245            .map(|voting_block| voting_block.reference())
246            .collect()
247    }

249    /// Check whether `potential_certificate` is a certificate for the leader
250    /// whose votes are captured in `vote_refs` — i.e. its ancestors include
251    /// a quorum of vote refs.
252    fn is_certificate(
253        &self,
254        potential_certificate: &VerifiedBlockHeader,
255        vote_refs: &HashSet<BlockRef>,
256    ) -> bool {
257        let mut votes_stake_aggregator = StakeAggregator::<QuorumThreshold>::new();
258        for reference in potential_certificate.ancestors() {
259            if vote_refs.contains(reference)
260                && votes_stake_aggregator.add(reference.author, &self.context.committee)
261            {
262                return true;
263            }
264        }
265        false
266    }

268    /// Decide the status of a target leader from the specified anchor. We
269    /// commit the target leader if it has a certified link to the anchor.
270    /// Otherwise, we skip the target leader.
271    fn decide_leader_from_anchor(
272        &mut self,
273        anchor: &VerifiedBlockHeader,
274        leader_slot: Slot,
275    ) -> LeaderStatus {
276        // Get the block(s) proposed by the leader. There could be more than one leader
277        // block in the slot from a Byzantine authority.
278        let leader_blocks = self
279            .dag_state
280            .read()
281            .get_block_headers_at_slot_above_last_commit(leader_slot);
282        if leader_blocks.is_empty() {
283            return LeaderStatus::Skip(leader_slot);
284        }
285
286        // TODO: Re-evaluate this check once we have a better way to handle/track
287        // byzantine authorities.
288        if leader_blocks.len() > 1 {
289            tracing::warn!(
290                "Multiple blocks found for leader slot {leader_slot}: {:?}",
291                leader_blocks
292            );
293        }
294
295        // Get all blocks that could be potential certificates for the target leader.
296        // These blocks are in the certifying round of the target leader and are
297        // linked to the anchor.
298        let wave = self.wave_number(leader_slot.round);
299        let certifying_round = self.certifying_round(wave);
300        let potential_certificates = self
301            .dag_state
302            .read()
303            .reachable_headers_at_round_above_last_commit(anchor, certifying_round);
304
305        // Use those potential certificates to determine which (if any) of the target
306        // leader blocks can be committed, and — when StarfishSpeed is enabled —
307        // classify the metastate in the same walk. When the flag is off we
308        // stick to the cheap regular-certificate-only check.
309        let starfish_speed = self.context.protocol_config.consensus_starfish_speed();
310        let voting_blocks = self
311            .dag_state
312            .read()
313            .get_block_headers_at_round_above_last_commit(leader_slot.round + 1);
314        let mut certified_leader_blocks: Vec<(
315            VerifiedBlockHeader,
316            Option<CommitMetastate>,
317            Vec<AuthorityIndex>,
318        )> = Vec::new();
319        for leader_block in leader_blocks {
320            let (any_cert, metastate, strong_voters) = if starfish_speed {
321                let (vote_refs, strong_vote_refs) =
322                    self.vote_and_strong_vote_refs_for_leader(&leader_block, &voting_blocks);
323                let mut any_cert = false;
324                let mut any_strong = false;
325                for potential_certificate in &potential_certificates {
326                    let (is_cert, is_strong) = self.classify_certificate_cached(
327                        potential_certificate,
328                        &leader_block,
329                        &vote_refs,
330                        &strong_vote_refs,
331                    );
332                    any_cert |= is_cert;
333                    any_strong |= is_strong;
334                    if any_cert && any_strong {
335                        break;
336                    }
337                }
338                let metastate = if any_cert {
339                    Some(if any_strong {
340                        CommitMetastate::Optimistic
341                    } else {
342                        CommitMetastate::Standard
343                    })
344                } else {
345                    None
346                };
347                let strong_voters = if any_strong {
348                    strong_vote_refs.iter().map(|r| r.author).collect()
349                } else {
350                    Vec::new()
351                };
352                (any_cert, metastate, strong_voters)
353            } else {
354                let vote_refs = self.vote_refs_for_leader(&leader_block, &voting_blocks);
355                let any_cert = potential_certificates.iter().any(|potential_certificate| {
356                    self.is_certificate(potential_certificate, &vote_refs)
357                });
358                (any_cert, None, Vec::new())
359            };
360            if any_cert {
361                certified_leader_blocks.push((leader_block, metastate, strong_voters));
362            }
363        }
364
365        // There can be at most one certified leader, otherwise it means the BFT
366        // assumption is broken.
367        if certified_leader_blocks.len() > 1 {
368            panic!("More than one certified block at wave {wave} from leader {leader_slot}")
369        }
370
371        match certified_leader_blocks.pop() {
372            Some((certified_leader_block, metastate, strong_voters)) => {
373                LeaderStatus::Commit(certified_leader_block, metastate, strong_voters)
374            }
375            None => LeaderStatus::Skip(leader_slot),
376        }
377    }

379    /// Check whether the specified leader has 2f+1 non-votes (blames) among
380    /// the given voting-round blocks to be directly skipped.
381    fn enough_leader_blame(
382        &self,
383        voting_round: Round,
384        voting_blocks: &[VerifiedBlockHeader],
385        leader: AuthorityIndex,
386    ) -> bool {
387        let mut blame_stake_aggregator = StakeAggregator::<QuorumThreshold>::new();
388        for voting_block in voting_blocks {
389            let voter = voting_block.reference().author;
390            if voting_block
391                .ancestors()
392                .iter()
393                .all(|ancestor| ancestor.author != leader)
394            {
395                tracing::trace!(
396                    "[{self}] {voting_block} is a blame for leader {}",
397                    Slot::new(voting_round - 1, leader)
398                );
399                if blame_stake_aggregator.add(voter, &self.context.committee) {
400                    return true;
401                }
402            } else {
403                tracing::trace!(
404                    "[{self}] {voting_block} is not a blame for leader {}",
405                    Slot::new(voting_round - 1, leader)
406                );
407            }
408        }
409        false
410    }

412    /// Apply the direct decision rule to one leader block using the given
413    /// voting-round (r+1) and certifying-round (r+2) blocks. Returns
414    /// `Some((metastate, strong_voters))` when 2f+1 certifying blocks carry a
415    /// certificate for the leader block, `None` otherwise.
416    ///
417    /// When StarfishSpeed is enabled, regular and strong certificates are
418    /// classified in a single walk over the certifying blocks, and the
419    /// metastate follows the direct rule: StrongQC quorum → `Optimistic`
420    /// (with the leader's strong-voter authorities, used by the linearizer
421    /// to seed the per-ref ack tracker), strong-blame quorum → `Standard`,
422    /// else `Pending`. When the flag is off the metastate is `None`.
423    fn direct_decide_leader_block(
424        &mut self,
425        leader_block: &VerifiedBlockHeader,
426        voting_blocks: &[VerifiedBlockHeader],
427        decision_blocks: &[VerifiedBlockHeader],
428    ) -> Option<(Option<CommitMetastate>, Vec<AuthorityIndex>)> {
429        if !self.context.protocol_config.consensus_starfish_speed() {
430            let vote_refs = self.vote_refs_for_leader(leader_block, voting_blocks);
431            let mut certificate_stake_aggregator = StakeAggregator::<QuorumThreshold>::new();
432            for decision_block in decision_blocks {
433                let authority = decision_block.reference().author;
434                if self.is_certificate(decision_block, &vote_refs)
435                    && certificate_stake_aggregator.add(authority, &self.context.committee)
436                {
437                    return Some((None, Vec::new()));
438                }
439            }
440            return None;
441        }
442
443        let (vote_refs, strong_vote_refs) =
444            self.vote_and_strong_vote_refs_for_leader(leader_block, voting_blocks);
445
446        let mut certificate_stake_aggregator = StakeAggregator::<QuorumThreshold>::new();
447        let mut strong_qc_stake_aggregator = StakeAggregator::<QuorumThreshold>::new();
448        let mut enough_support = false;
449        let mut strong_qc_quorum = false;
450        for decision_block in decision_blocks {
451            let (is_cert, is_strong) = self.classify_certificate_cached(
452                decision_block,
453                leader_block,
454                &vote_refs,
455                &strong_vote_refs,
456            );
457            let authority = decision_block.reference().author;
458            if is_cert && certificate_stake_aggregator.add(authority, &self.context.committee) {
459                enough_support = true;
460            }
461            if is_strong && strong_qc_stake_aggregator.add(authority, &self.context.committee) {
462                strong_qc_quorum = true;
463            }
464            if enough_support && strong_qc_quorum {
465                break;
466            }
467        }
468
469        if !enough_support {
470            return None;
471        }
472        if strong_qc_quorum {
473            let strong_voters = strong_vote_refs.iter().map(|r| r.author).collect();
474            return Some((Some(CommitMetastate::Optimistic), strong_voters));
475        }
476        if self.has_strong_blame_quorum(leader_block, voting_blocks) {
477            return Some((Some(CommitMetastate::Standard), Vec::new()));
478        }
479        Some((Some(CommitMetastate::Pending), Vec::new()))
480    }
481
482    /// `(vote_refs, strong_vote_refs)` for `leader_block` among the given
483    /// voting-round (r+1) blocks, built in a single pass. Strong votes are a
484    /// subset of votes; the strong-vote filter also requires the voter's
485    /// pinned leader (`strong_vote_leader`) to match `leader_block.author()`
486    /// — strong votes computed against a different leader (e.g. before a
487    /// schedule rotation) don't count toward this leader's quorum.
488    fn vote_and_strong_vote_refs_for_leader(
489        &self,
490        leader_block: &VerifiedBlockHeader,
491        voting_blocks: &[VerifiedBlockHeader],
492    ) -> (HashSet<BlockRef>, HashSet<BlockRef>) {
493        let mut vote_refs = HashSet::new();
494        let mut strong_vote_refs = HashSet::new();
495        for voter in voting_blocks {
496            if Self::is_vote(voter, leader_block) {
497                let r = voter.reference();
498                vote_refs.insert(r);
499                if voter.is_strong_vote_for(leader_block.author()) {
500                    strong_vote_refs.insert(r);
501                }
502            }
503        }
504        (vote_refs, strong_vote_refs)
505    }
506
507    /// `classify_certificate` memoized per (certifying block, leader block).
508    fn classify_certificate_cached(
509        &mut self,
510        potential_certificate: &VerifiedBlockHeader,
511        leader_block: &VerifiedBlockHeader,
512        vote_refs: &HashSet<BlockRef>,
513        strong_vote_refs: &HashSet<BlockRef>,
514    ) -> (bool, bool) {
515        let metrics = &self.context.metrics.node_metrics;
516        let key = (potential_certificate.reference(), leader_block.reference());
517        match self.certificates.entry(key.0.round).or_default().entry(key) {
518            Entry::Occupied(entry) => {
519                metrics.decision_cache_hits_total.inc();
520                *entry.get()
521            }
522            Entry::Vacant(entry) => {
523                metrics.decision_cache_misses_total.inc();
524                *entry.insert(Self::classify_certificate(
525                    &self.context.committee,
526                    potential_certificate,
527                    vote_refs,
528                    strong_vote_refs,
529                ))
530            }
531        }
532    }
533
534    /// Single-walk `(is_certificate, is_strong_certificate)` classifier.
535    /// `vote_refs`/`strong_vote_refs` are the pre-computed voter sets.
536    /// Invariant: `is_strong ⇒ is_certificate`.
537    fn classify_certificate(
538        committee: &Committee,
539        potential_certificate: &VerifiedBlockHeader,
540        vote_refs: &HashSet<BlockRef>,
541        strong_vote_refs: &HashSet<BlockRef>,
542    ) -> (bool, bool) {
543        let mut vote_agg = StakeAggregator::<QuorumThreshold>::new();
544        let mut strong_agg = StakeAggregator::<QuorumThreshold>::new();
545        let mut is_cert = false;
546        let mut is_strong = false;
547        for reference in potential_certificate.ancestors() {
548            if !is_cert
549                && vote_refs.contains(reference)
550                && vote_agg.add(reference.author, committee)
551            {
552                is_cert = true;
553            }
554            if !is_strong
555                && strong_vote_refs.contains(reference)
556                && strong_agg.add(reference.author, committee)
557            {
558                is_strong = true;
559            }
560            if is_cert && is_strong {
561                break;
562            }
563        }
564        (is_cert, is_strong)
565    }
566
567    /// Check whether 2f+1 of the given voting-round (r+1) blocks vote for
568    /// `leader_block`, pin their strong-blame to `leader_block.author()`, and
569    /// carry `is_strong_blame() == true`. Strong blames whose pinned leader
570    /// doesn't match are ignored — same misattribution rule as for strong
571    /// votes.
572    fn has_strong_blame_quorum(
573        &self,
574        leader_block: &VerifiedBlockHeader,
575        voting_blocks: &[VerifiedBlockHeader],
576    ) -> bool {
577        let mut strong_blame_stake_aggregator = StakeAggregator::<QuorumThreshold>::new();
578        for voting_block in voting_blocks {
579            if !voting_block.is_strong_blame_for(leader_block.author()) {
580                continue;
581            }
582            if !Self::is_vote(voting_block, leader_block) {
583                continue;
584            }
585            if strong_blame_stake_aggregator
586                .add(voting_block.reference().author, &self.context.committee)
587            {
588                return true;
589            }
590        }
591        false
592    }
593}
