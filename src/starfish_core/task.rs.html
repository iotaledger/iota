5use crate::error::{ConsensusError, ConsensusResult};
6
7/// Runs the closure on the blocking pool like `tokio::task::spawn_blocking`,
8/// but resolves to `ConsensusError::Shutdown` instead of a `JoinError` when
9/// the task is cancelled by runtime shutdown. A panic in the closure is
10/// resumed on the calling thread.
11pub(crate) async fn spawn_blocking<F, T>(f: F) -> ConsensusResult<T>
12where
13    F: FnOnce() -> T + Send + 'static,
14    T: Send + 'static,
15{
16    tokio::task::spawn_blocking(f)
17        .await
18        .map_err(spawn_blocking_join_error)
19}
20
21/// A blocking task's join error is either a panic, resumed here, or a
22/// cancellation, which only happens when the runtime is shutting down.
23fn spawn_blocking_join_error(e: tokio::task::JoinError) -> ConsensusError {
24    if e.is_panic() {
25        std::panic::resume_unwind(e.into_panic());
26    }
27    ConsensusError::Shutdown
28}

30#[cfg(test)]
31mod tests {
32    use super::*;
33
34    /// A blocking task racing with runtime shutdown is cancelled instead of
35    /// run, so its join handle resolves to a cancellation error.
36    #[tokio::test]
37    async fn cancelled_blocking_task_maps_to_shutdown() {
38        let runtime = tokio::runtime::Builder::new_current_thread()
39            .build()
40            .unwrap();
41        let handle = runtime.handle().clone();
42        runtime.shutdown_background();
43
44        let join_error = handle.spawn_blocking(|| ()).await.unwrap_err();
45        assert!(join_error.is_cancelled());
46        assert!(matches!(
47            spawn_blocking_join_error(join_error),
48            ConsensusError::Shutdown
49        ));
50    }
51
52    #[tokio::test]
53    #[should_panic(expected = "boom")]
54    async fn panic_in_blocking_task_is_resumed() {
55        let _ = spawn_blocking(|| panic!("boom")).await;
56    }
57}
