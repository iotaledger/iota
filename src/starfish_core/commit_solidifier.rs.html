17/// The `CommitSolidifier` is responsible for managing and handling
18/// the commit process for newly committed pending sub-dags. It ensures that
19/// sub-dags that are sent for execution are solid, i.e. all transactions
20/// included in the pending sub_dags are locally available and sub-dags are
21/// committed in order. The `CommitSolidifier` also tracks the highest committed
22/// index and maintains a buffer for pending sub-dags for which either the
23/// transactions are not yet available or the previous sub-dags are missing
24/// transactions and have not been output yet.
25///
26/// # Fields
27/// - `dag_state`: Shared state of the DAG.
28/// - `pending_subdags`: Buffer for sub-dags waiting to be committed.
29/// - `last_committed_index`: Tracks the highest committed sub-dag index.
30///
31/// # Usage
32/// The `CommitSolidifier` is used to process newly committed sub-dags by
33/// retrieving information about potentially missing transactions.
34pub(crate) struct CommitSolidifier {
35    dag_state: Arc<RwLock<DagState>>,
36    // Buffer for pending subdags, keyed by commit_ref.index for order
37    pending_subdags: BTreeMap<u32, PendingSubDag>,
38    // The highest solid commit_ref.index
39    last_solid_committed_index: CommitIndex,
40}
41
42impl CommitSolidifier {
43    pub(crate) fn new(dag_state: Arc<RwLock<DagState>>) -> Self {
44        // the last_solid_committed_index is set non-trivially during a recovery process
45        // before the first usage of try_commit method.
46        let last_solid_committed_index = 0;
47        Self {
48            dag_state,
49            pending_subdags: BTreeMap::new(),
50            last_solid_committed_index,
51        }
52    }
53
54    pub(crate) fn set_last_solid_committed_index(&mut self, index: CommitIndex) {
55        self.last_solid_committed_index = index;
56    }

58    /// Gets all missing transactions from pending subdags.
59    ///
60    /// # Returns
61    /// A `BTreeSet` of `GenericTransactionRef`s for which transactions are
62    /// missing.
63    pub(crate) fn get_missing_transaction_data(&self) -> BTreeSet<GenericTransactionRef> {
64        let mut missing = BTreeSet::new();
65        let dag_state = self.dag_state.read();
66
67        // Check all pending subdags for missing transactions
68        for subdag in self.pending_subdags.values() {
69            let exists = dag_state.contains_transactions(subdag.committed_transaction_refs.clone());
70            for (i, exists) in exists.iter().enumerate() {
71                if !exists {
72                    missing.insert(subdag.committed_transaction_refs[i]);
73                }
74            }
75        }
76        missing
77    }

79    /// Attempts to retrieve transactions included in the newly created commits.
80    /// Adds the PendingSubDag to the buffer if any transactions are missing and
81    /// outputs them once they are available.
82    ///
83    /// # Arguments
84    /// - `subdags`: A slice of `PendingSubDag` to be committed.
85    ///
86    /// # Returns
87    /// A tuple containing:
88    /// - `Vec<CommittedSubDag>`: Successfully committed sub-dags.
89    /// - `Vec<BlockRef>`: References to blocks with missing transactions
90    ///   preventing further commits.
91    pub(crate) fn try_get_solid_sub_dags(
92        &mut self,
93        subdags: &[PendingSubDag],
94    ) -> (Vec<CommittedSubDag>, Vec<GenericTransactionRef>) {
95        // Add new subdags to the buffer
96        for subdag in subdags {
97            self.pending_subdags
98                .entry(subdag.commit_ref.index)
99                .or_insert_with(|| subdag.clone());
100        }
101        let mut committed_subdags = Vec::new();
102        let mut last_solid_committed = self.last_solid_committed_index;
103        let mut missing = BTreeSet::new();
104
105        // Try to commit in order
106        loop {
107            let next_index = last_solid_committed + 1;
108            // If the next expected subdag is not in the buffer, we cannot commit anything
109            // further
110            let Some(pending_subdag) = self.pending_subdags.get(&next_index) else {
111                break;
112            };
113            match self.try_get_one_solid_sub_dag_internal(pending_subdag) {
114                Ok(committed_subdag) => {
115                    committed_subdags.push(committed_subdag);
116                    self.pending_subdags.remove(&next_index);
117                    last_solid_committed = next_index;
118                }
119                Err(missing_refs) => {
120                    // If we have missing refs, we cannot commit this subdag
121                    debug!(
122                        "Cannot create CommittedSubDag at index {}. Missing refs: {:?}",
123                        next_index, missing_refs
124                    );
125
126                    break; // Can't commit further until this one is ready
127                }
128            }
129        }
130
131        // Update with the latest solid_commit_refs which are used for commit syncer
132        if !committed_subdags.is_empty() {
133            let solid_commit_refs = committed_subdags
134                .iter()
135                .map(|subdag| subdag.commit_ref)
136                .collect();
137            self.dag_state
138                .write()
139                .update_pending_commit_votes(solid_commit_refs);
140        }
141
142        // Update last_committed_index
143        self.last_solid_committed_index = last_solid_committed;
144
145        // Only check for missing refs in the newly passed subdags that weren't
146        // processed yet
147        for subdag in subdags {
148            if subdag.commit_ref.index > self.last_solid_committed_index {
149                // Query dag_state directly for missing transactions
150                let dag_state_guard = self.dag_state.read();
151                let exists = dag_state_guard
152                    .contains_transactions(subdag.committed_transaction_refs.clone());
153                drop(dag_state_guard);
154                for (i, exists) in exists.iter().enumerate() {
155                    if !exists {
156                        let block_ref = subdag.committed_transaction_refs[i];
157                        if !missing.insert(block_ref) {
158                            // Transactions should only be committed by a single subdag, so
159                            // duplicates should never happen.
160                            panic!("Duplicate missing blockref detected: {block_ref:?}");
161                        }
162                    }
163                }
164            }
165        }
166
167        (committed_subdags, missing.into_iter().collect())
168    }
169
170    /// Internal method to retrieve all transactions committed to PendingSubDag
171    /// and form CommittedSubDAG if all of them are locally available
172    ///
173    /// # Arguments
174    /// - `subdag`: A reference to the `PendingSubDag` to be committed.
175    ///
176    /// # Returns
177    /// - `Ok(CommittedSubDag)`: If all required transactions are in DagState
178    /// - `Err(Vec<BlockRef>)`: If some transactions are missing.
179    fn try_get_one_solid_sub_dag_internal(
180        &self,
181        subdag: &PendingSubDag,
182    ) -> Result<CommittedSubDag, Vec<GenericTransactionRef>> {
183        let dag_state = self.dag_state.read();
184        // Get transactions and check if any are missing
185        let transaction_results =
186            dag_state.get_verified_transactions(&subdag.committed_transaction_refs);
187        let mut missing = Vec::new();
188        for (i, tx_opt) in transaction_results.iter().enumerate() {
189            if tx_opt.is_none() {
190                missing.push(subdag.committed_transaction_refs[i]);
191            }
192        }
193
194        if missing.is_empty() {
195            // All transactions exist, so we can create a CommittedSubDag
196            let transactions = transaction_results
197                .into_iter()
198                .map(|tx| tx.expect("Transaction must exist since we checked"))
199                .collect();
200
201            // Delayed subdags see current store state, not state at leader-round
202            // commit time. Safe: counts are absolute and consumers merge-max.
203            let misbehavior_counts = dag_state.misbehavior_store().snapshot_totals();
204            Ok(CommittedSubDag::new(
205                subdag.leader,
206                subdag.base.headers.clone(),
207                subdag.base.committed_header_refs.clone(),
208                transactions,
209                subdag.timestamp_ms,
210                subdag.commit_ref,
211                subdag.reputation_scores_desc.clone(),
212                misbehavior_counts,
213            ))
214        } else {
215            Err(missing)
216        }
217    }

269        /// Creates a selective DAG state that only contains transactions from
270        /// specified rounds
271        ///
272        /// # Arguments
273        /// * `included_rounds` - Vector of round numbers whose transactions
274        ///   should be included
275        /// * `excluded_transactions` - Vector of (round, block_index) pairs to
276        ///   exclude transactions from specific blocks
277        fn create_selective_dag_state(
278            &self,
279            included_rounds: Vec<u32>,
280            excluded_transactions: Vec<(u32, usize)>,
281        ) -> Arc<RwLock<DagState>> {
282            let selective_dag_state = Arc::new(RwLock::new(DagState::new(
283                self.context.clone(),
284                Arc::new(MemStore::new()),
285            )));
286
287            let mut state = selective_dag_state.write();
288
289            // Add genesis blocks if round 0 is included
290            if included_rounds.contains(&0) {
291                let genesis_blocks = genesis_blocks(&self.context);
292                for (i, block) in genesis_blocks.iter().enumerate() {
293                    state
294                        .accept_block_header(block.verified_block_header.clone(), DataSource::Test);
295                    if !excluded_transactions.contains(&(0, i)) {
296                        state.add_transactions(
297                            block.verified_transactions.clone(),
298                            DataSource::Test,
299                        );
300                    }
301                }
302            }
303
304            // Add blocks from specified rounds
305            for &round in &included_rounds {
306                if round == 0 {
307                    continue;
308                } // Genesis blocks already handled
309
310                let blocks = self.dag_builder.blocks(round..=round);
311                for (i, block) in blocks.iter().enumerate() {
312                    state
313                        .accept_block_header(block.verified_block_header.clone(), DataSource::Test);
314                    if !excluded_transactions.contains(&(round, i)) {
315                        state.add_transactions(
316                            block.verified_transactions.clone(),
317                            DataSource::Test,
318                        );
319                    }
320                }
321            }
322
323            drop(state);
324            selective_dag_state
325        }

378    #[derive(Clone)]
379    struct BlockSpec {

381        indices: Option<Vec<usize>>, // None means all blocks, Some(vec) means specific indices

396            }

408    impl SubDagBuilder {
409        fn new(setup: Arc<TestSetup>, index: u32) -> Self {
410            Self {
411                index,
412                leader_round: 0,
413                leader_index: 0,
414                block_specs: Vec::new(),
415                committed_refs: Vec::new(),
416                setup,
417            }
418        }
419
420        fn leader(mut self, round: u32, index: usize) -> Self {
421            self.leader_round = round;
422            self.leader_index = index;
423            self
424        }
425
426        fn with_blocks(mut self, specs: Vec<BlockSpec>) -> Self {
427            self.block_specs = specs;
428            self
429        }
430
431        fn with_committed_refs_from_round(mut self, round: u32) -> Self {
432            let refs = if round == 0 {
433                genesis_blocks(&self.setup.context)
434                    .iter()
435                    .map(|b| {
436                        GenericTransactionRef::TransactionRef(
437                            b.verified_block_header.transaction_ref(),
438                        )
439                    })
440                    .collect()
441            } else {
442                self.setup
443                    .dag_builder
444                    .block_headers(round..=round)
445                    .iter()
446                    .map(|b| GenericTransactionRef::TransactionRef(b.transaction_ref()))
447                    .collect()
448            };
449            self.committed_refs = refs;
450            self
451        }
452
453        fn with_committed_refs(mut self, refs: Vec<GenericTransactionRef>) -> Self {
454            self.committed_refs = refs;
455            self
456        }
457
458        fn build(self) -> PendingSubDag {
459            // Get leader block
460            let leader = if self.leader_round == 0 {
461                genesis_blocks(&self.setup.context)[self.leader_index].reference()
462            } else {
463                self.setup
464                    .dag_builder
465                    .block_headers(self.leader_round..=self.leader_round)[self.leader_index]
466                    .reference()
467            };
468
469            // Collect all blocks based on specs
470            let mut all_committed_block_headers = Vec::new();
471
472            for spec in &self.block_specs {
473                let headers = if spec.round == 0 {
474                    genesis_block_headers(&self.setup.context)
475                } else {
476                    self.setup
477                        .dag_builder
478                        .block_headers(spec.round..=spec.round)
479                };
480
481                match &spec.indices {
482                    None => all_committed_block_headers.extend(headers),
483                    Some(indices) => {
484                        if indices.is_empty() {
485                            // Special case: skip first
486                            all_committed_block_headers.extend(headers.into_iter().skip(1));
487                        } else {
488                            for &i in indices {
489                                if let Some(header) = headers.get(i) {
490                                    all_committed_block_headers.push(header.clone());
491                                }
492                            }
493                        }
494                    }
495                }
496            }
497
498            // Add a leader block if not already included
499            let leader_header = if self.leader_round == 0 {
500                genesis_blocks(&self.setup.context)[self.leader_index]
501                    .verified_block_header
502                    .clone()
503            } else {
504                self.setup
505                    .dag_builder
506                    .block_headers(self.leader_round..=self.leader_round)[self.leader_index]
507                    .clone()
508            };
509
510            if !all_committed_block_headers
511                .iter()
512                .any(|b| b.reference() == leader)
513            {
514                all_committed_block_headers.push(leader_header);
515            }
516
517            PendingSubDag::new(
518                leader,
519                all_committed_block_headers.clone(),
520                all_committed_block_headers
521                    .iter()
522                    .map(|h| h.reference())
523                    .collect(),
524                self.committed_refs,
525                123456,
526                CommitRef {
527                    index: self.index,
528                    digest: crate::commit::CommitDigest::MIN,
529                },
530                vec![],
531            )
532        }
533    }
