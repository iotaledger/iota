5//! CommitSyncer implements efficient synchronization of committed data.
6//!
7//! During the operation of a committee of authorities for consensus, one or
8//! more authorities can fall behind the quorum in their received and accepted
9//! blocks. This can happen due to network disruptions, host crash, or other
10//! reasons. Authorities fell behind need to catch up to the quorum to be able
11//! to vote on the latest leaders. So efficient synchronization is necessary
12//! to minimize the impact of temporary disruptions and maintain smooth
13//! operations of the network.
14//! CommitSyncer achieves efficient synchronization by relying on the following:
15//! when blocks are included in commits with >= 2f+1 certifiers by stake, these
16//! blocks must have passed verifications on some honest validators, so
17//! re-verifying them is unnecessary. In fact, the quorum certified commits
18//! themselves can be trusted to be sent to IOTA directly, but for simplicity
19//! this is not done. Blocks from trusted commits still go through Core and
20//! committer.
21//!
22//! Another way CommitSyncer improves the efficiency of synchronization is
23//! parallel fetching: commits have a simple dependency graph (linear), so it is
24//! easy to fetch ranges of commits in parallel.
25//!
26//! Commit synchronization is an expensive operation, involving transferring
27//! large amount of data via the network. And it is not on the critical path of
28//! block processing. So the heuristics for synchronization, including triggers
29//! and retries, should be chosen to favor throughput and efficient resource
30//! usage, over faster reactions.

90impl CommitSyncType {
91    pub(crate) fn commit_sync_batch_size(&self, context: &Context) -> u32 {
92        match self {
93            CommitSyncType::Fast => context.parameters.fast_commit_sync_batch_size,
94            CommitSyncType::Regular => context.parameters.commit_sync_batch_size,
95        }
96    }
97
98    /// Maximum number of commits a peer may return for `commit_range`. This is
99    /// the bound `verify_commits` enforces and the same bound the streaming
100    /// fetch loop applies before buffering. Fast sync extends past the
101    /// requested range end to reach a certifiable commit, so it accepts up to
102    /// twice the requested range; regular sync stays within it.
103    pub(crate) fn max_commits_per_response(&self, commit_range: &CommitRange) -> usize {
104        match self {
105            CommitSyncType::Fast => 2 * commit_range.size(),
106            CommitSyncType::Regular => commit_range.size(),
107        }
108    }
109
110    pub(crate) fn as_str(&self) -> &'static str {
111        match self {
112            CommitSyncType::Fast => "fast_commit_sync",
113            CommitSyncType::Regular => "commit_sync",
114        }
115    }
116
117    /// Fetch source under which peer responsiveness is tracked. The two sync
118    /// types are tracked separately because they call different endpoints: fast
119    /// sync asks one request for the commits and their transactions and keeps
120    /// only the prefix the peer covered, while regular sync fetches headers and
121    /// transactions in requests of their own and tolerates a peer that returns
122    /// no transactions. A peer that consistently fails one may serve the other
123    /// well.
124    pub(crate) fn data_source(&self) -> DataSource {
125        match self {
126            CommitSyncType::Fast => DataSource::FastCommitSyncer,
127            CommitSyncType::Regular => DataSource::CommitSyncer,
128        }
129    }
130
131    pub(crate) fn should_schedule(
132        &self,
133        gap: u32,
134        commit_sync_gap_threshold: u32,
135        fast_commit_sync_enabled: bool,
136    ) -> bool {
137        match self {
138            CommitSyncType::Fast => fast_commit_sync_enabled && gap > commit_sync_gap_threshold,
139            CommitSyncType::Regular => {
140                !fast_commit_sync_enabled || gap <= commit_sync_gap_threshold
141            }
142        }
143    }
144}

146/// Verifies that the fetched headers are exactly the requested ones, in any
147/// order. Requested refs must be distinct.
148pub(crate) fn verify_fetched_headers(
149    peer: AuthorityIndex,
150    request_block_refs: &[BlockRef],
151    serialized_block_headers: Vec<Bytes>,
152) -> ConsensusResult<Vec<VerifiedBlockHeader>> {
153    if serialized_block_headers.len() > request_block_refs.len() {
154        return Err(ConsensusError::TooManyFetchedHeadersReturned {
155            peer,
156            requested: request_block_refs.len(),
157            received: serialized_block_headers.len(),
158        });
159    }
160    if serialized_block_headers.len() < request_block_refs.len() {
161        return Err(ConsensusError::NotEnoughHeadersFetched {
162            peer,
163            requested: request_block_refs.len(),
164            received: serialized_block_headers.len(),
165        });
166    }
167    // Counts match, so consuming a distinct requested ref per header drains
168    // the set exactly when no header errors below.
169    let mut pending_refs: BTreeSet<BlockRef> = request_block_refs.iter().cloned().collect();
170    serialized_block_headers
171        .into_iter()
172        .map(|serialized| {
173            let header = VerifiedBlockHeader::new_from_bytes(serialized)?;
174            if !pending_refs.remove(&header.reference()) {
175                return Err(ConsensusError::UnexpectedBlockHeaderForCommit {
176                    peer,
177                    received: header.reference(),
178                });
179            }
180            Ok(header)
181        })
182        .collect()
183}

192    pub(crate) async fn stop(self) {
193        let _ = self.tx_shutdown.send(());
194        // Do not abort schedule task, which waits for fetches to shut down.
195        if let Err(e) = self.schedule_task.await {

225    /// Verifies the commits and also certifies them using the provided vote
226    /// blocks for the last commit. The method returns the trusted commits
227    /// and the verified voting block headers.

240            peer,
241            commit_range,
242            serialized_commits,
243            serialized_vote_blocks_headers,
244            max_commits,
245        )
246    }

248    /// Moves the peers that have voted for `commit_index` or later, other than
249    /// those in `excluded`, ahead of the rest, keeping the order within each
250    /// group. Does nothing unless
251    /// `enable_commit_sync_peer_selection_by_commit_votes` is set.
252    pub(crate) fn order_voters_first(
253        &self,
254        authorities: &mut [AuthorityIndex],
255        commit_index: CommitIndex,
256        excluded: &BTreeSet<AuthorityIndex>,
257    ) {
258        if !self
259            .context
260            .parameters
261            .enable_commit_sync_peer_selection_by_commit_votes
262        {
263            return;
264        }
265        authorities.sort_by_cached_key(|authority| {
266            excluded.contains(authority)
267                || !self
268                    .commit_vote_monitor
269                    .has_voted_for_commit(*authority, commit_index)
270        });
271    }

274/// Rejects a deserialized commit whose variant does not match the local
275/// protocol-flag configuration. The flag is uniform across the network within
276/// an epoch, so any mismatch implies either a malicious peer or a misconfigured
277/// upgrade path. Called on commits received over the network in
278/// `verify_commits`; recovery from local store is trusted and skips this check.
279pub(crate) fn check_commit_version_matches_flags(
280    commit: &Commit,
281    protocol_config: &iota_protocol_config::ProtocolConfig,
282) -> ConsensusResult<()> {
283    let starfish_speed = protocol_config.consensus_starfish_speed();
284    let variant_matches_flags = matches!(
285        (commit, starfish_speed),
286        (Commit::V2(_), false) | (Commit::V3(_), true)
287    );
288    if !variant_matches_flags {
289        let actual = match commit {
290            Commit::V1(_) => "V1",
291            Commit::V2(_) => "V2",
292            Commit::V3(_) => "V3",
293        };
294        return Err(ConsensusError::WrongCommitVersionForFlags {
295            actual,
296            starfish_speed,
297        });
298    }
299    Ok(())
300}

302/// Validates every `AuthorityIndex` carried by a fetched commit against the
303/// committee, so malformed indices are rejected at ingress with peer
304/// attribution instead of panicking later when commit content is indexed
305/// into per-authority state.
306fn verify_commit_authority_indices(
307    context: &Context,
308    peer: AuthorityIndex,
309    commit: &Commit,
310) -> ConsensusResult<()> {
311    let committee = &context.committee;
312    let check = |index: AuthorityIndex| -> ConsensusResult<()> {
313        if !committee.is_valid_index(index) {
314            return Err(ConsensusError::InvalidAuthorityIndexRequested {
315                index,
316                max: committee.size(),
317                peer,
318            });
319        }
320        Ok(())
321    };
322    check(commit.leader().author)?;
323    for block_ref in commit.block_headers() {
324        check(block_ref.author)?;
325    }
326    for transaction_ref in commit.committed_transactions() {
327        check(transaction_ref.author())?;
328    }
329    for (index, _) in commit.reputation_scores() {
330        check(*index)?;
331    }
332    Ok(())
333}

335/// Free-function form of `Inner::verify_commits`, taking only the inputs the
336/// verification actually uses (`Context` and `BlockVerifier`). Lets tests
337/// exercise the full deserialize-and-verify path without constructing a full
338/// `Inner<C>` fixture.
339pub(crate) fn verify_commits(
340    context: &Arc<Context>,
341    block_verifier: &dyn BlockVerifier,
342    misbehavior_store: &MisbehaviorStore,
343    peer: AuthorityIndex,
344    commit_range: CommitRange,
345    serialized_commits: Vec<Bytes>,
346    serialized_vote_blocks_headers: Vec<Bytes>,
347    max_commits: usize,
348) -> ConsensusResult<(Vec<TrustedCommit>, Vec<VerifiedBlockHeader>)> {
349    // Validate response size - peer should not return more than max_commits
350    if serialized_commits.len() > max_commits {
351        let e = ConsensusError::TooManyCommitsFromPeer {
352            peer,
353            count: serialized_commits.len() as CommitIndex,
354            limit: max_commits as CommitIndex,
355        };
356        misbehavior_store.record_fetch_fault(peer, &e);
357        return Err(e);
358    }
359
360    // One vote header per authority certifies a commit, but servers that do
361    // not dedup votes by author may legitimately serve a few more (e.g. an
362    // author re-including its vote in a block after crash recovery), so allow
363    // some multiplicity while still bounding signature verification work.
364    let max_vote_headers = context
365        .committee
366        .size()
367        .saturating_mul(MAX_COMMIT_VOTE_HEADERS_PER_AUTHORITY);
368    if serialized_vote_blocks_headers.len() > max_vote_headers {
369        let e = ConsensusError::TooManyCommitVoteHeaders {
370            peer,
371            count: serialized_vote_blocks_headers.len(),
372            limit: max_vote_headers,
373        };
374        misbehavior_store.record_fetch_fault(peer, &e);
375        return Err(e);
376    }
377
378    // Parse and verify commits.
379    let mut commits = Vec::new();
380    for serialized in &serialized_commits {
381        let commit: Commit =
382            bcs::from_bytes(serialized).map_err(ConsensusError::MalformedCommit)?;
383        check_commit_version_matches_flags(&commit, &context.protocol_config)?;
384        verify_commit_authority_indices(context, peer, &commit)?;
385        let digest = TrustedCommit::compute_digest(serialized);
386        if commits.is_empty() {
387            // start is inclusive, so first commit must be at the start index.
388            if commit.index() != commit_range.start() {
389                return Err(ConsensusError::UnexpectedStartCommit {
390                    peer,
391                    start: commit_range.start(),
392                    commit: Box::new(commit),
393                });
394            }
395        } else {
396            // Verify next commit increments index and references the previous digest.
397            let (last_commit_digest, last_commit): &(CommitDigest, Commit) =
398                commits.last().unwrap();
399            if commit.index() != last_commit.index() + 1
400                || &commit.previous_digest() != last_commit_digest
401            {
402                return Err(ConsensusError::UnexpectedCommitSequence {
403                    peer,
404                    prev_commit: Box::new(last_commit.clone()),
405                    curr_commit: Box::new(commit),
406                });
407            }
408        }
409        commits.push((digest, commit));
410    }
411    let Some((end_commit_digest, end_commit)) = commits.last() else {
412        return Err(ConsensusError::NoCommitReceived { peer });
413    };
414
415    // Parse and verify blocks. Then accumulate votes on the end commit.
416    let end_commit_ref = CommitRef::new(end_commit.index(), *end_commit_digest);
417    let mut stake_aggregator = StakeAggregator::<QuorumThreshold>::new();
418    let mut verified_voting_headers = Vec::new();
419    for serialized_block_header in serialized_vote_blocks_headers {
420        let signed_block_header: SignedBlockHeader = bcs::from_bytes(&serialized_block_header)
421            .map_err(ConsensusError::MalformedHeader)
422            .inspect_err(|e| {
423                // Author is unknown when deserialization fails — blame the peer.
424                misbehavior_store.record_faulty_block(peer, peer, e);
425            })?;
426        // The block signature needs to be verified.
427        if let Err(e) = block_verifier.verify(&signed_block_header) {
428            misbehavior_store.record_faulty_block(peer, signed_block_header.author(), &e);
429            return Err(e);
430        }
431        for vote in signed_block_header.commit_votes() {
432            if *vote == end_commit_ref {
433                stake_aggregator.add(signed_block_header.author(), &context.committee);
434            }
435        }
436        // Store the verified voting block header
437        verified_voting_headers.push(VerifiedBlockHeader::new_verified(
438            signed_block_header,
439            serialized_block_header,
440        ));
441    }
442
443    // Check if the end commit has enough votes.
444    if !stake_aggregator.reached_threshold(&context.committee) {
445        return Err(ConsensusError::NotEnoughCommitVotes {
446            stake: stake_aggregator.stake(),
447            peer,
448            commit: Box::new(end_commit.clone()),
449        });
450    }
451
452    let trusted_commits = commits
453        .into_iter()
454        .zip(serialized_commits)
455        .map(|((_d, c), s)| TrustedCommit::new_trusted(c, s))
456        .collect();
457    Ok((trusted_commits, verified_voting_headers))
458}

460/// Verifies each payload against the transactions commitment in its ref and
461/// returns the payloads keyed by transaction reference.
462pub(crate) fn verify_transactions_commitments(
463    context: &Arc<Context>,
464    peer: AuthorityIndex,
465    serialized_transactions: BTreeMap<TransactionRef, Bytes>,
466) -> ConsensusResult<BTreeMap<TransactionRef, CommitmentVerifiedTransactions>> {
467    let mut verified_transactions_map = BTreeMap::new();
468    let mut encoder = create_encoder(context);
469    let size_limit = serialized_transactions_size_limit(context);
470    for (transaction_ref, inner_serialized_transactions) in serialized_transactions {
471        // Range-check the peer-supplied author and round before any consumer
472        // indexes the committee by author.
473        if !context.committee.is_valid_index(transaction_ref.author) {
474            return Err(ConsensusError::InvalidAuthorityIndexRequested {
475                index: transaction_ref.author,
476                max: context.committee.size(),
477                peer,
478            });
479        }
480        if transaction_ref.round == GENESIS_ROUND {
481            return Err(ConsensusError::UnexpectedGenesisRequested { peer });
482        }
483        // Bound the peer-supplied payload before erasure-encoding it.
484        if inner_serialized_transactions.len() > size_limit {
485            return Err(ConsensusError::SerializedTransactionsTooLarge {
486                size: inner_serialized_transactions.len(),
487                limit: size_limit,
488            });
489        }
490        // Step 1: Verify that the transaction commitment matches.
491        if transaction_ref.transactions_commitment
492            != TransactionsCommitment::compute_transactions_commitment(
493                &inner_serialized_transactions,
494                context,
495                &mut encoder,
496            )?
497        {
498            return Err(ConsensusError::TransactionCommitmentFailure {
499                round: transaction_ref.round,
500                author: transaction_ref.author,
501                peer,
502            });
503        }
504
505        // Step 2: Deserialize the actual transactions vector.
506        let transactions: Vec<Transaction> = bcs::from_bytes(&inner_serialized_transactions)
507            .map_err(ConsensusError::MalformedTransactions)?;
508
509        // Step 3: Create a CommitmentVerifiedTransactions instance and insert into map
510        let verified_transactions = CommitmentVerifiedTransactions::new(
511            transactions,
512            transaction_ref,
513            None,
514            inner_serialized_transactions,
515        );
516
517        verified_transactions_map.insert(transaction_ref, verified_transactions);
518    }
519
520    Ok(verified_transactions_map)
521}

523/// How much of a requested commit range a fetch response actually delivered.
524/// Implemented by both syncers' payload types so [`fetch_loop`] can scale the
525/// latency it feeds to [`crate::peer_responsiveness::PeerResponsiveness`].
526pub(crate) trait FetchedCommits {
527    fn delivered_commits(&self) -> usize;
528}

530impl FetchedCommits for CertifiedCommits {

532        self.commits().len()
533    }

536/// Factor by which a peer's measured latency is inflated when it delivered less
537/// than the full requested range. A peer that answers a large request with a
538/// short prefix is as costly as a slow one, because the remainder has to be
539/// fetched again, so it must not rank as fast. Never below 1.0: fast sync may
540/// legitimately return more commits than requested.
541pub(crate) fn shortfall_factor(requested: usize, delivered: usize) -> f64 {
542    (requested as f64 / delivered.max(1) as f64).max(1.0)
543}

545/// Base timeout of one commit-sync request; every failed round adds one more
546/// of these, up to `MAX_FETCH_TIMEOUT_MULTIPLIER` times.
547#[cfg(not(test))]
548pub(crate) const FETCH_TIMEOUT: Duration = Duration::from_secs(10);
549#[cfg(test)]
550pub(crate) const FETCH_TIMEOUT: Duration = Duration::from_millis(500);
551/// Cap on the escalation: at the extreme a request waits 120 s.
552pub(crate) const MAX_FETCH_TIMEOUT_MULTIPLIER: u32 = 12;
553/// A whole fetch attempt may take this many request timeouts: the regular
554/// syncer pipelines several requests per attempt, the fast syncer sends one.
555pub(crate) const REGULAR_FETCH_ATTEMPT_MULTIPLIER: u32 = 4;
556pub(crate) const FAST_FETCH_ATTEMPT_MULTIPLIER: u32 = 2;
557/// The longest any commit syncer waits for one fetch attempt, 480 s in the
558/// node.
559pub(crate) fn max_fetch_attempt_timeout() -> Duration {
560    FETCH_TIMEOUT
561        * MAX_FETCH_TIMEOUT_MULTIPLIER
562        * REGULAR_FETCH_ATTEMPT_MULTIPLIER.max(FAST_FETCH_ATTEMPT_MULTIPLIER)
563}

565/// Generic fetch loop that retries fetching data from available authorities
566/// until a request succeeds. This is shared between RegularCommitSyncer and
567/// FastCommitSyncer.
568///
569/// # Type Parameters
570/// - `C`: Network client type
571/// - `T`: Fetched data type (CertifiedCommits for regular, (Vec<TrustedCommit>,
572///   Vec<CommittedSubDag>) for fast)
573/// - `F`: Fetch function type
574/// - `Fut`: Future returned by fetch function
575///
576/// # Parameters
577/// - `inner`: Shared context and dependencies
578/// - `commit_range`: The range of commits to fetch
579/// - `fetch_timeout_multiplier`: Multiplier for timeout calculation (4 for regular, 2 for fast)
580/// - `fetch_once_fn`: Implementation-specific fetch function
581///
582/// # Returns
583/// Tuple of (end_commit_index, fetched_data)
584#[cfg_attr(test, tracing::instrument(skip_all, fields(authority = %inner.context.own_index)))]
585pub(crate) async fn fetch_loop<C, T, F, Fut>(
586    inner: Arc<Inner<C>>,
587    commit_range: CommitRange,
588    fetch_timeout_multiplier: u32,
589    fetch_once_fn: F,
590) -> (CommitIndex, T)
591where
592    C: NetworkClient,
593    T: Send + FetchedCommits,
594    F: Fn(Arc<Inner<C>>, AuthorityIndex, CommitRange, Duration) -> Fut,
595    Fut: std::future::Future<Output = ConsensusResult<T>> + Send,
596{
597    // timeout * max number of targets should be reasonably small, so the
598    // system can adjust to slow network or large data sizes quickly.
599    const MAX_NUM_TARGETS: usize = 24;
600    let mut timeout_multiplier = 0;
601
602    // A failed attempt is reported at this fixed scale rather than at the
603    // escalating `fetch_timeout` below. The tracker keeps the worst latency it
604    // has seen for a peer and only decays it on success, so feeding the
605    // escalated value would leave a peer looking slow for many rounds after it
606    // recovered.
607    let failure_penalty = FETCH_TIMEOUT * fetch_timeout_multiplier;
608    let data_source = inner.sync_type.data_source();
609    let mut rng = StdRng::from_rng(&mut rng());
610
611    let _timer = inner
612        .context
613        .metrics
614        .node_metrics
615        .commit_sync_fetch_loop_latency
616        .start_timer();
617    info!(
618        "[{}] Starting to fetch commits in {commit_range:?} ...",
619        inner.sync_type.as_str()
620    );
621    loop {
622        // Attempt to fetch commits and blocks through min(committee size,
623        // MAX_NUM_TARGETS) peers.
624        let mut target_authorities = inner
625            .context
626            .committee
627            .authorities()
628            .filter_map(|(i, _)| {
629                if i != inner.context.own_index {
630                    Some(i)
631                } else {
632                    None
633                }
634            })
635            .collect_vec();
636        // Ranking runs before the truncation, so the peers actually tried are
637        // the best-ranked ones rather than a random subset of a committee
638        // larger than MAX_NUM_TARGETS. This changes which peers are reached in
639        // a round: a peer whose last attempt failed is ordered last and can
640        // fall outside the cut. That is safe here because a single peer serving
641        // the range is enough, one success clears the failed state, and the
642        // exploration fraction keeps drawing every peer - though on a committee
643        // much larger than MAX_NUM_TARGETS a given peer comes back slowly.
644        if inner.context.parameters.enable_peer_responsiveness_ranking {
645            inner.context.peer_responsiveness.prioritize(
646                data_source,
647                &mut target_authorities,
648                &mut rng,
649            );
650        } else {
651            #[cfg(not(test))]
652            target_authorities.shuffle(&mut rng);
653        }
654        // A peer that has voted for the end of the range has solidified every
655        // commit in it, so it is the one worth asking. A peer that has not may
656        // still serve the range: votes are only seen from blocks we received,
657        // and serving additionally needs the certifying headers in the peer's
658        // storage. So it is ordered behind, keeping the order chosen above
659        // within each group. Since the truncation below runs after this, a
660        // committee with more voters than MAX_NUM_TARGETS fills every round
661        // with voters, and a peer whose headers do not reach us is not asked
662        // until they do. Accepted: rounds retry forever over the up-to-24 peers
663        // that provably solidified the range, and any header from a
664        // behind-listed peer carrying a recent commit vote promotes it
665        // immediately.
666        inner.order_voters_first(
667            &mut target_authorities,
668            commit_range.end(),
669            &BTreeSet::new(),
670        );
671        target_authorities.truncate(MAX_NUM_TARGETS);
672        // Increase timeout multiplier for each loop until MAX_TIMEOUT_MULTIPLIER.
673        timeout_multiplier = (timeout_multiplier + 1).min(MAX_FETCH_TIMEOUT_MULTIPLIER);
674        let request_timeout = FETCH_TIMEOUT * timeout_multiplier;
675
676        let fetch_timeout = request_timeout * fetch_timeout_multiplier;
677        // Try fetching from the selected target authority.
678        for authority in target_authorities {
679            // Spans fetch and verification, so the cost of checking what a peer
680            // sent counts against it rather than only its time on the wire.
681            let started = Instant::now();
682            match tokio::time::timeout(
683                fetch_timeout,
684                fetch_once_fn(
685                    inner.clone(),
686                    authority,
687                    commit_range.clone(),
688                    request_timeout,
689                ),
690            )
691            .await
692            {
693                Ok(Ok(data)) => {
694                    // A failure is the ceiling of the scale: however short the
695                    // delivery, a peer that returned something never records
696                    // worse than one that failed.
697                    inner.context.peer_responsiveness.record_success(
698                        data_source,
699                        authority,
700                        started
701                            .elapsed()
702                            .mul_f64(shortfall_factor(
703                                commit_range.size(),
704                                data.delivered_commits(),
705                            ))
706                            .min(failure_penalty),
707                    );
708                    info!(
709                        "[{}] Finished fetching commits in {commit_range:?}",
710                        inner.sync_type.as_str()
711                    );
712                    return (commit_range.end(), data);
713                }
714                Ok(Err(e)) => {
715                    inner
716                        .context
717                        .peer_responsiveness
718                        .record_failure_with_timeout(data_source, authority, failure_penalty);
719                    let hostname = inner
720                        .context
721                        .committee
722                        .authority(authority)
723                        .hostname
724                        .clone();
725                    warn!(
726                        "[{}] Failed to fetch {commit_range:?} from {hostname}: {}",
727                        inner.sync_type.as_str(),
728                        e
729                    );
730                    let error: &'static str = e.into();
731                    inner
732                        .context
733                        .metrics
734                        .node_metrics
735                        .commit_sync_fetch_once_errors
736                        .with_label_values(&[hostname.as_str(), error, inner.sync_type.as_str()])
737                        .inc();
738                }
739                Err(_) => {
740                    inner
741                        .context
742                        .peer_responsiveness
743                        .record_failure_with_timeout(data_source, authority, failure_penalty);
744                    let hostname = inner
745                        .context
746                        .committee
747                        .authority(authority)
748                        .hostname
749                        .clone();
750                    warn!(
751                        "[{}] Timed out fetching {commit_range:?} from {authority}",
752                        inner.sync_type.as_str()
753                    );
754                    inner
755                        .context
756                        .metrics
757                        .node_metrics
758                        .commit_sync_fetch_once_errors
759                        .with_label_values(&[
760                            hostname.as_str(),
761                            "FetchTimeout",
762                            inner.sync_type.as_str(),
763                        ])
764                        .inc();
765                }
766            }
767        }
768        // Avoid busy looping, by waiting for a while before retrying.
769        sleep(FETCH_TIMEOUT).await;
770    }
771}

773/// Generic function to start pending fetches while respecting parallelism
774/// limits. This is shared between RegularCommitSyncer and FastCommitSyncer.
775///
776/// # Parameters
777/// - `inner`: Shared context and dependencies
778/// - `pending_fetches`: Set of commit ranges pending fetch
779/// - `fetched_ranges_count`: Number of fetched ranges waiting to be processed
780/// - `inflight_fetches_count`: Number of currently in-flight fetch tasks
781/// - `synced_commit_index`: Latest synced commit index
782/// - `spawn_fn`: Closure to spawn a new fetch task
783///
784/// # Returns
785/// Updated counts after spawning new fetches
786pub(crate) fn try_start_fetches<C, F>(
787    inner: &Arc<Inner<C>>,
788    pending_fetches: &mut BTreeSet<CommitRange>,
789    fetched_ranges_count: usize,
790    inflight_fetches_count: usize,
791    synced_commit_index: CommitIndex,
792    mut spawn_fn: F,
793) -> (usize, usize)
794where
795    C: NetworkClient,
796    F: FnMut(CommitRange),
797{
798    // Cap parallel fetches based on configured limit and committee size, to avoid
799    // overloading the network. Also when there are too many fetched block headers
800    // that cannot be sent to Core before an earlier fetch has not finished,
801    // reduce parallelism so the earlier fetch can retry on a better host and
802    // succeed.
803    let target_parallel_fetches = inner
804        .context
805        .parameters
806        .commit_sync_parallel_fetches
807        .min(inner.context.committee.size() * 2 / 3)
808        .min(
809            inner
810                .context
811                .parameters
812                .commit_sync_batches_ahead
813                .saturating_sub(fetched_ranges_count),
814        )
815        .max(1);
816
817    let mut new_inflight_count = inflight_fetches_count;
818
819    // Start new fetches if there are pending batches and available slots.
820    loop {
821        if new_inflight_count >= target_parallel_fetches {
822            break;
823        }
824        if !pending_fetches.is_empty() {
825            info!(
826                "[{}] Pending fetches: {:?}, target parallel fetches: {}, inflight fetch number: {}",
827                inner.sync_type.as_str(),
828                pending_fetches,
829                target_parallel_fetches,
830                new_inflight_count
831            );
832        }
833        let Some(commit_range) = pending_fetches.pop_first() else {
834            break;
835        };
836        spawn_fn(commit_range);
837        new_inflight_count += 1;
838    }
839
840    let metrics = &inner.context.metrics.node_metrics;
841    let sync_label = inner.sync_type.as_str();
842    metrics
843        .commit_sync_inflight_fetches
844        .with_label_values(&[sync_label])
845        .set(new_inflight_count as i64);
846    metrics
847        .commit_sync_pending_fetches
848        .with_label_values(&[sync_label])
849        .set(pending_fetches.len() as i64);
850    metrics
851        .commit_sync_highest_synced_index
852        .with_label_values(&[sync_label])
853        .set(synced_commit_index as i64);
854
855    (new_inflight_count, pending_fetches.len())
856}

862/// Result of scheduling commit range fetches.
863pub(crate) struct ScheduleResult {
864    /// The new highest scheduled commit index (if any ranges were scheduled).
865    pub new_highest_scheduled: Option<CommitIndex>,
866    /// The commit ranges that were scheduled for fetching.
867    pub ranges_scheduled: Vec<CommitRange>,
868}
869
870/// Creates commit range batches for fetching, respecting backpressure.
871///
872/// This function is shared between RegularCommitSyncer and FastCommitSyncer.
873/// It calculates which commit ranges should be fetched next based on:
874/// - The current gap between local and quorum commit indices
875/// - Backpressure from unhandled commits
876///
877/// # Parameters
878/// - `inner`: Shared context and dependencies
879/// - `fetch_after_index`: Start fetching from commits after this index
880/// - `quorum_commit_index`: The commit index that quorum has reached
881/// - `highest_handled_index`: The highest commit index that has been processed
882/// - `unhandled_commits_threshold`: Threshold for applying backpressure
883///
884/// # Returns
885/// A `ScheduleResult` with the new highest scheduled index and ranges to fetch.
886pub(crate) fn schedule_commit_ranges<C: NetworkClient>(
887    inner: &Inner<C>,
888    fetch_after_index: CommitIndex,
889    quorum_commit_index: CommitIndex,
890    highest_handled_index: CommitIndex,
891    unhandled_commits_threshold: CommitIndex,
892) -> ScheduleResult {
893    let step = inner.sync_type.commit_sync_batch_size(&inner.context);
894    let mut result = ScheduleResult {
895        new_highest_scheduled: None,
896        ranges_scheduled: Vec::new(),
897    };
898
899    for prev_end in (fetch_after_index..=quorum_commit_index).step_by(step as usize) {
900        let range_start = prev_end + 1;
901        let range_end = prev_end + step;
902
903        // Don't schedule incomplete batches
904        if quorum_commit_index < range_end {
905            break;
906        }
907
908        // Apply backpressure if handler is lagging
909        if highest_handled_index + unhandled_commits_threshold < range_end {
910            warn!(
911                "[{}] Skip scheduling new commit fetches: handler lagging. \
912                 highest_handled={}, threshold={}",
913                inner.sync_type.as_str(),
914                highest_handled_index,
915                unhandled_commits_threshold
916            );
917            break;
918        }
919
920        result
921            .ranges_scheduled
922            .push((range_start..=range_end).into());
923        result.new_highest_scheduled = Some(range_end);
924    }
925
926    result
927}

929/// Handles JoinError from fetch tasks.
930///
931/// # Returns
932/// `true` if the syncer should shutdown, `false` otherwise.
933pub(crate) fn handle_fetch_join_error(
934    error: &tokio::task::JoinError,
935    sync_type: &CommitSyncType,
936) -> bool {
937    if error.is_panic() {
938        // Re-panic in the main task
939        return true;
940    }
941    warn!(
942        "[{}] Fetch cancelled. Shutting down: {}",
943        sync_type.as_str(),
944        error
945    );
946    true // Signal to shutdown
947}

949/// Re-queues unfetched portion of a commit range for retry.
950///
951/// When a fetch returns partial results (fewer commits than requested),
952/// this function queues the remaining range for another fetch attempt.
953pub(crate) fn requeue_partial_range(
954    pending_fetches: &mut BTreeSet<CommitRange>,
955    commit_end: CommitIndex,
956    target_end: CommitIndex,
957) {
958    if commit_end < target_end {
959        pending_fetches.insert((commit_end + 1..=target_end).into());
960    }
