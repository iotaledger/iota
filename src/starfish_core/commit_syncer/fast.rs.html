71/// Which worker skipped a step because the fast commit syncer was active.
72/// Used as the `source` label on `syncer_paused_by_fast_sync`. All
73/// variants share a single metric; keeping them in one enum makes the
74/// label space disjoint and centrally visible.
75#[derive(Clone, Copy)]
76pub(crate) enum FastSyncPauseSource {
77    /// `RegularCommitSyncer::try_schedule_once` skipped adding new ranges
78    /// to `pending_fetches`.
79    RegularSchedule,
80    /// `RegularCommitSyncer::try_start_fetches` skipped moving ranges
81    /// from `pending_fetches` into `inflight_fetches`.
82    RegularStartFetches,
83    /// `HeaderSynchronizer` dropped a `FetchBlockHeaders` command
84    /// instead of dispatching it.
85    HeaderCommand,
86    /// `HeaderSynchronizer`'s periodic scheduler tick skipped starting
87    /// a new sync task.
88    HeaderScheduler,
89}
90
91impl FastSyncPauseSource {
92    fn as_str(self) -> &'static str {
93        match self {
94            Self::RegularSchedule => "regular_schedule",
95            Self::RegularStartFetches => "regular_start_fetches",
96            Self::HeaderCommand => "header_command",
97            Self::HeaderScheduler => "header_scheduler",
98        }
99    }
100}

102/// Returns true if the fast commit syncer is currently doing work and the
103/// caller should skip its gated step. Bumps the shared
104/// `syncer_paused_by_fast_sync` metric with the given source label when
105/// the gate fires. Returns false when fast sync is disabled at this
106/// deployment (`fast_sync_active` is `None`) — the call is then an
107/// unconditional pass-through with no metric mutation.
108pub(crate) fn paused_by_fast_sync(
109    fast_sync_active: Option<&Arc<AtomicBool>>,
110    metrics: &crate::metrics::NodeMetrics,
111    source: FastSyncPauseSource,
112) -> bool {
113    let paused = fast_sync_active.is_some_and(|flag| flag.load(Ordering::Relaxed));
114    if paused {
115        metrics
116            .syncer_paused_by_fast_sync
117            .with_label_values(&[source.as_str()])
118            .inc();
119    }
120    paused
121}

138pub(crate) struct FastCommitSyncer<C: NetworkClient> {
139    // States shared by scheduler and fetch tasks.
140
141    // Shared components wrapper.
142    inner: Arc<Inner<C>>,
143
144    block_stream_reset_sender: watch::Sender<()>,
145
146    // States only used by the scheduler.
147
148    // Inflight requests to fetch commits from different authorities.
149    inflight_fetches: JoinSet<(u32, FastSyncOutput)>,
150    // Additional ranges of commits to fetch.
151    pending_fetches: BTreeSet<CommitRange>,
152    // Fetched commits and blocks by commit range.
153    fetched_ranges: BTreeMap<CommitRange, FastSyncOutput>,

155    // Used to determine the start of new ranges to be fetched.
156    highest_scheduled_index: Option<CommitIndex>,
157    // Highest index among fetched commits, after commits and blocks are verified.
158    // Used for metrics.
159    highest_fetched_commit_index: CommitIndex,
160    // The commit index that is the max of highest local commit index and commit index inflight to
161    // Core. Used to determine if fetched blocks can be sent to Core without gaps.

167    // Whether the fast syncer has actually fetched any data. Close-to-quorum mode only
168    // activates after this is true. Reset to false after reinitialization completes.

218    pub(crate) fn start(self) -> CommitSyncerHandle {
219        let (tx_shutdown, rx_shutdown) = oneshot::channel();
220        let schedule_task = spawn_logged_monitored_task!(self.schedule_loop(rx_shutdown,));
221        CommitSyncerHandle {
222            schedule_task,
223            tx_shutdown,
224        }
225    }
226    #[cfg_attr(test,tracing::instrument(skip_all, name ="",fields(authority = %self.inner.context.own_index)))]
227    async fn schedule_loop(mut self, mut rx_shutdown: oneshot::Receiver<()>) {
228        let mut interval = tokio::time::interval(Duration::from_millis(500));
229        interval.set_missed_tick_behavior(MissedTickBehavior::Skip);
230
231        loop {
232            tokio::select! {
233                // Periodically, schedule new fetches if the node is falling behind.
234                _ = interval.tick() => {
235                    self.try_schedule_once();
236                }
237                // Handles results from fetch tasks.
238                Some(result) = self.inflight_fetches.join_next(), if !self.inflight_fetches.is_empty() => {
239                    if let Err(ref e) = result {
240                        if e.is_panic() {
241                            std::panic::resume_unwind(result.unwrap_err().into_panic());
242                        }
243                        if handle_fetch_join_error(e, &self.inner.sync_type) {
244                            // If any fetch is cancelled or panicked, try to shutdown and exit the loop.
245                            self.inflight_fetches.shutdown().await;
246                            return;
247                        }
248                    }
249                    let (target_end, output) = result.unwrap();
250                    self.handle_fetch_result(target_end, output).await;
251                }
252                _ = &mut rx_shutdown => {
253                    // Shutdown requested.
254                    info!("[{}] FastCommitSyncer shutting down ...", self.inner.sync_type.as_str());
255                    self.inflight_fetches.shutdown().await;
256                    return;
257                }
258            }
259
260            self.try_start_fetches();
261
262            // Handle close-to-quorum mode: when all fetches complete and we're close
263            // to the quorum, fetch block headers for a large enough number of rounds and
264            // reinitialize.
265            if self.close_to_quorum_mode
266                && self.inflight_fetches.is_empty()
267                && self.pending_fetches.is_empty()
268                && self.fetched_ranges.is_empty()
269            {
270                info!(
271                    "[{}] Close-to-quorum: all fetches complete, fetching headers for cached_rounds",
272                    self.inner.sync_type.as_str()
273                );
274
275                match Self::fetch_headers_for_reinitialization(self.inner.clone()).await {
276                    Ok(headers) => self.reinitialize_components(headers).await,
277                    Err(e) => {
278                        warn!(
279                            "[{}] Failed to fetch headers for cached rounds: {}",
280                            self.inner.sync_type.as_str(),
281                            e
282                        );
283                    }
284                }
285
286                // Reset state - regular syncer takes over for now.
287                // Keep the loop running so we can re-activate if the node
288                // falls behind significantly again.
289                self.close_to_quorum_mode = false;
290                self.has_fetched_data = false;
291                self.highest_scheduled_index = None;
292                self.synced_commit_index = self.inner.dag_state.read().last_solid_commit_index();
293
294                info!(
295                    "[{}] Fast sync complete, staying active for potential reactivation",
296                    self.inner.sync_type.as_str()
297                );
298            }
299
300            // Any of these being true means fast sync still has work; regular
301            // sync and the header synchronizer must stay paused.
302            let active = self.has_fetched_data
303                || self.close_to_quorum_mode
304                || !self.inflight_fetches.is_empty()
305                || !self.pending_fetches.is_empty()
306                || !self.fetched_ranges.is_empty();
307            if let Some(flag) = &self.inner.fast_sync_active {
308                flag.store(active, Ordering::Relaxed);
309            }
310        }
311    }

349    fn try_schedule_once(&mut self) {
350        let quorum_commit_index = self.inner.commit_vote_monitor.quorum_commit_index();
351        let last_solid_commit_index = self.inner.dag_state.read().last_solid_commit_index();
352        let highest_handled_index = self.inner.commit_consumer_monitor.highest_handled_commit();
353        let highest_scheduled_index = self.highest_scheduled_index.unwrap_or(0);
354        let unhandled_commits_threshold =
355            self.inner.context.parameters.unhandled_commits_threshold();
356        let step = self
357            .inner
358            .sync_type
359            .commit_sync_batch_size(&self.inner.context);
360
361        // Skip scheduling depending on sync type and gap threshold.
362        let gap = quorum_commit_index.saturating_sub(last_solid_commit_index);
363        let should_schedule = self.has_fetched_data
364            || self.inner.sync_type.should_schedule(
365                gap,
366                self.inner.context.parameters.commit_sync_gap_threshold,
367                self.inner.context.parameters.enable_fast_commit_syncer,
368            );
369
370        if should_schedule {
371            let metrics = &self.inner.context.metrics.node_metrics;
372            metrics
373                .commit_sync_quorum_index
374                .set(quorum_commit_index as i64);
375            metrics
376                .commit_sync_local_index
377                .set(last_solid_commit_index as i64);
378            // Update synced_commit_index periodically to make sure it is not smaller than
379            // local solid commit index.
380            self.synced_commit_index = self.synced_commit_index.max(last_solid_commit_index);
381
382            // TODO: cleanup inflight fetches that are no longer needed.
383            let fetch_after_index = self
384                .synced_commit_index
385                .max(self.highest_scheduled_index.unwrap_or(0));
386
387            debug!(
388                "[{}] Checking to schedule fetches: synced_commit_index={}, highest_handled_index={}, highest_scheduled_index={}, quorum_commit_index={}, unhandled_commits_threshold={}, fetch_after_index={}",
389                self.inner.sync_type.as_str(),
390                self.synced_commit_index,
391                highest_handled_index,
392                highest_scheduled_index,
393                quorum_commit_index,
394                unhandled_commits_threshold,
395                fetch_after_index,
396            );
397
398            // Schedule commit ranges for fetching using shared helper
399            let schedule_result = schedule_commit_ranges(
400                &self.inner,
401                fetch_after_index,
402                quorum_commit_index,
403                highest_handled_index,
404                unhandled_commits_threshold,
405            );
406
407            // Add scheduled ranges to pending fetches
408            for range in schedule_result.ranges_scheduled {
409                debug!(
410                    "[{}] Scheduling fetch for commit range {}..={}",
411                    self.inner.sync_type.as_str(),
412                    range.start(),
413                    range.end()
414                );
415                self.pending_fetches.insert(range);
416            }
417
418            // Update highest scheduled index
419            if let Some(new_highest) = schedule_result.new_highest_scheduled {
420                self.highest_scheduled_index = Some(new_highest);
421            }
422        }
423
424        // Detect close-to-quorum mode: when remaining gap is less than a full batch.
425        // Only activate if we've actually fetched data during this fast sync session.
426        //
427        // When close_to_quorum_mode is activated, the schedule_loop() will:
428        // 1. Wait for all inflight/pending fetches to complete
429        // 2. Fetch block headers for ~cached_rounds worth of commits
430        // 3. Send ReinitializeComponents to core thread to properly initialize DAG
431        //    state
432        // 4. Reset fast sync state so regular syncer can take over
433        if self.has_fetched_data && !self.close_to_quorum_mode {
434            let current_fetch_after = self
435                .synced_commit_index
436                .max(self.highest_scheduled_index.unwrap_or(0));
437            let remaining_gap = quorum_commit_index.saturating_sub(current_fetch_after);
438            if remaining_gap > 0 && remaining_gap < step {
439                let range_start = current_fetch_after + 1;
440                let range_end = quorum_commit_index;
441                debug!(
442                    "[{}] Scheduling final partial fetch for commit range {}..={} (remaining_gap={})",
443                    self.inner.sync_type.as_str(),
444                    range_start,
445                    range_end,
446                    remaining_gap
447                );
448                self.pending_fetches
449                    .insert((range_start..=range_end).into());
450                self.highest_scheduled_index = Some(range_end);
451            }
452            if remaining_gap < step {
453                self.close_to_quorum_mode = true;
454                info!(
455                    "[{}] Entering close-to-quorum mode: remaining_gap={}, step={}",
456                    self.inner.sync_type.as_str(),
457                    remaining_gap,
458                    step
459                );
460            }
461        }
462    }

464    async fn handle_fetch_result(&mut self, target_end: CommitIndex, output: FastSyncOutput) {
465        assert!(!output.committed_subdags.is_empty());
466
467        // Track that we have actually fetched data during this fast sync session.
468        self.has_fetched_data = true;
469
470        let total_transactions_size_bytes = output
471            .committed_subdags
472            .iter()
473            .flat_map(|subdag| &subdag.transactions)
474            .map(|txns| txns.serialized().len() as u64)
475            .sum();
476
477        let metrics = &self.inner.context.metrics.node_metrics;
478        let sync_label = self.inner.sync_type.as_str();
479        metrics
480            .commit_sync_fetched_commits
481            .with_label_values(&[sync_label])
482            .inc_by(output.committed_subdags.len() as u64);
483        metrics
484            .commit_sync_total_fetched_transactions_size
485            .with_label_values(&[sync_label])
486            .inc_by(total_transactions_size_bytes);
487
488        let (commit_start, commit_end) = (
489            output.committed_subdags.first().unwrap().commit_ref.index,
490            output.committed_subdags.last().unwrap().commit_ref.index,
491        );
492        self.highest_fetched_commit_index = self.highest_fetched_commit_index.max(commit_end);
493        metrics
494            .commit_sync_highest_fetched_index
495            .with_label_values(&[sync_label])
496            .set(self.highest_fetched_commit_index as i64);
497
498        // Allow returning partial results and try fetching the rest separately.
499        requeue_partial_range(&mut self.pending_fetches, commit_end, target_end);
500        // Make sure the synced_commit_index is up to date.
501        self.synced_commit_index = self
502            .synced_commit_index
503            .max(self.inner.dag_state.read().last_solid_commit_index());
504        // Only add new blocks if at least some of them are not already synced.
505        if self.synced_commit_index < commit_end {
506            self.fetched_ranges
507                .insert((commit_start..=commit_end).into(), output);
508        }
509        // Try to process as many fetched blocks as possible.
510        while let Some((fetched_commit_range, _)) = self.fetched_ranges.first_key_value() {
511            // Only pop fetched_ranges if there is no gap with blocks already synced.
512            // Note: start, end and synced_commit_index are all inclusive.
513            let (fetched_commit_range, output) =
514                if fetched_commit_range.start() <= self.synced_commit_index + 1 {
515                    self.fetched_ranges.pop_first().unwrap()
516                } else {
517                    // Found a gap between the earliest fetched block and the latest synced block,
518                    // so not sending additional blocks to Core.
519                    metrics
520                        .commit_sync_gap_on_processing
521                        .with_label_values(&[sync_label])
522                        .inc();
523                    break;
524                };
525            // Avoid sending to Core a whole batch of already synced blocks.
526            if fetched_commit_range.end() <= self.synced_commit_index {
527                continue;
528            }
529
530            debug!(
531                "[{}] Fetched {} subdags with transactions for commit range {:?}",
532                sync_label,
533                output.committed_subdags.len(),
534                fetched_commit_range,
535            );
536
537            // If the core thread cannot handle the incoming blocks, it is ok to block here.
538            if let Err(e) = self
539                .inner
540                .core_thread_dispatcher
541                .add_subdags_from_fast_sync(output.clone())
542                .await
543            {
544                info!(
545                    "[{}] Failed to dispatch subdags to core, shutting down: {}",
546                    sync_label, e
547                );
548                return;
549            }
550
551            // Once subdags are sent to Core, ratchet up synced_commit_index
552            self.synced_commit_index = self.synced_commit_index.max(fetched_commit_range.end());
553        }
554
555        metrics
556            .commit_sync_inflight_fetches
557            .with_label_values(&[sync_label])
558            .set(self.inflight_fetches.len() as i64);
559        metrics
560            .commit_sync_pending_fetches
561            .with_label_values(&[sync_label])
562            .set(self.pending_fetches.len() as i64);
563        metrics
564            .commit_sync_highest_synced_index
565            .with_label_values(&[sync_label])
566            .set(self.synced_commit_index as i64);
567    }

570        let inner = self.inner.clone();
571        shared_try_start_fetches(
572            &self.inner,
573            &mut self.pending_fetches,
574            self.fetched_ranges.len(),
575            self.inflight_fetches.len(),
576            self.synced_commit_index,

580            },

584    // Retries fetching commits and block headers from available authorities, until
585    // a request succeeds where at least a prefix of the commit range is
586    // fetched. Returns the fetched commits and block headers referenced by the
587    // commits.
588    #[cfg_attr(test,tracing::instrument(skip_all, name ="",fields(authority = %inner.context.own_index)))]
589    async fn fetch_loop(
590        inner: Arc<Inner<C>>,

593        shared_fetch_loop(inner, commit_range, 2, Self::fetch_once).await
594    }

596    // Fetches commits and transactions from a single authority. When the
597    // response covers only part of the range, returns the prefix of commits
598    // whose transactions were all fetched.
599    async fn fetch_once(
600        inner: Arc<Inner<C>>,
601        target_authority: AuthorityIndex,
602        commit_range: CommitRange,
603        timeout: Duration,
604    ) -> ConsensusResult<FastSyncOutput> {
605        let _timer = inner
606            .context
607            .metrics
608            .node_metrics
609            .commit_sync_fetch_once_latency
610            .with_label_values(&[inner.sync_type.as_str()])
611            .start_timer();
612
613        // 1. Fetch commits, voting headers, and transactions in the commit range from
614        //    the target authority. Each transaction is serialized as
615        //    SerializedTransactionsV2 which includes the TransactionRef.
616        let (
617            serialized_commits,
618            serialized_proof_for_last_commit,
619            serialized_transactions,
620            stream_error,
621        ) = inner
622            .network_client
623            .fetch_commits_and_transactions(target_authority, commit_range.clone(), timeout)
624            .await?;
625
626        // 2. Verify the response contains block headers that can certify the last
627        //    returned commit, and the returned commits are chained by digest,
628        // so earlier commits are certified as well.
629        let max_commits = inner.sync_type.max_commits_per_response(&inner.context);
630        let (mut commits, voting_block_headers) = Handle::current()
631            .spawn_blocking({
632                let inner = inner.clone();
633                move || {
634                    inner.verify_commits(
635                        target_authority,
636                        commit_range,
637                        serialized_commits,
638                        serialized_proof_for_last_commit,
639                        max_commits,
640                    )
641                }
642            })
643            .await
644            .expect("Spawn blocking should not fail")?;
645
646        // 3. Collect the committed transaction refs of each commit. Commits passing
647        //    verify_commits are V2/V3, which only carry `TransactionRef`s, so the
648        //    legacy `BlockRef` variant is an error.
649        let mut commits_tx_refs: Vec<Vec<TransactionRef>> = commits
650            .iter()
651            .map(|c| {
652                c.committed_transactions()
653                    .into_iter()
654                    .map(GenericTransactionRef::expect_transaction_ref)
655                    .collect()
656            })
657            .collect::<ConsensusResult<_>>()?;
658        let mut committed_tx_refs: BTreeSet<TransactionRef> =
659            commits_tx_refs.iter().flatten().copied().collect();
660
661        // 4. Process fetched transactions. Each serialized_transaction is a
662        //    SerializedTransactionsV2 containing both the TransactionRef and the actual
663        //    transaction data.
664        let mut fetched_transactions = process_serialized_transactions(
665            target_authority,
666            serialized_transactions,
667            &mut committed_tx_refs,
668        )
669        .inspect_err(|_| {
670            // Truncation drops whole entries and never corrupts one, so a
671            // malformed or uncommitted entry is the peer's fault.
672            inner.misbehavior_store.record_faulty_transactions(
673                target_authority,
674                false,
675                [target_authority],
676            );
677        })?;
678
679        // Empty payloads can be recovered from their commitments.
680        fill_missing_empty_transactions(
681            &inner.context,
682            &mut committed_tx_refs,
683            &mut fetched_transactions,
684        );
685
686        // The response may be missing transactions for a suffix of the commits,
687        // e.g. when the stream was cut off by the response byte limit or a
688        // mid-stream network error. Keep the prefix of commits whose
689        // transactions were all fetched so the fetch makes forward progress;
690        // the scheduler requeues the range after the prefix.
691        if !committed_tx_refs.is_empty() {
692            let fetched_commits = commits.len();
693            if let Err(mismatch) = truncate_to_fully_fetched_prefix(
694                target_authority,
695                &mut commits,
696                &mut commits_tx_refs,
697                &mut fetched_transactions,
698            ) {
699                // A cut stream explains the empty covered prefix: report the
700                // connection failure rather than transactions the peer was
701                // never able to send.
702                return Err(stream_error.unwrap_or(mismatch));
703            }
704            info!(
705                "[{}] Fetched transactions cover only {} out of {} commits received from {}, processing the covered prefix",
706                inner.sync_type.as_str(),
707                commits.len(),
708                fetched_commits,
709                target_authority,
710            );
711            inner
712                .context
713                .metrics
714                .node_metrics
715                .commit_sync_truncated_fetches
716                .with_label_values(&[inner.sync_type.as_str()])
717                .inc();
718        }
719
720        // 5. Verify the transactions against their commitments
721        let mut transactions_map = if !fetched_transactions.is_empty() {
722            Handle::current()
723                .spawn_blocking({
724                    let context = inner.context.clone();
725
726                    move || {
727                        verify_transactions_commitments(
728                            &context,
729                            target_authority,
730                            fetched_transactions,
731                        )
732                    }
733                })
734                .await
735                .expect("Spawn blocking should not fail")
736                .inspect_err(|_| {
737                    // Not provable against the author, whose commitment the
738                    // peer may have forged.
739                    inner.misbehavior_store.record_faulty_transactions(
740                        target_authority,
741                        false,
742                        [target_authority],
743                    );
744                })?
745        } else {
746            BTreeMap::new()
747        };
748
749        // 6. Now create the CommittedSubDags with the fetched transactions.
750        // For fast commit sync, we use block headers refs and reputation scores from
751        // the commit.
752        let mut committed_subdags = Vec::new();
753        // Replayed commits share one snapshot of current store state; downstream
754        // consumers merge-max against their last-seen, so repeating absolute
755        // totals across the batch is a no-op after the first.
756        let misbehavior_counts = inner.dag_state.read().misbehavior_store().snapshot_totals();
757        for (commit, commit_tx_refs) in commits.iter().zip(&commits_tx_refs) {
758            // Get block headers from the commit
759            let committed_header_refs = commit.block_headers().to_vec();
760
761            // Get reputation scores from the commit
762            let reputation_scores = commit.reputation_scores().to_vec();
763
764            // Collect transactions for this commit
765            let commit_transactions: Vec<CommitmentVerifiedTransactions> = commit_tx_refs
766                .iter()
767                .filter_map(|tx_ref| transactions_map.remove(tx_ref))
768                .collect();
769
770            committed_subdags.push(CommittedSubDag::new(
771                commit.leader(),
772                vec![], // headers - VerifiedBlockHeader, we don't have these in fast sync
773                committed_header_refs,
774                commit_transactions,
775                commit.timestamp_ms(),
776                commit.reference(),
777                reputation_scores,
778                misbehavior_counts.clone(),
779            ));
780        }
781
782        Ok(FastSyncOutput {
783            commits,
784            committed_subdags,
785            voting_block_headers,
786        })
787    }

789    /// Fetches block headers needed for component reinitialization from the
790    /// network. This is called when close_to_quorum mode is active and all
791    /// pending fetches complete. Fetches headers for the maximum of
792    /// cached_rounds, gc_depth * 2, leader_schedule_window, and
793    /// commits_since_schedule_update to satisfy DagState cache, linearizer,
794    /// and leader schedule recovery requirements.
795    async fn fetch_headers_for_reinitialization(
796        inner: Arc<Inner<C>>,
797    ) -> ConsensusResult<Vec<VerifiedBlockHeader>> {
798        // We need headers for three purposes:
799        // 1. DagState cache: at least cached_rounds commits back
800        // 2. Linearizer recovery: at least gc_depth * 2 commits back
801        // 3. Leader schedule recovery: at least leader_schedule_window commits back, or
802        //    all commits since the last stored commit info
803        //    (commits_since_schedule_update)
804        // Fetch the maximum to satisfy all requirements
805        let cached_rounds = inner.context.parameters.dag_state_cached_rounds;
806        let gc_depth = inner.context.protocol_config.gc_depth();
807        let leader_schedule_window = inner.context.protocol_config.commits_per_schedule();
808        // Get block refs from recent commits stored during fast sync
809        // TODO: The commits might not yet stored, but only fetched and pending
810        // processing.
811        let (commits_since_schedule_update, block_refs) = {
812            let dag_state = inner.dag_state.read();
813            let last_commit_index = dag_state.last_commit_index();
814            let last_commit_info_index = dag_state.last_commit_info_index();
815            let commits_since_schedule_update =
816                last_commit_index.saturating_sub(last_commit_info_index);
817            let num_commits = max(
818                commits_since_schedule_update,
819                max(leader_schedule_window, max(cached_rounds, gc_depth * 2)),
820            );
821            // When the sliding-window leader schedule is enabled, recovery replays
822            // the scorer over `replay_start..=last`; fetch enough commits to cover
823            // that range so the replayed window matches a fully-synced node
824            let num_commits = if inner
825                .context
826                .protocol_config
827                .consensus_enable_sliding_window_leader_schedule()
828            {
829                let window = inner.context.protocol_config.leader_schedule_window_size();
830                let replay_start = SlidingWindowSchedule::replay_start(last_commit_index, window);
831                max(
832                    num_commits,
833                    last_commit_index
834                        .saturating_sub(replay_start)
835                        .saturating_add(1),
836                )
837            } else {
838                num_commits
839            };
840            let block_refs = dag_state.get_block_refs_for_recent_commits(num_commits);
841            (commits_since_schedule_update, block_refs)
842        };
843
844        let max_headers_per_fetch = inner.context.parameters.max_headers_per_commit_sync_fetch;
845
846        info!(
847            "[{}] Fetching {} block headers for reinitialization (cached_rounds={}, gc_depth*2={}, leader_schedule_window={}, commits_since_schedule_update={})",
848            inner.sync_type.as_str(),
849            block_refs.len(),
850            cached_rounds,
851            gc_depth * 2,
852            leader_schedule_window,
853            commits_since_schedule_update
854        );
855
856        let mut target_authorities: Vec<_> = inner
857            .context
858            .committee
859            .authorities()
860            .filter_map(|(i, _)| {
861                if i != inner.context.own_index {
862                    Some(i)
863                } else {
864                    None
865                }
866            })
867            .collect();
868        let mut rng = StdRng::from_rng(thread_rng()).expect("thread_rng should be available");
869        // Without ranking, one shuffle for load balancing covers every chunk.
870        #[cfg(not(test))]
871        if !inner.context.parameters.enable_peer_responsiveness_ranking {
872            target_authorities.shuffle(&mut rng);
873        }
874
875        // Fetch headers in chunks to avoid overwhelming the network
876        let mut all_headers = Vec::new();
877        for chunk in block_refs.chunks(max_headers_per_fetch) {
878            let chunk_refs: Vec<_> = chunk.to_vec();
879
880            // Reordered per chunk rather than once, so a peer that stalls on one
881            // chunk is demoted for the rest instead of costing the full timeout
882            // on every one of them.
883            if inner.context.parameters.enable_peer_responsiveness_ranking {
884                inner.context.peer_responsiveness.prioritize(
885                    DataSource::FastCommitSyncer,
886                    &mut target_authorities,
887                    &mut rng,
888                );
889            }
890
891            // Try fetching from different authorities until successful
892            let mut fetched = false;
893            for &authority in &target_authorities {
894                let started = Instant::now();
895                match tokio::time::timeout(
896                    FETCH_HEADERS_TIMEOUT,
897                    inner.network_client.fetch_block_headers(
898                        authority,
899                        chunk_refs.clone(),
900                        vec![],
901                        FETCH_HEADERS_TIMEOUT,
902                    ),
903                )
904                .await
905                {
906                    Ok(Ok(serialized_headers)) => {
907                        // Verify headers match requested refs
908                        match verify_fetched_headers(authority, &chunk_refs, serialized_headers) {
909                            Ok(headers) => {
910                                inner.context.peer_responsiveness.record_success(
911                                    DataSource::FastCommitSyncer,
912                                    authority,
913                                    started.elapsed(),
914                                );
915                                info!(
916                                    "[{}] Fetched {} headers from authority {}",
917                                    inner.sync_type.as_str(),
918                                    headers.len(),
919                                    authority
920                                );
921                                all_headers.extend(headers);
922                                fetched = true;
923                                break;
924                            }
925                            Err(e) => {
926                                inner
927                                    .misbehavior_store
928                                    .record_faulty_block(authority, authority, &e);
929                                record_headers_for_reinitialization_failure(&inner, authority);
930                                warn!(
931                                    "[{}] Failed to verify headers from {}: {}",
932                                    inner.sync_type.as_str(),
933                                    authority,
934                                    e
935                                );
936                            }
937                        }
938                    }
939                    Ok(Err(e)) => {
940                        record_headers_for_reinitialization_failure(&inner, authority);
941                        warn!(
942                            "[{}] Failed to fetch headers from {}: {}",
943                            inner.sync_type.as_str(),
944                            authority,
945                            e
946                        );
947                    }
948                    Err(_) => {
949                        record_headers_for_reinitialization_failure(&inner, authority);
950                        warn!(
951                            "[{}] Timed out fetching headers from {}",
952                            inner.sync_type.as_str(),
953                            authority
954                        );
955                    }
956                }
957            }
958
959            if !fetched {
960                return Err(ConsensusError::FailedToFetchBlockHeaders {
961                    num_requested: chunk_refs.len(),
962                });
963            }
964        }
965
966        info!(
967            "[{}] Successfully fetched {} total block headers for reinitialization",
968            inner.sync_type.as_str(),
969            all_headers.len()
970        );
971
972        Ok(all_headers)
973    }

1006/// Deserializes fetched transaction entries and keys them by their
1007/// `TransactionRef`, consuming the matching refs from `committed_tx_refs`.
1008///
1009/// Every entry must deserialize and match a committed ref: a malformed entry
1010/// is a peer fault, not a benign byte cutoff — skipping it would be
1011/// indistinguishable from a truncated response and would let the peer cap
1012/// every fetch at one commit of progress via the prefix fallback.
1013fn process_serialized_transactions(
1014    peer: AuthorityIndex,
1015    serialized_transactions: Vec<Bytes>,
1016    committed_tx_refs: &mut BTreeSet<TransactionRef>,
1017) -> ConsensusResult<BTreeMap<TransactionRef, Bytes>> {
1018    let mut fetched_transactions = BTreeMap::new();
1019    for serialized_transaction in serialized_transactions {
1020        let tx_v2: SerializedTransactionsV2 = bcs::from_bytes(&serialized_transaction)
1021            .map_err(ConsensusError::MalformedTransactions)?;
1022        let transaction_ref = tx_v2.transaction_ref;
1023        if !committed_tx_refs.contains(&transaction_ref) {
1024            return Err(ConsensusError::UnexpectedTransactionForCommit {
1025                peer,
1026                received: transaction_ref,
1027            });
1028        }
1029        fetched_transactions.insert(transaction_ref, tx_v2.serialized_transactions);
1030        committed_tx_refs.remove(&transaction_ref);
1031    }
1032    Ok(fetched_transactions)
1033}

1035fn fill_missing_empty_transactions(
1036    context: &Context,
1037    committed_tx_refs: &mut BTreeSet<TransactionRef>,
1038    fetched_transactions: &mut BTreeMap<TransactionRef, Bytes>,
1039) {
1040    committed_tx_refs.retain(|transaction_ref| {
1041        let Some(empty) = context.empty_transactions_for_ref((*transaction_ref).into()) else {
1042            return true;
1043        };
1044        fetched_transactions.insert(*transaction_ref, empty.serialized().clone());
1045        false
1046    });
1047}

1049/// Truncates verified `commits` (and their aligned per-commit transaction
1050/// refs in `commits_tx_refs`) to the longest prefix whose committed
1051/// transactions are all present in `fetched_transactions`, and drops fetched
1052/// transactions not referenced by that prefix. Commits verified by
1053/// `verify_commits` are chained by digest up to a vote-certified last commit,
1054/// so any prefix of them remains trusted on its own.
1055///
1056/// Returns an error naming `peer` when even the first commit is missing
1057/// transactions, since the response then allows no forward progress. Not
1058/// recorded as misbehavior: an honest response can be cut to any prefix.
1059fn truncate_to_fully_fetched_prefix(
1060    peer: AuthorityIndex,
1061    commits: &mut Vec<TrustedCommit>,
1062    commits_tx_refs: &mut Vec<Vec<TransactionRef>>,
1063    fetched_transactions: &mut BTreeMap<TransactionRef, Bytes>,
1064) -> ConsensusResult<()> {
1065    let prefix_len = commits_tx_refs
1066        .iter()
1067        .take_while(|commit_tx_refs| {
1068            commit_tx_refs
1069                .iter()
1070                .all(|tx_ref| fetched_transactions.contains_key(tx_ref))
1071        })
1072        .count();
1073    if prefix_len == 0 {
1074        return Err(ConsensusError::FetchedTransactionsMismatch {
1075            peer,
1076            expected: commits_tx_refs
1077                .iter()
1078                .map(|commit_tx_refs| commit_tx_refs.len())
1079                .sum(),
1080            received: fetched_transactions.len(),
1081        });
1082    }
1083    if prefix_len < commits.len() {
1084        commits.truncate(prefix_len);
1085        commits_tx_refs.truncate(prefix_len);
1086        let prefix_tx_refs: BTreeSet<&TransactionRef> = commits_tx_refs.iter().flatten().collect();
1087        fetched_transactions.retain(|tx_ref, _| prefix_tx_refs.contains(tx_ref));
1088    }
1089    Ok(())
1090}

1258            );

1260                BlockRef::new(2, AuthorityIndex::new_for_test(1), BlockHeaderDigest::MIN);

1262                context,
1263                2,

2100    /// Drains all ready committed subdags from the running validators (those
2101    /// whose index is not in `stopped`), asserting commit indices advance
2102    /// monotonically and updating both the per-validator high-water marks and
2103    /// the consumer monitors.
2104    fn drain_running(
2105        output_receivers: &mut [UnboundedReceiver<CommittedSubDag>],
2106        committed_index: &mut [u32],
2107        consumer_monitors: &[Arc<CommitConsumerMonitor>],
2108        stopped: &[usize],
2109    ) {
2110        for (index, receiver) in output_receivers.iter_mut().enumerate() {
2111            if stopped.contains(&index) {
2112                continue;
2113            }
2114            while let Ok(committed_subdag) = receiver.try_recv() {
2115                let commit_index = committed_subdag.commit_ref.index;
2116                assert!(
2117                    commit_index > committed_index[index],
2118                    "Commit index {} should be greater than previous {}",
2119                    commit_index,
2120                    committed_index[index]
2121                );
2122                committed_index[index] = commit_index;
2123                consumer_monitors[index].set_highest_handled_commit(commit_index);
2124            }
2125        }
2126    }

2128    /// Runs the running validators (those not in `stopped`) until the highest
2129    /// commit index among them reaches `target`, draining their output
2130    /// meanwhile. Panics if `target` is not reached within `timeout`.
2131    ///
2132    /// Phases drive a target commit count rather than a fixed wall-clock
2133    /// duration so the gap a restarted validator must close is deterministic
2134    /// regardless of how fast the host commits: a too-small gap would fall
2135    /// under the fast-sync threshold and silently downgrade to regular
2136    /// sync.
2137    async fn run_until_commit_index(
2138        output_receivers: &mut [UnboundedReceiver<CommittedSubDag>],
2139        committed_index: &mut [u32],
2140        consumer_monitors: &[Arc<CommitConsumerMonitor>],
2141        stopped: &[usize],
2142        target: u32,
2143        timeout: Duration,
2144    ) {
2145        let start_time = Instant::now();
2146        loop {
2147            drain_running(
2148                output_receivers,
2149                committed_index,
2150                consumer_monitors,
2151                stopped,
2152            );
2153            let highest = committed_index
2154                .iter()
2155                .enumerate()
2156                .filter(|(i, _)| !stopped.contains(i))
2157                .map(|(_, v)| *v)
2158                .max()
2159                .unwrap_or(0);
2160            if highest >= target {
2161                return;
2162            }
2163            assert!(
2164                start_time.elapsed() < timeout,
2165                "running validators only reached commit {highest}, expected {target}, within {timeout:?}"
2166            );
2167            sleep(Duration::from_millis(50)).await;
2168        }
2169    }

2521    /// Test that a validator with pending subdags (gap between last_commit and
2522    /// last_solid_commit_leader_round) can successfully catch up via fast sync
2523    /// after restart.
2524    ///
2525    /// This test creates pending subdags using dynamic peer unsubscribe, then
2526    /// stops and restarts the validator to verify fast sync handles
2527    /// pre-existing pending subdags correctly.
2528    ///
2529    /// Test flow:
2530    /// - Phase 1: All validators run together, creating initial commits
2531    /// - Phase 2: Dynamically unsubscribe test validator from validator 1 +
2532    ///   stop txn synchronizer + stop shard reconstructor
2533    /// - Phase 3: Wait for commits with missing txs (creates pending subdags)
2534    ///   and verify gap
2535    /// - Phase 4: Stop test validator
2536    /// - Phase 5: Other validators continue (creates fast sync gap > threshold)
2537    /// - Phase 6: Restart test validator with full connectivity, but keep txn
2538    ///   synchronizer + shard reconstructor stopped to prevent pending subdags
2539    ///   from being solidified
2540    /// - Phase 7: Verify fast sync was used and validator caught up
2541    #[tokio::test(flavor = "current_thread")]
2542    #[serial_test::serial]
2543    async fn test_fast_sync_with_pending_subdags() {
2544        telemetry_subscribers::init_for_testing();
2545        let db_registry = Registry::new();
2546        DBMetrics::init(&db_registry);
2547
2548        const NUM_AUTHORITIES: usize = 4;
2549        const COMMIT_GAP_THRESHOLD: u32 = 30;
2550        const COMMIT_SYNC_BATCH_SIZE: u32 = 20;
2551
2552        // Work phases need to be long enough to create pending subdags during Phase 3.
2553        // During Phase 3, the validator keeps creating commits (headers arrive via
2554        // cordial dissemination), so there's no commit gap for syncers to act
2555        // on. Phase 5 creates a commit gap larger than the threshold for fast
2556        // sync to trigger on restart.
2557        let stable_work_duration = Duration::from_secs(10);
2558
2559        let (committee, keypairs) = local_committee_and_keys(0, vec![1; NUM_AUTHORITIES]);
2560        let protocol_config = ProtocolConfig::get_for_max_version_UNSAFE();
2561
2562        let temp_dirs: Vec<TempDir> = (0..NUM_AUTHORITIES)
2563            .map(|_| TempDir::new().unwrap())
2564            .collect();
2565
2566        let mut authorities = Vec::with_capacity(NUM_AUTHORITIES);
2567        let mut boot_counters = [0u64; NUM_AUTHORITIES];
2568        let mut consumer_monitors = Vec::with_capacity(NUM_AUTHORITIES);
2569        let mut output_receivers = Vec::with_capacity(NUM_AUTHORITIES);
2570
2571        let test_validator_index: usize = 0;
2572        let blocked_validator_index: usize = 1;
2573
2574        // Phase 1: Start all authorities and let them create initial commits.
2575        // Disable fast commit syncer for the test validator so it won't resolve
2576        // pending subdags during Phase 3 (the fast syncer uses last_solid_commit_index
2577        // for gap detection, which would trigger fetching when pending subdags exist).
2578        // Phase 6 restarts the test validator with enable_fast_commit_syncer: true.
2579        for (index, _) in committee.authorities() {
2580            let parameters = Parameters {
2581                db_path: temp_dirs[index.value()].path().to_path_buf(),
2582                dag_state_cached_rounds: 5,
2583                commit_sync_parallel_fetches: 2,
2584                commit_sync_batch_size: COMMIT_SYNC_BATCH_SIZE,
2585                commit_sync_gap_threshold: COMMIT_GAP_THRESHOLD,
2586                fast_commit_sync_batch_size: COMMIT_SYNC_BATCH_SIZE,
2587                enable_fast_commit_syncer: index.value() != test_validator_index,
2588                sync_last_known_own_block_timeout: Duration::from_millis(2_000),
2589                enable_peer_responsiveness_ranking: false,
2590                ..Default::default()
2591            };
2592            let (authority, receiver, monitor) = make_authority_with_params(
2593                index,
2594                &temp_dirs[index.value()],
2595                committee.clone(),
2596                keypairs.clone(),
2597                boot_counters[index],
2598                protocol_config.clone(),
2599                parameters,
2600                0,
2601            )
2602            .await;
2603            boot_counters[index] += 1;
2604            authorities.push(authority);
2605            output_receivers.push(receiver);
2606            consumer_monitors.push(monitor);
2607        }
2608
2609        let mut txn_counter = 0u64;
2610        let start_time = Instant::now();
2611        let mut committed_index = [0u32; NUM_AUTHORITIES];
2612        while start_time.elapsed() < stable_work_duration {
2613            // Submit transactions to all validators (rotating)
2614            let authority_index = txn_counter as usize % authorities.len();
2615            let txn = vec![txn_counter as u8; 16];
2616            authorities[authority_index]
2617                .transaction_client()
2618                .submit(vec![txn])
2619                .await
2620                .unwrap();
2621            txn_counter += 1;
2622
2623            for (index, receiver) in output_receivers.iter_mut().enumerate() {
2624                while let Ok(committed_subdag) = receiver.try_recv() {
2625                    let commit_index = committed_subdag.commit_ref.index;
2626                    assert!(
2627                        commit_index > committed_index[index],
2628                        "Commit index {} should be greater than previous {}",
2629                        commit_index,
2630                        committed_index[index]
2631                    );
2632                    committed_index[index] = commit_index;
2633                    consumer_monitors[index].set_highest_handled_commit(commit_index);
2634                }
2635            }
2636            sleep(Duration::from_millis(50)).await;
2637        }
2638
2639        // Phase 2: Dynamically unsubscribe from validator 1 + stop txn synchronizer +
2640        // stop shard reconstructor This will create pending subdags as headers
2641        // arrive via cordial dissemination but transactions from validator 1's
2642        // blocks are missing and shards cannot be reconstructed. Commit syncers
2643        // won't activate during Phase 3 because there's no commit gap - the
2644        // validator keeps up with commits, just missing transactions.
2645        authorities[test_validator_index].unsubscribe_from_peer_for_test(
2646            committee
2647                .to_authority_index(blocked_validator_index)
2648                .unwrap(),
2649        );
2650        authorities[test_validator_index]
2651            .stop_transactions_synchronizer_for_test()
2652            .await
2653            .expect("Transaction synchronizer should stop");
2654        authorities[test_validator_index]
2655            .stop_shard_reconstructor_for_test()
2656            .await
2657            .expect("Shard reconstructor should stop");
2658
2659        // Phase 3: Wait for headers to arrive via cordial dissemination and commits to
2660        // be created. Submit transactions to all validators (rotating).
2661        // This should create pending subdags (gap between
2662        // last_commit and last_solid_commit_leader_round)
2663
2664        // Track commits before the wait
2665        let commits_before = committed_index[test_validator_index];
2666
2667        // Submit transactions to all validators during Phase 3.
2668        // Validator 0 can process transactions from itself and validators 2 & 3.
2669        // However, validator 0 can't fetch transactions from validator 1's blocks
2670        // because:
2671        // - It's unsubscribed from validator 1
2672        // - Transaction synchronizer is stopped (blocks active transaction fetching)
2673        // - Shard reconstructor is stopped (blocks erasure-coded shard reconstruction)
2674        // This creates pending subdags.
2675        // The fast commit syncer is disabled for the test validator (see Phase 1).
2676        let phase3_start = Instant::now();
2677        while phase3_start.elapsed() < stable_work_duration {
2678            // Submit transactions to all validators (rotating)
2679            let authority_index = txn_counter as usize % authorities.len();
2680            let txn = vec![txn_counter as u8; 16];
2681            authorities[authority_index]
2682                .transaction_client()
2683                .submit(vec![txn])
2684                .await
2685                .unwrap();
2686            txn_counter += 1;
2687
2688            // Drain receivers
2689            for (index, receiver) in output_receivers.iter_mut().enumerate() {
2690                while let Ok(committed_subdag) = receiver.try_recv() {
2691                    let commit_index = committed_subdag.commit_ref.index;
2692                    if commit_index > committed_index[index] {
2693                        committed_index[index] = commit_index;
2694                        consumer_monitors[index].set_highest_handled_commit(commit_index);
2695                    }
2696                }
2697            }
2698            sleep(Duration::from_millis(50)).await;
2699        }
2700
2701        let commits_after = committed_index[test_validator_index];
2702        let new_commits = commits_after - commits_before;
2703        assert!(
2704            new_commits > 0,
2705            "Expected new commits during Phase 3, got 0"
2706        );
2707
2708        // Verify pending subdags gap exists
2709        let dag_state = authorities[test_validator_index].dag_state_for_test();
2710        let last_commit = dag_state.read().last_commit_round();
2711        let last_solid = dag_state.read().last_solid_commit_leader_round();
2712        let dag_round = dag_state.read().threshold_clock_round();
2713
2714        // Verify gap exists - we expect pending subdags
2715        let has_gap = last_commit > last_solid.unwrap_or(0);
2716        assert!(
2717            has_gap,
2718            "Expected pending subdags gap: last_commit={last_commit}, last_solid={last_solid:?}. \
2719             DAG round={dag_round}, new_commits={new_commits}. \
2720             Validator 1's new blocks should have missing transactions."
2721        );
2722
2723        // Record where the validator is now (with pending subdags)
2724        let last_processed_with_pending =
2725            consumer_monitors[test_validator_index].highest_handled_commit();
2726
2727        // Phase 4: Stop test validator (preserves pending subdags to disk)
2728        authorities.remove(test_validator_index).stop().await;
2729
2730        // Phase 5: Let other validators continue while the test validator is stopped
2731        // (creates fast sync gap > threshold)
2732        let start_time = Instant::now();
2733        while start_time.elapsed() < stable_work_duration * 2 {
2734            for (index, receiver) in output_receivers.iter_mut().enumerate() {
2735                if index == test_validator_index {
2736                    continue; // Skip stopped validator
2737                }
2738                while let Ok(committed_subdag) = receiver.try_recv() {
2739                    let commit_index = committed_subdag.commit_ref.index;
2740                    if commit_index > committed_index[index] {
2741                        committed_index[index] = commit_index;
2742                        consumer_monitors[index].set_highest_handled_commit(commit_index);
2743                    }
2744                }
2745            }
2746            sleep(Duration::from_millis(50)).await;
2747        }
2748
2749        let max_other = consumer_monitors
2750            .iter()
2751            .enumerate()
2752            .filter(|(i, _)| *i != test_validator_index)
2753            .map(|(_, m)| m.highest_handled_commit())
2754            .max()
2755            .unwrap_or(0);
2756
2757        let gap = max_other.saturating_sub(last_processed_with_pending);
2758        assert!(
2759            gap > COMMIT_GAP_THRESHOLD,
2760            "Gap {gap} should be greater than threshold {COMMIT_GAP_THRESHOLD}"
2761        );
2762
2763        // Phase 6: Restart test validator with full connectivity and fast commit
2764        // syncer enabled.
2765        let parameters = Parameters {
2766            db_path: temp_dirs[test_validator_index].path().to_path_buf(),
2767            dag_state_cached_rounds: 5,
2768            commit_sync_parallel_fetches: 2,
2769            commit_sync_batch_size: COMMIT_SYNC_BATCH_SIZE,
2770            commit_sync_gap_threshold: COMMIT_GAP_THRESHOLD,
2771            fast_commit_sync_batch_size: COMMIT_SYNC_BATCH_SIZE,
2772            sync_last_known_own_block_timeout: Duration::from_millis(2_000),
2773            enable_fast_commit_syncer: true,
2774            enable_peer_responsiveness_ranking: false,
2775            ..Default::default()
2776        };
2777        let (authority, receiver, monitor) = make_authority_with_params(
2778            committee.to_authority_index(test_validator_index).unwrap(),
2779            &temp_dirs[test_validator_index],
2780            committee.clone(),
2781            keypairs.clone(),
2782            boot_counters[test_validator_index],
2783            protocol_config.clone(),
2784            parameters,
2785            last_processed_with_pending,
2786        )
2787        .await;
2788        output_receivers[test_validator_index] = receiver;
2789        consumer_monitors[test_validator_index] = monitor;
2790        authorities.insert(test_validator_index, authority);
2791
2792        // Phase 7: Wait for the validator to catch up via fast sync
2793        let start_time = Instant::now();
2794        let mut caught_up = false;
2795        while start_time.elapsed() < Duration::from_secs(60) {
2796            for (index, receiver) in output_receivers.iter_mut().enumerate() {
2797                while let Ok(committed_subdag) = receiver.try_recv() {
2798                    let commit_index = committed_subdag.commit_ref.index;
2799                    if commit_index > committed_index[index] {
2800                        committed_index[index] = commit_index;
2801                        consumer_monitors[index].set_highest_handled_commit(commit_index);
2802                    }
2803                }
2804            }
2805
2806            let test_index = consumer_monitors[test_validator_index].highest_handled_commit();
2807            let max_other = consumer_monitors
2808                .iter()
2809                .enumerate()
2810                .filter(|(i, _)| *i != test_validator_index)
2811                .map(|(_, m)| m.highest_handled_commit())
2812                .max()
2813                .unwrap_or(0);
2814
2815            if test_index > last_processed_with_pending && test_index + 20 >= max_other {
2816                caught_up = true;
2817                break;
2818            }
2819            sleep(Duration::from_millis(50)).await;
2820        }
2821
2822        assert!(
2823            caught_up,
2824            "Validator {test_validator_index} should have caught up via fast sync"
2825        );
2826
2827        // Verify the validator progressed significantly after restart with pending
2828        // subdags
2829        let final_index = consumer_monitors[test_validator_index].highest_handled_commit();
2830        assert!(
2831            final_index > last_processed_with_pending,
2832            "Validator should have progressed after restart: with_pending_subdags={last_processed_with_pending}, final={final_index}"
2833        );
2834
2835        // Verify that fast sync was actually used by checking the fetched commits
2836        // metric
2837        let commit_sync_fetched_commits: u64 = authorities
2838            .iter()
2839            .map(|a| {
2840                a.context()
2841                    .metrics
2842                    .node_metrics
2843                    .commit_sync_fetched_commits
2844                    .with_label_values(&["fast_commit_sync"])
2845                    .get()
2846            })
2847            .sum();
2848
2849        assert!(
2850            commit_sync_fetched_commits > 0,
2851            "Expected commits fetched via fast sync > 0, got {commit_sync_fetched_commits}"
2852        );
2853
2854        // Stop all authorities
2855        for authority in authorities {
2856            authority.stop().await;
2857        }
