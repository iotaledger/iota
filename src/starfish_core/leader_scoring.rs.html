22#[derive(Clone, Default, Debug, Serialize, Deserialize, PartialEq, Eq)]
23pub(crate) struct ReputationScores {
24    /// Score per authority. Vec index is the `AuthorityIndex`.
25    pub(crate) scores_per_authority: Vec<u64>,
26    /// The committed interval ending at the rotation boundary these scores were
27    /// taken at. The scores are aggregated over a deeper window ending
28    /// `MAX_PENDING_COMMITS` commits earlier.
29    pub(crate) commit_range: CommitRange,
30}
31
32impl ReputationScores {
33    pub(crate) fn new(commit_range: CommitRange, scores_per_authority: Vec<u64>) -> Self {
34        Self {
35            scores_per_authority,
36            commit_range,
37        }
38    }
39
40    // Returns the authorities index with score tuples.
41    pub(crate) fn authorities_by_score(&self, context: Arc<Context>) -> Vec<(AuthorityIndex, u64)> {
42        self.scores_per_authority
43            .iter()
44            .enumerate()
45            .map(|(index, score)| {
46                (
47                    context
48                        .committee
49                        .to_authority_index(index)
50                        .expect("Should be a valid AuthorityIndex"),
51                    *score,
52                )
53            })
54            .collect()
55    }
56
57    pub(crate) fn update_metrics(&self, context: Arc<Context>) {
58        for (index, score) in self.scores_per_authority.iter().enumerate() {
59            let authority_index = context
60                .committee
61                .to_authority_index(index)
62                .expect("Should be a valid AuthorityIndex");
63            let authority = context.committee.authority(authority_index);
64            if !authority.hostname.is_empty() {
65                context
66                    .metrics
67                    .node_metrics
68                    .reputation_scores
69                    .with_label_values(&[&authority.hostname])
70                    .set(*score as i64);
71            }
72        }
73    }
74
75    /// Creates ReputationScores from reputation_scores_desc (as stored in
76    /// commits). The reputation_scores_desc is Vec<(AuthorityIndex, u64)>
77    /// sorted by score descending. This converts it to scores_per_authority
78    /// indexed by authority.
79    ///
80    /// Returns `InvalidAuthorityIndex` if an authority index is out of range
81    /// for the committee; the scores originate from fetched commits.
82    pub(crate) fn from_scores_desc(
83        num_authorities: usize,
84        commit_range: CommitRange,
85        reputation_scores_desc: &[(AuthorityIndex, u64)],
86    ) -> ConsensusResult<Self> {
87        let mut scores_per_authority = vec![0u64; num_authorities];
88        for (authority_index, score) in reputation_scores_desc {
89            let slot = scores_per_authority
90                .get_mut(authority_index.value())
91                .ok_or(ConsensusError::InvalidAuthorityIndex {
92                    index: *authority_index,
93                    max: num_authorities,
94                })?;
95            *slot = *score;
96        }
97        Ok(Self {
98            scores_per_authority,
99            commit_range,
100        })
101    }
102}

104/// Reputation multiplier applied to a strong vote relative to an ordinary vote.
105/// A strong vote is a stronger signal that the voter promptly saw and backed
106/// the leader — it attests holding the leader's transaction data, not just its
107/// header — so it scores double.
108const STRONG_VOTE_MULTIPLIER: u64 = 2;
109
110/// Compute the per-commit score contribution for the new commit `c`, scoring
111/// the oldest of the 3 pending commits (`c_minus_3`). Returns per-authority
112/// score deltas indexed by `AuthorityIndex`.
113///
114/// Distributed-vote scoring with propagation lookahead bounded at r+2: for
115/// each authority A, `contribution[A]` is the sum of stake of authorities
116/// whose round-(r+2) blocks strongly link to A's round-(r+1) voting block,
117/// where A's voting block strongly links to `c_minus_3`'s leader block (round
118/// r). The round-(r+1) and round-(r+2) blocks are collected from all three
119/// commits following `c_minus_3` (`c_minus_2`, `c_minus_1`, `c`).
120///
121/// A voting block that is a *strong* vote for the leader has its contribution
122/// multiplied by [`STRONG_VOTE_MULTIPLIER`].
123///
124/// Equivocation: if A has multiple voting blocks at r+1 within that window,
125/// `contribution[A] = 0`.
126pub(crate) fn compute_per_commit_contribution(
127    context: &Context,
128    c_minus_3: &SubDagBase,
129    c_minus_2: &SubDagBase,
130    c_minus_1: &SubDagBase,
131    c: &SubDagBase,
132) -> Vec<u64> {
133    let committee = &context.committee;
134    let leader_ref = c_minus_3.leader;
135    let r = leader_ref.round;
136    let vote_round = r + 1;
137    let certify_round = r + 2;
138
139    // Voting blocks at round r+1 that strongly link to the leader block, grouped
140    // by author, each tagged with whether it is a strong vote for the leader.
141    // Multiple blocks per author indicates equivocation in the lookback window.
142    let mut voting_blocks_by_author: BTreeMap<AuthorityIndex, Vec<(BlockRef, bool)>> =
143        BTreeMap::new();
144    for commit in [c_minus_2, c_minus_1, c] {
145        for header in &commit.headers {
146            if header.round() != vote_round {
147                continue;
148            }
149            if !header.ancestors().contains(&leader_ref) {
150                continue;
151            }
152            voting_blocks_by_author
153                .entry(header.author())
154                .or_default()
155                .push((
156                    header.reference(),
157                    header.is_strong_vote_for(leader_ref.author),
158                ));
159        }
160    }
161
162    // Voting-block ref → author, for authors with exactly one voting block.
163    // Equivocation in the lookback window → excluded, zero contribution.
164    // `strong_voter[A]` is true when A's single voting block is a strong vote.
165    let mut voting_author_by_ref: HashMap<BlockRef, AuthorityIndex> = HashMap::new();
166    let mut strong_voter = vec![false; committee.size()];
167    for (author, voting_refs) in &voting_blocks_by_author {
168        if let [(voting_ref, strong)] = voting_refs.as_slice() {
169            voting_author_by_ref.insert(*voting_ref, *author);
170            strong_voter[author.value()] = *strong;
171        }
172    }
173
174    // One pass over the round-(r+2) blocks' ancestors. The aggregator dedups
175    // by certifier so an equivocating certifier's stake counts once per author.
176    let mut certifying_stake: Vec<StakeAggregator<QuorumThreshold>> = (0..committee.size())
177        .map(|_| StakeAggregator::new())
178        .collect();
179    for commit in [c_minus_2, c_minus_1, c] {
180        for header in &commit.headers {
181            if header.round() != certify_round {
182                continue;
183            }
184            for ancestor in header.ancestors() {
185                if let Some(author) = voting_author_by_ref.get(ancestor) {
186                    certifying_stake[author.value()].add(header.author(), committee);
187                }
188            }
189        }
190    }
191
192    certifying_stake
193        .iter()
194        .enumerate()
195        .map(|(author, agg)| {
196            if strong_voter[author] {
197                agg.stake() * STRONG_VOTE_MULTIPLIER
198            } else {
199                agg.stake()
200            }
201        })
202        .collect()
203}

293    /// Helper: build 4 fully-connected commits using `DagBuilder` and return
294    /// their `SubDagBase`s in order [commit_1, commit_2, commit_3, commit_4].
295    fn build_four_commits(context: Arc<Context>) -> Vec<SubDagBase> {
296        let mut dag_builder = DagBuilder::new(context);
297        dag_builder.layers(1..=4).build();
298        dag_builder
299            .get_sub_dag_and_commits(1..=4)
300            .into_iter()
301            .map(|(sub_dag, _)| sub_dag.base)
302            .collect()
303    }

306    async fn test_compute_per_commit_contribution_full_connectivity() {
307        telemetry_subscribers::init_for_testing();
308        let context = Arc::new(Context::new_for_test(4).0);
309        let subdags = build_four_commits(context.clone());
310        assert_eq!(subdags.len(), 4);
311
312        let contribution = compute_per_commit_contribution(

314            &subdags[0],
315            &subdags[1],
316            &subdags[2],
317            &subdags[3],
318        );
319
320        // In a fully-connected DAG, every authority votes for commit-1's leader and
321        // every round-(r+2) block certifies every voting block. So every authority's
322        // contribution equals the full committee stake — doubled, since every

328            "expected each authority to get full committee stake as contribution"
329        );

333    async fn test_compute_per_commit_contribution_dedups_equivocating_certifier() {
334        // An equivocating certifier — two round-(r+2) blocks from one author,
335        // both linking the same voting block — contributes its stake once, not
336        // once per block.
337        let context = Arc::new(Context::new_for_test(4).0);
338        let committee = &context.committee;
339        let r: Round = 10;
340
341        // Minimal subdag: the scorer reads only the leader round and the
342        // headers' ancestor links.
343        let subdag = |leader_round: Round, headers: Vec<VerifiedBlockHeader>| SubDagBase {
344            leader: BlockRef::new(
345                leader_round,
346                AuthorityIndex::new_for_test(0),
347                BlockHeaderDigest::MIN,
348            ),
349            headers,
350            committed_header_refs: vec![],
351            timestamp_ms: 0,
352            commit_ref: CommitRef::new(0, CommitDigest::MIN),
353            reputation_scores_desc: vec![],
354        };
355
356        let c_minus_3 = subdag(r, vec![]);
357
358        // Authority 0 casts a single vote for the leader at round r+1.
359        let voting_block = VerifiedBlockHeader::new_for_test(
360            TestBlockHeader::new(r + 1, 0)
361                .set_ancestors(vec![c_minus_3.leader])
362                .build(),
363        );
364        let voting_ref = voting_block.reference();
365
366        // Authority 1 equivocates at round r+2 with two distinct blocks, each
367        // certifying the voting block.
368        let cert_a = VerifiedBlockHeader::new_for_test(
369            TestBlockHeader::new(r + 2, 1)
370                .set_ancestors(vec![voting_ref])
371                .set_timestamp_ms(1)
372                .build(),
373        );
374        let cert_b = VerifiedBlockHeader::new_for_test(
375            TestBlockHeader::new(r + 2, 1)
376                .set_ancestors(vec![voting_ref])
377                .set_timestamp_ms(2)
378                .build(),
379        );
380
381        let c_minus_2 = subdag(r + 1, vec![voting_block]);
382        let c_minus_1 = subdag(r + 2, vec![cert_a, cert_b]);
383        let c = subdag(r + 3, vec![]);
384
385        let scores =
386            compute_per_commit_contribution(&context, &c_minus_3, &c_minus_2, &c_minus_1, &c);
387
388        let mut expected = vec![0u64; committee.size()];
389        expected[0] = committee.stake(AuthorityIndex::new_for_test(1));
390        assert_eq!(
391            scores, expected,
392            "equivocating certifier's stake must be counted once, not per block"
393        );
394    }

396    #[tokio::test]
397    async fn test_compute_per_commit_contribution_attributes_partial_certification() {
398        // Each voting author is certified by a different subset of round-(r+2)
399        // blocks, so every authority expects a distinct score and any
400        // misattribution is visible.
401        let context = Arc::new(Context::new_for_test(4).0);
402        let committee = &context.committee;
403        let r: Round = 10;
404
405        let subdag = |leader_round: Round, headers: Vec<VerifiedBlockHeader>| SubDagBase {
406            leader: BlockRef::new(
407                leader_round,
408                AuthorityIndex::new_for_test(0),
409                BlockHeaderDigest::MIN,
410            ),
411            headers,
412            committed_header_refs: vec![],
413            timestamp_ms: 0,
414            commit_ref: CommitRef::new(0, CommitDigest::MIN),
415            reputation_scores_desc: vec![],
416        };
417
418        let c_minus_3 = subdag(r, vec![]);
419
420        // Authorities 0, 1, 2 vote for the leader at round r+1; authority 3's
421        // round-(r+1) block does not link the leader, so it is not a vote.
422        let vote = |author: u8| {
423            VerifiedBlockHeader::new_for_test(
424                TestBlockHeader::new(r + 1, author)
425                    .set_ancestors(vec![c_minus_3.leader])
426                    .build(),
427            )
428        };
429        let vote_0 = vote(0);
430        let vote_1 = vote(1);
431        let vote_2 = vote(2);
432        let non_vote_3 = VerifiedBlockHeader::new_for_test(TestBlockHeader::new(r + 1, 3).build());
433
434        // Certifiers at round r+2: authority 0 links all round-(r+1) blocks
435        // (including the non-vote, which must credit no one), authority 1
436        // links votes {0, 1}, authority 2 links vote {0} only.
437        let cert = |author: u8, ancestors: Vec<BlockRef>| {
438            VerifiedBlockHeader::new_for_test(
439                TestBlockHeader::new(r + 2, author)
440                    .set_ancestors(ancestors)
441                    .build(),
442            )
443        };
444        let cert_0 = cert(
445            0,
446            vec![
447                vote_0.reference(),
448                vote_1.reference(),
449                vote_2.reference(),
450                non_vote_3.reference(),
451            ],
452        );
453        let cert_1 = cert(1, vec![vote_0.reference(), vote_1.reference()]);
454        let cert_2 = cert(2, vec![vote_0.reference()]);
455
456        // Spread the blocks across the three commits following `c_minus_3`.
457        let c_minus_2 = subdag(r + 1, vec![vote_0, vote_1, vote_2, non_vote_3]);
458        let c_minus_1 = subdag(r + 2, vec![cert_0, cert_1]);
459        let c = subdag(r + 3, vec![cert_2]);
460
461        let scores =
462            compute_per_commit_contribution(&context, &c_minus_3, &c_minus_2, &c_minus_1, &c);
463
464        let stake = |index: u8| committee.stake(AuthorityIndex::new_for_test(index));
465        let expected = vec![
466            stake(0) + stake(1) + stake(2),
467            stake(0) + stake(1),
468            stake(0),
469            0,
470        ];
471        assert_eq!(
472            scores, expected,
473            "each voting author must be credited exactly its own certifiers' stake"
474        );
475    }

477    #[tokio::test]
478    async fn test_compute_per_commit_contribution_doubles_strong_vote() {
479        telemetry_subscribers::init_for_testing();
480        // Strong votes score double.
481        let context = Arc::new(Context::new_for_test(4).0);
482        let committee = &context.committee;
483        let r: Round = 10;
484        let leader_authority = AuthorityIndex::new_for_test(0);
485
486        let subdag = |leader_round: Round, headers: Vec<VerifiedBlockHeader>| SubDagBase {
487            leader: BlockRef::new(leader_round, leader_authority, BlockHeaderDigest::MIN),
488            headers,
489            committed_header_refs: vec![],
490            timestamp_ms: 0,
491            commit_ref: CommitRef::new(0, CommitDigest::MIN),
492            reputation_scores_desc: vec![],
493        };
494
495        let c_minus_3 = subdag(r, vec![]);
496
497        // Authority 1 casts an ordinary vote; authority 2 casts a strong vote
498        // for the same leader. Both link the leader at round r+1.
499        let ordinary_vote = VerifiedBlockHeader::new_for_test(
500            TestBlockHeader::new(r + 1, 1)
501                .set_ancestors(vec![c_minus_3.leader])
502                .build(),
503        );
504        let strong_vote = VerifiedBlockHeader::new_for_test(
505            TestBlockHeader::new(r + 1, 2)
506                .set_version(TestBlockHeaderVersion::V2)
507                .set_ancestors(vec![c_minus_3.leader])
508                .set_strong_vote(Some(StrongVote {
509                    leader_authority,
510                    missing: AuthoritySet::new(),
511                }))
512                .build(),
513        );
514        let ordinary_ref = ordinary_vote.reference();
515        let strong_ref = strong_vote.reference();
516
517        // One certifier (authority 3) certifies both voting blocks at r+2, so
518        // both voters collect the same certifying stake — the only difference
519        // is that the strong vote is doubled.
520        let certifier = VerifiedBlockHeader::new_for_test(
521            TestBlockHeader::new(r + 2, 3)
522                .set_ancestors(vec![ordinary_ref, strong_ref])
523                .build(),
524        );
525
526        let c_minus_2 = subdag(r + 1, vec![ordinary_vote, strong_vote]);
527        let c_minus_1 = subdag(r + 2, vec![certifier]);
528        let c = subdag(r + 3, vec![]);
529
530        let scores =
531            compute_per_commit_contribution(&context, &c_minus_3, &c_minus_2, &c_minus_1, &c);
532
533        let stake_3 = committee.stake(AuthorityIndex::new_for_test(3));
534        assert_eq!(
535            scores[1], stake_3,
536            "ordinary vote earns the certifying stake"
537        );
538        assert_eq!(scores[2], 2 * stake_3, "strong vote earns double");
539    }
