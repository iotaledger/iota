34        self.stop.send(()).ok();
35        self.handle.await.ok();

39pub(crate) struct LeaderTimeoutTask<D: CoreThreadDispatcher> {
40    dispatcher: Arc<D>,
41    transactions_synchronizer: Arc<TransactionsSynchronizerHandle>,
42    new_round_receiver: watch::Receiver<Round>,
43    new_block_receiver: broadcast::Receiver<VerifiedBlock>,
44    leader_timeout: Duration,
45    min_block_delay: Duration,
46    soft_leader_timeout: Duration,
47    starfish_speed_enabled: bool,
48    stop: Receiver<()>,
49}
50
51impl<D: CoreThreadDispatcher> LeaderTimeoutTask<D> {
52    /// Starts the leader timeout task, which monitors and manages the leader
53    /// election timeout mechanism.
54    pub fn start(
55        dispatcher: Arc<D>,
56        transactions_synchronizer: Arc<TransactionsSynchronizerHandle>,
57        signals_receivers: &CoreSignalsReceivers,
58        context: Arc<Context>,
59    ) -> LeaderTimeoutTaskHandle {
60        let (stop_sender, stop) = tokio::sync::oneshot::channel();
61        let mut me = Self {
62            dispatcher,
63            transactions_synchronizer,
64            stop,
65            new_block_receiver: signals_receivers.block_broadcast_receiver(),
66            new_round_receiver: signals_receivers.new_round_receiver(),
67            leader_timeout: context.parameters.leader_timeout,
68            min_block_delay: context.parameters.min_block_delay,
69            soft_leader_timeout: context.parameters.soft_leader_timeout,
70            starfish_speed_enabled: context.protocol_config.consensus_starfish_speed(),
71        };
72        let handle = tokio::spawn(async move { me.run().await });
73
74        LeaderTimeoutTaskHandle {
75            handle,
76            stop: stop_sender,
77        }
78    }
79
80    /// Runs the leader timeout task, managing the leader election timeout
81    /// mechanism in an asynchronous loop.
82    /// This mechanism ensures that if the current leader fails to produce a new
83    /// block within the specified timeout, the task forces the creation of a
84    /// new block, maintaining the continuity and robustness of the leader
85    /// election process.
86    /// In addition, if min block delay timeout is expired it attempts to
87    /// non-forcefully create a new block.
88    async fn run(&mut self) {
89        debug!("LeaderTimeoutTask is running");
90        let new_clock_round = &mut self.new_round_receiver;
91        let new_block = &mut self.new_block_receiver.resubscribe();
92        let mut clock_round: Round = *new_clock_round.borrow_and_update();
93        let mut last_own_block_round: Option<Round> = None;
94        let mut min_block_delay_timed_out = false;
95        let mut soft_timed_out = false;
96        let mut max_leader_round_timed_out = false;
97        let timer_start = Instant::now();
98        let min_block_delay_timeout = sleep_until(timer_start + self.min_block_delay);
99        let soft_timeout = sleep_until(timer_start + self.soft_leader_timeout);
100        let max_leader_timeout = sleep_until(timer_start + self.leader_timeout);
101
102        tokio::pin!(min_block_delay_timeout);
103        tokio::pin!(soft_timeout);
104        tokio::pin!(max_leader_timeout);
105
106        loop {
107            tokio::select! {
108                // When the min block delay timer expires, then we attempt to trigger the creation of a new block.
109                // If we already timed out before then, the branch gets disabled so we don't attempt
110                // all the time to produce already produced blocks for that round.
111
112                () = &mut min_block_delay_timeout, if !min_block_delay_timed_out && last_own_block_round.is_some() => {
113                    let next_round: Round = last_own_block_round.expect("We should expect some last own round") + 1;
114                    if Self::dispatch_new_block(&self.dispatcher, &self.transactions_synchronizer, next_round, ReasonToCreateBlock::MinBlockDelayTimeout).await {
115                        return;
116                    }
117                    min_block_delay_timed_out = true;
118                },
119                // When the soft leader timeout expires, attempt block creation without requiring
120                // a strong-vote quorum. Only active when starfish speed is enabled.
121                () = &mut soft_timeout, if !soft_timed_out && self.starfish_speed_enabled => {
122                    if Self::dispatch_new_block(&self.dispatcher, &self.transactions_synchronizer, clock_round, ReasonToCreateBlock::SoftTimeout).await {
123                        return;
124                    }
125                    soft_timed_out = true;
126                },
127                // When the max leader timer expires then we attempt to trigger the creation of a new block. This
128                // call is made with reason MaxLeaderTimeout to bypass any checks that allow to propose immediately if block
129                // not already produced.
130                () = &mut max_leader_timeout, if !max_leader_round_timed_out => {
131                    if Self::dispatch_new_block(&self.dispatcher, &self.transactions_synchronizer, clock_round, ReasonToCreateBlock::MaxLeaderTimeout).await {
132                        return;
133                    }
134                    max_leader_round_timed_out = true;
135                }
136
137                // A clock round has been advanced. Reset the leader timeout.
138                Ok(_) = new_clock_round.changed() => {
139                    clock_round = *new_clock_round.borrow_and_update();
140                    debug!("New clock round has been received {clock_round}, resetting timer");
141                    let _span = tracing::trace_span!("new_consensus_round_received", round = ?clock_round).entered();
142
143                    max_leader_round_timed_out = false;
144                    soft_timed_out = false;
145
146                    let now = Instant::now();
147
148                    max_leader_timeout
149                    .as_mut()
150                    .reset(now + self.leader_timeout);
151                    soft_timeout
152                    .as_mut()
153                    .reset(now + self.soft_leader_timeout);
154                },
155                 // A new block was created. Set a timer in min_block_delay
156                Ok(block) = new_block.recv() => {
157                    debug!("New block {:?} was created and seen in leader timeout task", block.verified_block_header);
158                    last_own_block_round = Some(block.round());
159
160                    min_block_delay_timed_out = false;
161                    soft_timed_out = false;
162
163                    let now = Instant::now();
164                    min_block_delay_timeout.as_mut().reset(now + self.min_block_delay);
165                    soft_timeout.as_mut().reset(now + self.soft_leader_timeout);
166                },
167                _ = &mut self.stop => {
168                    debug!("Stop signal has been received, now shutting down");
169                    return;
170                }
171            }
172        }
173    }
174
175    /// Request a new block and fetch any missing committed transactions.
176    /// Returns `true` if the dispatcher is shutting down.
177    async fn dispatch_new_block(
178        dispatcher: &D,
179        transactions_synchronizer: &TransactionsSynchronizerHandle,
180        round: Round,
181        reason: ReasonToCreateBlock,
182    ) -> bool {
183        match dispatcher.new_block(round, reason).await {
184            Ok(missing_committed_txns) => {
185                if !missing_committed_txns.is_empty() {
186                    debug!(
187                        "Missing committed transactions after creating new block: {missing_committed_txns:?}"
188                    );
189                    if let Err(err) = transactions_synchronizer
190                        .fetch_transactions(missing_committed_txns)
191                        .await
192                    {
193                        warn!(
194                            "Error while trying to fetch missing transactions via transactions synchronizer: {err}"
195                        );
196                    }
197                }
198                false
199            }
200            Err(err) => {
201                warn!(
202                    "Error received while calling dispatcher, probably dispatcher is shutting down, will now exit: {err:?}"
203                );
204                true
205            }
206        }
207    }
208}

300    #[rstest]
301    #[tokio::test(flavor = "current_thread", start_paused = true)]
302    async fn basic_leader_timeout(#[values(false, true)] starfish_speed: bool) {
303        telemetry_subscribers::init_for_testing();
304        let (mut context, _signers) = Context::new_for_test(4);
305        context
306            .protocol_config
307            .set_consensus_starfish_speed_for_testing(starfish_speed);
308        let dispatcher = Arc::new(MockCoreThreadDispatcher::default());
309        let leader_timeout = Duration::from_millis(500);
310        let min_block_delay = Duration::from_millis(50);
311        // Chosen so no timer deadline coincides with a sleep below — a tie
312        // would make the drained call sequence racy.
313        let soft_leader_timeout = Duration::from_millis(200);
314        let parameters = Parameters {
315            leader_timeout,
316            min_block_delay,
317            soft_leader_timeout,
318            ..Default::default()
319        };
320        let context = Arc::new(context.with_parameters(parameters));
321        let start = Instant::now();
322
323        let (mut signals, signal_receivers) = CoreSignals::new(context.clone());
324        let transactions_synchronizer = TransactionsSynchronizer::start(
325            Arc::new(FakeNetworkClient::default()),
326            context.clone(),
327            dispatcher.clone(),
328            Arc::new(RwLock::new(DagState::new(
329                context.clone(),
330                Arc::new(MemStore::new()),
331            ))),
332            Arc::new(NoopBlockVerifier),
333        );
334
335        // spawn the task
336        let _handle = LeaderTimeoutTask::start(
337            dispatcher.clone(),
338            transactions_synchronizer,
339            &signal_receivers,
340            context.clone(),
341        );
342
343        // send a signal that own block was created at round 8 to initialize the
344        // broadcaster
345        let mut encoder = create_encoder(&context);
346        let input_block = VerifiedBlock::new_for_test(
347            TestBlockHeader::new_with_commitment(8, 0, &context, &mut encoder).build(),
348        );
349        // send a signal that a new round has been advanced to initialize the watcher.
350        signals.new_round(9);
351
352        signals
353            .new_block(input_block)
354            .expect("We should expect correct sending a new block");
355        sleep(min_block_delay / 2).await;
356        // send a signal that own block was created at round 9, which will start min
357        // block delay timeout
358        let input_block = VerifiedBlock::new_for_test(
359            TestBlockHeader::new_with_commitment(9, 0, &context, &mut encoder).build(),
360        );
361        signals
362            .new_block(input_block)
363            .expect("We should expect correct sending a new block");
364        // send a signal that a new round has been advanced.
365        signals.new_round(10);
366
367        // wait enough until the min block delay has passed and a new_block call is
368        // triggered
369        sleep(2 * min_block_delay).await;
370        let all_calls = dispatcher.get_new_block_calls().await;
371        assert_eq!(all_calls.len(), 1);
372
373        let (round, reason, timestamp) = all_calls[0];
374        assert_eq!(round, 10);
375        assert_eq!(reason, ReasonToCreateBlock::MinBlockDelayTimeout);
376        assert!(
377            min_block_delay <= timestamp - start,
378            "Leader timeout min setting {:?} should be less than actual time difference {:?}",
379            min_block_delay,
380            timestamp - start
381        );
382
383        // wait enough until a new_block has been received
384        sleep(2 * leader_timeout).await;
385        let all_calls = dispatcher.get_new_block_calls().await;
386        // With starfish speed enabled the soft leader timeout fires before the
387        // max leader timeout.
388        let expected = if starfish_speed {
389            vec![
390                (ReasonToCreateBlock::SoftTimeout, soft_leader_timeout),
391                (ReasonToCreateBlock::MaxLeaderTimeout, leader_timeout),
392            ]
393        } else {
394            vec![(ReasonToCreateBlock::MaxLeaderTimeout, leader_timeout)]
395        };
396        assert_eq!(all_calls.len(), expected.len());
397        for ((round, reason, timestamp), (expected_reason, timeout)) in
398            all_calls.iter().zip(&expected)
399        {
400            assert_eq!(*round, 10);
401            assert_eq!(reason, expected_reason);
402            assert!(
403                *timeout <= *timestamp - start,
404                "Timeout setting {:?} should be less than actual time difference {:?}",
405                timeout,
406                *timestamp - start
407            );
408        }
409
410        // now wait another 2 * leader_timeout, no other call should be received
411        sleep(2 * leader_timeout).await;
412        let all_calls = dispatcher.get_new_block_calls().await;
413
414        assert_eq!(all_calls.len(), 0);
415    }

506                (ReasonToCreateBlock::MinBlockDelayTimeout, min_block_delay),

508            ]
509        };
510        assert_eq!(all_calls.len(), expected.len());
511        for ((round, reason, timestamp), (expected_reason, timeout)) in

513        {
514            assert_eq!(*round, 15);
515            assert_eq!(reason, expected_reason);
516            assert!(*timeout < *timestamp - now);
517        }
518    }
