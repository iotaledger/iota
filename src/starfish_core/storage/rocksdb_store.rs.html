36/// Persistent storage with RocksDB.
37#[derive(DBMapUtils)]
38pub(crate) struct RocksDBStore {
39    /// Stores SignedBlockHeader by refs.
40    block_headers: DBMap<(Round, AuthorityIndex, BlockHeaderDigest), Bytes>,
41    /// Stores Transactions by transaction refs
42    transactions_by_tx_refs: DBMap<(Round, AuthorityIndex, TransactionsCommitment), Bytes>,
43    /// A secondary index that orders refs first by authors.
44    #[rename = "digests"]
45    digests_by_authorities: DBMap<(AuthorityIndex, Round, BlockHeaderDigest), ()>,
46    /// A secondary index that orders transaction commitments first by authors.
47    transaction_commitments_by_authorities:
48        DBMap<(AuthorityIndex, Round, TransactionsCommitment), ()>,
49    /// Maps commit index to Commit.
50    commits: DBMap<(CommitIndex, CommitDigest), Bytes>,
51    /// Collects votes on commits.
52    /// TODO: batch multiple votes into a single row.
53    commit_votes: DBMap<(CommitIndex, CommitDigest, BlockRef), ()>,
54    /// Stores info related to Commit that helps recovery.
55    commit_info: DBMap<(CommitIndex, CommitDigest), CommitInfo>,
56    /// Stores voting block headers separately from regular block headers.
57    /// These are block headers that contain commit votes used to certify
58    /// commits.
59    voting_block_headers: DBMap<(Round, AuthorityIndex, BlockHeaderDigest), Bytes>,
60
61    fast_commit_sync_flag: DBMap<(), ()>,
62
63    /// Stores scoring metrics for each authority.
64    misbehavior_counts: DBMap<AuthorityIndex, MisbehaviorCounts>,
65}

80    /// Creates a new instance of RocksDB storage.
81    pub(crate) fn new(path: &str) -> Self {
82        // Consensus data has high write throughput (all transactions) and is rarely
83        // read (only during recovery and when helping peers catch up).
84        let db_options = default_db_options().optimize_db_for_write_throughput(2);
85        let mut metrics_conf = MetricConf::new("consensus");
86        metrics_conf.read_sample_interval = SamplingInterval::new(Duration::from_secs(60), 0);
87        let cf_options = default_db_options().optimize_for_write_throughput();
88        let column_family_options = DBMapTableConfigMap::new(BTreeMap::from([
89            (
90                Self::TRANSACTIONS_BY_TX_REF_CF.to_string(),
91                default_db_options()
92                    .optimize_for_write_throughput_no_deletion()
93                    // Using larger block is ok since there is not much point reads on the cf.
94                    .set_block_options(512, 128 << 10),
95            ),
96            (
97                Self::BLOCK_HEADERS_CF.to_string(),
98                default_db_options()
99                    .optimize_for_write_throughput_no_deletion()
100                    // TODO:think about these constants, for now it is a copy from blocks
101                    .set_block_options(512, 128 << 10),
102            ),
103            (
104                Self::DIGESTS_BY_AUTHORITIES_CF.to_string(),
105                cf_options.clone(),
106            ),
107            (
108                Self::TRANSACTION_COMMITMENTS_BY_AUTHORITIES_CF.to_string(),
109                cf_options.clone(),
110            ),
111            (Self::COMMITS_CF.to_string(), cf_options.clone()),
112            (Self::COMMIT_VOTES_CF.to_string(), cf_options.clone()),
113            (Self::COMMIT_INFO_CF.to_string(), cf_options.clone()),
114            // Voting block headers are much fewer than regular block headers,
115            // so using standard options is sufficient.
116            (
117                Self::VOTING_BLOCK_HEADERS_CF.to_string(),
118                cf_options.clone(),
119            ),
120            (
121                Self::FAST_COMMIT_SYNC_FLAG_CF.to_string(),
122                cf_options.clone(),
123            ),
124            (Self::MISBEHAVIOR_COUNTS_CF.to_string(), cf_options),
125        ]));
126        Self::open_tables_read_write(
127            path.into(),
128            metrics_conf,
129            Some(db_options.options),
130            Some(column_family_options),
131        )
132    }
133}

135impl Store for RocksDBStore {
136    fn write(&self, write_batch: WriteBatch) -> ConsensusResult<()> {
137        fail_point!("consensus-store-before-write");
138        // TODO: does it matter which CF we use here?
139        let mut batch = self.block_headers.batch();
140
141        // Store block headers and their associated commit votes
142        for block_header in write_batch.block_headers {
143            let block_ref = block_header.reference();
144            debug!("block header {} pushed to store", block_header);
145            // Store the block header
146            batch
147                .insert_batch(
148                    &self.block_headers,
149                    [(
150                        (block_ref.round, block_ref.author, block_ref.digest),
151                        block_header.serialized(),
152                    )],
153                )
154                .map_err(ConsensusError::RocksDBFailure)?;
155            // Store the authority digest
156            batch
157                .insert_batch(
158                    &self.digests_by_authorities,
159                    [((block_ref.author, block_ref.round, block_ref.digest), ())],
160                )
161                .map_err(ConsensusError::RocksDBFailure)?;
162            // Store commit votes from this block header using the BlockHeaderAPI trait
163            for vote in block_header.commit_votes() {
164                batch
165                    .insert_batch(
166                        &self.commit_votes,
167                        [((vote.index, vote.digest, block_ref), ())],
168                    )
169                    .map_err(ConsensusError::RocksDBFailure)?;
170            }
171        }
172
173        // Store transactions data
174        for transaction in write_batch.transactions {
175            let transaction_ref = transaction.transaction_ref();
176            batch
177                .insert_batch(
178                    &self.transactions_by_tx_refs,
179                    [(
180                        (
181                            transaction_ref.round,
182                            transaction_ref.author,
183                            transaction_ref.transactions_commitment,
184                        ),
185                        transaction.serialized(),
186                    )],
187                )
188                .map_err(ConsensusError::RocksDBFailure)?;
189            // Store the authority digest
190            batch
191                .insert_batch(
192                    &self.transaction_commitments_by_authorities,
193                    [(
194                        (
195                            transaction_ref.author,
196                            transaction_ref.round,
197                            transaction_ref.transactions_commitment,
198                        ),
199                        (),
200                    )],
201                )
202                .map_err(ConsensusError::RocksDBFailure)?;
203        }
204
205        // Handle commits
206        for commit in write_batch.commits {
207            batch
208                .insert_batch(
209                    &self.commits,
210                    [((commit.index(), commit.digest()), commit.serialized())],
211                )
212                .map_err(ConsensusError::RocksDBFailure)?;
213        }
214
215        // Handle commit info
216        for (commit_ref, commit_info) in write_batch.commit_info {
217            batch
218                .insert_batch(
219                    &self.commit_info,
220                    [((commit_ref.index, commit_ref.digest), commit_info)],
221                )
222                .map_err(ConsensusError::RocksDBFailure)?;
223        }
224
225        // Handle voting block headers
226        for header in write_batch.voting_block_headers {
227            let block_ref = header.reference();
228            batch
229                .insert_batch(
230                    &self.voting_block_headers,
231                    [(
232                        (block_ref.round, block_ref.author, block_ref.digest),
233                        header.serialized().clone(),
234                    )],
235                )
236                .map_err(ConsensusError::RocksDBFailure)?;
237            // Store commit votes from this block header
238            for vote in header.commit_votes() {
239                batch
240                    .insert_batch(
241                        &self.commit_votes,
242                        [((vote.index, vote.digest, block_ref), ())],
243                    )
244                    .map_err(ConsensusError::RocksDBFailure)?;
245            }
246        }
247
248        if let Some(flag) = write_batch.fast_commit_sync_flag {
249            if flag {
250                batch
251                    .insert_batch(&self.fast_commit_sync_flag, [((), ())])
252                    .map_err(ConsensusError::RocksDBFailure)?;
253            } else {
254                batch
255                    .delete_batch(&self.fast_commit_sync_flag, [()])
256                    .map_err(ConsensusError::RocksDBFailure)?;
257            }
258        }
259
260        batch
261            .insert_batch(&self.misbehavior_counts, write_batch.misbehavior_counts)
262            .map_err(ConsensusError::RocksDBFailure)?;
263
264        batch.write()?;
265        fail_point!("consensus-store-after-write");
266        Ok(())
267    }

269    fn read_blocks(&self, refs: &[BlockRef]) -> ConsensusResult<Vec<Option<VerifiedBlock>>> {
270        // Get both headers and transactions for the given references
271        let headers = self.read_verified_block_headers(refs)?;
272        let tx_refs = headers
273            .iter()
274            .map(|vh| {
275                if vh.is_none() {
276                    GenericTransactionRef::TransactionRef(TransactionRef::default())
277                } else {
278                    GenericTransactionRef::TransactionRef(vh.as_ref().unwrap().transaction_ref())
279                }
280            })
281            .collect::<Vec<GenericTransactionRef>>();
282        let transactions = self.read_verified_transactions(tx_refs.as_slice())?;
283
284        // Combine them into blocks if both parts exist
285        let mut blocks = Vec::with_capacity(refs.len());
286        for (header, transactions) in headers.into_iter().zip(transactions) {
287            match (header, transactions) {
288                (Some(hdr), Some(txs)) => {
289                    blocks.push(Some(VerifiedBlock::new(hdr, txs)));
290                }
291                _ => blocks.push(None),
292            }
293        }
294        Ok(blocks)
295    }

297    /// Return Verified Block Headers by reading their entries from storage and
298    /// deserializing them
299    fn read_verified_block_headers(
300        &self,
301        refs: &[BlockRef],
302    ) -> ConsensusResult<Vec<Option<VerifiedBlockHeader>>> {
303        let serialized_block_headers = self.read_serialized_block_headers(refs)?;
304        let mut block_headers = vec![];
305        for (key, serialized_block_header) in refs.iter().zip(serialized_block_headers) {
306            if let Some(serialized_block_header) = serialized_block_header {
307                let block_header = VerifiedBlockHeader::new_from_bytes(serialized_block_header)?;
308
309                // Makes sure block data is not corrupted by comparing digests.
310                assert_eq!(*key, block_header.reference());
311                block_headers.push(Some(block_header));
312            } else {
313                block_headers.push(None);
314            }
315        }
316        Ok(block_headers)
317    }

332    /// Return Verified Transactions by reading both transactions and headers
333    /// from storage, deserializing them and assembling into the required output
334    fn read_verified_transactions(
335        &self,
336        refs: &[GenericTransactionRef],
337    ) -> ConsensusResult<Vec<Option<CommitmentVerifiedTransactions>>> {
338        if !check_ref_consistency(refs) {
339            return Err(ConsensusError::InconsistentTransactionRefVariants);
340        }
341        if refs.is_empty() {
342            return Ok(vec![]);
343        }
344        let serialized_vec_transactions = self.read_serialized_transactions(refs)?;
345
346        let mut result = Vec::with_capacity(refs.len());
347        for (gen_tx_ref, serialized_transactions) in refs.iter().zip(serialized_vec_transactions) {
348            let GenericTransactionRef::TransactionRef(tx_ref) = gen_tx_ref else {
349                // Legacy variant, never stored under v29.
350                result.push(None);
351                continue;
352            };
353            if let Some(serialized_transactions) = serialized_transactions {
354                let transactions: Vec<Transaction> = bcs::from_bytes(&serialized_transactions)
355                    .map_err(ConsensusError::MalformedTransactions)?;
356                // We don't check the transactions commitment from the header as it's loaded
357                // from storage. Assemble verified transactions
358                let verified_transactions = CommitmentVerifiedTransactions::new(
359                    transactions,
360                    *tx_ref,
361                    None,
362                    serialized_transactions,
363                );
364                result.push(Some(verified_transactions));
365            } else {
366                result.push(None);
367            }
368        }
369        Ok(result)
370    }

372    /// Return Bytes of corresponding Transactions by reading their entries from
373    /// storage
374    fn read_serialized_transactions(
375        &self,
376        refs: &[GenericTransactionRef],
377    ) -> ConsensusResult<Vec<Option<Bytes>>> {
378        if !check_ref_consistency(refs) {
379            return Err(ConsensusError::InconsistentTransactionRefVariants);
380        }
381        if refs.is_empty() {
382            return Ok(vec![]);
383        }
384        // Legacy variant, never stored under v29.
385        if matches!(refs[0], GenericTransactionRef::BlockRef(_)) {
386            return Ok(vec![None; refs.len()]);
387        }
388        let keys: Result<Vec<_>, ConsensusError> = refs
389            .iter()
390            .map(|r| {
391                if let GenericTransactionRef::TransactionRef(tx_ref) = r {
392                    Ok((tx_ref.round, tx_ref.author, tx_ref.transactions_commitment))
393                } else {
394                    Err(ConsensusError::InconsistentTransactionRefVariants)
395                }
396            })
397            .collect();
398        Ok(self.transactions_by_tx_refs.multi_get(keys?)?)
399    }

401    fn contains_transactions(&self, refs: &[GenericTransactionRef]) -> ConsensusResult<Vec<bool>> {
402        if !check_ref_consistency(refs) {
403            return Err(ConsensusError::InconsistentTransactionRefVariants);
404        }
405        if refs.is_empty() {
406            return Ok(vec![]);
407        }
408        // Legacy variant, never stored under v29.
409        if matches!(refs[0], GenericTransactionRef::BlockRef(_)) {
410            return Ok(vec![false; refs.len()]);
411        }
412        let keys: Result<Vec<_>, ConsensusError> = refs
413            .iter()
414            .map(|r| {
415                if let GenericTransactionRef::TransactionRef(tx_ref) = r {
416                    Ok((tx_ref.round, tx_ref.author, tx_ref.transactions_commitment))
417                } else {
418                    Err(ConsensusError::InconsistentTransactionRefVariants)
419                }
420            })
421            .collect();
422        Ok(self.transactions_by_tx_refs.multi_contains_keys(keys?)?)
423    }

483    fn scan_blocks_by_author(
484        &self,
485        author: AuthorityIndex,
486        start_round: Round,
487    ) -> ConsensusResult<Vec<VerifiedBlock>> {
488        let mut refs = vec![];
489        for kv in self.digests_by_authorities.safe_range_iter((
490            Included((author, start_round, BlockHeaderDigest::MIN)),
491            Included((author, Round::MAX, BlockHeaderDigest::MAX)),
492        )) {
493            let ((author, round, digest), _) = kv?;
494            refs.push(BlockRef::new(round, author, digest));
495        }
496        let results = self.read_blocks(refs.as_slice())?;
497        let mut blocks = Vec::with_capacity(refs.len());
498        for (r, block) in refs.into_iter().zip(results) {
499            blocks.push(
500                block.unwrap_or_else(|| panic!("Storage inconsistency: block {r:?} not found!")),
501            );
502        }
503        Ok(blocks)
504    }

506    // The method returns the last `num_of_rounds` rounds blocks by author in round
507    // ascending order. When a `before_round` is defined then the blocks of
508    // round `<=before_round` are returned. If not then the max value for round
509    // will be used as cut off.
510    fn scan_last_blocks_by_author(
511        &self,
512        author: AuthorityIndex,
513        num_of_rounds: u64,
514        before_round: Option<Round>,
515    ) -> ConsensusResult<Vec<VerifiedBlock>> {
516        let before_round = before_round.unwrap_or(Round::MAX);
517        let mut refs = std::collections::VecDeque::new();
518        for kv in self
519            .digests_by_authorities
520            .safe_range_iter_reversed(
521                (author, Round::MIN, BlockHeaderDigest::MIN)
522                    ..=(author, before_round, BlockHeaderDigest::MAX),
523            )
524            .take(num_of_rounds as usize)
525        {
526            let ((author, round, digest), _) = kv?;
527            refs.push_front(BlockRef::new(round, author, digest));
528        }
529        // Use make_contiguous() rather than as_slices().0 so all refs are read
530        // when the VecDeque ring buffer wraps; otherwise trailing entries would
531        // be silently dropped.
532        let refs_slice = refs.make_contiguous();
533        let results = self.read_blocks(refs_slice)?;
534        let mut blocks = vec![];
535        for (r, block) in refs.into_iter().zip(results) {
536            blocks.push(
537                block.unwrap_or_else(|| panic!("Storage inconsistency: block {r:?} not found!")),
538            );
539        }
540        Ok(blocks)
541    }

552    fn read_last_commit(&self) -> ConsensusResult<Option<TrustedCommit>> {
553        let Some(result) = self.commits.safe_range_iter_reversed(..).next() else {
554            return Ok(None);
555        };
556        let ((_index, digest), serialized) = result?;
557        let commit = TrustedCommit::new_trusted(
558            bcs::from_bytes(&serialized).map_err(ConsensusError::MalformedCommit)?,
559            serialized,
560        );
561        assert_eq!(commit.digest(), digest);
562        Ok(Some(commit))
563    }

565    fn scan_commits(&self, range: CommitRange) -> ConsensusResult<Vec<TrustedCommit>> {
566        let mut commits = vec![];
567        for result in self.commits.safe_range_iter((
568            Included((range.start(), CommitDigest::MIN)),
569            Included((range.end(), CommitDigest::MAX)),
570        )) {
571            let ((_index, digest), serialized) = result?;
572            let commit = TrustedCommit::new_trusted(
573                bcs::from_bytes(&serialized).map_err(ConsensusError::MalformedCommit)?,
574                serialized,
575            );
576            assert_eq!(commit.digest(), digest);
577            commits.push(commit);
578        }
579        Ok(commits)
580    }

598    fn read_highest_commit_index_with_votes(
599        &self,
600        up_to_index: CommitIndex,
601    ) -> ConsensusResult<Option<CommitIndex>> {
602        // Do a reverse iteration from up_to_index to find the first entry with votes.
603        // The commit_votes table is keyed by (CommitIndex, CommitDigest, BlockRef).
604        let result = self
605            .commit_votes
606            .safe_range_iter_reversed(
607                (CommitIndex::MIN, CommitDigest::MIN, BlockRef::MIN)
608                    ..=(up_to_index, CommitDigest::MAX, BlockRef::MAX),
609            )
610            .next();
611
612        match result {
613            Some(Ok(((index, _, _), _))) => Ok(Some(index)),
614            Some(Err(e)) => Err(ConsensusError::RocksDBFailure(e)),
615            None => Ok(None),
616        }
617    }

638    fn read_last_commit_info(&self) -> ConsensusResult<Option<(CommitRef, CommitInfo)>> {

640            return Ok(None);
641        };
642        let (key, commit_info) = result.map_err(ConsensusError::RocksDBFailure)?;
643        Ok(Some((CommitRef::new(key.0, key.1), commit_info)))
644    }
645
646    fn read_voting_block_headers(
647        &self,
648        refs: &[BlockRef],
649    ) -> ConsensusResult<Vec<Option<VerifiedBlockHeader>>> {
650        let keys: Vec<_> = refs.iter().map(|r| (r.round, r.author, r.digest)).collect();
651        let results = self
652            .voting_block_headers
653            .multi_get(keys)
654            .map_err(ConsensusError::RocksDBFailure)?;
655        results
656            .into_iter()
657            .map(|r| r.map(VerifiedBlockHeader::new_from_bytes).transpose())
658            .collect()
659    }
660
661    fn read_fast_sync_ongoing(&self) -> ConsensusResult<bool> {
662        self.fast_commit_sync_flag
663            .contains_key(&())
664            .map_err(ConsensusError::RocksDBFailure)
665    }
666
667    fn scan_serialized_transactions(
668        &self,
669        refs: &BTreeSet<TransactionRef>,
670        byte_budget: usize,
671    ) -> ConsensusResult<BTreeMap<TransactionRef, Bytes>> {
672        let Some((lower, upper)) = transaction_scan_bounds(refs) else {
673            return Ok(BTreeMap::new());
674        };
675        collect_transactions_within_budget(
676            refs,
677            byte_budget,
678            self.transactions_by_tx_refs
679                .safe_range_iter(lower..=upper)
680                .map(|entry| entry.map_err(ConsensusError::RocksDBFailure)),
681        )
682    }
683}
684
685/// Returns true if all elements in the slice have the same variant (by
686/// discriminant)
687pub(crate) fn check_ref_consistency(refs: &[GenericTransactionRef]) -> bool {
688    if refs.is_empty() {
689        return true;
690    }
691    let first = std::mem::discriminant(&refs[0]);
692    refs.iter().all(|r| std::mem::discriminant(r) == first)
693}

709        debug!("Deleted all transactions from store");
710        Ok(())
711    }
