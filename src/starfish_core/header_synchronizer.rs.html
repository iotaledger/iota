164impl BlocksGuard {
165    /// Unlocks the given subset of the guarded block refs ahead of the guard
166    /// being dropped.
167    fn release(&mut self, block_refs: &BTreeSet<BlockRef>) {
168        if block_refs.is_empty() {
169            return;
170        }
171        self.map.unlock_headers(block_refs, self.peer);
172        self.block_refs
173            .retain(|block_ref| !block_refs.contains(block_ref));
174    }
175}
176
177impl Drop for BlocksGuard {
178    fn drop(&mut self) {
179        self.map.unlock_headers(&self.block_refs, self.peer);
180    }
181}

183// Keeps a mapping between the missing headers that have been instructed to be
184// fetched and the authorities that are currently fetching them. For a block ref
185// there is a maximum number of authorities that can concurrently fetch it. The
186// authority ids that are currently fetching a block are set on the
187// corresponding `BTreeSet` and basically they act as "locks".
188struct InflightBlockHeadersMap {
189    inner: Mutex<HashMap<BlockRef, BTreeSet<AuthorityIndex>>>,
190}
191
192impl InflightBlockHeadersMap {
193    fn new() -> Arc<Self> {
194        Arc::new(Self {
195            inner: Mutex::new(HashMap::new()),
196        })
197    }
198
199    /// Locks the block headers to be fetched for the assigned `peer_index`. We
200    /// want to avoid re-fetching the missing block headers from too many
201    /// authorities at the same time, thus we limit the concurrency per
202    /// block by attempting to lock per block. If a header is already
203    /// fetched by the maximum allowed number of authorities, then the block
204    /// ref will not be included in the returned set. The method returns all
205    /// the block refs that have been successfully locked and allowed to be
206    /// fetched.
207    ///
208    /// Different limits apply based on the sync method:
209    /// - Periodic sync: Can lock if total authorities <
210    ///   MAX_AUTHORITIES_TO_FETCH_PER_BLOCK_HEADER (3)
211    /// - Live sync: Can lock if total authorities <
212    ///   MAX_AUTHORITIES_TO_LIVE_FETCH_PER_BLOCK_HEADER (1)
213    fn lock_headers(
214        self: &Arc<Self>,
215        missing_block_refs: BTreeSet<BlockRef>,
216        peer: AuthorityIndex,
217        method: SyncMethod,
218    ) -> Option<BlocksGuard> {
219        let mut block_refs = BTreeSet::new();
220        let mut inner = self.inner.lock();
221
222        for block_ref in missing_block_refs {
223            let authorities = inner.entry(block_ref).or_default();
224
225            // Check if this peer is already fetching this header
226            if authorities.contains(&peer) {
227                continue;
228            }
229
230            // Count total authorities currently fetching this header
231            let total_count = authorities.len();
232
233            // Determine the limit based on the sync method
234            let max_limit = method.max_authorities_to_fetch_per_block_header();
235
236            // Check if we can acquire the lock
237            if total_count < max_limit {
238                assert!(authorities.insert(peer));
239                block_refs.insert(block_ref);
240            }
241        }
242
243        if block_refs.is_empty() {
244            None
245        } else {
246            Some(BlocksGuard {
247                map: self.clone(),
248                block_refs,
249                peer,
250                method,
251            })
252        }
253    }
254
255    /// Unlocks the provided block references for the given `peer`. The
256    /// unlocking is strict, meaning that if this method is called for a
257    /// specific block ref and peer more times than the corresponding lock
258    /// has been called, it will panic.
259    fn unlock_headers(self: &Arc<Self>, block_refs: &BTreeSet<BlockRef>, peer: AuthorityIndex) {
260        // Now mark all the blocks as fetched from the map
261        let mut headers_to_fetch = self.inner.lock();
262        for block_ref in block_refs {
263            let authorities = headers_to_fetch
264                .get_mut(block_ref)
265                .expect("Should have found a non empty map");
266
267            assert!(authorities.remove(&peer), "Peer index should be present!");
268
269            // if the last one then just clean up
270            if authorities.is_empty() {
271                headers_to_fetch.remove(block_ref);
272            }
273        }
274    }
275
276    /// Drops the provided `blocks_guard` which will force to unlock the blocks,
277    /// and lock now again the referenced block refs. The swap is best
278    /// effort and there is no guarantee that the `peer` will be able to
279    /// acquire the new locks.
280    fn swap_locks(
281        self: &Arc<Self>,
282        blocks_guard: BlocksGuard,
283        peer: AuthorityIndex,
284    ) -> Option<BlocksGuard> {
285        let block_refs = blocks_guard.block_refs.clone();
286        let method = blocks_guard.method;
287
288        // Explicitly drop the guard
289        drop(blocks_guard);
290
291        // Now create a new guard with the same sync method
292        self.lock_headers(block_refs, peer, method)
293    }
294
295    #[cfg(test)]
296    fn num_of_locked_headers(self: &Arc<Self>) -> usize {
297        let inner = self.inner.lock();
298        inner.len()
299    }
300}

319impl HeaderSynchronizerHandle {
320    /// Explicitly asks from the synchronizer to fetch block headers - provided
321    /// the block_refs set - from the peer authority. The request is sent after
322    /// a short wait, minus whatever the block streams delivered in the
323    /// meantime.
324    pub(crate) async fn fetch_headers(
325        &self,
326        missing_block_refs: BTreeSet<BlockRef>,
327        peer_index: AuthorityIndex,
328    ) -> ConsensusResult<()> {
329        let (sender, receiver) = oneshot::channel();
330        self.commands_sender
331            .send(Command::FetchBlockHeaders {
332                missing_block_refs,
333                peer_index,
334                result: sender,
335            })
336            .await
337            .map_err(|_err| ConsensusError::Shutdown)?;
338        receiver.await.map_err(|_err| ConsensusError::Shutdown)?
339    }
340
341    pub(crate) fn clear_verified_headers_cache(&self) {
342        self.verified_headers_cache.lock().clear();
343    }
344
345    /// Digests of the block headers received over the block streams, filled by
346    /// the authority service.
347    pub(crate) fn received_block_headers(&self) -> &Arc<FilterForHeaders> {
348        &self.received_block_headers
349    }
350
351    pub(crate) async fn stop(&self) -> Result<(), JoinError> {
352        let mut tasks = self.tasks.lock().await;
353        tasks.abort_all();
354        while let Some(result) = tasks.join_next().await {
355            match result {
356                // task finished successfully
357                Ok(_) => (),
358                // task was cancelled, which is expected on shutdown
359                Err(e) if e.is_cancelled() => (),
360                // propagate other errors (e.g. panics)
361                Err(e) => return Err(e),
362            }
363        }
364        Ok(())
365    }

368/// `Synchronizer` oversees live block synchronization, crucial for node
369/// progress. Live synchronization refers to the process of retrieving missing
370/// blocks, particularly those essential for advancing a node when data from
371/// only a few rounds is absent. If a node significantly lags behind the
372/// network, `commit_syncer` handles fetching missing headers via a more
373/// efficient approach. Transactions that are sequenced and included in commit
374/// are fetched through the transaction synchronizer once block headers are
375/// processed. `Synchronizer` aims for swift catch-up employing two
376/// mechanisms:
377///
378/// 1. Explicitly requesting missing headers from designated authorities via the
379///    bundle streaming path. This includes attempting to fetch any missing
380///    ancestors necessary for processing a received bundle of block and
381///    headers. Such requests prioritize the block author, maximizing the chance
382///    of prompt retrieval. A locking mechanism allows concurrent requests for
383///    missing blocks from up to three authorities simultaneously, enhancing the
384///    chances of timely retrieval. Notably, if additional missing blocks arise
385///    during block processing, requests are deferred to the scheduler.
386///
387/// 2. Periodically requesting missing block headers via a scheduler. This
388///    primarily serves to retrieve missing headers that were not ancestors of a
389///    received block bundle via the bundle streaming path. The scheduler
390///    operates on either a fixed periodic basis or is triggered immediately
391///    after explicit fetches described in (1), ensuring continued block
392///    retrieval if gaps persist.
393///
394/// Additionally to the above, the synchronizer can synchronize and fetch the
395/// last own proposed header from the network peers as best effort approach to
396/// recover node from amnesia and avoid making the node equivocate.
397pub(crate) struct HeaderSynchronizer<C: NetworkClient, V: BlockVerifier, D: CoreThreadDispatcher> {
398    context: Arc<Context>,
399    commands_receiver: Receiver<Command>,
400    fetch_block_senders: BTreeMap<AuthorityIndex, Sender<BlocksGuard>>,
401    core_dispatcher: Arc<D>,
402    commit_vote_monitor: Arc<CommitVoteMonitor>,
403    dag_state: Arc<RwLock<DagState>>,
404    transactions_synchronizer: Arc<TransactionsSynchronizerHandle>,
405    fetch_block_headers_scheduler_task: JoinSet<()>,
406    fetch_own_last_header_task: JoinSet<()>,
407    network_client: Arc<C>,
408    block_verifier: Arc<V>,
409    inflight_block_headers_map: Arc<InflightBlockHeadersMap>,
410    verified_headers_cache: Arc<Mutex<LruCache<BlockHeaderDigest, ()>>>,
411    commands_sender: Sender<Command>,
412    /// Present only when the fast commit syncer is running at this
413    /// deployment. When it reads `true`, the header synchronizer skips
414    /// dispatching new fetches (explicit commands and periodic scheduler)
415    /// because fast sync is about to supply the corresponding headers via
416    /// reinitialization. `None` on deployments where fast sync is disabled
417    /// — the gate becomes an unconditional pass-through.
418    fast_sync_active: Option<Arc<AtomicBool>>,
419    misbehavior_store: Arc<MisbehaviorStore>,
420}

422impl<C: NetworkClient, V: BlockVerifier, D: CoreThreadDispatcher> HeaderSynchronizer<C, V, D> {
423    /// Starts the synchronizer, which is responsible for fetching block headers
424    /// from other authorities and managing block synchronization tasks.
425    pub fn start(
426        network_client: Arc<C>,
427        context: Arc<Context>,
428        core_dispatcher: Arc<D>,
429        commit_vote_monitor: Arc<CommitVoteMonitor>,
430        transactions_synchronizer: Arc<TransactionsSynchronizerHandle>,
431        block_verifier: Arc<V>,
432        dag_state: Arc<RwLock<DagState>>,
433        sync_last_known_own_block: bool,
434        fast_sync_active: Option<Arc<AtomicBool>>,
435        misbehavior_store: Arc<MisbehaviorStore>,
436    ) -> Arc<HeaderSynchronizerHandle> {
437        let (commands_sender, commands_receiver) =
438            channel("consensus_synchronizer_commands", 1_000);
439        let inflight_block_headers_map = InflightBlockHeadersMap::new();
440        let verified_headers_cache = Arc::new(Mutex::new(LruCache::new(
441            NonZeroUsize::new(VERIFIED_HEADERS_CACHE_CAP).unwrap(),
442        )));
443        let received_block_headers = Arc::new(FilterForHeaders::new());
444
445        // Spawn the tasks to fetch the blocks from the others
446        let mut fetch_block_senders = BTreeMap::new();
447        let mut tasks = JoinSet::new();
448        for (index, _) in context.committee.authorities() {
449            if index == context.own_index {
450                continue;
451            }
452            let (sender, receiver) = channel(
453                "consensus_synchronizer_fetches",
454                FETCH_BLOCK_HEADERS_CONCURRENCY,
455            );
456            let fetch_blocks_from_authority_async = Self::fetch_headers_from_authority_live(
457                index,
458                network_client.clone(),
459                block_verifier.clone(),
460                verified_headers_cache.clone(),
461                received_block_headers.clone(),
462                transactions_synchronizer.clone(),
463                commit_vote_monitor.clone(),
464                context.clone(),
465                core_dispatcher.clone(),
466                dag_state.clone(),
467                receiver,
468                commands_sender.clone(),
469                misbehavior_store.clone(),
470            );
471            tasks.spawn(monitored_future!(fetch_blocks_from_authority_async));
472            fetch_block_senders.insert(index, sender);
473        }
474
475        let commands_sender_clone = commands_sender.clone();
476        let verified_headers_cache_clone = verified_headers_cache.clone();
477
478        if sync_last_known_own_block {
479            commands_sender
480                .try_send(Command::FetchOwnLastBlockHeader)
481                .expect("Failed to sync our last block");
482        }
483
484        // Spawn the task to listen to the requests & periodic runs
485        tasks.spawn(monitored_future!(async move {
486            let mut s = Self {
487                context,
488                commands_receiver,
489                fetch_block_senders,
490                core_dispatcher,
491                commit_vote_monitor,
492                fetch_block_headers_scheduler_task: JoinSet::new(),
493                fetch_own_last_header_task: JoinSet::new(),
494                transactions_synchronizer,
495                network_client,
496                block_verifier,
497                inflight_block_headers_map,
498                verified_headers_cache,
499                commands_sender: commands_sender_clone,
500                dag_state,
501                fast_sync_active,
502                misbehavior_store,
503            };
504            s.run().await;
505        }));
506
507        Arc::new(HeaderSynchronizerHandle {
508            commands_sender,
509            tasks: tokio::sync::Mutex::new(tasks),
510            verified_headers_cache: verified_headers_cache_clone,
511            received_block_headers,
512        })
513    }
514
515    // The main loop to listen for the submitted commands.
516    #[cfg_attr(test,tracing::instrument(skip_all, name ="",fields(authority = %self.context.own_index)))]
517    async fn run(&mut self) {
518        // We want the synchronizer to run periodically every 200ms to fetch any missing
519        // blocks.
520        const PERIODIC_SYNCHRONIZER_TIMEOUT: Duration = Duration::from_millis(200);
521        let scheduler_timeout = sleep_until(Instant::now() + PERIODIC_SYNCHRONIZER_TIMEOUT);
522
523        tokio::pin!(scheduler_timeout);
524
525        loop {
526            tokio::select! {
527                Some(command) = self.commands_receiver.recv() => {
528                    match command {
529                        Command::FetchBlockHeaders{ missing_block_refs, peer_index, result } => {
530                            if peer_index == self.context.own_index {
531                                error!("We should never attempt to fetch block headers from our own node");
532                                continue;
533                            }
534
535                            // Skip while fast commit syncer is active — headers for
536                            // committed ranges will be supplied by fast sync's
537                            // reinitialization. Prevents racing with fast sync and
538                            // avoids fetching now-stale ancestors post-reinit.
539                            if paused_by_fast_sync(
540                                self.fast_sync_active.as_ref(),
541                                &self.context.metrics.node_metrics,
542                                FastSyncPauseSource::HeaderCommand,
543                            ) {
544                                result.send(Ok(())).ok();
545                                continue;
546                            }
547
548                            // Keep only the max allowed blocks to request. It is ok to reduce here as the scheduler
549                            // task will take care syncing whatever is leftover.
550                            let missing_block_refs = missing_block_refs
551                                .into_iter()
552                                .take(self.context.parameters.max_headers_per_header_sync_fetch)
553                                .collect();
554
555                            let blocks_guard = self.inflight_block_headers_map.lock_headers(missing_block_refs, peer_index, SyncMethod::Live);
556                            let Some(blocks_guard) = blocks_guard else {
557                                result.send(Ok(())).ok();
558                                continue;
559                            };
560
561                            // We don't block if the corresponding peer task is saturated - but we rather drop the request. That's ok as the periodic
562                            // synchronization task will handle any still missing blocks in next run.
563                            let r = self
564                                .fetch_block_senders
565                                .get(&peer_index)
566                                .expect("Fatal error, sender should be present")
567                                .try_send(blocks_guard)
568                                .map_err(|err| {
569                                    match err {
570                                        TrySendError::Full(_) => ConsensusError::SynchronizerSaturated(peer_index),
571                                        TrySendError::Closed(_) => ConsensusError::Shutdown
572                                    }
573                                });
574
575                            result.send(r).ok();
576                        }
577                        Command::FetchOwnLastBlockHeader => {
578                            if self.fetch_own_last_header_task.is_empty() {
579                                self.start_fetch_own_last_block_header_task();
580                            }
581                        }
582                        Command::KickOffScheduler => {
583                            // just reset the scheduler timeout timer to run immediately if not already running.
584                            // If the scheduler is already running then just reduce the remaining time to run.
585                            let timeout = if self.fetch_block_headers_scheduler_task.is_empty() {
586                                Instant::now()
587                            } else {
588                                Instant::now() + PERIODIC_SYNCHRONIZER_TIMEOUT.checked_div(2).unwrap()
589                            };
590
591                            // only reset if it is earlier than the next deadline
592                            if timeout < scheduler_timeout.deadline() {
593                                scheduler_timeout.as_mut().reset(timeout);
594                            }
595                        }
596                    }
597                },
598                Some(result) = self.fetch_own_last_header_task.join_next(), if !self.fetch_own_last_header_task.is_empty() => {
599                    match result {
600                        Ok(()) => {},
601                        Err(e) => {
602                            if e.is_cancelled() {
603                            } else if e.is_panic() {
604                                std::panic::resume_unwind(e.into_panic());
605                            } else {
606                                panic!("fetch our last block task failed: {e}");
607                            }
608                        },
609                    };
610                },
611                Some(result) = self.fetch_block_headers_scheduler_task.join_next(), if !self.fetch_block_headers_scheduler_task.is_empty() => {
612                    match result {
613                        Ok(()) => {},
614                        Err(e) => {
615                            if e.is_cancelled() {
616                            } else if e.is_panic() {
617                                std::panic::resume_unwind(e.into_panic());
618                            } else {
619                                panic!("fetch blocks scheduler task failed: {e}");
620                            }
621                        },
622                    };
623                },
624                () = &mut scheduler_timeout => {
625                    // Skip firing the periodic sync task while fast commit
626                    // syncer is active. The timer is still rearmed so the
627                    // scheduler resumes at the next tick after fast sync
628                    // becomes idle.
629                    if paused_by_fast_sync(
630                        self.fast_sync_active.as_ref(),
631                        &self.context.metrics.node_metrics,
632                        FastSyncPauseSource::HeaderScheduler,
633                    ) {
634                        scheduler_timeout
635                            .as_mut()
636                            .reset(Instant::now() + PERIODIC_SYNCHRONIZER_TIMEOUT);
637                        continue;
638                    }
639
640                    // we want to start a new task only if the previous one has already finished.
641                    if self.fetch_block_headers_scheduler_task.is_empty() {
642                        if let Err(err) = self.start_periodic_sync_task().await {
643                            debug!("Core is shutting down, synchronizer is shutting down: {err:?}");
644                            return;
645                        };
646                    }
647
648                    scheduler_timeout
649                        .as_mut()
650                        .reset(Instant::now() + PERIODIC_SYNCHRONIZER_TIMEOUT);
651                }
652            }
653        }

656    async fn fetch_headers_from_authority_live(
657        peer_index: AuthorityIndex,
658        network_client: Arc<C>,
659        block_verifier: Arc<V>,
660        verified_cache: Arc<Mutex<LruCache<BlockHeaderDigest, ()>>>,
661        received_block_headers: Arc<FilterForHeaders>,
662        transactions_synchronizer: Arc<TransactionsSynchronizerHandle>,
663        commit_vote_monitor: Arc<CommitVoteMonitor>,
664        context: Arc<Context>,
665        core_dispatcher: Arc<D>,
666        dag_state: Arc<RwLock<DagState>>,
667        mut receiver: Receiver<BlocksGuard>,
668        commands_sender: Sender<Command>,
669        misbehavior_store: Arc<MisbehaviorStore>,
670    ) {
671        const MAX_RETRIES: u32 = 3;
672        let peer_hostname = &context.committee.authority(peer_index).hostname;
673
674        let mut requests = FuturesUnordered::new();
675
676        loop {
677            tokio::select! {
678                Some(mut headers_guard) = receiver.recv(), if requests.len() < FETCH_BLOCK_HEADERS_CONCURRENCY => {
679                    let context = context.clone();
680                    let network_client = network_client.clone();
681                    let received_block_headers = received_block_headers.clone();
682                    requests.push(async move {
683                        // Most missing ancestors are a round or two behind and already
684                        // on their way over the block streams: let them arrive first
685                        // and ask only for the rest, without gap-fill (no frontier).
686                        sleep(context.parameters.min_block_delay * 2).await;
687                        let arrived = headers_guard
688                            .block_refs
689                            .iter()
690                            .filter(|block_ref| received_block_headers.contains(&block_ref.digest))
691                            .copied()
692                            .collect::<BTreeSet<_>>();
693                        let metrics = &context.metrics.node_metrics;
694                        metrics.synchronizer_live_block_headers_resolved_before_fetch.inc_by(arrived.len() as u64);
695                        headers_guard.release(&arrived);
696                        if headers_guard.block_refs.is_empty() {
697                            metrics.synchronizer_live_fetches_skipped.inc();
698                            let nothing_fetched = FetchedHeaders { serialized_headers: Vec::new(), elapsed: Duration::ZERO, holds_requested: false };
699                            return (Ok(nothing_fetched), headers_guard, 1, peer_index, Vec::new());
700                        }
701
702                        let peer_hostname = &context.committee.authority(peer_index).hostname;
703                        metrics
704                            .synchronizer_requested_block_headers_by_peer
705                            .with_label_values(&[peer_hostname.as_str(), "live"])
706                            .inc_by(headers_guard.block_refs.len() as u64);
707                        let authors = headers_guard.block_refs.iter().map(|block_ref| block_ref.author).collect::<HashSet<_>>();
708                        for author in authors {
709                            let host = &context.committee.authority(author).hostname;
710                            metrics
711                                .synchronizer_requested_block_headers_by_authority
712                                .with_label_values(&[host.as_str(), "live"])
713                                .inc();
714                        }
715
716                        Self::fetch_block_headers_request(context, network_client, peer_index, headers_guard, Vec::new(), FETCH_REQUEST_TIMEOUT, true, 1).await
717                    }.boxed())
718                },
719                Some((response, blocks_guard, retries, _peer, highest_rounds)) = requests.next() => {
720                    match response {
721                        Ok(fetched) => {
722                            if let Err(err) = Self::process_fetched_headers_from_authority(fetched,
723                                peer_index,
724                                blocks_guard,
725                                highest_rounds,
726                                core_dispatcher.clone(),
727                                dag_state.clone(),
728                                block_verifier.clone(),
729                                verified_cache.clone(),
730                                commit_vote_monitor.clone(),
731                                transactions_synchronizer.clone(),
732                                context.clone(),
733                                commands_sender.clone(),
734                                "live",
735                                misbehavior_store.clone(),
736                            ).await {
737                                warn!("Error while processing fetched blocks from peer {peer_index} {peer_hostname}: {err}");
738                                context.metrics.node_metrics.synchronizer_process_fetched_failures_by_peer.with_label_values(&[peer_hostname.as_str(), "live"]).inc();
739                            }
740                        },
741                        Err(_) => {
742                            context.metrics.node_metrics.synchronizer_fetch_failures_by_peer.with_label_values(&[peer_hostname.as_str(), "live"]).inc();
743                            if retries <= MAX_RETRIES {
744                                requests.push(Self::fetch_block_headers_request(context.clone(), network_client.clone(), peer_index, blocks_guard, highest_rounds, FETCH_REQUEST_TIMEOUT, true, retries).boxed())
745                            } else {
746                                warn!("Max retries {retries} reached while trying to fetch blocks from peer {peer_index} {peer_hostname}.");
747                                // we don't necessarily need to do, but dropping the guard here to unlock the blocks
748                                drop(blocks_guard);
749                            }
750                        }
751                    }
752                },
753                else => {
754                    info!("Fetching blocks from authority {peer_index} task will now abort.");
755                    break;
756                }
757            }
758        }

761    /// Processes the requested raw fetched headers from peer `peer_index`. If
762    /// no error is returned then the verified blocks are immediately sent
763    /// to Core for processing.
764    ///
765    /// Records the fetch in the responsiveness tracker once the payload has
766    /// been verified: a verification failure or a gap-fill window violation
767    /// counts as a failure, a success is credited with the fetch-plus-verify
768    /// latency scaled by how many of the requested headers were actually
769    /// delivered.
770    async fn process_fetched_headers_from_authority(
771        fetched: FetchedHeaders,
772        peer_index: AuthorityIndex,
773        requested_blocks_guard: BlocksGuard,
774        highest_rounds: Vec<Round>,
775        core_dispatcher: Arc<D>,
776        dag_state: Arc<RwLock<DagState>>,
777        block_verifier: Arc<V>,
778        verified_cache: Arc<Mutex<LruCache<BlockHeaderDigest, ()>>>,
779        commit_vote_monitor: Arc<CommitVoteMonitor>,
780        transactions_synchronizer: Arc<TransactionsSynchronizerHandle>,
781        context: Arc<Context>,
782        commands_sender: Sender<Command>,
783        sync_method: &str,
784        misbehavior_store: Arc<MisbehaviorStore>,
785    ) -> ConsensusResult<()> {
786        let mut serialized_headers = fetched.serialized_headers;
787        if serialized_headers.is_empty() {
788            return Ok(());
789        }
790        let _s = context
791            .metrics
792            .node_metrics
793            .scope_processing_time
794            .with_label_values(&["Synchronizer::process_fetched_blocks"])
795            .start_timer();
796        if serialized_headers.len() > context.parameters.max_headers_per_header_sync_fetch {
797            debug!(
798                "Truncating fetched headers from peer {} to max allowed {} blocks",
799                peer_index, context.parameters.max_headers_per_header_sync_fetch
800            );
801            serialized_headers.truncate(context.parameters.max_headers_per_header_sync_fetch);
802        }
803
804        // Verify all the fetched block headers
805        let verify_start = Instant::now();
806        let block_headers = Handle::current()
807            .spawn_blocking({
808                let block_verifier = block_verifier.clone();
809                let verified_cache = verified_cache.clone();
810                let context = context.clone();
811                let sync_method = sync_method.to_string();
812                let misbehavior_store = misbehavior_store.clone();
813                move || {
814                    Self::verify_block_headers(
815                        serialized_headers,
816                        block_verifier,
817                        verified_cache,
818                        &context,
819                        peer_index,
820                        &sync_method,
821                        &misbehavior_store,
822                    )
823                }
824            })
825            .await
826            .expect("Spawn blocking should not fail");
827        // Fetch and verification both count against the peer, matching the
828        // commit syncer.
829        let elapsed = fetched.elapsed + verify_start.elapsed();
830        let block_headers = block_headers.inspect_err(|_| {
831            // A fast response of invalid headers must not earn rank.
832            context.peer_responsiveness.record_failure_with_timeout(
833                DataSource::HeaderSynchronizerRequested,
834                peer_index,
835                FETCH_REQUEST_TIMEOUT,
836            );
837        })?;
838
839        // Record commit votes from the verified blocks.
840        for block in &block_headers {
841            commit_vote_monitor.observe_block(block);
842        }
843
844        let metrics = &context.metrics.node_metrics;
845        let peer_hostname = &context.committee.authority(peer_index).hostname;
846        metrics
847            .synchronizer_fetched_block_headers_by_peer
848            .with_label_values(&[peer_hostname.as_str(), sync_method])
849            .inc_by(block_headers.len() as u64);
850        for block_header in &block_headers {
851            let block_header_hostname =
852                &context.committee.authority(block_header.author()).hostname;
853            metrics
854                .synchronizer_fetched_block_headers_by_authority
855                .with_label_values(&[block_header_hostname.as_str(), sync_method])
856                .inc();
857        }
858
859        debug!(
860            "Synced {} missing blocks from peer {peer_index} {peer_hostname}: {}",
861            block_headers.len(),
862            block_headers
863                .iter()
864                .map(|b| b.reference().to_string())
865                .join(", "),
866        );
867
868        // Partition the response. Refs we asked for are admitted unconditionally:
869        // we named them by digest, so they are needed by definition. Anything
870        // else is the server's gap-fill, legitimate only for an author we
871        // requested from, at a round strictly between the frontier we sent and
872        // the highest round we requested for that author. The bound is the
873        // highest rather than the lowest requested round because the server may
874        // truncate an oversized request and derive its gap-fill window from the
875        // surviving refs.
876        let mut max_requested_rounds: Vec<Option<Round>> = vec![None; context.committee.size()];
877        for block_ref in &requested_blocks_guard.block_refs {
878            let entry = &mut max_requested_rounds[block_ref.author];
879            *entry = Some(entry.map_or(block_ref.round, |round| round.max(block_ref.round)));
880        }
881
882        let mut requested_headers = Vec::new();
883        let mut additional_headers = Vec::new();
884        let mut out_of_window: Option<BlockRef> = None;
885        for header in block_headers {
886            let block_ref = header.reference();
887            if requested_blocks_guard.block_refs.contains(&block_ref) {
888                requested_headers.push(header);
889                continue;
890            }
891            // The live synchronizer sends no frontier and so gets no gap-fill;
892            // the periodic synchronizer sends one and admits extras inside its
893            // window.
894            let in_window = !highest_rounds.is_empty()
895                && max_requested_rounds[block_ref.author].is_some_and(|max_round| {
896                    highest_rounds[block_ref.author] < block_ref.round
897                        && block_ref.round < max_round
898                });
899            if !in_window {
900                out_of_window.get_or_insert(block_ref);
901            }
902            additional_headers.push(header);
903        }
904
905        // Both sides derive the gap-fill window from the same request inputs,
906        // so an out-of-window header cannot come from an honest server — and
907        // the unrequested part of the response is exactly what we cannot
908        // verify by digest, so none of it is accepted.
909        if let Some(block_ref) = out_of_window {
910            let e = ConsensusError::UnrequestedHeaderOutOfWindow {
911                peer: peer_index,
912                author: block_ref.author,
913                round: block_ref.round,
914            };
915            misbehavior_store.record_faulty_block(peer_index, peer_index, &e);
916            metrics
917                .synchronizer_fetch_window_violations
918                .with_label_values(&[peer_hostname.as_str(), sync_method])
919                .inc();
920            warn!(
921                "Dropping {} unrequested headers fetched from peer {peer_index} {peer_hostname}: {e}",
922                additional_headers.len()
923            );
924            additional_headers.clear();
925            // A proven violation is charged like a failed fetch: the peer is
926            // dishonest, so its response speed must not earn rank.
927            context.peer_responsiveness.record_failure_with_timeout(
928                DataSource::HeaderSynchronizerRequested,
929                peer_index,
930                FETCH_REQUEST_TIMEOUT,
931            );
932        } else {
933            // Credit the fetch now that the payload verified. Only requested
934            // headers count as delivered, each ref once, so neither unsolicited
935            // extras nor duplicated copies can mask a shortfall.
936            let factor = if fetched.holds_requested {
937                let delivered = requested_headers
938                    .iter()
939                    .map(|header| header.reference())
940                    .collect::<BTreeSet<_>>()
941                    .len();
942                shortfall_factor(requested_blocks_guard.block_refs.len(), delivered)
943            } else {
944                1.0
945            };
946            // Capped at the failure penalty: a partial delivery never records
947            // worse than a failed fetch.
948            context.peer_responsiveness.record_success(
949                DataSource::HeaderSynchronizerRequested,
950                peer_index,
951                elapsed.mul_f64(factor).min(FETCH_REQUEST_TIMEOUT),
952            );
953        }
954
955        let requested_headers = drop_far_future(
956            &context,
957            &dag_state,
958            requested_headers,
959            DataSource::HeaderSynchronizerRequested,
960            |header| header.round(),
961        );
962        let additional_headers = drop_far_future(
963            &context,
964            &dag_state,
965            additional_headers,
966            DataSource::HeaderSynchronizerAdditional,
967            |header| header.round(),
968        );
969
970        // Now send them to core for processing — additional headers first, as
971        // they sit below the requested ones and connect them to our frontier.
972        // They go in under their own source so the block manager's per-slot cap
973        // applies to them but never to the requested refs. Ignore the returned
974        // missing blocks as we don't want this mechanism to keep feedback
975        // looping on fetching more blocks. The periodic synchronization will
976        // take care of that.
977        let mut missing_blocks = BTreeSet::new();
978        let mut missing_committed_txns = BTreeMap::new();
979        for (headers, source) in [
980            (additional_headers, DataSource::HeaderSynchronizerAdditional),
981            (requested_headers, DataSource::HeaderSynchronizerRequested),
982        ] {
983            if headers.is_empty() {
984                continue;
985            }
986            let (blocks, committed_txns) = core_dispatcher
987                .add_block_headers(headers, source)
988                .await
989                .map_err(|_| ConsensusError::Shutdown)?;
990            missing_blocks.extend(blocks);
991            missing_committed_txns.extend(committed_txns);
992        }
993
994        // now release all the locked blocks as they have been fetched, verified &
995        // processed
996        drop(requested_blocks_guard);
997
998        // kick off immediately the scheduled synchronizer
999        if !missing_blocks.is_empty() {
1000            // do not block here, so we avoid any possible cycles.
1001            if let Err(TrySendError::Full(_)) = commands_sender.try_send(Command::KickOffScheduler)
1002            {
1003                warn!("Commands channel is full")
1004            }
1005        }
1006
1007        context
1008            .metrics
1009            .node_metrics
1010            .missing_block_headers_after_fetch_total
1011            .inc_by(missing_blocks.len() as u64);
1012
1013        if !missing_committed_txns.is_empty() {
1014            debug!(
1015                "Missing committed transactions after fetching blocks: {:?}",
1016                missing_committed_txns
1017            );
1018            if let Err(err) = transactions_synchronizer
1019                .fetch_transactions(missing_committed_txns)
1020                .await
1021            {
1022                warn!(
1023                    "Error while trying to fetch missing transactions via transactions synchronizer: {err}"
1024                );
1025            }
1026        }
1027
1028        Ok(())
1029    }

1046    fn verify_block_headers(
1047        serialized_block_headers: Vec<Bytes>,
1048        block_verifier: Arc<V>,
1049        verified_cache: Arc<Mutex<LruCache<BlockHeaderDigest, ()>>>,
1050        context: &Context,
1051        peer_index: AuthorityIndex,
1052        sync_method: &str,
1053        misbehavior_store: &MisbehaviorStore,
1054    ) -> ConsensusResult<Vec<VerifiedBlockHeader>> {
1055        let mut verified_block_headers = Vec::new();
1056        let mut skipped_count = 0u64;
1057
1058        for serialized_block_header in serialized_block_headers {
1059            let block_header_digest = VerifiedBlockHeader::compute_digest(&serialized_block_header);
1060            // A cache hit only skips the signature check. The header is still
1061            // returned: it may have been dropped after the earlier
1062            // verification (gap-fill window, far-future bound, per-slot cap),
1063            // and the block manager filters real duplicates against its own
1064            // state.
1065            let already_verified = verified_cache.lock().get(&block_header_digest).is_some();
1066
1067            let signed_block_header: SignedBlockHeader = bcs::from_bytes(&serialized_block_header)
1068                .map_err(ConsensusError::MalformedHeader)
1069                .inspect_err(|e| {
1070                    // Author is unknown when deserialization fails — blame the peer.
1071                    misbehavior_store.record_faulty_block(peer_index, peer_index, e);
1072                })?;
1073
1074            if already_verified {
1075                skipped_count += 1;
1076            } else {
1077                if let Err(e) = block_verifier.verify(&signed_block_header) {
1078                    // TODO: we might want to use a different metric to track the invalid "served"
1079                    // blocks from the invalid "proposed" ones.
1080                    let hostname = context.committee.authority(peer_index).hostname.clone();
1081
1082                    context
1083                        .metrics
1084                        .node_metrics
1085                        .synchronizer_invalid_block_headers
1086                        .with_label_values(&[hostname.as_str(), "synchronizer", e.name()])
1087                        .inc();
1088                    misbehavior_store.record_faulty_block(
1089                        peer_index,
1090                        signed_block_header.author(),
1091                        &e,
1092                    );
1093                    warn!("Invalid block received from {}: {}", peer_index, e);
1094                    return Err(e);
1095                }
1096
1097                verified_cache.lock().put(block_header_digest, ());
1098            }
1099
1100            let verified_block_header = VerifiedBlockHeader::new_verified_with_digest(
1101                signed_block_header,
1102                serialized_block_header,
1103                block_header_digest,
1104            );
1105
1106            // Dropping is ok because the block will be refetched.
1107            // TODO: improve efficiency, maybe suspend and continue processing the block
1108            // asynchronously.
1109            let now = context.clock.timestamp_utc_ms();
1110            if now < verified_block_header.timestamp_ms() {
1111                trace!(
1112                    "Synced block header {} timestamp {} is in the future (now={}). Will not ignore as median based timestamp is enabled.",
1113                    verified_block_header.reference(),
1114                    verified_block_header.timestamp_ms(),
1115                    now
1116                );
1117            }
1118
1119            verified_block_headers.push(verified_block_header);
1120        }
1121
1122        // Record skipped block headers metric
1123        if skipped_count > 0 {
1124            let peer_hostname = &context.committee.authority(peer_index).hostname;
1125            context
1126                .metrics
1127                .node_metrics
1128                .synchronizer_skipped_block_headers_by_peer
1129                .with_label_values(&[peer_hostname.as_str(), sync_method])
1130                .inc_by(skipped_count);
1131        }
1132
1133        Ok(verified_block_headers)
1134    }

1136    /// Fetches the headers held by `blocks_guard` from `peer`, recording
1137    /// failures (network error, timeout, empty response from a peer that owes
1138    /// the request) in the responsiveness tracker. A non-empty response is
1139    /// credited only after it verifies, in
1140    /// `process_fetched_headers_from_authority`, so a fast but invalid
1141    /// response earns no rank. Both the live and the periodic path go through
1142    /// here.
1143    ///
1144    /// A peer known to hold the headers (`holds_requested`) owes the whole
1145    /// request: an empty response is a failure. A peer asked speculatively
1146    /// owes nothing: it is left unrecorded on an empty response, since
1147    /// charging its routine empty answers would park most of the committee at
1148    /// the timeout.
1149    ///
1150    /// Recording is unconditional; `enable_peer_responsiveness_ranking` gates
1151    /// only the ordering.
1152    async fn fetch_block_headers_request(
1153        context: Arc<Context>,
1154        network_client: Arc<C>,
1155        peer: AuthorityIndex,
1156        blocks_guard: BlocksGuard,
1157        highest_rounds: Vec<Round>,
1158        request_timeout: Duration,
1159        holds_requested: bool,
1160        mut retries: u32,
1161    ) -> (
1162        ConsensusResult<FetchedHeaders>,
1163        BlocksGuard,
1164        u32,
1165        AuthorityIndex,
1166        Vec<Round>,
1167    ) {
1168        let start = Instant::now();
1169        let resp = timeout(
1170            request_timeout,
1171            network_client.fetch_block_headers(
1172                peer,
1173                blocks_guard
1174                    .block_refs
1175                    .clone()
1176                    .into_iter()
1177                    .collect::<Vec<_>>(),
1178                highest_rounds.clone(),
1179                request_timeout,
1180            ),
1181        )
1182        .await;
1183
1184        // Taken before the retry backoff below, which would otherwise pad every
1185        // failed request out to the full timeout.
1186        let elapsed = start.elapsed();
1187        match &resp {
1188            Ok(Ok(headers)) => {
1189                // Nothing to report about a peer that was never expected to
1190                // hold these headers and turned out not to.
1191                if headers.is_empty() && holds_requested {
1192                    context.peer_responsiveness.record_failure_with_timeout(
1193                        DataSource::HeaderSynchronizerRequested,
1194                        peer,
1195                        request_timeout,
1196                    );
1197                }
1198            }
1199            _ => {
1200                context.peer_responsiveness.record_failure_with_timeout(
1201                    DataSource::HeaderSynchronizerRequested,
1202                    peer,
1203                    request_timeout,
1204                );
1205            }
1206        }
1207
1208        fail_point_async!("consensus-delay");
1209
1210        let resp = match resp {
1211            Ok(Err(err)) => {
1212                // Add a delay before retrying - if that is needed. If request has timed out
1213                // then eventually this will be a no-op.
1214                sleep_until(start + request_timeout).await;
1215                retries += 1;
1216                Err(err)
1217            } // network error
1218            Err(err) => {
1219                // timeout
1220                sleep_until(start + request_timeout).await;
1221                retries += 1;
1222                Err(ConsensusError::NetworkRequestTimeout(err.to_string()))
1223            }
1224            Ok(Ok(serialized_headers)) => Ok(FetchedHeaders {
1225                serialized_headers,
1226                elapsed,
1227                holds_requested,
1228            }),
1229        };
1230        (resp, blocks_guard, retries, peer, highest_rounds)
1231    }
1232    #[cfg_attr(test,tracing::instrument(skip_all, name ="",fields(authority = %self.context.own_index)))]
1233    fn start_fetch_own_last_block_header_task(&mut self) {
1234        const FETCH_OWN_BLOCK_HEADER_RETRY_DELAY: Duration = Duration::from_millis(1_000);
1235        const MAX_RETRY_DELAY_STEP: Duration = Duration::from_millis(4_000);
1236
1237        let context = self.context.clone();
1238        let dag_state = self.dag_state.clone();
1239        let network_client = self.network_client.clone();
1240        let block_verifier = self.block_verifier.clone();
1241        let core_dispatcher = self.core_dispatcher.clone();
1242        let misbehavior_store = self.misbehavior_store.clone();
1243
1244        self.fetch_own_last_header_task
1245            .spawn(monitored_future!(async move {
1246                let _scope = monitored_scope("FetchOwnLastBlockHeaderTask");
1247
1248                // Timed: the probe seeds the responsiveness track for the whole
1249                // committee before any other fetch source has a sample.
1250                let fetch_own_block_header = |authority_index: AuthorityIndex, fetch_own_block_header_delay: Duration| {
1251                    let network_client_cloned = network_client.clone();
1252                    let own_index = context.own_index;
1253                    async move {
1254                        sleep(fetch_own_block_header_delay).await;
1255                        let started = Instant::now();
1256                        let r = network_client_cloned.fetch_latest_block_headers(authority_index, vec![own_index], FETCH_REQUEST_TIMEOUT).await;
1257                        (r, authority_index, started.elapsed())
1258                    }
1259                };
1260
1261                let process_block_headers = |block_headers: Vec<Bytes>, authority_index: AuthorityIndex| -> ConsensusResult<Vec<VerifiedBlockHeader >> {
1262                                    let mut result = Vec::new();
1263                                    for serialized_block_header in block_headers {
1264                                        let signed_block_header: SignedBlockHeader = bcs::from_bytes(&serialized_block_header)
1265                                            .map_err(ConsensusError::MalformedHeader)
1266                                            .inspect_err(|e| {
1267                                                // Author unknown when deserialization fails — blame the peer.
1268                                                misbehavior_store.record_faulty_block(
1269                                                    authority_index,
1270                                                    authority_index,
1271                                                    e,
1272                                                );
1273                                            })?;
1274                                        block_verifier.verify(&signed_block_header).tap_err(|err|{
1275                                            let hostname = context.committee.authority(authority_index).hostname.clone();
1276                                            context
1277                                                .metrics
1278                                                .node_metrics
1279                                                .synchronizer_invalid_block_headers
1280                                                .with_label_values(&[hostname.as_str(), "synchronizer_own_block_header", err.clone().name()])
1281                                                .inc();
1282                                            misbehavior_store.record_faulty_block(
1283                                                authority_index,
1284                                                signed_block_header.author(),
1285                                                err,
1286                                            );
1287                                            warn!("Invalid block header received from {}: {}", authority_index, err);
1288                                        })?;
1289
1290                                        let verified_block_header = VerifiedBlockHeader::new_verified(signed_block_header, serialized_block_header);
1291                                        if verified_block_header.author() != context.own_index {
1292                                            return Err(ConsensusError::UnexpectedLastOwnHeader { index: authority_index, block_ref: verified_block_header.reference()});
1293                                        }
1294                                        result.push(verified_block_header);
1295                                    }
1296                                    Ok(result)
1297                };
1298
1299                // Get the highest of all the results. Retry until at least `2f+1` results have been gathered.
1300                let mut highest_round = GENESIS_ROUND;
1301                // Keep track of the received responses to avoid fetching the own block header from same peer
1302                let mut received_response = vec![false; context.committee.size()];
1303                // Assume that our node is not Byzantine
1304                received_response[context.own_index] = true;
1305                let mut total_stake = context.committee.stake(context.own_index);
1306                let mut retries = 0;
1307                let mut retry_delay_step = Duration::from_millis(500);
1308                'main:loop {
1309                    if context.committee.size() == 1 {
1310                        highest_round = dag_state.read().get_last_proposed_block_header().round();
1311                        info!("Only one node in the network, will not try fetching own last block header from peers.");
1312                        break 'main;
1313                    }
1314
1315                    // Ask all the other peers about our last block header
1316                    let mut results = FuturesUnordered::new();
1317
1318                    for (authority_index, _authority) in context.committee.authorities() {
1319                        // Skip our own index and the ones that have already responded
1320                        if !received_response[authority_index] {
1321                            results.push(fetch_own_block_header(authority_index, Duration::from_millis(0)));
1322                        }
1323                    }
1324
1325                    // Gather the results but wait to timeout as well
1326                    let timer = sleep_until(Instant::now() + context.parameters.sync_last_known_own_block_timeout);
1327                    tokio::pin!(timer);
1328
1329                    'inner: loop {
1330                        tokio::select! {
1331                            result = results.next() => {
1332                                let Some((result, authority_index, latency)) = result else {
1333                                    break 'inner;
1334                                };
1335                                // An empty answer that verified still counts: a peer
1336                                // legitimately holds no header of ours.
1337                                let record_probe = |ok: bool| {
1338                                    if ok {
1339                                        context.peer_responsiveness.record_success(
1340                                            DataSource::HeaderSynchronizerRequested,
1341                                            authority_index,
1342                                            latency,
1343                                        );
1344                                    } else {
1345                                        context.peer_responsiveness.record_failure_with_timeout(
1346                                            DataSource::HeaderSynchronizerRequested,
1347                                            authority_index,
1348                                            FETCH_REQUEST_TIMEOUT,
1349                                        );
1350                                    }
1351                                };
1352                                match result {
1353                                    Ok(result) => {
1354                                        match process_block_headers(result, authority_index) {
1355                                            Ok(block_headers) => {
1356                                                record_probe(true);
1357                                                received_response[authority_index] = true;
1358                                                let max_round = block_headers.into_iter().map(|b|b.round()).max().unwrap_or(0);
1359                                                highest_round = highest_round.max(max_round);
1360
1361                                                total_stake += context.committee.stake(authority_index);
1362                                            },
1363                                            Err(err) => {
1364                                                record_probe(false);
1365                                                warn!("Invalid result returned from {authority_index} while fetching last own block header: {err}");
1366                                            }
1367                                        }
1368                                    },
1369                                    Err(err) => {
1370                                        record_probe(false);
1371                                        warn!("Error {err} while fetching our own block header from peer {authority_index}. Will retry.");
1372                                        results.push(fetch_own_block_header(authority_index, FETCH_OWN_BLOCK_HEADER_RETRY_DELAY));
1373                                    }
1374                                }
1375                            },
1376                            () = &mut timer => {
1377                                info!("Timeout while trying to sync our own last block header from peers");
1378                                break 'inner;
1379                            }
1380                        }
1381                    }
1382
1383                    // Request at least a quorum of 2f+1 stake to have replied back.
1384                    if context.committee.reached_quorum(total_stake) {
1385                        info!("A quorum, {} out of {} total stake, returned acceptable results for our own last block header with highest round {}, with {retries} retries.", total_stake, context.committee.total_stake(), highest_round);
1386                        break 'main;
1387                    } else {
1388                        info!("Only {} out of {} total stake returned acceptable results for our own last block header with highest round {}, with {retries} retries.", total_stake, context.committee.total_stake(), highest_round);
1389                    }
1390
1391                    retries += 1;
1392                    context.metrics.node_metrics.sync_last_known_own_block_header_retries.inc();
1393                    warn!("Not enough stake: {} out of {} total stake returned acceptable results for our own last block header with highest round {}. Will now retry {retries}.", total_stake, context.committee.total_stake(), highest_round);
1394
1395                    sleep(retry_delay_step).await;
1396
1397                    retry_delay_step = Duration::from_secs_f64(retry_delay_step.as_secs_f64() * 1.5);
1398                    retry_delay_step = retry_delay_step.min(MAX_RETRY_DELAY_STEP);
1399                }
1400
1401                // Update the Core with the highest detected round
1402                context.metrics.node_metrics.last_known_own_block_header_round.set(highest_round as i64);
1403
1404                if let Err(err) = core_dispatcher.set_last_known_proposed_round(highest_round) {
1405                    warn!("Error received while calling dispatcher, probably dispatcher is shutting down, will now exit: {err:?}");
1406                }
1407            }));
1408    }

1410    /// Starts the periodic synchronization task to fetch missing block headers
1411    /// from peers. This task runs periodically to ensure that the missing
1412    /// block headers are synced.
1413    async fn start_periodic_sync_task(&mut self) -> ConsensusResult<()> {
1414        let mut missing_blocks_refs = self
1415            .core_dispatcher
1416            .get_missing_block_headers()
1417            .await
1418            .map_err(|_err| ConsensusError::Shutdown)?;
1419
1420        // No reason to kick off the scheduler if there are no missing blocks to fetch
1421        if missing_blocks_refs.is_empty() {
1422            return Ok(());
1423        }
1424
1425        // While solidification lags, new headers only widen the round window
1426        // in which shards and payloads are retained (fetched headers raise the
1427        // accepted frontier), so the scheduler stands down entirely until the
1428        // transactions synchronizer closes the gap.
1429        if self.dag_state.read().is_solidification_lagging() {
1430            trace!(
1431                "Scheduled synchronizer temporarily disabled as the last solid commit is lagging the last commit too much."
1432            );
1433            self.context
1434                .metrics
1435                .node_metrics
1436                .synchronizer_fetch_block_headers_scheduler_skipped
1437                .with_label_values(&["solid_commit_lagging"])
1438                .inc();
1439            return Ok(());
1440        }
1441
1442        let context = self.context.clone();
1443        let network_client = self.network_client.clone();
1444        let block_verifier = self.block_verifier.clone();
1445        let verified_cache = self.verified_headers_cache.clone();
1446        let commit_vote_monitor = self.commit_vote_monitor.clone();
1447        let core_dispatcher = self.core_dispatcher.clone();
1448        let inflight_block_headers_map = self.inflight_block_headers_map.clone();
1449        let commands_sender = self.commands_sender.clone();
1450        let dag_state = self.dag_state.clone();
1451        let transactions_synchronizer = self.transactions_synchronizer.clone();
1452        let misbehavior_store = self.misbehavior_store.clone();
1453
1454        let (commit_lagging, last_commit_index, quorum_commit_index) = self.is_commit_lagging();
1455        trace!(
1456            "Commit lagging: {commit_lagging}, last commit index: {last_commit_index}, quorum commit index: {quorum_commit_index}"
1457        );
1458        if commit_lagging {
1459            // As node is commit lagging try to sync only the missing block headers that are
1460            // within the acceptable round thresholds to sync. The rest we don't
1461            // attempt to sync yet.
1462            let highest_accepted_round = dag_state.read().highest_accepted_round();
1463            missing_blocks_refs = missing_blocks_refs
1464                .into_iter()
1465                .take_while(|(block_ref, _)| {
1466                    block_ref.round
1467                        <= highest_accepted_round + self.missing_block_header_round_threshold()
1468                })
1469                .collect::<BTreeMap<_, _>>();
1470
1471            // If no missing block headers are within the acceptable thresholds to sync
1472            // while we commit lag, then we disable the scheduler completely for
1473            // this run.
1474            if missing_blocks_refs.is_empty() {
1475                trace!(
1476                    "Scheduled synchronizer temporarily disabled as local commit is falling behind from quorum {last_commit_index} << {quorum_commit_index} and missing block headers are too far in the future."
1477                );
1478                self.context
1479                    .metrics
1480                    .node_metrics
1481                    .synchronizer_fetch_block_headers_scheduler_skipped
1482                    .with_label_values(&["commit_lagging"])
1483                    .inc();
1484                return Ok(());
1485            }
1486        }
1487
1488        self.fetch_block_headers_scheduler_task
1489            .spawn(monitored_future!(async move {
1490                let _scope = monitored_scope("FetchMissingBlockHeadersScheduler");
1491
1492                context
1493                    .metrics
1494                    .node_metrics
1495                    .synchronizer_fetch_block_headers_scheduler_inflight
1496                    .inc();
1497                let total_requested = missing_blocks_refs.len();
1498
1499                fail_point_async!("consensus-delay");
1500
1501                // Fetch blocks from peers
1502                let results = Self::fetch_block_headers_from_authorities_periodic(
1503                    context.clone(),
1504                    inflight_block_headers_map.clone(),
1505                    network_client,
1506                    missing_blocks_refs,
1507                    dag_state.clone(),
1508                )
1509                .await;
1510                context
1511                    .metrics
1512                    .node_metrics
1513                    .synchronizer_fetch_block_headers_scheduler_inflight
1514                    .dec();
1515                if results.is_empty() {
1516                    warn!("No results returned while requesting missing block headers");
1517                    return;
1518                }
1519
1520                // Now process the returned results
1521                let mut total_fetched = 0;
1522                for (blocks_guard, fetched, peer, highest_rounds) in results {
1523                    total_fetched += fetched.serialized_headers.len();
1524
1525                    if let Err(err) = Self::process_fetched_headers_from_authority(
1526                        fetched,
1527                        peer,
1528                        blocks_guard,
1529                        highest_rounds,
1530                        core_dispatcher.clone(),
1531                        dag_state.clone(),
1532                        block_verifier.clone(),
1533                        verified_cache.clone(),
1534                        commit_vote_monitor.clone(),
1535                        transactions_synchronizer.clone(),
1536                        context.clone(),
1537                        commands_sender.clone(),
1538                        "periodic",
1539                        misbehavior_store.clone(),
1540                    )
1541                    .await
1542                    {
1543                        warn!(
1544                            "Error occurred while processing fetched block headers from peer {peer}: {err}"
1545                        );
1546                    }
1547                }
1548
1549                debug!(
1550                    "Total block headers requested to fetch: {}, total fetched: {}",
1551                    total_requested, total_fetched
1552                );
1553            }));
1554        Ok(())
1555    }

1557    fn is_commit_lagging(&self) -> (bool, CommitIndex, CommitIndex) {
1558        let last_commit_index = self.dag_state.read().last_commit_index();
1559        let quorum_commit_index = self.commit_vote_monitor.quorum_commit_index();
1560        let commit_threshold = last_commit_index
1561            + self.context.parameters.commit_sync_batch_size * COMMIT_LAG_MULTIPLIER;
1562        (
1563            commit_threshold < quorum_commit_index,
1564            last_commit_index,
1565            quorum_commit_index,
1566        )

1569    /// The number of rounds above the highest accepted round to allow fetching
1570    /// missing block headers via the periodic synchronization. Any missing
1571    /// block headers of higher rounds are considered too far in the future
1572    /// to fetch. This property is used only when it's detected that the
1573    /// node has fallen behind on its commit compared to the rest of the
1574    /// network, otherwise scheduler will attempt to fetch any missing block
1575    /// header.
1576    fn missing_block_header_round_threshold(&self) -> Round {
1577        self.context.parameters.commit_sync_batch_size
1578    }

1580    /// Fetches the given `missing_blocks` from up to `MAX_PEERS` authorities in
1581    /// parallel:
1582    ///
1583    /// Select `MAX_PEERS - MAX_RANDOM_PEERS` peers from those who are known to
1584    /// hold some missing block, requesting up to `MAX_BLOCKS_PER_FETCH` block
1585    /// refs per peer.
1586    ///
1587    /// Select `MAX_RANDOM_PEERS` additional peers from the committee (excluding
1588    /// self and those already selected).
1589    ///
1590    /// Both groups, and the peers held back to retry a failed request, are
1591    /// ordered by responsiveness when `enable_peer_responsiveness_ranking` is
1592    /// set, and drawn uniformly otherwise.
1593    ///
1594    /// The method returns a vector with the fetched blocks from each peer that
1595    /// successfully responded and any corresponding additional ancestor blocks.
1596    /// Each element of the vector is a tuple which contains the requested
1597    /// missing block refs, the returned blocks and the peer authority index.
1598    #[cfg_attr(test,tracing::instrument(skip_all, name ="",fields(authority = %context.own_index)))]
1599    async fn fetch_block_headers_from_authorities_periodic(
1600        context: Arc<Context>,
1601        inflight_block_headers: Arc<InflightBlockHeadersMap>,
1602        network_client: Arc<C>,
1603        missing_block_headers_refs: BTreeMap<BlockRef, BTreeSet<AuthorityIndex>>,
1604        dag_state: Arc<RwLock<DagState>>,
1605    ) -> Vec<(BlocksGuard, FetchedHeaders, AuthorityIndex, Vec<Round>)> {
1606        // Step 1: Map authorities to missing block headers refs that they are aware of
1607        let mut authority_to_block_headers_refs: HashMap<AuthorityIndex, Vec<BlockRef>> =
1608            HashMap::new();
1609        for (missing_block_header_ref, authorities) in &missing_block_headers_refs {
1610            for author in authorities {
1611                if author == &context.own_index {
1612                    // Skip our own index as we don't want to fetch block headers from ourselves
1613                    continue;
1614                }
1615                authority_to_block_headers_refs
1616                    .entry(*author)
1617                    .or_default()
1618                    .push(*missing_block_header_ref);
1619            }
1620        }
1621
1622        // Step 2: Choose at most MAX_PEERS-MAX_RANDOM_PEERS peers from those who are
1623        // aware of some missing block headers
1624
1625        let mut rng = StdRng::from_entropy();
1626        let rank_peers = |candidates: &mut Vec<AuthorityIndex>, rng: &mut StdRng| {
1627            if context.parameters.enable_peer_responsiveness_ranking {
1628                context.peer_responsiveness.prioritize(
1629                    DataSource::HeaderSynchronizerRequested,
1630                    candidates,
1631                    rng,
1632                );
1633            } else {
1634                // Under test the unranked order is the committee order, so the
1635                // peers a scenario expects stay fixed.
1636                #[cfg(not(test))]
1637                candidates.shuffle(rng);
1638            }
1639        };
1640
1641        // Pick MAX_PEERS - MAX_RANDOM_PEERS authorities aware of missing block
1642        // headers, ranked before the truncation so the best-ranked holders are
1643        // the ones asked.
1644        let mut peers_with_block_headers: Vec<AuthorityIndex> =
1645            authority_to_block_headers_refs.keys().copied().collect();
1646        #[cfg(test)]
1647        peers_with_block_headers.sort();
1648        rank_peers(&mut peers_with_block_headers, &mut rng);
1649        let mut chosen_peers_with_block_headers: Vec<(
1650            AuthorityIndex,
1651            Vec<BlockRef>,
1652            PeriodicPeerChoice,
1653        )> = peers_with_block_headers
1654            .into_iter()
1655            .take(MAX_PERIODIC_SYNC_PEERS - MAX_PERIODIC_SYNC_RANDOM_PEERS)
1656            .map(|peer| {
1657                let limited_block_refs = authority_to_block_headers_refs[&peer]
1658                    .iter()
1659                    .copied()
1660                    .take(context.parameters.max_headers_per_header_sync_fetch)
1661                    .collect();
1662                (peer, limited_block_refs, PeriodicPeerChoice::Known)
1663            })
1664            .collect();
1665
1666        // Step 3: Choose at most MAX_PERIODIC_SYNC_RANDOM_PEERS peers not known to
1667        // hold the missing headers. Ranked too; the exploration fraction of
1668        // `prioritize` keeps the whole committee reachable through these slots.
1669        let already_chosen: HashSet<AuthorityIndex> = chosen_peers_with_block_headers
1670            .iter()
1671            .map(|(peer, _, _)| *peer)
1672            .collect();
1673
1674        let mut random_candidates: Vec<_> = context
1675            .committee
1676            .authorities()
1677            .filter_map(|(peer_index, _)| {
1678                (peer_index != context.own_index && !already_chosen.contains(&peer_index))
1679                    .then_some(peer_index)
1680            })
1681            .collect();
1682        rank_peers(&mut random_candidates, &mut rng);
1683        let random_peers: Vec<AuthorityIndex> = random_candidates
1684            .into_iter()
1685            .take(MAX_PERIODIC_SYNC_RANDOM_PEERS)
1686            .collect();
1687
1688        #[cfg_attr(test, allow(unused_mut))]
1689        let mut all_missing_block_headers_refs: Vec<BlockRef> =
1690            missing_block_headers_refs.keys().cloned().collect();
1691        #[cfg(not(test))]
1692        all_missing_block_headers_refs.shuffle(&mut rng);
1693
1694        let mut block_headers_chunks = all_missing_block_headers_refs
1695            .chunks(context.parameters.max_headers_per_header_sync_fetch);
1696
1697        for peer in random_peers {
1698            if let Some(chunk) = block_headers_chunks.next() {
1699                chosen_peers_with_block_headers.push((
1700                    peer,
1701                    chunk.to_vec(),
1702                    PeriodicPeerChoice::Random,
1703                ));
1704            } else {
1705                break;
1706            }
1707        }
1708
1709        let mut request_futures = FuturesUnordered::new();
1710
1711        let highest_rounds = Self::get_highest_accepted_rounds(dag_state, &context);
1712
1713        // Record the missing blocks per authority for metrics
1714        let mut missing_block_headers_per_authority = vec![0; context.committee.size()];
1715        for block_ref in &all_missing_block_headers_refs {
1716            missing_block_headers_per_authority[block_ref.author] += 1;
1717        }
1718        for (missing, (_, authority)) in missing_block_headers_per_authority
1719            .into_iter()
1720            .zip(context.committee.authorities())
1721        {
1722            context
1723                .metrics
1724                .node_metrics
1725                .synchronizer_missing_block_headers_by_authority
1726                .with_label_values(&[&authority.hostname.as_str()])
1727                .inc_by(missing as u64);
1728            context
1729                .metrics
1730                .node_metrics
1731                .synchronizer_current_missing_block_headers_by_authority
1732                .with_label_values(&[&authority.hostname.as_str()])
1733                .set(missing as i64);
1734        }
1735
1736        // Peers not chosen yet serve as retry targets for failed requests,
1737        // ranked so retries go to the most responsive peer left.
1738        let mut remaining_peers: Vec<_> = context
1739            .committee
1740            .authorities()
1741            .filter_map(|(peer_index, _)| {
1742                if peer_index != context.own_index
1743                    && !chosen_peers_with_block_headers
1744                        .iter()
1745                        .any(|(chosen_peer, _, _)| *chosen_peer == peer_index)
1746                {
1747                    Some(peer_index)
1748                } else {
1749                    None
1750                }
1751            })
1752            .collect();
1753
1754        rank_peers(&mut remaining_peers, &mut rng);
1755        let mut remaining_peers = remaining_peers.into_iter();
1756
1757        // Send the initial requests
1758        for (peer, block_headers_to_request, choice) in chosen_peers_with_block_headers {
1759            let label = choice.as_str();
1760            let peer_hostname = &context.committee.authority(peer).hostname;
1761            let block_refs = block_headers_to_request
1762                .iter()
1763                .cloned()
1764                .collect::<BTreeSet<_>>();
1765
1766            // Lock the block headers to be fetched. If no lock can be acquired for any of
1767            // the block headers then don't bother.
1768            if let Some(blocks_guard) =
1769                inflight_block_headers.lock_headers(block_refs.clone(), peer, SyncMethod::Periodic)
1770            {
1771                debug!(
1772                    "Periodic sync of {} missing block headers from peer {} {}: {}",
1773                    block_refs.len(),
1774                    peer,
1775                    peer_hostname,
1776                    block_refs
1777                        .iter()
1778                        .map(|b| b.to_string())
1779                        .collect::<Vec<_>>()
1780                        .join(", ")
1781                );
1782                // Record metrics about requested blocks
1783                let metrics = &context.metrics.node_metrics;
1784                metrics
1785                    .synchronizer_requested_block_headers_by_peer
1786                    .with_label_values(&[peer_hostname.as_str(), label])
1787                    .inc_by(block_refs.len() as u64);
1788                for block_ref in &block_refs {
1789                    let block_hostname = &context.committee.authority(block_ref.author).hostname;
1790                    metrics
1791                        .synchronizer_requested_block_headers_by_authority
1792                        .with_label_values(&[block_hostname.as_str(), label])
1793                        .inc();
1794                }
1795                request_futures.push(Self::fetch_block_headers_request(
1796                    context.clone(),
1797                    network_client.clone(),
1798                    peer,
1799                    blocks_guard,
1800                    highest_rounds.clone(),
1801                    FETCH_REQUEST_TIMEOUT,
1802                    choice.holds_requested(),
1803                    1,
1804                ));
1805            }
1806        }
1807
1808        let mut results = Vec::new();
1809        let fetcher_timeout = sleep(FETCH_FROM_PEERS_TIMEOUT);
1810
1811        tokio::pin!(fetcher_timeout);
1812
1813        loop {
1814            tokio::select! {
1815                Some((response, blocks_guard, _retries, peer_index, highest_rounds)) = request_futures.next() => {
1816                    let peer_hostname = &context.committee.authority(peer_index).hostname;
1817                    match response {
1818                        Ok(fetched) => {
1819                            debug!("Fetched {} block headers from peer {}", fetched.serialized_headers.len(), peer_hostname);
1820                            results.push((blocks_guard, fetched, peer_index, highest_rounds));
1821
1822                            // no more pending requests are left, just break the loop
1823                            if request_futures.is_empty() {
1824                                break;
1825                            }
1826                        },
1827                        Err(_) => {
1828                            context.metrics.node_metrics.synchronizer_fetch_failures_by_peer.with_label_values(&[peer_hostname.as_str(), "periodic"]).inc();
1829                            // try again if there is any peer left
1830                            if let Some(next_peer) = remaining_peers.next() {
1831                                // do best effort to lock guards. If we can't lock then don't bother at this run.
1832                                if let Some(blocks_guard) = inflight_block_headers.swap_locks(blocks_guard, next_peer) {
1833                                    debug!(
1834                                        "Retrying syncing {} missing block headers from peer {}: {}",
1835                                        blocks_guard.block_refs.len(),
1836                                        peer_hostname,
1837                                        blocks_guard.block_refs
1838                                            .iter()
1839                                            .map(|b| b.to_string())
1840                                            .collect::<Vec<_>>()
1841                                            .join(", ")
1842                                    );
1843                                    let block_refs = blocks_guard.block_refs.clone();
1844                                    // Record metrics about requested blocks
1845                                    let metrics = &context.metrics.node_metrics;
1846                                    metrics
1847                                        .synchronizer_requested_block_headers_by_peer
1848                                        .with_label_values(&[peer_hostname.as_str(), PeriodicPeerChoice::Retry.as_str()])
1849                                        .inc_by(block_refs.len() as u64);
1850                                    for block_ref in &block_refs {
1851                                        let block_hostname =
1852                                            &context.committee.authority(block_ref.author).hostname;
1853                                        metrics
1854                                            .synchronizer_requested_block_headers_by_authority
1855                                            .with_label_values(&[block_hostname.as_str(), PeriodicPeerChoice::Retry.as_str()])
1856                                            .inc();
1857                                    }
1858                                    request_futures.push(Self::fetch_block_headers_request(
1859                                        context.clone(),
1860                                        network_client.clone(),
1861                                        next_peer,
1862                                        blocks_guard,
1863                                        highest_rounds,
1864                                        FETCH_REQUEST_TIMEOUT,
1865                                        PeriodicPeerChoice::Retry.holds_requested(),
1866                                        1,
1867                                    ));
1868                                } else {
1869                                    debug!("Couldn't acquire locks to fetch block headers from peer {next_peer}.")
1870                                }
1871                            } else {
1872                                debug!("No more peers left to fetch block headers");
1873                            }
1874                        }
1875                    }
1876                },
1877                _ = &mut fetcher_timeout => {
1878                    debug!("Timed out while fetching missing block headers");
1879                    // Drop all pending requests immediately — frees all block guards
1880                    drop(request_futures);
1881                    break;
1882                }
1883            }
1884        }
1885
1886        results
1887    }
