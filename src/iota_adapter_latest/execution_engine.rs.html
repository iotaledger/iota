77    /// The main entry point to the adapter's transaction execution. It
78    /// prepares a transaction for execution, then executes it through an
79    /// inner execution method and finally produces an instance of
80    /// transaction effects. It also returns the inner temporary store, which
81    /// contains the objects resulting from the transaction execution, the gas
82    /// status instance, which tracks the gas usage, and the execution result.
83    /// The function handles transaction execution based on the provided
84    /// `TransactionKind`. It checks for any expensive operations, manages
85    /// shared object references, and ensures transaction dependencies are
86    /// met. The returned objects are not committed to the store until the
87    /// resulting effects are applied by the caller.
88    #[instrument(name = "tx_execute_to_effects", level = "debug", skip_all)]
89    pub fn execute_transaction_to_effects<Mode: ExecutionMode>(
90        store: &dyn BackingStore,
91        input_objects: CheckedInputObjects,
92        gas_data: GasPayment,
93        gas_status: IotaGasStatus,
94        transaction_kind: TransactionKind,
95        transaction_signer: Address,
96        transaction_digest: TransactionDigest,
97        move_vm: &Arc<MoveVM>,
98        epoch_id: &EpochId,
99        epoch_timestamp_ms: u64,
100        protocol_config: &ProtocolConfig,
101        metrics: Arc<LimitsMetrics>,
102        enable_expensive_checks: bool,
103        certificate_deny_set: &HashSet<TransactionDigest>,
104        trace_builder_opt: &mut Option<MoveTraceBuilder>,
105    ) -> (
106        InnerTemporaryStore,
107        IotaGasStatus,
108        TransactionEffects,
109        Vec<ExecutionTiming>,
110        Result<Mode::ExecutionResults, ExecutionError>,
111    ) {
112        let input_objects = input_objects.into_inner();
113        let mutable_inputs = if enable_expensive_checks {
114            input_objects.mutable_inputs().keys().copied().collect()
115        } else {
116            HashSet::new()
117        };
118        let shared_object_refs = input_objects.filter_shared_objects();
119        let receiving_objects = transaction_kind.receiving_objects();
120        let transaction_dependencies = input_objects.transaction_dependencies();
121        let contains_deleted_input = input_objects.contains_deleted_objects();
122        let cancelled_objects = input_objects.get_cancelled_objects();
123
124        let temporary_store = TemporaryStore::new(
125            store,
126            input_objects,
127            receiving_objects,
128            transaction_digest,
129            protocol_config,
130            *epoch_id,
131        );
132
133        let sponsor = resolve_sponsor(&gas_data, &transaction_signer);
134        let gas_price = gas_status.gas_price();
135        let rgp = gas_status.reference_gas_price();
136        let gas_charger = GasCharger::new(
137            transaction_digest,
138            gas_data.objects,
139            gas_status,
140            protocol_config,
141        );
142
143        let tx_ctx = TxContext::new_from_components(
144            &transaction_signer,
145            &transaction_digest,
146            epoch_id,
147            epoch_timestamp_ms,
148            rgp,
149            gas_price,
150            gas_data.budget,
151            sponsor,
152            protocol_config,
153        );
154        let tx_ctx = Rc::new(RefCell::new(tx_ctx));
155
156        execute_transaction_to_effects_inner::<Mode>(
157            temporary_store,
158            gas_charger,
159            tx_ctx,
160            &mutable_inputs,
161            shared_object_refs,
162            transaction_dependencies,
163            contains_deleted_input,
164            cancelled_objects,
165            transaction_kind,
166            transaction_signer,
167            transaction_digest,
168            move_vm,
169            epoch_id,
170            protocol_config,
171            metrics,
172            enable_expensive_checks,
173            certificate_deny_set,
174            trace_builder_opt,
175            None,
176        )
177    }

179    /// The main execution function that processes a transaction and produces
180    /// effects. It handles gas charging and execution logic.
181    #[instrument(name = "tx_execute_to_effects_inner", level = "debug", skip_all)]
182    fn execute_transaction_to_effects_inner<Mode: ExecutionMode>(
183        mut temporary_store: TemporaryStore,
184        mut gas_charger: GasCharger,
185        tx_ctx: Rc<RefCell<TxContext>>,
186        mutable_inputs: &HashSet<ObjectId>,
187        shared_object_refs: Vec<SharedInput>,
188        mut transaction_dependencies: BTreeSet<TransactionDigest>,
189        contains_deleted_input: bool,
190        cancelled_objects: Option<(CancelledObjects, Version)>,
191        transaction_kind: TransactionKind,
192        transaction_signer: Address,
193        transaction_digest: TransactionDigest,
194        move_vm: &Arc<MoveVM>,
195        epoch_id: &EpochId,
196        protocol_config: &ProtocolConfig,
197        metrics: Arc<LimitsMetrics>,
198        enable_expensive_checks: bool,
199        certificate_deny_set: &HashSet<TransactionDigest>,
200        trace_builder_opt: &mut Option<MoveTraceBuilder>,
201        pre_execution_result_opt: Option<
202            Result<
203                <execution_mode::Authentication as ExecutionMode>::ExecutionResults,
204                ExecutionError,
205            >,
206        >,
207    ) -> (
208        InnerTemporaryStore,
209        IotaGasStatus,
210        TransactionEffects,
211        Vec<ExecutionTiming>,
212        Result<Mode::ExecutionResults, ExecutionError>,
213    ) {
214        let is_epoch_change = transaction_kind.is_end_of_epoch();
215        let deny_cert = is_certificate_denied(&transaction_digest, certificate_deny_set);
216
217        let (gas_cost_summary, execution_result, timings) = execute_transaction::<Mode>(
218            &mut temporary_store,
219            transaction_kind,
220            &mut gas_charger,
221            tx_ctx,
222            move_vm,
223            protocol_config,
224            metrics,
225            enable_expensive_checks,
226            deny_cert,
227            contains_deleted_input,
228            cancelled_objects,
229            trace_builder_opt,
230            pre_execution_result_opt,
231        );
232
233        let status = if let Err(error) = &execution_result {
234            elaborate_error_logs(error, transaction_digest)
235        } else {
236            ExecutionStatus::Success
237        };
238
239        #[skip_checked_arithmetic]
240        trace!(
241            tx_digest = ?transaction_digest,
242            computation_gas_cost = gas_cost_summary.computation_cost,
243            computation_gas_cost_burned = gas_cost_summary.computation_cost_burned,
244            storage_gas_cost = gas_cost_summary.storage_cost,
245            storage_gas_rebate = gas_cost_summary.storage_rebate,
246            "Finished execution of transaction with status {:?}",
247            status
248        );
249
250        // Genesis writes a special digest to indicate that an object was created during
251        // genesis and not written by any normal transaction - remove that from the
252        // dependencies
253        transaction_dependencies.remove(&TransactionDigest::GENESIS_MARKER);
254
255        if enable_expensive_checks && !Mode::allow_arbitrary_function_calls() {
256            temporary_store
257                .check_ownership_invariants(
258                    &transaction_signer,
259                    &mut gas_charger,
260                    mutable_inputs,
261                    is_epoch_change,
262                )
263                .unwrap()
264        } // else, in dev inspect mode and anything goes--don't check
265
266        let (inner, effects) = temporary_store.into_effects(
267            shared_object_refs,
268            &transaction_digest,
269            transaction_dependencies,
270            gas_cost_summary,
271            status,
272            &mut gas_charger,
273            *epoch_id,
274        );
275
276        (
277            inner,
278            gas_charger.into_gas_status(),
279            effects,
280            timings,
281            execution_result,
282        )
283    }

285    /// This function produces transaction effects for a transaction that
286    /// requires the Move authentication.
287    /// It creates a temporary store, gas charger, and transaction context for
288    /// the authentication execution and then reuses these for the normal
289    /// transaction execution.
290    /// Running the Move authentication can have two outcomes:
291    ///   - If it fails, then it charges gas for the failed execution of the
292    ///     authentication and produces transaction effects with the appropriate
293    ///     error status.
294    ///   - Else, if the authentication is successful, it continues with the
295    ///     normal transaction execution.
296    /// It combines the input objects from both the authentication and
297    /// transaction.
298    #[instrument(
299        name = "tx_authenticate_then_execute_to_effects",
300        level = "debug",
301        skip_all
302    )]
303    pub fn authenticate_then_execute_transaction_to_effects<Mode: ExecutionMode>(
304        store: &dyn BackingStore,
305        // Configuration
306        protocol_config: &ProtocolConfig,
307        metrics: Arc<LimitsMetrics>,
308        enable_expensive_checks: bool,
309        certificate_deny_set: &HashSet<TransactionDigest>,
310        // Epoch
311        epoch_id: &EpochId,
312        epoch_timestamp_ms: u64,
313        // Gas related
314        gas_data: GasPayment,
315        gas_status: IotaGasStatus,
316        // Authentication
317        authenticators: Vec<(
318            MoveAuthenticator,
319            AuthenticatorFunctionRefForExecution,
320            CheckedInputObjects,
321        )>,
322        authenticator_and_transaction_input_objects: CheckedInputObjects,
323        // Transaction
324        transaction_kind: TransactionKind,
325        transaction_signer: Address,
326        transaction_digest: TransactionDigest,
327        auth_context_data: AuthContextData,
328        // Tracing
329        trace_builder_opt: &mut Option<MoveTraceBuilder>,
330        // VM
331        move_vm: &Arc<MoveVM>,
332    ) -> (
333        InnerTemporaryStore,
334        IotaGasStatus,
335        TransactionEffects,
336        Vec<ExecutionTiming>,
337        Result<Mode::ExecutionResults, ExecutionError>,
338    ) {
339        // Preparation
340        // It involves setting up the TemporaryStore, GasCharger, and TxContext, that
341        // will be common for both the authentication and transaction execution.
342
343        // Input objects come from both authentication and transaction inputs
344        let input_objects = authenticator_and_transaction_input_objects.into_inner();
345        // Mutable inputs come only from the transaction inputs
346        let mutable_inputs = if enable_expensive_checks {
347            input_objects.mutable_inputs().keys().copied().collect()
348        } else {
349            HashSet::new()
350        };
351        // Shared object refs come from both authentication and transaction inputs
352        let shared_object_refs = input_objects.filter_shared_objects();
353        // Receiving objects can only come from the transaction inputs
354        let transaction_receiving_objects = transaction_kind.receiving_objects();
355        // Transaction dependencies come from both authentication and transaction inputs
356        let transaction_dependencies = input_objects.transaction_dependencies();
357        // Deleted and cancelled objects come from both authentication and transaction
358        // inputs
359        let contains_deleted_input = input_objects.contains_deleted_objects();
360        let cancelled_objects = input_objects.get_cancelled_objects();
361
362        // Prepare the temporary store.
363        let mut temporary_store = TemporaryStore::new(
364            store,
365            input_objects,
366            transaction_receiving_objects,
367            transaction_digest,
368            protocol_config,
369            *epoch_id,
370        );
371
372        // Prepare the gas charger.
373        let sponsor = resolve_sponsor(&gas_data, &transaction_signer);
374        let gas_price = gas_status.gas_price();
375        let rgp = gas_status.reference_gas_price();
376        let mut gas_charger = GasCharger::new(
377            transaction_digest,
378            gas_data.objects,
379            gas_status,
380            protocol_config,
381        );
382
383        // Prepare the transaction context.
384        let tx_ctx = TxContext::new_from_components(
385            &transaction_signer,
386            &transaction_digest,
387            epoch_id,
388            epoch_timestamp_ms,
389            rgp,
390            gas_price,
391            gas_data.budget,
392            sponsor,
393            protocol_config,
394        );
395        let tx_ctx = Rc::new(RefCell::new(tx_ctx));
396
397        // Prepare the authenticators for execution.
398        // Store the loaded object metadata in the `TemporaryStore` before the
399        // authenticators are executed.
400        // The temporary store must contain all the required information at this
401        // point.
402        let authenticators = authenticators
403            .into_iter()
404            .map(
405                |(
406                    authenticator,
407                    authenticator_function_ref_for_execution,
408                    authenticator_input_objects,
409                )| {
410                    let AuthenticatorFunctionRefForExecution {
411                        authenticator_function_ref,
412                        loaded_object_id,
413                        loaded_object_metadata,
414                    } = authenticator_function_ref_for_execution;
415
416                    // Save the loaded object metadata, i.e., the field object containing the
417                    // AuthenticatorFunctionRef, in the temporary store.
418                    temporary_store.save_loaded_runtime_objects(BTreeMap::from([(
419                        loaded_object_id,
420                        loaded_object_metadata,
421                    )]));
422
423                    (
424                        authenticator,
425                        authenticator_function_ref,
426                        authenticator_input_objects,
427                    )
428                },
429            )
430            .collect::<Vec<_>>();
431
432        // Authentication execution.
433        // It does not alter the state, if not for command execution gas charging, and
434        // produces no effects other than possible errors.
435
436        // Run each authenticator in sequence; the first failure aborts the chain.
437        let authentication_execution_result = authenticators.into_iter().try_for_each(
438            |(authenticator, authenticator_function_ref, authenticator_input_objects)| {
439                match authenticator_function_ref {
440                    AuthenticatorFunctionRef::V1(authenticator_function_ref_v1) => {
441                        authenticate_transaction_inner(
442                            &mut temporary_store,
443                            protocol_config,
444                            metrics.clone(),
445                            &mut gas_charger,
446                            authenticator,
447                            authenticator_function_ref_v1,
448                            &authenticator_input_objects.into_inner(),
449                            transaction_kind.clone(),
450                            transaction_digest,
451                            auth_context_data.clone(),
452                            tx_ctx.clone(),
453                            trace_builder_opt,
454                            move_vm,
455                        )
456                    }
457                }
458            },
459        );
460
461        let authentication_execution_result =
462            report_authentication_error(authentication_execution_result, protocol_config);
463
464        // Transaction execution.
465        // At this stage we arrive with gas charged for the execution of the
466        // authenticate function and a result which is either empty or an error.
467        // We can now start the creation of the transaction effects, either for an
468        // authentication failure or for a normal execution of the transaction.
469
470        // Run the transaction execution and return the effects.
471        execute_transaction_to_effects_inner::<Mode>(
472            temporary_store,
473            gas_charger,
474            tx_ctx,
475            &mutable_inputs,
476            shared_object_refs,
477            transaction_dependencies,
478            contains_deleted_input,
479            cancelled_objects,
480            transaction_kind,
481            transaction_signer,
482            transaction_digest,
483            move_vm,
484            epoch_id,
485            protocol_config,
486            metrics,
487            enable_expensive_checks,
488            certificate_deny_set,
489            trace_builder_opt,
490            Some(authentication_execution_result),
491        )
492    }

494    /// This function checks the authentication of a transaction without
495    /// returning effects. It executes an authenticate function using the
496    /// information of an authenticator. If the execution fails, it returns
497    /// an execution error; otherwise it returns an empty value.
498    #[instrument(name = "tx_validate", level = "debug", skip_all)]
499    pub fn authenticate_transaction(
500        store: &dyn BackingStore,
501        // Configuration
502        protocol_config: &ProtocolConfig,
503        metrics: Arc<LimitsMetrics>,
504        // Epoch
505        epoch_id: &EpochId,
506        epoch_timestamp_ms: u64,
507        // Gas related
508        gas_data: GasPayment,
509        gas_status: IotaGasStatus,
510        // Authentication
511        authenticators: Vec<(
512            MoveAuthenticator,
513            AuthenticatorFunctionRef,
514            CheckedInputObjects,
515        )>,
516        aggregated_authenticator_input_objects: CheckedInputObjects,
517        // Transaction
518        transaction_kind: TransactionKind,
519        transaction_signer: Address,
520        transaction_digest: TransactionDigest,
521        auth_context_data: AuthContextData,
522        // Tracing
523        trace_builder_opt: &mut Option<MoveTraceBuilder>,
524        // VM
525        move_vm: &Arc<MoveVM>,
526    ) -> Result<<execution_mode::Authentication as ExecutionMode>::ExecutionResults, ExecutionError>
527    {
528        // Prepare the gas charger for authentication execution.
529        let sponsor = resolve_sponsor(&gas_data, &transaction_signer);
530        let gas_price = gas_status.gas_price();
531        let rgp = gas_status.reference_gas_price();
532        let mut gas_charger =
533            GasCharger::new(transaction_digest, vec![], gas_status, protocol_config);
534
535        // Prepare the transaction context, equal for both authentication and
536        // transaction execution.
537        let tx_ctx = TxContext::new_from_components(
538            &transaction_signer,
539            &transaction_digest,
540            epoch_id,
541            epoch_timestamp_ms,
542            rgp,
543            gas_price,
544            gas_data.budget,
545            sponsor,
546            protocol_config,
547        );
548        let tx_ctx = Rc::new(RefCell::new(tx_ctx));
549
550        let mut temporary_store = TemporaryStore::new(
551            store,
552            aggregated_authenticator_input_objects.into_inner(),
553            vec![],
554            transaction_digest,
555            protocol_config,
556            *epoch_id,
557        );
558
559        // Run each authenticator in sequence; return on first failure.
560        let authentication_execution_result = authenticators.into_iter().try_for_each(
561            |(authenticator, authenticator_function_ref, authenticator_input_objects)| {
562                match authenticator_function_ref {
563                    AuthenticatorFunctionRef::V1(authenticator_function_ref_v1) => {
564                        authenticate_transaction_inner(
565                            &mut temporary_store,
566                            protocol_config,
567                            metrics.clone(),
568                            &mut gas_charger,
569                            authenticator,
570                            authenticator_function_ref_v1,
571                            &authenticator_input_objects.into_inner(),
572                            transaction_kind.clone(),
573                            transaction_digest,
574                            auth_context_data.clone(),
575                            tx_ctx.clone(),
576                            trace_builder_opt,
577                            move_vm,
578                        )
579                    }
580                }
581            },
582        );
583
584        report_authentication_error(authentication_execution_result, protocol_config)
585    }

587    // This function implements the authentication execution. It checks that the
588    // authentication method used by the authenticator is valid. It prepares a
589    /// `MoveAuthenticator` PTB with a single move call for execution, then
590    /// executes it through an inner execution method. The
591    /// `MoveAuthenticator` provides the inputs to use for the
592    /// authentication function found in `AuthenticatorFunctionRef`,
593    /// that is retrieved from an account.
594    /// If the execution fails, it returns an execution error; otherwise it
595    /// returns an empty value.
596    #[instrument(name = "tx_validate", level = "debug", skip_all)]
597    pub fn authenticate_transaction_inner(
598        temporary_store: &mut TemporaryStore<'_>,
599        // Configuration
600        protocol_config: &ProtocolConfig,
601        metrics: Arc<LimitsMetrics>,
602        // Gas related
603        gas_charger: &mut GasCharger,
604        // Authenticator
605        authenticator: MoveAuthenticator,
606        authenticator_function_ref: AuthenticatorFunctionRefV1,
607        authenticator_input_objects: &InputObjects,
608        // Transaction
609        transaction_kind: TransactionKind,
610        transaction_digest: TransactionDigest,
611        auth_context_data: AuthContextData,
612        tx_ctx: Rc<RefCell<TxContext>>,
613        // Tracing
614        trace_builder_opt: &mut Option<MoveTraceBuilder>,
615        // VM
616        move_vm: &Arc<MoveVM>,
617    ) -> Result<<execution_mode::Authentication as ExecutionMode>::ExecutionResults, ExecutionError>
618    {
619        // Check the preconditions.
620        debug_assert!(
621            transaction_kind.is_programmable(),
622            "Only programmable transactions are allowed"
623        );
624        debug_assert!(
625            authenticator_input_objects
626                .mutable_inputs()
627                .keys()
628                .copied()
629                .collect::<HashSet<_>>()
630                .is_empty(),
631            "No mutable inputs are allowed"
632        );
633        debug_assert!(
634            authenticator.receiving_objects().is_empty(),
635            "No receiving inputs are allowed"
636        );
637
638        let contains_deleted_input = authenticator_input_objects.contains_deleted_objects();
639        let cancelled_objects = authenticator_input_objects.get_cancelled_objects();
640
641        // Prepare the authentication context.
642        let auth_ctx = {
643            let TransactionKind::Programmable(ptb) = &transaction_kind else {
644                unreachable!("Only programmable transactions are allowed");
645            };
646            AuthContext::new_from_components(
647                authenticator.digest().into(),
648                auth_context_data.sender_auth_digest,
649                auth_context_data.sponsor_auth_digest,
650                auth_context_data
651                    .sender_authenticator_function_ref
652                    .and_then(Into::into),
653                auth_context_data
654                    .sponsor_authenticator_function_ref
655                    .and_then(Into::into),
656                ptb,
657                auth_context_data.transaction_data_bytes,
658            )
659        };
660        let auth_ctx = Rc::new(RefCell::new(auth_ctx));
661
662        // Store the authentication context in the temporary store.
663        // It will be added to the authentication's parameter list later, just before
664        // execution.
665        temporary_store.store_auth_context(auth_ctx);
666
667        // Execute the authentication.
668        let authentication_execution_result = execute_authenticator_move_call(
669            temporary_store,
670            authenticator,
671            authenticator_function_ref,
672            gas_charger,
673            tx_ctx,
674            move_vm,
675            protocol_config,
676            metrics,
677            false,
678            contains_deleted_input,
679            cancelled_objects,
680            trace_builder_opt,
681        );
682
683        // Check the authentication result.
684        let authentication_execution_status = if let Err(error) = &authentication_execution_result {
685            elaborate_error_logs(error, transaction_digest)
686        } else {
687            ExecutionStatus::Success
688        };
689
690        #[skip_checked_arithmetic]
691        trace!(
692            tx_digest = ?transaction_digest,
693            computation_gas_cost = gas_charger.summary().gas_used(),
694            "Finished authenticator execution of transaction with status {:?}",

793    /// Executes a transaction by processing the specified `TransactionKind`,
794    /// applying the necessary gas charges and running the main execution logic.
795    /// The function handles certain error conditions such as denied
796    /// certificate, deleted input objects, exceeded execution meter limits,
797    /// failed conservation checks. It also accounts for unmetered storage
798    /// rebates and adjusts for special cases like epoch change
799    /// transactions. Gas costs are managed through the `GasCharger`
800    /// argument; gas is also charged in case of errors.
801    #[instrument(name = "tx_execute", level = "debug", skip_all)]
802    fn execute_transaction<Mode: ExecutionMode>(
803        temporary_store: &mut TemporaryStore<'_>,
804        transaction_kind: TransactionKind,
805        gas_charger: &mut GasCharger,
806        tx_ctx: Rc<RefCell<TxContext>>,
807        move_vm: &Arc<MoveVM>,
808        protocol_config: &ProtocolConfig,
809        metrics: Arc<LimitsMetrics>,
810        enable_expensive_checks: bool,
811        deny_cert: bool,
812        contains_deleted_input: bool,
813        cancelled_objects: Option<(CancelledObjects, Version)>,
814        trace_builder_opt: &mut Option<MoveTraceBuilder>,
815        pre_execution_result_opt: Option<
816            Result<
817                <execution_mode::Authentication as ExecutionMode>::ExecutionResults,
818                ExecutionError,
819            >,
820        >,
821    ) -> (
822        GasCostSummary,
823        Result<Mode::ExecutionResults, ExecutionError>,
824        Vec<ExecutionTiming>,
825    ) {
826        gas_charger.smash_gas(temporary_store);
827
828        // At this point, either no charges have been applied yet or we have
829        // already a pre execution result to handle.
830        debug_assert!(
831            pre_execution_result_opt.is_some() || gas_charger.no_charges(),
832            "No gas charges must be applied yet"
833        );
834
835        let is_genesis_or_epoch_change_tx = matches!(transaction_kind, TransactionKind::Genesis(_))
836            || transaction_kind.is_end_of_epoch();
837
838        let advance_epoch_gas_summary = transaction_kind.get_advance_epoch_tx_gas_summary();
839
840        let tx_digest = tx_ctx.borrow().digest();
841
842        // We must charge object read here during transaction execution, because if this
843        // fails we must still ensure an effect is committed and all objects
844        // versions incremented
845        let result = gas_charger.charge_input_objects(temporary_store);
846        let result: ResultWithTimings<Mode::ExecutionResults, ExecutionError> =
847            result.map_err(|e| (e, vec![])).and_then(
848                |()| -> ResultWithTimings<Mode::ExecutionResults, ExecutionError> {
849                    run_inputs_checks(
850                        protocol_config,
851                        deny_cert,
852                        contains_deleted_input,
853                        cancelled_objects,
854                    )
855                    .map_err(|e| (e, vec![]))?;
856
857                    // If the pre-execution succeeded, proceed with the main execution loop
858                    // else propagate the pre-execution error
859                    let mut execution_result = pre_execution_result_opt
860                        .unwrap_or(Ok(()))
861                        .map_err(|e| (e, vec![]))
862                        .and_then(|_| {
863                            execution_loop::<Mode>(
864                                temporary_store,
865                                transaction_kind,
866                                tx_ctx,
867                                move_vm,
868                                gas_charger,
869                                protocol_config,
870                                metrics.clone(),
871                                trace_builder_opt,
872                            )
873                        });
874
875                    let meter_check = check_meter_limit(
876                        temporary_store,
877                        gas_charger,
878                        protocol_config,
879                        metrics.clone(),
880                    );
881                    if let Err(e) = meter_check {
882                        execution_result = Err((e, vec![]));
883                    }
884
885                    if execution_result.is_ok() {
886                        let gas_check = check_written_objects_limit(
887                            temporary_store,
888                            gas_charger,
889                            protocol_config,
890                            metrics,
891                        );
892                        if let Err(e) = gas_check {
893                            execution_result = Err((e, vec![]));
894                        }
895                    }
896
897                    execution_result
898                },
899            );
900
901        let (mut result, timings) = match result {
902            Ok((r, t)) => (Ok(r), t),
903            Err((e, t)) => (Err(e), t),
904        };
905
906        let cost_summary = gas_charger.charge_gas(temporary_store, &mut result);
907        // For advance epoch transaction, we need to provide epoch rewards and rebates
908        // as extra information provided to check_iota_conserved, because we
909        // mint rewards, and burn the rebates. We also need to pass in the
910        // unmetered_storage_rebate because storage rebate is not reflected in
911        // the storage_rebate of gas summary. This is a bit confusing.
912        // We could probably clean up the code a bit.
913        // Put all the storage rebate accumulated in the system transaction
914        // to the 0x5 object so that it's not lost.
915        temporary_store.conserve_unmetered_storage_rebate(gas_charger.unmetered_storage_rebate());
916
917        if let Err(e) = run_conservation_checks::<Mode>(
918            temporary_store,
919            gas_charger,
920            tx_digest,
921            move_vm,
922            enable_expensive_checks,
923            &cost_summary,
924            is_genesis_or_epoch_change_tx,
925            advance_epoch_gas_summary,
926        ) {
927            // FIXME: we cannot fail the transaction if this is an epoch change transaction.
928            result = Err(e);
929        }
930
931        (cost_summary, result, timings)
932    }

995    /// Performs IOTA conservation checks during transaction execution, ensuring
996    /// that the transaction does not create or destroy IOTA. If
997    /// conservation is violated, the function attempts to recover
998    /// by resetting the gas charger, recharging gas, and rechecking
999    /// conservation. If recovery fails, it panics to avoid IOTA creation or
1000    /// destruction. These checks include both simple and expensive
1001    /// checks based on the configuration and are skipped for genesis or epoch
1002    /// change transactions.
1003    #[instrument(name = "run_conservation_checks", level = "debug", skip_all)]
1004    fn run_conservation_checks<Mode: ExecutionMode>(
1005        temporary_store: &mut TemporaryStore<'_>,
1006        gas_charger: &mut GasCharger,
1007        tx_digest: TransactionDigest,
1008        move_vm: &Arc<MoveVM>,
1009        enable_expensive_checks: bool,
1010        cost_summary: &GasCostSummary,
1011        is_genesis_or_epoch_change_tx: bool,
1012        advance_epoch_gas_summary: Option<(u64, u64)>,
1013    ) -> Result<(), ExecutionError> {
1014        let mut result: std::result::Result<(), iota_types::error::ExecutionError> = Ok(());
1015        if !is_genesis_or_epoch_change_tx && !Mode::skip_conservation_checks() {
1016            // ensure that this transaction did not create or destroy IOTA, try to recover
1017            // if the check fails
1018            let conservation_result = {
1019                temporary_store
1020                    .check_iota_conserved(cost_summary)
1021                    .and_then(|()| {
1022                        if enable_expensive_checks {
1023                            // ensure that this transaction did not create or destroy IOTA, try to
1024                            // recover if the check fails
1025                            let mut layout_resolver =
1026                                TypeLayoutResolver::new(move_vm, Box::new(&*temporary_store));
1027                            temporary_store.check_iota_conserved_expensive(
1028                                cost_summary,
1029                                advance_epoch_gas_summary,
1030                                &mut layout_resolver,
1031                            )
1032                        } else {
1033                            Ok(())
1034                        }
1035                    })
1036            };
1037            if let Err(conservation_err) = conservation_result {
1038                // conservation violated. try to avoid panic by dumping all writes, charging for
1039                // gas, re-checking conservation, and surfacing an aborted
1040                // transaction with an invariant violation if all of that works
1041                result = Err(conservation_err);
1042                gas_charger.reset(temporary_store);
1043                gas_charger.charge_gas(temporary_store, &mut result);
1044                // check conservation once more
1045                if let Err(recovery_err) = {
1046                    temporary_store
1047                        .check_iota_conserved(cost_summary)
1048                        .and_then(|()| {
1049                            if enable_expensive_checks {
1050                                // ensure that this transaction did not create or destroy IOTA, try
1051                                // to recover if the check fails
1052                                let mut layout_resolver =
1053                                    TypeLayoutResolver::new(move_vm, Box::new(&*temporary_store));
1054                                temporary_store.check_iota_conserved_expensive(
1055                                    cost_summary,
1056                                    advance_epoch_gas_summary,
1057                                    &mut layout_resolver,
1058                                )
1059                            } else {
1060                                Ok(())
1061                            }
1062                        })
1063                } {
1064                    // if we still fail, it's a problem with gas
1065                    // charging that happens even in the "aborted" case--no other option but panic.
1066                    // we will create or destroy IOTA otherwise
1067                    panic!(
1068                        "IOTA conservation fail in tx block {}: {}\nGas status is {}\nTx was ",
1069                        tx_digest,
1070                        recovery_err,
1071                        gas_charger.summary()
1072                    )
1073                }
1074            }
1075        } // else, we're in the genesis transaction which mints the IOTA supply, and hence
1076        // does not satisfy IOTA conservation, or we're in the non-production
1077        // dev inspect mode which allows us to violate conservation
1078        result
1079    }

1081    /// Runs checks on the input objects of a transaction to ensure that they
1082    /// meet the necessary conditions for execution.
1083    ///
1084    /// It checks for denied certificates, deleted input objects, and cancelled
1085    /// objects due to congestion or randomness unavailability. If any of
1086    /// these conditions are met, it returns an appropriate
1087    /// `ExecutionError`.
1088    ///
1089    /// If all checks pass, it returns `Ok(())`, indicating that the transaction
1090    /// can proceed with execution.
1091    #[instrument(name = "run_inputs_checks", level = "debug", skip_all)]
1092    fn run_inputs_checks(
1093        protocol_config: &ProtocolConfig,
1094        deny_cert: bool,
1095        contains_deleted_input: bool,
1096        cancelled_objects: Option<(CancelledObjects, Version)>,
1097    ) -> Result<(), ExecutionError> {
1098        if deny_cert {
1099            Err(ExecutionError::new(
1100                ExecutionErrorKind::CertificateDenied,
1101                None,
1102            ))
1103        } else if contains_deleted_input {
1104            Err(ExecutionError::new(
1105                ExecutionErrorKind::InputObjectDeleted,
1106                None,
1107            ))
1108        } else if let Some((cancelled_objects, reason)) = cancelled_objects {
1109            match reason {
1110                version if version.is_congested() => Err(ExecutionError::new(
1111                    match cancelled_objects {
1112                        // A transaction cancelled through its gas object has no shared
1113                        // inputs, so the execution workers are the congested resource
1114                        // and no individual object is responsible.
1115                        CancelledObjects::GasObject => {
1116                            ExecutionErrorKind::ExecutionCanceledDueToExecutionWorkerCongestion {
1117                                suggested_gas_price: version
1118                                    .get_congested_version_suggested_gas_price()
1119                                    .expect(
1120                                        "execution-worker congestion control requires the gas \
1121                                        price feedback mechanism",
1122                                    ),
1123                            }
1124                        }
1125                        CancelledObjects::SharedObjects(congested_objects) => {
1126                            if protocol_config.congestion_control_gas_price_feedback_mechanism() {
1127                                ExecutionErrorKind::ExecutionCanceledDueToSharedObjectCongestionV2 {
1128                                    congested_objects,
1129                                    suggested_gas_price: version
1130                                        .get_congested_version_suggested_gas_price()
1131                                        .unwrap(),
1132                                }
1133                            } else {
1134                                // WARN: do not remove this `else` branch even after
1135                                // `congestion_control_gas_price_feedback_mechanism` is enabled
1136                                // on the mainnet. It must be kept to be able to replay old
1137                                // transaction data.
1138                                ExecutionErrorKind::ExecutionCanceledDueToSharedObjectCongestion {
1139                                    congested_objects,
1140                                }
1141                            }
1142                        }
1143                    },
1144                    None,
1145                )),
1146                Version::RANDOMNESS_UNAVAILABLE => Err(ExecutionError::new(
1147                    ExecutionErrorKind::ExecutionCanceledDueToRandomnessUnavailable,
1148                    None,
1149                )),
1150                _ => panic!("invalid cancellation reason Version: {reason}"),
1151            }
1152        } else {
1153            Ok(())
1154        }
1155    }

1247    /// Executes the given transaction based on its `TransactionKind` by
1248    /// processing it through corresponding handlers such as epoch changes,
1249    /// genesis transactions, consensus commit prologues, and programmable
1250    /// transactions. For each type of transaction, the corresponding logic is
1251    /// invoked, such as advancing the epoch, setting up consensus commits, or
1252    /// executing a programmable transaction.
1253    #[instrument(level = "debug", skip_all)]
1254    fn execution_loop<Mode: ExecutionMode>(
1255        temporary_store: &mut TemporaryStore<'_>,
1256        transaction_kind: TransactionKind,
1257        tx_ctx: Rc<RefCell<TxContext>>,
1258        move_vm: &Arc<MoveVM>,
1259        gas_charger: &mut GasCharger,
1260        protocol_config: &ProtocolConfig,
1261        metrics: Arc<LimitsMetrics>,
1262        trace_builder_opt: &mut Option<MoveTraceBuilder>,
1263    ) -> ResultWithTimings<Mode::ExecutionResults, ExecutionError> {
1264        let result = match transaction_kind {
1265            TransactionKind::Genesis(GenesisTransaction { objects, events }) => {
1266                if tx_ctx.borrow().epoch() != 0 {
1267                    panic!("BUG: Genesis Transactions can only be executed in epoch 0");
1268                }
1269
1270                for genesis_object in objects {
1271                    let object = ObjectInner {
1272                        data: genesis_object.data,
1273                        owner: genesis_object.owner,
1274                        previous_transaction: tx_ctx.borrow().digest(),
1275                        storage_rebate: 0,
1276                    };
1277                    temporary_store.create_object(object.into());
1278                }
1279
1280                temporary_store.record_execution_results(ExecutionResults::V1(
1281                    ExecutionResultsV1 {
1282                        user_events: events,
1283                        ..Default::default()
1284                    },
1285                ));
1286
1287                Ok((Mode::empty_results(), vec![]))
1288            }
1289            TransactionKind::ConsensusCommitPrologueV1(prologue) => {
1290                setup_consensus_commit(
1291                    prologue.commit_timestamp_ms,
1292                    temporary_store,
1293                    tx_ctx,
1294                    move_vm,
1295                    gas_charger,
1296                    protocol_config,
1297                    metrics,
1298                    trace_builder_opt,
1299                )
1300                .expect("ConsensusCommitPrologueV1 cannot fail");
1301                Ok((Mode::empty_results(), vec![]))
1302            }
1303            TransactionKind::Programmable(pt) => {
1304                programmable_transactions::execution::execute::<Mode>(
1305                    protocol_config,
1306                    metrics,
1307                    move_vm,
1308                    temporary_store,
1309                    tx_ctx,
1310                    gas_charger,
1311                    pt,
1312                    trace_builder_opt,
1313                )
1314            }
1315            TransactionKind::EndOfEpoch(txns) => {
1316                // Non-change-epoch kinds accumulate commands into the builder;
1317                // the change-epoch kind is always last and executes the built

1460    /// Mints epoch rewards by creating both storage and computation charges
1461    /// using a `ProgrammableTransactionBuilder`. The function takes in the
1462    /// `AdvanceEpochParams`, serializes the storage and computation
1463    /// charges, and invokes the reward creation function within the IOTA
1464    /// Prepares invocations for creating both storage and computation charges
1465    /// with a `ProgrammableTransactionBuilder` using the `AdvanceEpochParams`.
1466    /// The corresponding functions from the IOTA framework can be invoked later
1467    /// during execution of the programmable transaction.
1468    fn mint_epoch_rewards_in_pt(
1469        builder: &mut ProgrammableTransactionBuilder,
1470        params: &AdvanceEpochParams,
1471    ) -> (Argument, Argument) {
1472        // Create storage charges.
1473        let storage_charge_arg = builder
1474            .input(CallArg::pure(&params.storage_charge))
1475            .unwrap();
1476        let storage_charges = builder.programmable_move_call(
1477            ObjectId::FRAMEWORK,
1478            Identifier::BALANCE_MODULE,
1479            BALANCE_CREATE_REWARDS_FUNCTION_NAME,
1480            vec![TypeTag::from(StructTag::new_gas())],
1481            vec![storage_charge_arg],
1482        );
1483
1484        // Create computation charges.
1485        let computation_charge_arg = builder
1486            .input(CallArg::pure(&params.computation_charge))
1487            .unwrap();
1488        let computation_charges = builder.programmable_move_call(
1489            ObjectId::FRAMEWORK,
1490            Identifier::BALANCE_MODULE,
1491            BALANCE_CREATE_REWARDS_FUNCTION_NAME,
1492            vec![TypeTag::from(StructTag::new_gas())],
1493            vec![computation_charge_arg],
1494        );
1495        (storage_charges, computation_charges)
1496    }

1498    /// Constructs a `ProgrammableTransaction` to advance the epoch. It creates
1499    /// storage charges and computation charges by invoking
1500    /// `mint_epoch_rewards_in_pt`, advances the epoch by setting up the
1501    /// necessary arguments, such as epoch number, protocol version, storage
1502    /// rebate, and slashing rate, and executing the `advance_epoch` function
1503    /// within the IOTA system. Then, it destroys the storage rebates to
1504    /// complete the transaction.
1505    pub fn construct_advance_epoch_pt_impl(
1506        mut builder: ProgrammableTransactionBuilder,
1507        params: &AdvanceEpochParams,
1508        call_arg_vec: Vec<CallArg>,
1509    ) -> Result<ProgrammableTransaction, ExecutionError> {
1510        // Create storage and computation charges and add them as arguments.
1511        let (storage_charges, computation_charges) = mint_epoch_rewards_in_pt(&mut builder, params);
1512        let mut arguments = vec![
1513            builder
1514                .pure(params.validator_subsidy)
1515                .expect("bcs encoding a u64 should not fail"),
1516            storage_charges,
1517            computation_charges,
1518        ];
1519
1520        let call_arg_arguments = call_arg_vec
1521            .into_iter()
1522            .map(|a| builder.input(a))
1523            .collect::<Result<_, _>>();
1524
1525        assert_invariant!(
1526            call_arg_arguments.is_ok(),
1527            "Unable to generate args for advance_epoch transaction!"
1528        );
1529
1530        arguments.append(&mut call_arg_arguments.unwrap());
1531
1532        info!("Call arguments to advance_epoch transaction: {:?}", params);
1533
1534        let storage_rebates = builder.programmable_move_call(
1535            ObjectId::SYSTEM,
1536            Identifier::IOTA_SYSTEM_MODULE,
1537            ADVANCE_EPOCH_FUNCTION_NAME,
1538            vec![],
1539            arguments,
1540        );
1541
1542        // Step 3: Destroy the storage rebates.
1543        builder.programmable_move_call(
1544            ObjectId::FRAMEWORK,
1545            Identifier::BALANCE_MODULE,
1546            BALANCE_DESTROY_REBATES_FUNCTION_NAME,
1547            vec![TypeTag::from(StructTag::new_gas())],
1548            vec![storage_rebates],
1549        );
1550        Ok(builder.finish())
1551    }

1647    /// Advances the epoch by executing a `ProgrammableTransaction`. If the
1648    /// transaction fails, it switches to safe mode and retries the epoch
1649    /// advancement in a more controlled environment. The function also
1650    /// handles the publication and upgrade of system packages for the new
1651    /// epoch. If any system package is added or upgraded, it ensures the
1652    /// proper execution and storage of the changes.
1653    fn advance_epoch_impl(
1654        advance_epoch_pt: ProgrammableTransaction,
1655        params: AdvanceEpochParams,
1656        system_packages: Vec<SystemPackage>,
1657        temporary_store: &mut TemporaryStore<'_>,
1658        tx_ctx: Rc<RefCell<TxContext>>,
1659        move_vm: &Arc<MoveVM>,
1660        gas_charger: &mut GasCharger,
1661        protocol_config: &ProtocolConfig,
1662        metrics: Arc<LimitsMetrics>,
1663        trace_builder_opt: &mut Option<MoveTraceBuilder>,
1664    ) -> Result<(), ExecutionError> {
1665        let result = programmable_transactions::execution::execute::<execution_mode::System>(
1666            protocol_config,
1667            metrics.clone(),
1668            move_vm,
1669            temporary_store,
1670            tx_ctx.clone(),
1671            gas_charger,
1672            advance_epoch_pt,
1673            trace_builder_opt,
1674        );
1675
1676        #[cfg(msim)]
1677        let result = maybe_modify_result(result, params.epoch);
1678
1679        if let Err(err) = &result {
1680            tracing::error!(
1681                "Failed to execute advance epoch transaction. Switching to safe mode. Error: {:?}. Input objects: {:?}. Tx params: {:?}",
1682                err.0,
1683                temporary_store.objects(),
1684                params,
1685            );
1686            temporary_store.drop_writes();
1687            // Must reset the storage rebate since we are re-executing.
1688            gas_charger.reset_storage_cost_and_rebate();
1689
1690            temporary_store.advance_epoch_safe_mode(&params, protocol_config);
1691        }
1692
1693        let new_vm = new_move_vm(
1694            all_natives(/* silent */ true, protocol_config),
1695            protocol_config,
1696            // enable_profiler
1697            None,
1698        )
1699        .expect("Failed to create new MoveVM");
1700        process_system_packages(
1701            system_packages,
1702            temporary_store,
1703            tx_ctx,
1704            &new_vm,
1705            gas_charger,
1706            protocol_config,
1707            metrics,
1708            trace_builder_opt,
1709        );
1710
1711        Ok(())
1712    }

1901    fn process_system_packages(
1902        system_packages: Vec<SystemPackage>,
1903        temporary_store: &mut TemporaryStore<'_>,
1904        tx_ctx: Rc<RefCell<TxContext>>,
1905        move_vm: &MoveVM,
1906        gas_charger: &mut GasCharger,
1907        protocol_config: &ProtocolConfig,
1908        metrics: Arc<LimitsMetrics>,
1909        trace_builder_opt: &mut Option<MoveTraceBuilder>,
1910    ) {
1911        let binary_config = to_binary_config(protocol_config);
1912        for SystemPackage {
1913            version,
1914            modules,
1915            dependencies,
1916        } in system_packages.into_iter()
1917        {
1918            let deserialized_modules: Vec<_> = modules
1919                .iter()
1920                .map(|m| CompiledModule::deserialize_with_config(m, &binary_config).unwrap())
1921                .collect();
1922
1923            if version == OBJECT_START_VERSION {
1924                let package_id = deserialized_modules.first().unwrap().address();
1925                info!("adding new system package {package_id}");
1926
1927                let publish_pt = {
1928                    let mut b = ProgrammableTransactionBuilder::new();
1929                    b.command(Command::new_publish(modules, dependencies));
1930                    b.finish()
1931                };
1932
1933                programmable_transactions::execution::execute::<execution_mode::System>(
1934                    protocol_config,
1935                    metrics.clone(),
1936                    move_vm,
1937                    temporary_store,
1938                    tx_ctx.clone(),
1939                    gas_charger,
1940                    publish_pt,
1941                    trace_builder_opt,
1942                )
1943                .map_err(|(e, _)| e)
1944                .expect("System Package Publish must succeed");
1945            } else {
1946                let mut new_package = Object::new_system_package(
1947                    &deserialized_modules,
1948                    version,
1949                    dependencies,
1950                    tx_ctx.borrow().digest(),
1951                );
1952
1953                info!("upgraded system package {:?}", new_package.object_ref());
1954
1955                // Decrement the version before writing the package so that the store can record
1956                // the version growing by one in the effects.
1957                new_package
1958                    .data
1959                    .as_opt_mut_package()
1960                    .unwrap()
1961                    .decrement_version()
1962                    .expect("package version should never underflow");
1963
1964                // upgrade of a previously existing framework module
1965                temporary_store.upgrade_system_package(new_package);
1966            }
1967        }
1968    }
