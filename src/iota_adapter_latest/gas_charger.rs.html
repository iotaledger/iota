25    /// Tracks all gas operations for a single transaction.
26    /// This is the main entry point for gas accounting.
27    /// All the information about gas is stored in this object.
28    /// The objective here is two-fold:
29    /// 1- Isolate al version info into a single entry point. This file and the
30    /// other gas related files are the only one that check for gas
31    /// version. 2- Isolate all gas accounting into a single implementation.
32    /// Gas objects are not passed around, and they are retrieved from
33    /// this instance.
34    #[derive(Debug)]
35    pub struct GasCharger {
36        tx_digest: TransactionDigest,
37        #[expect(unused)]
38        gas_model_version: u64,
39        gas_coins: Vec<ObjectReference>,
40        // this is the first gas coin in `gas_coins` and the one that all others will
41        // be smashed into. It can be None for system transactions when `gas_coins` is empty.
42        smashed_gas_coin: Option<ObjectId>,
43        gas_status: IotaGasStatus,
44    }

115        /// Runs `f` with the gas left lowered to what `gas_budget` buys, when
116        /// given and that is not more, and gives the rest back once `f`
117        /// returns. Gas `f` uses stays used. Returns the result of `f` and
118        /// whether `gas_budget` bounded the gas `f` could use, also when it
119        /// buys exactly the gas left: then running out of gas in `f` means
120        /// running out of `gas_budget`, which more gas left would not change.
121        pub fn with_gas_limit<T>(
122            &mut self,
123            gas_budget: Option<u64>,
124            f: impl FnOnce(&mut Self) -> T,
125        ) -> (T, bool) {
126            let Some(gas_budget) = gas_budget else {
127                return (f(self), false);
128            };
129            let taken = self.move_gas_status_mut().limit_gas_left(gas_budget);
130            let result = f(self);
131            if let Some(taken) = taken {
132                self.move_gas_status_mut().add_gas_left(taken);
133            }
134            (result, taken.is_some())
135        }

141        /// Sets the first gas coin as the transaction's gas coin without merging
142        /// the other gas coins into it, for a transaction that charges nobody.
143        /// After this call, `gas_coin` returns its id, so the effects name it
144        /// as the gas object.
145        pub(crate) fn set_gas_coin_without_smashing(&mut self) {
146            self.smashed_gas_coin = self
147                .gas_coins
148                .first()
149                .map(|gas_coin| gas_coin.object_id)
150                .filter(|gas_coin_id| *gas_coin_id != ObjectId::ZERO);
151        }
152
153        // This function is called when the transaction is about to be executed.
154        // It will smash all gas coins into a single one and set the logical gas coin
155        // to be the first one in the list.
156        // After this call, `gas_coin` will return it id of the gas coin.
157        // This function panics if errors are found while operation on the gas coins.
158        // Transaction and certificate input checks must have insured that all gas coins
159        // are correct.
160        pub fn smash_gas(&mut self, temporary_store: &mut TemporaryStore<'_>) {
161            let gas_coin_count = self.gas_coins.len();
162            if gas_coin_count == 0
163                || (gas_coin_count == 1 && self.gas_coins[0].object_id == ObjectId::ZERO)
164            {
165                return; // self.smashed_gas_coin is None
166            }
167            // set the first coin to be the transaction only gas coin.
168            // All others will be smashed into this one.
169            let gas_coin_id = self.gas_coins[0].object_id;
170            self.smashed_gas_coin = Some(gas_coin_id);
171            if gas_coin_count == 1 {
172                return;
173            }
174
175            // sum the value of all gas coins
176            let new_balance = self
177                .gas_coins
178                .iter()
179                .map(|obj_ref| {
180                    let obj = temporary_store.objects().get(&obj_ref.object_id).unwrap();
181                    let ObjectData::Struct(move_obj) = &obj.data else {
182                        return Err(ExecutionError::invariant_violation(
183                            "Provided non-gas coin object as input for gas!",
184                        ));
185                    };
186                    if !move_obj.struct_tag().is_gas_coin() {
187                        return Err(ExecutionError::invariant_violation(
188                            "Provided non-gas coin object as input for gas!",
189                        ));
190                    }
191                    Ok(move_obj.get_coin_value_unchecked())
192                })
193                .collect::<Result<Vec<u64>, ExecutionError>>()
194                // transaction and certificate input checks must have insured that all gas coins
195                // are valid
196                .unwrap_or_else(|_| {
197                    panic!(
198                        "Invariant violation: non-gas coin object as input for gas in txn {}",
199                        self.tx_digest
200                    )
201                })
202                .iter()
203                .sum();
204            let mut primary_gas_object = temporary_store
205                .objects()
206                .get(&gas_coin_id)
207                // unwrap should be safe because we checked that this exists in `self.objects()`
208                // above
209                .unwrap_or_else(|| {
210                    panic!(
211                        "Invariant violation: gas coin not found in store in txn {}",
212                        self.tx_digest
213                    )
214                })
215                .clone();
216            // delete all gas objects except the primary_gas_object
217            for gas_coin in &self.gas_coins[1..] {
218                debug_assert_ne!(gas_coin.object_id, primary_gas_object.id());
219                temporary_store.delete_input_object(&gas_coin.object_id);
220            }
221            primary_gas_object
222                .data
223                .as_opt_mut_struct()
224                // unwrap should be safe because we checked that the primary gas object was a coin
225                // object above.
226                .unwrap_or_else(|| {
227                    panic!(
228                        "Invariant violation: invalid coin object in txn {}",
229                        self.tx_digest
230                    )
231                })
232                .set_coin_value_unchecked(new_balance);
233            temporary_store.mutate_input_object(primary_gas_object);
234        }

261        pub fn charge_input_objects(
262            &mut self,
263            temporary_store: &TemporaryStore<'_>,
264        ) -> Result<(), ExecutionError> {
265            let objects = temporary_store.objects();
266            // TODO: Charge input object count.
267            let _object_count = objects.len();
268            // Charge bytes read
269            let total_size = temporary_store
270                .objects()
271                .iter()
272                // don't charge for loading IOTA Framework or Move stdlib
273                .filter(|(id, _)| !id.is_system_package())
274                .map(|(_, obj)| obj.object_size_for_gas_metering())
275                .sum();
276            self.gas_status.charge_storage_read(total_size)
277        }

279        pub fn charge_coin_transfers(
280            &mut self,
281            protocol_config: &ProtocolConfig,
282            num_non_gas_coin_owners: u64,
283        ) -> Result<(), ExecutionError> {
284            // times two for the global pause and per-address settings
285            // this "overcharges" slightly since it does not check the global pause for each
286            // owner but rather each coin type.
287            let bytes_read_per_owner = CONFIG_SETTING_DYNAMIC_FIELD_SIZE_FOR_GAS;
288            // associate the cost with dynamic field access so that it will increase if/when
289            // this cost increases
290            let cost_per_byte =
291                protocol_config.dynamic_field_borrow_child_object_type_cost_per_byte() as usize;
292            let cost_per_owner = bytes_read_per_owner * cost_per_byte;
293            let owner_cost = cost_per_owner * (num_non_gas_coin_owners as usize);
294            self.gas_status.charge_storage_read(owner_cost)
295        }

297        /// Resets any mutations, deletions, and events recorded in the store,
298        /// as well as any storage costs and rebates, then Re-runs gas
299        /// smashing. Effects on store are now as if we were about to begin
300        /// execution
301        pub fn reset(&mut self, temporary_store: &mut TemporaryStore<'_>) {
302            temporary_store.drop_writes();
303            self.gas_status.reset_storage_cost_and_rebate();
304            self.smash_gas(temporary_store);
305        }

307        /// Entry point for gas charging.
308        /// 1. Compute tx storage gas costs and tx storage rebates, update storage_rebate field of
309        /// mutated objects
310        /// 2. Deduct computation gas costs and storage costs, credit storage rebates.
311        /// The happy path of this function follows (1) + (2) and is fairly
312        /// simple. Most of the complexity is in the unhappy paths:
313        /// - if execution aborted before calling this function, we have to dump all writes +
314        ///   re-smash gas, then charge for storage
315        /// - if we run out of gas while charging for storage, we have to dump all writes + re-smash
316        ///   gas, then charge for storage again
317        pub fn charge_gas<T>(
318            &mut self,
319            temporary_store: &mut TemporaryStore<'_>,
320            execution_result: &mut Result<T, ExecutionError>,
321        ) -> GasCostSummary {
322            // at this point, we have done *all* charging for computation,
323            // but have not yet set the storage rebate or storage gas units
324            debug_assert!(self.gas_status.storage_rebate() == 0);
325            debug_assert!(self.gas_status.storage_gas_units() == 0);
326
327            if self.smashed_gas_coin.is_some() {
328                // bucketize computation cost
329                if let Err(err) = self.gas_status.bucketize_computation() {
330                    if execution_result.is_ok() {
331                        *execution_result = Err(err);
332                    }
333                }
334
335                // On error we need to dump writes, deletes, etc before charging storage gas
336                if execution_result.is_err() {
337                    self.reset(temporary_store);
338                }
339            }
340
341            // compute and collect storage charges
342            temporary_store.ensure_active_inputs_mutated();
343            temporary_store.collect_storage_and_rebate(self);
344
345            if self.smashed_gas_coin.is_some() {
346                #[skip_checked_arithmetic]
347                trace!(target: "replay_gas_info", "Gas smashing has occurred for this transaction");
348            }
349
350            // system transactions (None smashed_gas_coin)  do not have gas and so do not
351            // charge for storage, however they track storage values to check
352            // for conservation rules
353            if let Some(gas_object_id) = self.smashed_gas_coin {
354                self.handle_storage_and_rebate(temporary_store, execution_result);
355
356                let cost_summary = self.gas_status.summary();
357                let gas_used = cost_summary.net_gas_usage();
358
359                let mut gas_object = temporary_store.read_object(&gas_object_id).unwrap().clone();
360                deduct_gas(&mut gas_object, gas_used);
361                #[skip_checked_arithmetic]
362                trace!(gas_used, gas_obj_id =? gas_object.id(), gas_obj_ver =? gas_object.version(), "Updated gas object");
363
364                temporary_store.mutate_input_object(gas_object);
365                cost_summary
366            } else {
367                GasCostSummary::default()
368            }
369        }
370
371        fn handle_storage_and_rebate<T>(
372            &mut self,
373            temporary_store: &mut TemporaryStore<'_>,
374            execution_result: &mut Result<T, ExecutionError>,
375        ) {
376            if let Err(err) = self.gas_status.charge_storage_and_rebate() {
377                // we run out of gas charging storage, reset and try charging for storage again.
378                // Input objects are touched and so they have a storage cost
379                self.reset(temporary_store);
380                temporary_store.ensure_active_inputs_mutated();
381                temporary_store.collect_storage_and_rebate(self);
382                if let Err(err) = self.gas_status.charge_storage_and_rebate() {
383                    // we run out of gas attempting to charge for the input objects exclusively,
384                    // deal with this edge case by not charging for storage
385                    self.reset(temporary_store);
386                    self.gas_status.adjust_computation_on_out_of_gas();
387                    temporary_store.ensure_active_inputs_mutated();
388                    temporary_store.collect_rebate(self);
389                    if execution_result.is_ok() {
390                        *execution_result = Err(err);
391                    }
392                } else if execution_result.is_ok() {
393                    *execution_result = Err(err);
394                }
395            }
396        }
397    }

410    const GAS_PRICE: u64 = 1000;
411
412    /// A metered charger with `units` gas units left at [`GAS_PRICE`].
413    fn gas_charger(units: u64) -> GasCharger {
414        let config = ProtocolConfig::get_for_max_version_UNSAFE();
415        let gas_status = IotaGasStatus::new(units * GAS_PRICE, GAS_PRICE, GAS_PRICE, &config)
416            .expect("the reference gas price is within the gas price bounds");
417        GasCharger::new(TransactionDigest::ZERO, vec![], gas_status, &config)
418    }
419
420    /// Charges `units` gas units.
421    fn charge(gas_charger: &mut GasCharger, units: u64) -> PartialVMResult<()> {
422        let gas: InternalGas = Gas::new(units).to_unit();
423        gas_charger.move_gas_status_mut().deduct_gas(gas)
424    }
425
426    fn remaining_gas(gas_charger: &GasCharger) -> Gas {
427        gas_charger.move_gas_status().remaining_gas()
428    }

430    /// Without a budget, `f` runs on all the gas left, and the gas is not
431    /// reported as bounded.
432    #[test]
433    fn with_gas_limit_runs_on_all_the_gas_left_without_a_budget() {
434        let mut gas_charger = gas_charger(100);
435
436        let (seen, bounded) = gas_charger.with_gas_limit(None, |gas_charger| {
437            let seen = remaining_gas(gas_charger);
438            charge(gas_charger, 10).expect("100 gas units cover 10");
439            seen
440        });
441
442        assert_eq!(seen, Gas::new(100));
443        assert!(!bounded);
444        assert_eq!(remaining_gas(&gas_charger), Gas::new(90));
445    }

447    /// With a budget below the gas left, `f` runs on what the budget buys, and
448    /// the rest is there again once it returns, less what `f` used.
449    #[test]
450    fn with_gas_limit_runs_on_a_budget_below_the_gas_left_and_gives_the_rest_back() {
451        let mut gas_charger = gas_charger(100);
452
453        let (seen, bounded) = gas_charger.with_gas_limit(Some(30 * GAS_PRICE), |gas_charger| {
454            let seen = remaining_gas(gas_charger);
455            charge(gas_charger, 10).expect("the limit covers 10 gas units");
456            seen
457        });
458
459        assert_eq!(seen, Gas::new(30));
460        assert!(bounded);
461        assert_eq!(remaining_gas(&gas_charger), Gas::new(90));
462        assert_eq!(gas_charger.move_gas_status().gas_used_pre_gas_price(), 10);
463    }
