6//! IMPORTANT: Bodies of all public functions in this module should be enclosed
7//! in a large if statement checking if tracing is enabled or not to make sure
8//! that any errors coming from these functions only manifest itself
9//! when tracing is enabled.

36/// Inserts Move call start event into the trace. As is the case for all other
37/// public functions in this module, its body is (and must be) enclosed in an if
38/// statement checking if tracing is enabled.
39pub fn trace_move_call_start(trace_builder_opt: &mut Option<MoveTraceBuilder>) {
40    if let Some(trace_builder) = trace_builder_opt {
41        trace_builder.push_event(TraceEvent::External(Box::new(serde_json::json!(
42            PTBEvent::MoveCallStart
43        ))));
44    }
45}

58/// Inserts transfer event into the trace. As is the case for all other public
59/// functions in this module, its body is (and must be) enclosed in an if
60/// statement checking if tracing is enabled.
61pub fn trace_transfer(
62    context: &mut ExecutionContext<'_, '_, '_>,
63    trace_builder_opt: &mut Option<MoveTraceBuilder>,
64    obj_values: &[ObjectValue],
65) -> Result<(), ExecutionError> {
66    if let Some(trace_builder) = trace_builder_opt {
67        let mut to_transfer = vec![];
68        for (idx, v) in obj_values.iter().enumerate() {
69            let obj_info = obj_info_from_obj_value(context, v)?;
70            to_transfer.push(ExtMoveValue::Single {
71                name: format!("obj{idx}"),
72                info: obj_info,
73            });
74        }
75        trace_builder.push_event(TraceEvent::External(Box::new(serde_json::json!(
76            PTBEvent::ExternalEvent(ExternalEvent {
77                description: "TransferObjects: obj0...objN => ()".to_string(),
78                name: "Transfer".to_string(),
79                values: to_transfer,
80            })
81        ))));
82    }
83    Ok(())
84}

86/// Inserts PTB summary event into the trace. As is the case for all other
87/// public functions in this module, its body is (and must be) enclosed in an if
88/// statement checking if tracing is enabled.
89pub fn trace_ptb_summary(
90    trace_builder_opt: &mut Option<MoveTraceBuilder>,
91    commands: &[Command],
92) -> Result<(), ExecutionError> {
93    if let Some(trace_builder) = trace_builder_opt {
94        let events = commands
95            .iter()
96            .map(|c| match c {
97                Command::MoveCall(move_call) => {
98                    let pkg = move_call.package.to_string();
99                    let module = move_call.module.to_string();
100                    let function = move_call.function.to_string();
101                    PTBCommandInfo::MoveCall {
102                        pkg,
103                        module,
104                        function,
105                    }
106                }
107                Command::TransferObjects(..) => {
108                    PTBCommandInfo::ExternalEvent("TransferObjects".to_string())
109                }
110                Command::SplitCoins(..) => PTBCommandInfo::ExternalEvent("SplitCoins".to_string()),
111                Command::MergeCoins(..) => PTBCommandInfo::ExternalEvent("MergeCoins".to_string()),
112                Command::Publish(..) => PTBCommandInfo::ExternalEvent("Publish".to_string()),
113                Command::MakeMoveVector(..) => {
114                    PTBCommandInfo::ExternalEvent("MakeMoveVec".to_string())
115                }
116                Command::Upgrade(..) => PTBCommandInfo::ExternalEvent("Upgrade".to_string()),
117                _ => unimplemented!("a new Command enum variant was added and needs to be handled"),
118            })
119            .collect();
120        trace_builder.push_event(TraceEvent::External(Box::new(serde_json::json!(
121            PTBEvent::Summary(SummaryEvent {
122                name: "PTBSummary".to_string(),
123                events,
124            })
125        ))));
126    }
127
128    Ok(())
129}

131/// Inserts split coins event into the trace. As is the case for all other
132/// public functions in this module, its body is (and must be) enclosed in an if
133/// statement checking if tracing is enabled.
134pub fn trace_split_coins(
135    context: &mut ExecutionContext<'_, '_, '_>,
136    trace_builder_opt: &mut Option<MoveTraceBuilder>,
137    coin_type: &Type,
138    input_coin: &Coin,
139    split_coin_values: &[Value],
140) -> Result<(), ExecutionError> {
141    if let Some(trace_builder) = trace_builder_opt {
142        let type_tag_with_refs = trace_type_to_type_tag_with_refs(context, coin_type)?;
143        let mut split_coin_move_values = vec![];
144        for coin_val in split_coin_values {
145            let Value::Object(ObjectValue {
146                contents: ObjectContents::Coin(coin),
147                ..
148            }) = coin_val
149            else {
150                invariant_violation!("Expected result of split coins PTB command to be a coin");
151            };
152            split_coin_move_values.push(
153                coin_obj_info(
154                    type_tag_with_refs.clone(),
155                    *coin.id.object_id(),
156                    coin.balance.value(),
157                )?
158                .value,
159            );
160        }
161
162        let input = coin_obj_info(
163            type_tag_with_refs.clone(),
164            *input_coin.id.object_id(),
165            input_coin.value(),
166        )?;
167        trace_builder.push_event(TraceEvent::External(Box::new(serde_json::json!(
168            PTBEvent::ExternalEvent(ExternalEvent {
169                description: "SplitCoins: input => result".to_string(),
170                name: "SplitCoins".to_string(),
171                values: vec![
172                    ExtMoveValue::Single {
173                        name: "input".to_string(),
174                        info: input
175                    },
176                    ExtMoveValue::Vector {
177                        name: "result".to_string(),
178                        type_: type_tag_with_refs,
179                        value: split_coin_move_values
180                    },
181                ],
182            })
183        ))));
184    }
185    Ok(())
186}

188/// Creates `ObjectInfo` from `ObjectValue`.
189fn obj_info_from_obj_value(
190    context: &mut ExecutionContext<'_, '_, '_>,
191    obj_val: &ObjectValue,
192) -> Result<ExtMoveValueInfo, ExecutionError> {
193    let type_tag_with_refs = trace_type_to_type_tag_with_refs(context, &obj_val.type_)?;
194    match &obj_val.contents {
195        ObjectContents::Coin(coin) => {
196            coin_obj_info(type_tag_with_refs, *coin.id.object_id(), coin.value())
197        }
198        ObjectContents::Raw(bytes) => {
199            let layout = context
200                .vm
201                .get_runtime()
202                .type_to_fully_annotated_layout(&obj_val.type_)
203                .map_err(|e| {
204                    ExecutionError::new_with_source(ExecutionErrorKind::InvariantViolation, e)
205                })?;
206            let move_value = BoundedVisitor::deserialize_value(bytes, &layout).map_err(|e| {
207                ExecutionError::new_with_source(ExecutionErrorKind::InvariantViolation, e)
208            })?;
209            let serialized_move_value = SerializableMoveValue::from(move_value);
210            Ok(ExtMoveValueInfo {
211                type_: type_tag_with_refs,
212                value: serialized_move_value,
213            })
214        }
215    }
216}

218/// Creates `ObjectInfo` for a coin.
219fn coin_obj_info(
220    type_tag_with_refs: TypeTagWithRefs,
221    object_id: ObjectId,
222    balance: u64,
223) -> Result<ExtMoveValueInfo, ExecutionError> {
224    let coin_type_tag = match type_tag_with_refs.type_.clone() {
225        TypeTag::Struct(tag) => tag,
226        _ => invariant_violation!("Expected a struct type tag when creating a Move coin value"),
227    };
228    // object.ID
229    let object_id = SerializableMoveValue::Address(AccountAddress::new(object_id.into_bytes()));
230    let object_id_struct_tag = StructTag {
231        address: coin_type_tag.address,
232        module: Identifier::new("object").unwrap(),
233        name: Identifier::new("ID").unwrap(),
234        type_params: vec![],
235    };
236    let object_id_struct = SimplifiedMoveStruct {
237        type_: object_id_struct_tag,
238        fields: vec![(Identifier::new("value").unwrap(), object_id)],
239    };
240    let serializable_object_id = SerializableMoveValue::Struct(object_id_struct);
241    // object.UID
242    let object_uid_struct_tag = StructTag {
243        address: coin_type_tag.address,
244        module: Identifier::new("object").unwrap(),
245        name: Identifier::new("UID").unwrap(),
246        type_params: vec![],
247    };
248    let object_uid_struct = SimplifiedMoveStruct {
249        type_: object_uid_struct_tag,
250        fields: vec![(Identifier::new("id").unwrap(), serializable_object_id)],
251    };
252    let serializable_object_uid = SerializableMoveValue::Struct(object_uid_struct);
253    // coin.Balance
254    let serializable_value = SerializableMoveValue::U64(balance);
255    let balance_struct_tag = StructTag {
256        address: coin_type_tag.address,
257        module: Identifier::new("balance").unwrap(),
258        name: Identifier::new("Balance").unwrap(),
259        type_params: coin_type_tag.type_params.clone(),
260    };
261    let balance_struct = SimplifiedMoveStruct {
262        type_: balance_struct_tag,
263        fields: vec![(Identifier::new("value").unwrap(), serializable_value)],
264    };
265    let serializable_balance = SerializableMoveValue::Struct(balance_struct);
266    // coin.Coin
267    let coin_obj = SimplifiedMoveStruct {
268        type_: *coin_type_tag,
269        fields: vec![
270            (Identifier::new("id").unwrap(), serializable_object_uid),
271            (Identifier::new("balance").unwrap(), serializable_balance),
272        ],
273    };
274    Ok(ExtMoveValueInfo {
275        type_: type_tag_with_refs,
276        value: SerializableMoveValue::Struct(coin_obj),
277    })
278}

280/// Converts a type to type tag format used in tracing.
281fn trace_type_to_type_tag_with_refs(
282    context: &mut ExecutionContext<'_, '_, '_>,
283    type_: &Type,
284) -> Result<TypeTagWithRefs, ExecutionError> {
285    let (deref_type, ref_type) = match type_ {
286        Type::Reference(t) => (t.as_ref(), Some(RefType::Imm)),
287        Type::MutableReference(t) => (t.as_ref(), Some(RefType::Mut)),
288        t => (t, None),
289    };
290    let type_ = context
291        .vm
292        .get_runtime()
293        .get_type_tag(deref_type)
294        .map_err(|e| ExecutionError::new_with_source(ExecutionErrorKind::InvariantViolation, e))?;
295    Ok(TypeTagWithRefs { type_, ref_type })
296}
