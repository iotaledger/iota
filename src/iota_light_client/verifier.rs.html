27pub fn extract_verified_effects_and_events(
28    checkpoint: &CheckpointData,
29    committee: &Committee,
30    transaction_digest: TransactionDigest,
31) -> Result<(TransactionEffects, Option<TransactionEvents>)> {
32    let summary = &checkpoint.checkpoint_summary;
33
34    // Verify the checkpoint summary using the committee
35    summary.verify_with_contents(committee, Some(&checkpoint.checkpoint_contents))?;
36
37    // Check the validity of the transaction
38    let contents = &checkpoint.checkpoint_contents;
39    let (matching_tx, _) = checkpoint
40        .transactions
41        .iter()
42        .zip(contents.iter())
43        // Note that we get the digest of the effects to ensure this is
44        // indeed the correct effects that are authenticated in the contents.
45        .find(|(tx, digest)| {
46            tx.effects.execution_digests() == *digest && digest.transaction == transaction_digest
47        })
48        .ok_or_else(|| anyhow!("Transaction not found in checkpoint contents"))?;
49
50    // Check the events are all correct.
51    let events_digest = matching_tx.events.as_ref().map(|events| events.digest());
52    anyhow::ensure!(
53        events_digest.as_ref() == matching_tx.effects.events_digest(),
54        "Events digest does not match"
55    );
56
57    // Since we do not check objects we do not return them
58    Ok((matching_tx.effects.clone(), matching_tx.events.clone()))
59}

61pub async fn get_verified_object(config: &Config, object_id: ObjectId) -> Result<Object> {
62    let iota_client = Arc::new(
63        IotaClientBuilder::default()
64            .build(config.rpc_url.as_str())
65            .await?,
66    );
67
68    info!("Getting object: {object_id}");
69
70    let read_api = iota_client.read_api();
71    let object_json = read_api
72        .get_object_with_options(object_id, IotaObjectDataOptions::bcs_lossless())
73        .await
74        .expect("Cannot get object");
75    let object = object_json
76        .into_object()
77        .expect("Cannot make into object data");
78    let object: Object = object.try_into().expect("Cannot reconstruct object");
79
80    // Need to authenticate this object
81    let (effects, _) = get_verified_effects_and_events(config, object.previous_transaction)
82        .await
83        .expect("Cannot get effects and events");
84
85    // check that this object ID, version and hash is in the effects
86    let target_object_ref = object.object_ref();
87    effects
88        .all_changed_objects()
89        .iter()
90        .find(|(changed, _)| changed.reference == target_object_ref)
91        .ok_or_else(|| anyhow!("Object not found"))?;
92
93    Ok(object)
94}

96pub async fn get_verified_effects_and_events(
97    config: &Config,
98    transaction_digest: TransactionDigest,
99) -> Result<(TransactionEffects, Option<TransactionEvents>)> {
100    let iota_client = IotaClientBuilder::default()
101        .build(config.rpc_url.as_str())
102        .await?;
103    let read_api = iota_client.read_api();
104
105    info!("Getting effects and events for transaction: {transaction_digest}");
106
107    // Lookup the transaction digest and get the checkpoint sequence number
108    let options = IotaTransactionBlockResponseOptions::new();
109    let seq = read_api
110        .get_transaction_with_options(transaction_digest, options)
111        .await
112        .context("Cannot get transaction")?
113        .checkpoint
114        .ok_or_else(|| anyhow!("Transaction not found"))?;
115
116    let checkpoint = if config.checkpoint_store_config.is_some() {
117        let checkpoint_store = CheckpointStore::new(config)?;
118
119        // Download the full checkpoint for this sequence number
120        checkpoint_store
121            .fetch_full_checkpoint(seq)
122            .await
123            .context("Cannot get full checkpoint")?
124    } else {
125        bail!("No checkpoint store configured to fetch checkpoint data")
126    };
127
128    // Load the list of stored checkpoints
129    let checkpoints_list: CheckpointList = read_checkpoint_list(config)?;
130
131    // find the stored checkpoint before the seq checkpoint
132    let prev_ckp_id = checkpoints_list
133        .checkpoints
134        .iter()
135        .rfind(|ckp_id| **ckp_id < seq);
136
137    let committee = if let Some(prev_ckp_id) = prev_ckp_id {
138        // Read it from the store
139        let prev_ckp = read_checkpoint_summary(config, *prev_ckp_id)?;
140
141        // Check we have the right checkpoint
142        anyhow::ensure!(
143            prev_ckp.epoch().checked_add(1).unwrap() == checkpoint.checkpoint_summary.epoch(),
144            "Checkpoint sequence number does not match. Need to Sync."
145        );
146
147        // Get the committee from the previous checkpoint
148        let next_epoch_committee = &prev_ckp
149            .end_of_epoch_data
150            .as_ref()
151            .ok_or_else(|| anyhow!("Expected all checkpoints to be end-of-epoch checkpoints"))?
152            .next_epoch_committee;
153
154        // Make a committee object using this
155        Committee::from_committee_members(
156            prev_ckp.epoch().checked_add(1).unwrap(),
157            next_epoch_committee,
158        )
159    } else {
160        // Since we did not find a small committee checkpoint we use the genesis
161        Genesis::load(config.genesis_blob_file_path())?
162            .committee()
163            .context("Cannot load Genesis")?
164    };
165
166    info!("Extracting effects and events for transaction: {transaction_digest}");
167
168    extract_verified_effects_and_events(&checkpoint, &committee, transaction_digest)
169        .context("Cannot extract effects and events")
170}

172/// Get the verified checkpoint sequence number for an object.
173/// This function will verify that the object is in the transaction's effects,
174/// and that the transaction is in the checkpoint
175/// and that the checkpoint is signed by the committee
176/// and the committee is read from the verified checkpoint summary
177/// which is signed by the previous committee.
178pub async fn get_verified_checkpoint(
179    config: &Config,
180    object_id: ObjectId,
181) -> Result<CheckpointSequenceNumber> {
182    let iota_client = IotaClientBuilder::default()
183        .build(config.rpc_url.as_str())
184        .await?;
185    let read_api = iota_client.read_api();
186    let object_json = read_api
187        .get_object_with_options(object_id, IotaObjectDataOptions::bcs_lossless())
188        .await
189        .expect("Cannot get object");
190    let object = object_json
191        .into_object()
192        .expect("Cannot make into object data");
193    let object: Object = object.try_into().expect("Cannot reconstruct object");
194
195    // Lookup the transaction id and get the checkpoint sequence number
196    let options = IotaTransactionBlockResponseOptions::new();
197    let seq = read_api
198        .get_transaction_with_options(object.previous_transaction, options)
199        .await
200        .context("Cannot get transaction")?
201        .checkpoint
202        .ok_or_else(|| anyhow!("Transaction not found"))?;
203
204    // Need to authenticate this object
205    let (effects, _) = get_verified_effects_and_events(config, object.previous_transaction)
206        .await
207        .expect("Cannot get effects and events");
208
209    // check that this object ID, version and hash is in the effects
210    let target_object_ref = object.object_ref();
211    effects
212        .all_changed_objects()
213        .iter()
214        .find(|(changed, _)| changed.reference == target_object_ref)
215        .ok_or_else(|| anyhow!("Object not found"))?;
216
217    // Create object store
218    let object_store = CheckpointStore::new(config)?;
219
220    // Download the full checkpoint for this sequence number
221    let full_check_point = object_store
222        .fetch_full_checkpoint(seq)
223        .await
224        .context("Cannot get full checkpoint")?;
225
226    // Load the list of stored checkpoints
227    let checkpoints_list: CheckpointList = read_checkpoint_list(config)?;
228
229    // find the stored checkpoint before the seq checkpoint
230    let prev_ckp_id = checkpoints_list
231        .checkpoints
232        .iter()
233        .rfind(|ckp_id| **ckp_id < seq);
234
235    let committee = if let Some(prev_ckp_id) = prev_ckp_id {
236        // Read it from the store
237        let prev_ckp = read_checkpoint_summary(config, *prev_ckp_id)?;
238
239        // Check we have the right checkpoint
240        anyhow::ensure!(
241            prev_ckp.epoch().checked_add(1).unwrap() == full_check_point.checkpoint_summary.epoch(),
242            "Checkpoint sequence number does not match. Need to Sync."
243        );
244
245        // Get the committee from the previous checkpoint
246        let next_epoch_committee = &prev_ckp
247            .end_of_epoch_data
248            .as_ref()
249            .ok_or_else(|| anyhow!("Expected all checkpoints to be end-of-epoch checkpoints"))?
250            .next_epoch_committee;
251
252        // Make a committee object using this
253        Committee::from_committee_members(
254            prev_ckp.epoch().checked_add(1).unwrap(),
255            next_epoch_committee,
256        )
257    } else {
258        // Since we did not find a small committee checkpoint we use the genesis
259        Genesis::load(config.genesis_blob_file_path())?
260            .committee()
261            .context("Cannot load Genesis")?
262    };
263
264    // Verify that committee signed this checkpoint and checkpoint contents with
265    // digest
266    full_check_point
267        .checkpoint_summary
268        .verify_with_contents(&committee, Some(&full_check_point.checkpoint_contents))?;
269
270    anyhow::ensure!(
271        full_check_point
272            .transactions
273            .iter()
274            .any(|t| *t.transaction.digest() == object.previous_transaction),
275        "Transaction not found in checkpoint"
276    );
277    Ok(seq)
278}

325    async fn read_data(committee_seq: u64, seq: u64) -> (Committee, CheckpointData) {
326        let checkpoint_summary_path = PathBuf::from(env!("CARGO_MANIFEST_DIR"))
327            .join(FIXTURES_DIR)
328            .join(format!("{committee_seq}.sum"));
329        let summary = read_checkpoint_summary(&checkpoint_summary_path)
330            .await
331            .unwrap();
332        let prev_committee = &summary
333            .end_of_epoch_data
334            .as_ref()
335            .expect("Expected all checkpoints to be end-of-epoch checkpoints")
336            .next_epoch_committee;
337
338        let committee = Committee::from_committee_members(
339            summary.epoch().checked_add(1).unwrap(),
340            prev_committee,
341        );
342
343        let full_checkpoint_path = PathBuf::from(env!("CARGO_MANIFEST_DIR"))
344            .join(FIXTURES_DIR)
345            .join(format!("{seq}.chk"));
346        let full_checkpoint = read_full_checkpoint(&full_checkpoint_path).await.unwrap();
347
348        (committee, full_checkpoint)
349    }

351    async fn read_checkpoint_summary(
352        checkpoint_path: &PathBuf,
353    ) -> anyhow::Result<CertifiedCheckpointSummary> {
354        let mut reader = fs::File::open(checkpoint_path.clone())?;
355        let metadata = fs::metadata(checkpoint_path)?;
356        let mut buffer = vec![0; metadata.len() as usize];
357        reader.read_exact(&mut buffer)?;
358        bcs::from_bytes(&buffer).context("failed to deserialize summary from bcs bytes")
359    }

362        let mut reader = fs::File::open(checkpoint_path.clone())?;
363        let metadata = fs::metadata(checkpoint_path)?;
364        let mut buffer = vec![0; metadata.len() as usize];
365        reader.read_exact(&mut buffer)?;
366        bcs::from_bytes(&buffer).context("failed to deserialize full checkpoint from bcs bytes")
367    }
