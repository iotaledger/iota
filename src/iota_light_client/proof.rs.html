20/// A proof for specific targets. It certifies a checkpoint summary and
21/// optionally includes transaction evidence to certify objects and events.
22#[derive(Debug, Serialize, Deserialize)]
23pub struct Proof {
24    /// Targets of the proof are a committee, objects, or events that need to be
25    /// certified.
26    pub targets: ProofTargets,
27
28    /// A summary of the checkpoint being certified.
29    pub checkpoint_summary: CertifiedCheckpointSummary,
30
31    /// Optional transaction proof to certify objects and events.
32    pub contents_proof: Option<TransactionProof>,
33}
34
35/// Define aspects of IOTA state that need to be certified in a proof
36#[derive(Default, Debug, Serialize, Deserialize)]
37pub struct ProofTargets {
38    /// Objects that need to be certified.
39    pub objects: Vec<(ObjectReference, Object)>,
40
41    /// Events that need to be certified.
42    pub events: Vec<(EventID, Event)>,
43
44    /// The next committee being certified.
45    pub committee: Option<Committee>,
46}
47
48impl ProofTargets {
49    /// Create a new empty proof target. An empty proof target still ensures
50    /// that the checkpoint summary is correct.
51    pub fn new() -> Self {
52        Self::default()
53    }
54
55    /// Add an object to be certified by object reference and content. A
56    /// verified proof will ensure that both the reference and content are
57    /// correct. Note that some content is metadata such as the transaction
58    /// that created this object.
59    pub fn add_object(mut self, object_ref: ObjectReference, object: Object) -> Self {
60        self.objects.push((object_ref, object));
61        self
62    }
63
64    /// Add an event to be certified by event ID and content. A verified proof
65    /// will ensure that both the ID and content are correct.
66    pub fn add_event(mut self, event_id: EventID, event: Event) -> Self {
67        self.events.push((event_id, event));
68        self
69    }
70
71    /// Add the next committee to be certified. A verified proof will ensure
72    /// that the next committee is correct.
73    pub fn set_committee(mut self, committee: Committee) -> Self {
74        self.committee = Some(committee);
75        self
76    }
77}

79/// Part of a proof that provides evidence relating to a specific transaction to
80/// certify objects and events.
81#[derive(Debug, Serialize, Deserialize)]
82pub struct TransactionProof {
83    /// Checkpoint contents including this transaction.
84    pub checkpoint_contents: CheckpointContents,
85
86    /// The transaction being certified.
87    pub transaction: TransactionEnvelope,
88
89    /// The effects of the transaction being certified.
90    pub effects: TransactionEffects,
91
92    /// The events of the transaction being certified.
93    pub events: Option<TransactionEvents>,
94}
95
96/// Verify a proof against a committee. A proof is valid if it certifies the
97/// checkpoint summary and optionally includes transaction evidence to certify
98/// objects and events.
99///
100/// If the result is `Ok(())` then the proof is valid. If Err is returned then
101/// the proof is invalid and the error message will describe the reason. Once a
102/// proof is verified it can be trusted, and information in `targets` as well as
103/// `checkpoint_summary` or `contents_proof` can be trusted as being authentic.
104///
105/// The authoritative committee is required to verify the proof. The sequence of
106/// committees can be verified through a Committee proof target on the last
107/// checkpoint of each epoch, sequentially since the first epoch.
108pub fn verify_proof(committee: &Committee, proof: &Proof) -> anyhow::Result<()> {
109    // Get checkpoint summary and optional contents
110    let summary = &proof.checkpoint_summary;
111    let contents_ref = proof
112        .contents_proof
113        .as_ref()
114        .map(|x| &x.checkpoint_contents);
115
116    // Verify the checkpoint summary using the committee
117    summary.verify_with_contents(committee, contents_ref)?;
118
119    // MILESTONE 1 : summary and contents is correct
120    // Note: this is unconditional on the proof targets, and always checked.
121
122    // If the proof target is the next committee check it
123    if let Some(committee) = &proof.targets.committee {
124        let Some(EndOfEpochData {
125            next_epoch_committee,
126            ..
127        }) = &summary.end_of_epoch_data
128        else {
129            bail!("No end of epoch committee in the checkpoint summary");
130        };
131
132        // Extract the end of epoch committee
133        let new_committee = Committee::from_committee_members(
134            summary.epoch().checked_add(1).unwrap(),
135            next_epoch_committee,
136        );
137
138        if new_committee != *committee {
139            bail!("Given committee does not match the end of epoch committee");
140        }
141    }
142
143    // MILESTONE 2: committee if requested is correct
144
145    // Non empty object or event targets require the optional contents proof
146    // If it is not present return an error
147
148    if (!proof.targets.objects.is_empty() || !proof.targets.events.is_empty())
149        && proof.contents_proof.is_none()
150    {
151        bail!("Contents proof is missing");
152    }
153
154    // MILESTONE 3: contents proof is present if required
155
156    if let Some(contents_proof) = &proof.contents_proof {
157        // Extract Transaction Digests and check they are in contents
158        let digests = contents_proof.effects.execution_digests();
159        if contents_proof.transaction.digest() != &digests.transaction {
160            bail!("Transaction digest does not match the execution digest");
161        }
162
163        // Ensure the digests are in the checkpoint contents
164        if !contents_proof
165            .checkpoint_contents
166            .enumerate_transactions(summary)
167            .any(|x| x.1 == digests)
168        {
169            // Could not find the digest in the checkpoint contents
170            bail!("Transaction digest not found in the checkpoint contents");
171        }
172
173        // MILESTONE 4: Transaction & Effect correct and in contents
174
175        if contents_proof.effects.events_digest()
176            != contents_proof.events.as_ref().map(|e| e.digest()).as_ref()
177        {
178            bail!("Events digest does not match the execution digest");
179        }
180
181        // If the target includes any events ensure the events digest is not None
182        if !proof.targets.events.is_empty() && contents_proof.events.is_none() {
183            bail!("Events digest is missing");
184        }
185
186        // MILESTONE 5: Events digest & Events are correct and present if required
187
188        // Now we verify the content of any target events
189
190        for (event_id, event) in &proof.targets.events {
191            // Check the event corresponds to the transaction
192            if event_id.tx_digest != digests.transaction {
193                bail!("Event does not belong to the transaction");
194            }
195
196            // The sequence number must be a valid index
197            // Note: safe to unwrap as we have checked that its not None above
198            if event_id.event_seq as usize >= contents_proof.events.as_ref().unwrap().len() {
199                bail!("Event sequence number out of bounds");
200            }
201
202            // Now check that the contents of the event are the same
203            if &contents_proof.events.as_ref().unwrap()[event_id.event_seq as usize] != event {
204                bail!("Event contents do not match");
205            }
206        }
207
208        // MILESTONE 6: Event contents are correct
209
210        // Now check all object references are correct and in the effects
211        let changed_objects = contents_proof.effects.all_changed_objects();
212
213        for (object_ref, object) in &proof.targets.objects {
214            // Is the given reference correct?
215            if object_ref != &object.object_ref() {
216                bail!("Object reference does not match the object");
217            }
218
219            // Has this object been created in these effects?
220            changed_objects
221                .iter()
222                .find(|(changed, _)| &changed.reference == object_ref)
223                .ok_or_else(|| anyhow!("Object not found"))?;
224        }
225
226        // MILESTONE 7: Object references are correct and in the effects
227    }
228
229    Ok(())
230}
