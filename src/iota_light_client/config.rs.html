16const GENESIS_FILE_NAME: &str = "genesis.blob";
17const CHECKPOINTS_FILE_NAME: &str = "checkpoints.yaml";
18
19/// The config file for the light client.
20#[derive(Clone, Debug, Deserialize, Serialize)]
21pub struct Config {
22    /// An RPC endpoint to a full node.
23    pub rpc_url: Url,
24    /// A GraphQL endpoint to a full node.
25    pub graphql_url: Option<Url>,
26    /// The directory containing synced checkpoints.
27    pub checkpoints_dir: PathBuf,
28    /// The URL to download the genesis.blob file from.
29    pub genesis_blob_download_url: Option<Url>,
30    /// Flag to enable automatic syncing before running one of the check
31    /// commands.
32    pub sync_before_check: bool,
33    /// A config to sync the light client from a checkpoint store. If provided,
34    /// will also be used to check objects/transactions for inclusion.
35    pub checkpoint_store_config: Option<ObjectStoreConfig>,
36}
37
38impl Config {
39    /// Loads the config from file.
40    pub async fn load<P: AsRef<Path>>(path: P) -> Result<Self> {
41        let content = read_to_string(path).await?;
42        let config: Config = serde_yaml::from_str(&content)?;
43        config.validate()?;
44        Ok(config)
45    }
46
47    /// Creates the necessary checkpoint directory and files if not already
48    /// present.
49    pub async fn setup(&self) -> Result<()> {
50        // Create the checkpoints directory if it doesn't exist yet
51        if !self.checkpoints_dir.is_dir() {
52            create_dir_all(&self.checkpoints_dir).await?;
53        }
54        // Download or copy the genesis blob if it doesn't exist yet
55        if !self.genesis_blob_file_path().is_file() {
56            if let Some(url) = &self.genesis_blob_download_url {
57                match url.scheme() {
58                    "file" => {
59                        let path = url
60                            .to_file_path()
61                            .map_err(|_| anyhow!("invalid file path '{url}'"))?;
62                        tokio::fs::copy(path, self.genesis_blob_file_path()).await?;
63                    }
64                    _ => {
65                        let contents = reqwest::get(url.as_str()).await?.bytes().await?;
66                        tokio::fs::write(self.genesis_blob_file_path(), contents).await?;
67                    }
68                }
69            }
70        }
71        // Create an empty `checkpoints.yaml` if it doesn't exist yet
72        if !self.checkpoints_list_file_path().is_file() {
73            write_checkpoint_list(self, &CheckpointList::default())?;
74        }
75        Ok(())
76    }
77
78    pub fn validate(&self) -> Result<()> {
79        if self.graphql_url.is_none() && self.checkpoint_store_config.is_none() {
80            bail!(
81                "Invalid config: either a GraphQL URL or a checkpoint store config must be provided"
82            );
83        }
84        Ok(())
85    }
86
87    pub fn checkpoints_list_file_path(&self) -> PathBuf {
88        self.checkpoints_dir.join(CHECKPOINTS_FILE_NAME)
89    }
90
91    pub fn genesis_blob_file_path(&self) -> PathBuf {
92        self.checkpoints_dir.join(GENESIS_FILE_NAME)
93    }
94
95    pub fn checkpoint_summary_file_path(&self, seq: u64) -> PathBuf {
96        Path::new(&self.checkpoints_dir).join(format!("{seq}.sum"))
97    }
98
99    pub fn mainnet() -> Self {
100        Self::create_config_from_network_name("mainnet")
101    }
102
103    pub fn testnet() -> Self {
104        Self::create_config_from_network_name("testnet")
105    }
106
107    pub fn devnet() -> Self {
108        Self::create_config_from_network_name("devnet")
109    }
110
111    fn create_config_from_network_name(network: &str) -> Self {
112        Self {
113            rpc_url: Url::parse(&format!("https://api.{network}.iota.cafe")).unwrap(),
114            graphql_url: Some(Url::parse(&format!("https://graphql.{network}.iota.cafe")).unwrap()),
115            checkpoints_dir: PathBuf::from_str(&format!("checkpoints_{network}")).unwrap(),
116            genesis_blob_download_url: Some(
117                Url::parse(&format!(
118                    "https://dbfiles.{network}.iota.cafe/{GENESIS_FILE_NAME}"
119                ))
120                .unwrap(),
121            ),
122            sync_before_check: false,
123            checkpoint_store_config: Some(ObjectStoreConfig {
124                object_store: Some(ObjectStoreType::S3),
125                object_store_connection_limit: 20,
126                aws_endpoint: Some(format!(
127                    "https://checkpoints.{network}.iota.cafe/ingestion/historical"
128                )),
129                aws_virtual_hosted_style_request: true,
130                aws_region: Some("weur".to_string()),
131                no_sign_request: true,
132                ..Default::default()
133            }),
134        }
135    }
136}

138#[cfg(test)]
139mod tests {
140    use tempfile::TempDir;
141
142    use super::*;
143
144    fn create_test_config() -> (Config, TempDir) {
145        let temp_dir = TempDir::new().unwrap();
146        std::fs::File::create(temp_dir.path().join(GENESIS_FILE_NAME)).unwrap();
147        let config = Config {
148            rpc_url: "http://localhost:9000".parse().unwrap(),
149            graphql_url: Some("http://localhost:9003".parse().unwrap()),
150            checkpoints_dir: temp_dir.path().to_path_buf(),
151            genesis_blob_download_url: None,
152            sync_before_check: false,
153            checkpoint_store_config: Some(ObjectStoreConfig {
154                object_store: Some(ObjectStoreType::S3),
155                aws_endpoint: Some("http://localhost:9001".to_string()),
156                ..Default::default()
157            }),
158        };
159        config.validate().expect("invalid");
160        (config, temp_dir)
161    }
162
163    #[test]
164    fn test_config_validation() {
165        let (config, _temp_dir) = create_test_config();
166        assert!(config.validate().is_ok());
167    }
168
169    #[test]
170    fn test_checkpoint_paths() {
171        let (config, _temp_dir) = create_test_config();
172
173        let list_path = config.checkpoints_list_file_path();
174        assert_eq!(list_path.file_name().unwrap(), "checkpoints.yaml");
175
176        let checkpoint_path = config.checkpoint_summary_file_path(123);
177        assert_eq!(checkpoint_path.file_name().unwrap(), "123.sum");
178    }
179
180    #[test]
181    fn test_genesis_path() {
182        let (config, _temp_dir) = create_test_config();
183        let genesis_path = config.genesis_blob_file_path();
184        assert_eq!(genesis_path.file_name().unwrap(), GENESIS_FILE_NAME);
185    }
