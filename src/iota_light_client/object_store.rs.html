5use std::num::NonZeroUsize;
6
7use anyhow::{Result, anyhow, bail};
8use iota_data_ingestion_core::history::{
9    epoch_boundaries::EpochBoundaries,
10    reader::{HistoricalReader, HistoricalReaderConfig},
11};
12use iota_types::{
13    full_checkpoint_content::CheckpointData, messages_checkpoint::CertifiedCheckpointSummary,
14};
15use tracing::debug;
16
17use crate::config::Config;
18
19pub struct CheckpointStore {
20    historical_reader: HistoricalReader,
21}
22
23impl CheckpointStore {
24    pub fn new(config: &Config) -> Result<Self> {
25        let Some(remote_store_config) = config.checkpoint_store_config.clone() else {
26            bail!("missing checkpoint store config");
27        };
28
29        let config = HistoricalReaderConfig {
30            remote_store_config,
31            download_concurrency: NonZeroUsize::new(5).unwrap(),
32        };
33
34        Ok(Self {
35            historical_reader: HistoricalReader::new(config)?,
36        })
37    }
38
39    /// Returns the archive's recorded end-of-epoch checkpoints (`epoch → last
40    /// checkpoint of that epoch`).
41    pub async fn end_of_epoch_checkpoints(&self) -> Result<EpochBoundaries> {
42        Ok(self.historical_reader.epoch_boundaries().await?)
43    }
44
45    pub async fn fetch_checkpoint_summary(&self, seq: u64) -> Result<CertifiedCheckpointSummary> {
46        let full_checkpoint = self.fetch_full_checkpoint(seq).await?;
47
48        Ok(full_checkpoint.checkpoint_summary)
49    }
50
51    pub async fn fetch_full_checkpoint(&self, seq: u64) -> Result<CheckpointData> {
52        self.historical_reader.sync_manifest_once().await?;
53        let checkpoint = self
54            .historical_reader
55            .iter_for_range(seq..seq + 1)
56            .await?
57            .next()
58            .ok_or_else(|| anyhow!("missing full checkpoint"))?;
59        debug!("Fetched checkpoint '{seq}' from checkpoint store",);
60
61        Ok(checkpoint)
62    }
63}
